400class PipelinedEngine:
401 """Drive a real AsyncScheduler like EngineCore.step_with_batch_queue:
402 schedule until the batch queue is full, then process the oldest step's
403 output. Async PP runs pp_size+1 concurrent batches, so up to pp_size
404 steps are in flight at each schedule() call -- the window in which
405 preemption must handle output that has not yet returned. (Single-GPU e2e
406 tests can never create this window: at PP=1, exactly one step is in
407 flight and it is processed before a preempted request can resume.)
408
409 The model runner is emulated with the V2 runner's own bookkeeping, from
410 only what the scheduler serializes to it: slots flushed on
411 preempted_req_ids, resumed requests re-added from the NewRequestData
412 snapshot, sampling when a step reaches the end of the runner's own view
413 of the sequence. This makes preemption races observable: a stale token
414 delivered after a resume is scheduled extends the scheduler's sequence
415 but not the runner's.
416
417 Every sample emits a globally unique token tagged with its sampled
418 position, so tests can assert exact delivery.
419 """
420
421 def __init__(
422 self,
423 scheduler: AsyncScheduler,
424 queue_size: int,
425 accept_drafts: Callable[[int, str, int], int] | None = None,
426 ):
427 self.scheduler = scheduler
428 self.queue_size = queue_size
429 self.accept_drafts = accept_drafts
430 # In-flight steps: (scheduler_output, new_reqs snapshot) in FIFO order.
431 self.queue: deque[tuple[SchedulerOutput, list[tuple[str, int, int]]]] = deque()
432 # Runner-side request state: req_id -> [seq_len, num_computed] as the
433 # runner sees them (its own sampled tokens, not the scheduler's).
434 self.runner_view: dict[str, list[int]] = {}
435 # All tokens the fake runner ever sampled, per request, in order.
436 self.emitted: dict[str, list[int]] = defaultdict(list)
437 # Sequence position each (globally unique) token was sampled for.
438 self.emitted_position: dict[int, int] = {}
439 self.step_idx = 0
440 self._next_token = 1000
441
442 def _schedule(self) -> bool:
443 scheduler_output = self.scheduler.schedule()
444 self.step_idx += 1
445 # Snapshot what NewRequestData serializes at schedule time (both new
446 # and resumed requests for the V2 runner).
447 new_reqs = [
448 (r.req_id, len(r.prefill_token_ids), r.num_computed_tokens)
449 for r in scheduler_output.scheduled_new_reqs
450 ]
451 # Enqueue empty steps too (the engine executes them), so the runner
452 # still observes their preempted/finished request ids in step order.
453 self.queue.appendleft((scheduler_output, new_reqs))
454 return True
455
456 def _process_oldest_step(self) -> None:
457 scheduler_output, new_reqs = self.queue.pop()
458 # Worker-side state updates, in step order: flush preempted/finished
459 # slots, then (re-)add new/resumed requests.
460 for req_id in scheduler_output.preempted_req_ids or ():
461 self.runner_view.pop(req_id, None)
462 for req_id in scheduler_output.finished_req_ids or ():
463 self.runner_view.pop(req_id, None)
464 for req_id, seq_len, num_computed in new_reqs:
465 self.runner_view[req_id] = [seq_len, num_computed]
466
467 req_ids = list(scheduler_output.num_scheduled_tokens.keys())
468 sampled_token_ids: list[list[int]] = []
469 for req_id in req_ids:
470 num_scheduled = scheduler_output.num_scheduled_tokens[req_id]
471 view = self.runner_view.get(req_id)
472 if view is None:
473 # Slot already flushed (request finished/aborted mid-flight).
474 sampled_token_ids.append([])
475 continue
476 seq_len, num_computed = view
477 end = num_computed + num_scheduled
478 if end < seq_len:
479 # Partial prefill by the runner's own bookkeeping: no sample.
480 view[1] = end
481 sampled_token_ids.append([])
482 continue
483 drafts = scheduler_output.scheduled_spec_decode_tokens.get(req_id, ())
484 num_accepted = (
485 min(self.accept_drafts(self.step_idx, req_id, len(drafts)), len(drafts))
486 if drafts and self.accept_drafts
487 else 0
488 )
489 num_rejected = len(drafts) - num_accepted
490 tokens = list(range(self._next_token, self._next_token + 1 + num_accepted))
491 self._next_token += 1 + num_accepted
492 self.emitted[req_id].extend(tokens)
493 sampled_token_ids.append(tokens)
494 # Rejected drafts roll back computed; the sampled tokens extend
495 # the runner's sequence.
496 view[1] = end - num_rejected
497 view[0] = view[1] + 1
498 for offset, token in enumerate(tokens):
499 self.emitted_position[token] = view[0] - len(tokens) + offset
500 model_runner_output = ModelRunnerOutput(
501 req_ids=req_ids,
502 req_id_to_index={req_id: i for i, req_id in enumerate(req_ids)},
503 sampled_token_ids=sampled_token_ids,
504 logprobs=None,
505 prompt_logprobs_dict={},
506 pooler_output=[],
507 )
508 self.scheduler.update_from_output(scheduler_output, model_runner_output)
509
510 def run(
511 self,
512 max_steps: int = 2000,
513 before_step: Callable[[int, "PipelinedEngine"], None] | None = None,
514 ) -> None:
515 for i in range(max_steps):
516 if not self.scheduler.has_requests() and not self.queue:
517 return
518 if before_step is not None:
519 before_step(i, self)
520 scheduled = (
521 self.scheduler.has_requests()
522 and len(self.queue) < self.queue_size
523 and self._schedule()
524 )
525 if scheduled and len(self.queue) < self.queue_size:
526 # Queue not yet full: the engine returns without blocking.
527 continue
528 if self.queue:
529 self._process_oldest_step()
530 raise AssertionError("engine loop did not converge")
531

533def _create_async_pp_scheduler(
534 num_spec: int, pp_size: int = 3, num_blocks: int = 5
535) -> AsyncScheduler:
536 scheduler = create_scheduler(
537 async_scheduling=True,
538 num_speculative_tokens=num_spec or None,
539 speculative_method="ngram_gpu" if num_spec else None,
540 use_v2_model_runner=True,
541 num_blocks=num_blocks,
542 block_size=16,
543 max_num_batched_tokens=512,
544 )
545 # Emulate PP at the scheduler level; constructing with
546 # pipeline_parallel_size>1 requires that many visible GPUs. Drive with
547 # queue_size=pp_size+1 (V2 async PP runs pp_size+1 concurrent batches).
548 scheduler.pp_size = pp_size
549 scheduler.use_pp = pp_size > 1
550 return scheduler

561def _assert_positions_consistent(req, engine: PipelinedEngine) -> None:
562 """The i-th delivered output token must be one the runner sampled for
563 exactly sequence position prompt_len + i: catches a preempted request's
564 stale output landing on a position the resumed request resampled (or

575@pytest.mark.parametrize("num_spec", [0, 3])
576def test_kv_pressure_preemption_with_inflight_output(num_spec: int):
577 """KV-pressure preemption of requests with in-flight async output.
578
579 PP=3 + async scheduling (batch queue of 4), a block pool small enough
580 that decodes contend and preempt mid-flight, and staggered arrivals so
581 the batch queue actually pipelines. A preempted request's in-flight steps
582 still return: their tokens must be delivered exactly once, their stale
583 spec-rejection counts must not corrupt the rolled-back counters, and the
584 resume must not resample a position that output later delivers.
585
586 Regression for the num_output_placeholders underflow EngineCore crash:
587 with the fix reverted, the num_spec=3 variant fails with exactly
588 ``assert request.num_output_placeholders >= 0`` when a stale spec output
589 returns after the preempted request was resumed and sampled.
590 """
591 max_tokens = 24
592 scheduler = _create_async_pp_scheduler(num_spec)
593 requests = create_requests(
594 num_requests=8, num_tokens=8, max_tokens=max_tokens, ignore_eos=True
595 )
596 pending = list(requests)
597 for _ in range(2):
598 scheduler.add_request(pending.pop(0))
599
600 # Observe that the scenario under test actually occurs.
601 preempts_with_inflight_output = 0
602 orig_preempt = scheduler._preempt_request
603
604 def counting_preempt(request, timestamp, **kwargs):
605 nonlocal preempts_with_inflight_output
606 if request.num_in_flight_tokens > 0:
607 preempts_with_inflight_output += 1
608 return orig_preempt(request, timestamp, **kwargs)
609
610 scheduler._preempt_request = counting_preempt
611
612 def add_requests(step: int, engine: PipelinedEngine):
613 if pending:
614 scheduler.add_request(pending.pop(0))
615
616 engine = PipelinedEngine(
617 scheduler,
618 queue_size=4,
619 # Deterministically vary spec acceptance so stale outputs carry
620 # nonzero rejection counts.
621 accept_drafts=lambda step, req_id, n: (step + int(req_id)) % (n + 1),
622 )
623 engine.run(before_step=add_requests)
624
625 assert preempts_with_inflight_output > 0, (
626 "test did not exercise preemption with in-flight output"
627 )
628 for req in requests:

726@pytest.mark.parametrize(
727 ("kv_role", "defer_free"),
728 [
729 ("kv_producer", False),
730 ("kv_consumer", False),
731 ("kv_consumer", True),
732 ],
733)
734def test_kv_pressure_preempt_mid_handoff(kv_role: str, defer_free: bool):
735 """P/D race: KV pressure hits while the output of a request's final
736 prefill chunk -- the hand-off token that would finish it -- is in flight.
737
738 When the victim's blocks free immediately, the request is preempted. On a
739 producer, that output must be dropped so the request recomputes; delivering
740 it would finish the request and hand off blocks the preemption already
741 freed, so the consumer pulls garbage. A consumer hands nothing off, so it
742 keeps the lossless deliver-stale path.
743
744 A consumer with overlapping batches (async scheduling or PP) instead fences
745 the victim's free behind its in-flight output, so the allocation retry
746 stops instead of preempting; the request then finishes from that output
747 once it lands. The gate depends on the platform (async scheduling is
748 force-disabled on CPU), so force the flag to cover both paths everywhere.
749 """
750 is_producer = kv_role == "kv_producer"
751 scheduler = create_scheduler(
752 async_scheduling=True,
753 use_kv_connector=True,
754 kv_role=kv_role,
755 num_blocks=5,
756 block_size=16,
757 max_num_batched_tokens=512,
758 )
759 assert scheduler.requires_kv_delivery is is_producer
760 # The production gate requires overlapping batches, which async
761 # scheduling only provides off-CPU.
762 scheduler.defer_block_free = defer_free
763
764 # 32-token prompts fill 2 blocks each, exhausting the usable pool, so the
765 # next decode allocation targets the tail of the running queue (the
766 # handoff request) while its prefill output is still in flight.
767 decoder = create_requests(
768 num_requests=1, num_tokens=32, max_tokens=8, req_ids=["decoder"]
769 )[0]
770 handoff = create_requests(
771 num_requests=1, num_tokens=32, max_tokens=1, req_ids=["handoff"]
772 )[0]
773 scheduler.add_request(decoder)
774 scheduler.add_request(handoff)
775 sched_output = scheduler.schedule()
776 assert handoff.status == RequestStatus.RUNNING
777 assert handoff.num_output_placeholders == 1
778
779 scheduler.schedule()
780 if defer_free:
781 # The victim's blocks are fenced behind its in-flight output, so
782 # preempting it could not satisfy the allocation; the retry stops
783 # instead of preempting.
784 assert handoff.status == RequestStatus.RUNNING
785 assert handoff.num_stale_output_tokens == 0
786 else:
787 # Blocks free immediately, so the handoff request is preempted.
788 assert handoff.status == RequestStatus.PREEMPTED
789 assert handoff.num_stale_output_tokens == handoff.num_prompt_tokens
790 assert handoff.drop_stale_output is is_producer
791
792 scheduler.update_from_output(sched_output, _make_model_runner_output(sched_output))
793
794 assert handoff.num_stale_output_tokens == 0
795 if is_producer:
796 # Dropped: recomputed from the waiting queue, so the hand-off happens
797 # against real KV.
798 assert not handoff.is_finished()
799 assert handoff.status == RequestStatus.PREEMPTED
800 assert handoff.num_output_tokens == 0
801 assert handoff.request_id in scheduler.requests
802 else:
803 assert handoff.is_finished()
804 assert handoff.num_output_tokens == 1
