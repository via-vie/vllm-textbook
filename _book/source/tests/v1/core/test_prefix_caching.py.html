125def make_kv_cache_manager(kv_cache_config: KVCacheConfig, **kwargs) -> KVCacheManager:
126 """Build a ``KVCacheManager``, deriving ``scheduler_block_size`` from the
127 config (LCM of group block sizes) unless explicitly provided. This mirrors
128 ``resolve_kv_cache_block_sizes`` for the non-context-parallel case used by
129 these tests, so callers don't have to pass it at every site."""
130 kwargs.setdefault(
131 "scheduler_block_size",
132 lcm(*(g.kv_cache_spec.block_size for g in kv_cache_config.kv_cache_groups)),
133 )
134 if "retention_interval" in kwargs:
135 kv_cache_config = replace(
136 kv_cache_config,
137 prefix_cache_retention_interval=kwargs.pop("retention_interval"),
138 )
139 manager = KVCacheManager(kv_cache_config, **kwargs)
140 if kv_cache_config.hisparse_host_num_blocks is not None:
141 get_hisparse_coordinator(manager)
142 return manager
