591def test_throttle_capacity_bound_guard_admits():
592 """Saturation guard: if a cadence-aligned release step cannot drain the
593 waiting prefill queue (it ran out of token budget), the throttle backs off on
594 the next step so the backlog cannot grow into a TTFT avalanche -- prefills are
595 admitted even though throttle_prefills is set."""
596 scheduler = create_scheduler(
597 max_num_seqs=16, max_num_batched_tokens=200, enable_chunked_prefill=True
598 )
599 a, b = create_requests(num_requests=2, num_tokens=200, req_ids=["a", "b"])
600 scheduler.add_request(a)
601 scheduler.add_request(b)
602
603 # Release step (throttle off): `a` fills the 200-token budget; `b` cannot be
604 # reached, so the waiting queue is not drained -> capacity-bound.
605 output = scheduler.schedule()
606 assert "a" in output.num_scheduled_tokens
607 assert "b" not in output.num_scheduled_tokens
608 assert scheduler.prefill_capacity_bound
609
610 # Throttle. Because the previous release was capacity-bound, the guard backs
