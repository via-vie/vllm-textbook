3"""Tests for deferred block freeing under async scheduling.
4
5With async scheduling, a finished/preempted request's blocks may still be
6written by a speculatively over-scheduled in-flight GPU step (mamba/GDN
7layers rewrite the whole state block every step). If such a block is
8reallocated to a request arriving via PD disaggregation, the NIC/RDMA write
9of the received state races with the in-flight stale write. The scheduler
10closes the race by deferring the return of blocks to the block pool until
11the newest scheduled step's output has been processed.
12"""

51def _create_deferring_scheduler(scheduling_policy="fcfs"):

53
54 The production gate additionally requires a PD KV-consumer connector;
55 the mechanism itself is independent of it.
56 """

66def _setup_request_with_inflight_step(scheduler, max_tokens: int = 5):
67 """Schedule a request's prefill (step 1) and one speculatively
68 over-scheduled decode (step 2), mimicking async scheduling depth 1.
69
70 Returns (request, out0, out1).
71 """
72 request = create_requests(
73 num_requests=1,
74 num_tokens=NUM_PROMPT_TOKENS,
75 max_tokens=max_tokens,
76 stop_token_ids=[STOP_TOKEN_ID],
77 )[0]
78 scheduler.add_request(request)
79 out0 = scheduler.schedule()
80 assert out0.num_scheduled_tokens[request.request_id] == NUM_PROMPT_TOKENS
81 out1 = scheduler.schedule()
82 assert out1.num_scheduled_tokens[request.request_id] == 1
83 return request, out0, out1

86def _fail_one_allocation(scheduler, call_number: int):
87 real_allocate = scheduler.kv_cache_manager.allocate_slots
88 calls = 0
89
90 def allocate(*args, **kwargs):
91 nonlocal calls
92 calls += 1
93 if calls == call_number:
94 return None
95 return real_allocate(*args, **kwargs)
96
97 return patch.object(
98 scheduler.kv_cache_manager,
99 "allocate_slots",
100 side_effect=allocate,
101 )

219def test_preempt_defers_free_and_clears_bookkeeping():
220 scheduler = _create_deferring_scheduler()
221 pool = scheduler.kv_cache_manager.block_pool
222 num_free_initially = pool.get_num_free_blocks()
223
224 request, out0, out1 = _setup_request_with_inflight_step(scheduler)
225 num_free_running = pool.get_num_free_blocks()
226
227 # Preempt the request while steps are in flight (mirrors the
228 # preemption path inside schedule()).
229 scheduler.running.remove(request)
230 scheduler._preempt_request(request, time.monotonic())
231 assert request.status == RequestStatus.PREEMPTED
232
233 # Blocks are withheld from the pool, but the manager bookkeeping is
234 # cleared immediately so the request can be rescheduled safely.
235 assert len(scheduler.deferred_frees) == 1
236 assert pool.get_num_free_blocks() == num_free_running
237 for manager in scheduler.kv_cache_manager.coordinator.single_type_managers:
238 assert request.request_id not in manager.req_to_blocks
239
240 # Outputs of both in-flight steps are processed: blocks return to the
241 # pool only after the newest one.
242 scheduler.update_from_output(out0, _make_model_runner_output(out0))
243 assert len(scheduler.deferred_frees) == 1
244 scheduler.update_from_output(out1, _make_model_runner_output(out1))
245 assert not scheduler.deferred_frees
246 assert pool.get_num_free_blocks() == num_free_initially

256def test_allocation_retry_waits_for_fence_then_succeeds(policy, failed_call):
257 scheduler = _create_deferring_scheduler(policy)
258 requests = create_requests(
259 num_requests=3,
260 num_tokens=NUM_PROMPT_TOKENS,
261 max_tokens=10,
262 stop_token_ids=[STOP_TOKEN_ID],
263 )
264 for request in requests:
265 scheduler.add_request(request)
266 out0 = scheduler.schedule()
267 out1 = scheduler.schedule()
268
269 worst, trigger, tail = requests
270 worst.priority, trigger.priority, tail.priority = 9, 0, 1
271
272 with _fail_one_allocation(scheduler, failed_call) as allocate:
273 blocked = scheduler.schedule()
274 assert allocate.call_count == failed_call
275 assert not blocked.preempted_req_ids
276 retry_trigger = requests[failed_call - 1]
277 assert retry_trigger.request_id not in blocked.num_scheduled_tokens
278
279 for output in (out0, out1, blocked):
280 if output.total_num_scheduled_tokens:
281 scheduler.update_from_output(
282 output,
283 _make_model_runner_output(output),
284 )
285
286 expected_victim = worst if policy == "priority" else tail
287 with _fail_one_allocation(scheduler, failed_call) as allocate:
288 resumed = scheduler.schedule()
289
290 assert allocate.call_count > failed_call
291 assert expected_victim.request_id in resumed.preempted_req_ids
292 assert retry_trigger.request_id in resumed.num_scheduled_tokens
293 if policy == "priority":
294 assert tail.request_id in resumed.num_scheduled_tokens

326def test_fence_held_across_multiple_inflight_steps():
327 """Pipeline-parallel / deep async: with several steps scheduled ahead,
328 a freed request's blocks must stay held until the *newest* in-flight
329 step's output is processed, not the first.
330
331 Depth-1 tests only check a single intervening update; with PP the
332 scheduler can dispatch up to pp_size steps ahead, so the fence must
333 survive multiple intervening update_from_output calls.
334 """
335 scheduler = _create_deferring_scheduler()
336 pool = scheduler.kv_cache_manager.block_pool
337 num_free_initially = pool.get_num_free_blocks()
338
339 request = create_requests(
340 num_requests=1,
341 num_tokens=NUM_PROMPT_TOKENS,
342 max_tokens=10,
343 )[0]
344 scheduler.add_request(request)
345
346 # Schedule three steps ahead without processing any output: a prefill
347 # plus two speculatively over-scheduled decodes, all in flight at once.
348 outs = [scheduler.schedule() for _ in range(3)]
349 assert outs[0].num_scheduled_tokens[request.request_id] == NUM_PROMPT_TOKENS
350 assert outs[1].num_scheduled_tokens[request.request_id] == 1
351 assert outs[2].num_scheduled_tokens[request.request_id] == 1
352 assert scheduler.sched_step_seq == 3
353 num_free_running = pool.get_num_free_blocks()
354 assert num_free_running < num_free_initially
355
356 # Abort while all three steps are in flight: the fence is the newest
357 # scheduled step (3), since any of them may still write the blocks.
358 scheduler.finish_requests(request.request_id, RequestStatus.FINISHED_ABORTED)
359 assert len(scheduler.deferred_frees) == 1
360 assert scheduler.deferred_frees[0][0] == 3
361 assert pool.get_num_free_blocks() == num_free_running
362
363 # Draining the two earlier in-flight steps must NOT release the blocks:
364 # their outputs don't fence the still-pending newest write.
365 for out in (outs[0], outs[1]):
366 scheduler.update_from_output(out, _make_model_runner_output(out))
367 assert len(scheduler.deferred_frees) == 1
368 assert pool.get_num_free_blocks() == num_free_running
369
370 # Only once the newest scheduled step's output is processed do the
371 # blocks return to the pool.
372 scheduler.update_from_output(outs[2], _make_model_runner_output(outs[2]))
373 assert not scheduler.deferred_frees
374 assert pool.get_num_free_blocks() == num_free_initially
375

377def test_max_tokens_finish_frees_immediately_with_other_inflight():
378 """A request finishing by reaching max_tokens is never over-scheduled past
379 its final-token step, so no in-flight step writes its blocks: it is freed
380 immediately even while another request's step is still in flight.
381 """
382 scheduler = _create_deferring_scheduler()
383 pool = scheduler.kv_cache_manager.block_pool
384
385 # Short request finishes at max_tokens=1; long request keeps running.
386 short = create_requests(
387 num_requests=1, num_tokens=NUM_PROMPT_TOKENS, max_tokens=1, req_ids=["short"]
388 )[0]
389 long = create_requests(
390 num_requests=1, num_tokens=NUM_PROMPT_TOKENS, max_tokens=100, req_ids=["long"]
391 )[0]
392 scheduler.add_request(short)
393 scheduler.add_request(long)
394
395 out0 = scheduler.schedule() # prefill both
396 out1 = scheduler.schedule() # short is skipped (at max_tokens); long decodes
397 assert "short" not in out1.num_scheduled_tokens
398 assert "long" in out1.num_scheduled_tokens
399
400 free_before = pool.get_num_free_blocks()
401 # Process step 0: `short` reaches max_tokens and finishes while step 1
402 # (which scheduled `long`, not `short`) is still in flight.
403 scheduler.update_from_output(out0, _make_model_runner_output(out0))
404
405 assert short.is_finished()
406 # A step IS globally in flight (the old global fence would have deferred),
407 # but the per-request gate frees `short` immediately since nothing writes
408 # its blocks anymore.
409 assert scheduler.sched_step_seq > scheduler.processed_step_seq
410 assert not scheduler.deferred_frees
411 assert pool.get_num_free_blocks() > free_before # short's blocks returned

488def test_cow_retentions_deferred_until_copy_step_processed():
489 """The endpoints of a queued KV block copy must stay out of the free
490 pool until the step that runs the copy has been processed. Freed
491 earlier, an endpoint can be reallocated (e.g. as a PD KV-load
492 destination) and overwritten by a transfer that is not ordered against
493 the copy still pending in the in-flight step.
494 """
495 scheduler = _create_deferring_scheduler()
496 pool = scheduler.kv_cache_manager.block_pool
497 manager = scheduler.kv_cache_manager.coordinator.single_type_managers[0]
498
499 request = create_requests(
500 num_requests=1,
501 num_tokens=NUM_PROMPT_TOKENS,
502 max_tokens=5,
503 stop_token_ids=[STOP_TOKEN_ID],
504 )[0]
505 scheduler.add_request(request)
506
507 # Simulate a partial-hit CoW performed while scheduling step 1, whose
508 # hitting request was freed within the same step: the copy rides out0
509 # and each endpoint stays alive only through its copy retention.
510 src_block, dst_block = pool.get_new_blocks(2)
511 block_copy = KVCacheBlockCopy(
512 src_block_id=src_block.block_id, dst_block_id=dst_block.block_id
513 )
514 manager._pending_cow_copies.append((src_block, dst_block))
515 out0 = scheduler.schedule()
516 assert out0.kv_cache_block_copies == [block_copy]
517
518 # Exhaust the rest of the pool so the copy endpoints are the only blocks
519 # a new request could receive, then add one that fits exactly in them.
520 pool.get_new_blocks(pool.get_num_free_blocks())
521 late_request = create_requests(
522 num_requests=1,
523 num_tokens=2 * scheduler.block_size,
524 max_tokens=5,
525 req_ids=["late"],
526 )[0]
527 scheduler.add_request(late_request)
528
529 # Step 2 is scheduled while step 1 (which runs the copy) is still in
530 # flight: the retentions are released against the copy's fence, so the
531 # endpoints must not reach the free pool -- the late request must not be
532 # scheduled onto them.
533 out1 = scheduler.schedule()
534 assert src_block.ref_cnt == 1
535 assert dst_block.ref_cnt == 1
536 assert scheduler.deferred_frees
537 assert not out1.scheduled_new_reqs
538
539 # Step 1's output is processed: the copy has run, endpoints return to
540 # the pool and the late request can be scheduled onto them safely.
541 scheduler.update_from_output(out0, _make_model_runner_output(out0))
542 assert src_block.ref_cnt == 0
543 assert dst_block.ref_cnt == 0
544 assert not scheduler.deferred_frees
545 out2 = scheduler.schedule()
546 assert [r.req_id for r in out2.scheduled_new_reqs] == ["late"]
