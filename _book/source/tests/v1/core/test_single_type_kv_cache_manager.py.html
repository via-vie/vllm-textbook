144@pytest.mark.parametrize("block_size", [3584, 4608])
145@pytest.mark.parametrize("in_flight_chunks", [0, 1, 2])
146def test_mamba_retirement_bounds_prefill_states(block_size, in_flight_chunks):
147 spec = MambaSpec(
148 block_size=block_size,
149 shapes=((1, 1),),
150 dtypes=(torch.float32,),
151 mamba_cache_mode="align",
152 num_speculative_blocks=5,
153 )
154 pool = BlockPool(num_gpu_blocks=1000, enable_caching=True, hash_block_size=256)
155 manager = MambaManager(
156 spec,
157 block_pool=pool,
158 enable_caching=True,
159 kv_cache_group_id=0,
160 scheduler_block_size=block_size,
161 )
162 initial_free = pool.get_num_free_blocks()
163 chunk = 8192 // block_size * block_size
164 peak_held = 0
165 # Replay full prefill chunks with an async committed-token lag.
166 for target in range(chunk, 480031 + 1, chunk):
167 computed = target - chunk
168 committed = max(0, computed - in_flight_chunks * chunk)
169 manager.remove_skipped_blocks("r", committed)
170 manager.get_num_blocks_to_allocate(
171 "r", target + 5, [], computed, computed, target
172 )
173 manager.allocate_new_blocks("r", target + 5, target)
174 peak_held = max(peak_held, initial_free - pool.get_num_free_blocks())
175
176 # Five speculative blocks, current/previous states, and bounded in-flight states.
177 assert peak_held == 7 + in_flight_chunks
178 manager.free("r")
179 assert pool.get_num_free_blocks() == initial_free
