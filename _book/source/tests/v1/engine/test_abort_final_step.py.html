4"""
5Test for the fix in PR #29987: Eagerly abort cancelled final-step requests.
6
7This test verifies that when a request is aborted during its final execution
8step (when it would naturally complete), it is properly marked as aborted
9rather than being treated as normally completed.
10
11The test uses a dummy KV connector to verify that the connector receives
12the correct finish status (FINISHED_ABORTED, not FINISHED_LENGTH_CAPPED).
13"""

144@pytest.mark.parametrize("async_scheduling", [False, True])
145@pytest.mark.asyncio
146async def test_abort_during_final_step(async_scheduling: bool):
147 """
148 Test that a request aborted during its final execution step is treated as
149 aborted rather than completed.
150
151 This test:
152 1. Monkeypatches execute_model to wait for a file to be deleted
153 2. Configures a dummy KV connector to capture finish statuses
154 3. Starts a request with max_tokens=1 (will complete on first decode step)
155 4. Aborts the request, then deletes the file to unblock execute_model
156 5. Verifies the KV connector received FINISHED_ABORTED not FINISHED_LENGTH_CAPPED
157
158 See https://github.com/vllm-project/vllm/pull/29987.
159
160 Without the fix, the KV connector would see FINISHED_LENGTH_CAPPED because
161 update_from_output() would mark the request as completed before processing
162 the abort. This causes KV cache blocks to not be freed properly in
163 disaggregated prefill scenarios.
164
165 With the fix, _process_aborts_queue() runs before update_from_output(), so the
166 abort takes precedence and the KV connector sees FINISHED_ABORTED.
167 """
168
169 # Create three temporary files:
170 # 1. ready_file: deleted by execute_model to signal it has started
171 # 2. block_file: execute_model waits for this to be deleted
172 # 3. status_file: KV connector writes finish statuses here
173 with tempfile.NamedTemporaryFile(delete=False) as f:
174 ready_file = Path(f.name)
175 with tempfile.NamedTemporaryFile(delete=False) as f2:
176 block_file = Path(f2.name)
177 with tempfile.NamedTemporaryFile(delete=False, mode="w") as f3:
178 status_file = Path(f3.name)
179
180 try:
181 # Get the original execute_model method
182 from vllm.v1.worker.gpu_worker import Worker
183
184 original_execute_model = Worker.execute_model
185
186 def execute_model_with_wait(self, scheduler_output):
187 # V2's `gpu_worker.compile_or_warm_up_model` calls
188 # `warmup_kernels(...)` during engine init, which itself calls
189 # `Worker.execute_model` three times (prefill / decode / cleanup)
190 # to JIT compile triton kernels. None of those carry the test's
191 # request id, so we only stall when our actual request is being
192 # processed.
193 scheduled = scheduler_output.num_scheduled_tokens or {}
194 finished = scheduler_output.finished_req_ids or set()
195
196 def is_target_request(req_ids):
197 return any(
198 rid == request_id or rid.startswith(f"{request_id}-")
199 for rid in req_ids
200 )
201
202 if is_target_request(scheduled) or is_target_request(finished):
203 # Signal that execute_model has been called by deleting ready_file
204 if ready_file.exists():
205 ready_file.unlink()
206
207 # Wait for the block file to be deleted (triggered from test after
208 # abort). This runs in the worker process (after fork), so we poll
209 # the filesystem.
210 while block_file.exists():
211 time.sleep(0.01)
212 return original_execute_model(self, scheduler_output)
213
214 # Patch execute_model to inject the wait
215 # This happens before the worker process is forked, so the patch applies there
216 with patch.object(Worker, "execute_model", execute_model_with_wait):
217 request_id = "test-abort-final-step"
218
219 # Configure engine with dummy KV connector
220 # Pass the status file path so the connector can write to it
221 kv_transfer_config = KVTransferConfig(
222 kv_connector="DummyKVConnector",
223 kv_role="kv_both",
224 kv_connector_extra_config={"status_file": str(status_file)},
225 )
226 engine_args = AsyncEngineArgs(
227 model="meta-llama/Llama-3.2-1B-Instruct",
228 enforce_eager=True,
229 async_scheduling=async_scheduling,
230 kv_transfer_config=kv_transfer_config,
231 )
232
233 with set_default_torch_num_threads(1):
234 engine = AsyncLLM.from_engine_args(engine_args)
235
236 try:
237 # Create a request that will complete after just 1 token
238 sampling_params = SamplingParams(
239 max_tokens=1,
240 ignore_eos=True,
241 output_kind=RequestOutputKind.DELTA,
242 )
243
244 # Start generation in a task
245 outputs = []
246
247 async def generate():
248 async for output in engine.generate(
249 request_id=request_id,
250 prompt=TEXT_PROMPT,
251 sampling_params=sampling_params,
252 ):
253 outputs.append(output)
254
255 gen_task = asyncio.create_task(generate())
256
257 # Wait for execute_model to signal it has started (with timeout)
258 timeout = 5.0 # 5 second timeout
259 start_time = time.time()
260 while ready_file.exists():
261 if time.time() - start_time > timeout:
262 raise TimeoutError(
263 "Timeout waiting for execute_model to start. "
264 "The monkeypatch may not be working correctly, "
265 "for example if spawn was used instead of fork."
266 )
267 await asyncio.sleep(0.01)
268
269 # Abort the request while execute_model is blocked
270 await engine.abort(request_id)
271
272 # Now unblock execute_model by deleting the file
273 # The abort should be processed before the model output
274 block_file.unlink()
275
276 # Wait for generation to complete
277 await gen_task
278
279 # Poll for the KV connector to record the finish status
280 timeout = 5.0
281 start = time.time()
282 captured_statuses = []
283 while time.time() - start < timeout:
284 with open(status_file) as f4:
285 status_lines = f4.read().strip().split("\n")
286 captured_statuses = [
287 line
288 for line in status_lines
289 if line and line.startswith("FINISHED_")
290 ]
291 if captured_statuses:
292 break
293 await asyncio.sleep(0.05)
294 else:
295 raise TimeoutError(
296 "Timeout waiting for KV connector to record finish status."
297 )
298
299 # Verify we got output
300 assert len(outputs) > 0, "Should have received at least one output"
301
302 # The final output should have finish_reason="abort"
303 final_output = outputs[-1]
304 assert final_output.finished, (
305 "Final output should be marked as finished"
306 )
307 assert final_output.outputs[0].finish_reason == "abort", (
308 f"Expected finish_reason='abort' but got "
309 f"'{final_output.outputs[0].finish_reason}'. "
310 )
311
312 assert len(captured_statuses) >= 1, (
313 f"Expected at least 1 captured finish status, got "
314 f"{len(captured_statuses)}. File content: {status_lines}"
315 )
316
317 assert "FINISHED_ABORTED" in captured_statuses, (
318 f"KV connector should see FINISHED_ABORTED but got "
319 f"{captured_statuses}. "
320 )
321
322 # Verify cleanup
323 assert not engine.output_processor.has_unfinished_requests()
324
325 finally:
326 # Shutdown the engine
327 engine.shutdown()
328
329 finally:
330 # Clean up temporary files if they still exist
331 if ready_file.exists():
332 ready_file.unlink()
333 if block_file.exists():
334 block_file.unlink()
335 if status_file.exists():
336 status_file.unlink()
