246 Test that the engine can handle multiple concurrent batches.

249 def make_request_with_max_tokens(req_id: str, max_tokens: int) -> EngineCoreRequest:

251 request.request_id = req_id

255 class DummyExecutor(UniProcExecutor):
256 def initialize_from_config(self, kv_cache_configs: list[KVCacheConfig]) -> None:
257 super().initialize_from_config(kv_cache_configs)
258
259 # Create a thread pool with a single worker
260 self.thread_pool = ThreadPoolExecutor(max_workers=1)
261
262 def execute_model(
263 self,
264 scheduler_output,
265 non_block=False,
266 ) -> Future[ModelRunnerOutput | None]:
267 """Make execute_model non-blocking."""
268
269 # DummyExecutor used only for testing async case.
270 assert non_block
271
272 def _execute():
273 output = self.collective_rpc("execute_model", args=(scheduler_output,))
274 # Make a copy because output[0] may be reused
275 # by the next batch.
276 return copy.deepcopy(output[0])
277
278 # Use the thread pool instead of creating a new thread
279 return self.thread_pool.submit(_execute)
280
281 def sample_tokens(
282 self, grammar_output, non_block=False
283 ) -> Future[ModelRunnerOutput]:
284 """Make sample_tokens non-blocking."""
285
286 # DummyExecutor used only for testing async case.
287 assert non_block
288
289 def _execute():
290 output = self.collective_rpc("sample_tokens", args=(grammar_output,))
291 # Make a copy because output[0] may be reused
292 # by the next batch.
293 return copy.deepcopy(output[0])
294
295 # Use the thread pool instead of creating a new thread
296 return self.thread_pool.submit(_execute)
297
298 def shutdown(self):
299 if hasattr(self, "thread_pool"):
300 self.thread_pool.shutdown(wait=False)
301
302 engine_args = EngineArgs(
303 model=MODEL_NAME,
304 # To test concurrent batches.
305 max_num_seqs=2,
306 # Avoid all requests being scheduled once.
307 enable_prefix_caching=False,
308 max_num_batched_tokens=10,
309 # Reduce startup time.
310 enforce_eager=True,
311 # Test concurrent batch behaviour independently of async scheduling.
312 async_scheduling=False,
313 )
314 vllm_config = engine_args.create_engine_config()
315 # Force two concurrent batches to exercise the batch queue independently
316 # of async scheduling (which is disabled above).
317 with (
318 set_default_torch_num_threads(1),
319 patch.object(
320 VllmConfig,
321 "max_concurrent_batches",
322 new_callable=PropertyMock,
323 return_value=2,
324 ),
325 ):
326 engine_core = EngineCore(
327 vllm_config=vllm_config, log_stats=False, executor_class=DummyExecutor
328 )
329 assert engine_core.batch_queue is not None
330
331 # Add two requests in a row. Each request have 12 prompt tokens.
332 req0 = make_request_with_max_tokens("0", 5)
333 engine_core.add_request(*engine_core.preprocess_add_request(req0))
334 req1 = make_request_with_max_tokens("1", 5)
335 engine_core.add_request(*engine_core.preprocess_add_request(req1))
336
337 # Schedule Batch 1: (10, req0)
338 assert engine_core.step_with_batch_queue()[0] is None
339 assert len(engine_core.batch_queue) == 1
340 scheduler_output = engine_core.batch_queue[-1][1]
341 assert scheduler_output.num_scheduled_tokens["0"] == 10
342 # num_computed_tokens should have been updated immediately.
343 assert engine_core.scheduler.requests[req0.request_id].num_computed_tokens == 10
344
345 # Schedule Batch 2: (2, req0), (8, req1)
346 assert engine_core.step_with_batch_queue()[0] == {}
347 assert len(engine_core.batch_queue) == 1
348 scheduler_output = engine_core.batch_queue[-1][1]
349 assert scheduler_output.num_scheduled_tokens["0"] == 2
350 assert scheduler_output.num_scheduled_tokens["1"] == 8
351 # num_computed_tokens should have been updated immediately.
352 assert engine_core.scheduler.requests["0"].num_computed_tokens == 12
353 assert engine_core.scheduler.requests["1"].num_computed_tokens == 8
354
355 assert engine_core.scheduler.get_num_unfinished_requests() == 2
356
357 # Finish Batch 1 and schedule Batch 3: (4, req1).
358 # Note that req0 cannot be scheduled
359 # because it is in the decoding stage now.
360 engine_core.step_with_batch_queue()
361 assert len(engine_core.batch_queue) == 1
362 scheduler_output = engine_core.batch_queue[-1][1]
363 assert scheduler_output.num_scheduled_tokens["1"] == 4
364
365 # Finish Batch 2. Get first token of req0.
366 # Schedule Batch 4: (1, req0).
367 output = engine_core.step_with_batch_queue()[0].get(0)
368 assert output is not None
369 assert len(output.outputs) == 1
370 assert engine_core.scheduler.requests[req0.request_id].num_tokens == 13
371 scheduler_output = engine_core.batch_queue[-1][1]
372 assert scheduler_output.num_scheduled_tokens["0"] == 1
373
374 # Finish Batch 3. Get first token of req1. Schedule Batch 5: (1, req1).
375 output = engine_core.step_with_batch_queue()[0].get(0)
376 assert output is not None
377 assert len(output.outputs) == 1
378 assert engine_core.scheduler.requests[req1.request_id].num_tokens == 13
379 scheduler_output = engine_core.batch_queue[-1][1]
380 assert scheduler_output.num_scheduled_tokens["1"] == 1
381
382 # Loop until req0 is finished.
383 req_id = 0
384 expected_num_tokens = [
385 engine_core.scheduler.requests["0"].num_tokens + 1,
386 engine_core.scheduler.requests["1"].num_tokens + 1,
387 ]
388 while engine_core.scheduler.get_num_unfinished_requests() == 2:
389 output = engine_core.step_with_batch_queue()[0]
390 # Every step consumes an output.
391 assert output is not None
392 assert len(output[0].outputs) == 1
393 if req_id in engine_core.scheduler.requests:
394 assert (
395 engine_core.scheduler.requests[req_id].num_tokens
396 == expected_num_tokens[req_id]
397 )
398 expected_num_tokens[req_id] += 1
399 req_id = (req_id + 1) % 2

614def _cadenced_dp_engine_core(
615 monkeypatch, results: list[tuple[bool, bool]], dp_sync_interval: int = 32
616):
617 """A bare DPEngineCoreProc whose all-reduce is scripted by `results`;
618 returns the core and the step numbers at which the all-reduce ran."""
619 core = object.__new__(DPEngineCoreProc)
620 core.dp_group = object()
621 core.dp_sync_interval = dp_sync_interval
622 core.step_counter = 0
623 core.pending_pause = False
624 core.ignore_start_dp_wave = False
625 synced: list[int] = []
626
627 def scripted_sync(dp_group, has_unfinished, pending_pause):
628 synced.append(core.step_counter)
629 return results.pop(0)
630
631 monkeypatch.setattr(ParallelConfig, "sync_dp_state", staticmethod(scripted_sync))
632 return core, synced

678@pytest.mark.parametrize("deferred", [False, True])
679def test_pause_synchronizes_device_before_cache_reset(deferred: bool):
680 """A resolved pause promises an idle device: the barrier must run before
681 caches are cleared and before the caller is unblocked."""
682 core = _pausable_engine_core_proc()
683 core.engines_running = deferred
684 order: list[str] = []
685 core.model_executor.collective_rpc.side_effect = lambda method: order.append(method)
686 core._reset_caches = lambda: order.append("reset_caches")
687
688 result = EngineCoreProc.pause_scheduler(core, mode="keep", clear_cache=True)
689 if deferred:
690 assert isinstance(result, Future)
691 assert not result.done() and order == []
692 core.engines_running = False
693 core._notify_idle_state_callbacks()
694 assert result.result(timeout=0) is None
695 else:
696 assert result is None
697 assert order == ["synchronize_device", "reset_caches"]
