106async def generate(
107 engine: AsyncLLM,
108 request_id: str,
109 prompt: PromptType,
110 output_kind: RequestOutputKind,
111 max_tokens: int,
112 n: int = 1,
113 prompt_logprobs: int | None = None,
114 cancel_after: int | None = None,
115) -> tuple[int, str]:
116 # Ensure generate doesn't complete too fast for cancellation test.
117 await asyncio.sleep(0.2)
118
119 count = 0
120 sampling_params = SamplingParams(
121 max_tokens=max_tokens,
122 ignore_eos=True,
123 output_kind=output_kind,
124 temperature=0.5,
125 seed=33,
126 n=n,
127 prompt_logprobs=prompt_logprobs,
128 )
129 async for out in engine.generate(
130 request_id=request_id, prompt=prompt, sampling_params=sampling_params
131 ):
132 num_tokens = sum(len(output.token_ids) for output in out.outputs)
133 if output_kind == RequestOutputKind.DELTA:
134 count += num_tokens
135 else:
136 count = num_tokens
137
138 if cancel_after is not None and count >= cancel_after:
139 return count, request_id
140
141 await asyncio.sleep(0.0)
142
143 return count, request_id

643async def test_abort_final_output(output_kind: RequestOutputKind):
644 """Test that abort() returns a final output with correct information."""
645
646 with ExitStack() as after:
647 with set_default_torch_num_threads(1):
648 engine = AsyncLLM.from_engine_args(TEXT_ENGINE_ARGS)
649 after.callback(engine.shutdown)
650
651 request_id = "test-abort-final-output"
652
653 # Start a long-running request
654 sampling_params = SamplingParams(
655 max_tokens=3000, # Long enough to allow abort
656 ignore_eos=True,
657 output_kind=output_kind,
658 temperature=0.5,
659 seed=42,
660 )
661
662 outputs: list[RequestOutput] = []
663 generated = asyncio.create_task(
664 collect_outputs(engine, request_id, TEXT_PROMPT, sampling_params, outputs)
665 )
666
667 # Let it generate some tokens
668 await asyncio.sleep(0.5)
669
670 # Abort the request
671 await engine.abort(request_id, internal=False)
672
673 # Wait for generation to complete and return final output
674 final_output = await generated
675
676 # Verify we got a final output
677 assert final_output is not None
678 assert final_output.finished
679 assert len(final_output.outputs) == 1
680
681 assert final_output.outputs[0].finish_reason == "abort"
682 assert final_output.outputs[0].stop_reason is None
683
684 # Verify num_cached_tokens is set correctly
685 assert hasattr(final_output, "num_cached_tokens")
686 assert final_output.num_cached_tokens >= 0
687
688 # If we got intermediate outputs, verify they are consistent
689 if output_kind == RequestOutputKind.DELTA:
690 # For DELTA, sum all intermediate tokens should <= final tokens
691 token_count = sum(len(output.outputs[0].token_ids) for output in outputs)
692 assert token_count > 0
693 # This would ordinarily be 0, but could end up > 0 if the
694 # final abort is coalesced with another chunk in the output queue.
695 assert len(final_output.outputs[0].token_ids) >= 0
696 else:
697 # For FINAL_ONLY, we should only get the final output
698 assert len(outputs) == 0
699 assert len(final_output.outputs[0].token_ids) > 0
700
