63def _ref_convert_id_to_token(

67 """Reference impl of logprobs detokenization.

69 Args:
70 tokenizer: tokenizer used by the model under test

207 pooling_params=None,

250def _validate_logprobs(
251 gen_tokens: dict[str, list[int]],
252 gen_logprobs: dict[str, SampleLogprobs | None],
253 gen_prompt_logprobs: dict[str, PromptLogprobs | None],
254 gen_cumulative_logprob: dict[str, float],
255 dtv: DummyOutputProcessorTestVectors,
256 request_id_list: list[str],
257 num_sample_logprobs: int | None,
258 num_prompt_logprobs: int | None,
259) -> None:
260 for req_idx, req_id in enumerate(request_id_list):
261 new_tokens = gen_tokens[req_id]
262 logprobs = gen_logprobs[req_id]
263 prompt_logprobs = gen_prompt_logprobs[req_id]
264 cumulative_logprob = gen_cumulative_logprob[req_id]
265 prompt_token_ids = dtv.prompt_tokens[req_idx]
266 ref_logprobs = dtv.generation_logprobs[req_idx]
267 ref_prompt_logprobs = dtv.prompt_logprobs[req_idx]
268 if num_sample_logprobs is not None:
269 # Validate sample logprobs
270 assert logprobs is not None, (
271 f"Request {req_id} requires sample"
272 " logprobs but sample logprobs are"
273 " None."
274 )
275 # Require num sampled tokens to match num
276 # sampled logprobs - especially important
277 # to check since the detokenizer can cause
278 # a request to finish early due to a stop
279 # string being hit
280 num_new_tokens = len(new_tokens)
281 len_sample_logprobs = len(logprobs)
282 assert num_new_tokens == len_sample_logprobs, (
283 f"Request {req_id} has {num_new_tokens}"
284 " completion tokens but has"
285 f" {len_sample_logprobs} sample logprobs."
286 )
287 ref_cumulative_logprob = 0.0
288 for idx, (sampled_token, pos_logprob_dict) in enumerate(
289 zip(new_tokens, logprobs)
290 ):
291 # Break out the reference log probability value &
292 # logprob token id tensors associated with this
293 # position in the completion. Also break out the
294 # sampled token ranks
295 (ref_pos_logprob_toks, ref_pos_logprob_vals, ref_sampled_token_rank) = (
296 ref_logprobs[idx]
297 )
298 # For each position in the completion sequence,
299 # ensure the actual sampled token is among the
300 # logprobs
301 assert sampled_token in pos_logprob_dict, (
302 f"Sampled token {sampled_token} not"
303 f" present in logprob at index {idx}"
304 )
305
306 # Validate number of sample logprobs
307 num_lp_toks = len(pos_logprob_dict)
308 assert (
309 num_lp_toks == num_sample_logprobs
310 or num_lp_toks == num_sample_logprobs + 1
311 ), (
312 "Valid numbers of sample logprobs are"
313 f" {num_sample_logprobs} or"
314 f" {num_sample_logprobs + 1} but"
315 f" {num_lp_toks} logprobs found at"
316 f" position {idx}. Logprobs dict:"
317 f" {pos_logprob_dict}"
318 )
319
320 # Validate sampled token logprob rank
321 smp_lp = pos_logprob_dict[sampled_token]
322 smp_lp_rank = smp_lp.rank
323 assert ref_sampled_token_rank == smp_lp_rank, (
324 "Sampled token logprob rank"
325 f" {smp_lp_rank} does not match"
326 " correct value"
327 f" {ref_sampled_token_rank}"
328 f" in Logprob {smp_lp}"
329 )
330
331 # Validate that the logprob processor yields
332 # the correct log probabilities and valid
333 # rankings
334 rank_one_appears = False
335 for jdx in range(1, len(ref_pos_logprob_toks)):
336 # Iterate over the (logprob val,logprob tok id)
337 # pairs expected by the test fixture at this
338 # position in the completion.
339 ref_lp_val = ref_pos_logprob_vals[jdx]
340 ref_tok_id = ref_pos_logprob_toks[jdx]
341 assert ref_tok_id in pos_logprob_dict, (
342 f"Expected token {ref_tok_id} to be"
343 f" in logprob dict but it is not."
344 )
345
346 # Extract actually-generated logprob
347 # info
348 lp = pos_logprob_dict[ref_tok_id]
349 lp_val = lp.logprob
350 lp_rank = lp.rank
351
352 # A "top" (rank 1) logprob must be
353 # present
354 rank_one_appears = True if lp_rank == 1 else rank_one_appears
355
356 # Rank must be >= 1
357 assert lp_rank >= 1, (
358 f"Logprob {lp} has invalid"
359 f" rank {lp_rank} < 1."
360 f" Logprob dict: {pos_logprob_dict}"
361 )
362
363 # Validate log probability
364 assert math.isclose(lp_val, ref_lp_val), (
365 f"Token id {ref_tok_id} appears in logprobs dict"
366 f" at position {idx} in completion with log"
367 f" probability {lp_val} but {ref_lp_val} was"
368 f" expected. Logprob: {lp}"
369 )
370
371 assert rank_one_appears, (
372 f"No Logprob has rank 1"
373 " in the following Logprob"
374 f" dict: {pos_logprob_dict}"
375 )
376
377 # Validate logprobs detokenization
378 for lp_tok in pos_logprob_dict:
379 # Confirm that sample logprob decoded token matches
380 # the logprob token id at this sequence position
381 decoded_token = pos_logprob_dict[lp_tok].decoded_token
382 ref_decoded_token = _ref_convert_id_to_token(dtv.tokenizer, lp_tok)
383
384 # With UTF-8 correction logic, tokens ending with "�"
385 # (incomplete byte sequences) are corrected to either
386 # empty string or proper UTF-8 characters
387 if ref_decoded_token.endswith("�"):
388 # Token needs UTF-8 correction
389 assert not decoded_token.endswith("�"), (
390 f"Sampled logprob token id {lp_tok} decodes to"
391 f" '{ref_decoded_token}' (ends with replacement char)"
392 f" but corrected decoded token '{decoded_token}'"
393 f" still ends with replacement char"
394 f" (at position {idx}). UTF-8 correction should"
395 f" have removed it."
396 )
397 else:
398 # No correction needed, should match exactly
399 assert decoded_token == ref_decoded_token, (
400 f"Sampled logprob token id {lp_tok} decodes to"
401 f" {ref_decoded_token} but Logprob decoded"
402 f" token is {decoded_token} instead"
403 f" (at position {idx})"
404 )
405
406 ref_cumulative_logprob += pos_logprob_dict[sampled_token].logprob
407 # Assert that cumulative logprobs are correct
408 assert math.isclose(cumulative_logprob, ref_cumulative_logprob)
409 else:
410 # Sample logprobs disabled for this request
411 assert logprobs is None
412 assert cumulative_logprob is None
413
414 if num_prompt_logprobs is not None:
415 # Validate prompt logprobs
416 assert prompt_logprobs is not None, (
417 f"Request {req_id} requires prompt"
418 " logprobs but prompt logprobs are"

420 )

422 # prompt logprobs
