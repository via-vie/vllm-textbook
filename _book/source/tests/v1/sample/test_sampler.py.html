55def _create_bad_words_token_ids(
56 batch_size: int,
57 vocab_size: int,
58 bad_words_lengths: tuple[int, ...],
59) -> dict[int, list[list[int]]]:
60 bad_words_token_ids = {}
61 for batch_idx in range(batch_size):
62 token_ids_single_batch = []
63 for bad_words_length in bad_words_lengths:
64 token_ids = np.random.choice(
65 vocab_size, size=bad_words_length, replace=True
66 ).tolist()
67 token_ids_single_batch.append(token_ids)
68 bad_words_token_ids[batch_idx] = token_ids_single_batch
69 if batch_size >= 2:
70 # Test no bad_words for some batch
71 no_bad_words_batch_idx = np.random.choice(batch_size)
72 bad_words_token_ids.pop(no_bad_words_batch_idx, None)
73 return bad_words_token_ids

76# Returns all last tokens of bad word sequences that share the same prefix
77# as `given_prefix` (excluding the last token).
78def _collect_suffixes_with_same_prefix(
79 given_prefix: list[int], bad_words_token_ids: list[list[int]]
80) -> list[int]:
81 return [bwt[-1] for bwt in bad_words_token_ids if bwt[:-1] == given_prefix]

84# generate a valid token id that is not in bad_words_token_ids
85def _generate_valid_token_id(
86 bad_words_token_ids: list[list[int]], vocab_size: int
87) -> int:
88 forbidden_start_tokens = set()
89 for bad_word in bad_words_token_ids:
90 forbidden_start_tokens.add(bad_word[0])
91 # Get a safe token that's not in forbidden starts
92 safe_token_candidates = list(set(range(vocab_size)) - forbidden_start_tokens)
93 # Pick a random safe token
94 return np.random.choice(safe_token_candidates)

97def _update_output_token_ids_for_bad_words(
98 metadata: SamplingMetadata, vocab_size: int
99) -> dict[int, list[int]]:
100 bad_words_last_tokens = {}
101 for batch_idx, bad_words_token_ids in metadata.bad_words_token_ids.items():
102 output_token_ids = metadata.output_token_ids[batch_idx]
103 bad_words_last_token: list[int] = []
104 for i, bad_word_token_ids in enumerate(bad_words_token_ids):
105 if len(bad_word_token_ids) == 1:
106 # Single token id always affects logits
107 bad_words_last_token.append(bad_word_token_ids[0])
108 else:
109 prefix_length = len(bad_word_token_ids) - 1
110 has_bad_words = np.random.choice([True, False])
111 if has_bad_words:
112 prefix = bad_word_token_ids[:-1]
113 output_token_ids[-prefix_length:] = prefix
114 # Collect all last tokens from other bad words
115 # that share this prefix
116 bad_words_last_token.extend(
117 _collect_suffixes_with_same_prefix(prefix, bad_words_token_ids)
118 )
119 break # Maximum one update to output_token_ids
120 else: # Make sure no accidental match to bad words
121 output_token_ids[-1] = _generate_valid_token_id(
122 bad_words_token_ids, vocab_size
123 )
124 bad_words_last_tokens[batch_idx] = bad_words_last_token
125 return bad_words_last_tokens

128def _create_default_sampling_metadata(
129 num_output_tokens: int,
130 batch_size: int,
131 vocab_size: int,
132 device: torch.device,
133) -> SamplingMetadata:
134 output_token_ids: list[list[int]] = []
135 prompt_token_ids: list[list[int]] = []
136 for _ in range(batch_size):
137 output_token_ids.append(
138 np.random.randint(0, vocab_size, size=num_output_tokens).tolist()
139 )
140 prompt_token_ids.append(
141 np.random.randint(
142 0, vocab_size, size=np.random.randint(1, MAX_NUM_PROMPT_TOKENS)
143 ).tolist()
144 )
145 fake_sampling_metadata = SamplingMetadata(
146 temperature=torch.full((batch_size,), 0.0),
147 all_greedy=True,
148 all_random=False,
149 top_p=None,
150 top_k=None,
151 generators={},
152 max_num_logprobs=0,
153 prompt_token_ids=_create_prompt_tokens_tensor(
154 prompt_token_ids, vocab_size, device
155 ),
156 output_token_ids=output_token_ids,
157 spec_token_ids=[[] for _ in range(batch_size)],
158 frequency_penalties=_create_penalty_tensor(batch_size, 0.0, device),
159 presence_penalties=_create_penalty_tensor(batch_size, 0.0, device),
160 repetition_penalties=_create_penalty_tensor(batch_size, 1.0, device),
161 no_penalties=True,
162 allowed_token_ids_mask=None,
163 bad_words_token_ids={},
164 logitsprocs=LogitsProcessors(),
165 )
166 return fake_sampling_metadata

169def _create_weighted_output_token_list(
170 batch_size: int, vocab_size: int
171) -> tuple[list[list[int]], list[list[int]]]:
172 """
173 Creates an output token list where each token occurs a distinct
174 number of times.
175
176 For each batch, a random subset of token IDs is selected from the
177 vocabulary. The selected tokens are then added to the output token
178 list, each with a different frequency.
179
180 Returns:
181 tuple[list[list[int]], list[list[int]]]:
182 - The first element is the output token list, where each sublist
183 corresponds to a batch and contains tokens with weighted
184 frequencies.
185 - The second element is a list of distinct token IDs for each
186 batch, ordered by their frequency in the corresponding output
187 list.
188 """
189 output_token_ids: list[list[int]] = []
190 sorted_token_ids_in_output: list[list[int]] = []
191 for _ in range(batch_size):
192 distinct_token_ids = np.random.choice(
193 vocab_size, size=np.random.randint(1, 10), replace=False
194 ).tolist()
195 sorted_token_ids_in_output.append(distinct_token_ids)
196 output_token_ids_for_batch = []
197 for index, token_id in enumerate(distinct_token_ids):
198 output_token_ids_for_batch.extend([token_id for _ in range(index + 1)])
199 output_token_ids.append(output_token_ids_for_batch)
200 return output_token_ids, sorted_token_ids_in_output

203@pytest.mark.parametrize("device", DEVICES)
204@pytest.mark.parametrize("batch_size", [1, 2, 32])
205@pytest.mark.parametrize("presence_penalty", [-2.0, 2.0])
206def test_sampler_presence_penalty(
207 device: str, batch_size: int, presence_penalty: float
208):
209 """
210 Test to verify that if presence penalty is enabled then tokens
211 are penalized as per their presence in the existing output.
212 """
213 torch.set_default_device(device)
214 # Create fake logits where each token is assigned the same
215 # logit value.
216 fake_logits = _create_fake_logits(batch_size, VOCAB_SIZE)
217 sampling_metadata = _create_default_sampling_metadata(
218 NUM_OUTPUT_TOKENS, batch_size, VOCAB_SIZE, torch.device(device)
219 )
220 output_token_ids = sampling_metadata.output_token_ids
221 sampling_metadata.presence_penalties = _create_penalty_tensor(
222 batch_size, presence_penalty, torch.device(device)
223 )
224 sampling_metadata.no_penalties = False
225 sampler = Sampler()
226 logits = sampler.apply_penalties(
227 fake_logits, sampling_metadata, sampling_metadata.output_token_ids
228 )
229 logits = logits.cpu()
230 for batch_idx in range(batch_size):
231 # Since all tokens initially have the same logits, the non-penalized
232 # token ID will be the one with the highest logit value, while the
233 # penalized token ID will be the one with the lowest logit value.
234 non_penalized_token_id = logits[batch_idx].argmax().item()
235 penalized_token_id = logits[batch_idx].argmin().item()
236 if presence_penalty > 0:
237 # If `presence_penalty` is set to a value greater than 0, it
238 # indicates a preference for new tokens over those already
239 # present in the output.
240 # Verify that the penalized token ID exists in the output, while the
241 # non-penalized token ID does not.
242 assert penalized_token_id in output_token_ids[batch_idx]
243 assert non_penalized_token_id not in output_token_ids[batch_idx]
244 elif presence_penalty < 0:
245 # If `presence_penalty` is set to a value less than 0, it indicates
246 # a preference for existing tokens over new ones. Verify that the
247 # non-penalized token ID exists in the output, while the penalized
248 # token ID does not.
249 assert non_penalized_token_id in output_token_ids[batch_idx]
250 assert penalized_token_id not in output_token_ids[batch_idx]

256def test_sampler_frequency_penalty(
257 device: str, batch_size: int, frequency_penalty: float

260 Test to verify that if frequency penalty is enabled then tokens are
261 penalized as per their frequency of occurrence.
262 """

268 NUM_OUTPUT_TOKENS, batch_size, VOCAB_SIZE, torch.device(device)
269 )
270 sampling_metadata.frequency_penalties = _create_penalty_tensor(
271 batch_size, frequency_penalty, torch.device(device)
272 )
273 output_token_ids, sorted_token_ids_in_output = _create_weighted_output_token_list(
274 batch_size,
275 VOCAB_SIZE,
276 )
277 sampling_metadata.output_token_ids = output_token_ids
278 sampling_metadata.no_penalties = False
279 sampler = Sampler()
280 logits = sampler.apply_penalties(
281 fake_logits, sampling_metadata, sampling_metadata.output_token_ids
282 )

285 non_penalized_token_id = logits[batch_idx].argmax().item()
286 penalized_token_id = logits[batch_idx].argmin().item()
287 distinct_sorted_token_ids_in_output = sorted_token_ids_in_output[batch_idx]
288 most_frequent_token_id = distinct_sorted_token_ids_in_output[
289 len(distinct_sorted_token_ids_in_output) - 1
290 ]
291 if frequency_penalty > 0:
292 # If `frequency_penalty` is set to > 0, it indicates

298 assert penalized_token_id == most_frequent_token_id
299 elif frequency_penalty < 0:
300 # If `frequency_penalty` is set to < 0, it indicates
301 # a preference for existing tokens over new ones. Verify that the
302 # non-penalized token ID is the one that occurs most frequently
303 # in the output, while the penalized token ID is one that has not
304 # yet appeared.
305 assert non_penalized_token_id == most_frequent_token_id
306 assert penalized_token_id not in distinct_sorted_token_ids_in_output

309@pytest.mark.parametrize("device", DEVICES)
310@pytest.mark.parametrize("batch_size", [1, 2, 32])
311@pytest.mark.parametrize("repetition_penalty", [0.1, 1.9])
312def test_sampler_repetition_penalty(
313 device: str, batch_size: int, repetition_penalty: float
314):
315 """
316 Test to verify that when the repetition penalty is enabled, tokens
317 are penalized based on their presence in the prompt or the existing
318 output.
319 """
320 torch.set_default_device(device)
321 # Create fake logits where each token is assigned the same
322 # logit value.
323 fake_logits = _create_fake_logits(batch_size, VOCAB_SIZE)
324 sampling_metadata = _create_default_sampling_metadata(
325 NUM_OUTPUT_TOKENS, batch_size, VOCAB_SIZE, torch.device(device)
326 )
327 sampling_metadata.repetition_penalties = _create_penalty_tensor(
328 batch_size, repetition_penalty, torch.device(device)
329 )
330 sampling_metadata.no_penalties = False
331 sampler = Sampler()
332 logits = sampler.apply_penalties(
333 fake_logits, sampling_metadata, sampling_metadata.output_token_ids
334 )
335 logits = logits.cpu()
336 for batch_idx in range(batch_size):
337 non_penalized_token_id = logits[batch_idx].argmax().item()
338 penalized_token_id = logits[batch_idx].argmin().item()
339 prompt_tokens = sampling_metadata.prompt_token_ids[batch_idx][:].tolist()
340 output_tokens = sampling_metadata.output_token_ids[batch_idx]
341 if repetition_penalty > 1.0:
342 # If `repetition_penalty` > 1.0, verify that the non-penalized
343 # token ID has not been seen before, while the penalized token ID
344 # exists either in the prompt or the output.
345 assert (
346 non_penalized_token_id not in prompt_tokens
347 and non_penalized_token_id not in output_tokens
348 )
349 assert (
350 penalized_token_id in prompt_tokens
351 or penalized_token_id in output_tokens
352 )
353 elif repetition_penalty < 1.0:
354 # If `repetition_penalty` < 1.0, verify that the penalized
355 # token ID has not been seen before, while the non-penalized
356 # token ID exists either in the prompt or the output.
357 assert (
358 penalized_token_id not in prompt_tokens
359 and penalized_token_id not in output_tokens
360 )
361 assert (
362 non_penalized_token_id in prompt_tokens
363 or non_penalized_token_id in output_tokens
364 )

367@pytest.mark.parametrize("device", DEVICES)
368@pytest.mark.parametrize("batch_size", [1, 2, 32])
369@pytest.mark.parametrize("num_allowed_token_ids", [0, 1, 2])
370def test_sampler_allowed_token_ids(
371 device: str, batch_size: int, num_allowed_token_ids: int
372):
373 """
374 Test to verify that when the repetition penalty is enabled, tokens
375 are penalized based on their presence in the prompt or the existing
376 output.
377 """
378 torch.set_default_device(device)
379 # Create fake logits where each token is assigned the same
380 # logit value.
381 fake_logits = _create_fake_logits(batch_size, VOCAB_SIZE)
382 sampling_metadata = _create_default_sampling_metadata(
383 NUM_OUTPUT_TOKENS, batch_size, VOCAB_SIZE, torch.device(device)
384 )
385 mask = create_allowed_token_ids(
386 batch_size=batch_size,
387 vocab_size=VOCAB_SIZE,
388 num_allowed_token_ids=num_allowed_token_ids,
389 device=device,
390 )
391 sampling_metadata.allowed_token_ids_mask = mask
392 sampler = Sampler()
393 logits = sampler.apply_logits_processors(
394 fake_logits, sampling_metadata, predict_bonus_token=False
395 )
396 logits = logits.cpu()
397 for batch_idx in range(batch_size):
398 logits_for_req = logits[batch_idx]
399 if batch_idx % 2 == 1:
400 assert torch.all(logits_for_req != -float("inf"))
401 continue
402 for token_id in range(VOCAB_SIZE):
403 start = min(batch_idx, VOCAB_SIZE - 1)
404 end = min(batch_idx + num_allowed_token_ids, VOCAB_SIZE - 1)
405 if token_id >= start and token_id < end:
406 assert logits_for_req[token_id] == -float("inf"), (
407 f"{batch_idx}, {token_id}"
408 )
409 else:
410 assert logits_for_req[token_id] != -float("inf")

413@pytest.mark.parametrize("device", DEVICES)
414@pytest.mark.parametrize("batch_size", [1, 2, 32])
415@pytest.mark.parametrize("bad_words_lengths", [(1,), (1, 3), (2, 2)])
416def test_sampler_bad_words(
417 device: str, batch_size: int, bad_words_lengths: tuple[int, ...]
418):
419 """
420 Test to verify that when the bad words restriction is present, tokens
421 are penalized based on their match with the bad words.
422 """
423 torch.set_default_device(device)
424 # Create fake logits where each token is assigned the same
425 # logit value.
426 fake_logits = _create_fake_logits(batch_size, VOCAB_SIZE)
427 sampling_metadata = _create_default_sampling_metadata(
428 NUM_OUTPUT_TOKENS, batch_size, VOCAB_SIZE, torch.device(device)

432 )
433 bad_words_last_tokens = _update_output_token_ids_for_bad_words(
434 sampling_metadata, VOCAB_SIZE
435 )
436 sampler = Sampler()
437 logits = sampler.apply_logits_processors(
438 fake_logits, sampling_metadata, predict_bonus_token=False
439 )
440 logits = logits.cpu()
441 for batch_idx in range(batch_size):
442 logits_for_req = logits[batch_idx]
443 for token_id in range(VOCAB_SIZE):

445 batch_idx in bad_words_last_tokens
446 and token_id in bad_words_last_tokens[batch_idx]

448 assert logits_for_req[token_id] == -float("inf")

450 assert logits_for_req[token_id] != -float("inf")
