35def _verify_num_sequences(value: int, parameter_name: str) -> None:
36 if not isinstance(value, int):
37 raise VLLMValidationError(
38 f"{parameter_name} must be an int, but is of type {type(value)}"

51def validate_thinking_token_budget(value: int | float | bool | None) -> int | None:
52 """Validate ``thinking_token_budget``; return ``None`` if unset."""
53 if value is None:
54 return None
55 if isinstance(value, (bool, float)) or not isinstance(value, int):
56 raise VLLMValidationError(
57 "`thinking_token_budget` must be a non-negative integer "
58 "or -1 for unlimited.",
59 parameter="thinking_token_budget",
60 value=value,
61 )
62 if value == -1:
63 return None
64 if value < 0:
65 raise VLLMValidationError(
66 "`thinking_token_budget` must be a non-negative integer "
67 "or -1 for unlimited.",
68 parameter="thinking_token_budget",
69 value=value,
70 )
71 return value

162class RepetitionDetectionParams:
163 """Parameters for detecting repetitive N-gram patterns in output tokens."""
164
165 max_pattern_size: int = 0
166 """Maximum size of N-gram pattern to detect for sequence repetition.
167 Set to 0 to disable. Must be used together with min_count."""
168
169 min_pattern_size: int = 0
170 """Minimum N-gram pattern size to check for sequence repetition.
171 If set to 0, it defaults to 1.
172 Must be <= max_pattern_size."""
173
174 min_count: int = 0
175 """Minimum number of times an N-gram pattern must repeat to trigger
176 detection. Must be >= 2. Example: 3 for detecting a phrase repeated
177 3 times. Must be used together with max_pattern_size."""
178
179 def __post_init__(self):
180 if (
181 self.max_pattern_size < 0
182 or self.min_pattern_size < 0
183 or self.min_pattern_size > self.max_pattern_size
184 ):
185 raise VLLMValidationError(
186 "max_pattern_size, min_pattern_size must be >=0, "
187 "with min_pattern_size <= max_pattern_size. "
188 "Set both to 0 to disable repetitive pattern detection."
189 )
190 if self.max_pattern_size > 0 and self.min_count < 2:
191 raise VLLMValidationError(
192 "min_count must be >= 2 to detect repetitive patterns "
193 "in engine output. If you do not wish to detect repetitive "
194 "patterns, set max_pattern_size to 0."

215class SamplingParams(
216 PydanticMsgspecMixin,
217 msgspec.Struct,
218 omit_defaults=True, # type: ignore[call-arg]
219 # required for @cached_property.
220 dict=True,
221): # type: ignore[call-arg]
222 """Sampling parameters for text generation.
223
224 Overall, we follow the sampling parameters from the OpenAI text completion
225 API (https://platform.openai.com/docs/api-reference/completions/create).
226 In addition, we support beam search, which is not supported by OpenAI.
227 """
228
229 n: int = 1
230 """Number of outputs to return for the given prompt request.
231
232 The maximum allowed value is controlled by the ``VLLM_MAX_N_SEQUENCES``
233 environment variable (default: 16384).
234
235 NOTE:
236 `AsyncLLM` streams outputs by default. When `n > 1`, all `n` outputs
237 are generated and streamed cumulatively per request. To see all `n`
238 outputs upon completion, use `output_kind=RequestOutputKind.FINAL_ONLY`
239 in `SamplingParams`."""
240 presence_penalty: float = 0.0
241 """Penalizes new tokens based on whether they appear in the generated text
242 so far. Values > 0 encourage the model to use new tokens, while values < 0
243 encourage the model to repeat tokens."""
244 frequency_penalty: float = 0.0
245 """Penalizes new tokens based on their frequency in the generated text so
246 far. Values > 0 encourage the model to use new tokens, while values < 0
247 encourage the model to repeat tokens."""
248 repetition_penalty: float = 1.0
249 """Penalizes new tokens based on whether they appear in the prompt and the
250 generated text so far. Values > 1 encourage the model to use new tokens,
251 while values < 1 encourage the model to repeat tokens."""
252 temperature: float = 1.0
253 """Controls the randomness of the sampling. Lower values make the model
254 more deterministic, while higher values make the model more random. Zero
255 means greedy sampling."""
256 watermarking: bool = True
257 """Whether to apply the engine's configured watermark to this request."""
258 top_p: float = 1.0
259 """Controls the cumulative probability of the top tokens to consider. Must
260 be in (0, 1]. Set to 1 to consider all tokens."""
261 top_k: int = 0
262 """Controls the number of top tokens to consider. Set to 0 (or -1) to
263 consider all tokens."""
264 min_p: float = 0.0
265 """Represents the minimum probability for a token to be considered,
266 relative to the probability of the most likely token. Must be in [0, 1].
267 Set to 0 to disable this."""
268 seed: int | None = None
269 """Random seed to use for the generation."""
270 stop: str | list[str] | None = None
271 """String(s) that stop the generation when they are generated. The returned
272 output will not contain the stop strings."""
273 stop_token_ids: list[int] | None = None
274 """Token IDs that stop the generation when they are generated. The returned
275 output will contain the stop tokens unless the stop tokens are special
276 tokens."""
277 ignore_eos: bool = False
278 """Whether to ignore the EOS token and continue generating
279 tokens after the EOS token is generated."""
280 max_tokens: int | None = 16
281 """Maximum number of tokens to generate per output sequence."""
282 min_tokens: int = 0
283 """Minimum number of tokens to generate per output sequence before EOS or
284 `stop_token_ids` can be generated"""
285 logprobs: int | None = None
286 """Number of log probabilities to return per output token. When set to
287 `None`, no probability is returned. If set to a non-`None` value, the
288 result includes the log probabilities of the specified number of most
289 likely tokens, as well as the chosen tokens. Note that the implementation
290 follows the OpenAI API: The API will always return the log probability of
291 the sampled token, so there may be up to `logprobs+1` elements in the
292 response. When set to -1, return all `vocab_size` log probabilities."""
293 prompt_logprobs: int | None = None
294 """Number of log probabilities to return per prompt token.
295 When set to -1, return all `vocab_size` log probabilities."""
296 logprob_token_ids: list[int] | None = None
297 """Specific token IDs to return logprobs for. More efficient than
298 logprobs=-1 when you only need logprobs for a small set of tokens.
299 When set, logprobs for exactly these token IDs will be returned,
300 in addition to the sampled token. This is useful for scoring tasks
301 where you want to compare probabilities of specific label tokens."""
302 flat_logprobs: bool = False
303 """Whether to return logprobs in flatten format (i.e. FlatLogprob)
304 for better performance.
305 NOTE: GC costs of FlatLogprobs is significantly smaller than
306 list[dict[int, Logprob]]. After enabled, PromptLogprobs and
307 SampleLogprobs would populated as FlatLogprobs."""
308 # NOTE: This parameter is only exposed at the engine level for now.
309 # It is not exposed in the OpenAI API server, as the OpenAI API does
310 # not support returning only a list of token IDs.
311 detokenize: bool = True
312 """Whether to detokenize the output."""
313 skip_special_tokens: bool = True
314 """Whether to skip special tokens in the output."""
315 spaces_between_special_tokens: bool = True
316 """Whether to add spaces between special tokens in the output."""
317 include_stop_str_in_output: bool = False
318 """Whether to include the stop strings in output text."""
319 output_kind: RequestOutputKind = RequestOutputKind.CUMULATIVE
320 stream_interval: int | None = None
321 """Number of newly generated tokens to batch into each streamed
322 `RequestOutput`. Raises the interval above the engine-level
323 `--stream-interval`. Values below engine setting are clamped up to it.
324 The first and final outputs are always emitted immediately."""
325 skip_clone: bool = False
326 """Internal flag indicating that this SamplingParams instance is safe to
327 reuse without cloning. When True, clone() will return self without
328 performing a deep copy. This should only be set when the params object
329 is guaranteed to be dedicated to a single request and won't be modified
330 in ways that would affect other uses."""
331
332 # The below fields are not supposed to be used as an input.
333 # They are set in post_init.
334 output_text_buffer_length: int = 0
335 _eos_token_id: int | None = None
336 _all_stop_token_ids: set[int] = msgspec.field(default_factory=set)
337
338 # Fields used to construct logits processors
339 structured_outputs: StructuredOutputsParams | None = None
340 """Parameters for configuring structured outputs."""
341 logit_bias: dict[int, float] | None = None
342 """If provided, the engine will construct a logits processor that applies
343 these logit biases."""
344 allowed_token_ids: list[int] | None = None
345 """If provided, the engine will construct a logits processor which only
346 retains scores for the given token ids."""
347 extra_args: dict[str, Any] | None = None
348 """Arbitrary additional args, that can be used by custom sampling
349 implementations, plugins, etc. Not used by any in-tree sampling
350 implementations."""
351 # Fields used for bad words
352 bad_words: list[str] | None = None
353 """Words that are not allowed to be generated. More precisely, only the
354 last token of a corresponding token sequence is not allowed when the next
355 generated token can complete the sequence."""
356 _bad_words_token_ids: list[list[int]] | None = None
357
358 skip_reading_prefix_cache: bool | None = None
359 thinking_token_budget: int | None = None
360 """Maximum number of tokens allowed for thinking operations."""
361
362 repetition_detection: RepetitionDetectionParams | None = None
363 """Parameters for detecting repetitive N-gram patterns in output tokens.
364 If such repetition is detected, generation will be ended early. LLMs can
365 sometimes generate repetitive, unhelpful token patterns, stopping only
366 when they hit the maximum output length (e.g. 'abcdabcdabcd...' or
367 '\\emoji \\emoji \\emoji ...'). This feature can detect such behavior
368 and terminate early, saving time and tokens."""
369
370 # Debugging / RL-specific parameters. Not intended for production serving.
371 routed_experts_prompt_start: int = 0
372 """When enable_return_routed_experts is active, skip the first
373 routed_experts_prompt_start prompt tokens from the returned routing
374 data. In multi-turn agent scenarios, set this to the length of the
375 already-returned prefix to avoid duplicating routing for prompt tokens
376 covered by earlier turns. Default 0 returns routing for all prompt
377 tokens."""
378 trace_decode_token_ids: list[int] | None = None
379 """If provided, forces the engine to emit this predetermined sequence of
380 token IDs during decoding instead of sampling randomly. Real logprobs are
381 still computed. Conflict checking is performed at the engine level."""

487 def __post_init__(self) -> None:
488 if 0 < self.temperature < _MAX_TEMP:
489 logger.warning(
490 "temperature %s is less than %s, which may cause numerical "
491 "errors nan or inf in tensors. We have maxed it out to %s.",
492 self.temperature,
493 _MAX_TEMP,
494 _MAX_TEMP,
495 )
496 self.temperature = max(self.temperature, _MAX_TEMP)
497
498 if self.seed == -1:
499 self.seed = None
500
501 self.thinking_token_budget = validate_thinking_token_budget(
502 self.thinking_token_budget
503 )
504
505 if self.stop is None:
506 self.stop = []
507 elif isinstance(self.stop, str):
508 self.stop = [self.stop]
509
510 if self.stop_token_ids is None:
511 self.stop_token_ids = []
512 else:
513 self.stop_token_ids = list(dict.fromkeys(self.stop_token_ids))
514
515 if self.bad_words is None:
516 self.bad_words = []
517 else:
518 self.bad_words = list(dict.fromkeys(self.bad_words))
519
520 if self.logprobs is True:
521 self.logprobs = 1
522
523 if self.prompt_logprobs is True:
524 self.prompt_logprobs = 1
525
526 # Number of characters to hold back for stop string evaluation
527 # until sequence is finished.
528 if self.stop and not self.include_stop_str_in_output:
529 self.output_text_buffer_length = max(len(s) for s in self.stop) - 1
530
531 self._verify_args()
532
533 if self.temperature < _SAMPLING_EPS:
534 # Zero temperature means greedy sampling.
535 self.top_p = 1.0
536 self.top_k = 0
537 self.min_p = 0.0
538 self._verify_greedy_sampling()
539
540 # eos_token_id is added to this by the engine
541 self._all_stop_token_ids.update(self.stop_token_ids)
542
543 if self.skip_reading_prefix_cache is None:
544 # If prefix caching is enabled,
545 # the output of prompt logprobs may less than n_prompt_tokens,
546 # we need to skip reading cache at this request.
547 self.skip_reading_prefix_cache = self.prompt_logprobs is not None
548
549 def _verify_args(self) -> None:
550 _verify_num_sequences(self.n, "n")
551 if self.extra_args:
552 self._verify_extra_args()
553 if not -2.0 <= self.presence_penalty <= 2.0:
554 raise VLLMValidationError(
555 f"presence_penalty must be in [-2, 2], got {self.presence_penalty}."
556 )
557 if not -2.0 <= self.frequency_penalty <= 2.0:
558 raise VLLMValidationError(
559 f"frequency_penalty must be in [-2, 2], got {self.frequency_penalty}."
560 )
561 if not math.isfinite(self.repetition_penalty):
562 raise VLLMValidationError(
563 "repetition_penalty must be a finite number, "
564 f"got {self.repetition_penalty}."
565 )
566 if self.repetition_penalty <= 0.0:
567 raise VLLMValidationError(
568 "repetition_penalty must be greater than zero, got "
569 f"{self.repetition_penalty}."
570 )
571 if not math.isfinite(self.temperature):
572 raise VLLMValidationError(
573 f"temperature must be a finite number, got {self.temperature}.",
574 parameter="temperature",
575 value=self.temperature,
576 )
577 if self.temperature < 0.0:
578 raise VLLMValidationError(
579 f"temperature must be non-negative, got {self.temperature}.",
580 parameter="temperature",
581 value=self.temperature,
582 )
583 if self.temperature > 2.0:
584 raise VLLMValidationError(
585 f"temperature must be in [0, 2], got {self.temperature}.",
586 parameter="temperature",
587 value=self.temperature,
588 )
589 if not 0.0 < self.top_p <= 1.0:
590 raise VLLMValidationError(
591 f"top_p must be in (0, 1], got {self.top_p}.",
592 parameter="top_p",
593 value=self.top_p,
594 )
595 # quietly accept -1 as disabled, but prefer 0
596 if self.top_k < -1:
597 raise VLLMValidationError(
598 f"top_k must be 0 (disable), or at least 1, got {self.top_k}."
599 )
600 if not isinstance(self.top_k, int):
601 raise VLLMValidationError(
602 f"top_k must be an integer, got {type(self.top_k).__name__}"
603 )
604 if not 0.0 <= self.min_p <= 1.0:
605 raise VLLMValidationError(f"min_p must be in [0, 1], got {self.min_p}.")
606 if self.max_tokens is not None and self.max_tokens < 1:
607 raise VLLMValidationError(
608 f"max_tokens must be at least 1, got {self.max_tokens}.",
609 parameter="max_tokens",
610 value=self.max_tokens,
611 )
612 if self.min_tokens < 0:
613 raise VLLMValidationError(
614 f"min_tokens must be greater than or equal to 0, got {self.min_tokens}."
615 )
616 if self.max_tokens is not None and self.min_tokens > self.max_tokens:
617 raise VLLMValidationError(
618 f"min_tokens must be less than or equal to "
619 f"max_tokens={self.max_tokens}, got {self.min_tokens}."
620 )
621 if self.stream_interval is not None and self.stream_interval < 1:
622 raise VLLMValidationError(
623 f"stream_interval must be at least 1, got {self.stream_interval}.",
624 parameter="stream_interval",
625 value=self.stream_interval,
626 )
627 if self.logprobs is not None and self.logprobs != -1 and self.logprobs < 0:
628 raise VLLMValidationError(
629 f"logprobs must be non-negative or -1, got {self.logprobs}.",
630 parameter="logprobs",
631 value=self.logprobs,
632 )
633 if (
634 self.prompt_logprobs is not None
635 and self.prompt_logprobs != -1
636 and self.prompt_logprobs < 0
637 ):
638 raise VLLMValidationError(
639 f"prompt_logprobs must be non-negative or -1, got "
640 f"{self.prompt_logprobs}.",
641 parameter="prompt_logprobs",
642 value=self.prompt_logprobs,
643 )
644 assert isinstance(self.stop_token_ids, list)
645 if not all(isinstance(st_id, int) for st_id in self.stop_token_ids):
646 raise VLLMValidationError(
647 f"stop_token_ids must contain only integers, got {self.stop_token_ids}."
648 )
649 assert isinstance(self.stop, list)
650 if any(not stop_str for stop_str in self.stop):
651 raise VLLMValidationError("stop cannot contain an empty string.")
652 if self.stop and not self.detokenize:
653 raise VLLMValidationError(
654 "stop strings are only supported when detokenize is True. "
655 "Set detokenize=True to use stop."
656 )
657 assert isinstance(self.bad_words, list)
658 if any(not bad_word for bad_word in self.bad_words):

660 f"bad_words cannot contain an empty string. "
661 f"Got bad_words={self.bad_words}"

664 def _verify_extra_args(self) -> None:
665 # JSON accepts arbitrary integers, but the engine's MessagePack
666 # transport only supports signed/unsigned 64-bit integers.
667 pending: list[Any] = [self.extra_args]
668 visited: set[int] = set()
669 while pending:
670 value = pending.pop()
671 if isinstance(value, int) and not -(2**63) <= value < 2**64:
672 raise VLLMValidationError(
673 "extra_args integers must be between -2**63 and 2**64 - 1.",
674 parameter="extra_args",
675 )
676 if isinstance(value, (dict, list, tuple)) and id(value) not in visited:
677 visited.add(id(value))
678 if isinstance(value, dict):
679 pending.extend(value.keys())
680 pending.extend(value.values())
681 else:
682 pending.extend(value)

685 if self.n > 1:
686 raise VLLMValidationError(

690 def update_from_generation_config(
691 self,
692 generation_config: dict[str, Any],
693 eos_token_id: int | None = None,
694 ) -> None:
695 """Update if there are non-default values from generation_config"""
696 if not self.ignore_eos:
697 self._eos_token_id = eos_token_id
698
699 if eos_token_id is not None:
700 # Add the eos token id into the sampling_params to support
701 # min_tokens processing.
702 self._all_stop_token_ids.add(eos_token_id)
703
704 # Update eos_token_id for generation
705 if (eos_ids := generation_config.get("eos_token_id")) is not None:

710 # stop_token_ids since it's handled separately for stopping

713 if eos_ids:
714 self._all_stop_token_ids.update(eos_ids)
715 if not self.ignore_eos:
716 assert self.stop_token_ids is not None
717 eos_ids.update(self.stop_token_ids)
718 self.stop_token_ids = list(eos_ids)

720 def update_from_tokenizer(self, tokenizer: TokenizerLike) -> None:
721 if not self.bad_words:
722 return
723 self._bad_words_token_ids = []
724 max_num_bad_words = envs.VLLM_MAX_NUM_BAD_WORDS
725 for bad_word in self.bad_words:
726 # To prohibit words both at the beginning
727 # and in the middle of text
728 # (related to add_prefix_space tokenizer parameter)
729 for add_prefix_space in [False, True]:
730 prefix = " " if add_prefix_space else ""
731 prompt = prefix + bad_word.lstrip()
732 prompt_token_ids = tokenizer.encode(
733 text=prompt, add_special_tokens=False
734 )
735
736 if not prompt_token_ids:
737 if not add_prefix_space:
738 raise VLLMValidationError(
739 "bad_words entries must tokenize to at least one token.",
740 parameter="bad_words",
741 value=self.bad_words,
742 )
743 # The unprefixed form is still enforceable when only the
744 # optional space-prefixed form tokenizes to nothing.
745 continue
746
747 # If no space at the beginning
748 # or if prefix space produces a new word token
749 if (not add_prefix_space) or (
750 add_prefix_space
751 and prompt_token_ids[0] != self._bad_words_token_ids[-1][0]
752 and len(prompt_token_ids) == len(self._bad_words_token_ids[-1])
753 ):
754 self._bad_words_token_ids.append(prompt_token_ids)
755 if len(self._bad_words_token_ids) > max_num_bad_words:
756 raise VLLMValidationError(
757 f"Too many bad words after tokenization: "
758 f"{len(self._bad_words_token_ids)}. "
759 f"The max number is {max_num_bad_words}.",
760 parameter="bad_words",
761 value=self.bad_words,
762 )
763
764 invalid_token_ids = [
765 token_id
766 for bad_words_token_ids in self._bad_words_token_ids
767 for token_id in bad_words_token_ids
768 if token_id < 0 or token_id > tokenizer.max_token_id
769 ]
770 if len(invalid_token_ids) > 0:
771 raise VLLMValidationError(
772 f"The model vocabulary size is {tokenizer.max_token_id + 1},"
773 f" but the following tokens"
774 f" were specified as bad: {invalid_token_ids}."
775 f" All token id values should be integers satisfying:"
776 f" 0 <= token_id <= {tokenizer.max_token_id}.",
777 parameter="bad_words",
778 value=self.bad_words,
779 )

838 def _validate_logprobs(self, model_config: ModelConfig) -> None:
839 max_logprobs = model_config.max_logprobs
840 if max_logprobs == -1:
841 max_logprobs = model_config.get_vocab_size()
842
843 # Validate sample logprobs.
844 if num_logprobs := self.logprobs:
845 if num_logprobs == -1:
846 num_logprobs = model_config.get_vocab_size()
847 if num_logprobs > max_logprobs:
848 raise VLLMValidationError(
849 f"Requested sample logprobs of {num_logprobs}, "
850 f"which is greater than max allowed: {max_logprobs}",
851 parameter="logprobs",
852 value=num_logprobs,
853 )
854
855 # Validate logprob_token_ids.
856 if self.logprob_token_ids is not None:
857 n = len(self.logprob_token_ids)
858 if n > MAX_LOGPROB_TOKEN_IDS:
859 raise VLLMValidationError(
860 f"Requested logprob_token_ids of length {n}, "
861 f"which is greater than max allowed: {MAX_LOGPROB_TOKEN_IDS}",
862 parameter="logprob_token_ids",
863 value=n,
864 )
865 vocab_size = model_config.get_vocab_size()
866 invalid_token_ids = [
867 token_id
868 for token_id in self.logprob_token_ids
869 if token_id < 0 or token_id >= vocab_size
870 ]
871 if invalid_token_ids:
872 raise VLLMValidationError(
873 f"token_id(s) {invalid_token_ids} in logprob_token_ids "
874 f"contain out-of-vocab token ids. Vocabulary size: "
875 f"{vocab_size}",
876 parameter="logprob_token_ids",
877 value=invalid_token_ids,
878 )
879 if self.logprobs is not None and self.logprobs != n:
880 raise VLLMValidationError(
881 f"When both logprobs and logprob_token_ids are set, "
882 f"logprobs must equal len(logprob_token_ids). Got "
883 f"logprobs={self.logprobs}, len(logprob_token_ids)={n}.",
884 parameter="logprob_token_ids",
885 value=n,
886 )
887
888 # Validate prompt logprobs.
889 if num_prompt_logprobs := self.prompt_logprobs:
890 if num_prompt_logprobs == -1:
891 num_prompt_logprobs = model_config.get_vocab_size()
892 if num_prompt_logprobs > max_logprobs:
893 raise VLLMValidationError(
894 f"Requested prompt logprobs of {num_prompt_logprobs}, "
895 f"which is greater than max allowed: {max_logprobs}",
896 parameter="prompt_logprobs",
897 value=num_prompt_logprobs,
898 )

1074 def _validate_structured_outputs(
1075 self,
1076 model_config: ModelConfig,
1077 structured_outputs_config: StructuredOutputsConfig | None,
1078 tokenizer: TokenizerLike | None,
1079 ) -> None:
1080 if structured_outputs_config is None or self.structured_outputs is None:
1081 return
1082
1083 if model_config.is_diffusion:
1084 # Diffusion LLMs denoise a whole canvas of tokens in parallel
1085 # rather than sampling left-to-right, which the grammar FSM
1086 # requires. Without this check, requests fail mid-generation
1087 # with an FSM rejection (HTTP 500). See issue #45436.
1088 raise VLLMValidationError(
1089 "Structured outputs are not yet supported for diffusion "
1090 "language models. Remove the structured output constraint "
1091 "(e.g. `response_format`, `structured_outputs`) from the "
1092 "request."
1093 )
1094
1095 if tokenizer is None:
1096 raise VLLMValidationError(
1097 "Structured outputs requires a tokenizer so it can't be used with 'skip_tokenizer_init'" # noqa: E501
1098 )
1099
1100 backend = structured_outputs_config.backend
1101 if _backend := self.structured_outputs._backend:
1102 # Request-level backend selection is not supported.
1103 # The values may differ if `params` is reused and was set
1104 # to a specific backend based on `auto` behavior in a previous
1105 # request. We remember that it was set as a result of `auto`
1106 # using the `_backend_was_auto` field set in the params.
1107 if backend != _backend and not (
1108 backend == "auto" and self.structured_outputs._backend_was_auto
1109 ):
1110 raise VLLMValidationError(
1111 "Request-level structured output backend selection is not "
1112 f"supported. The request specified '{_backend}', but vLLM "
1113 f"was initialised with '{backend}'. This error can be "
1114 "resolved by removing '_backend' from the request."
1115 )
1116 else:
1117 self.structured_outputs._backend = backend
1118
1119 # Request content validation
1120 if (
1121 isinstance(self.structured_outputs.choice, list)
1122 and not self.structured_outputs.choice
1123 ):
1124 # It is invalid for choice to be an empty list
1125 raise VLLMValidationError(
1126 f"Choice '{self.structured_outputs.choice}' cannot be an empty list" # noqa: E501
1127 )
1128 # Reject empty string grammar early to avoid engine-side crashes
1129 if (
1130 isinstance(self.structured_outputs.grammar, str)
1131 and self.structured_outputs.grammar.strip() == ""
1132 ):
1133 raise VLLMValidationError(
1134 "structured_outputs.grammar cannot be an empty string"
1135 )
1136 # Reject empty string json schema early to avoid engine-side crashes
1137 if (
1138 isinstance(self.structured_outputs.json, str)
1139 and self.structured_outputs.json.strip() == ""
1140 ):
1141 raise VLLMValidationError(
1142 "structured_outputs.json cannot be an empty string"
1143 )
1144 # Reject json_object=False early to avoid engine-side crashes
1145 if self.structured_outputs.json_object is False:
1146 raise VLLMValidationError(
1147 "structured_outputs.json_object must be True if set; omit "
1148 "structured_outputs to disable structured outputs"
1149 )
1150 # Reject a regex containing a NUL byte early, in every backend mode. A
1151 # NUL is never meaningful in a regex pattern and is not handled by the
1152 # regex-to-grammar conversion. Checked here, before backend selection,
1153 # so it is a clean 400 rather than a silent fallback in the default
1154 # "auto" mode.
1155 if self.structured_outputs.regex and "\x00" in self.structured_outputs.regex:
1156 raise VLLMValidationError(
1157 "structured_outputs.regex must not contain a NUL character ('\\x00')"
1158 )
1159
1160 from vllm.v1.structured_output.backend_guidance import (
1161 has_guidance_unsupported_json_features,
1162 validate_guidance_grammar,
1163 )
1164 from vllm.v1.structured_output.backend_lm_format_enforcer import (
1165 validate_structured_output_request_lm_format_enforcer,
1166 )
1167 from vllm.v1.structured_output.backend_outlines import (
1168 validate_structured_output_request_outlines,
1169 )
1170 from vllm.v1.structured_output.backend_xgrammar import validate_xgrammar_grammar
1171
1172 if backend.startswith("xgrammar"):
1173 # xgrammar with no fallback
1174 validate_xgrammar_grammar(self)
1175 elif backend.startswith("guidance"):
1176 if _is_non_tekken_mistral(tokenizer=tokenizer):
1177 raise VLLMValidationError(
1178 "Non-tekken Mistral tokenizers are not supported for the 'guidance'"
1179 " structured output backend. Please either use a more recent "
1180 "Mistral model, the ['xgrammar', 'outlines'] "
1181 "backends or tokenizer_mode='hf' instead."
1182 )
1183 # TODO: ideally we would have the LLTokenizer here as Lark syntax
1184 # allows <|special_token|> and similar, see
1185 # https://github.com/guidance-ai/llguidance/blob/main/docs/syntax.md#special-tokens
1186 # Without tokenizer these are disallowed in grammars.
1187 validate_guidance_grammar(
1188 self,
1189 tokenizer=_get_llg_tokenizer(tokenizer),
1190 )
1191 elif backend == "outlines":
1192 # outlines backend
1193 validate_structured_output_request_outlines(self)
1194 elif backend == "lm-format-enforcer":
1195 # lm format enforcer backend
1196 if is_mistral_tokenizer(tokenizer):
1197 raise VLLMValidationError(
1198 "Mistral tokenizer is not supported for the 'lm-format-enforcer' "
1199 "structured output backend. Please use ['xgrammar', 'outlines'] "
1200 "backends or tokenizer_mode='hf' instead."
1201 )
1202 validate_structured_output_request_lm_format_enforcer(self)
1203 else:
1204 # NOTE: backend must be "auto" here, because we have
1205 # checked supported_backends above.
1206 # In this mode, we set opinionated defaults based on what we think
1207 # will satisfy the most use cases without having to worry about
1208 # this setting. We include fallback behavior here, but not with any
1209 # other setting where a specific backend was specified.
1210 try:
1211 validate_xgrammar_grammar(self)
1212 self.structured_outputs._backend = "xgrammar"
1213 except VLLMValidationError:
1214 # The request either failed validation
1215 # or includes some jsonschema feature(s) that
1216 # are not supported in xgrammar.
1217
1218 skip_guidance = _is_non_tekken_mistral(tokenizer)
1219
1220 # Check if schema has features unsupported by guidance
1221 so_params = self.structured_outputs
1222 if not skip_guidance and so_params.json:
1223 if isinstance(so_params.json, str):
1224 try:
1225 schema = json_mod.loads(so_params.json)
1226 except json_mod.JSONDecodeError as e:
1227 raise VLLMValidationError(
1228 "Invalid JSON grammar specification."
1229 ) from e
1230 else:
1231 schema = so_params.json
1232 skip_guidance = has_guidance_unsupported_json_features(schema)
1233
1234 if skip_guidance:
1235 # Fall back to outlines if the tokenizer is non-tekken Mistral or
1236 # the schema contains features unsupported by guidance
1237 validate_structured_output_request_outlines(self)
1238 self.structured_outputs._backend = "outlines"
1239 else:
1240 # Fall back to guidance by default.
1241 validate_guidance_grammar(
1242 self,
1243 tokenizer=_get_llg_tokenizer(tokenizer),
1244 )
1245 self.structured_outputs._backend = "guidance"
1246 # Remember that this backend was set automatically
1247 self.structured_outputs._backend_was_auto = True
1248
1249 # Run post-init validation. This is also important to ensure subsequent
1250 # roundtrip serialization/deserialization won't fail.
1251 self.structured_outputs.__post_init__()
1252
