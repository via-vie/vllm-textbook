74 def __init__(
75 self,
76 model: SupportsLoRAModel,
77 max_num_seqs: int,

79 vocab_size: int,

84 """Create a LoRAModelManager and adapter for a given model.
85
86 Args:
87 model: the model to be adapted.
88 max_num_seqs: the maximum number of sequences model can run in a

99 )
100
101 self.adapter_type = "LoRA"
102 self.lora_config = lora_config
103 self.device = device
104 self.max_num_seqs = max_num_seqs
105 assert self.capacity >= self.lora_slots
106 self._registered_adapters: AdapterLRUCache[LoRAModel] = AdapterLRUCache(
107 self.capacity, self.deactivate_adapter
108 )
109 self._active_adapters: AdapterLRUCache[None] = AdapterLRUCache(
110 self.lora_slots, self._deactivate_adapter
111 )
112 self.max_num_batched_tokens = math.ceil(max_num_batched_tokens / 8) * 8
113 self.lora_index_to_id: list[int | None] = [None] * self.lora_slots
114 self.vocab_size = vocab_size
115
116 self.is_pooling_model = is_pooling_model(self.model)
117 self.packed_modules: dict[str, list[str]] = {}
118 self.modules: dict[str, BaseLayerWithLoRA] = {}
119 self._last_mapping: LoRAMapping | None = None
120 self._last_slot_layout: tuple[int | None, ...] | None = None

126 # the model's 3D flag, so 2D and 3D adapters can coexist.

179 def _maybe_init_mm(
180 self,
181 vllm_config: VllmConfig,
182 max_num_batched_tokens: int,
183 ) -> None:
184 mm_registry = MULTIMODAL_REGISTRY
185
186 self.supports_tower_connector_lora = False
187 self.mm_mapping: MultiModelKeys = self.model.get_mm_mapping()
188
189 # Only one language model can be included in the model.
190 assert len(self.mm_mapping.language_model) == 1
191
192 # Language model punica wrapper
193 llm_punica_wrapper = get_punica_wrapper(
194 max_num_batched_tokens,
195 max_batches=self.max_num_seqs,
196 device=self.device,
197 lora_config=self.lora_config,
198 )
199
200 lm_prefix = self.mm_mapping.language_model[0]
201 self.punica_wrapper_mapping[lm_prefix] = llm_punica_wrapper
202
203 # First, determine if the model supports tower connector LoRA.
204 self.supports_tower_connector_lora = (
205 self.supports_mm and self.model.supports_tower_connector_lora
206 )
207
208 # Then, handle the case where the feature is disabled in the config.
209 if not self.lora_config.enable_tower_connector_lora:
210 if self.supports_tower_connector_lora:
211 logger.info(
212 "%s supports adding LoRA to the tower modules. If needed, "
213 "please set `enable_tower_connector_lora=True`.",
214 self.model.__class__.__name__,
215 )
216 self.supports_tower_connector_lora = False
217 return
218
219 # After this point, the feature is enabled in the config.
220 # Now check if it's supported by the model.
221 if not self.supports_tower_connector_lora:
222 # Enabled but not supported: log warning and return.
223 logger.warning(
224 "LoRA with tower connector is enabled, but the model %s "
225 "does not support it. This will be ignored.",
226 self.model.__class__.__name__,
227 )
228 return
229
230 # Check if initialize the language model only.
231 if (
232 vllm_config.model_config.multimodal_config
233 and vllm_config.model_config.multimodal_config.language_model_only
234 ):
235 logger.warning(
236 "Disabling `enable_tower_connector_lora` because the multimodal "
237 "model is configured to initialize the language model only."
238 )
239 self.supports_tower_connector_lora = False
240 return
241
242 logger.warning(
243 "LoRA for the tower and connector of multimodal models is "
244 "experimental and may contain bugs. Please report any related issues on "
245 "GitHub if you encounter them."
246 )
247
248 mm_budget = MultiModalBudget(vllm_config, mm_registry)
249 limit_per_prompt = max(mm_budget.mm_max_items_per_prompt.values())
250 max_lora_tokens = mm_budget.get_encoder_budget()
251 lora_token_counts_by_modality = [
252 self.model.get_mm_lora_token_counts(
253 modality=modality,
254 mm_kwargs=None,
255 num_mm_embeds=max_lora_tokens,
256 )
257 for modality in mm_budget.mm_max_toks_per_item
258 ]
259 num_encoder_tokens = max(
260 tower_tokens for tower_tokens, _ in lora_token_counts_by_modality
261 )
262
263 # Tower wrappers
264 tower_punica_wrapper = get_punica_wrapper(
265 num_encoder_tokens,
266 max_batches=self.max_num_seqs * limit_per_prompt,
267 device=self.device,
268 lora_config=self.lora_config,
269 )
270 for prefix in self.mm_mapping.tower_model:
271 self.punica_wrapper_mapping[prefix] = tower_punica_wrapper
272
273 # Use wrapper for connector if present.
274 if self.mm_mapping.connector:
275 connector_tokens = max(
276 (
277 connector_tokens
278 for _, connector_tokens in lora_token_counts_by_modality
279 if connector_tokens is not None
280 ),
281 default=None,
282 )
283 if connector_tokens is not None:
284 connector_punica_wrapper = get_punica_wrapper(
285 connector_tokens,
286 max_batches=self.max_num_seqs * limit_per_prompt,
287 device=self.device,
288 lora_config=self.lora_config,
289 )
290 for prefix in self.mm_mapping.connector:
291 self.punica_wrapper_mapping[prefix] = connector_punica_wrapper
292 else:
293 logger.warning_once(
294 "Connector LoRA support disabled: model does not implement "
295 "get_num_mm_connector_tokens(). This method is required to "
296 "determine the connector's token budget for LoRA operations."
297 )

315 def activate_adapter(
316 self,
317 lora_id: int,
318 ) -> bool:
319 """Move LoRA into a GPU buffer to be used in the forward pass."""
320 if lora_id in self._active_adapters:
321 return False
322 first_free_slot = next(
323 (
324 (i, lora_id)
325 for i, lora_id in enumerate(self.lora_index_to_id)
326 if lora_id is None
327 ),
328 None,
329 )
330 if first_free_slot is None:
331 raise ValueError("No free lora slots")
332 index, _ = first_free_slot
333 self._active_adapters[lora_id] = None
334 lora_model = self._registered_adapters[lora_id]
335 logger.debug(
336 "Activating LoRA. int id: %d, slot index: %d", lora_model.id, index
337 )
338 self.lora_index_to_id[index] = lora_model.id
339 num_lora_weights_applied = 0
340 for module_name, module in self.modules.items():
341 module_lora = self._get_lora_layer_weights(lora_model, module_name)
342 if not module_lora:
343 module.reset_lora(index)
344 logger.debug(
345 "No LoRA weights found for module %s, skipping.", module_name
346 )
347 continue
348
349 module.set_lora(
350 index,
351 module_lora.lora_a,
352 module_lora.lora_b,
353 )
354 num_lora_weights_applied += 1
355 logger.debug("Successfully loaded LoRA weights for module %s.", module_name)
356 if num_lora_weights_applied == 0:
357 logger.debug_once(
358 "No LoRA weights were applied for adapter %s on this worker. "
359 "Requests may use the base model; check --lora-target-modules. "
360 "This may be expected with pipeline or expert parallelism.",
361 lora_model.id,
362 )
363 return True

370 pass

416 def _create_lora_modules(self):
417 def _parent_module(module_name: str) -> str:
418 # module name is a dot separated name.
419 # for example:
420 # - given an input 'x.y.z' return 'x.y'
421 # - given an input 'x' return ''
422 return module_name.rpartition(".")[0]
423
424 wrapped_by_id: dict[int, BaseLayerWithLoRA] = {}
425
426 for module_name, module in self.model.named_modules(remove_duplicate=False):
427 if isinstance(module, PPMissingLayer):
428 continue
429
430 if self.lora_config.target_modules is None:
431 if not is_supported_lora_module(
432 module_name,
433 module,
434 self.supported_lora_modules,
435 ):
436 continue
437 elif not self._match_target_modules(module_name):
438 continue
439
440 punica_wrapper = self._get_punica_wrapper(module_name)
441 if punica_wrapper is None:
442 logger.warning(
443 "Regarding %s, no matching PunicaWrapper "
444 "is found; %s will be ignored.",
445 self.model.__class__.__name__,
446 module_name,
447 )
448 continue
449
450 # TODO: Remove this restriction
451 # peft error when generating LoRA adapter with "gate" module:
452 # "Target module NemotronHTopkRouter() is not supported."
453 # Working LoRA adapter was created using peft with:
454 # LoraConfig(target_modules="all-linear", ...)
455 if self._is_non_gated_moe and module_name.endswith("mixer.gate"):
456 logger.debug_once(
457 "LoRA is not supported for non-gated MoE gate module."
458 " %s will be ignored.",
459 module_name,
460 )
461 continue
462
463 existing_wrapper = wrapped_by_id.get(id(module))
464 if existing_wrapper is not None and "lm_head" not in module_name:
465 # Same underlying module was already wrapped under another
466 # path (e.g. a MoE gate held both directly on the block and
467 # inside the MoE runner). The adapter targets the canonical
468 # path (`mlp.gate`); rewire the alias attribute
469 # (`runner.gate`) to the SAME wrapper so the forward path
470 # through the alias still applies LoRA, but do NOT add a
471 # second entry to self.modules — otherwise `activate_adapter`
472 # would call `reset_lora` on the alias and wipe the weights
473 # just set under the canonical name, because the alias can't
474 # load LoRA weights due to name mismatch.
475 parent = self.model.get_submodule(_parent_module(module_name))
476 # reference
477 setattr(parent, module_name.rpartition(".")[-1], existing_wrapper)
478 continue
479
480 parts = module_name.split(".")[-1]
481 packed_moduled_lst = self.packed_modules_mapping.get(parts, [])
482 if isinstance(module, MoERunner):
483 # packed_moduled_lst is used here to just determine whether to
484 # instantiate FusedMoE3DWithLoRA or FusedMoEWithLoRA, and the
485 # difference between these two LoRA layers is whether the
486 # LoRA weights of w1 and w3 have already been fused on disk.
487
488 packed_moduled_lst = ["w13"] if self._is_3d_moe_model else ["w1", "w3"]
489 new_module = replace_submodule(
490 self.model,
491 module_name,
492 from_layer(
493 module,
494 self.lora_slots,
495 self.lora_config,
496 packed_moduled_lst,
497 self.model.config,
498 ),
499 )
500 if isinstance(new_module, BaseLayerWithLoRA):
501 wrapped_by_id[id(module)] = new_module
502 wrapped_by_id[id(new_module)] = new_module
503
504 # (yard1): TODO make this more robust
505 if "lm_head" in module_name:
506 logits_processor_module_name = "logits_processor"
507 parent_module = _parent_module(module_name)
508 if parent_module:
509 logits_processor_module_name = (
510 f"{parent_module}.{logits_processor_module_name}"
511 )
512
513 logits_processor_module = self.model.get_submodule(
514 logits_processor_module_name
515 )
516
517 new_module = replace_submodule(
518 self.model,
519 logits_processor_module_name,
520 from_layer_logits_processor(
521 logits_processor_module,
522 module,
523 self.lora_slots,
524 self.lora_config,
525 self.model.config,
526 ),
527 )
528
529 # Some matched modules can be unsupported by LoRA wrappers
530 # (e.g. subclasses with specialized forward behavior).
531 if not isinstance(new_module, BaseLayerWithLoRA):
532 error_msg = (
533 "LoRA target module "
534 f"{module_name} ({type(module).__name__}) matched the "
535 "deployment configuration but could not be wrapped by any "
536 "LoRA layer implementation."
537 )
538 if self.lora_config.target_modules is not None:
539 raise ValueError(
540 f"{error_msg} target_modules="
541 f"{sorted(self.lora_config.target_modules)}"
542 )
543 logger.warning_once("%s It will be ignored.", error_msg)
544 continue
545 self.register_module(module_name, new_module)
546
547 self._register_packed_modules(module_name)
548 # All lora layers share the same punica_wrapper based on reference.
549 new_module.set_mapping(punica_wrapper)

558 @staticmethod
559 def _pad_lora_pairs_to_triplets(
560 loras: list[LoRALayerWeights | None],
561 ) -> list[LoRALayerWeights | None]:
562 """Pad LoRA weight pairs to triplets for non-gated MoE.
563
564 For non-gated MoE, each expert has 2 entries (w1, w2) that need to be
565 padded to triplets (w1, w2, None) to match pack_moe expectations.
566 """
567 assert len(loras) % 2 == 0, "Expected pairs of LoRA weights for non-gated MoE."
568 padded: list[LoRALayerWeights | None] = []
569 for i in range(0, len(loras), 2):
570 padded.extend(loras[i : i + 2])
571 padded.append(None)
572 return padded

574 def create_dummy_lora(
575 self,
576 lora_id: int,
577 rank: int,
578 embedding_modules: dict[str, str] | None = None,
579 ) -> LoRAModel:
580 """Create zero-initialized LoRAModel for warmup."""
581 model = LoRAModel(lora_id, rank, {})
582 for module_name, module in self.model.named_modules():
583 if (
584 not is_supported_lora_module(
585 module_name,
586 module,
587 self.supported_lora_modules,
588 )
589 or not self._match_target_modules(module_name)
590 or not isinstance(module, BaseLayerWithLoRA)
591 or self._get_punica_wrapper(module_name) is None
592 ):
593 continue
594 parts = module_name.split(".")
595 if module_name not in self.packed_modules:
596 assert embedding_modules is not None
597 if parts[-1] in embedding_modules:
598 # Special-case lm_head: wrapped by LogitsProcessorWithLoRA.
599 # LoRA input dim is hidden_size, output dim is vocab size.
600 # LogitsProcessorWithLoRA handles extra vocab size directly.
601 if parts[-1] == "lm_head":
602 input_dim = module.lora_a_stacked[0].shape[-1]
603 output_dim = module.lora_b_stacked[0].shape[-2]
604 else:
605 input_dim = (
606 module.base_layer.org_vocab_size
607 if hasattr(module.base_layer, "org_vocab_size")
608 else module.base_layer.weight.shape[1]
609 )
610 output_dim = (
611 module.base_layer.embedding_dim
612 if hasattr(module.base_layer, "embedding_dim")
613 else module.base_layer.weight.shape[0]
614 )
615 lora = LoRALayerWeights.create_dummy_lora_weights(
616 module_name,
617 input_dim,
618 output_dim,
619 rank,
620 module.lora_a_stacked[0].dtype,
621 "cpu",
622 )
623 model.loras[module_name] = lora
624 elif module.__class__.__name__ == "FusedMoE3DWithLoRA":
625 # Case for 3D moe model
626 # w2
627 lora = LoRALayerWeights.create_dummy_lora_weights(
628 module_name,
629 module.w2_input_size,
630 module.w2_output_size,
631 rank * module.w2_lora_a_stacked[0].shape[1], # rank*num_experts
632 module.w2_lora_a_stacked[0].dtype,
633 "cpu",
634 )
635 model.loras[module_name] = lora
636 # w13
637 lora = LoRALayerWeights.create_dummy_lora_weights(
638 module_name,
639 module.w13_input_size,
640 module.w13_output_size,
641 rank
642 * module.w13_lora_a_stacked[0].shape[1], # rank*num_experts
643 module.w13_lora_a_stacked[0].dtype,
644 "cpu",
645 )
646 model.loras[module_name + ".base_layer"] = lora
647 else:
648 lora = LoRALayerWeights.create_dummy_lora_weights(
649 module_name,
650 module.lora_a_stacked[0].shape[-1],
651 module.lora_b_stacked[0].shape[-2],
652 rank,
653 module.lora_a_stacked[0].dtype,

655 )

657 else:
658 parts = module_name.split(".")

662 replacements = replacements[
663 : len(module.lora_a_stacked) // self.lora_slots

672 module_name + "." + r,
673 module.lora_a_stacked[i].shape[-1],
674 module.lora_b_stacked[i].shape[-2],
675 rank,
676 module.lora_a_stacked[i].dtype,
677 "cpu",
678 )

680 if module.__class__.__name__ == "FusedMoEWithLoRA":
681 # For non-gated MoE, pad subloras to 3 elements per expert
682 # to match pack_moe expectations (w1, w2, None for w3)
683 if self._is_non_gated_moe and len(subloras) > 0:
684 subloras = self._pad_lora_pairs_to_triplets(subloras)
685 lora = PackedLoRALayerWeights.pack_moe(
686 subloras, module_name, is_non_gated_moe=self._is_non_gated_moe
687 )
688 else:
689 lora = PackedLoRALayerWeights.pack(subloras)
690 model.loras[module_name] = lora
691 return model

693 def get_dummy_lora_warmup_rank(self, default_rank: int) -> int:
694 """Return a dummy LoRA rank compatible with wrapped modules.
695
696 Dummy LoRAs keep warmup memory low by using a small rank. Fully
697 sharded MoE wrappers additionally require the dummy rank to be divisible
698 by tensor parallel size because they shard W13 along the rank axis.
699 """
700 if not self.lora_config.fully_sharded_loras:
701 return default_rank
702
703 required_multiple = 1
704 for module in self.modules.values():
705 if not getattr(module, "fully_sharded", False):
706 continue
707 required_multiple = math.lcm(required_multiple, module.tp_size)
708
709 if required_multiple == 1 or default_rank % required_multiple == 0:
710 return default_rank
711
712 adjusted_rank = (
713 (default_rank + required_multiple - 1) // required_multiple
714 ) * required_multiple
715 if adjusted_rank > self.lora_config.max_lora_rank:
716 raise ValueError(
717 "Unable to choose a dummy LoRA warmup rank compatible with "
718 "fully sharded MoE modules: "
719 f"default_rank={default_rank}, "
720 f"required_multiple={required_multiple}, "
721 f"max_lora_rank={self.lora_config.max_lora_rank}"
722 )
723 return adjusted_rank

771 def _create_merged_loras_inplace(self, lora_model: LoRAModel) -> None:
772 for module_name, new_module_names in self.packed_modules.items():
773 # For 2D RoutedExperts modules with EP, narrow the per-expert
774 # sub-module list to this rank's owned experts so pack_moe
775 # produces a tensor sized to local_num_experts directly.
776 packed_module_names = new_module_names
777 if module_name.endswith(".experts"):
778 new_module_names = self._restrict_to_local_experts(
779 module_name, new_module_names
780 )
781 replacement_loras: list[LoRALayerWeights | None] = []
782 has_replacement = False
783 for r in new_module_names:
784 lora = self._get_lora_layer_weights(lora_model, r)
785 replacement_loras.append(lora)
786 if lora:
787 has_replacement = True
788 if not has_replacement:
789 continue
790 for i in range(len(replacement_loras)):
791 if replacement_loras[i]:
792 continue
793 replacement_loras[i] = None
794 # HACK Temporary solution for the pool model.
795 if self.is_pooling_model and not lora_model.check_lora_name(module_name):
796 replaced_module_name = module_name.removeprefix("model.")
797 if lora_model.check_lora_name(replaced_module_name):
798 module_name = replaced_module_name
799 if module_name.endswith(".experts"):
800 if self._enable_moe_shared_loras:
801 lora_model.loras[module_name] = (
802 PackedLoRALayerWeights.pack_moe_stacked(
803 replacement_loras,
804 module_name,
805 )
806 )
807 for module in packed_module_names:
808 lora_model.loras.pop(module, None)
809 continue
810 if self._is_non_gated_moe and len(replacement_loras) > 0:
811 replacement_loras = self._pad_lora_pairs_to_triplets(
812 replacement_loras
813 )
814 lora_model.loras[module_name] = PackedLoRALayerWeights.pack_moe(
815 replacement_loras,
816 module_name,
817 is_non_gated_moe=self._is_non_gated_moe,
818 )
819 else:
820 lora_model.loras[module_name] = PackedLoRALayerWeights.pack(
821 replacement_loras
822 )
823 # Drop every candidate sub-module, including non-local expert
824 # entries that were loaded but did not contribute to the
825 # packed result. Without this they would keep extra CPU
826 # memory alive after pack_moe.
827 for module in packed_module_names:
828 lora_model.loras.pop(module, None)
829
830 for lora in lora_model.loras.values():
831 lora.optimize()
832
833 for module_name, module in self.modules.items():
834 if isinstance(module, FusedMoE3DWithLoRA):
835 self._stack_moe_lora_weights(lora_model, module, module_name)
836 elif isinstance(module, FusedMoEWithLoRA):
837 # When mixed mode is enabled the universal 2D wrapper has to
838 # absorb both 2D and 3D-format adapters. 3D-format adapters
839 # need to be split into per-(w1, w2, w3) tensors before the
840 # 2D set_lora can copy them into the stacked buffers.
841 if self._enable_mixed_moe_lora_format and getattr(
842 lora_model, "is_3d_lora_weight", False
843 ):
844 self._convert_3d_to_2d_moe_lora(lora_model, module, module_name)
845 else:
846 self._slice_moe_lora_ep(lora_model, module, module_name)
847
848 first_lora: LoRALayerWeights = next(iter(lora_model.loras.values()))
849 assert first_lora.lora_a is not None
850 if isinstance(first_lora.lora_a, list):
851 lora_device = next(iter(first_lora.lora_a))
852 else:
853 lora_device = first_lora.lora_a.device
854 # Execute pin_memory after LoRA weight merging, mainly because:
855 # 1. Some MoE models have a large number of LoRA weights. If we
856 # perform # pin_memory immediately after loading weights, the
857 # overhead is significant.
858 # 2. The weight packing above (e.g., pack_moe) may invalidate the
859 # pin_memory allocation, so we execute it after packing.
860
861 pin_memory = str(lora_device) == "cpu" and PIN_MEMORY
862 if pin_memory:
863 for lora in lora_model.loras.values():
864 if isinstance(lora.lora_a, list):
865 for index in range(len(lora.lora_a)):
866 if lora.lora_a[index] is None:
867 continue
868 lora.lora_a[index] = lora.lora_a[index].pin_memory()
869 lora.lora_b[index] = lora.lora_b[index].pin_memory()
870 else:
871 lora.lora_a = lora.lora_a.pin_memory()
872 lora.lora_b = lora.lora_b.pin_memory()

874 def _stack_moe_lora_weights(
875 self, lora_model: LoRAModel, module: FusedMoE3DWithLoRA, module_name: str
876 ):
877 module_lora = self._get_lora_layer_weights(lora_model, module_name)
878
879 # Note (gnovack) - If MOE lora weights are not split into
880 # num_experts chunks, we split them here
881 if module_lora and torch.is_tensor(module_lora.lora_a):
882 # Handle PEFT file format where experts.base_layer is the
883 # gate_up_proj and experts is the down_proj
884 gate_up_proj_lora = self._get_lora_layer_weights(
885 lora_model, module_name + ".base_layer"
886 )
887 down_proj_lora = module_lora
888 # FIXME Edge case where LoRA is not added to gate_up_proj
889 # or down_proj
890 assert gate_up_proj_lora is not None
891 assert down_proj_lora is not None
892 if self._is_3d_moe_model:
893 local_num_experts = module.w13_lora_a_stacked[0].shape[1]
894 # The checkpoint holds weights for all global experts, but
895 # each EP rank owns only local_num_experts. Reshape against
896 # the adapter's actual expert count, then slice this rank's
897 # owned expert range before it gets copied into the local
898 # stacked buffer. For non-EP (local == global) this is a
899 # no-op slice.
900 global_num_experts = module.global_num_experts
901 ep_rank = module.ep_rank
902 expert_start = ep_rank * local_num_experts
903 expert_end = expert_start + local_num_experts
904
905 # (num_experts,rank,input_size)
906 gate_up_proj_lora.lora_a = gate_up_proj_lora.lora_a.reshape(
907 global_num_experts, -1, gate_up_proj_lora.lora_a.shape[-1]
908 )[expert_start:expert_end].contiguous()
909 down_proj_lora.lora_a = down_proj_lora.lora_a.reshape(
910 global_num_experts, -1, down_proj_lora.lora_a.shape[-1]
911 )[expert_start:expert_end].contiguous()
912
913 # (output_size,rank,num_experts)
914 gate_up_proj_lora.lora_b = gate_up_proj_lora.lora_b.reshape(
915 gate_up_proj_lora.lora_b.shape[0], -1, global_num_experts
916 )[..., expert_start:expert_end]
917 down_proj_lora.lora_b = down_proj_lora.lora_b.reshape(
918 down_proj_lora.lora_b.shape[0], -1, global_num_experts
919 )[..., expert_start:expert_end]
920
921 # (num_experts,output_size,rank)
922 gate_up_proj_lora.lora_b = gate_up_proj_lora.lora_b.permute(
923 2, 0, 1
924 ).contiguous()
925 down_proj_lora.lora_b = down_proj_lora.lora_b.permute(
926 2, 0, 1
927 ).contiguous()
928
929 module_lora.lora_a = [
930 gate_up_proj_lora.lora_a,
931 down_proj_lora.lora_a,
932 ]
933 module_lora.lora_b = [
934 gate_up_proj_lora.lora_b,
935 down_proj_lora.lora_b,
936 ]
937 else:
938 # Some 3D MoE models haven't added the `is_3d_moe_weight`
939 # attribute yet, so fallback here
940 num_experts = module_lora.lora_a.shape[0] // module_lora.rank
941
942 gate_proj_a = gate_up_proj_lora.lora_a.chunk(num_experts, dim=0)
943 up_proj_a = gate_up_proj_lora.lora_a.chunk(num_experts, dim=0)
944
945 gate_proj_b = gate_up_proj_lora.lora_b[::2, ...].chunk(
946 num_experts, dim=-1
947 )
948 up_proj_b = gate_up_proj_lora.lora_b[1::2, ...].chunk(
949 num_experts, dim=-1
950 )
951
952 down_proj_a = down_proj_lora.lora_a.chunk(num_experts, dim=0)
953 down_proj_b = down_proj_lora.lora_b.chunk(num_experts, dim=-1)
954
955 lora_a = []
956 lora_b = []
957 for i in range(num_experts):
958 lora_a.append(gate_proj_a[i])
959 lora_a.append(down_proj_a[i])
960 lora_a.append(up_proj_a[i])
961
962 lora_b.append(gate_proj_b[i])
963 lora_b.append(down_proj_b[i])
964 lora_b.append(up_proj_b[i])
965
966 module_lora.lora_a = lora_a
967 module_lora.lora_b = lora_b
968
969 def _convert_3d_to_2d_moe_lora(
970 self,
971 lora_model: LoRAModel,
972 module: FusedMoEWithLoRA,
973 module_name: str,
974 ) -> None:
975 """Convert a 3D-format MoE LoRA checkpoint into the 2D pack layout
976 that `FusedMoEWithLoRA.set_lora` expects.
977
978 On disk the 3D PEFT layout stores two flat tensor pairs per layer:
979 - `{module}.base_layer.lora_{A,B}`: gate_up_proj, with shapes
980 `(rank * num_experts, hidden)` / `(intermediate * 2,
981 rank * num_experts)`
982 - `{module}.lora_{A,B}`: down_proj, with shapes
983 `(rank * num_experts, intermediate)` / `(hidden,
984 rank * num_experts)`
985 The 2D wrapper expects three stacked per-expert tensors,
986 `[w1, w2, w3]`, with `(num_experts, rank, in)` for lora_a and
987 `(num_experts, out, rank)` for lora_b. In the 3D layout w1
988 (gate_proj) and w3 (up_proj) share the rank-r intermediate
989 representation, so both halves use the same lora_a tensor.
990
991 Only invoked when `enable_mixed_moe_lora_format=True` and the
992 source LoRARequest declares `is_3d_lora_weight=True`.
993 """
994 gate_up_proj_lora = self._get_lora_layer_weights(
995 lora_model, module_name + ".base_layer"
996 )
997 down_proj_lora = self._get_lora_layer_weights(lora_model, module_name)
998 if gate_up_proj_lora is None or down_proj_lora is None:
999 # Either the adapter omits the experts entirely or the file
1000 # layout differs from what this path supports; leave the entry
1001 # untouched so set_lora can raise a clear error if needed.
1002 return
1003
1004 local_num_experts = module.local_num_experts
1005 global_num_experts = module.global_num_experts
1006 ep_rank = module.ep_rank
1007 expert_start = ep_rank * local_num_experts
1008 expert_end = expert_start + local_num_experts
1009
1010 # Reshape and EP-slice into per-expert 3D tensors. This mirrors
1011 # `_stack_moe_lora_weights`; for non-EP runs the slice is a no-op.
1012 gate_up_a = gate_up_proj_lora.lora_a.reshape(
1013 global_num_experts, -1, gate_up_proj_lora.lora_a.shape[-1]
1014 )[expert_start:expert_end].contiguous()
1015 gate_up_b = (
1016 gate_up_proj_lora.lora_b.reshape(
1017 gate_up_proj_lora.lora_b.shape[0], -1, global_num_experts
1018 )[..., expert_start:expert_end]
1019 .permute(2, 0, 1)
1020 .contiguous()
1021 )
1022 down_a = down_proj_lora.lora_a.reshape(
1023 global_num_experts, -1, down_proj_lora.lora_a.shape[-1]
1024 )[expert_start:expert_end].contiguous()
1025 down_b = (
1026 down_proj_lora.lora_b.reshape(
1027 down_proj_lora.lora_b.shape[0], -1, global_num_experts
1028 )[..., expert_start:expert_end]
1029 .permute(2, 0, 1)
1030 .contiguous()
1031 )
1032
1033 # Split the fused gate_up_proj output dim into separate w1 / w3
1034 # halves. GPT-OSS interleaves them along the output dim, all other
1035 # 3D MoE checkpoints we know about concatenate them.
1036 intermediate_x2 = gate_up_b.shape[1]
1037 if intermediate_x2 % 2 != 0:
1038 raise ValueError(
1039 "Expected gate_up_proj LoRA-B output dim to be 2 * intermediate, "
1040 f"got {intermediate_x2}."
1041 )
1042 intermediate = intermediate_x2 // 2
1043 base_arch = self.model.config.architectures[0]
1044 if base_arch == "GptOssForCausalLM":
1045 w1_b = gate_up_b[:, ::2, :].contiguous()
1046 w3_b = gate_up_b[:, 1::2, :].contiguous()
1047 else:
1048 w1_b = gate_up_b[:, :intermediate, :].contiguous()
1049 w3_b = gate_up_b[:, intermediate:, :].contiguous()
1050
1051 # In the 3D layout w1 and w3 share the same rank-r mid
1052 # representation, so they reuse the same lora_a tensor. The 2D
1053 # wrapper's set_lora copies whatever it gets here into independent
1054 # per-slice buffers, so the sharing is purely a CPU-side memory
1055 # optimization and does not affect numerics.
1056 down_proj_lora.lora_a = [gate_up_a, down_a, gate_up_a]
1057 down_proj_lora.lora_b = [w1_b, down_b, w3_b]
1058 # Drop the redundant base_layer entry to avoid double pin_memory
1059 # and to keep the activation path looking up only the wrapper key.
1060 lora_model.loras.pop(module_name + ".base_layer", None)
1061
1062 def _slice_moe_lora_ep(
1063 self,
1064 lora_model: LoRAModel,
1065 module: FusedMoEWithLoRA,
1066 module_name: str,
1067 ) -> None:
1068 """Slice cached LoRA tensors down to this rank's local experts.
1069
1070 Shared factors have expert dimension one and remain available on every
1071 rank. Per-expert factors are narrowed to the local expert block.
1072 """
1073 if not module.use_ep:
1074 return
1075 module_lora = self._get_lora_layer_weights(lora_model, module_name)
1076 if module_lora is None or not isinstance(module_lora.lora_a, list):
1077 return
1078
1079 local_num_experts = module.local_num_experts
1080 global_num_experts = module.global_num_experts
1081 ep_rank = module.ep_rank
1082 expert_start = ep_rank * local_num_experts
1083 expert_end = expert_start + local_num_experts
1084
1085 def _slice_local(t: torch.Tensor | None) -> torch.Tensor | None:
1086 if t is not None and t.shape[0] == global_num_experts:
1087 return t[expert_start:expert_end].contiguous()
1088 return t
1089
1090 module_lora.lora_a = [_slice_local(a) for a in module_lora.lora_a]
1091 module_lora.lora_b = [_slice_local(b) for b in module_lora.lora_b]

1093 def _restrict_to_local_experts(
1094 self, module_name: str, new_module_names: list[str]
1095 ) -> list[str]:
1096 """Narrow a flat expert-major sub-module list to this rank's experts.
1097
1098 ``new_module_names`` is produced by
1099 ``fused_moe_make_expert_params_mapping`` and is ordered
1100 ``[e=0,w1, e=0,w2, e=0,w3, e=1,w1, ...]`` (non-gated MoE has 2
1101 entries per expert instead of 3). When the module is a 2D
1102 ``FusedMoEWithLoRA`` with EP enabled, we slice the list to the
1103 contiguous block of experts owned by this rank so the downstream
1104 ``pack_moe`` call only consumes local weights and produces a
1105 tensor sized to ``local_num_experts`` directly.
1106
1107 Returns the original list unchanged for non-MoE modules, the 3D
1108 MoE path (handled separately by ``_stack_moe_lora_weights``),
1109 modules without EP, or layouts we cannot cleanly partition.
1110 """
1111 module = self.modules.get(module_name)
1112 if not isinstance(module, FusedMoEWithLoRA):
1113 return new_module_names
1114 if isinstance(module, FusedMoE3DWithLoRA):
1115 return new_module_names
1116 if not getattr(module, "use_ep", False):
1117 return new_module_names
1118 global_num_experts = module.global_num_experts
1119 local_num_experts = module.local_num_experts
1120 ep_rank = module.ep_rank
1121 if global_num_experts <= 0 or len(new_module_names) % global_num_experts != 0:
1122 return new_module_names
1123 per_expert = len(new_module_names) // global_num_experts
1124 start = ep_rank * local_num_experts * per_expert
1125 end = start + local_num_experts * per_expert
1126 return new_module_names[start:end]

1183 def set_adapter_mapping(self, mapping: LoRAMapping) -> None:
1184 # The punica metadata derives from the slot layout as well as the
1185 # mapping: an out-of-band add_lora() can LRU-evict and reassign slots
1186 # while the running batch, and thus the mapping, is unchanged.
1187 slot_layout = tuple(self.lora_index_to_id)
1188 if self._last_mapping != mapping or self._last_slot_layout != slot_layout:
1189 self._set_adapter_mapping(mapping)
1190 self._last_mapping = mapping
1191 self._last_slot_layout = slot_layout

1193 def remove_adapter(self, adapter_id: int) -> bool:
1194 self.deactivate_adapter(adapter_id)
1195 if adapter_id not in self._registered_adapters:
1196 return False
1197 self._registered_adapters.pop(adapter_id, None)
1198 return True
1199
1200 def list_adapters(self) -> dict[int, LoRAModel]:
1201 return dict(self._registered_adapters.cache)
1202
1203 def get_adapter(self, adapter_id: int) -> LoRAModel | None:
1204 return self._registered_adapters.get(adapter_id)

1207class LRUCacheLoRAModelManager(LoRAModelManager):
1208 """A model manager that manages multiple LoRAs with LRU cache."""
1209
1210 def add_adapter(self, lora: LoRAModel) -> bool:
1211 """Add a LoRAModel to the manager."""
1212 logger.debug("Adding lora. Model id: %d, int id: %d", lora.id, lora.id)
1213 if lora.id not in self._registered_adapters:
1214 self._add_adapter(lora)
1215 was_added = True
1216 else:
1217 # We always touch to update the LRU cache order
1218 self._registered_adapters.touch(lora.id)
1219 was_added = False
1220 return was_added
1221
1222 def activate_adapter(
1223 self,
1224 lora_id: int,
1225 ) -> bool:
1226 if (
1227 lora_id not in self._active_adapters
1228 and len(self._active_adapters) >= self.lora_slots
1229 ):
1230 self._active_adapters.remove_oldest()
1231 result = super().activate_adapter(lora_id)
1232 # We always touch to update the LRU cache order
1233 self._active_adapters.touch(lora_id)
1234 return result
1235
1236 def remove_oldest_adapter(self) -> bool:
1237 if len(self._registered_adapters) > 0:
1238 self._registered_adapters.remove_oldest()
1239 return True
1240 return False
1241
1242 def pin_adapter(self, lora_id: int) -> bool:
1243 """Pin a LoRAModel in the manager cache."""
1244 self._pin_lora_in_cpu_cache(lora_id)
1245 self._pin_lora_in_gpu_cache(lora_id)
1246 return True
1247
1248 def _pin_lora_in_cpu_cache(self, lora_id: int):
1249 try:
1250 self._registered_adapters.pin(lora_id)
1251 except ValueError as err:
1252 raise ValueError(
1253 f"Pinning failed. LoRA {lora_id} is not registered."
1254 ) from err
1255
1256 def _pin_lora_in_gpu_cache(self, lora_id: int):
1257 if lora_id not in self._active_adapters:
1258 # move lora to gpu if not already active
1259 self.activate_adapter(lora_id)
1260
1261 self._active_adapters.pin(lora_id)
