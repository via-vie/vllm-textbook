3"""
4KVConnectorBase_V1 Class for Distributed KV Cache & Hidden State
5communication in vLLM v1
6
7The class provides the following primitives:
8 Scheduler-side: runs in the scheduler, binds metadata, which
9 is used by the worker-side to load/save KV cache.
10 get_num_new_matched_tokens() - get number of new tokens
11 that exist in the remote KV cache. Might be called multiple
12 times for a given request and should be side-effect free.
13 update_state_after_alloc() - update KVConnector state after
14 temporary buffer alloc by the CacheManager.
15 update_connector_output() - update KVConnector state after
16 output is received from worker-side connectors.
17 request_finished() - called once when a request is finished,
18 with the computed kv cache blocks for the request.
19 Returns whether KV cache should be freed now or if the
20 connector now assumes responsibility for freeing the
21 the blocks asynchronously. Also optionally returns KV
22 transfer params.
23 take_events() - returns new KV events that were collected
24 by the connector since the last call.
25
26 Worker-side: runs in each worker, loads/saves KV cache to/from
27 the Connector based on the metadata.
28 handle_preemptions() - called for handling preempted requests
29 or request evicted blocks before they are overwritten
30
31 start_load_kv() - starts loading all KVs (maybe async)
32 wait_for_layer_load() - blocks until layer i load is done
33
34 save_kv_layer() - starts saving KV for layer i (maybe async)
35 wait_for_save() - blocks until all saves are done
36
37 get_transfer_results() - returns async send/receive completions and
38 receive failures in one snapshot.
39 build_connector_worker_meta() - builds metadata to be sent
40 back to the scheduler-side connector
41"""

406 """

408

410 Set of block IDs that encountered load errors.
411 Empty set if no load errors occurred.
412

414 - Applies to both sync- and async-loading requests.
415 - Async loading: failed blocks may be reported in any forward pass
416 up to and including the pass where the request ID is returned by

488 @abstractmethod
489 def get_num_new_matched_tokens(
490 self,
491 request: "Request",
492 num_computed_tokens: int,
493 ) -> tuple[int | None, bool]:
494 """
495 Get number of new tokens that can be loaded from the
496 external KV cache beyond the num_computed_tokens.
497
498 Args:
499 request (Request): the request object.
500 num_computed_tokens (int): the number of locally
501 computed tokens for this request
502
503 Returns:
504 A tuple with the following elements:
505 - An optional number of tokens that can be loaded from the
506 external KV cache beyond what is already computed.
507 If None, it means that the connector needs more time to
508 determine the number of matched tokens, and the scheduler
509 should query for this request again later.
510 - `True` if external KV cache tokens will be loaded
511 asynchronously (between scheduler steps). Must be
512 'False' if the first element is 0.
513
514 Notes:
515 The connector should only consider the largest prefix of prompt-
516 tokens for which KV cache is actually available at the time of the
517 call. If the cache cannot be loaded for some tokens (e.g., due to
518 connectivity issues or eviction), those tokens must not be taken
519 into account.
520 """
521 pass

523 @abstractmethod
524 def update_state_after_alloc(
525 self, request: "Request", blocks: "KVCacheBlocks", num_external_tokens: int
526 ):
527 """
528 Update KVConnector state after block allocation.
529
530 If get_num_new_matched_tokens previously returned True for a
531 request, this function may be called twice for that same request -
532 first when blocks are allocated for the connector tokens to be
533 asynchronously loaded into, and second when any additional blocks
534 are allocated, after the load/transfer is complete.
535
536 Decide whether to load based on ``num_external_tokens``, not on
537 whether ``blocks`` is empty: ``blocks`` may be non-empty even when
538 ``num_external_tokens == 0`` (e.g. a non-chosen sub-connector of
539 MultiConnector still receives the request's real blocks).
540
541 Args:
542 request (Request): the request object.
543 blocks (KVCacheBlocks): the blocks allocated for the request.
544 num_external_tokens (int): the number of tokens to load from the
545 external KV cache. 0 means nothing should be loaded.
546 """
547 pass
