82class _RetainedHandle:
83 """Handle that retains source tensors until all posted work completes.
84
85 Used by ``isend_object`` to keep the serialized CPU tensors alive while
86 gloo copies them in the background; ``wait`` drops the refs once drained.
87
88 ``wait`` must be idempotent: gloo ``Work.wait()`` is single-shot for
89 point-to-point operations — calling it a second time on an already
90 completed send blocks forever (verified empirically). Both the worker's
91 top-of-step drain of ``_pp_send_work`` and ``_reap_completed_isends``
92 may wait the same entry's metadata handle, so the second caller must
93 not re-wait the underlying works.
94 """
95
96 def __init__(self, works: list[Any], retained: tuple[torch.Tensor, ...]) -> None:
97 self._works = works
98 self._retained = retained
99 self._waited = False
100
101 def is_completed(self) -> bool:
102 return all(w.is_completed() for w in self._works)
103
104 def wait(self) -> None:
105 if self._waited:
106 return
107 for w in self._works:
108 w.wait()
109 self._waited = True
110 self._retained = ()

113def _split_tensor_dict(
114 tensor_dict: dict[str, torch.Tensor | Any],
115) -> tuple[list[tuple[str, Any]], list[torch.Tensor]]:
116 """Split the tensor dictionary into two parts:
117 1. A list of (key, value) pairs. If the value is a tensor, it is replaced
118 by its metadata.
119 2. A list of tensors.
120 """
121 metadata_list: list[tuple[str, Any]] = []
122 tensor_list: list[torch.Tensor] = []
123 for key, value in tensor_dict.items():
124 if isinstance(value, torch.Tensor):
125 # Note: we cannot use `value.device` here,
126 # because it contains not only the device type but also the device
127 # index (e.g. "cuda:0"). We only need the device type.
128 # receiving side will set the device index.
129 device = value.device.type
130 metadata_list.append(
131 (key, TensorMetadata(device, value.dtype, value.size()))
132 )
133 tensor_list.append(value)
134 else:
135 metadata_list.append((key, value))
136 return metadata_list, tensor_list

162def _apply_to_device_comms(
163 action: Callable[[DeviceCommunicatorBase], None],
164) -> None:

324def _create_subgroups_split_group(
325 group_ranks: list[list[int]],
326 group_name: str,
327 torch_distributed_backend: str | Backend,
328) -> tuple[ProcessGroup, ProcessGroup]:
329 """Create the device + CPU subgroups for ``GroupCoordinator`` via
330 ``torch.distributed.split_group``.
331
332 ``split_group`` is collective on the parent group, so every parent rank
333 must enter with the same ``split_ranks`` definition. Each rank receives
334 the subgroup it belongs to.
335 """
336 from vllm.distributed.utils import (
337 get_cpu_distributed_timeout_or_none,
338 get_distributed_timeout_or_none,
339 )
340
341 device_backend_str = _device_backend_str(torch_distributed_backend)
342 self_device_group = torch.distributed.split_group(
343 split_ranks=group_ranks,
344 group_desc=f"{group_name}:device",
345 backend=device_backend_str,
346 timeout=get_distributed_timeout_or_none(),
347 )
348 # CPU subgroup: split_group requires the requested backend filter to
349 # include the parent's default device type (= the device the parent PG
350 # was bound to via ``device_id``), so a cpu-only filter is rejected.
351 # Include the device backend in the filter; only the gloo backend is
352 # actually used for CPU collectives on this group.
353 self_cpu_group = torch.distributed.split_group(
354 split_ranks=group_ranks,
355 group_desc=f"{group_name}:cpu",
356 backend=f"cpu:gloo,{device_backend_str}",
357 timeout=get_cpu_distributed_timeout_or_none(),
358 )
359 return self_device_group, self_cpu_group

422class GroupCoordinator:
423 """
424 PyTorch ProcessGroup wrapper for a group of processes.
425 PyTorch ProcessGroup is bound to one specific communication backend,
426 e.g. NCCL, Gloo, MPI, etc.
427 GroupCoordinator takes charge of all the communication operations among
428 the processes in the group. It manages both CPU and device
429 communication.
430 """
431
432 # available attributes:
433 rank: int # global rank
434 ranks: list[int] # global ranks in the group
435 world_size: int # size of the group
436 # difference between `local_rank` and `rank_in_group`:
437 # if we have a group of size 4 across two nodes:
438 # Process | Node | Rank | Local Rank | Rank in Group
439 # 0 | 0 | 0 | 0 | 0
440 # 1 | 0 | 1 | 1 | 1
441 # 2 | 1 | 2 | 0 | 2
442 # 3 | 1 | 3 | 1 | 3
443 local_rank: int # local rank used to assign devices
444 rank_in_group: int # rank inside the group
445 cpu_group: ProcessGroup # group for CPU communication
446 device_group: ProcessGroup # group for device communication
447 # device communicator (if use_device_communicator=True)
448 device_communicator: DeviceCommunicatorBase | None
449 mq_broadcaster: Any | None # shared memory broadcaster

452 self,

454 local_rank: int,

458 group_name: str | None = None,
459 use_all2all: bool = False,

1071 all_gather_group: "GroupCoordinator | None" = None,
1072 all_gather_tensors: dict[str, bool] | None = None,
1073 ) -> dict[str, torch.Tensor | Any] | None:
1074 """Send the input tensor dictionary.
1075 NOTE: `dst` is the local rank of the source rank.
1076
1077 all_gather_group: The group for the all-gather operation. If provided,
1078 an optimization is enabled where each rank in the group sends a
1079 slice of a tensor and the receiver reconstructs it using an
1080 all-gather, which can improve performance. This is typically the
1081 tensor-parallel group.

1104 # appended by ``isend_tensor_dict`` is already drained; drop it
1105 # instead of leaking the tensor refs until the next async send.

1111 def _reap_completed_isends(self) -> None:
1112 """Lazily drop self-retained ``isend_tensor_dict`` entries that have
1113 completed, oldest first (FIFO).
1114
1115 gloo ``Work.is_completed()`` is unreliable (it can report completion
1116 before the background copy of the source buffer finishes), so the
1117 metadata handle (``handles[0]``, a gloo-backed ``_RetainedHandle``) is
1118 never used as the completion gate. Instead we gate on the tensor-send
1119 handles (``handles[1:]``), which run on the device backend where
1120 ``is_completed()`` is reliable. Once all tensor sends are done the
1121 peer must already have posted the matching tensor recvs — and since
1122 the receiver ingests metadata before tensors, the gloo metadata send
1123 is then guaranteed complete, so ``wait()`` on it is ~instant and only
1124 serves to drop the retained refs.
1125
1126 Metadata-only entries (``handles[1:]`` empty, e.g. an empty
1127 ``IntermediateTensors``) have no reliable completion signal; drop them
1128 best-effort without blocking on the gloo metadata wait.
1129 """
1130 pending = getattr(self, "_pending_isends", None)
1131 if pending is None:
1132 # Subclasses that replicate __init__ without calling
1133 # super().__init__ may not have the attribute yet.
1134 self._pending_isends = pending = deque()
1135 while pending:
1136 handles, _ = pending[0]
1137 tensor_handles = handles[1:]
1138 if not tensor_handles:
1139 pending.popleft()
1140 continue
1141 if not all(h.is_completed() for h in tensor_handles):
1142 # The oldest send is still in flight; FIFO ordering means the
1143 # rest are no further along, so stop here.
1144 break
1145 handles[0].wait()
1146 pending.popleft()

1156

1158 ``self._pending_isends`` (together with the sent source tensors) and
1159 reaped lazily on subsequent calls, so fire-and-forget callers that
1160 drop the handles cannot free or overwrite the send buffers while the
1161 sends are still in flight. Callers that do keep the handles may
1162 ``wait()`` them as before; the retention entry is dropped on reap

1178 )

1184 )

1187 metadata_group = self.cpu_group
1188
1189 metadata_list, tensor_list = _split_tensor_dict(tensor_dict)
1190
1191 tensor_keys = [k for k, v in tensor_dict.items() if isinstance(v, torch.Tensor)]
1192 assert len(tensor_keys) == len(tensor_list)
1193
1194 # Reap completed sends before posting new ones, bounding the
1195 # self-retention FIFO.

1331 def _postprocess(
1332 key: str = key,
1333 slice_tensor: torch.Tensor = slice_tensor,
1334 orig_shape: tuple[int, ...] = tuple(orig_shape),
1335 all_gather_group=all_gather_group,
1336 ) -> None:
1337 assert all_gather_group is not None
1338 tensor_dict[key] = all_gather_group.all_gather(
1339 slice_tensor, dim=0
1340 ).reshape(orig_shape)
1341
1342 postprocess.append(_postprocess)
1343 tensor_dict[key] = slice_tensor
1344 else:
1345 comm_group = metadata_group if full_tensor.is_cpu else group
1346 handle = torch.distributed.irecv(
1347 full_tensor, src=self.ranks[src], group=comm_group
1348 )
1349 handles.append(handle)
1350 tensor_dict[key] = full_tensor
1351 else:
1352 tensor_dict[key] = value
1353

1954def _engram_dp_shard_size(
1955 world_size: int,
1956 data_parallel_size: int,
1957 replica_size: int,
1958) -> int:
1959 """Size of DP shards aligned with complete replicas and node boundaries."""
1960 node_count = get_node_count()
1961 replicas_per_node, remainder = divmod(world_size, node_count * replica_size)
1962 if remainder:
1963 return 1
1964 # Shards must divide both the DP group and each node's replica count.
1965 shard_size = gcd(data_parallel_size, replicas_per_node)
1966 if shard_size == 1 or node_count == 1:
1967 return shard_size
1968 ranks_per_node = replicas_per_node * replica_size
1969 for start in range(0, world_size, ranks_per_node):
1970 same_node = in_the_same_node_as(get_world_group().cpu_group, start)
1971 node_ranks = [rank for rank, local in enumerate(same_node) if local]
1972 if node_ranks != list(range(start, start + ranks_per_node)):
1973 return 1
1974 return shard_size

2482 memory system (shared access to shared memory).
2483 """
2484 if isinstance(pg, ProcessGroup):

2490 world_size = torch.distributed.get_world_size(group=pg)

2493 ranks = torch.distributed.get_process_group_ranks(pg)

2501 [0] * world_size, dtype=torch.int32, device="cpu"
2502 )

2505 shm = None
2506
2507 try:
2508 with contextlib.suppress(OSError):

2510 # create a shared memory segment
2511 shm = shared_memory.SharedMemory(create=True, size=128)
2512 assert shm.buf is not None, "Buffer was not created"
2513 shm.buf[: len(magic_message)] = magic_message

2515 torch.distributed.broadcast_object_list(
2516 [shm.name], src=ranks[source_rank], group=pg
2517 )
2518 else:
2519 pg.broadcast_obj(shm.name, src=source_rank)
2520 is_in_the_same_node[rank] = 1
2521 else:
2522 # try to open the shared memory segment
2523 if isinstance(pg, ProcessGroup):
2524 recv = [None]

2526 recv, src=ranks[source_rank], group=pg

2532 # Python incorrectly tracks shared memory even if it is not
