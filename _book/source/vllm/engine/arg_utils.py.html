155def parse_type(return_type: Callable[[str], T]) -> Callable[[str], T]:
156 def _parse_type(val: str) -> T:
157 try:
158 return return_type(val)
159 except ValueError as e:
160 raise argparse.ArgumentTypeError(
161 f"Value {val} cannot be converted to {return_type}."
162 ) from e
163
164 return _parse_type

167def optional_type(return_type: Callable[[str], T]) -> Callable[[str], T | None]:
168 def _optional_type(val: str) -> T | None:
169 if val == "" or val == "None":
170 return None
171 return parse_type(return_type)(val)
172
173 return _optional_type

182def is_type(type_hint: TypeHint, type: TypeHintT) -> TypeIs[TypeHintT]:
183 """Check if the type hint is a specific type."""
184 return type_hint is type or get_origin(type_hint) is type

187def contains_type(type_hints: set[TypeHint], type: TypeHintT) -> bool:
188 """Check if the type hints contain a specific type."""
189 return any(is_type(type_hint, type) for type_hint in type_hints)

197def literal_to_kwargs(type_hints: set[TypeHint]) -> dict[str, Any]:
198 """Get the `type` and `choices` from a `Literal` type hint in `type_hints`.
199
200 If `type_hints` also contains `str`, we use `metavar` instead of `choices`.
201 """
202 type_hint = get_type(type_hints, Literal)
203 options = get_args(type_hint)
204 option_type = type(options[0])
205 if not all(isinstance(option, option_type) for option in options):
206 raise ValueError(
207 "All options must be of the same type. "
208 f"Got {options} with types {[type(c) for c in options]}"
209 )
210 kwarg = "metavar" if contains_type(type_hints, str) else "choices"
211 return {"type": option_type, kwarg: sorted(options)}

214def collection_to_kwargs(type_hints: set[TypeHint], type: TypeHint) -> dict[str, Any]:
215 type_hint = get_type(type_hints, type)
216 types = get_args(type_hint)
217 elem_type = types[0]
218
219 # Handle Ellipsis
220 assert all(t is elem_type for t in types if t is not Ellipsis), (
221 f"All non-Ellipsis elements must be of the same type. Got {types}."
222 )
223
224 # Handle Union types
225 if get_origin(elem_type) in {Union, UnionType}:
226 # Union for Union[X, Y] and UnionType for X | Y
227 assert str in get_args(elem_type), (
228 "If element can have multiple types, one must be 'str' "
229 f"(i.e. 'list[int | str]'). Got {elem_type}."
230 )
231 elem_type = str
232
233 return {
234 "type": elem_type,
235 "nargs": "+" if type is not tuple or Ellipsis in types else len(types),
236 }

239def is_not_builtin(type_hint: TypeHint) -> bool:
240 """Check if the class is not a built-in type."""
241 return type_hint.__module__ != "builtins"

244def get_type_hints(type_hint: TypeHint) -> set[TypeHint]:
245 """Extract type hints from Annotated or Union type hints."""
246 type_hints: set[TypeHint] = set()
247 origin = get_origin(type_hint)
248 args = get_args(type_hint)
249
250 if origin is Annotated:
251 type_hints.update(get_type_hints(args[0]))
252 elif origin in {Union, UnionType}:
253 # Union for Union[X, Y] and UnionType for X | Y
254 for arg in args:
255 type_hints.update(get_type_hints(arg))
256 else:
257 type_hints.add(type_hint)
258
259 return type_hints

279def _expand_json_human_readable_numbers(val: str) -> str:
280 """Expand human-readable number suffixes in a JSON string.
281
282 Based on :func:`human_readable_int` so that the ``k/m/g/t`` (decimal) and
283 ``K/M/G/T`` (binary) conventions work out the box.
284 Also works inside JSON config arguments such
285 as ``--kv-transfer-config '{"cpu_bytes_to_use": 80m}'``.
286
287 Only bare (unquoted) tokens are replaced so that JSON string values
288 like ``"model_name"`` are never modified.
289 """
290 # Split on quoted strings so we only touch non-string regions.
291 parts = re.split(r'("(?:[^"\\]|\\.)*")', val)
292 for i in range(0, len(parts), 2): # even indices = outside strings
293 parts[i] = re.sub(
294 r"\b\d+(?:\.\d+)?[kKmMgGtT]\b",
295 lambda m: str(human_readable_int(m.group())),
296 parts[i],
297 )
298 return "".join(parts)

301@functools.lru_cache(maxsize=30)
302def _compute_kwargs(cls: ConfigType) -> dict[str, dict[str, Any]]:
303 # Save time only getting attr docs if we're generating help text
304 cls_docs = get_attr_docs(cls) if NEEDS_HELP else {}
305 kwargs = {}
306 for field in fields(cls):
307 # Get the set of possible types for the field
308 type_hints: set[TypeHint] = get_type_hints(field.type)
309
310 # If the field is a dataclass, we can use the model_validate_json
311 generator = (th for th in type_hints if is_dataclass(th))
312 dataclass_cls = next(generator, None)
313
314 # Get the default value of the field
315 if field.default is not MISSING:
316 default = field.default
317 # Handle pydantic.Field defaults
318 if isinstance(default, FieldInfo):
319 if default.default_factory is None:
320 default = default.default
321 else:
322 # VllmConfig's Fields have default_factory set to config classes.
323 # These could emit logs on init, which would be confusing.
324 with suppress_logging():
325 default = default.default_factory() # type: ignore[call-arg]
326 elif field.default_factory is not MISSING:
327 default = field.default_factory()
328
329 # Get the help text for the field
330 name = field.name
331 help = cls_docs.get(name, "").strip()
332 # Escape % for argparse
333 help = help.replace("%", "%%")
334
335 # Initialise the kwargs dictionary for the field
336 kwargs[name] = {"default": default, "help": help}
337
338 # Set other kwargs based on the type hints
339 json_tip = (
340 "Should either be a valid JSON string or JSON keys passed individually."
341 )
342 if dataclass_cls is not None:
343
344 def parse_dataclass(val: str, cls=dataclass_cls) -> Any:
345 try:
346 val = _expand_json_human_readable_numbers(val)
347 return TypeAdapter(cls).validate_json(val)
348 except ValidationError as e:
349 raise argparse.ArgumentTypeError(repr(e)) from e
350
351 kwargs[name]["type"] = parse_dataclass
352 kwargs[name]["help"] += _maybe_add_docs_url(dataclass_cls)
353 kwargs[name]["help"] += f"\n\n{json_tip}"
354 elif type_hints == {bool, str, type(None)}:
355 # Optional-valued flag: bare flag -> True, value -> str.
356 kwargs[name]["type"] = str
357 kwargs[name]["nargs"] = "?"
358 kwargs[name]["const"] = True
359 elif contains_type(type_hints, bool):
360 # Creates --no-<name> and --<name> flags
361 kwargs[name]["action"] = argparse.BooleanOptionalAction
362 elif contains_type(type_hints, Literal):
363 kwargs[name].update(literal_to_kwargs(type_hints))
364 elif contains_type(type_hints, tuple):
365 kwargs[name].update(collection_to_kwargs(type_hints, tuple))
366 elif contains_type(type_hints, list):
367 kwargs[name].update(collection_to_kwargs(type_hints, list))
368 elif contains_type(type_hints, set):
369 kwargs[name].update(collection_to_kwargs(type_hints, set))
370 elif contains_type(type_hints, int):
371 # Arguments that accept human-readable integer strings (e.g., 1K, 2M, 1G)
372 human_readable_int_args = {
373 "max_num_batched_tokens",
374 "max_num_scheduled_tokens",
375 "kv_cache_memory_bytes",
376 "safetensors_prefetch_block_size",
377 "max_num_queued_tokens",
378 }
379 if name == "max_model_len":
380 kwargs[name]["type"] = human_readable_int_or_auto
381 kwargs[name]["help"] += f"\n\n{human_readable_int_or_auto.__doc__}"
382 elif name in human_readable_int_args:
383 kwargs[name]["type"] = human_readable_int
384 kwargs[name]["help"] += f"\n\n{human_readable_int.__doc__}"
385 else:
386 kwargs[name]["type"] = int
387 elif contains_type(type_hints, float):
388 kwargs[name]["type"] = float
389 elif contains_type(type_hints, dict) and (
390 contains_type(type_hints, str)
391 or any(is_not_builtin(th) for th in type_hints)
392 ):
393 kwargs[name]["type"] = union_dict_and_str
394 elif contains_type(type_hints, dict):
395 kwargs[name]["type"] = parse_type(json.loads)
396 kwargs[name]["help"] += f"\n\n{json_tip}"
397 elif contains_type(type_hints, str) or any(
398 is_not_builtin(th) for th in type_hints
399 ):
400 kwargs[name]["type"] = str
401 else:
402 raise ValueError(f"Unsupported type {type_hints} for argument {name}.")
403
404 # If the type hint was a sequence of literals, use the helper function
405 # to update the type and choices
406 if get_origin(kwargs[name].get("type")) is Literal:
407 kwargs[name].update(literal_to_kwargs({kwargs[name]["type"]}))
408
409 # If None is in type_hints, make the argument optional.
410 # But not if it's a bool, argparse will handle this better.
411 if type(None) in type_hints and not contains_type(type_hints, bool):
412 kwargs[name]["type"] = optional_type(kwargs[name]["type"])
413 if kwargs[name].get("choices"):
414 kwargs[name]["choices"].append("None")
415 return kwargs

418def get_kwargs(cls: ConfigType) -> dict[str, dict[str, Any]]:
419 """Return argparse kwargs for the given Config dataclass.
420
421 If `--help` or `mkdocs` are not present in the command line command, the
422 attribute documentation will not be included in the help output.
423
424 The heavy computation is cached via functools.lru_cache, and a deep copy
425 is returned so callers can mutate the dictionary without affecting the
426 cached version.
427 """
428 return copy.deepcopy(_compute_kwargs(cls))

2873 model_config: ModelConfig,
2874 ) -> tuple[int, str] | None:
2875 """Get the minimum max_num_batched_tokens needed for a multimodal
2876 prefix-LM model to process at least one item of any supported modality.
2877
2878 Returns (token_count, modality_name) for the most expensive modality,
2879 or None if the value cannot be determined at this stage.
2880 """
2881 try:
2882 from vllm.multimodal import MULTIMODAL_REGISTRY
2883
2884 # get_processing_info returns the model's multimodal processing
2885 # metadata (supported modalities, token limits) without loading
2886 # model weights or generating dummy data.
2887 info = MULTIMODAL_REGISTRY.get_processing_info(model_config)
2888 mm_counts = {modality: 1 for modality in info.supported_mm_limits}
2889 # get_mm_max_tokens_per_item returns pre-computed per-item token
2890 # ceilings for models that override it (e.g., Gemma4), or None
2891 # for models that rely on dummy-input profiling. When None is
2892 # returned we bail out — no dummy generation is triggered here.
2893 max_tokens = info.get_mm_max_tokens_per_item(
2894 seq_len=model_config.max_model_len,
2895 mm_counts=mm_counts,
2896 )
2897 if max_tokens is not None:
2898 modality = max(max_tokens, key=max_tokens.__getitem__)
2899 return (max_tokens[modality], modality)
2900 except Exception as e:
2901 logger.warning("Failed to determine min multimodal batched tokens: %s", e)
2902 return None

2905 self,
2906 usage_context: UsageContext | None,
2907 model_config: ModelConfig,
2908 parallel_config: ParallelConfig,
2909 ):
2910 world_size = self.pipeline_parallel_size * self.tensor_parallel_size
2911 (
2912 default_max_num_batched_tokens,
2913 default_max_num_seqs,
2914 ) = self.get_batch_defaults(world_size)
2915
2916 orig_max_num_batched_tokens = self.max_num_batched_tokens
2917 orig_max_num_seqs = self.max_num_seqs
2918
2919 if self.max_num_batched_tokens is None:
2920 if parallel_config.use_batched_dp_moe:
2921 self.max_num_batched_tokens = (
2922 SchedulerConfig.DEFAULT_MAX_NUM_BATCHED_TOKENS_FOR_BATCHED_DP

2932 usage_context,
2933 SchedulerConfig.DEFAULT_MAX_NUM_SEQS,
2934 )
2935

2937 if self.performance_mode == "throughput":
2938 if orig_max_num_batched_tokens is None:
2939 self.max_num_batched_tokens *= 2
2940 if orig_max_num_seqs is None:
2941 self.max_num_seqs *= 2
2942
2943 if orig_max_num_batched_tokens is None:
2944 assert model_config.max_model_len is not None, (
2945 "max_model_len must be set by this point"
2946 )
2947 if not self.enable_chunked_prefill:
2948 # If max_model_len is too short, use the default for higher throughput.
2949 self.max_num_batched_tokens = max(
2950 model_config.max_model_len,
2951 self.max_num_batched_tokens,
2952 )
2953
2954 # For multimodal prefix-LM models (e.g., Gemma 4) that disable
2955 # chunked MM input, a single multimodal item must fit in one batch.
2956 # Raise the floor to accommodate the largest per-item token count.
2957 if model_config.is_multimodal_model and model_config.is_mm_prefix_lm:

2964 self.max_num_batched_tokens,
2965 mm_min,
2966 modality,
2967 model_config.model,
2968 )

2972 # Ensure max_num_batched_tokens does not exceed model limit.
2973 # Some models (e.g., Whisper) have embeddings tied to max length.
2974 self.max_num_batched_tokens = min(
2975 self.max_num_seqs * model_config.max_model_len,
2976 self.max_num_batched_tokens,
2977 )
2978
2979 logger.debug(

2981 self.max_num_batched_tokens,
2982 usage_context.value if usage_context else None,
2983 )
2984
2985 if orig_max_num_seqs is None:
2986 assert self.max_num_batched_tokens is not None # For type checking
2987 self.max_num_seqs = min(self.max_num_seqs, self.max_num_batched_tokens)
2988
2989 logger.debug(
2990 "Defaulting max_num_seqs to %d for %s usage context.",
2991 self.max_num_seqs,
2992 usage_context.value if usage_context else None,
2993 )
