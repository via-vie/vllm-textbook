58def ignore_torch_compile(cls: type[_T]) -> type[_T]:
59 """
60 A decorator to ignore support_torch_compile decorator
61 on the class. This is useful when a parent class has
62 a support_torch_compile decorator, but we don't want to
63 compile the class `cls` that inherits the parent class.
64 This only ignores compiling the forward of the class the
65 decorator is applied to.
66
67 If the parent has ignore_torch_compile but the child has
68 support_torch_compile, the child will still be compiled.
69
70 If the class has one or more submodules
71 that have support_torch_compile decorator applied, compile will
72 not be ignored for those submodules.
73 """
74 setattr(cls, IGNORE_COMPILE_KEY, True)
75 return cls

120def support_torch_compile(
121 cls: type[_T] | None = None,
122 *,
123 dynamic_arg_dims: dict[str, int | list[int] | dict[int, str]] | None = None,
124 mark_unbacked_dims: dict[str, int | list[int]] | None = None,
125 enable_if: Callable[[VllmConfig], bool] | None = None,
126 is_encoder: bool = False,
127) -> Callable[[type[_T]], type[_T]] | type[_T]:
128 """
129 A decorator to add support for compiling the forward method of a class.
130
131 Usage 1: use directly as a decorator without arguments:
132
133 ```python
134 @support_torch_compile
135 class MyModel(nn.Module):
136 def forward(self, x: torch.Tensor, y: Optional[torch.Tensor]): ...
137 ```
138
139 Usage 2: use as a decorator with arguments:
140
141 ```python
142 @support_torch_compile(dynamic_arg_dims={"x": 0, "y": 0})
143 class MyModel(nn.Module):
144 def forward(self, x: torch.Tensor, y: Optional[torch.Tensor]): ...
145 ```
146
147 `dynamic_arg_dims` is a dictionary that maps argument names to the dynamic
148 dimensions of the argument. The value can be:
149 - int: a single dimension index (e.g., 0)
150 - list[int]: multiple dimension indices (e.g., [0, 1])
151 - dict[int, str]: dimension to shape_id mapping for shape relations
152 (e.g., {0: "b"}). Dimensions with the same shape_id share the same
153 unbacked symbol.
154
155 if `dynamic_arg_dims` is `None`, it is inferred from the type annotation
156 of the `forward` method, based on the following default rules:
157
158 - if the argument is annotated as `torch.Tensor` or
159 `Optional[torch.Tensor]`, the first dimension will be
160 marked as dynamic.
161 - if the argument is annotated as `IntermediateTensors`, the first
162 dimension of all the tensors in the intermediate tensors
163 will be marked as dynamic.
164
165 During runtime, when we actually mark dimensions of tensors,
166 it depends on the value of arguments:
167
168 - if it is a single integer (can be negative), the corresponding dimension
169 of the argument will be marked as dynamic.
170 - if it is `None`, ignored.
171 - if it is `IntermediateTensors`, all the tensors in the intermediate
172 tensors will be marked as dynamic.
173 - otherwise, it will raise an error.
174
175 NOTE: if an argument is `None`, it should always be passed as `None` during
176 the lifetime of the model, otherwise, it cannot be captured as a single
177 computation graph.
178
179 `enable_if` is a function that takes a `VllmConfig` object as input and
180 returns a boolean value indicating whether to compile the model or not.
181 This is useful if you want to compile the model only when certain
182 conditions are met.
183
184 `mark_unbacked_dims` is a dictionary that maps argument names with a dynamic
185 dim to be decorated with `mark_unbacked`. This is useful if we would like to
186 enforce that dynamo does not specialize on 0/1 values in the case of dummy input
187 such as for vision model compilation
188
189 `is_encoder` marks this module as a portion of an multimodal encoder.
190 When True, the compile range upper bound is set to MAX_INT32 instead of
191 max_num_batched_tokens, since encoder input shapes are unpredictable.
192 This is typically used for vision encoder sub-modules in multimodal models.
193
194 `shape_invariants` is a function that gets compiled right before forward.
195 The function should have the torch._check calls that are needed to set
196 the relationships between different input sizes. For example:
197 torch._check(input_ids.size()[0] == inputs_embeds.size()[0])
198 This enforces constraints on the symbolic shapes without hardcoding
199 specific values. It is needed for some models to avoid data dependent
200 errors and maximize perf when unbacked shapes are used.
201 """
202
203 def cls_decorator_helper(cls: type[_T]) -> type[_T]:
204 # helper to pass `dynamic_arg_dims` to `_support_torch_compile`
205 # to avoid too much indentation for `_support_torch_compile`
206 if not hasattr(cls, "forward"):
207 raise TypeError("decorated class should have a forward method.")
208 sig = inspect.signature(cls.forward)
209 inferred_dynamic_arg_dims = dynamic_arg_dims
210 if inferred_dynamic_arg_dims is None:
211 inferred_dynamic_arg_dims = {}
212 for k, v in sig.parameters.items():
213 if v.annotation in [
214 torch.Tensor,
215 torch.Tensor | None,
216 torch.FloatTensor,
217 torch.FloatTensor | None,
218 IntermediateTensors,
219 IntermediateTensors | None,
220 ]:
221 inferred_dynamic_arg_dims[k] = 0
222
223 logger.debug(
224 ("Inferred dynamic dimensions for forward method of %s: %s"),
225 cls,
226 list(inferred_dynamic_arg_dims.keys()),
227 )
228
229 if len(inferred_dynamic_arg_dims) == 0:
230 raise ValueError(
231 "No dynamic dimensions found in the forward method of "
232 f"{cls}. Please provide dynamic_arg_dims explicitly."
233 )
234
235 for k in inferred_dynamic_arg_dims:
236 if k not in sig.parameters:
237 raise ValueError(
238 f"Argument {k} not found in the forward method of {cls}"
239 )
240
241 return _support_torch_compile(
242 cls,
243 inferred_dynamic_arg_dims,
244 mark_unbacked_dims,
245 enable_if,
246 is_encoder,
247 )
248
249 if cls is not None:
250 # use `support_torch_compile` as a decorator without arguments
251 assert isinstance(cls, type)
252 return cls_decorator_helper(cls)
253
254 return cls_decorator_helper

257def _model_hash_key(fn: Callable[..., Any]) -> str:
258 import vllm
259
260 sha256_hash = hashlib.sha256()
261 sha256_hash.update(vllm.__version__.encode())
262 sha256_hash.update(fn.__qualname__.encode())
263 sha256_hash.update(str(fn.__code__.co_firstlineno).encode())
264 return sha256_hash.hexdigest()

267def _verify_source_unchanged(
268 source_info: "SourceInfo", vllm_config: VllmConfig
269) -> None:
270 from .caching import _compute_code_hash, _compute_code_hash_with_content
271
272 file_contents = {}
273 for source in source_info.inlined_sources:
274 module = sys.modules[source.module]
275 file = inspect.getfile(module)
276 vllm_config.compilation_config.traced_files.add(file)
277 file_contents[file] = source.content
278 expected_checksum = _compute_code_hash_with_content(file_contents)
279 actual_checksum = _compute_code_hash(set(file_contents.keys()))
280 if expected_checksum != actual_checksum:
281 raise RuntimeError(
282 "Source code has changed since the last compilation. Recompiling the model."
283 )

286def _try_load_aot_compiled_fn(
287 model: Any,
288 aot_compilation_path: str,
289) -> Any | None:
290 """Try to load an AOT-compiled function from disk.
291
292 Returns the loaded callable on success, or None on failure.
293 Re-raises on failure when ``VLLM_FORCE_AOT_LOAD`` is set.
294 """
295 try:
296 with monitor_torch_compile(model.vllm_config, is_encoder=model._is_encoder):
297 with (
298 set_current_vllm_config(model.vllm_config),
299 open(aot_compilation_path, "rb") as f,
300 ):
301 loaded_fn = torch.compiler.load_compiled_function(
302 f, f_globals=model.forward.__globals__
303 )
304 _verify_source_unchanged(loaded_fn.source_info(), model.vllm_config)
305 ds_config = model.compilation_config.dynamic_shapes_config
306 if not ds_config.evaluate_guards:
307 loaded_fn.disable_guard_check()
308 # Eagerly load compiled artifacts now that traced_files
309 # is populated by _verify_source_unchanged.
310 with maybe_use_cudagraph_partition_wrapper(model.vllm_config):
311 loaded_fn._artifacts.compiled_fn.finalize_loading(model.vllm_config)
312 compilation_counter.num_aot_artifacts_loaded += 1
313 logger.info(
314 "Directly load AOT compilation from path %s", aot_compilation_path
315 )
316 return loaded_fn
317 except Exception as e:
318 if os.path.exists(aot_compilation_path):
319 if isinstance(e, EOFError):
320 message = "Compile cache file corrupted."
321 else:
322 message = str(e)
323 logger.warning(
324 "Compiling model again due to a load failure from %s, reason: %s",
325 aot_compilation_path,
326 message,
327 )
328 if envs.VLLM_FORCE_AOT_LOAD:
329 raise e
330 return None

333def _support_torch_compile(
334 cls: type[_T],
335 dynamic_arg_dims: dict[str, int | list[int] | dict[int, str]],
336 mark_unbacked_dims: dict[str, int | list[int]] | None = None,
337 enable_if: Callable[[VllmConfig], bool] | None = None,
338 is_encoder: bool = False,
339) -> type[_T]:
340 """Internal implementation of support_torch_compile decorator."""
341
342 if TorchCompileWithNoGuardsWrapper in cls.__bases__:
343 # support decorating multiple times
344 return cls
345
346 # take care of method resolution order
347 # make sure super().__init__ is called on the base class
348 # other than TorchCompileWithNoGuardsWrapper
349 cls.__bases__ = cls.__bases__ + (TorchCompileWithNoGuardsWrapper,)
350
351 old_init = cls.__init__
352
353 setattr(cls, IGNORE_COMPILE_KEY, False)
354
355 def __init__(
356 self: _T,
357 *args,
358 vllm_config: VllmConfig | None = None,
359 prefix: str = "",
360 **kwargs: Any,
361 ) -> None:
362 if vllm_config is None:
363 vllm_config = get_current_vllm_config()
364
365 # NOTE: to support multimodal models (such as encoder),
366 # we may not have vllm_config so we may need to patch it
367 sig = inspect.signature(old_init)
368 # Check that any positional arguments match the old_init method signature
369 annotations = [p.annotation for p in sig.parameters.values()]
370 for arg, annotation in zip(args, annotations):
371 if annotation is inspect._empty:
372 continue
373 if not isinstance(arg, annotation):
374 init = f"'{type(self).__name__}.__init__'"
375 arg_type = f"'{type(arg).__name__}'"
376 raise TypeError(
377 f"{init} received a positional argument of type {arg_type}, "
378 "but no parameter of that type was found in the method signature. "
379 f"Please either annotate {init} or pass it as a keyword argument."
380 )
381 if "vllm_config" in sig.parameters:
382 kwargs["vllm_config"] = vllm_config
383 if "prefix" in sig.parameters:
384 kwargs["prefix"] = prefix
385 old_init(self, *args, **kwargs)
386
387 self.vllm_config = vllm_config
388 self.compilation_config = self.vllm_config.compilation_config
389 enable_compile = enable_if is None or enable_if(vllm_config)
390 # for CompilationMode.STOCK_TORCH_COMPILE , the upper level model runner
391 # will handle the compilation, so we don't need to do anything here.
392 self.do_not_compile = (
393 self.compilation_config.mode
394 in [CompilationMode.NONE, CompilationMode.STOCK_TORCH_COMPILE]
395 or _should_ignore_torch_compile(self.__class__)
396 or not enable_compile
397 )
398 if self.do_not_compile:
399 return
400
401 self._dynamic_arg_dims = dynamic_arg_dims
402
403 self.was_aot_compile_fn_loaded_from_disk = False
404 compilation_counter.num_models_seen += 1
405 self.compiled = False
406
407 # Handled by monkeypatching `TorchCompileWithNoGuardsWrapper` into base class
408 TorchCompileWithNoGuardsWrapper.__init__(
409 self,
410 compile_prefix=cls.__name__ if is_encoder else "",
411 is_encoder=is_encoder,
412 )
413
414 cls.__init__ = __init__

416 def _mark_dynamic_inputs(
417 mod: type[_T], ds_type: DynamicShapesType, *args: Any, **kwargs: Any
418 ) -> None:
419 def mark_dynamic(
420 arg: torch.Tensor, dim_shape_pairs: list[tuple[int, str | None]]
421 ) -> None:
422 if ds_type == DynamicShapesType.UNBACKED:
423 if is_torch_equal_or_newer("2.10.0"):
424 for dim, shape_id in dim_shape_pairs:
425 if shape_id is not None:
426 if not _SUPPORTS_SHAPE_ID:
427 raise RuntimeError(
428 f"shape_id='{shape_id}' requires PyTorch >= 2.11.0"
429 )
430 torch._dynamo.decorators.mark_unbacked(
431 arg,
432 dim,
433 hint_override=arg.size()[dim],
434 shape_id=shape_id,
435 )
436 else:
437 torch._dynamo.decorators.mark_unbacked(
438 arg,
439 dim,
440 hint_override=arg.size()[dim],
441 )
442 else:
443 # For older versions, we can't use hint_override or shape_id
444 dims = [dim for dim, _ in dim_shape_pairs]
445 torch._dynamo.decorators.mark_unbacked(arg, dims)
446 else:
447 dims = [dim for dim, _ in dim_shape_pairs]
448 torch._dynamo.mark_dynamic(arg, dims)
449
450 sig = inspect.signature(mod.__class__.forward) # type: ignore[attr-defined]
451 bound_args = sig.bind(mod, *args, **kwargs)
452 bound_args.apply_defaults()
453
454 # Normalize dynamic_arg_dims to dict[str, dict[int, str | None]]
455 normalized_dims: dict[str, dict[int, str | None]] = {}
456 for k, v in dynamic_arg_dims.items():
457 if isinstance(v, dict):
458 normalized_dims[k] = {dim: shape_id for dim, shape_id in v.items()}
459 elif isinstance(v, int):
460 normalized_dims[k] = {v: None}
461 else:
462 normalized_dims[k] = {d: None for d in v}
463
464 for k, dim_to_shape_id in normalized_dims.items():
465 arg = bound_args.arguments.get(k)
466
467 if arg is not None:
468 dims = list(dim_to_shape_id.keys())
469
470 if isinstance(arg, torch.Tensor):
471 dim_shape_pairs = [
472 (arg.ndim + d if d < 0 else d, dim_to_shape_id.get(d))
473 for d in dims
474 ]
475 mark_dynamic(arg, dim_shape_pairs)
476 elif isinstance(arg, IntermediateTensors):
477 for tensor in arg.tensors.values():
478 dim_shape_pairs = [
479 (tensor.ndim + d if d < 0 else d, dim_to_shape_id.get(d))
480 for d in dims
481 ]
482 mark_dynamic(tensor, dim_shape_pairs)
483 else:
484 raise ValueError(
485 f"Unsupported dynamic dimensions {dims} "
486 f"for argument {k} with type {type(arg)}."
487 )
488
489 if mark_unbacked_dims:
490 for k, dims_val in mark_unbacked_dims.items():
491 arg = bound_args.arguments.get(k)
492 if arg is not None:
493 dims = [dims_val] if isinstance(dims_val, int) else list(dims_val)
494 if isinstance(arg, torch.Tensor):
495 dims = [arg.ndim + d if d < 0 else d for d in dims]
496 if is_torch_equal_or_newer("2.10.0"):
497 for dim in dims:
498 torch._dynamo.decorators.mark_unbacked(
499 arg, dim, hint_override=arg.size()[dim]
500 )
501 else:
502 torch._dynamo.decorators.mark_unbacked(arg, dims)
503
504 def __call__(self: type[_T], *args: Any, **kwargs: Any) -> Any:
505 # torch.compiler.is_compiling() means we are inside the compilation
506 # e.g. TPU has the compilation logic in model runner, so we don't
507 # need to compile the model inside.
508 if self.do_not_compile or torch.compiler.is_compiling():
509 return self.forward(*args, **kwargs)
510
511 # If skip_compiled is set, bypass compiled model call. This is used e.g. for
512 # enc-dec models where tensor shapes/types vary across invocations, preventing
513 # the capture of a single computational graph.
514 if is_forward_context_available() and get_forward_context().skip_compiled:
515 return self.forward(*args, **kwargs)
516
517 # if aot_compiled_fn is set, call it with partition wrapper context.
518 # The partition wrapper must be active at runtime for CUDA graph
519 # capture to work correctly with inductor graph partitioning.
520 if getattr(self, "aot_compiled_fn", None) is not None:
521 with maybe_use_cudagraph_partition_wrapper(self.vllm_config):
522 return self.aot_compiled_fn(self, *args, **kwargs)
523
524 ds_type = self.compilation_config.dynamic_shapes_config.type
525 cache_dir = None
526 aot_compilation_path = None
527 if envs.VLLM_USE_AOT_COMPILE:
528 """
529 When using torch.compile in AOT mode, we store the cache artifacts
530 under VLLM_CACHE_ROOT/torch_compile_cache/torch_aot_compile/{hash}
531 The {hash} contains all of the factors except for the source files
532 being traced through, because we don't actually know which source
533 files to check at this point (before dynamo runs).
534 On loading we will actually look at the source files being traced
535 through. If any source file have changed (compared with the
536 serialized backend artifacts), then we need to generate a new AOT
537 compile artifact from scratch.
538 """
539 from .caching import aot_compile_hash_factors
540
541 factors: list[str] = aot_compile_hash_factors(self.vllm_config)
542
543 factors.append(_model_hash_key(self.forward))
544 hash_key = hashlib.sha256(str(factors).encode()).hexdigest()
545 cache_dir = os.path.join(
546 envs.VLLM_CACHE_ROOT,
547 "torch_compile_cache",
548 "torch_aot_compile",
549 hash_key,
550 )
551
552 # Hash-level dir; shared across ranks on the same node.
553 self.compilation_config.local_cache_dir = cache_dir
554 inductor_cache = os.path.join(cache_dir, "inductor_cache")
555 os.makedirs(inductor_cache, exist_ok=True)
556 # Process-wide: post-load execution, CUDA-graph capture, and later
557 # autotune/recompile all need to write under {hash}/inductor_cache/.
558 # Unconditional because torch's cache_dir() may have pre-filled the
559 # /tmp default during import, making setdefault a no-op.
560 os.environ["TORCHINDUCTOR_CACHE_DIR"] = inductor_cache
561
562 rank = self.vllm_config.parallel_config.rank
563 dp_rank = self.vllm_config.parallel_config.data_parallel_index
564 cache_dir = os.path.join(cache_dir, f"rank_{rank}_{dp_rank}")
565 aot_compilation_path = os.path.join(cache_dir, "model")
566 if not envs.VLLM_DISABLE_COMPILE_CACHE:
567 loaded_fn = _try_load_aot_compiled_fn(self, aot_compilation_path)
568 if loaded_fn is not None:
569 self.aot_compiled_fn = loaded_fn
570 self.was_aot_compile_fn_loaded_from_disk = True
571 with (
572 monitor_profiling_run(),
573 maybe_use_cudagraph_partition_wrapper(self.vllm_config),
574 ):
575 output = self.aot_compiled_fn(self, *args, **kwargs)
576 return output
577
578 if self.compiled:
579 assert (
580 not envs.VLLM_USE_AOT_COMPILE
581 or self.vllm_config.compilation_config.backend == "eager"
582 )
583 return TorchCompileWithNoGuardsWrapper.__call__(self, *args, **kwargs) # type: ignore[arg-type]
584
585 # This is the path for the first compilation.
586 # the first compilation needs to have dynamic shapes marked
587 _mark_dynamic_inputs(
588 self,
589 ds_type,
590 *args,
591 **kwargs,
592 )
593
594 # Tensors that enter the graph through the forward context rather than
595 # model args (e.g. the MoE padding mask) need their batch dim marked
596 # dynamic too, otherwise AOT-compiled artifacts bake a static guard
597 # for the compile-time batch size.
598 if is_forward_context_available():
599 is_padding = get_forward_context().is_padding
600 if is_padding is not None:
601 torch._dynamo.mark_dynamic(is_padding, 0)
602
603 original_code_object = self.original_code_object()
604 logger.debug("Start compiling function %s", original_code_object)
605
606 # we do not want tp delete the original code object entries since
607 # we depend on them now to look up cached compiled functions.
608 # torch._dynamo.eval_frame.remove_from_cache(original_code_object)
609
610 # collect all relevant files traced by Dynamo,
611 # so that the compilation cache can trigger re-compilation
612 # properly when any of these files change.
613
614 # 1. the file containing the top-level forward function
615 self.compilation_config.traced_files.add(original_code_object.co_filename)
616
617 # 2. every time Dynamo sees a function call, it will inline
618 # the function by calling InliningInstructionTranslator.inline_call_
619 # we hijack this function to know all the functions called
620 # during Dynamo tracing, and their corresponding files
621 inline_call = InliningInstructionTranslator.inline_call_
622
623 def patched_inline_call(self_: Any) -> Any:
624 code = self_.f_code
625 self.compilation_config.traced_files.add(code.co_filename)
626 return inline_call(self_)
627
628 # Disable the C++ compilation of symbolic shape guards. C++-fication
629 # of symbolic shape guards can improve guard overhead. But, since
630 # vllm skip guards anyways, setting this flag to False can improve
631 # compile time.
632 dynamo_config_patches = {}
633 try:
634 _ = torch._dynamo.config.enable_cpp_symbolic_shape_guards
635 dynamo_config_patches["enable_cpp_symbolic_shape_guards"] = False
636 except AttributeError:
637 # Note: this config is not available in torch 2.6, we can skip
638 # if the config doesn't exist
639 logger.debug("enable_cpp_symbolic_shape_guards config not available")
640
641 # Prepare backed_size_oblivious config patch if needed
642 fx_config_patches = {}
643 if ds_type == DynamicShapesType.BACKED_SIZE_OBLIVIOUS:
644 fx_config_patches["backed_size_oblivious"] = True
645
646 # Prepare inductor config patches
647 # assume_32bit_indexing is only available in torch 2.10.0+
648 inductor_config_patches = {}
649 if is_torch_equal_or_newer("2.10.0"):
650 inductor_config_patches["assume_32bit_indexing"] = (
651 self.compilation_config.dynamic_shapes_config.assume_32_bit_indexing
652 )
653
654 with (
655 patch.object(
656 InliningInstructionTranslator, "inline_call_", patched_inline_call
657 ),
658 torch._dynamo.config.patch(**dynamo_config_patches),
659 maybe_use_cudagraph_partition_wrapper(self.vllm_config),
660 torch.fx.experimental._config.patch(**fx_config_patches),
661 torch._inductor.config.patch(**inductor_config_patches),
662 ):
663 use_aot_compile = envs.VLLM_USE_AOT_COMPILE
664 if self.vllm_config.compilation_config.backend == "eager":
665 logger.warning("Detected eager backend, disabling AOT compile.")
666 use_aot_compile = False
667 if use_aot_compile:
668 # store the path for saving after warmup
669 self._aot_compilation_path = aot_compilation_path
670 self._aot_cache_dir = cache_dir
671 with monitor_torch_compile(
672 self.vllm_config, is_encoder=self._is_encoder
673 ):
674 self.aot_compiled_fn = self.aot_compile(*args, **kwargs)
675 compilation_counter.num_aot_compiles += 1
676 # All compilation is done at this point, save the
677 # AOT artifact.
678 self.save_aot_compiled_function()
679
680 with monitor_profiling_run():
681 output = self.aot_compiled_fn(self, *args, **kwargs)
682 else:
683 with monitor_torch_compile(
684 self.vllm_config,
685 "torch.compile and initial profiling/warmup "
686 "run together took %.2f s in total",
687 is_encoder=self._is_encoder,
688 ):
689 output = TorchCompileWithNoGuardsWrapper.__call__(
690 self, # type: ignore[arg-type]
691 *args,
692 **kwargs,
693 )
694
695 self.compiled = True
696 return output
697
698 # triggers VllmSerializableFunction.serialize()
699 def save_aot_compiled_function(self: type[_T]) -> None:
700 if envs.VLLM_DISABLE_COMPILE_CACHE:
701 return
702
703 if self.was_aot_compile_fn_loaded_from_disk:
704 logger.debug("AOT compiled function was loaded from cache, skipping save")
705 return
706
707 assert (
708 self.aot_compiled_fn and self._aot_compilation_path and self._aot_cache_dir
709 )
710
711 try:
712 os.makedirs(self._aot_cache_dir, exist_ok=True)
713 # File saving should be atomic, so we will save to a temporary location
714 # first. Should be upstreamed to PyTorch 2.12 as well.
715 tmp_file = f"{self._aot_compilation_path}.{os.getpid()}.tmp"
716 self.aot_compiled_fn.save_compiled_function(tmp_file)
717 os.replace(tmp_file, self._aot_compilation_path)
718 compilation_counter.num_aot_artifacts_saved += 1
719 logger.info_once(
720 "saved AOT compiled function to %s",
721 self._aot_compilation_path,
722 )
723 except Exception as e:
724 logger.warning(
725 "unable to save AOT compiled function to %s: %s",
726 self._aot_compilation_path,
727 e,
728 )
729
730 cls.__call__ = __call__
731 cls.save_aot_compiled_function = save_aot_compiled_function
732 return cls

735@contextlib.contextmanager
736def maybe_use_cudagraph_partition_wrapper(
737 vllm_config: VllmConfig,
738) -> Generator[None, None, None]:
739 """
740 Context manager to set/unset customized cudagraph partition wrappers.
741
742 If we're using Inductor-based graph partitioning, we currently have the
743 whole `fx.Graph` before Inductor lowering and the piecewise
744 splitting happens after all graph passes and fusions. Here, we add
745 a custom hook for Inductor to wrap each partition with our static
746 graph wrapper class to maintain more control over static graph
747 capture and replay.
748 """
749 from vllm.config import CUDAGraphMode
750
751 compilation_config = vllm_config.compilation_config
752 if (
753 compilation_config.cudagraph_mode.has_piecewise_cudagraphs()
754 and compilation_config.use_inductor_graph_partition
755 ):
756 from torch._inductor.utils import CUDAGraphWrapperMetadata
757
758 from vllm.compilation.cuda_graph import CUDAGraphOptions
759 from vllm.platforms import current_platform
760
761 static_graph_wrapper_class = resolve_obj_by_qualname(
762 current_platform.get_static_graph_wrapper_cls()
763 )
764
765 def customized_cudagraph_wrapper(
766 f: Callable[..., Any], metadata: CUDAGraphWrapperMetadata
767 ) -> Any:
768 partition_id = metadata.partition_index
769 num_partitions = metadata.num_partitions
770 return static_graph_wrapper_class(
771 runnable=f,
772 vllm_config=vllm_config,
773 runtime_mode=CUDAGraphMode.PIECEWISE,
774 cudagraph_options=CUDAGraphOptions(
775 debug_log_enable=partition_id == 0,
776 gc_disable=partition_id != 0,
777 weak_ref_output=partition_id == num_partitions - 1,
778 ),
779 )
780
781 torch._inductor.utils.set_customized_partition_wrappers(
782 customized_cudagraph_wrapper
783 )
784
785 yield
786
787 if (
788 compilation_config.cudagraph_mode.has_piecewise_cudagraphs()
789 and compilation_config.use_inductor_graph_partition
790 ):
791 torch._inductor.utils.set_customized_partition_wrappers(None)
