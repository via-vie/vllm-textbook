72 def __init__(

85 Initializes the SingleTypeKVCacheManager.
86 Args:
87 kv_cache_spec: The kv_cache_spec for this manager.
88 block_pool: The block pool.
89 kv_cache_group_id: The id of the kv cache group of this manager.
90 scheduler_block_size: The scheduling granularity (LCM of all group
91 block sizes); a multiple of this manager's ``block_size``.
92 needs_kv_cache_zeroing: Whether worker-side KV cache zeroing needs
93 newly allocated block IDs from this manager.
94 max_admission_blocks_per_request: Recycling-aware per-request
95 block cap used by `get_num_blocks_to_allocate`. Only set for
96 spec types that recycle blocks across chunks (SWA,
97 chunked-local); `None` (the default) means no cap, which is
98 correct for full-attention-style specs that hold every
99 block until the request finishes.
100 """

102 # Hybrid fine-grained lookup may lower this after all participating
103 # managers have been validated by the coordinator.
104 self.cache_hit_alignment_tokens = scheduler_block_size
105 # The block size for this manager; used for actual block allocation.
106 self.block_size = kv_cache_spec.block_size
107 self.dcp_world_size = dcp_world_size
108 self.pcp_world_size = pcp_world_size
109 if dcp_world_size > 1:

112 self.block_pool = block_pool
113 self.enable_caching = enable_caching
114 self._max_admission_blocks_per_request = max_admission_blocks_per_request

116 # consume them and this manager holds a spec type that gets zeroed.
117 self._record_new_block_ids = (
118 needs_kv_cache_zeroing
119 and isinstance(kv_cache_spec, AttentionSpec)
120 and not isinstance(kv_cache_spec, CircularBufferSpec)
121 )
122 self.new_block_ids: list[int] = []

125 # for each request, so that we can free the blocks when the request
126 # is finished.
127 self.req_to_blocks: defaultdict[str, list[KVCacheBlock]] = defaultdict(list)

129 # {req_id: The number of cached blocks for this given request}
130 # This is used to track the number of cached blocks for each request.
131 # This is only used to track the RUNNING requests, we do not track the
132 # data for preempted ones.

158 tuple[str, int, KVCacheBlock, int]

177 def get_num_blocks_to_allocate(
178 self,
179 request_id: str,
180 num_tokens: int,
181 new_computed_blocks: Sequence[KVCacheBlock],
182 total_computed_tokens: int,
183 num_local_computed_tokens: int,
184 num_tokens_main_model: int,
185 apply_admission_cap: bool = False,
186 ) -> int:
187 """
188 Get the number of blocks needed to be allocated for the request.
189
190 Args:
191 request_id: The request ID.
192 num_tokens: The total number of tokens that need a slot (including
193 tokens that are already allocated).
194 new_computed_blocks: The new computed blocks just hitting the
195 prefix caching.
196 total_computed_tokens: Include both local and external computed
197 tokens.
198 num_local_computed_tokens: The number of local prefix-cache computed
199 tokens.
200 num_tokens_main_model: The number of tokens for the main model (aka target
201 model in spec decode). w/o spec decode, it is num_tokens;
202 with spec decode, it is num_tokens - num_lookahead_tokens.
203 apply_admission_cap: If True, clamp by `num_required_blocks` by
204 `_max_admission_blocks_per_request`for recycling-aware specs
205 (SWA, chunked-local).
206
207 Returns:
208 The number of blocks to allocate.
209 """
210
211 num_required_blocks = cdiv(num_tokens, self.block_size)
212 if apply_admission_cap and self._max_admission_blocks_per_request is not None:
213 # Recycling-aware specs (SWA, chunked-local) cap the per-request
214 # reservation here so admission matches the startup pool sizer
215 # (`SlidingWindowSpec.max_admission_blocks_per_request` / its
216 # chunked-local counterpart). `remove_skipped_blocks` runs from
217 # `allocate_slots` before each chunk's `get_num_blocks_to_allocate`,
218 # so per-request peak real-held blocks <= this cap, which keeps
219 # `sum(reservations) <= pool` <=> `sum(peak_real_held) <= pool`.
220 # Drift between the two would re-introduce the deadlock from
221 # issue #39734 or, worse, mid-prefill OOM.
222 num_required_blocks = min(
223 num_required_blocks, self._max_admission_blocks_per_request
224 )
225 num_req_blocks = len(self.req_to_blocks.get(request_id, ()))
226
227 if request_id in self.num_cached_block:
228 # Fast-path: a running request won't have any new prefix-cache hits.
229 assert len(new_computed_blocks) == 0
230 # NOTE: With speculative decoding, request's blocks may be allocated
231 # for draft tokens which are later rejected. In this case,
232 # num_required_blocks may be smaller than num_req_blocks.
233 return max(num_required_blocks - num_req_blocks, 0)
234
235 num_skipped_tokens = self.get_num_skipped_tokens(total_computed_tokens)
236 num_local_computed_blocks = len(new_computed_blocks) + num_req_blocks
237 # Number of whole blocks that are skipped by the attention window.
238 # If nothing is skipped, this is 0.
239 num_skipped_blocks = num_skipped_tokens // self.block_size
240 # We need blocks for the non-skipped suffix. If there are still
241 # local-computed blocks inside the window, they contribute to the
242 # required capacity; otherwise, skipped blocks dominate.
243 num_new_blocks = max(
244 num_required_blocks - max(num_skipped_blocks, num_local_computed_blocks),
245 0,
246 )
247
248 # Among the `new_computed_blocks`, the first `num_skipped_blocks` worth
249 # of blocks are skipped; `num_req_blocks` of those may already be in
250 # `req_to_blocks`, so only skip the remainder from `new_computed_blocks`.
251 num_skipped_new_computed_blocks = max(0, num_skipped_blocks - num_req_blocks)
252
253 # If a computed block is an eviction candidate (in the free queue and
254 # ref_cnt == 0), it will be removed from the free queue when touched by
255 # the allocated request, so we must count it in the free-capacity check.
256 num_evictable_blocks = self._get_num_evictable_blocks(
257 new_computed_blocks[num_skipped_new_computed_blocks:]
258 )
259 if self._has_partial_local_hit(new_computed_blocks, num_local_computed_tokens):
260 # Reserve the extra block that allocate_new_blocks pulls for the
261 # partial-hit CoW redirect.
262 num_new_blocks += 1
263 return num_new_blocks + num_evictable_blocks

265 def add_local_computed_blocks(
266 self,
267 request_id: str,
268 new_computed_blocks: Sequence[KVCacheBlock],
269 num_local_computed_tokens: int,
270 num_external_computed_tokens: int,
271 ) -> None:
272 """
273 Add the locally cached (prefix-hit) blocks to the request:
274 1. Touch the computed blocks (paired with adding them to `req_blocks`)
275 so their ref_cnt exactly tracks the referencing requests.
276 1.5. (Optional) For sliding window, skipped blocks are padded with nulls.
277 2. Add the remaining computed blocks.
278
279 Args:
280 request_id: The request ID.
281 new_computed_blocks: The new computed blocks just hitting the
282 prefix cache.
283 num_local_computed_tokens: The number of local computed tokens.
284 num_external_computed_tokens: The number of external computed tokens.
285 """
286 # The coordinator only calls this for first-time allocations (running
287 # requests are short-circuited there), so the request has no blocks yet.
288 req_blocks = self.req_to_blocks[request_id]
289 assert len(req_blocks) == 0
290 num_total_computed_tokens = (
291 num_local_computed_tokens + num_external_computed_tokens
292 )
293 num_skipped_tokens = self.get_num_skipped_tokens(num_total_computed_tokens)
294 num_skipped_blocks = num_skipped_tokens // self.block_size
295 if num_skipped_blocks > 0:
296 # It is possible that all new computed blocks are skipped when
297 # num_skipped_blocks > len(new_computed_blocks).
298 new_computed_blocks = new_computed_blocks[num_skipped_blocks:]
299
300 # Touch the computed blocks to make sure they won't be evicted.
301 if self.enable_caching:
302 self.block_pool.touch(new_computed_blocks)
303 else:
304 assert not any(new_computed_blocks), (
305 "Computed blocks should be empty when prefix caching is disabled"
306 )
307
308 # Skip blocks are padded with null blocks.
309 req_blocks.extend([self._null_block] * num_skipped_blocks)
310 # Add the remaining computed blocks.
311 req_blocks.extend(new_computed_blocks)
312 # All cached hits (including skipped nulls) are already cached; mark
313 # them so cache_blocks() will not try to re-cache blocks that already
314 # have a block_hash set.
315 self.num_cached_block[request_id] = len(req_blocks)
316 if self._has_partial_local_hit(new_computed_blocks, num_local_computed_tokens):
317 # Record the partial tail for the CoW redirect in
318 # allocate_new_blocks; cap the cached count at the full blocks so
319 # cache_blocks() re-caches the private copy once full.
320 block_idx = num_local_computed_tokens // self.block_size
321 self._partial_hit_reqs[request_id] = (block_idx, new_computed_blocks[-1])
322 self.num_cached_block[request_id] = block_idx

324 def allocate_external_computed_blocks(
325 self,
326 request_id: str,
327 num_local_computed_tokens: int,
328 num_external_computed_tokens: int,
329 ) -> None:
330 """
331 Allocate new blocks for external (KV-connector) computed tokens.
332
333 Must run only after every group's local blocks have been touched via
334 `add_local_computed_blocks`, so this group's `get_new_blocks` cannot
335 evict another group's cache-hit blocks (issue #33775).
336
337 Args:
338 request_id: The request ID.
339 num_local_computed_tokens: The number of local computed tokens.
340 num_external_computed_tokens: The number of external computed tokens.
341 """
342 num_total_computed_tokens = (
343 num_local_computed_tokens + num_external_computed_tokens
344 )
345 num_skipped_tokens = self.get_num_skipped_tokens(num_total_computed_tokens)
346 if num_skipped_tokens > 0:
347 # Some external computed tokens may be skipped too.
348 num_external_computed_tokens = min(
349 num_total_computed_tokens - num_skipped_tokens,
350 num_external_computed_tokens,
351 )
352 if num_external_computed_tokens <= 0:
353 return
354
355 req_blocks = self.req_to_blocks[request_id]
356 num_new_blocks = max(
357 0, cdiv(num_total_computed_tokens, self.block_size) - len(req_blocks)
358 )
359 allocated_blocks = self.block_pool.get_new_blocks(num_new_blocks)
360 req_blocks.extend(allocated_blocks)
361 if self._record_new_block_ids:
362 self.new_block_ids.extend(b.block_id for b in allocated_blocks)

364 def allocate_new_blocks(
365 self, request_id: str, num_tokens: int, num_tokens_main_model: int
366 ) -> list[KVCacheBlock]:
367 """
368 Allocate new blocks for the request to give it at least `num_tokens`
369 token slots.
370
371 Args:
372 request_id: The request ID.
373 num_tokens: The total number of tokens that need a slot (including
374 tokens that are already allocated).
375 num_tokens_main_model: The number of tokens for the main model (aka target
376 model in spec decode). w/o spec decode, it is num_tokens;
377 with spec decode, it is num_tokens - num_lookahead_tokens.
378 Returns:
379 The new allocated blocks.
380 """
381 cow_blocks: list[KVCacheBlock] = []
382 if request_id in self._partial_hit_reqs:
383 # Partial hit: redirect the shared tail to a private CoW block.
384 # Replacing in place keeps the length-based allocation below
385 # correct; the extra block was reserved by
386 # get_num_blocks_to_allocate.
387 block_idx, source_block = self._partial_hit_reqs.pop(request_id)
388 cow_block = self.block_pool.get_new_blocks(1)[0]
389 self._apply_cow(request_id, block_idx, source_block, cow_block)
390 self.new_block_ids.append(cow_block.block_id)
391 cow_blocks.append(cow_block)
392
393 req_blocks = self.req_to_blocks[request_id]
394 num_required_blocks = cdiv(num_tokens, self.block_size)
395 num_new_blocks = num_required_blocks - len(req_blocks)
396 if num_new_blocks <= 0:
397 return cow_blocks
398 else:
399 new_blocks = self.block_pool.get_new_blocks(num_new_blocks)
400 req_blocks.extend(new_blocks)
401 if self._record_new_block_ids:
402 self.new_block_ids.extend(b.block_id for b in new_blocks)
403 return cow_blocks + new_blocks

448 def _apply_cow(
449 self,
450 request_id: str,
451 block_idx: int,
452 source_block: KVCacheBlock,
453 cow_block: KVCacheBlock,
454 ) -> None:
455 """Redirect a partial prefix-cache hit to a private CoW block.
456
457 Both copy endpoints stay retained until the copy has run on the worker,
458 so a same-step free cannot recycle them: ``source_block`` keeps its
459 hit-ref, ``cow_block`` takes an extra ref beyond the one handed to the
460 request.
461 """
462 req_blocks = self.req_to_blocks[request_id]
463 assert block_idx < len(req_blocks)
464 assert req_blocks[block_idx] is source_block
465 assert not source_block.is_null and source_block.ref_cnt > 0
466 req_blocks[block_idx] = cow_block
467 self._pending_cow_copies.append((source_block, cow_block))
468 cow_block.ref_cnt += 1

470 def cache_blocks(
471 self,
472 request: Request,
473 num_tokens: int,
474 retention_interval: int | None = None,
475 *,
476 replay_boundaries: Sequence[int],
477 ) -> None:
478 """
479 Cache the blocks for the request.
480
481 Args:
482 request: The request.
483 num_tokens: The total number of tokens that need to be cached
484 (including tokens that are already cached).
485 retention_interval: Sparse local-checkpoint granularity. ``None``
486 keeps dense checkpointing; ``0`` keeps only the latest replay
487 boundary; a positive multiple of ``scheduler_block_size`` keeps
488 a tail once per that-sized segment. Only SWA acts on it.
489 replay_boundaries: Positions a later request replaying this prompt
490 can resume at, from ``get_replay_boundaries``.
491 """
492 num_cached_blocks = self.num_cached_block.get(request.request_id, 0)
493 num_full_blocks = num_tokens // self.block_size
494
495 if num_cached_blocks >= num_full_blocks:
496 return
497
498 # Token boundaries whose reachable tail must be retained under sparse
499 # retention: every position a replaying sibling can resume at (see
500 # ``get_replay_boundaries``) and any detected shared-prefix junction.
501 reachable_boundaries = [*replay_boundaries]
502 if request.shared_prefix_boundary:
503 reachable_boundaries.append(request.shared_prefix_boundary)
504
505 block_mask = self.reachable_block_mask(
506 start_block=num_cached_blocks,
507 end_block=num_full_blocks,
508 alignment_tokens=self.cache_hit_alignment_tokens,
509 kv_cache_spec=self.kv_cache_spec,
510 use_eagle=self.use_eagle,
511 retention_interval=retention_interval,
512 reachable_boundaries=reachable_boundaries,
513 dcp_world_size=self.dcp_world_size,
514 )
515 self.block_pool.cache_full_blocks(
516 request=request,
517 blocks=self.req_to_blocks[request.request_id],
518 num_cached_blocks=num_cached_blocks,
519 num_full_blocks=num_full_blocks,
520 block_size=self.block_size,
521 kv_cache_group_id=self.kv_cache_group_id,
522 block_mask=block_mask,
523 )
524
525 self.num_cached_block[request.request_id] = num_full_blocks

528 def reachable_block_mask(
529 cls,

531 end_block: int,

533 kv_cache_spec: KVCacheSpec,

545 ``reachable_boundaries`` are token positions whose reachable tail must be
546 retained; the base (dense) policy ignores them.
547 ``final_segment_end_block`` is the exclusive end of the request's final
548 segment. It may be later than ``end_block`` while prefill is still in
549 progress. Non-EAGLE sparse managers may keep its reachable tail. The
550 base policy already keeps every block, so it ignores this value.
551 """
552 return None

554 def pop_blocks_for_free(self, request_id: str) -> list[KVCacheBlock]:
555 """
556 Pop the request's bookkeeping and return its blocks without yet
557 returning them to the block pool. The caller is responsible for
558 eventually passing the returned blocks to `block_pool.free_blocks`,
559 freeing them in reverse order (so that tail blocks are evicted first).
560
561 Args:
562 request_id: The request ID.
563
564 Returns:
565 The request's blocks in allocation order.
566 """
567 # Default to [] in case a request is freed (aborted) before alloc.
568 req_blocks = self.req_to_blocks.pop(request_id, [])
569 self.num_cached_block.pop(request_id, None)
570 self._partial_hit_reqs.pop(request_id, None)
571 return req_blocks

615 `max_length`. The prefix should be a common prefix hit for all the
616 kv cache groups in `kv_cache_group_ids`. If no cache hit is found,
617 return an empty list.
618 If eagle is enabled, drop the last matched block to force recompute the
619 last block to get the required hidden states for eagle drafting head.
620 For multi-module MTP, this recompute also rewrites the dropped block's
621 draft-layer KVs, which depend on up to num_speculative_tokens - 1
622 tokens past the matched prefix (i.e. on the cache writer's
623 continuation, which the block hash does not cover); the coordinator
624 asserts the block size covers that window.
625 Need to be customized for each attention type.
626
627 Args:
628 block_hashes: The block hashes of the request.
629 max_length: The maximum length of the cache hit prefix.
630 kv_cache_group_ids: The ids of the kv cache groups.
631 block_pool: The block pool.
632 kv_cache_spec: The kv cache spec.
633 drop_eagle_block: Whether to drop the last matched block for EAGLE/MTP.
634 Always False for non-EAGLE/MTP groups, but can be False for EAGLE/MTP
635 groups too if the last block is already dropped (e.g., in a
636 convergence loop in `find_longest_cache_hit`).
637 alignment_tokens: The returned cache hit length (in tokens) should
638 be a multiple of this value (in tokens). By default, it should
639 be set to the block_size.
640 dcp_world_size: The world size of decode context parallelism.
641 pcp_world_size: The world size of prefill context parallelism.
642
643 Returns:
644 A tuple containing cached blocks and the exact cache-hit length in
645 tokens. The cached block tuple has skipped blocks replaced by null
646 blocks for each kv cache group in `kv_cache_group_ids`.
647 For example, sliding window manager should return a list like
648 ([NULL, NULL, KVCacheBlock(7), KVCacheBlock(8)]) for block size 4
649 and sliding window 8 and len(kv_cache_group_ids) = 1.
650 """

654 def _remove_blocks_in_range(
655 self,
656 request_id: str,
657 first_block: int,
658 last_block: int,
659 ) -> None:
660 """Free blocks in ``[first_block, last_block)`` and replace with null_block.
661
662 Iterates backward so newly-evictable tail blocks are reached even after
663 earlier blocks in the range were nulled in a prior call.
664 """
665 if request_id not in self.req_to_blocks:
666 return
667 if first_block >= last_block:
668 return
669 blocks = self.req_to_blocks[request_id]
670 last_block = min(last_block, len(blocks))
671
672 freed: list[KVCacheBlock] = []
673 for i in range(last_block - 1, first_block - 1, -1):
674 if blocks[i] == self._null_block:
675 break
676 freed.append(blocks[i])
677 blocks[i] = self._null_block
678 if freed:
679 self.block_pool.free_blocks(freed)

788 [] for _ in range(len(kv_cache_group_ids))
789 )
790 # Phase 1: longest run of cached full blocks from the start. A missing
791 # block implies every later block misses too (chained hashes).
792 for block_hash in itertools.islice(full_block_hashes, max_length // block_size):
793 cached_block = block_pool.get_cached_block(block_hash, kv_cache_group_ids)
794 if not cached_block:
795 break
796 for computed, cached in zip(computed_blocks, cached_block):
797 computed.append(cached)
798 hit_length = len(computed_blocks[0]) * block_size
799
800 # Phase 2 (fine-grained only): extend into the first non-full block by
801 # probing its interior hash boundaries high-to-low (longest hit first).
802 if fine_grained:

804 assert isinstance(block_hashes, Sequence)

808 first_partial_idx + scale_factor - 1,
809 max_length // alignment_tokens,
810 len(block_hashes),
811 )

815 )

819 computed.append(cached)
820 hit_length = (fine_idx + 1) * alignment_tokens
821 break
822

824 # drop one hash unit when fine-grained (the tail block's KV is
825 # append-only, so it still covers the reduced length), else one cache
826 # block.
827 if drop_eagle_block and hit_length > 0:
828 hit_length -= min(alignment_tokens, block_size)
829 # Round down to the alignment; a no-op when fine-grained (hits land on
830 # hash boundaries by construction) and when alignment_tokens ==
831 # block_size. Then trim blocks past the new tail.
832 hit_length -= hit_length % alignment_tokens
833 num_blocks = cdiv(hit_length, block_size)
834 for computed in computed_blocks:
835 del computed[num_blocks:]
836 return computed_blocks, hit_length

918

920 [ceil(prefix_len / block_size) * block_size,

953 @classmethod
954 def _contiguous_blocks_for_hit(
955 cls, window_size: int, block_size: int, use_eagle: bool
956 ) -> int:
957 blocks = cdiv(window_size - 1, block_size)
958 if use_eagle:
959 # Need to drop the last matched block if eagle is enabled. For
960 # sliding window layer, we achieve this by increasing the number of
961 # contiguous blocks needed for prefix cache hit by one and dropping
962 # the last matched block.
963 blocks += 1

966 @classmethod
967 def find_longest_cache_hit(
968 cls,
969 block_hashes: BlockHashList,
970 max_length: int,
971 kv_cache_group_ids: list[int],
972 block_pool: BlockPool,
973 kv_cache_spec: KVCacheSpec,
974 drop_eagle_block: bool,
975 alignment_tokens: int,
976 dcp_world_size: int = 1,
977 pcp_world_size: int = 1,
978 ) -> tuple[tuple[list[KVCacheBlock], ...], int]:
979 assert isinstance(kv_cache_spec, SlidingWindowSpec), (
980 "SlidingWindowManager can only be used for sliding window groups"
981 )
982 assert dcp_world_size == 1, "DCP not support sliding window attn now."
983 assert pcp_world_size == 1, "PCP not support sliding window attn now."
984 # Sliding-window cache hits must stay at the group's physical block
985 # granularity. resolve_block_hashes() converts finer-grained hashes to
986 # that view when the hybrid-cache alignment is smaller than block_size.
987 block_hashes = resolve_block_hashes(
988 block_hashes,
989 block_pool.hash_block_size,
990 kv_cache_spec.block_size,
991 supports_fine_grained_hash_lookup=cls.supports_fine_grained_hash_lookup,
992 alignment_tokens=alignment_tokens,
993 )
994
995 # The number of contiguous blocks needed for a prefix cache hit.
996 sliding_window_contiguous_blocks = cls._contiguous_blocks_for_hit(
997 kv_cache_spec.sliding_window, kv_cache_spec.block_size, drop_eagle_block
998 )
999
1000 # TODO: reduce i by sliding_window_contiguous_blocks when cache miss, to
1001 # optimize the time complexity from O(max_num_blocks) to
1002 # O(max_num_blocks / sliding_window_contiguous_blocks +
1003 # sliding_window_contiguous_blocks),
1004 # which is good for low cache hit rate scenarios.
1005 max_num_blocks = max_length // kv_cache_spec.block_size
1006 computed_blocks: tuple[list[KVCacheBlock], ...] = tuple(
1007 [block_pool.null_block] * max_num_blocks
1008 for _ in range(len(kv_cache_group_ids))
1009 )
1010 block_size = kv_cache_spec.block_size
1011 num_contiguous_blocks = 0
1012 match_found = False
1013 # Search from right to left and early stop when a match is found.
1014 for i in range(max_num_blocks - 1, -1, -1):
1015 if cached_block := block_pool.get_cached_block(
1016 block_hashes[i], kv_cache_group_ids
1017 ):
1018 # Skip prefix matching check if the block is not aligned with
1019 # `alignment_tokens`.
1020 if num_contiguous_blocks == 0 and block_size != alignment_tokens:
1021 post_pop_blocks = i if drop_eagle_block else i + 1
1022 if (post_pop_blocks * block_size) % alignment_tokens != 0:
1023 continue
1024 # Add the cached block to the computed blocks.
1025 for computed, cached in zip(computed_blocks, cached_block):
1026 computed[i] = cached
1027 num_contiguous_blocks += 1
1028 if num_contiguous_blocks >= sliding_window_contiguous_blocks:
1029 # Trim the trailing blocks.
1030 # E.g., [NULL, NULL, 8, 3, NULL, 9] -> [NULL, NULL, 8, 3]
1031 # when sliding_window_contiguous_blocks=2.
1032 for computed in computed_blocks:
1033 del computed[i + num_contiguous_blocks :]
1034 match_found = True
1035 break
1036 else:
1037 num_contiguous_blocks = 0
1038 if not match_found:
1039 # The first `num_contiguous_blocks` is a cache hit even if
1040 # `num_contiguous_blocks < sliding_window_contiguous_blocks`.
1041 for computed in computed_blocks:
1042 del computed[num_contiguous_blocks:]
1043 while (
1044 block_size != alignment_tokens # Faster for common case.
1045 and len(computed_blocks[0]) * block_size % alignment_tokens != 0
1046 ):
1047 for computed in computed_blocks:
1048 computed.pop()
1049 if drop_eagle_block and computed_blocks[0]:
1050 for computed in computed_blocks:
1051 computed.pop()
1052 # Re-align after eagle pop: the pop may break the alignment
1053 # when block_size != alignment_tokens (hybrid models with
1054 # different page sizes, e.g. Gemma4).
1055 while (
1056 block_size != alignment_tokens
1057 and len(computed_blocks[0]) * block_size % alignment_tokens != 0
1058 ):
1059 for computed in computed_blocks:
1060 computed.pop()
1061 hit_length = len(computed_blocks[0]) * block_size
1062 return computed_blocks, hit_length

1064 @classmethod
1065 def reachable_block_mask(
1066 cls,
1067 start_block: int,
1068 end_block: int,
1069 alignment_tokens: int | None,
1070 kv_cache_spec: KVCacheSpec,
1071 use_eagle: bool,
1072 retention_interval: int | None = None,
1073 reachable_boundaries: Sequence[int] = (),
1074 dcp_world_size: int = 1,
1075 final_segment_end_block: int | None = None,
1076 ) -> list[bool] | None:
1077 assert isinstance(kv_cache_spec, SlidingWindowSpec)
1078 if alignment_tokens is None:
1079 # Fast path: when the coordinator imposes no alignment constraint.
1080 return None
1081 block_size = kv_cache_spec.block_size * dcp_world_size
1082 if alignment_tokens % block_size != 0:
1083 # The mask is block-granular, so a sub-block alignment cannot be
1084 # represented exactly. This happens for hybrid offloading, where
1085 # ``alignment_tokens`` is the full-attention chunk size and need not
1086 # be a multiple of this SWA group's (DCP-scaled) block size (e.g.
1087 # Gemma). Fall back to dense: every block is reachable, which never
1088 # drops a block that could serve a hit.
1089 return None
1090
1091 # Contiguous blocks a hit needs at a boundary (incl. the EAGLE peek).
1092 need = cls._contiguous_blocks_for_hit(
1093 window_size=kv_cache_spec.sliding_window,
1094 block_size=block_size,
1095 use_eagle=use_eagle,
1096 )
1097 # The matched run's right edge sits on the aligned boundary block when
1098 # EAGLE peeks one block past it (shift=1), otherwise on the last block
1099 # before the boundary (shift=0).
1100 shift = 1 if use_eagle else 0
1101
1102 mask = [False] * (end_block - start_block)
1103
1104 # (1) Segment-boundary tails. ``retention_interval``:
1105 # None -> dense (a tail at every ``alignment_tokens`` boundary);
1106 # 0 -> no dense tails (only the replay boundary below);
1107 # >0 -> a tail once per ``retention_interval``-sized segment.
1108 segment_tokens = (
1109 alignment_tokens
1110 if retention_interval is None
1111 else (None if retention_interval == 0 else retention_interval)
1112 )
1113 if segment_tokens is not None:
1114 per_segment = segment_tokens // block_size
1115 if need >= per_segment:
1116 # Every block is reachable; cache them all.
1117 return None
1118 for i in range(start_block, end_block):
1119 if i >= shift and (i - shift) % per_segment >= per_segment - need:
1120 mask[i - start_block] = True
1121
1122 if final_segment_end_block is not None and not use_eagle:
1123 final_segment_size = final_segment_end_block % per_segment
1124 if final_segment_size:
1125 final_segment_start = final_segment_end_block - final_segment_size
1126 final_tail_start = max(
1127 final_segment_start, final_segment_end_block - need
1128 )
1129 final_tail_end = min(end_block, final_segment_end_block)
1130 for i in range(max(start_block, final_tail_start), final_tail_end):
1131 mask[i - start_block] = True
1132
1133 # (2) Reachable-boundary tails: the replay boundary (``num_prompt - 1``,
1134 # capped by ``get_computed_blocks``) and any shared-prefix junction. Both
1135 # land before segments would cover them under sparse retention, so keep
1136 # the ``need``-block tail ending on each boundary explicitly.
1137 if retention_interval is not None:
1138 for boundary_tokens in reachable_boundaries:
1139 aligned = boundary_tokens // alignment_tokens * alignment_tokens
1140 end = aligned // block_size + shift
1141 for j in range(max(start_block, end - need), min(end_block, end)):
1142 mask[j - start_block] = True
1143
1144 return mask

1149

1152
1153 Example:
1154 sliding_window=4, num_computed_tokens=7
1155
1156 Tokens: [ 0 1 2 3 4 5 6 7 ]
1157 | ---- computed -----|
1158 ^ next token to be computed
1159 |-----------| sliding window for next token
1160 |--skipped---|
1161

1163 attention computation since they are outside the sliding window.

1173 num_computed_tokens: The number of tokens that have been computed.
1174
1175 Returns:
1176 The number of tokens that will be skipped for attention computation.
1177 """
1178 return max(
1179 0,
1180 num_computed_tokens - self.sliding_window + 1 - self.extra_retained_tokens,
1181 )

1583 ``0`` -> keep only the ``reachable_boundaries`` states
1584 ``> 0`` -> keep one state per ``retention_interval``-sized segment

1841 and not has_partial_hit
1842 and not checkpoint_block

1852 # (1 + num_speculative_blocks) block

1854 prev_block_len - 1 - self.num_speculative_blocks
1855 )

1858 # saves the hit state.

2081 request.num_prompt_tokens // hash_block_size

2492 def get_num_blocks_to_allocate(
2493 self,
2494 request_id: str,
2495 num_tokens: int,
2496 new_computed_blocks: Sequence[KVCacheBlock],

2498 num_local_computed_tokens: int,
2499 num_tokens_main_model: int,
2500 apply_admission_cap: bool = False,
2501 ) -> int:
2502 del num_tokens_main_model
2503 assert not new_computed_blocks
2504 if total_computed_tokens > num_local_computed_tokens:
2505 # Keep the last imported page writable, including the last-token
2506 # replay on a full prompt hit. Only earlier pages stay host-only.
2507 imported_pages = cdiv(total_computed_tokens, self.block_size)
2508 existing = len(self.req_to_blocks.get(request_id, ()))
2509 needs_tail = self.get_resident_page(request_id, imported_pages - 1) is None
2510 return int(needs_tail) + max(
2511 cdiv(num_tokens, self.block_size) - max(imported_pages, existing), 0
2512 )
2513 existing = len(self.req_to_blocks.get(request_id, ()))
2514 host_pages = cdiv(num_local_computed_tokens, self.block_size)
2515 required = cdiv(num_tokens, self.block_size)
2516 if apply_admission_cap:
2517 assert self._max_admission_blocks_per_request is not None
2518 required = min(required, self._max_admission_blocks_per_request)
2519 return max(required - max(existing, host_pages), 0)

2522 self,
2523 request_id: str,
2524 num_local_computed_tokens: int,
2525 num_external_computed_tokens: int,
2526 ) -> None:

2528 assert num_external_computed_tokens > 0

2600def get_manager_for_kv_cache_spec(
2601 kv_cache_spec: KVCacheSpec,
2602 max_in_flight_tokens: int,
2603 max_model_len: int,
2604 role: str | None = None,
2605 **kwargs,
2606) -> SingleTypeKVCacheManager:
2607 """
2608 Get the appropriate manager for a given KVCacheSpec.
2609
2610 Uses the KVCacheSpecRegistry to look up the manager class, supporting
2611 both built-in and custom specs registered via @register_kv_cache_spec
2612 and KVCacheSpecRegistry.register.
2613
2614 Args:
2615 kv_cache_spec: The KVCacheSpec instance
2616 max_in_flight_tokens: The max tokens scheduled but not yet settled
2617 (one batch per concurrent step); see `VllmConfig.max_in_flight_tokens`
2618 max_model_len: The maximum context length the model could serve
2619 Returns:
2620 An instance of the appropriate SingleTypeKVCacheManager subclass
2621 """
2622 manager_class = KVCacheSpecRegistry.get_manager_class(kv_cache_spec, role)
2623 assert manager_class is not None, (
2624 f"No manager registered for KVCacheSpec {type(kv_cache_spec)}"
2625 )
2626 # SlidingWindow / ChunkedLocalAttention managers recycle blocks;
2627 # the runtime admission cap must match the recycling-aware bound the
2628 # startup pool sizer uses (single source of truth: the spec method).
2629 # R-SWA also recycles gap blocks but peak physical KV still fits the
2630 # full-attention bound (prefix + window <= max_model_len), so it inherits
2631 # FullAttentionSpec sizing without a separate admission cap.
2632 if isinstance(
2633 kv_cache_spec,
2634 (SlidingWindowSpec, ChunkedLocalAttentionSpec, HiSparseResidentSpec),
2635 ):
2636 kwargs["max_admission_blocks_per_request"] = (
2637 kv_cache_spec.max_admission_blocks_per_request(
2638 max_in_flight_tokens=max_in_flight_tokens,
2639 max_model_len=max_model_len,
2640 )
2641 )
2642 manager = manager_class(kv_cache_spec, **kwargs)
2643 return manager
