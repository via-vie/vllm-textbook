24class PauseState(enum.IntEnum):
25 """Scheduler pause state.
26
27 - UNPAUSED: Normal operation
28 - PAUSE_NEW: No new requests are scheduled, requests already in
29 running state are scheduled.
30 - PAUSE_ALL: No requests are scheduled
31 """
32
33 UNPAUSED = 0
34 PAUSED_NEW = 1
35 PAUSED_ALL = 2

39 @abstractmethod
40 def __init__(

45 block_size: int,

50 ) -> None:
51 raise NotImplementedError

53 @abstractmethod
54 def schedule(self, throttle_prefills: bool = False) -> "SchedulerOutput":
55 """Schedule the requests to process in this scheduling step.
56
57 The scheduling decision is made at the iteration level. Each scheduling
58 step corresponds to a single forward pass of the model. Therefore, this
59 method is called repeatedly by a busy loop in the engine.
60
61 Essentially, the scheduler produces a dictionary of {req_id: num_tokens}
62 that specifies how many tokens to process for each request in this
63 scheduling step. For example, num_tokens can be as large as the number
64 of prompt tokens for new requests, or it can be 1 for the requests that
65 are auto-regressively generating new tokens one by one. Otherwise, it
66 can be somewhere in between in case of chunked prefills, prefix caching,
67 speculative decoding, etc.
68
69 Additionally, the scheduler also returns useful data about each request
70 or the batch as a whole. The model runner will use this information in
71 preparing inputs to the model.
72
73 Args:
74 throttle_prefills: DP prefill balancing. When True (set by the DP
75 engine core on non-cadence-aligned steps), new prefill compute is
76 deferred to a later step so prefills stay aligned across DP ranks;
77 automatically overridden when the rank is saturated.
78
79 Returns:
80 A SchedulerOutput object containing information about the scheduled
81 requests.
82 """
83 raise NotImplementedError

98

102 update its states, checks the finished requests, and returns the output

104
105 Returns:

107 outputs for each request originating from that client.
108 """
109 raise NotImplementedError

144 @abstractmethod
145 def finish_requests(
146 self,
147 request_ids: str | Iterable[str] | None,
148 finished_status: "RequestStatus",
149 ) -> "list[Request]":
150 """Finish the requests in the scheduler's internal queue. If the request
151 is not in the queue, this method will do nothing for that request.
152
153 This method is called in two cases:
154 1. When the request is aborted by the client.
155 2. When the frontend process detects a stop string of the request after
156 de-tokenizing its generated tokens.
157
158 Args:
159 request_ids: A single or a list of request IDs, or None to finish all.
160 finished_status: The finished status of the given requests.
161
162 Returns:
163 List of requests that were aborted. Will not include any that were
164 already finished.
165 """
166 raise NotImplementedError
167
168 @abstractmethod
169 def get_num_unfinished_requests(self) -> int:
170 """Number of unfinished requests in the scheduler's internal queue."""
171 raise NotImplementedError
172
173 def has_unfinished_requests(self) -> bool:
174 """Returns True if there are unfinished requests in the scheduler's
175 internal queue."""
176 return self.get_num_unfinished_requests() > 0
177
178 @abstractmethod
179 def has_finished_requests(self) -> bool:
180 """Returns True if there are finished requests that need to be cleared.
181 NOTE: This is different from `not self.has_unfinished_requests()`.
182
183 The scheduler maintains an internal list of the requests finished in the
184 previous step. This list is returned from the next call to schedule(),
185 to be sent to the model runner in the next step to clear cached states
186 for these finished requests.
187
188 This method checks if this internal list of finished requests is
189 non-empty. This information is useful for DP attention.
190 """
191 raise NotImplementedError
192
193 def has_requests(self) -> bool:
194 """Returns True if there are unfinished requests, or finished requests
195 not yet returned in SchedulerOutputs."""
196 return self.has_unfinished_requests() or self.has_finished_requests()
