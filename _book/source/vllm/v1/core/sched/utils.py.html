10def _has_repeating_pattern(
11 token_ids: Sequence[int],
12 pattern_len: int,
13 repetition_min_count: int,
14) -> bool:
15 """Check if the tail of token_ids contains a repeating pattern.
16
17 Compares the last pattern_len tokens against the preceding
18 (repetition_min_count - 1) repetitions of the same length.
19 """
20 for n in range(1, pattern_len + 1):
21 target_token = token_ids[-n]
22 for m in range(1, repetition_min_count):
23 if token_ids[-(pattern_len * m + n)] != target_token:
24 return False
25 return True

28def check_sequence_repetition(
29 token_ids: Sequence[int],
30 params: RepetitionDetectionParams,
31) -> bool:
32 """Check if a sequence of token IDs has a repetition pattern.
33 Args:
34 token_ids: List of token IDs
35 params: Repetition detection parameters.
36 Returns:
37 True if a repetition pattern is found, False otherwise.
38 """
39 max_pattern_size = params.max_pattern_size
40 min_pattern_size = params.min_pattern_size
41 min_count = params.min_count
42
43 if min_pattern_size <= 0:
44 min_pattern_size = 1
45
46 if max_pattern_size <= 0 or min_count < 2 or min_pattern_size > max_pattern_size:
47 return False
48
49 for pattern_len in range(
50 min_pattern_size,
51 max_pattern_size + 1,
52 ):
53 if pattern_len * min_count > len(token_ids):
54 return False
55
56 if _has_repeating_pattern(token_ids, pattern_len, min_count):
57 return True
58
59 return False

62def remove_all(lst: list, items_to_remove: set) -> list:
63 """Remove all items from a list that are in the items_to_remove set.
64
65 This method optimizes for the common case of removing a single item,
66 falling back to list comprehension for multiple items.
67
68 Args:
69 lst: The list to remove items from
70 items_to_remove: Set of items to remove
71
72 Returns:
73 Either the modified original list (for single item removal) or
74 a new list (for multiple item removal). Callers should use the
75 returned value.
76
77 Note:
78 For single item removal, this modifies the original list in-place
79 and returns it. For multiple items, it creates and returns a new list.
80 """
81 if not items_to_remove:
82 return lst
83
84 if len(items_to_remove) == 1:
85 # Fast path for single item removal (most common case)
86 item = next(iter(items_to_remove))
87 with contextlib.suppress(ValueError):
88 lst.remove(item)
89 return lst
90 # For multiple items, use list comprehension
91 return [item for item in lst if item not in items_to_remove]

94def check_stop(request: Request, max_model_len: int) -> bool:
95 assert not request.pooling_params
96
97 sampling_params = request.sampling_params
98 assert sampling_params is not None
99
100 last_token_id = request.output_token_ids[-1]
101 if last_token_id == sampling_params.eos_token_id:
102 request.status = RequestStatus.FINISHED_STOPPED
103 return True
104
105 if last_token_id in (sampling_params.stop_token_ids or ()):
106 request.status = RequestStatus.FINISHED_STOPPED
107 request.stop_reason = last_token_id
108 return True
109
110 if (
111 request.num_tokens >= max_model_len
112 or request.num_output_tokens >= request.max_tokens
113 ):
114 request.status = RequestStatus.FINISHED_LENGTH_CAPPED
115 return True
116
117 # Note(arpera):
118 # Order of checks is important for min_tokens
119 # If you decide to change the order, first look at these PRs please:
120 # 47489, 49521, and 51299
121 # These PRs show some reasoning about the order of checks
122 if request.num_output_tokens < sampling_params.min_tokens:
123 return False
124
125 repetition_detection = sampling_params.repetition_detection
126 if repetition_detection is not None and (
127 check_sequence_repetition(
128 request.output_token_ids,
129 repetition_detection,
130 )
131 ):
132 request.status = RequestStatus.FINISHED_REPETITION
133 request.stop_reason = "repetition_detected"
134 return True
135
136 return False
