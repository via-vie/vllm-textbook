4from vllm.logger import init_logger
5from vllm.v1.core.sched.output import SchedulerOutput
6from vllm.v1.core.sched.scheduler import Scheduler
7from vllm.v1.request import Request, RequestStatus
8
9logger = init_logger(__name__)
10
11
12class AsyncScheduler(Scheduler):
13 def __init__(self, *args, **kwargs) -> None:
14 super().__init__(*args, **kwargs)
15 # reusable read-only placeholder list for speculative decoding.
16 self._spec_token_placeholders: list[int] = [-1] * self.num_spec_tokens
17 self.pp_size = self.parallel_config.pipeline_parallel_size
18
19 def _update_after_schedule(self, scheduler_output: SchedulerOutput) -> None:
20 super()._update_after_schedule(scheduler_output)
21 spec_decode_tokens = scheduler_output.scheduled_spec_decode_tokens
22 # Use the latest num of scheduled draft tokens in next step as placeholder.
23 self._spec_token_placeholders = [
24 -1
25 ] * scheduler_output.num_spec_tokens_to_schedule
26 for req_id in scheduler_output.num_scheduled_tokens:
27 request = self.requests[req_id]
28 if request.is_prefill_chunk:
29 continue
30
31 scheduler_output.pending_structured_output_tokens |= (
32 request.use_structured_output and request.num_output_placeholders > 0
33 )
34 # The request will generate num_sampled_tokens_per_step new tokens
35 # plus num_spec_tokens in this scheduling step. Diffusion has no AR
36 # bonus token (num_sampled_tokens_per_step == 0) — only the canvas
37 # (spec) tokens.
38 cur_num_spec_tokens = len(spec_decode_tokens.get(req_id, ()))
39 request.num_output_placeholders += (
40 self.num_sampled_tokens_per_step + cur_num_spec_tokens
41 )
42 # Add placeholders for the new draft/spec tokens.
43 # We will update the actual spec token ids in the worker process.
44 request.spec_token_ids = self._spec_token_placeholders
45
46 if self.use_v2_model_runner:
47 # Set the next step index in which this request is eligible to be
48 # scheduled for decode (for PP microbatching).
49 request.next_decode_eligible_step = self.current_step + self.pp_size
50
51 def _update_request_with_output(
52 self, request: Request, new_token_ids: list[int], is_stale: bool = False
53 ) -> tuple[list[int], bool]:
54 status_before_update = request.status
55 new_token_ids, stopped = super()._update_request_with_output(
56 request, new_token_ids
57 )
58
59 # Placeholders were zeroed at preemption; a stale delivery must not
60 # decrement them (it would underflow).
61 if not is_stale:
62 request.num_output_placeholders -= len(new_token_ids)
63 assert request.num_output_placeholders >= 0
64
65 # Cache the new tokens. Preempted requests should be skipped.
66 if status_before_update == RequestStatus.RUNNING:
67 self.kv_cache_manager.cache_blocks(
68 request, request.num_computed_tokens - request.num_output_placeholders
69 )
70 return new_token_ids, stopped
