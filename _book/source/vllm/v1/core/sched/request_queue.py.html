20class RequestQueue(ABC):
21 """Abstract base class for request queues."""

75class FCFSRequestQueue(deque[Request], RequestQueue):
76 """A first-come-first-served queue that supports deque operations."""
77
78 def add_request(self, request: Request) -> None:
79 """Add a request to the queue according to FCFS policy."""
80 self.append(request)
81
82 def pop_request(self) -> Request:
83 """Pop a request from the queue according to FCFS policy."""
84 return self.popleft()
85
86 def peek_request(self) -> Request:
87 """Peek at the next request in the queue without removing it."""
88 if not self:
89 raise IndexError("peek from an empty queue")
90 return self[0]
91
92 def prepend_request(self, request: Request) -> None:
93 """Prepend a request to the front of the queue."""
94 self.appendleft(request)
95
96 def prepend_requests(self, requests: RequestQueue) -> None:
97 """Prepend all requests from another queue to the front of this
98 queue.
99
100 Note: The requests will be prepended in reverse order of their
101 appearance in the `requests` queue.
102 """
103 self.extendleft(requests)
104
105 def remove_request(self, request: Request) -> None:
106 """Remove a specific request from the queue."""
107 self.remove(request)
108
109 def remove_requests(self, requests: Iterable[Request]) -> None:
110 """Remove multiple specific requests from the queue."""
111 requests_to_remove = set(requests)
112 filtered_requests = [req for req in self if req not in requests_to_remove]
113 # deque does not support in-place filtering, so we need to clear
114 # and extend
115 self.clear()
116 self.extend(filtered_requests)
117
118 def __bool__(self) -> bool:
119 """Check if queue has any requests."""
120 return len(self) > 0
121
122 def __len__(self) -> int:
123 """Get number of requests in queue."""
124 return super().__len__()
125
126 def __iter__(self) -> Iterator[Request]:
127 """Iterate over the queue according to FCFS policy."""
128 return super().__iter__()

131class PriorityRequestQueue(RequestQueue):
132 """
133 A priority queue that supports heap operations.
134
135 Respects the ordering defined in the Request class, where
136 requests with a smaller value of `priority` are processed first.
137 If multiple requests have the same priority, the one with the earlier
138 `arrival_time` is processed first.
139 """
140
141 def __init__(self) -> None:
142 self._heap: list[Request] = []
143
144 def add_request(self, request: Request) -> None:
145 """Add a request to the queue according to priority policy."""
146 heapq.heappush(self._heap, request)
147
148 def pop_request(self) -> Request:
149 """Pop a request from the queue according to priority policy."""
150 if not self._heap:
151 raise IndexError("pop from empty heap")
152 return heapq.heappop(self._heap)
153
154 def peek_request(self) -> Request:
155 """Peek at the next request in the queue without removing it."""
156 if not self._heap:
157 raise IndexError("peek from empty heap")
158 return self._heap[0]
159
160 def prepend_request(self, request: Request) -> None:
161 """Add a request to the queue according to priority policy.
162
163 Note: In a priority queue, there is no concept of prepending to the
164 front. Requests are ordered by (priority, arrival_time)."""
165 self.add_request(request)
166
167 def prepend_requests(self, requests: RequestQueue) -> None:
168 """Add all requests from another queue according to priority policy.
169
170 Note: In a priority queue, there is no concept of prepending to the
171 front. Requests are ordered by (priority, arrival_time)."""
172 for request in requests:
173 self.add_request(request)
174
175 def remove_request(self, request: Request) -> None:
176 """Remove a specific request from the queue."""
177 self._heap.remove(request)
178 heapq.heapify(self._heap)
179
180 def remove_requests(self, requests: Iterable[Request]) -> None:
181 """Remove multiple specific requests from the queue."""
182 requests_to_remove = requests if isinstance(requests, set) else set(requests)
183 self._heap = [r for r in self._heap if r not in requests_to_remove]
184 heapq.heapify(self._heap)
185
186 def __bool__(self) -> bool:
187 """Check if queue has any requests."""
188 return bool(self._heap)
189
190 def __len__(self) -> int:
191 """Get number of requests in queue."""
192 return len(self._heap)
193
194 def __iter__(self) -> Iterator[Request]:
195 """Iterate over the queue according to priority policy."""
196 heap_copy = self._heap[:]
197 while heap_copy:
198 yield heapq.heappop(heap_copy)
199

201def create_request_queue(policy: SchedulingPolicy) -> RequestQueue:
202 """Create request queue based on scheduling policy."""
203 if policy == SchedulingPolicy.PRIORITY:
204 return PriorityRequestQueue()
205 elif policy == SchedulingPolicy.FCFS:
206 return FCFSRequestQueue()
207 else:
208 raise ValueError(f"Unknown scheduling policy: {policy}")
