413 def _mamba_block_aligned_split(
414 self,
415 request: Request,
416 num_new_tokens: int,
417 num_new_local_computed_tokens: int = 0,
418 num_external_computed_tokens: int = 0,
419 ) -> int:
420 """Clip a prefill chunk so it ends where Mamba state must be cached.
421
422 In "align" cache mode reusable SSM states are materialized at block
423 boundaries, plus mandatory early stops (the prompt's partial-tail hash
424 boundary, a detected shared-prefix junction). If a block is larger
425 than the configured prefill chunk limit, intermediate chunks keep
426 private running state until they reach the next cacheable position.
427 """
428 start = (
429 request.num_computed_tokens
430 + num_new_local_computed_tokens
431 + num_external_computed_tokens
432 )
433 # Split only during prefill: `request.num_tokens - 1` extends this to
434 # resumed requests replaying their output tokens.
435 prefill_end = max(request.num_prompt_tokens, request.num_tokens - 1)
436 if start >= prefill_end:
437 return num_new_tokens
438
439 block_size = self.cache_config.block_size
440 # The last block-aligned position whose state can be cached. With
441 # Eagle, FullAttn prunes the last matching block, so back off one
442 # block to avoid a Mamba cache miss.
443 last_cache_position = request.num_tokens - request.num_tokens % block_size
444 if self.use_eagle_block_drop:
445 last_cache_position = max(last_cache_position - block_size, 0)
446
447 end = start + num_new_tokens
448 checkpoint_position = get_mamba_prefill_checkpoint_position(
449 prefill_end,
450 self.hash_block_size,
451 drop_eagle_block=self.use_eagle_block_drop,
452 )
453 use_internal_checkpoint = (
454 self.mamba_has_prefill_checkpoint_blocks
455 and end >= prefill_end
456 and is_mamba_prefill_checkpoint_valid(
457 query_start=start,
458 query_end=end,
459 checkpoint_position=checkpoint_position,
460 hash_block_size=self.hash_block_size,
461 mamba_block_size=block_size,
462 checkpoint_alignment=self.mamba_prefill_checkpoint_alignment,
463 )
464 )
465 if use_internal_checkpoint:
466 last_cache_position = 0
467 # Invariant: slot p holds the state after exactly (p + 1) * block_size
468 # tokens. State is written at chunk ends, so chunk ends must be block
469 # aligned. Exempt: the prompt's last chunk, whose slot decode advances
470 # to the boundary. A block too wide for one chunk advances sub-block
471 # and re-aligns at the next boundary.
472 if end < prefill_end:
473 max_prefill_tokens = self.max_num_scheduled_tokens
474 long_prefill_threshold = self.scheduler_config.long_prefill_token_threshold
475 if long_prefill_threshold > 0:
476 max_prefill_tokens = min(max_prefill_tokens, long_prefill_threshold)
477 aligned_end = end // block_size * block_size
478 if aligned_end > start or block_size <= max_prefill_tokens:
479 end = aligned_end
480
481 next_block_boundary = (start // block_size + 1) * block_size
482 tail_boundary = (
483 request.num_prompt_tokens // self.hash_block_size * self.hash_block_size
484 if self.mamba_partial_cache_hit and not use_internal_checkpoint
485 else 0
486 )
487 if tail_boundary and self.use_eagle_block_drop:
488 # Eagle matches one hash unit past the candidate and drops it, so
489 # nothing proves the prompt's own last hash boundary. Materialize
490 # the state one unit lower, where the hit can actually land. Keyed on
491 # the block-drop bit, not plain use_eagle: this shift exists only to
492 # compensate for the drop, and the Mamba manager's matching gate
493 # reads the same bit (the coordinator is handed use_eagle_block_drop).
494 tail_boundary = max(tail_boundary - self.hash_block_size, 0)
495 junction = request.shared_prefix_boundary
496 # Block-floored: a sub-block junction's state is not separately cacheable.
497 block_floored = start + (junction - start) // block_size * block_size
498 # Past the prompt the manager writes nothing, so fall back to the
499 # block-floored stop rather than dropping it: a resumed request replaying
500 # output tokens can still observe a junction there.
501 junction_stop = (
502 junction
503 if self.mamba_fine_grained_prefix_cache
504 and junction <= request.num_prompt_tokens
505 else block_floored
506 )
507 stops = (
508 # Same invariant: a chunk starting mid-block stops at the boundary
509 # rather than running past it.
510 next_block_boundary
511 if start % block_size != 0 and not use_internal_checkpoint
512 else 0,
513 # Never run past the last cacheable block boundary mid-chunk.
514 last_cache_position,
515 # Fine-grained hits: the prompt's partial-tail entry can only be
516 # registered by a chunk ending exactly at its last hash boundary.
517 tail_boundary
518 if last_cache_position < tail_boundary < request.num_prompt_tokens
519 else 0,
520 # Marconi shared-prefix junction: cache its state so sibling
521 # requests sharing the prefix can reuse it.
522 junction_stop if start < junction < end else 0,
523 )
524 # Stop at the earliest mandatory position strictly inside the chunk.
525 end = min((s for s in stops if start < s < end), default=end)
526 return max(end - start, 0)

540 def _reserve_prefill_lookahead(
541 self,
542 request: Request,
543 num_computed_tokens: int,
544 num_new_tokens: int,
545 ) -> int:
546 """Never end a prefill chunk within num_prefill_lookahead of the
547 prefill end.
548
549 At a chunked-prefill boundary, the multi-module MTP drafter consumes
550 the next num_prefill_lookahead known prefill tokens as draft inputs. A
551 boundary closer to the end than that would make it fall back to
552 sampled drafts, permanently polluting the trailing modules' KV caches.
553 Either finish the prefill or leave at least num_prefill_lookahead for
554 the next chunk. No-op for eagle-family drafters (lookahead 1).
555 """
556 remaining = request.num_tokens - num_computed_tokens - num_new_tokens
557 if 0 < remaining < self.num_prefill_lookahead:
558 num_new_tokens -= self.num_prefill_lookahead - remaining
559 return max(num_new_tokens, 0)
560
561 def schedule(self, throttle_prefills: bool = False) -> SchedulerOutput:
562 self.current_step += 1
563 # NOTE(woosuk) on the scheduling algorithm:
564 # There's no "decoding phase" nor "prefill phase" in the scheduler.
565 # Each request just has the num_computed_tokens and
566 # num_tokens_with_spec. num_tokens_with_spec =
567 # len(prompt_token_ids) + len(output_token_ids) + len(spec_token_ids).
568 # At each step, the scheduler tries to assign tokens to the requests
569 # so that each request's num_computed_tokens can catch up its
570 # num_tokens_with_spec. This is general enough to cover
571 # chunked prefills, prefix caching, speculative decoding,
572 # and the "jump decoding" optimization in the future.
573
574 scheduled_new_reqs: list[Request] = []
575 scheduled_resumed_reqs: list[Request] = []
576 scheduled_running_reqs: list[Request] = []
577 preempted_reqs: list[Request] = []
578
579 req_to_new_blocks: dict[str, KVCacheBlocks] = {}
580 num_scheduled_tokens: dict[str, int] = {}
581 token_budget = self.max_num_scheduled_tokens
582 spec = self.vllm_config.speculative_config
583 draft_slots = spec.max_num_new_slots_for_drafting if spec is not None else 0
584 input_budget = self.scheduler_config.max_num_batched_tokens
585 if self._pause_state == PauseState.PAUSED_ALL:
586 # Do not schedule any requests when paused.
587 token_budget = 0
588
589 # Encoder-related.
590 scheduled_encoder_inputs: dict[str, list[int]] = {}
591 encoder_compute_budget = self.max_num_encoder_input_tokens
592 # Spec decode-related.
593 scheduled_spec_decode_tokens: dict[str, list[int]] = {}
594 # Whether the running batch contains any prefill requests.
595 prefill_scheduled = False
596 # Whether any scheduled request has a synchronous connector KV load.
597 has_sync_kv_loads = False
598
599 # For logging.
600 scheduled_timestamp = time.monotonic()
601
602 self.kv_cache_manager.new_step_starts()
603
604 # DP prefill balancing: on a throttled (non-cadence-aligned) step, defer
605 # all prefill compute unless saturated.
606 defer_prefills = (
607 throttle_prefills and not self.prefill_capacity_bound
608 ) and any(not r.is_prefill_chunk for r in self.running)
609
610 # First, schedule the RUNNING requests.
611 req_index = 0
612 while req_index < len(self.running) and token_budget > 0:
613 request = self.running[req_index]
614 if input_budget <= draft_slots:
615 break
616
617 if (
618 request.num_output_placeholders > 0
619 # This is (num_computed_tokens + 1) - (num_output_placeholders - 1).
620 # Since output placeholders are also included in the computed tokens
621 # count, we subtract (num_output_placeholders - 1) to remove any draft
622 # tokens, so that we can be sure no further steps are needed even if
623 # they are all rejected.
624 and request.num_computed_tokens + 2 - request.num_output_placeholders
625 >= request.num_prompt_tokens + request.max_tokens
626 ):
627 # Async scheduling: Avoid scheduling an extra step when we are sure that
628 # the previous step has reached request.max_tokens. We don't schedule
629 # partial draft tokens since this prevents uniform decode optimizations.
630 req_index += 1
631 continue
632
633 if self.current_step < request.next_decode_eligible_step:
634 # V2+PP+async: enforce `pp_size` steps between same-req decodes
635 # to match worker-side sampled-tokens broadcast slot ring cadence.
636 req_index += 1
637 continue
638
639 if defer_prefills and request.is_prefill_chunk:
640 # DP prefill balancing: defer this in-progress prefill chunk to a
641 # cadence-aligned step; decodes still run to fill this step.
642 req_index += 1
643 continue
644
645 if (
646 self.ec_connector is not None
647 and request.mm_features
648 and not self.ec_connector.ensure_cache_available(
649 request,
650 request.num_computed_tokens - request.num_output_placeholders,
651 )
652 ):
653 req_index += 1
654 continue
655
656 num_new_tokens = (
657 request.num_tokens_with_spec
658 + request.num_output_placeholders
659 - request.num_computed_tokens
660 )
661 if 0 < self.scheduler_config.long_prefill_token_threshold < num_new_tokens:
662 num_new_tokens = self.scheduler_config.long_prefill_token_threshold
663 num_new_tokens = min(
664 num_new_tokens, token_budget, input_budget - draft_slots
665 )
666
667 # Make sure the input position does not exceed the max model len.
668 # This is necessary when using spec decoding.
669 num_new_tokens = min(
670 num_new_tokens,
671 self.max_model_len
672 - request.num_computed_tokens
673 - self.num_sampled_tokens_per_step,
674 )
675
676 # Apply Mamba alignment before encoder caps.
677 if self.need_mamba_block_aligned_split:
678 num_new_tokens = self._mamba_block_aligned_split(
679 request, num_new_tokens
680 )
681
682 # Schedule encoder inputs.
683 encoder_inputs_to_schedule = None
684 external_load_encoder_input: list[int] = []
685 new_encoder_compute_budget = encoder_compute_budget
686 if request.has_encoder_inputs:
687 (
688 encoder_inputs_to_schedule,
689 num_new_tokens,
690 new_encoder_compute_budget,
691 external_load_encoder_input,
692 ) = self._try_schedule_encoder_inputs(
693 request,
694 request.num_computed_tokens,
695 num_new_tokens,
696 encoder_compute_budget,
697 shift_computed_tokens=self.num_prefill_lookahead,
698 )
699
700 # Multi-module MTP: avoid ending a prefill chunk within
701 # num_prefill_lookahead of the prefill end.
702 num_new_tokens = self._reserve_prefill_lookahead(
703 request, request.num_computed_tokens, num_new_tokens
704 )
705
706 if num_new_tokens == 0:
707 # The request cannot be scheduled because one of the following
708 # reasons:
709 # 1. No new tokens to schedule. This may happen when
710 # (1) PP>1 and we have already scheduled all prompt tokens
711 # but they are not finished yet.
712 # (2) Async scheduling and the request has reached to either
713 # its max_total_tokens or max_model_len.
714 # 2. The encoder budget is exhausted.
715 # 3. The encoder cache is exhausted.
716 # 4. Insufficient budget for a block-aligned chunk in hybrid
717 # models with mamba cache mode \"align\".
718 # 5. Insufficient budget to keep a multi-module MTP prefill
719 # chunk out of the prefill-lookahead window.
720 # NOTE(woosuk): Here, by doing `continue` instead of `break`,
721 # we do not strictly follow the FCFS scheduling policy and
722 # allow the lower-priority requests to be scheduled.
723 req_index += 1
724 continue
725
726 # Schedule newly needed KV blocks for the request.
727 with record_function_or_nullcontext("schedule: allocate_slots"):
728 while True:
729 new_blocks = self.kv_cache_manager.allocate_slots(
730 request,
731 num_new_tokens,
732 num_lookahead_tokens=self.num_lookahead_tokens,
733 )
734
735 if new_blocks is not None:
736 # The request can be scheduled.
737 break
738
739 if (
740 self.connector is not None
741 and self.connector.has_pending_block_frees()
742 ):
743 break
744
745 # The request cannot be scheduled.
746 # Preempt the lowest-priority request.
747 if self.policy == SchedulingPolicy.PRIORITY:
748 preempted_req = max(
749 self.running,
750 key=lambda r: (r.priority, r.arrival_time),
751 )
752 else:
753 preempted_req = self.running[-1]
754
755 # A deferred free will not help with immediate allocation.
756 if not self._request_blocks_can_be_freed(preempted_req):
757 break
758
759 if self.policy == SchedulingPolicy.PRIORITY:
760 victim_index = self.running.index(preempted_req)
761 del self.running[victim_index]
762 if victim_index < req_index:
763 req_index -= 1
764
765 if preempted_req in scheduled_running_reqs:
766 preempted_req_id = preempted_req.request_id
767 scheduled_running_reqs.remove(preempted_req)
768 restored = num_scheduled_tokens.pop(preempted_req_id)
769 token_budget += restored
770 input_budget += restored + draft_slots
771 req_to_new_blocks.pop(preempted_req_id)
772 scheduled_spec_decode_tokens.pop(preempted_req_id, None)
773 preempted_encoder_inputs = scheduled_encoder_inputs.pop(
774 preempted_req_id, None
775 )
776 if preempted_encoder_inputs:
777 # Restore encoder compute budget if the preempted
778 # request had encoder inputs scheduled in this step.
779 num_embeds_to_restore = sum(
780 preempted_req.get_num_encoder_embeds(i)
781 for i in preempted_encoder_inputs
782 )
783 encoder_compute_budget += num_embeds_to_restore
784 else:
785 preempted_req = self.running.pop()
786
787 self._preempt_request(
788 preempted_req,
789 scheduled_timestamp,
790 drop_stale_output=self.requires_kv_delivery,
791 )
792 preempted_reqs.append(preempted_req)
793 if preempted_req == request:
794 # No more request to preempt. Cannot schedule this request.
795 break
796
797 if new_blocks is None:
798 # Cannot schedule this request.
799 break
800
801 # Schedule the request.
802 scheduled_running_reqs.append(request)
803 prefill_scheduled |= request.is_prefill_chunk
804 request_id = request.request_id
805 req_to_new_blocks[request_id] = new_blocks
806 num_scheduled_tokens[request_id] = num_new_tokens
807 token_budget -= num_new_tokens
808 input_budget -= num_new_tokens + draft_slots
809 req_index += 1
810
811 # Speculative decode related.
812 if request.spec_token_ids:
813 num_scheduled_spec_tokens = (
814 num_new_tokens
815 + request.num_computed_tokens
816 - request.num_tokens
817 - request.num_output_placeholders
818 )
819 if num_scheduled_spec_tokens > 0:
820 spec_token_ids = request.spec_token_ids
821 if len(spec_token_ids) > num_scheduled_spec_tokens:
822 spec_token_ids = spec_token_ids[:num_scheduled_spec_tokens]
823 scheduled_spec_decode_tokens[request.request_id] = spec_token_ids
824
825 # New spec tokens will be set in `update_draft_token_ids` before the
826 # next step when applicable.
827 request.spec_token_ids = []
828
829 # Encoder-related.
830 if encoder_inputs_to_schedule:
831 scheduled_encoder_inputs[request_id] = encoder_inputs_to_schedule
832 # Allocate the encoder cache.
833 for i in encoder_inputs_to_schedule:
834 self.encoder_cache_manager.allocate(request, i)
835 if self.ec_connector is not None:
836 self.ec_connector.update_state_after_alloc(request, i)
837 encoder_compute_budget = new_encoder_compute_budget
838 if external_load_encoder_input:
839 for i in external_load_encoder_input:
840 self.encoder_cache_manager.allocate(request, i)
841 if self.ec_connector is not None:
842 self.ec_connector.update_state_after_alloc(request, i)
843
844 # Record the LoRAs in scheduled_running_reqs
845 scheduled_loras: set[int] = set()
846 if self.lora_config:
847 scheduled_loras = set(
848 req.lora_request.lora_int_id
849 for req in scheduled_running_reqs
850 if req.lora_request and req.lora_request.lora_int_id > 0
851 )
852 assert len(scheduled_loras) <= self.lora_config.max_loras
853
854 # Next, schedule the WAITING requests.
855 if not preempted_reqs and self._pause_state == PauseState.UNPAUSED:
856 step_skipped_waiting = create_request_queue(self.policy)
857
858 while (self.waiting or self.skipped_waiting) and token_budget > 0:
859 if input_budget <= draft_slots:
860 break
861 # Paused streaming sessions (WAITING_FOR_STREAMING_REQ) are not
862 # in `running` but still hold a model-runner request slot.
863 num_running = len(self.running) + self.num_waiting_for_streaming_input
864 if num_running >= self.max_num_running_reqs:
865 break
866
867 request_queue = self._select_waiting_queue_for_scheduling()
868 assert request_queue is not None
869
870 request = request_queue.peek_request()
871 request_id = request.request_id
872
873 # try to promote blocked statuses while traversing skipped queue.
874 if self._is_blocked_waiting_status(
875 request.status
876 ) and not self._try_promote_blocked_waiting_request(request):
877 if request.status == RequestStatus.WAITING_FOR_REMOTE_KVS:
878 logger.debug(
879 "%s is still in WAITING_FOR_REMOTE_KVS state.",
880 request_id,
881 )
882 request_queue.pop_request()
883 step_skipped_waiting.prepend_request(request)
884 continue
885
886 if (
887 request.num_stale_output_tokens > 0
888 and not request.drop_stale_output
889 ):
890 # Deliverable stale output still in flight: resuming now
891 # could resample a position that output later delivers.
892 # It drains within the pipeline depth.
893 request_queue.pop_request()
894 step_skipped_waiting.prepend_request(request)
895 continue
896
897 # Check that adding the request still respects the max_loras
898 # constraint.
899 if (
900 self.lora_config
901 and request.lora_request
902 and (
903 len(scheduled_loras) == self.lora_config.max_loras
904 and request.lora_request.lora_int_id not in scheduled_loras
905 )
906 ):
907 # Scheduling would exceed max_loras, skip.
908 request_queue.pop_request()
909 step_skipped_waiting.prepend_request(request)
910 continue
911
912 num_external_computed_tokens = 0
913 load_kv_async = False
914 connector_prefix_cache_queries, connector_prefix_cache_hits = 0, 0
915 did_prefix_cache_lookup = False
916
917 # Get already-cached tokens.
918 if request.num_computed_tokens == 0:
919 did_prefix_cache_lookup = True
920 (
921 new_computed_blocks,
922 num_new_local_computed_tokens,
923 request.shared_prefix_boundary,
924 hit_diverged,
925 ) = self._get_local_prefix_cache_hit(request)
926
927 # Get externally-cached tokens if using a KVConnector.
928 if self.connector is not None:
929 # Present a block-aligned local hit to the connector so
930 # a strictly longer remote hit can supersede a local
931 # sub-block tail without racing its copy-on-write.
932 partial_tail = num_new_local_computed_tokens % self.block_size
933 block_aligned_local = (
934 num_new_local_computed_tokens - partial_tail
935 )
936 ext_tokens, load_kv_async = (
937 self.connector.get_num_new_matched_tokens(
938 request, block_aligned_local
939 )
940 )
941
942 if ext_tokens is None:
943 # The request cannot be scheduled because
944 # the KVConnector couldn't determine
945 # the number of matched tokens.
946 request_queue.pop_request()
947 step_skipped_waiting.prepend_request(request)
948 continue
949
950 if partial_tail and ext_tokens > partial_tail:
951 # Remote strictly exceeds the full local hit: drop the
952 # sub-block tail so no CoW is needed, and let the load
953 # cover it. Trim the partial block out of the local
954 # computed blocks so it is not adopted from the cache.
955 new_computed_blocks = (
956 self.kv_cache_manager.truncate_computed_blocks(
957 new_computed_blocks, block_aligned_local
958 )
959 )
960 num_new_local_computed_tokens = block_aligned_local
961 num_external_computed_tokens = ext_tokens
962 elif partial_tail:
963 # Remote does not exceed the full local hit: keep the
964 # local sub-block tail and load nothing external.
965 num_external_computed_tokens = 0
966 # Nothing to load remotely -> not an async-load step;
967 # clearing avoids the `load_kv_async` assert below.
968 load_kv_async = False
969 else:
970 num_external_computed_tokens = ext_tokens
971
972 if hit_diverged and num_external_computed_tokens == 0:
973 # No external tokens back the deeper local hit, so its
974 # resume boundary would have no valid Mamba state.
975 # Reconcile to the boundary every group agrees on.
976 (
977 new_computed_blocks,
978 num_new_local_computed_tokens,
979 request.shared_prefix_boundary,
980 ) = self.kv_cache_manager.get_computed_blocks(request)
981
982 connector_prefix_cache_queries = (
983 request.num_tokens - num_new_local_computed_tokens
984 )
985 connector_prefix_cache_hits = num_external_computed_tokens
986
987 # Total computed tokens (local + external).
988 num_computed_tokens = (
989 num_new_local_computed_tokens + num_external_computed_tokens
990 )
991 assert num_computed_tokens <= request.num_tokens
992
993 # Skip request with pending mm encoding prefetches
994 if self._ec_transfer_pending(request, num_computed_tokens):
995 request_queue.pop_request()
996 step_skipped_waiting.prepend_request(request)
997 continue
998
999 # Track first scheduled prefill, not post-preemption repeat prefills
1000 if request.prefill_stats and request.num_preemptions <= 0:
1001 assert num_computed_tokens <= request.num_prompt_tokens
1002 request.prefill_stats.set(
1003 num_prompt_tokens=request.num_prompt_tokens,
1004 num_local_cached_tokens=num_new_local_computed_tokens,
1005 num_external_cached_tokens=num_external_computed_tokens,
1006 )
1007 else:
1008 # KVTransfer: WAITING reqs have num_computed_tokens > 0
1009 # after async KV recvs are completed. A streaming-input
1010 # session resumes here too, carrying whatever media its
1011 # latest chunk added, so this branch needs the same gate.
1012 new_computed_blocks = self.kv_cache_manager.empty_kv_cache_blocks
1013 num_new_local_computed_tokens = 0
1014 num_computed_tokens = request.num_computed_tokens
1015
1016 if self._ec_transfer_pending(request, num_computed_tokens):
1017 request_queue.pop_request()
1018 step_skipped_waiting.prepend_request(request)
1019 continue
1020
1021 encoder_inputs_to_schedule = None
1022 external_load_encoder_input = []
1023 new_encoder_compute_budget = encoder_compute_budget
1024 pad_spec_decode = False
1025
1026 if load_kv_async:
1027 # KVTransfer: loading remote KV, do not allocate for new work.
1028 assert num_external_computed_tokens > 0
1029 num_new_tokens = 0
1030 elif defer_prefills and num_computed_tokens < request.num_tokens - 1:
1031 # DP prefill balancing: defer this step's local prefill
1032 # compute to a cadence-aligned step.
1033 break
1034 else:
1035 request_token_budget = min(token_budget, input_budget - draft_slots)
1036 # Number of tokens to be scheduled.
1037 # We use `request.num_tokens` instead of
1038 # `request.num_prompt_tokens` to consider the resumed
1039 # requests, which have output tokens.
1040 num_new_tokens = request.num_tokens - num_computed_tokens
1041
1042 # Pad new decode requests to uniform spec decoding size to
1043 # preserve full cudagraph for this step.
1044 # Not for diffusion where draft tokens can't be padded.
1045 if (
1046 (self.num_spec_tokens > 0 and self.dynamic_sd_lookup is None)
1047 and self.num_sampled_tokens_per_step > 0
1048 and num_new_tokens == 1
1049 and not prefill_scheduled
1050 and (scheduled_running_reqs or num_computed_tokens > 0)
1051 ):
1052 padded_num_tokens = 1 + self.num_spec_tokens
1053 # Pad only when there is room for the sampled token(s).
1054 if (
1055 num_computed_tokens
1056 + padded_num_tokens
1057 + self.num_sampled_tokens_per_step
1058 <= self.max_model_len
1059 ):
1060 if padded_num_tokens > request_token_budget:
1061 # Prefer to not schedule than schedule un-padded.
1062 break
1063 num_new_tokens = padded_num_tokens
1064 pad_spec_decode = True
1065
1066 threshold = self.scheduler_config.long_prefill_token_threshold
1067 if 0 < threshold < num_new_tokens:
1068 num_new_tokens = threshold
1069
1070 # chunked prefill has to be enabled explicitly to allow
1071 # pooling requests to be chunked
1072 if (
1073 not self.scheduler_config.enable_chunked_prefill
1074 and num_new_tokens > request_token_budget
1075 ):
1076 # If chunked_prefill is disabled,
1077 # we can stop the scheduling here.
1078 break
1079
1080 num_new_tokens = min(num_new_tokens, request_token_budget)
1081 assert num_new_tokens > 0
1082
1083 # Apply Mamba alignment before encoder caps.
1084 if self.need_mamba_block_aligned_split:
1085 num_new_tokens = self._mamba_block_aligned_split(
1086 request,
1087 num_new_tokens,
1088 num_new_local_computed_tokens,
1089 num_external_computed_tokens,
1090 )
1091 if num_new_tokens == 0:
1092 break
1093 if (
1094 pad_spec_decode
1095 and num_new_tokens != 1 + self.num_spec_tokens
1096 ):
1097 # Alignment clipped the placeholder rows. The split
1098 # aligns prefill chunks, but the padded tail rows are
1099 # speculative positions, not prefill tokens. A padded
1100 # request must keep all 1 + num_spec rows or the
1101 # sampler's row count stops matching its query rows,
1102 # so drop the padding instead of shortening it.
1103 num_new_tokens = 1
1104 pad_spec_decode = False
1105
1106 # Schedule encoder inputs.
1107 if request.has_encoder_inputs:
1108 (
1109 encoder_inputs_to_schedule,
1110 num_new_tokens,
1111 new_encoder_compute_budget,
1112 external_load_encoder_input,
1113 ) = self._try_schedule_encoder_inputs(
1114 request,
1115 num_computed_tokens,
1116 num_new_tokens,
1117 encoder_compute_budget,
1118 shift_computed_tokens=self.num_prefill_lookahead,
1119 )
1120
1121 # Multi-module MTP: avoid ending a prefill chunk within
1122 # num_prefill_lookahead of the prefill end.
1123 num_new_tokens = self._reserve_prefill_lookahead(
1124 request, num_computed_tokens, num_new_tokens
1125 )
1126
1127 if num_new_tokens == 0:
1128 # The request cannot be scheduled.
1129 break
1130
1131 # During async KV load, no forward pass is run yet.
1132 # Allocate speculative lookahead slots later to avoid
1133 # mismatching local and remote block counts.
1134 limit_lookahead_tokens = load_kv_async and self.num_lookahead_tokens > 0
1135 effective_lookahead_tokens = (
1136 0 if limit_lookahead_tokens else self.num_lookahead_tokens
1137 )
1138
1139 # Determine if we need to allocate cross-attention blocks.
1140 num_encoder_tokens = 0
1141 if (
1142 self.is_encoder_decoder
1143 and request.has_encoder_inputs
1144 and encoder_inputs_to_schedule
1145 ):
1146 num_encoder_tokens = sum(
1147 request.get_num_encoder_embeds(i)
1148 for i in encoder_inputs_to_schedule
1149 )
1150
1151 reserved_blocks = 0
1152 if load_kv_async:
1153 # An async load holds its blocks for the whole transfer with
1154 # no forward progress and isn't preemptible here. Admit it
1155 # only if it fits in (free - other in-flight reservations), to
1156 # avoid deadlock and predictable preemptions.
1157 reserved_blocks = self._inflight_prefill_reserved_blocks()
1158
1159 new_blocks = self.kv_cache_manager.allocate_slots(
1160 request,
1161 num_new_tokens,
1162 num_new_computed_tokens=num_new_local_computed_tokens,
1163 new_computed_blocks=new_computed_blocks,
1164 num_lookahead_tokens=effective_lookahead_tokens,
1165 num_external_computed_tokens=num_external_computed_tokens,
1166 delay_cache_blocks=load_kv_async,
1167 num_encoder_tokens=num_encoder_tokens,
1168 full_sequence_must_fit=self.scheduler_reserve_full_isl,
1169 reserved_blocks=reserved_blocks,
1170 has_scheduled_reqs=bool(self.running),
1171 )
1172
1173 if new_blocks is None:
1174 # The request cannot be scheduled.
1175
1176 # NOTE: we need to untouch the request from the encode cache
1177 # manager
1178 if request.has_encoder_inputs:
1179 self.encoder_cache_manager.free(request)
1180 break
1181
1182 # KVTransfer: the connector uses this info to determine
1183 # if a load is needed. Note that
1184 # This information is used to determine if a load is
1185 # needed for this request.
1186 if self.connector is not None:
1187 self.connector.update_state_after_alloc(
1188 request,
1189 self.kv_cache_manager.get_blocks(request_id),
1190 num_external_computed_tokens,
1191 )
1192 if (
1193 self.connector_prefix_cache_stats is not None
1194 and connector_prefix_cache_queries != 0
1195 ):
1196 self.connector_prefix_cache_stats.record(
1197 num_tokens=connector_prefix_cache_queries,
1198 num_hits=connector_prefix_cache_hits,
1199 preempted=request.num_preemptions > 0,
1200 )
1201
1202 # Record at admission so unscheduled lookups are not counted.
1203 if did_prefix_cache_lookup:
1204 self.kv_cache_manager.record_prefix_cache_stats(
1205 request, num_new_local_computed_tokens
1206 )
1207
1208 request = request_queue.pop_request()
1209 if load_kv_async:
1210 # If loading async, allocate memory and put request
1211 # into the WAITING_FOR_REMOTE_KV state.
1212 request.status = RequestStatus.WAITING_FOR_REMOTE_KVS
1213 step_skipped_waiting.prepend_request(request)

1215 # request.num_computed_tokens will not be used anywhere until
1216 # the request finished the KV transfer.
1217 #
1218 # If a transfer error is reported by the connector,
1219 # request.num_computed_tokens will be re-set accordingly in
1220 # _update_requests_with_invalid_blocks.
1221 #
1222 # When the transfer is finished, either successfully or not,
1223 # request.num_computed_tokens will correctly reflect the number
1224 # of computed tokens.
1225 # _update_waiting_for_remote_kv will then cache
1226 # only the successfully loaded tokens.
1227 request.num_computed_tokens = num_computed_tokens
1228 self._inflight_prefills.add(request)

1230 # Skip zeroing of the blocks the async load will
1231 # overwrite; the zeroing could race the write.
1232 self._skip_zero_block_ids.update(
1233 self.kv_cache_manager.get_zeroing_block_ids_in_range(
1234 request.request_id,
1235 num_new_local_computed_tokens,
1236 num_computed_tokens,
1237 )
1238 )

1241 self.running.append(request)
1242 if num_external_computed_tokens > 0:
1243 # load_kv_async is False here
1244 has_sync_kv_loads = True
1245 if self.log_stats:
1246 request.record_event(
1247 EngineCoreEventType.SCHEDULED, scheduled_timestamp
1248 )

1250 scheduled_new_reqs.append(request)
1251 elif request.status == RequestStatus.PREEMPTED:
1252 scheduled_resumed_reqs.append(request)
1253 else:

1257 scheduled_loras.add(request.lora_request.lora_int_id)

1259 request_id
1260 )

1263 input_budget -= num_new_tokens + draft_slots

1267 assert num_new_tokens == 1 + self.num_spec_tokens

1269 -1

1280 if self.ec_connector is not None:
1281 self.ec_connector.update_state_after_alloc(request, i)

1283 # Allocate for external load encoder cache
1284 if external_load_encoder_input:
1285 for i in external_load_encoder_input:
1286 self.encoder_cache_manager.allocate(request, i)
1287 if self.ec_connector is not None:
1288 self.ec_connector.update_state_after_alloc(request, i)

1291 if step_skipped_waiting:
1292 self.skipped_waiting.prepend_requests(step_skipped_waiting)

1294 # DP prefill balancing: on a step that admitted prefills (release),
1295 # record whether it was capacity-bound.
1296 if not defer_prefills:
1297 self.prefill_capacity_bound = bool(self.waiting)

1300 total_num_scheduled_tokens = sum(num_scheduled_tokens.values())
1301 assert total_num_scheduled_tokens <= self.max_num_scheduled_tokens
1302
1303 assert token_budget >= 0
1304 assert input_budget >= 0
1305 assert len(self.running) <= self.max_num_running_reqs

1307 # this step, the total number of scheduled requests can be smaller than
1308 # len(self.running).

1310 scheduled_running_reqs

1320 self.kv_cache_manager.get_num_common_prefix_blocks(any_request_id)

1349 scheduled_resumed_reqs,
1350 num_scheduled_tokens,
1351 scheduled_spec_decode_tokens,
1352 req_to_new_blocks,
1353 )

1355 # Record the request ids that were scheduled in this step (MRV1-only).
1356 if not self.use_v2_model_runner:
1357 self.prev_step_scheduled_req_ids.clear()

1360 # Mamba "align" boundary states must be handed off with exact block ids;

1362 # table. Drained every step so stale offers cannot accumulate.
1363 boundary_state_offloads = self.kv_cache_manager.take_boundary_state_offloads()

1365 kv_connector_block_state = None
1366 if self.connector is not None:

1368 # not only the ones that were allocated blocks: a store save lands

1370 block_state_req_ids = set(num_scheduled_tokens)

1372 req_id for req_id in boundary_state_offloads if req_id in self.requests

1375 req_ids=block_state_req_ids,
1376 resolve_block_ids=self.kv_cache_manager.get_block_ids,

1378 )

1380 kv_cache_block_copies, cow_retained_blocks = (
1381 self.kv_cache_manager.take_kv_cache_block_copies()
1382 )
1383 if kv_cache_block_copies:

1393 if self.dynamic_sd_lookup is not None and len(num_scheduled_tokens) > 0:
1394 num_spec_tokens_to_schedule = self.dynamic_sd_lookup[
1395 len(num_scheduled_tokens)
1396 ]

1398 scheduled_encoder_input_stats = None
1399 if (
1400 self.log_stats
1401 and self.observability_config.enable_logging_iteration_details
1402 ):
1403 scheduled_encoder_input_stats = self._make_scheduled_encoder_input_stats(
1404 scheduled_encoder_inputs
1405 )

1407 scheduler_output = SchedulerOutput(
1408 scheduled_new_reqs=new_reqs_data,
1409 scheduled_cached_reqs=cached_reqs_data,

1414 scheduled_encoder_input_stats=scheduled_encoder_input_stats,

1420 # the previous and the current steps.

1422 free_encoder_mm_hashes=self.encoder_cache_manager.get_freed_mm_hashes(),

1426 kv_connector_block_state=kv_connector_block_state,
1427 num_spec_tokens_to_schedule=num_spec_tokens_to_schedule,

1432 # 1. Plan the KV cache store
1433 # 2. Wrap up all the KV cache load / save ops into an opaque object
1434 # 3. Clear the internal states of the connector

1436 meta = self._build_kv_connector_meta(self.connector, scheduler_output)

1443 )
1444 scheduler_output.ec_connector_metadata = ec_meta

1447 scheduler_output.kv_connector_block_state = None

1450 # write KV and have their output processed later in update_from_output).
1451 if self.defer_block_free and total_num_scheduled_tokens > 0:
1452 self.sched_step_seq += 1

1477 def _preempt_request(
1478 self, request: Request, timestamp: float, drop_stale_output: bool = False
1479 ) -> None:
1480 """Preempt a request and put it back to the waiting queue.
1481
1482 NOTE: The request should be popped from the running queue outside of this
1483 method.
1484
1485 drop_stale_output: drop (rather than deliver) any in-flight output; used
1486 by reset_prefix_cache, whose same-step resume would otherwise deliver
1487 tokens out of order, and for connectors with a pending KV hand-off,
1488 which the preemption's block free would leave without valid KV.
1489 """
1490 assert request.status == RequestStatus.RUNNING, (
1491 "Only running requests can be preempted"
1492 )
1493 self._free_request_blocks(request)
1494 self.encoder_cache_manager.free(request)
1495 self._inflight_prefills.discard(request)
1496 request.status = RequestStatus.PREEMPTED
1497 request.num_computed_tokens = 0
1498 if request.spec_token_ids:
1499 request.spec_token_ids = []
1500 # Async scheduling: mark all in-flight output as stale. Its tokens are
1501 # still delivered on return (dropping them would perturb spec-decode
1502 # acceptance) but must not mutate the reset counters; each step drains
1503 # its share in update_from_output. num_in_flight_tokens already
1504 # includes any undrained stale share, so assign rather than accumulate.
1505 # An undrained drop-mode share stays dropped: its positions have
1506 # already been resampled.
1507 request.drop_stale_output = drop_stale_output or (
1508 request.drop_stale_output and request.num_stale_output_tokens > 0
1509 )
1510 request.num_stale_output_tokens = request.num_in_flight_tokens
1511 request.num_output_placeholders = 0
1512 request.num_preemptions += 1
1513 if self.log_stats:
1514 request.record_event(EngineCoreEventType.PREEMPTED, timestamp)
1515
1516 # Put the request back to the waiting queue.
1517 self.waiting.prepend_request(request)
1518 self.reset_preempted_req_ids.add(request.request_id)

1520 def _update_after_schedule(self, scheduler_output: SchedulerOutput) -> None:
1521 # Advance the number of computed tokens for the request AFTER
1522 # the request is scheduled.
1523 # 1. The scheduler_output of the current step has to include the
1524 # original number of scheduled tokens to determine input IDs.
1525 # 2. Advance the number of computed tokens here allowing us to
1526 # schedule the prefill request again immediately in the next
1527 # scheduling step.
1528 # 3. If some tokens (e.g. spec tokens) are rejected later, the number of
1529 # computed tokens will be adjusted in update_from_output.
1530 num_scheduled_tokens = scheduler_output.num_scheduled_tokens
1531 for req_id, num_scheduled_token in num_scheduled_tokens.items():
1532 request = self.requests[req_id]
1533 request.num_computed_tokens += num_scheduled_token
1534 request.num_in_flight_tokens += num_scheduled_token
1535 if self.defer_block_free:
1536 # Record the in-flight step, to fence deferred block freeing.
1537 request.last_sched_seq = self.sched_step_seq
1538 request.is_prefill_chunk = request.num_computed_tokens < (
1539 request.num_tokens + request.num_output_placeholders
1540 )
1541 scheduler_output.has_structured_output_requests |= (
1542 request.use_structured_output and not request.is_prefill_chunk
1543 )
1544 # Drop from the in-flight-prefill set once it's no longer prefilling.
1545 if not request.is_prefill_chunk:
1546 self._inflight_prefills.discard(request)
1547
1548 # Snapshot block IDs for routed experts before forward starts.
1549 # A concurrent schedule() may preempt requests and free blocks
1550 # before update_from_output runs; the snapshot survives that.
1551 # Use update() to preserve entries from the previous step that
1552 # have not yet been consumed by update_from_output (async
1553 # scheduling may call _update_after_schedule again before the
1554 # prior update_from_output runs).
1555 if self.enable_return_routed_experts:
1556 gid = self.routed_experts_mgr.attn_gid
1557 self._re_block_ids.update(
1558 {
1559 rid: self.kv_cache_manager.get_blocks(rid).get_block_ids()[gid]
1560 for rid in num_scheduled_tokens
1561 }
1562 )
1563
1564 # Clear the finished and preempted request IDs.
1565 # NOTE: We shouldn't just clear() here because it will also affect
1566 # the scheduler output.
1567 self.finished_req_ids = set()
1568 self.reset_preempted_req_ids = set()

1613 def _make_cached_request_data(

1616 resumed_reqs: list[Request],

1659 req.num_output_tokens + req.num_output_placeholders

1669 num_output_tokens=num_output_tokens,

1672 def _try_schedule_encoder_inputs(
1673 self,
1674 request: Request,
1675 num_computed_tokens: int,
1676 num_new_tokens: int,
1677 encoder_compute_budget: int,
1678 shift_computed_tokens: int = 0,
1679 ) -> tuple[list[int], int, int, list[int]]:
1680 """
1681 Determine which encoder inputs need to be scheduled in the current step,
1682 and update `num_new_tokens` and encoder token budget accordingly.
1683
1684 An encoder input will be scheduled if:
1685 - Its output tokens overlap with the range of tokens being computed
1686 in this step, i.e.,
1687 [num_computed_tokens, num_computed_tokens + num_new_tokens).
1688 - It is not already computed and stored in the encoder cache.
1689 - It is not exist on remote encoder cache (via ECConnector)
1690 - There is sufficient encoder token budget to process it.
1691 - The encoder cache has space to store it.
1692
1693 If an encoder input cannot be scheduled due to cache or budget
1694 limitations, the method adjusts `num_new_tokens` to schedule only the
1695 decoder tokens up to just before the unschedulable encoder input.
1696
1697 Note that num_computed_tokens includes both locally cached
1698 blocks and externally cached blocks (via KVConnector).
1699 """
1700 if num_new_tokens == 0 or not request.has_encoder_inputs:
1701 return [], num_new_tokens, encoder_compute_budget, []
1702 encoder_inputs_to_schedule: list[int] = []
1703 mm_features = request.mm_features
1704 assert mm_features is not None
1705 assert len(mm_features) > 0
1706 external_load_encoder_input = []
1707
1708 # NOTE: since scheduler operates on the request level (possibly with
1709 # multiple encoder inputs per request), we need to create temporary
1710 # trackers for accounting at the encoder input level.
1711 mm_hashes_to_schedule = set()
1712 num_embeds_to_schedule = 0
1713
1714 encoder_window_end = (
1715 num_computed_tokens + num_new_tokens + shift_computed_tokens
1716 )
1717 lo, hi = get_mm_features_in_window(
1718 mm_features,
1719 start=num_computed_tokens,
1720 end=encoder_window_end,
1721 )
1722 # For encoder-decoder, all inputs sit at start_pos=0, so lo=0 always.
1723 if self.is_encoder_decoder:
1724 lo = 0
1725
1726 for i in range(lo, hi):
1727 mm_feature = mm_features[i]
1728 start_pos = mm_feature.mm_position.offset
1729 num_encoder_tokens = mm_feature.mm_position.length
1730 num_encoder_embeds = mm_feature.mm_position.get_num_embeds()
1731 item_identifier = mm_feature.identifier
1732
1733 if self.is_encoder_decoder and num_computed_tokens > 0:
1734 assert start_pos == 0, (
1735 "Encoder input should be processed at the beginning of "
1736 "the sequence when encoder-decoder models are used."
1737 )
1738 # Encoder input has already been computed
1739 # The calculation here is a bit different. We don't turn encoder
1740 # output into tokens that get processed by the decoder and
1741 # reflected in num_computed_tokens. Instead, start_pos reflects
1742 # the position where we need to ensure we calculate encoder
1743 # inputs. This should always be 0 to ensure we calculate encoder
1744 # inputs before running the decoder. Once we've calculated some
1745 # decoder tokens (num_computed_tokens > 0), then we know we
1746 # already calculated encoder inputs and can skip here.
1747 continue
1748
1749 if not self.is_encoder_decoder:
1750 # We are not using the encoder cache for encoder-decoder models,
1751 # yet.
1752 if item_identifier in mm_hashes_to_schedule:
1753 # The same encoder input has already been scheduled in the
1754 # current step.
1755 continue
1756
1757 if self.encoder_cache_manager.check_and_update_cache(request, i):
1758 # The encoder input is already computed and cached from a
1759 # previous step.
1760 continue

1821 num_embeds_to_schedule += num_encoder_embeds

2498 def finish_requests(

2502
2503 For example, the API server can abort a request when the client
2504 disconnects.
2505
2506 If request_ids is None, all requests will be finished.
2507
2508 Returns:
2509 List of requests that were aborted. Will not include any that were
2510 already finished.
2511 """
2512 assert RequestStatus.is_finished(finished_status)
2513 if isinstance(request_ids, str):
2514 request_ids = (request_ids,)
2515 elif request_ids is not None:
2516 request_ids = set(request_ids)
2517 else:
2518 request_ids = self.requests.keys()
2519
2520 running_requests_to_remove = set()
2521 waiting_requests_to_remove = []
2522 valid_requests = []

2524 # First pass: collect requests to remove from queues
2525 for req_id in request_ids:
2526 request = self.requests.get(req_id)
2527 if request is None or request.is_finished():

2529 continue

2533 running_requests_to_remove.add(request)

2543 self.waiting.remove_requests(waiting_requests_to_remove)
2544 self.skipped_waiting.remove_requests(waiting_requests_to_remove)
2545
2546 # Second pass: set status and free requests

2551 request.request_id not in self.finished_recving_kv_req_ids
2552 )

2554 self.failed_recving_kv_req_ids.discard(request.request_id)

2688 def reset_prefix_cache(
2689 self, reset_running_requests: bool = False, reset_connector: bool = False
2690 ) -> bool:
2691 """Reset the KV prefix cache.
2692
2693 If reset_running_requests is True, all the running requests will be
2694 preempted and moved to the waiting queue.
2695 Otherwise, this method will only reset the KV prefix cache when there
2696 is no running requests taking KV cache.
2697 """

2699 # For logging.
2700 timestamp = time.monotonic()
2701 # Invalidate all the current running requests KV's by pushing them to
2702 # the waiting queue. In this case, we can reduce the ref count of all
2703 # the kv blocks to 0 and thus we can make sure the reset is successful.
2704 # Preempt in reverse order so the requests will be added back to the
2705 # running queue in FIFO order.

2711 # + resumption in the same step, we must act as if these requests were
2712 # not scheduled in the prior step. They will be flushed from the
2713 # persistent batch in the model runner.
2714 self.prev_step_scheduled_req_ids.clear()

2716 reset_successful = self.kv_cache_manager.reset_prefix_cache()
2717 if reset_running_requests and not reset_successful:
2718 raise RuntimeError(
2719 "Failed to reset KV cache even when all the running requests are "
2720 "preempted and moved to the waiting queue. This is likely due to "
2721 "the presence of running requests waiting for remote KV transfer, "
2722 "which is not supported yet."
2723 )
2724
2725 if reset_connector:
2726 reset_successful = self.reset_connector_cache() and reset_successful

2919 def _update_waiting_for_remote_kv(self, request: Request) -> None:
2920 """
2921 KV Connector: update request state after async recv is finished.
2922
2923 When the kv transfer is ready, we cache the blocks
2924 and the request state will be moved back to WAITING from
2925 WAITING_FOR_REMOTE_KV.
2926 """
2927 assert self.connector is not None
2928
2929 if request.request_id in self.failed_recving_kv_req_ids:
2930 # Request had KV load failures; num_computed_tokens was already
2931 # updated in _update_requests_with_invalid_blocks
2932 if request.num_computed_tokens:
2933 # Cache any valid computed tokens.
2934 self.kv_cache_manager.cache_blocks(request, request.num_computed_tokens)
2935 if self.needs_kv_cache_zeroing:
2936 # The failed load left the blocks beyond the valid
2937 # prefix unwritten and their zeroing was skipped; zero
2938 # them before they are recomputed locally.
2939 self.kv_cache_manager.record_blocks_for_zeroing(
2940 request.request_id, request.num_computed_tokens
2941 )
2942 else:
2943 # No valid computed tokens, release allocated blocks.
2944 # There may be a local cache hit on retry.
2945 # (Freed blocks are re-recorded for zeroing when
2946 # reallocated, so the skipped blocks need no handling.)
2947 self.kv_cache_manager.free(request)
2948
2949 self.failed_recving_kv_req_ids.remove(request.request_id)
2950 else:
2951 # Now that the blocks are ready, actually cache them.
2952 # This will cache the blocks iff caching is enabled.
2953 self.kv_cache_manager.cache_blocks(request, request.num_computed_tokens)
2954

3027 def _update_requests_with_invalid_blocks(
3028 self,
3029 requests: Iterable[Request],
3030 invalid_block_ids: set[int],
3031 num_scheduled_tokens: dict[str, int],
3032 evict_blocks: bool = True,
3033 ) -> tuple[set[str], int, set[int]]:
3034 """
3035 Identify and update requests affected by invalid KV cache blocks.
3036
3037 This method scans the given requests, detects those with invalid blocks
3038 and adjusts their `num_computed_tokens` to the longest valid prefix.
3039 For observability, it also accumulates the total number of tokens that
3040 will need to be recomputed across all affected requests.
3041
3042 A hybrid model spreads one token span over KV cache groups that disagree
3043 on block size, so a block index maps to a different token position in
3044 each group and there is no single longest valid prefix. Those requests
3045 recompute in full instead.
3046
3047 Args:
3048 requests: The set of requests to scan for invalid blocks.
3049 invalid_block_ids: IDs of invalid blocks.
3050 num_scheduled_tokens: req_id -> number of scheduled tokens.
3051 evict_blocks: Whether to collect blocks for eviction (False for
3052 async requests which aren't cached yet).
3053
3054 Returns:
3055 tuple:
3056 - affected_req_ids (set[str]): IDs of requests impacted by
3057 invalid blocks.
3058 - total_affected_tokens (int): Total number of tokens that must
3059 be recomputed across all affected requests.
3060 - blocks_to_evict (set[int]): Block IDs to evict from cache,
3061 including invalid blocks and downstream dependent blocks.
3062 """
3063 affected_req_ids: set[str] = set()
3064 total_affected_tokens = 0
3065 blocks_to_evict: set[int] = set()
3066 # If a block is invalid and shared by multiple requests in the batch,
3067 # these requests must be rescheduled, but only the first will recompute
3068 # it. This set tracks blocks already marked for recomputation.
3069 marked_invalid_block_ids: set[int] = set()
3070 # Positions a group holds no KV for -- skipped sliding-window blocks,
3071 # and every Mamba block but the aligned snapshot -- all point at this
3072 # one block, shared by every request. It carries no request's data, so
3073 # it can neither fail for a request nor be evicted on its behalf.
3074 null_block_id = self.kv_cache_manager.block_pool.null_block.block_id
3075 for request in requests:
3076 is_affected = False
3077 marked_invalid_block = False
3078 req_id = request.request_id
3079 # One block-id list per KV cache group.
3080 req_block_ids_per_group = self.kv_cache_manager.get_block_ids(req_id)
3081 # We iterate only over blocks that may contain externally computed
3082 # tokens
3083 req_num_computed_tokens = (
3084 request.num_computed_tokens - num_scheduled_tokens.get(req_id, 0)
3085 )
3086
3087 if len(req_block_ids_per_group) > 1:
3088 # No single truncation point exists across groups, so the whole
3089 # request is recomputed. Blocks are still marked, so a request
3090 # sharing one of them is rescheduled too.
3091 request_invalid_block_ids = {
3092 block_id
3093 for req_block_ids in req_block_ids_per_group
3094 for block_id in req_block_ids
3095 if block_id != null_block_id and block_id in invalid_block_ids
3096 }
3097 if request_invalid_block_ids:
3098 marked_invalid_block_ids |= request_invalid_block_ids
3099 total_affected_tokens += req_num_computed_tokens
3100 request.num_computed_tokens = 0
3101 if evict_blocks:
3102 for req_block_ids in req_block_ids_per_group:
3103 blocks_to_evict.update(
3104 block_id
3105 for block_id in req_block_ids
3106 if block_id != null_block_id
3107 )
3108 affected_req_ids.add(req_id)
3109 continue
3110
3111 (req_block_ids,) = req_block_ids_per_group
3112
3113 req_num_computed_blocks = (
3114 req_num_computed_tokens + self.block_size - 1
3115 ) // self.block_size
3116 for idx, block_id in zip(range(req_num_computed_blocks), req_block_ids):
3117 if block_id not in invalid_block_ids:
3118 continue
3119
3120 is_affected = True
3121
3122 if block_id in marked_invalid_block_ids:
3123 # This invalid block is shared with a previous request
3124 # and was already marked for recomputation.
3125 # This means this request can still consider this block
3126 # as computed when rescheduled.
3127 # Currently this only applies to sync loading; Async
3128 # loading does not yet support block sharing
3129 continue
3130
3131 marked_invalid_block_ids.add(block_id)
3132
3133 if marked_invalid_block:
3134 # This request has already marked an invalid block for
3135 # recomputation and updated its num_computed_tokens.
3136 continue
3137
3138 marked_invalid_block = True
3139 # Truncate the computed tokens at the first failed block
3140 request.num_computed_tokens = idx * self.block_size
3141 num_affected_tokens = (
3142 req_num_computed_tokens - request.num_computed_tokens
3143 )
3144 total_affected_tokens += num_affected_tokens
3145
3146 # collect invalid block and all downstream dependent blocks
3147 if evict_blocks:
3148 blocks_to_evict.update(req_block_ids[idx:])
3149
3150 if is_affected:
3151 if not marked_invalid_block:
3152 # All invalid blocks of this request are shared with
3153 # previous requests and will be recomputed by them.
3154 # Revert to considering only cached tokens as computed.
3155 # Currently this only applies to sync loading; Async
3156 # loading does not yet support block sharing
3157 total_affected_tokens += (
3158 request.num_computed_tokens - req_num_computed_tokens
