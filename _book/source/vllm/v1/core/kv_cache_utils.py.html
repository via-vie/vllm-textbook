58# BlockHash represents the hash of a single KV-cache block used for
59# prefix caching. Treating it as a distinct type from `bytes` helps
60# catch accidental misuse when passing around raw byte strings.
61BlockHash = NewType("BlockHash", bytes)
62
63# `BlockHashWithGroupId` combines a `BlockHash` with its KV cache group ID.
64# It is represented as raw bytes for compactness and efficiency. The helper
65# functions below pack/unpack the `BlockHash` and group id into/from the key.
66BlockHashWithGroupId = NewType("BlockHashWithGroupId", bytes)

68# ExternalBlockHash is used for reproducible prefix-cache block hashing.
69# It's a union of `bytes` and `int` to keep backward compatibility
70# after we default block hashing to use sha256 bytes.
71ExternalBlockHash: TypeAlias = bytes | int

74def make_block_hash_with_group_id(
75 block_hash: BlockHash, group_id: int
76) -> BlockHashWithGroupId:
77 """Pack a `BlockHash` and group id into a `BlockHashWithGroupId`.
78
79 The group id is encoded using 4 bytes in big-endian order and appended to
80 the block hash bytes. This representation avoids creating tuples while
81 still allowing us to recover both components when needed.
82 """
83 return BlockHashWithGroupId(block_hash + group_id.to_bytes(4, "big", signed=False))

104# The hash seed for the first block of any prefix block sequence.
105#
106# For cryptographic hash algorithms it is derived deterministically from a fixed
107# default seed, so independent vLLM processes compute identical block hashes for
108# identical content and can share a prefix cache (e.g. KV cache reuse across
109# nodes) without extra configuration. This does not weaken collision resistance,
110# which for SHA-256 does not depend on keeping the seed secret; ``cache_salt``
111# remains the mechanism for intentional cache isolation.
112#
113# Non-cryptographic algorithms keep a per-process random seed, because a
114# predictable seed would let an attacker precompute colliding blocks offline
115# (see #12621). Setting PYTHONHASHSEED overrides the seed in both cases.
116#
117# The function `init_none_hash` initializes this variable globally.
118NONE_HASH: BlockHash
119
120# Fixed seed used when the PYTHONHASHSEED environment variable is not set and
121# the hash algorithm is cryptographic.
122DEFAULT_NONE_HASH_SEED = "vllm-none-hash"
123
124# Algorithms that are not collision resistant, so the seed must stay secret.
125_NON_CRYPTO_HASH_FUNCTIONS = frozenset({xxhash, xxhash_cbor})
126
127# The seed NONE_HASH was derived from, set by init_none_hash.
128_NONE_HASH_SEED: str | None = None
129
130
131def resolve_none_hash_seed(hash_fn: Callable[[Any], bytes]) -> str:
132 """Resolve the seed to derive NONE_HASH from.
133
134 PYTHONHASHSEED wins if set. Otherwise cryptographic algorithms get the
135 fixed default (shareable across processes) and non-cryptographic ones get
136 fresh random bytes, keeping the seed unpredictable where collision
137 resistance depends on it.
138 """
139 hash_seed = os.getenv("PYTHONHASHSEED")
140 if hash_seed is not None:
141 return hash_seed
142 if hash_fn in _NON_CRYPTO_HASH_FUNCTIONS:
143 return os.urandom(32).hex()
144 return DEFAULT_NONE_HASH_SEED
145
146
147def get_none_hash_seed() -> str:
148 """Return the seed NONE_HASH was derived from.
149
150 Components that must agree on NONE_HASH across processes (the P2P tier
151 advertises this during its connect handshake) read the resolved seed here
152 instead of re-deriving it, so they observe the random seed too. Falls back
153 to the deterministic seed before ``init_none_hash`` has run.
154 """
155 if _NONE_HASH_SEED is None:
156 return DEFAULT_NONE_HASH_SEED
157 return _NONE_HASH_SEED
158
159
160def init_none_hash(hash_fn: Callable[[Any], bytes]):
161 global NONE_HASH, _NONE_HASH_SEED
162
163 _NONE_HASH_SEED = resolve_none_hash_seed(hash_fn)
164 if hash_fn in _NON_CRYPTO_HASH_FUNCTIONS and os.getenv("PYTHONHASHSEED") is None:
165 logger.warning(
166 "Using a random per-process NONE_HASH seed because %s is not "
167 "collision resistant. Block hashes are therefore not reproducible "
168 "across processes; set PYTHONHASHSEED to a shared value to reuse "
169 "the prefix cache across instances, or use sha256.",
170 hash_fn.__name__,
171 )
172 NONE_HASH = BlockHash(hash_fn(_NONE_HASH_SEED))

175@dataclass(slots=True)
176class KVCacheBlock:
177 """KV-cache block metadata."""
178
179 # Block ID, ranging from 0 to num_gpu_blocks - 1.
180 block_id: int
181 # Reference count.
182 ref_cnt: int = 0
183 # Deferred frees may contain blocks from multiple pools.
184 pool: "BlockPool | None" = field(
185 default=None, repr=False, compare=False, kw_only=True
186 )
187 # The hash key (block hash + group id) of the block, only available
188 # when the block is full and cached.
189 _block_hash: BlockHashWithGroupId | None = None
190 # Number of prefix tokens covered by _block_hash. For full blocks this is
191 # the full block boundary; partial entries can end inside a cache block.
192 _block_hash_num_tokens: int | None = None
193
194 # Used to construct a doubly linked list for free blocks.
195 # These two attributes should only be manipulated by FreeKVCacheBlockQueue.
196 prev_free_block: "KVCacheBlock | None" = None
197 next_free_block: "KVCacheBlock | None" = None
198
199 # Whether the block is a null block that should never be cached.
200 is_null: bool = False
201
202 @property
203 def block_hash(self) -> BlockHashWithGroupId | None:
204 return self._block_hash
205
206 @property
207 def block_hash_num_tokens(self) -> int | None:
208 return self._block_hash_num_tokens
209
210 def set_block_hash(
211 self,
212 block_hash: BlockHashWithGroupId,
213 num_tokens: int | None = None,
214 ) -> None:
215 assert self.block_hash is None and self._block_hash_num_tokens is None, (
216 "The block already has a hash. This should not happen."
217 )
218 self._block_hash = block_hash
219 self._block_hash_num_tokens = num_tokens
220
221 def reset_hash(self):
222 """Reset the block hash when the block is evicted."""
223 self._block_hash = None
224 self._block_hash_num_tokens = None
225
226 def __repr__(self) -> str:
227 # Use block_id instead of KVCacheBlock object to avoid calling __repr__
228 # on KVCacheBlock object recursively.
229 prev_block_id = self.prev_free_block.block_id if self.prev_free_block else None
230 next_block_id = self.next_free_block.block_id if self.next_free_block else None
231 return (
232 f"KVCacheBlock(block_id={self.block_id}, "
233 f"ref_cnt={self.ref_cnt}, "
234 f"_block_hash={self._block_hash!r}, "
235 f"_block_hash_num_tokens={self._block_hash_num_tokens}, "
236 f"prev_free_block={prev_block_id}, "
237 f"next_free_block={next_block_id})"
238 )

246class FreeKVCacheBlockQueue:
247 """This class organizes a list of KVCacheBlock objects to a doubly linked
248 list of free blocks. We implement this class instead of using Python
249 builtin deque to support removing a block in the middle of the queue
250 in O(1) time. To close the performance gap to the builtin deque which is
251 implemented in C++, this class does not allocate any Python objects when
252 manipulating the linked list. Instead, this class manipulates the
253 prev_free_block and next_free_block attributes of the given blocks.
254
255 The queue is ordered by block ID in the beginning. When a block is allocated
256 and then freed, it will be appended back with the eviction order:
257 1. The least recent used block is at the front (LRU).
258 2. If two blocks have the same last accessed time (allocated by the
259 same sequence), the one with more hash tokens (the tail of a block
260 chain) is at the front.
261 Note that we maintain this order by reversing the block order when free
262 blocks of a request. This operation is outside of this class.
263
264 Args:
265 blocks: A list of KVCacheBlock objects.
266 """
267
268 def __init__(self, blocks: list[KVCacheBlock]) -> None:
269 self.num_free_blocks = len(blocks)
270
271 # Initialize doubly links of consecutive blocks
272 for i in range(self.num_free_blocks):
273 if i > 0:
274 blocks[i].prev_free_block = blocks[i - 1]
275 if i < self.num_free_blocks - 1:
276 blocks[i].next_free_block = blocks[i + 1]
277
278 # Create a fake head and a tail block for the doubly linked list to
279 # reduce branching in the code
280 #
281 # The implementation guaranteed that the fake head and tail
282 # are NEVER got popped, so we could safely assume each real blocks
283 # in the queue has prev and next blocks.
284 self.fake_free_list_head = KVCacheBlock(block_id=-1)
285 self.fake_free_list_tail = KVCacheBlock(block_id=-1)
286 if self.num_free_blocks > 0:
287 # Connect fake_head and fake_tail to the first and last block
288 # respectively.
289 self.fake_free_list_head.next_free_block = blocks[0]
290 blocks[0].prev_free_block = self.fake_free_list_head
291 self.fake_free_list_tail.prev_free_block = blocks[-1]
292 blocks[-1].next_free_block = self.fake_free_list_tail
293 else:
294 # For empty list, simply connect the fake head and tail.
295 self.fake_free_list_head.next_free_block = self.fake_free_list_tail
296 self.fake_free_list_tail.prev_free_block = self.fake_free_list_head
297
298 def popleft(self) -> KVCacheBlock:
299 """Pop the first free block and reduce num_free_blocks by 1.
300
301 Returns:
302 The first free block.
303 """
304 if (
305 self.fake_free_list_head.next_free_block is self.fake_free_list_tail
306 or self.fake_free_list_head.next_free_block is None
307 ):
308 assert self.num_free_blocks == 0, (
309 f"num_free_blocks ({self.num_free_blocks}) is out of sync "
310 "with the free list."
311 )
312 raise ValueError("No free blocks available")
313
314 first_block: KVCacheBlock = self.fake_free_list_head.next_free_block
315
316 if first_block.next_free_block is None:
317 # This should not happen if the block is from the free list.
318 # It indicates a bug in the caller's logic.
319 raise RuntimeError(
320 "Invalid block found in popleft() "
321 "which doesn't have a valid next_free_block"
322 )
323
324 # Connect fake_head and the next block of first_block (i.e. second block
325 # or fake tail).
326 self.fake_free_list_head.next_free_block = first_block.next_free_block
327 first_block.next_free_block.prev_free_block = self.fake_free_list_head
328
329 # Remove the block from the linked list.
330 first_block.prev_free_block = first_block.next_free_block = None
331
332 self.num_free_blocks -= 1
333 return first_block
334
335 def popleft_n(self, n: int) -> list[KVCacheBlock]:
336 """Pop the first n free blocks and reduce num_free_blocks by n.
337
338 Args:
339 n: The number of blocks to pop.
340
341 Returns:
342 A list of n free blocks.
343 """
344 if n == 0:
345 return []
346 assert self.num_free_blocks >= n
347 self.num_free_blocks -= n
348
349 curr_block = self.fake_free_list_head.next_free_block
350 # Pop n blocks from the head of the list
351 ret = []
352 for _ in range(n):
353 assert curr_block is not None
354 ret.append(curr_block)
355 last_block = curr_block
356 curr_block = curr_block.next_free_block
357 # Reset prev_free_block and next_free_block of all popped blocks
358 last_block.prev_free_block = None
359 last_block.next_free_block = None
360
361 if curr_block is not None:
362 # The queue is not empty, connect the fake head to
363 # the new first block.
364 self.fake_free_list_head.next_free_block = curr_block
365 curr_block.prev_free_block = self.fake_free_list_head
366 return ret
367
368 def remove(self, block: KVCacheBlock) -> None:
369 """Remove a block in the free list and reduce num_free_blocks by 1.
370
371 Args:
372 block: The block to remove.
373 """
374 if block.prev_free_block is None or block.next_free_block is None:
375 # This should not happen if the block is from the free list.
376 # It indicates a bug in the caller's logic.
377 raise RuntimeError(f"remove() called on an invalid block: {block}")
378
379 # Link the previous block to the next block.
380 block.prev_free_block.next_free_block = block.next_free_block
381 # Link the next block to the previous block.
382 block.next_free_block.prev_free_block = block.prev_free_block
383
384 # Remove the block from the linked list.
385 block.prev_free_block = block.next_free_block = None
386 self.num_free_blocks -= 1
387
388 def append(self, block: KVCacheBlock) -> None:
389 """Put a block back into the free list and increase
390 num_free_blocks by 1.
391
392 Args:
393 block: The block to append.
394 """
395 if self.fake_free_list_tail.prev_free_block is None:
396 raise RuntimeError(
397 "prev_free_block of fake_free_list_tail should always exist"
398 )
399 last_block: KVCacheBlock = self.fake_free_list_tail.prev_free_block
400
401 # Connect the new block after the last block.
402 last_block.next_free_block = block
403 block.prev_free_block = last_block
404
405 # Connect the fake tail after the new block.
406 block.next_free_block = self.fake_free_list_tail
407 self.fake_free_list_tail.prev_free_block = block
408
409 self.num_free_blocks += 1
410
411 def prepend_n(self, blocks: list[KVCacheBlock]) -> None:
412 """Put a list of blocks at the front of the free list."""
413 if len(blocks) == 0:
414 return
415
416 first_block = self.fake_free_list_head.next_free_block
417 assert first_block is not None, (
418 "next_free_block of fake_free_list_head should always exist"
419 )
420
421 prev_block = self.fake_free_list_head
422 for block in blocks:
423 block.prev_free_block = prev_block
424 prev_block.next_free_block = block
425 prev_block = block
426
427 prev_block.next_free_block = first_block
428 first_block.prev_free_block = prev_block
429
430 self.num_free_blocks += len(blocks)
431
432 def append_n(self, blocks: list[KVCacheBlock]) -> None:
433 """Put a list of blocks back into the free list
434
435 Args:
436 blocks: The blocks to append.
437 """
438 if len(blocks) == 0:
439 return
440
441 last_block = self.fake_free_list_tail.prev_free_block
442 assert last_block is not None, (
443 "prev_free_block of fake_free_list_tail should always exist"
444 )
445 # Add inter-connections between consecutive blocks
446 for block in blocks:
447 block.prev_free_block = last_block
448 last_block.next_free_block = block
449 last_block = block
450
451 # Connect the last block of <blocks> to the fake tail
452 last_block.next_free_block = self.fake_free_list_tail
453 self.fake_free_list_tail.prev_free_block = last_block
454
455 self.num_free_blocks += len(blocks)

457 def get_all_free_blocks(self) -> list[KVCacheBlock]:
458 """Get all free blocks in the free list. Mainly used for testing.
459
460 Returns:
461 A list of free blocks.
462 """
463 ret = []
464 if self.fake_free_list_head.next_free_block is None:
465 raise RuntimeError(
466 "next_free_block of fake_free_list_head should always exist"
467 )
468 # Start from the first block
469 curr_block: KVCacheBlock = self.fake_free_list_head.next_free_block
470 # As long as next_free_block is available, we haven't reached to
471 # the fake tail yet.
472 while curr_block.next_free_block is not None:
473 ret.append(curr_block)
474 curr_block = curr_block.next_free_block
475 return ret

477 def iter_blocks_after(
478 self,
479 cursor: KVCacheBlock | None,
480 ) -> Iterator[KVCacheBlock]:
481 """Iterate free blocks in eviction order after the cursor."""
482 if cursor is None:
483 curr_block = self.fake_free_list_head.next_free_block
484 else:
485 curr_block = cursor.next_free_block
486
487 while curr_block is not None and curr_block is not self.fake_free_list_tail:

489 curr_block = curr_block.next_free_block

492def _gen_mm_extra_hash_keys(
493 request: Request, start_token_idx: int, end_token_idx: int, start_mm_idx: int
494) -> tuple[list[Any], int]:
495 """Generate extra keys related to MultiModal request for block hash
496 computation. For multi-modal inputs, the extra keys are
497 (mm_hash, start_offset) that indicate a mm input contained in the
498 block and its starting offset in the block tokens.
499
500 Args:
501 request: The request object.
502 start_token_idx: The start token index of the block.
503 end_token_idx: The end token index of the block.
504 start_mm_idx: The start multi-modal index of the block.
505
506 Returns:
507 A tuple of extra keys and the next multi-modal index.
508 """
509 extra_keys: list[Any] = []
510
511 mm_features = request.mm_features
512 if not mm_features:
513 return extra_keys, start_mm_idx
514
515 # Note that we assume mm_features are sorted by mm_position.offset.
516 # We do not need to check all mm inputs if the start token index is out of
517 # range. This usually happens in the late prefill phase and decoding phase.
518 last_pos = mm_features[-1].mm_position
519 if last_pos.offset + last_pos.length <= start_token_idx:
520 return extra_keys, start_mm_idx
521
522 # Support start_mm_idx == -1 to indicate the last mm input.
523 if start_mm_idx < 0:
524 assert -start_mm_idx <= len(mm_features)
525 start_mm_idx = len(mm_features) + start_mm_idx
526
527 curr_mm_idx = start_mm_idx
528 while mm_features and curr_mm_idx < len(mm_features):
529 mm_feature = mm_features[curr_mm_idx]
530 assert mm_feature.identifier is not None
531 offset = mm_feature.mm_position.offset
532 length = mm_feature.mm_position.length
533 if end_token_idx > offset:
534 if start_token_idx >= offset + length:
535 # This block has passed the current mm input.
536 curr_mm_idx += 1
537 continue
538
539 # The block contains the current mm input. Include its offset
540 # relative to the start of the block so prefix-cache keys stay
541 # distinct when the same MM item appears at different positions
542 # within otherwise-identical placeholder blocks.
543 extra_keys.append((mm_feature.identifier, offset - start_token_idx))
544
545 if end_token_idx >= offset + length:
546 # If this block contains the end of the current mm input,
547 # move to the next mm input as this block may also contain
548 # the next mm input.
549 curr_mm_idx += 1
550 else:
551 # Otherwise this block is done with mm inputs.
552 break
553 else:
554 # This block has not reached the current mm input.
555 break
556 return extra_keys, curr_mm_idx

574def _gen_prompt_embeds_extra_hash_keys(
575 request: Request, start_token_idx: int, end_token_idx: int
576) -> list[bytes]:
577 """Generate extra keys related to prompt embeds for block hash computation.
578
579 Args:
580 request: The request object.
581 start_token_idx: The start token index of the block.
582 end_token_idx: The end token index of the block.
583
584 Returns:
585 Return a stable hash of the block prompt embeddings if prompt embeds
586 are present. Return empty list otherwise.
587 """
588 if request.prompt_embeds is None:
589 return []
590 block_range = (start_token_idx, end_token_idx)
591 embeds_hash = request._prompt_embeds_per_block_hashes.get(block_range)
592 if embeds_hash is None:
593 block_prompt_embeds = request.prompt_embeds[start_token_idx:end_token_idx]
594 # Hash prompt embeds once per block and cache on request
595 embeds_hash = hashlib.sha256(tensor_data(block_prompt_embeds)).digest()
596 request._prompt_embeds_per_block_hashes[block_range] = embeds_hash
597 return [embeds_hash]

600def generate_block_hash_extra_keys(
601 request: Request, start_token_idx: int, end_token_idx: int, start_mm_idx: int
602) -> tuple[tuple[Any, ...] | None, int]:
603 """Generate extra keys for the block hash. The extra keys can come from
604 the multi-modal inputs, request specific metadata (e.g., LoRA names), and
605 hashed data from prompt embeddings.
606
607 Args:
608 request: The request object.
609 start_token_idx: The start token index of the block.
610 end_token_idx: The end token index of the block.
611 start_mm_idx: The start multi-modal index of the block.
612
613 Returns:
614 A tuple of extra keys and the next multi-modal index.
615 """
616 mm_extra_keys: list[Any]
617 mm_extra_keys, new_start_mm_idx = _gen_mm_extra_hash_keys(
618 request, start_token_idx, end_token_idx, start_mm_idx
619 )
620 lora_extra_keys: list[str] = _gen_lora_extra_hash_keys(request)
621 cache_salt_keys: list[str] = (
622 [request.cache_salt] if (start_token_idx == 0 and request.cache_salt) else []
623 )
624 prompt_embeds_keys = _gen_prompt_embeds_extra_hash_keys(
625 request, start_token_idx, end_token_idx
626 )
627
628 extra_keys: list[Any] = (
629 lora_extra_keys + mm_extra_keys + cache_salt_keys + prompt_embeds_keys
630 )
631
632 if not extra_keys:
633 return None, new_start_mm_idx
634
635 return tuple(extra_keys), new_start_mm_idx

657 The entire tuple is used as the hash key of the block.

660 parent_block_hash = NONE_HASH
661
662 curr_block_token_ids_tuple = tuple(curr_block_token_ids)

717def resolve_kv_cache_block_sizes(
718 kv_cache_config: KVCacheConfig,
719 vllm_config: VllmConfig,
720) -> tuple[int, int]:
721 """Resolve (scheduler_block_size, hash_block_size).
722
723 - ``scheduler_block_size`` is the token-alignment invariant used by the
724 scheduler (e.g. for ``num_computed_tokens`` rounding). Single group:
725 ``cache_config.block_size * dcp``. Multiple groups: LCM of every
726 group's effective block size. Attention groups are scaled by DCP;
727 Mamba groups keep their full per-rank state and are not scaled.
728 - ``hash_block_size`` is the granularity at which ``Request.block_hashes``
729 is computed. Single group: equals scheduler block size. Multiple groups:
730 ``cache_config.prefix_match_unit`` override if set, else the GCD of
731 group block sizes; every group's block size must be divisible by it.
732 Returns the scheduler block size (i.e. disables finer hashing) if block
733 hashing is inactive or a mamba group is not using cache mode "align".
734 """
735 cache_config = vllm_config.cache_config
736 dcp = vllm_config.parallel_config.decode_context_parallel_size
737 groups = kv_cache_config.kv_cache_groups
738
739 if len(groups) <= 1:
740 bs = cache_config.block_size * dcp
741 return bs, bs
742
743 group_block_sizes = [
744 resolve_dcp_kv_block_size(g.kv_cache_spec, dcp) for g in groups
745 ]
746 scheduler_block_size = math.lcm(*group_block_sizes)
747
748 logger.info("kv cache group sizes %s", group_block_sizes)
749 logger.info("kv lcm block sizes %s", scheduler_block_size)
750
751 # Block hashes are only consumed by prefix caching and KV connectors
752 # (P/D, offloading); when neither is active, keep hash_block_size equal
753 # to the scheduler block size.
754 connector_enabled = vllm_config.kv_transfer_config is not None
755 if not (cache_config.enable_prefix_caching or connector_enabled):
756 return scheduler_block_size, scheduler_block_size
757
758 # Mamba groups outside align mode break divisibility; back off to the
759 # scheduler block size. Read the mode from the resolved group spec because
760 # its block size may have been updated independently of cache_config.
761 if any(
762 isinstance(spec, MambaSpec) and spec.mamba_cache_mode != "align"
763 for group in groups
764 for spec in iter_layer_specs(group.kv_cache_spec)
765 ):
766 return scheduler_block_size, scheduler_block_size
767
768 hashing_sizes = [
769 block_size
770 for group, block_size in zip(groups, group_block_sizes)
771 if group.kv_cache_spec.prefix_cacheable
772 ] or group_block_sizes
773 requested = cache_config.prefix_match_unit
774 hash_block_size = requested if requested is not None else math.gcd(*hashing_sizes)
775 if any(bs % hash_block_size != 0 for bs in hashing_sizes):
776 raise ValueError(
777 f"Invalid prefix_match_unit={hash_block_size}; prefix-cacheable "
778 "KV cache group block sizes must be divisible by prefix_match_unit. "
779 f"Got group block sizes={group_block_sizes}, "
780 f"prefix-cacheable={hashing_sizes}."
781 )
782 prefix_alignments = {
783 spec.tokens_per_state
784 for group in groups
785 for spec in iter_layer_specs(group.kv_cache_spec)
786 if spec.prefix_cacheable
787 and isinstance(spec.tokens_per_state, int)
788 and spec.tokens_per_state > 1
789 }
790 has_partial_mamba_group = any(
791 isinstance(spec, MambaSpec)
792 and spec.mamba_cache_mode == "align"
793 and (
794 (dcp == 1 and block_size > hash_block_size)
795 or (dcp > 1 and block_size >= hash_block_size)
796 )
797 for group, block_size in zip(groups, group_block_sizes)
798 for spec in iter_layer_specs(group.kv_cache_spec)
799 )
800 cache_hit_alignment = (
801 hash_block_size if has_partial_mamba_group else scheduler_block_size
802 )
803 if any(cache_hit_alignment % alignment for alignment in prefix_alignments):
804 raise ValueError(
805 f"Invalid prefix_match_unit={hash_block_size}; prefix-cache boundaries "
806 "must align with each spec's per-state compression. "
807 f"Got alignments={sorted(prefix_alignments)}."
808 )
809 return scheduler_block_size, hash_block_size

812def get_request_block_hasher(
813 hash_block_size: int,
814 caching_hash_fn: Callable[[Any], bytes],
815) -> Callable[[Request], list[BlockHash]]:
816 """
817 Returns a function which computes the list of un-computed block hashes
818 of a request.
819
820 Hashes are computed at ``hash_block_size`` granularity and chained over the
821 full prefix, so each hash uniquely fingerprints the prefix ending at its
822 boundary. Coarser group block sizes and partial-cache boundaries reuse
823 these hashes directly (see ``BlockHashListWithBlockSize``).
824 """
825
826 def request_block_hasher(request: Request) -> list[BlockHash]:
827 start_token_idx = len(request.block_hashes) * hash_block_size
828 num_tokens = request.num_tokens
829
830 if start_token_idx + hash_block_size > num_tokens:
831 # Early stop when there no new full blocks created.
832 return []
833
834 curr_mm_idx = 0
835 if start_token_idx > 0:
836 # Set curr_mm_idx = -1 to indicate the last mm input.
837 # Note that since we reach to this branch only when the block is
838 # completed with generated tokens, we only need to consider the
839 # last mm input.
840 curr_mm_idx = -1
841
842 prev_block_hash_value = (
843 request.block_hashes[-1] if request.block_hashes else None
844 )
845 new_block_hashes: list[BlockHash] = []
846 while True:
847 end_token_idx = start_token_idx + hash_block_size
848 if end_token_idx > num_tokens:
849 # We only hash full blocks
850 break
851
852 # MM and LoRA requests need extra keys for block-hash computation.
853 extra_keys, curr_mm_idx = generate_block_hash_extra_keys(
854 request, start_token_idx, end_token_idx, curr_mm_idx
855 )
856
857 # Compute the hash of the current block
858 block_tokens = request.all_token_ids[start_token_idx:end_token_idx]
859 block_hash = hash_block_tokens(
860 caching_hash_fn, prev_block_hash_value, block_tokens, extra_keys
861 )
862
863 new_block_hashes.append(block_hash)
864 start_token_idx += hash_block_size
865 prev_block_hash_value = block_hash
866
867 return new_block_hashes
868
869 return request_block_hasher

921def estimate_max_model_len(
922 vllm_config: VllmConfig,
923 kv_cache_spec: dict[str, KVCacheSpec],
924 available_memory: int,
925) -> int:
926 """
927 Estimates the maximum model length that can fit in the available memory
928 using binary search.
929
930 This function temporarily modifies max_model_len during estimation but
931 restores the original value before returning, ensuring no side effects.
932
933 Args:
934 vllm_config: The global VllmConfig
935 kv_cache_spec: The kv cache spec of each attention layer in the model
936 available_memory: Memory available for KV cache in bytes.
937
938 Returns:
939 The estimated maximum model length that can fit in the available memory.
940 """
941 # Save the original max_model_len to restore after estimation
942 original_max_model_len = vllm_config.model_config.max_model_len
943
944 # Define a function to check if a given model length fits in memory
945 def fits_in_memory(model_len: int) -> bool:
946 # Temporarily modify the max_model_len for this calculation
947 vllm_config.model_config.max_model_len = model_len
948 # Calculate memory needed for the given model length
949 memory_needed = max_memory_usage_bytes(vllm_config, kv_cache_spec.values())
950 return memory_needed <= available_memory
951
952 try:
953 # Binary search for the maximum model length
954 left, right = 1, original_max_model_len
955
956 # If even the smallest model length doesn't fit, return 0
957 if not fits_in_memory(left):
958 return 0
959
960 # Binary search for the maximum model length that fits
961 result = 1
962 while left <= right:
963 mid = (left + right) // 2
964 if fits_in_memory(mid):
965 result = mid
966 left = mid + 1
967 else:
968 right = mid - 1
969 return result
970 finally:
971 # Always restore the original max_model_len to avoid side effects
972 vllm_config.model_config.max_model_len = original_max_model_len

1067def get_max_concurrency_for_kv_cache_config(
1068 vllm_config: VllmConfig, kv_cache_config: KVCacheConfig
1069) -> float:
1070 """
1071 Get the maximum concurrency for the given KV cache configuration.
1072
1073 A request at max_model_len consumes whole blocks from each group's block
1074 table — cdiv(per-request bytes, page bytes) of the group's spec — and all
1075 device groups draw those block ids from one shared pool, so the per-request
1076 total is the sum over groups. The memory/page ratio is identical whether
1077 a group carries an aggregated UniformTypeKVCacheSpecs (worker config) or
1078 a representative per-layer spec (scheduler config), so both capacity
1079 call sites agree.
1080
1081 Host groups use a separate pool; the smaller concurrency limit applies.
1082 """
1083 num_blocks_per_request = 0
1084 host_blocks_per_request = 0
1085 for group in kv_cache_config.kv_cache_groups:
1086 required = cdiv(
1087 group.kv_cache_spec.max_memory_usage_bytes(vllm_config),
1088 group.kv_cache_spec.page_size_bytes,
1089 )
1090 if group.host_resident:
1091 host_blocks_per_request += required
1092 else:
1093 num_blocks_per_request += required
1094 limits = [kv_cache_config.num_blocks / num_blocks_per_request]
1095 if host_blocks_per_request:
1096 assert kv_cache_config.hisparse_host_num_blocks is not None
1097 limits.append(
1098 kv_cache_config.hisparse_host_num_blocks / host_blocks_per_request
1099 )
1100 return min(limits)

1365def unify_kv_cache_spec_page_size(
1366 kv_cache_spec: dict[str, KVCacheSpec],
1367) -> dict[str, KVCacheSpec]:
1368 """
1369 Unify the page size of the given KVCacheSpec. If the page size of all layers
1370 are the same, return the original KVCacheSpec. If not same, unify the page
1371 size by increasing the block size of layers with smaller page size. Two
1372 cases cannot be unified by block size alone and pad their physical page to
1373 the maximum instead: Mamba layers, whose page size comes from state shapes
1374 and is independent of block size; and non-MLA attention layers whose page
1375 does not evenly divide the maximum (the padded page is read through a
1376 strided view). MLA is excluded because sparse MLA indexes the cache in
1377 whole token rows (see ``flat_kv_row_view``), so its block stride can only
1378 be padded by its own row-aligned ``alignment``, not to an arbitrary page
1379 size. Raise NotImplementedError if failed to unify the page size;
1380 ``get_kv_cache_groups`` catches it to try the full-allocation fallback
1381 (e.g. MLA next to an incompatible sliding-window draft).
1382
1383 Args:
1384 kv_cache_spec: The KVCacheSpec of each attention layer in the model
1385
1386 Returns:
1387 The updated KVCacheSpec with the same page_size_bytes.
1388 """
1389 page_sizes = {layer.page_size_bytes for layer in kv_cache_spec.values()}
1390 if len(page_sizes) <= 1:
1391 # All layers have the same page size, no need to unify.
1392 return kv_cache_spec
1393
1394 max_page_size = max(page_sizes)
1395 new_kv_cache_spec = {}
1396 for layer_name, layer_spec in kv_cache_spec.items():
1397 if layer_spec.page_size_bytes == max_page_size:
1398 new_kv_cache_spec[layer_name] = layer_spec
1399 elif isinstance(layer_spec, MambaSpec):
1400 # MambaSpec's page size is determined by its state shapes and does
1401 # not scale with block_size, so pad the page instead. This is the
1402 # same padding mechanism the platform uses to align Mamba pages
1403 # with the main model's attention page size; it is needed here
1404 # when another layer (e.g. from a draft model) has a larger page
1405 # than the already-aligned Mamba page.
1406 new_spec: KVCacheSpec = replace(layer_spec, page_size_padded=max_page_size)
1407 assert new_spec.page_size_bytes == max_page_size
1408 new_kv_cache_spec[layer_name] = new_spec
1409 else:
1410 layer_page_size = layer_spec.page_size_bytes
1411 if max_page_size % layer_page_size == 0:
1412 ratio = max_page_size // layer_page_size
1413 new_block_size = layer_spec.block_size * ratio
1414 new_spec = replace(layer_spec, block_size=new_block_size)
1415 elif isinstance(layer_spec, AttentionSpec) and not isinstance(
1416 layer_spec, MLAAttentionSpec
1417 ):
1418 new_spec = replace(layer_spec, page_size_padded=max_page_size)
1419 else:
1420 raise NotImplementedError(
1421 f"Layer {layer_name}: page size is not divisible by the "
1422 "maximum page size and cannot be padded. Padding is only "
1423 "supported for non-MLA attention layers."
1424 )
1425 assert new_spec.page_size_bytes == max_page_size
1426 new_kv_cache_spec[layer_name] = new_spec
1427 return new_kv_cache_spec

1435def _get_kv_cache_groups_uniform_page_size(
1436 kv_cache_spec: dict[str, KVCacheSpec],
1437) -> list[KVCacheGroupSpec]:
1438 """
1439 Generates the KV cache groups for hybrid models with multiple
1440 attention types but still with a uniform page size (physical memory per
1441 block per layer) for all layers.
1442
1443 Detailed explanation about kv cache management of hybrid models:
1444 The layers in the models are repeated with some patterns, e.g., a model
1445 with 10 full attention layers and 20 sliding window attention layers can be
1446 regarded as repeating the pattern (1 * full, 2 * sw) 10 times.
1447 The KVCacheManager allocates different block tables for each of the 3 layers
1448 in the pattern, and repeats each of them 10 times to generate the
1449 block_table for the 30 layers in the model.
1450 Therefore, we can group the layers in the model into 3 kv_cache_groups, each
1451 of which contains 10 layers in the model.
1452 The KVCacheManager allocates the block_table for each group based on its
1453 kv_cache spec, and the model runner applies the block table to each layer
1454 in the group.
1455 For example:
1456 1. A model only uses full attention. The pattern is
1457 (num_hidden_layers * full), so there is only one group and the block table
1458 is shared by all layers. It is already handled by
1459 `_get_kv_cache_config_uniform_type`.
1460 2. A model with 10 full attention layers and 20 sliding window
1461 attention layers. There are 3 layers in the pattern (1 * full, 2 * sw), so
1462 there are 3 kv_cache_groups, each of which represents 10 layers.
1463
1464 To simplify the implementation, we make the following assumptions:
1465 1. Physical memory per block: Must be the same across all KV cache groups.
1466 Breaking this assumption is non-trivial due to memory fragmentation concerns
1467 when allocating blocks of different sizes.
1468 2. Tokens per block (block_size): Currently, we directly use
1469 `CacheConfig.block_size` for all layers. It can be extended to vary by KV
1470 cache group, but within each KV cache group, all layers must share the same
1471 block size.
1472 3. Physical memory per token per layer: This property is decided by model
1473 config. Currently we only support models that have the same physical memory
1474 per token per layer for all layers. Can be relaxed with a simple extension,
1475 but still need to keep physical memory per block the same for all groups.
1476 4. Number of layers per group: Currently assumed the same for all layers.
1477 Can be relaxed with a simple extension, but still need to keep physical
1478 memory per block the same for all groups.
1479 5. Attention type within groups: All layers in a group must share the same
1480 attention type. One exception is that, when
1481 `--disable-hybrid-kv-cache-manager` is true, the single group for full
1482 attention layers may also include attention layers using sliding window or
1483 LLaMA 4 local attention. See `unify_hybrid_kv_cache_specs` for more details.
1484 6. Support for multiple attention types: The design for most components is
1485 general to an arbitrary number of attention types. But
1486 `find_longest_cache_hit` only supports one attention type or two
1487 types of full-attention plus exactly one another type. The general
1488 implementation of this function is feasible but we don't know how to
1489 implement it cleanly yet.
1490
1491 As we assume tokens per block, physical memory per token per layer, and
1492 number of layers per group are the same now, we can ensure that physical
1493 memory per block is the same for all groups.
1494
1495 Args:
1496 kv_cache_spec: The KVCacheSpec of each attention layer in the model
1497 Returns:
1498 The generated KVCacheGroupSpecs
1499 """
1500 # Group all layers by kv_cache_spec.
1501 # E.g., 2 full attention layers and 3 sliding window attention layers,
1502 # -> (full.0, full.1), (sw.0, sw.1, sw.2).
1503 same_type_layers: dict[KVCacheSpec, list[str]] = defaultdict(list)
1504 for layer_name, layer_spec in kv_cache_spec.items():
1505 same_type_layers[layer_spec].append(layer_name)
1506
1507 # Attempt to further merge same-type layers based on whether their KV
1508 # cache specs can be merged, to minimize the group count. This benefits
1509 # situations where specs share a block layout and differ only in a
1510 # property it can reconcile (e.g. full attention layers differing only in
1511 # sliding window / attention chunk size).
1512 layer_buckets: list[list[str]] = []
1513 spec_buckets: list[list[KVCacheSpec]] = []
1514 for layer_spec, layer_names in same_type_layers.items():
1515 for names, specs in zip(layer_buckets, spec_buckets):
1516 try:
1517 # A raise means that the specs are incompatible.
1518 type(specs[0]).merge([*specs, layer_spec])
1519 except (AssertionError, ValueError):
1520 continue
1521 names.extend(layer_names)
1522 specs.append(layer_spec)
1523 break
1524 else:
1525 layer_buckets.append(list(layer_names))
1526 spec_buckets.append([layer_spec])
1527
1528 # Split each group into smaller groups, to make the number of layers in each
1529 # group identical. Add padding to the last group of each type if necessary.
1530 # E.g., (full.0, full.1), (sw.0, sw.1, sw.2)
1531 # split to 3 groups with 2 layers each:
1532 # (full.0, full.1), (sw.0, sw.2), (sw.1, padding).
1533 # FIXME(Chen): At the moment of writing this code (2025-06-02), all
1534 # open-source hybrid model follows a n:1 pattern between different attention
1535 # types (e.g., Gemma3 5:1 between sw and full, LLaMA4 3:1 between local and
1536 # full), so we can use the "1" in the n:1 pattern as the group size, which
1537 # is the minimum number of layers among all attention types. Need a better
1538 # strategy if we want to support more complex patterns (e.g., 20 full + 30
1539 # sw, where the group size should be 10).
1540 min_num_layers = min([len(layers) for layers in layer_buckets])
1541 group_size = min_num_layers
1542 max_num_layers = max([len(layers) for layers in layer_buckets])
1543 if max_num_layers < min_num_layers * 1.5:
1544 # If the number of layers is not much larger than the minimum number of
1545 # layers, use the maximum number of layers as the group size to avoid
1546 # too many padding layers. A typical example is gpt-oss-20b + eagle,
1547 # with 12 sw + 13 full. We pad it to (13 sw, 13 full) instead of
1548 # (12 sw, 24 full). 1.5 is a heuristic to avoid too many padding
1549 # layers while accommodating speculative decoding drafters that add
1550 # extra layers to one attention type.
1551 group_size = max_num_layers
1552 grouped_layers = []
1553 for layers in layer_buckets:
1554 num_padding_layers = group_size - len(layers) % group_size
1555 if num_padding_layers != group_size:
1556 logger.warning(
1557 "Add %d padding layers, may waste at most %.2f%% KV cache memory", # noqa
1558 num_padding_layers,
1559 num_padding_layers / len(layers) * 100,
1560 )
1561 num_groups = cdiv(len(layers), group_size)
1562 # In PP case, say if we have
1563 # - stage 0: full.0, sw.0, sw.1
1564 # - stage 1: full.1, sw.2, sw.3
1565 # We should have 3 groups: (full.0, full.1), (sw.0, sw.2), (sw.1, sw.3)
1566 # It can't be (full.0, full.1), (sw.0, sw.1), (sw.2, sw.3) because
1567 # the 3 groups in stage 0 will be (full.0), (sw.0, sw.1), (empty group)
1568 # and it will be padded to (full.0, padding), (sw.0, sw.1),
1569 # (padding, padding) to ensure the number of layers in each group is
1570 # the same and will cause memory waste.
1571 # To avoid this, we assign layers[i::num_groups] to the i-th group
1572 # instead of layers[i * group_size: (i + 1) * group_size]
1573 for i in range(num_groups):
1574 grouped_layers.append(layers[i::num_groups])
1575 return create_kv_cache_group_specs(kv_cache_spec, grouped_layers)

1648def get_kv_cache_config_from_groups(
1649 vllm_config: VllmConfig,
1650 kv_cache_groups: list[KVCacheGroupSpec],
1651 available_memory: int,
1652) -> KVCacheConfig:
1653 """
1654 Generate the KV cache configuration from the KV cache groups and spec
1655 of each layer.
1656
1657 Args:
1658 vllm_config: The global VllmConfig
1659 kv_cache_groups: The KV cache groups
1660 available_memory: Memory available for KV cache in bytes
1661 Returns:
1662 The generated KVCacheConfig
1663 """
1664 if len(kv_cache_groups) == 0:
1665 # Attention free models do not have KV cache.
1666 # Return num_blocks=1 as BlockPool always needs a null_block.
1667 return KVCacheConfig(
1668 num_blocks=1,
1669 kv_cache_tensors=[],
1670 kv_cache_groups=kv_cache_groups,
1671 prefix_cache_retention_interval=(
1672 vllm_config.cache_config.prefix_cache_retention_interval
1673 ),
1674 )
1675
1676 if vllm_config.attention_config.hisparse_config is not None:
1677 host_budget = get_hisparse_host_pool_bytes(vllm_config)
1678 return get_hisparse_kv_cache_config(
1679 vllm_config, kv_cache_groups, available_memory, host_budget
1680 )
1681
1682 if (glm5_layout := _glm5_next_tensor_layout(kv_cache_groups)) is not None:
1683 (
1684 attn_group,
1685 mamba_groups,
1686 mla_names,
1687 idx_names,
1688 mla_page,
1689 idx_page,
1690 tail_names,
1691 _,
1692 ) = glm5_layout
1693 bytes_per_block = len(mla_names) * mla_page + len(idx_names) * idx_page
1694 num_blocks = may_override_num_blocks(
1695 vllm_config, available_memory // bytes_per_block
1696 )
1697 size = bytes_per_block * num_blocks
1698 attn_specs = cast(
1699 UniformTypeKVCacheSpecs, attn_group.kv_cache_spec
1700 ).kv_cache_specs
1701
1702 kv_cache_tensors: list[KVCacheTensor] = []
1703
1704 def add_tensor(layer_name: str, spec: KVCacheSpec, offset: int) -> None:
1705 kv_cache_tensors.append(
1706 KVCacheTensor(
1707 size=size,
1708 layers=[layer_name],
1709 layer_stride=spec.page_size_bytes * num_blocks,
1710 block_stride=spec.page_size_bytes,
1711 offset=offset,
1712 )
1713 )
1714
1715 for index, mla_name in enumerate(mla_names):
1716 offset = index * mla_page * num_blocks
1717 add_tensor(mla_name, attn_specs[mla_name], offset)
1718 for group in mamba_groups:
1719 if index < len(group.layer_names):
1720 add_tensor(group.layer_names[index], group.kv_cache_spec, offset)
1721
1722 idx_base = len(mla_names) * mla_page * num_blocks
1723 for index, idx_name in enumerate(idx_names):
1724 offset = idx_base + index * idx_page * num_blocks
1725 add_tensor(idx_name, attn_specs[idx_name], offset)
1726 if tail_names:
1727 tail_name = tail_names[index]
1728 tail_group = next(
1729 group for group in kv_cache_groups if tail_name in group.layer_names
1730 )
1731 tail_specs = cast(
1732 UniformTypeKVCacheSpecs, tail_group.kv_cache_spec
1733 ).kv_cache_specs
1734 add_tensor(tail_name, tail_specs[tail_name], offset)
1735
1736 return KVCacheConfig(
1737 num_blocks=num_blocks,
1738 kv_cache_tensors=kv_cache_tensors,
1739 kv_cache_groups=kv_cache_groups,
1740 prefix_cache_retention_interval=(
1741 vllm_config.cache_config.prefix_cache_retention_interval
1742 ),
1743 )
1744
1745 layout = vllm_config.cache_config.get_resolved_kv_cache_layout()
1746 validate_kv_cache_layout(layout, kv_cache_groups)
1747 bytes_per_block = _get_kv_cache_bytes_per_block(kv_cache_groups)
1748 interleaved_block_stride = bytes_per_block if layout.is_block_outermost else None
1749
1750 num_blocks = available_memory // bytes_per_block
1751 num_blocks = may_override_num_blocks(vllm_config, num_blocks)
1752 size = bytes_per_block * num_blocks
1753
1754 # Groups alias from byte 0. Spec regions are laid out differently:
1755 #
1756 # block-outer (the same packing repeats for every block):
1757 # group 0: | blk 0 [ A | B | pad ] | blk 1 [ A | B | pad ] | ...
1758 # group 1: | blk 0 [ C | D ] | blk 1 [ C | D ] | ...
1759 # |<--- bytes_per_block -->|
1760 #
1761 # layer-outer (only supported for uniform page sizes or single-group models):
1762 # group 0: | A [ blk 0 | blk 1 | ... ] | B [ blk 0 | blk 1 | ... ] |
1763 # group 1: | C [ blk 0 | blk 1 | ... ] | D [ blk 0 | blk 1 | ... ] |
1764
1765 kv_cache_tensors = []
1766 for group in kv_cache_groups:
1767 group_spec = group.kv_cache_spec
1768 layers_by_spec: defaultdict[KVCacheSpec, list[str]] = defaultdict(list)
1769 if isinstance(group_spec, UniformTypeKVCacheSpecs):
1770 for layer_name, spec in group_spec.kv_cache_specs.items():
1771 layers_by_spec[spec].append(layer_name)
1772 elif group.layer_names:
1773 layers_by_spec[group_spec].extend(group.layer_names)
1774
1775 byte_offset = 0
1776 for spec, layer_names in layers_by_spec.items():
1777 layer_stride, block_stride, _, _, _ = compute_layout_strides(
1778 spec,
1779 num_blocks,
1780 len(layer_names),
1781 layout,
1782 fixed_strides=(None, interleaved_block_stride, None, None, None),
1783 )
1784 offset = (
1785 byte_offset
1786 * max(layer_stride, spec.page_size_bytes)
1787 // spec.page_size_bytes
1788 )
1789 kv_cache_tensors.append(
1790 KVCacheTensor(
1791 size=size,
1792 layers=layer_names,
1793 layer_stride=layer_stride,
1794 block_stride=block_stride,
1795 offset=offset,
1796 )
1797 )
1798 byte_offset += len(layer_names) * spec.page_size_bytes
1799
1800 return KVCacheConfig(
1801 num_blocks=num_blocks,

1943def _approximate_gcd(values: Sequence[int], *, lower_bound: int | None = None) -> int:
1944 """Pick a chunk size that minimizes total upward padding.
1945
1946 Each x is rounded up to a multiple of d:
1947
1948 x -> ceil(x / d) * d
1949
1950 Total padding is:
1951
1952 pad(d) = sum_i (ceil(x_i / d) * d - x_i)
1953
1954 We brute-force d in [lower_bound, max(values)] (fine for small lists / small
1955 maxima) and return the d with minimum padding. Ties prefer larger d.
1956 """
1957 if not values:
1958 raise ValueError("values must be non-empty")
1959 if any(x <= 0 for x in values):
1960 raise ValueError(f"values must be positive, got: {list(values)!r}")
1961
1962 min_d = max(1, lower_bound if lower_bound is not None else 1)
1963 max_d = max(values)
1964 if min_d > max_d:
1965 return min_d
1966
1967 best_d = min_d
1968 best_pad: int | None = None
1969 for d in range(min_d, max_d + 1):
1970 pad = sum((d - (x % d)) % d for x in values)
1971 if best_pad is None or pad < best_pad or (pad == best_pad and d > best_d):
1972 best_pad = pad
1973 best_d = d
1974
1975 return best_d

1978def _get_packed_kv_cache_groups(
1979 vllm_config: VllmConfig,
1980 kv_cache_spec: dict[str, KVCacheSpec],
1981) -> list[KVCacheGroupSpec] | None:
1982 """Group mixed-page-size layers for contiguous block-outermost packing.
1983
1984 Greedily buckets layers into uniform-type specs. Buckets with equal layer
1985 counts per page size are treated as a repeating layer pattern (one layer
1986 per page size) and split into groups covering the same number of pattern
1987 repeats (picked by ``_approximate_gcd`` to minimize padding), so all
1988 groups pack into the same per-block layout. Mamba buckets are additionally
1989 split to fit the block the attention buckets already need.
1990 Returns None when the layout is not block-outermost or all layers already
1991 share one page size.
1992 """
1993 layout = vllm_config.cache_config.get_resolved_kv_cache_layout()
1994 page_sizes = {spec.page_size_bytes for spec in kv_cache_spec.values()}
1995 if not layout.is_block_outermost or len(page_sizes) <= 1:
1996 return None
1997
1998 buckets: list[dict[str, KVCacheSpec]] = []
1999 for name, spec in kv_cache_spec.items():
2000 for bucket in buckets:
2001 candidate = {**bucket, name: spec}
2002 if UniformTypeKVCacheSpecs.is_uniform_type(candidate):
2003 bucket[name] = spec
2004 break
2005 else:
2006 buckets.append({name: spec})
2007
2008 bucketed = []
2009 for bucket in buckets:
2010 uniform_spec = UniformTypeKVCacheSpecs.from_specs(bucket)
2011 assert uniform_spec is not None
2012 page_size_layers: dict[int, list[str]] = defaultdict(list)
2013 for layer_name, layer_spec in bucket.items():
2014 page_size_layers[layer_spec.page_size_bytes].append(layer_name)
2015 # Only 1:1 patterns (one layer of each page size per repeat) are
2016 # supported; counts sharing a gcd > 1 (e.g. 2:1) could in principle
2017 # repeat too, but such buckets are emitted whole instead.
2018 balanced = len(set(map(len, page_size_layers.values()))) == 1
2019 bucketed.append((uniform_spec, page_size_layers, balanced))
2020
2021 # Balanced buckets that mix page sizes must stay whole, so the largest one
2022 # sets a floor on the repeats per group; larger single-size buckets are
2023 # split down toward it. No such bucket means nothing needs packing.
2024 min_repeats_per_group = max(
2025 (
2026 spec.get_max_layers_per_page_size()
2027 for spec, page_size_layers, balanced in bucketed
2028 if balanced and len(page_size_layers) > 1
2029 ),
2030 default=0,
2031 )
2032 repeats_per_group = (
2033 _approximate_gcd(
2034 [
2035 spec.get_max_layers_per_page_size()
2036 for spec, _, balanced in bucketed
2037 if balanced
2038 ],
2039 lower_bound=min_repeats_per_group,
2040 )
2041 if min_repeats_per_group
2042 else None
2043 )

2045 def num_groups_for(spec: UniformTypeKVCacheSpecs, balanced: bool) -> int:
2046 if balanced and repeats_per_group is not None:
2047 return cdiv(spec.get_max_layers_per_page_size(), repeats_per_group)
2048 return 1

2050 def widest_group_bytes(page_size_layers: dict[int, list[str]], n: int) -> int:
2051 """Page bytes of the largest of the n groups a bucket splits into."""
2052 return sum(
2053 cdiv(len(names), n) * page for page, names in page_size_layers.items()
2054 )

2056 # Bytes a block must hold however the state buckets end up split: mamba,
2057 # circular-buffer and unsplit sliding-window buckets can go down to one
2058 # state per group, every other bucket's split is fixed by the repeat pattern.
2059 def is_state_bucket(spec: UniformTypeKVCacheSpecs) -> bool:
2060 if isinstance(spec.first_spec, (MambaSpec, CircularBufferSpec)):
2061 return True
2062 return repeats_per_group is None and isinstance(
2063 spec.first_spec, SlidingWindowSpec
2064 )
2065
2066 anchor_bytes = max(
2067 (
2068 widest_group_bytes(
2069 page_size_layers,
2070 len(spec.kv_cache_specs)
2071 if is_state_bucket(spec)
2072 else num_groups_for(spec, balanced),
2073 )
2074 for spec, page_size_layers, balanced in bucketed
2075 ),
2076 default=0,
2077 )

2079 groups = []
2080 for spec, page_size_layers, balanced in bucketed:
2081 num_groups = num_groups_for(spec, balanced)

2083 # it widen the block.
2084 if anchor_bytes and is_state_bucket(spec):
2085 states_per_block = max(anchor_bytes // spec.first_spec.page_size_bytes, 1)
2086 num_groups = max(
2087 num_groups, cdiv(len(spec.kv_cache_specs), states_per_block)
2088 )
2089 if num_groups == 1:
2090 groups.append(KVCacheGroupSpec(list(spec.kv_cache_specs), spec))
2091 continue
2092
2093 pattern_repeats = list(zip(*page_size_layers.values()))
2094 for i in range(num_groups):
2095 group_layer_names = [
2096 name for repeat in pattern_repeats[i::num_groups] for name in repeat
2097 ]
2098 group_layer_specs = {
2099 name: spec.kv_cache_specs[name] for name in group_layer_names
2100 }
2101 group_spec = UniformTypeKVCacheSpecs.from_specs(group_layer_specs)
2102 assert group_spec is not None
2103 groups.append(KVCacheGroupSpec(group_layer_names, group_spec))

2399 f"{num_tokens:,}",
2400 f"{max_model_len:,}",
2401 max_concurrency,

2405def _max_memory_usage_bytes_from_groups(
2406 vllm_config: VllmConfig,
2407 kv_cache_groups: list[KVCacheGroupSpec],
2408) -> int:
2409 """
2410 Calculate maximum memory usage in bytes from KV cache groups.
2411
2412 This correctly accounts for padding in hybrid models. For example, if a
2413 model has 8 full attention layers and 9 sliding window layers, they will
2414 be padded to 9 full + 9 sliding window for uniform group sizes.
2415
2416 Each group independently claims blocks from the shared pool, so a request consumes
2417 the sum of the per-group block counts, i.e. ``bytes_per_block * total_blocks``.
2418 """
2419 if not kv_cache_groups:
2420 return 0
2421
2422 if vllm_config.attention_config.hisparse_config is not None:
2423 return get_hisparse_gpu_memory_usage(vllm_config, kv_cache_groups)
2424
2425 if (glm5_layout := _glm5_next_tensor_layout(kv_cache_groups)) is not None:
2426 (
2427 attn_group,
2428 mamba_groups,
2429 mla_names,
2430 idx_names,
2431 mla_page,
2432 idx_page,
2433 tail_names,
2434 _,
2435 ) = glm5_layout
2436 uniform_spec = cast(UniformTypeKVCacheSpecs, attn_group.kv_cache_spec)
2437 total_blocks = uniform_spec.max_memory_usage_pages(vllm_config)
2438 total_blocks += sum(
2439 cdiv(
2440 group.kv_cache_spec.max_memory_usage_bytes(vllm_config),
2441 group.kv_cache_spec.page_size_bytes,
2442 )
2443 for group in mamba_groups
2444 )
2445 if tail_names:
2446 total_blocks += 1
2447 return total_blocks * (len(mla_names) * mla_page + len(idx_names) * idx_page)
2448
2449 bytes_per_block = _pool_bytes_per_block(kv_cache_groups)
2450 total_blocks = 0
2451 for group in kv_cache_groups:
2452 spec = group.kv_cache_spec
2453 if isinstance(spec, UniformTypeKVCacheSpecs):
2454 total_blocks += spec.max_memory_usage_pages(vllm_config)
2455 else:
2456 total_blocks += cdiv(
2457 spec.max_memory_usage_bytes(vllm_config),
2458 spec.page_size_bytes,
2459 )
2460
2461 return bytes_per_block * total_blocks

2464def _estimate_max_model_len_from_groups(
2465 vllm_config: VllmConfig,
2466 kv_cache_groups: list[KVCacheGroupSpec],
2467 available_memory: int,
2468) -> int:
2469 """
2470 Binary search for the maximum model length that fits in available memory.
2471 Returns 0 if even 1 token doesn't fit.
2472 """
2473 original_max = vllm_config.model_config.max_model_len
2474 hisparse_enabled = (
2475 vllm_config.attention_config.hisparse_config is not None
2476 and bool(kv_cache_groups)
2477 )
2478
2479 def fits(model_len: int) -> bool:
2480 vllm_config.model_config.max_model_len = model_len
2481 if hisparse_enabled:
2482 try:
2483 config = get_kv_cache_config_from_groups(
2484 vllm_config, kv_cache_groups, available_memory
2485 )
2486 except ValueError:
2487 return False
2488 return get_max_concurrency_for_kv_cache_config(vllm_config, config) >= 1
2489 return (
2490 _max_memory_usage_bytes_from_groups(vllm_config, kv_cache_groups)
2491 <= available_memory
2492 )
2493
2494 try:
2495 left, right = 1, original_max
2496 if not fits(left):
2497 return 0
2498 result = 1
2499 while left <= right:
2500 mid = (left + right) // 2
2501 if fits(mid):
2502 result = mid
2503 left = mid + 1
2504 else:
2505 right = mid - 1
2506 return result
2507 finally:
2508 vllm_config.model_config.max_model_len = original_max
2509

2511def _auto_fit_max_model_len(
2512 vllm_config: VllmConfig,
2513 projected_groups_per_worker: list[list[KVCacheGroupSpec]],
2514 available_memory: list[int],
2515) -> None:
2516 """
2517 When max_model_len is set to -1, this function estimates the largest
2518 context length that can be supported with the available GPU memory.
2519 It uses binary search to find the maximum length that fits across all
2520 workers.
2521
2522 Args:
2523 vllm_config: The global VllmConfig (will be modified in-place)
2524 projected_groups_per_worker: KV cache groups projected to each worker.
2525 available_memory: Memory available for KV cache in bytes for each
2526 worker.
2527 """
2528 original_max = vllm_config.model_config.max_model_len
2529
2530 if all(not groups for groups in projected_groups_per_worker):
2531 # All workers have empty specs (attention-free model)
2532 logger.info_once(
2533 "Auto-fit max_model_len: attention-free model, "
2534 "using derived max_model_len=%d",
2535 original_max,
2536 )
2537 return
2538
2539 # Find the max_model_len that fits across all workers.
2540 auto_fit_max = original_max
2541 limiting_worker_mem = available_memory[0]
2542 for groups, avail_mem in zip(projected_groups_per_worker, available_memory):
2543 if not groups:
2544 continue
2545 worker_max = _estimate_max_model_len_from_groups(vllm_config, groups, avail_mem)
2546 if worker_max < auto_fit_max:
2547 auto_fit_max = worker_max
2548 limiting_worker_mem = avail_mem
2549
2550 if auto_fit_max <= 0:
2551 raise ValueError(
2552 "Cannot auto-fit max_model_len: not enough GPU memory available "
2553 "to serve even a single token. Try increasing `gpu_memory_utilization`."
2554 )
2555
2556 if auto_fit_max >= original_max:
2557 # The model's full context length fits in memory
2558 logger.info_once(
2559 "Auto-fit max_model_len: full model context length %d fits in "
2560 "available GPU memory",
2561 original_max,
2562 )
2563 else:
2564 # Need to reduce max_model_len to fit in memory
2565 vllm_config.model_config.max_model_len = auto_fit_max
2566 logger.info_once(
2567 "Auto-fit max_model_len: reduced from %d to %d to fit in "
2568 "available GPU memory (%s GiB available for KV cache)",
2569 original_max,
2570 auto_fit_max,
2571 format_gib(limiting_worker_mem),
2572 )

2575def _project_kv_cache_groups_to_worker(
2576 global_kv_cache_groups: list[KVCacheGroupSpec],
2577 worker_spec: dict[str, KVCacheSpec],
2578) -> list[KVCacheGroupSpec]:
2579 """
2580 Projects global KV cache groups onto a single worker's assigned layers.
2581
2582 In pipeline parallelism, each worker only owns a subset of layers. This
2583 function filters the global groups to include only layers present on the
2584 given worker, adjusting UniformTypeKVCacheSpecs accordingly.
2585
2586 Args:
2587 global_kv_cache_groups: The global KV cache groups for the whole model.
2588 worker_spec: The KV cache spec of each layer on this worker.
2589
2590 Returns:
2591 The projected KV cache groups containing only this worker's layers.
2592 """
2593 projected_groups: list[KVCacheGroupSpec] = []
2594 for group in global_kv_cache_groups:
2595 worker_layer_names = [
2596 layer_name for layer_name in group.layer_names if layer_name in worker_spec
2597 ]
2598 group_spec = group.kv_cache_spec
2599 if worker_layer_names and isinstance(group_spec, UniformTypeKVCacheSpecs):
2600 group_spec = UniformTypeKVCacheSpecs(
2601 block_size=group_spec.block_size,
2602 kv_cache_specs={
2603 layer_name: group_spec.kv_cache_specs[layer_name]
2604 for layer_name in worker_layer_names
2605 },
2606 )
2607 projected_groups.append(
2608 KVCacheGroupSpec(
2609 worker_layer_names,
2610 group_spec,
2611 is_eagle_group=group.is_eagle_group and bool(worker_layer_names),
2612 )
2613 )
2614 return projected_groups

2617def get_kv_cache_configs(
2618 vllm_config: VllmConfig,
2619 kv_cache_specs: list[dict[str, KVCacheSpec]],
2620 available_memory: list[int],
2621) -> list[KVCacheConfig]:
2622 """
2623 Generates the KV cache configurations for a model.
2624 Since we use a shared centralized controller for all workers, we need the
2625 `kv_cache_config` to be consistent across all workers to make sure
2626 the KV cache allocation can be applied to all workers. However, different
2627 workers may have different memory available, and different type of layers
2628 (when pipeline parallel is enabled). To handle the difference between
2629 workers, the current implementation is:
2630 1. Merge the KV cache specs of all workers to get the KVCacheSpecs for
2631 the whole model.
2632 2. Generate the KV cache groups based on the layer ratio of the whole model.
2633 This also handles spec unification for hybrid models.
2634 3. Handle auto-fit max_model_len and memory checks using per-worker
2635 projected groups to account for PP sharding.
2636 4. Generate the KV cache configs for each worker based on the KV cache
2637 grouping strategy. (This is reasonable because the layer ratio of
2638 different PP stages are similar.)
2639 5. Change the num_blocks of each worker to the smallest among all workers
2640 and shrink tensor sizes proportionally to avoid allocating unused memory.
2641
2642 Args:
2643 vllm_config: The global VllmConfig
2644 kv_cache_specs: List of dict[layer_name, KVCacheSpec] for each worker.
2645 available_memory: Memory available for KV cache in bytes for each
2646 worker.
2647
2648 Returns:
2649 The generated KVCacheConfigs for each worker.
2650 """
2651
2652 # Merge the KV cache specs of all workers. Different PP stages may have
2653 # different layer names, and different TP ranks of the same PP stage should
2654 # have the same KV cache spec.
2655 merged_kv_cache_specs: dict[str, KVCacheSpec] = {}
2656 for kv_cache_spec_one_worker in kv_cache_specs:
2657 for layer_name, layer_spec in kv_cache_spec_one_worker.items():
2658 if layer_name not in merged_kv_cache_specs:
2659 merged_kv_cache_specs[layer_name] = layer_spec
2660 else:
2661 assert merged_kv_cache_specs[layer_name] == layer_spec, (
2662 "The KV cache specs for the same layer are different "
2663 "across workers. This is not supported yet."
2664 )
2665
2666 # Check if the KV cache specs are registered correctly.
2667 # This is to prevent that some layers are initialized with unregistered specs.
2668 KVCacheSpecRegistry.check_kv_cache_spec_registry(merged_kv_cache_specs)
2669
2670 # When speculating with more than 1 speculative module (e.g. multi-layered MTP)
2671 # tag every SlidingWindowSpec with how many extra tokens to retain in the window.
2672 extra_retained_tokens = (
2673 vllm_config.speculative_config.num_speculative_tokens - 1
2674 if vllm_config.speculative_config is not None
2675 and vllm_config.speculative_config.use_multi_module_mtp()
2676 else 0
2677 )
2678 for layer_name, layer_spec in merged_kv_cache_specs.items():
2679 if isinstance(layer_spec, SlidingWindowSpec):
2680 merged_kv_cache_specs[layer_name] = replace(
2681 layer_spec, extra_retained_tokens=extra_retained_tokens
2682 )
2683
2684 # Get global KV cache groups. This also handles spec unification for
2685 # hybrid models when disable_hybrid_kv_cache_manager is enabled.
2686 # After this call, merged_kv_cache_specs may be modified in-place.
2687 global_kv_cache_groups = get_kv_cache_groups(vllm_config, merged_kv_cache_specs)
2688
2689 # If original_max_model_len was -1, automatically
2690 # determine the maximum model length that fits in available GPU memory.
2691 # We use per-worker projected groups to account for PP sharding.
2692 projected_groups_per_worker = [
2693 _project_kv_cache_groups_to_worker(global_kv_cache_groups, worker_spec)
2694 for worker_spec in kv_cache_specs
2695 ]
2696
2697 # If `num_gpu_blocks_override` is set, the cache size that will actually
2698 # be allocated is decoupled from the profiled `available_memory`:
2699 # `may_override_num_blocks` in `get_kv_cache_config_from_groups` clamps
2700 # `num_blocks` to the override. Reflect that in `available_memory` here so
2701 # auto-fit, the admission check, and the per-worker config builder all
2702 # plan against the same effective capacity.
2703 override = vllm_config.cache_config.num_gpu_blocks_override
2704 if override is not None:
2705 adjusted_memory: list[int] = []
2706 for groups, avail_mem in zip(projected_groups_per_worker, available_memory):
2707 if not groups:
2708 adjusted_memory.append(avail_mem)
2709 continue
2710 bytes_per_block = _pool_bytes_per_block(groups)
2711 logger.info(
2712 "Overriding num_gpu_blocks=%d with num_gpu_blocks_override=%d",
2713 avail_mem // bytes_per_block,
2714 override,
2715 )
2716 adjusted_memory.append(override * bytes_per_block)
2717 available_memory = adjusted_memory
2718
2719 if vllm_config.attention_config.hisparse_config is not None:
2720 available_memory = [min(available_memory)] * len(available_memory)
2721
2722 # Reserve the null block BlockPool permanently holds back, so auto-fit and
2723 # the capacity check both plan against usable blocks. Allocation below
2724 # still uses the full memory.
2725 check_memory = [
2726 avail_mem - _pool_bytes_per_block(groups) if groups else avail_mem
2727 for groups, avail_mem in zip(projected_groups_per_worker, available_memory)
2728 ]
2729
2730 if vllm_config.model_config.original_max_model_len == -1:
2731 _auto_fit_max_model_len(vllm_config, projected_groups_per_worker, check_memory)
2732
2733 # Check if the available memory is enough per worker.
2734 for groups, avail_mem in zip(projected_groups_per_worker, check_memory):
2735 if not groups:
2736 continue
2737 _check_enough_kv_cache_memory(
2738 avail_mem,
2739 partial(_max_memory_usage_bytes_from_groups, vllm_config, groups),
2740 vllm_config.model_config.max_model_len,
2741 partial(_estimate_max_model_len_from_groups, vllm_config, groups),
2742 )
2743
2744 kv_cache_configs: list[KVCacheConfig] = []
2745 for projected_groups, kv_cache_spec_one_worker, available_memory_one_worker in zip(
2746 projected_groups_per_worker, kv_cache_specs, available_memory
2747 ):
2748 assert sum(len(group.layer_names) for group in projected_groups) == len(
2749 kv_cache_spec_one_worker
2750 ), "Some layers are not assigned to any group."
2751 kv_cache_configs.append(
2752 get_kv_cache_config_from_groups(
2753 vllm_config, projected_groups, available_memory_one_worker
2754 )
2755 )
2756
2757 # Change the num_blocks of each rank to the smallest among all ranks.
2758 # We also need to shrink the tensor size proportionally to avoid
2759 # allocating unused memory.
2760 min_num_blocks = min(
2761 kv_cache_config.num_blocks for kv_cache_config in kv_cache_configs
2762 )
2763 for i, kv_cache_config in enumerate(kv_cache_configs):
2764 if kv_cache_config.num_blocks == min_num_blocks:
2765 continue
2766 # Re-plan with exactly the memory the smallest rank can afford, so
2767 # strides and offsets stay consistent with the shrunken allocation.
2768 groups = kv_cache_config.kv_cache_groups
2769 kv_cache_configs[i] = get_kv_cache_config_from_groups(
2770 vllm_config, groups, min_num_blocks * _pool_bytes_per_block(groups)
2771 )
2772
2773 return kv_cache_configs

2776class BlockHashListWithBlockSize:
2777 """
2778 Convert block-hash granularity from `hash_block_size` to `target_block_size`.
2779 Used when KV cache groups have different block sizes: `hash_block_size`
2780 is the size used to compute the original `block_hashes`; `target_block_size`
2781 is the group's actual block size.
2782
2783 Currently, only scaling up by an integer factor is supported (i.e.,
2784 `target_block_size` is a multiple of `hash_block_size`). Conversion is
2785 performed lazily on access for efficiency. Each `hash_block_size` hash is
2786 already chained over its entire prefix, so the hash at the last
2787 `hash_block_size` boundary of a `target_block_size` block uniquely
2788 fingerprints that block's prefix; we use it directly.
2789
2790 Example (`hash_block_size` = 16, `target_block_size` = 32):
2791 the second 16-size hash already covers tokens 0-31, so it is the 32-size
2792 hash:
2793
2794 Block hashes with block_size 16:
2795 | Token Range | 0-15 | 16-31 | 32-47 | 48-63 |
2796 |-------------|------|-------|-------|-------|
2797 | Hash | A | B | C | D |
2798
2799 Block hashes with block_size 32:
2800 | Token Range | 0-31 | 32-63 |
2801 |-------------|------|-------|
2802 | Hash | B | D |
2803
2804 Args:
2805 block_hashes: Block hashes to convert, computed at `hash_block_size`.
2806 hash_block_size: Block size at which `block_hashes` were computed.
2807 target_block_size: Desired block size; must be a multiple of `hash_block_size`.
2808 """
2809
2810 def __init__(
2811 self,
2812 block_hashes: list[BlockHash],
2813 hash_block_size: int,
2814 target_block_size: int,
2815 ):
2816 self.block_hashes = block_hashes
2817 assert target_block_size % hash_block_size == 0
2818 self.scale_factor = target_block_size // hash_block_size
2819
2820 def __len__(self) -> int:
2821 return len(self.block_hashes) // self.scale_factor
2822
2823 @overload
2824 def __getitem__(self, idx: int) -> BlockHash: ...
2825
2826 @overload
2827 def __getitem__(self, idx: slice) -> list[BlockHash]: ...
2828
2829 def __getitem__(self, idx):
2830 if isinstance(idx, int):
2831 return self._get_value_at(idx)
2832
2833 if isinstance(idx, slice):
2834 start, stop, step = idx.indices(len(self))
2835 return [self._get_value_at(i) for i in range(start, stop, step)]
2836
2837 raise TypeError(f"Invalid index type: {type(idx)!r}")
2838
2839 def __iter__(self) -> Iterator[BlockHash]:
2840 for i in range(len(self)):
2841 yield self._get_value_at(i)
2842
2843 def _get_value_at(self, idx: int) -> BlockHash:
2844 # The last hash_block_size hash within the target block already chains
2845 # over the whole prefix, so it is the target block's hash.
2846 return self.block_hashes[(idx + 1) * self.scale_factor - 1]
2847
2848
2849BlockHashList = list[BlockHash] | BlockHashListWithBlockSize
2850
2851
2852def resolve_block_hashes(
2853 block_hashes: BlockHashList,
2854 hash_block_size: int,
2855 block_size: int,
2856 *,
2857 supports_fine_grained_hash_lookup: bool = False,
2858 alignment_tokens: int | None = None,
2859) -> BlockHashList:
2860 """Resolve the block-hash view at ``block_size``.
2861
2862 When ``block_size`` equals ``hash_block_size``, reuse the precomputed block
2863 hashes directly; otherwise view them at ``block_size`` granularity.
2864 Fine-grained lookup keeps the original hashes for partial cache hits.
2865 """
2866 if block_size == hash_block_size:
2867 return block_hashes
2868 if isinstance(block_hashes, BlockHashListWithBlockSize):
2869 # Already a block-size view
2870 assert block_hashes.scale_factor == block_size // hash_block_size
2871 return block_hashes
2872 # Fine-grained partial hits keep the raw hashes. The caller passes
2873 # alignment_tokens = hash_block_size to enable them, else >= block_size.
2874 if (
2875 supports_fine_grained_hash_lookup
2876 and alignment_tokens is not None
2877 and alignment_tokens < block_size
2878 and block_size % alignment_tokens == 0
2879 ):
2880 return block_hashes
2881 assert block_size % hash_block_size == 0
2882 return BlockHashListWithBlockSize(block_hashes, hash_block_size, block_size)
