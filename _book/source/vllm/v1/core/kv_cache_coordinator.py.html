34def _validate_prefix_cache_retention_interval(
35 retention_interval: int | None,
36 scheduler_block_size: int,
37 kv_cache_config: KVCacheConfig,
38) -> None:
39 if retention_interval is None:
40 return
41
42 # Retention sparsifies sliding-window and Mamba (linear-attention)
43 # checkpoints; full-attention and chunked-local groups cache densely and
44 # ignore it (their hit granularity must stay fine).
45 if not any(
46 isinstance(g.kv_cache_spec, (SlidingWindowSpec, MambaSpec))
47 for g in kv_cache_config.kv_cache_groups
48 ):
49 if retention_interval == 0:
50 return
51 raise ValueError(
52 "prefix_cache_retention_interval is set but this model has "
53 "no sliding-window or Mamba KV cache group, so retention has no "
54 "effect. Set it to 0 (it only applies to sliding-window and Mamba "
55 "attention)."
56 )
57
58 if retention_interval < 0 or retention_interval % scheduler_block_size != 0:
59 raise ValueError(
60 f"prefix_cache_retention_interval ({retention_interval}) "
61 "must be non-negative and a multiple of scheduler_block_size "
62 f"({scheduler_block_size})."
63 )

73 def __init__(
74 self,
75 kv_cache_config: KVCacheConfig,
76 max_model_len: int,
77 max_in_flight_tokens: int,
78 use_eagle: bool,
79 enable_caching: bool,
80 enable_kv_cache_events: bool,
81 dcp_world_size: int,
82 pcp_world_size: int,
83 scheduler_block_size: int,
84 hash_block_size: int,
85 metrics_collector: KVCacheMetricsCollector | None = None,
86 num_prefill_lookahead: int = 0,
87 ):
88 self.kv_cache_config = kv_cache_config
89 # The scheduling granularity (LCM of all group block sizes), must be a multiple
90 # of the hash_block_size and the block size of each group.
91 assert scheduler_block_size % hash_block_size == 0 and all(
92 scheduler_block_size % g.kv_cache_spec.block_size == 0
93 for g in kv_cache_config.kv_cache_groups
94 )
95 self.scheduler_block_size = scheduler_block_size
96 self.num_reprefillable_tokens = max(0, num_prefill_lookahead - 1)
97
98 self.block_pool = BlockPool(
99 num_gpu_blocks=kv_cache_config.num_blocks,
100 enable_caching=enable_caching,
101 hash_block_size=hash_block_size,
102 enable_kv_cache_events=enable_kv_cache_events,
103 metrics_collector=metrics_collector,
104 )
105
106 # KV cache group indices that get the EAGLE last-block drop.
107 self.eagle_group_ids: set[int] = {
108 i for i, g in enumerate(kv_cache_config.kv_cache_groups) if g.is_eagle_group
109 }
110 # Conservatively fall back to flag all groups when no group is flagged.
111 if use_eagle and not self.eagle_group_ids:
112 self.eagle_group_ids = set(range(len(kv_cache_config.kv_cache_groups)))
113
114 # During chunked prefill with EAGLE, the single next prefill lookahead
115 # token past the chunk boundary is combined with the final hidden state
116 # and written to the KV cache. Therefore, the final chunk token must be
117 # excluded from prefix cache hits to prevent requests from acquiring the
118 # KV cache slot polluted with the next prefill token, which may or may not
119 # be present after the matching prefix. The last-block drop handles this
120 # edge case. During multi-module MTP, the issue generalizes to a prefill
121 # lookahead of num_speculative_tokens, so the dropped tail must be large
122 # enough to contain them. Hits land on scheduler-block boundaries (see
123 # `_cache_hit_alignment_tokens`), so the excluded tail is
124 # scheduler_block_size, not the group's own block size.
125 if (
126 enable_caching
127 and self.eagle_group_ids
128 and scheduler_block_size < num_prefill_lookahead
129 ):
130 raise ValueError(
131 f"Multi-module MTP with prefix caching requires scheduler_block_size"
132 f" (={scheduler_block_size}) >= num_speculative_tokens"
133 f" (={num_prefill_lookahead})."
134 )
135
136 self.single_type_managers = tuple(
137 get_manager_for_kv_cache_spec(
138 kv_cache_spec=kv_cache_group.kv_cache_spec,
139 max_in_flight_tokens=max_in_flight_tokens,
140 max_model_len=max_model_len,
141 block_pool=self.block_pool,
142 role=kv_cache_group.role,
143 enable_caching=enable_caching,
144 kv_cache_group_id=i,
145 dcp_world_size=dcp_world_size_for_kv_cache_spec(
146 kv_cache_group.kv_cache_spec, dcp_world_size
147 ),
148 pcp_world_size=pcp_world_size,
149 scheduler_block_size=self.scheduler_block_size,
150 needs_kv_cache_zeroing=self.kv_cache_config.needs_kv_cache_zeroing,
151 )
152 for i, kv_cache_group in enumerate(self.kv_cache_config.kv_cache_groups)
153 )
154 # Match Mamba checkpoints to Eagle's attention replay boundary.
155 if use_eagle:
156 for manager in self.single_type_managers:
157 if isinstance(manager, MambaManager):
158 manager.drop_eagle_checkpoint_block = True
159 self.group_block_sizes = tuple(
160 manager.block_size for manager in self.single_type_managers
161 )
162
163 # A positive retention interval must be a multiple of the base hit granularity
164 # (``scheduler_block_size``) to land on real cache-hit boundaries.
165 # 0 = keep only the latest replay boundary; None = dense;
166 self.retention_interval = kv_cache_config.prefix_cache_retention_interval
167 _validate_prefix_cache_retention_interval(
168 self.retention_interval, self.scheduler_block_size, kv_cache_config
169 )

171 def get_num_blocks_to_allocate(
172 self,
173 request_id: str,
174 num_tokens: int,
175 new_computed_blocks: tuple[Sequence[KVCacheBlock], ...],
176 num_encoder_tokens: int,
177 total_computed_tokens: int,
178 num_local_computed_tokens: int,
179 num_tokens_main_model: int,
180 apply_admission_cap: bool = False,
181 ) -> int:
182 """
183 Get the number of device blocks needed to be allocated for the request.
184
185 Args:
186 request_id: The request ID.
187 num_tokens: The total number of tokens that need a slot (including
188 tokens that are already allocated).
189 new_computed_blocks: The new computed blocks just hitting the
190 prefix caching.
191 num_encoder_tokens: The number of encoder tokens for allocating
192 blocks for cross-attention.
193 total_computed_tokens: Include both local and external tokens.
194 num_local_computed_tokens: The number of local prefix-cache computed
195 tokens.
196 num_tokens_main_model: The number of tokens for the main model (aka target
197 model in spec decode). w/o spec decode, it is num_tokens;
198 with spec decode, it is num_tokens - num_lookahead_tokens.
199 apply_admission_cap: If True, apply the recycling-aware
200 per-request admission cap (SWA / chunked-local). Set only by
201 the full-sequence admission gate; per-step allocation must
202 leave it False so the predictor matches `allocate_new_blocks`.
203
204 Returns:
205 The number of blocks to allocate.
206 """
207 num_blocks_to_allocate = 0
208 for i, manager in enumerate(self.single_type_managers):
209 if isinstance(manager, CrossAttentionManager):
210 # For cross-attention, we issue a single static allocation
211 # of blocks based on the number of encoder input tokens.
212 num_blocks_to_allocate += manager.get_num_blocks_to_allocate(
213 request_id,
214 num_encoder_tokens,
215 [],
216 0,
217 0,
218 num_encoder_tokens,
219 apply_admission_cap=apply_admission_cap,
220 )
221 else:
222 num_blocks_to_allocate += manager.get_num_blocks_to_allocate(
223 request_id,
224 num_tokens,
225 new_computed_blocks[i],
226 total_computed_tokens,
227 num_local_computed_tokens,
228 num_tokens_main_model,
229 apply_admission_cap=apply_admission_cap,
230 )
231 return num_blocks_to_allocate

233 def allocate_new_computed_blocks(
234 self,
235 request_id: str,
236 new_computed_blocks: tuple[Sequence[KVCacheBlock], ...],
237 num_local_computed_tokens: int,
238 num_external_computed_tokens: int,
239 ) -> None:
240 """
241 Add the new computed blocks to the request. Optionally allocate new
242 blocks for external computed tokens (if any).
243
244 Args:
245 request_id: The request ID.
246 new_computed_blocks: The new computed blocks just hitting the
247 prefix cache.
248 num_local_computed_tokens: The number of local computed tokens.
249 num_external_computed_tokens: The number of external computed tokens.
250 """
251 # A running request is already tracked in num_cached_block and won't
252 # have new prefix-cache hits, so this is a no-op for it.
253 if any(
254 request_id in manager.num_cached_block
255 for manager in self.single_type_managers
256 ):
257 assert all(len(blocks) == 0 for blocks in new_computed_blocks)
258 return
259
260 # Two-phase allocation (issue #33775): first touch every group's local
261 # cache-hit blocks, then allocate external blocks for every group. This
262 # ensures an earlier group's external `get_new_blocks` cannot evict a
263 # later group's not-yet-touched cache-hit blocks.
264 for i, manager in enumerate(self.single_type_managers):
265 manager.add_local_computed_blocks(
266 request_id,
267 new_computed_blocks[i],
268 num_local_computed_tokens,
269 num_external_computed_tokens,
270 )
271 if num_external_computed_tokens > 0:
272 for manager in self.single_type_managers:
273 manager.allocate_external_computed_blocks(
274 request_id,
275 num_local_computed_tokens,
276 num_external_computed_tokens,
277 )

279 def allocate_new_blocks(
280 self,
281 request_id: str,
282 num_tokens: int,
283 num_tokens_main_model: int,
284 num_encoder_tokens: int = 0,
285 ) -> tuple[list[KVCacheBlock], ...]:
286 """
287 Allocate new blocks for the request to give it at least `num_tokens`
288 token slots.
289
290 Args:
291 request_id: The request ID.
292 num_tokens: The total number of tokens that need a slot (including
293 tokens that are already allocated).
294 num_tokens_main_model: The number of tokens for the main model (aka target
295 model in spec decode). w/o spec decode, it is num_tokens;
296 with spec decode, it is num_tokens - num_lookahead_tokens.
297 num_encoder_tokens: The number of encoder tokens for allocating
298 blocks for cross-attention.
299
300 Returns:
301 The new allocated blocks.
302 """
303 return tuple(
304 manager.allocate_new_blocks(
305 request_id,
306 num_encoder_tokens
307 if isinstance(manager, CrossAttentionManager)
308 else num_tokens,
309 num_tokens_main_model,
310 )
311 for manager in self.single_type_managers
312 )

314 def get_replay_boundaries(self, request: Request) -> tuple[int, ...]:
315 """Positions a later request replaying this prompt can resume at.
316
317 A hit is the shortest across all groups, so every group retains state
318 at each position; EAGLE groups also keep the block above, which they
319 match and drop back from (see ``reachable_block_mask``).
320
321 Two positions are reachable: a resend of the identical prompt is capped
322 at ``num_tokens - 1`` (its last token is recomputed for logits), a
323 longer sibling matches the final aligned block. They differ only on a
324 block-aligned prompt, where retaining just the higher one collapses the
325 resend's hit to 0. The alignment is the scheduler block size, not the
326 finer hash granularity, which would over-estimate the reach.
327 """
328 if not self.eagle_group_ids:
329 return (request.num_prompt_tokens - 1,)
330 block = self.scheduler_block_size
331 resend = (request.num_prompt_tokens - 1) // block * block
332 extension = request.num_prompt_tokens // block * block
333 return tuple(sorted({max(resend - block, 0), max(extension - block, 0)}))

335 def cache_blocks(self, request: Request, num_computed_tokens: int) -> None:
336 """
337 Cache the blocks for the request.
338
339 Args:
340 request: The request.
341 num_computed_tokens: The total number of tokens
342 that need to be cached
343 (including tokens that are already cached).
344 """
345 boundaries = self.get_replay_boundaries(request)
346 for manager in self.single_type_managers:
347 if not manager.enable_caching:
348 continue
349 # Only cache tokens with finalized KV. The last num_reprefillable_tokens
350 # tokens can be re-prefilled during multi-module MTP.
351 num_tokens_to_cache = max(
352 0, num_computed_tokens - self.num_reprefillable_tokens
353 )
354 manager.cache_blocks(
355 request,
356 num_tokens_to_cache,
357 retention_interval=self.retention_interval,
358 replay_boundaries=boundaries,
359 )

361 def emit_cached_block_events(
362 self, request: Request, computed_blocks: tuple[list[KVCacheBlock], ...]
363 ) -> None:
364 for group_idx, group_blocks in enumerate(computed_blocks):
365 if group_blocks:
366 manager = self.single_type_managers[group_idx]
367 group = self.kv_cache_config.kv_cache_groups[group_idx]
368 manager.block_pool.emit_cached_block_events(
369 request,
370 len(group_blocks),
371 group.kv_cache_spec.block_size,
372 group_idx,
373 )

392 def pop_blocks_for_free(self, request_id: str) -> list[KVCacheBlock]:
393 """
394 Pop the request's bookkeeping from all single-type managers and
395 return its blocks without returning them to the block pool. The
396 caller must eventually pass the returned blocks to
397 `block_pool.free_blocks`, freeing them in reverse order (so that
398 tail blocks are evicted first).
399
400 Args:
401 request_id: The request ID.
402
403 Returns:
404 The request's blocks in allocation order.
405 """
406 blocks: list[KVCacheBlock] = []
407 for manager in self.single_type_managers:
408 blocks.extend(manager.pop_blocks_for_free(request_id))
409 return blocks

585 hit_blocks, hit_length = self.single_type_managers[0].find_longest_cache_hit(

606 is necessarily decided for the whole spec group.

611 manager_cls: type[SingleTypeKVCacheManager]

615class HybridKVCacheCoordinator(KVCacheCoordinator):
616 """
617 KV cache coordinator for hybrid models with multiple KV cache types, and
618 thus multiple kv cache groups.
619 """
620
621 def __init__(
622 self,
623 kv_cache_config: KVCacheConfig,
624 max_model_len: int,
625 max_in_flight_tokens: int,
626 use_eagle: bool,
627 enable_caching: bool,
628 enable_kv_cache_events: bool,
629 dcp_world_size: int,
630 pcp_world_size: int,
631 scheduler_block_size: int,
632 hash_block_size: int,
633 metrics_collector: KVCacheMetricsCollector | None = None,
634 num_prefill_lookahead: int = 0,
635 allow_partial_hash_hits: bool = True,
636 ):
637 super().__init__(
638 kv_cache_config,
639 max_model_len,
640 max_in_flight_tokens,
641 use_eagle,
642 enable_caching,
643 enable_kv_cache_events,
644 dcp_world_size=dcp_world_size,
645 pcp_world_size=pcp_world_size,
646 scheduler_block_size=scheduler_block_size,
647 hash_block_size=hash_block_size,
648 metrics_collector=metrics_collector,
649 num_prefill_lookahead=num_prefill_lookahead,
650 )
651 # hash_block_size: the block size used to compute block hashes.
652 # The actual block size usually equals hash_block_size, but in cases where
653 # different KV cache groups have different block sizes, the actual block size
654 # can be a multiple of hash_block_size.
655 self.hash_block_size = hash_block_size
656 self.dcp_world_size = dcp_world_size
657 # Only groups that participate in prefix caching must satisfy the
658 # divisibility constraint; groups that opt out (e.g. GLM-5.3-Flash kpool
659 # tail, block_size=kpool) are scratch buffers and excluded.
660 cacheable_block_sizes = [
661 manager.block_size
662 for manager, group in zip(
663 self.single_type_managers, kv_cache_config.kv_cache_groups
664 )
665 if group.kv_cache_spec.prefix_cacheable
666 ]
667 assert all(
668 block_size % hash_block_size == 0 for block_size in cacheable_block_sizes
669 ), (
670 "Each KV cache group's real block_size must be divisible by "
671 f"hash_block_size. block_sizes={cacheable_block_sizes}, "
672 f"hash_block_size={hash_block_size}"
673 )
674 assert pcp_world_size == 1, "PCP not support hybrid attn now."
675 if dcp_world_size > 1:
676 # DCP shards full-attention KV across ranks and replicates Mamba
677 # state; other spec types (e.g. sliding window) have no DCP-aware
678 # handling yet, so reject them explicitly.
679 for g in kv_cache_config.kv_cache_groups:
680 assert isinstance(g.kv_cache_spec, (FullAttentionSpec, MambaSpec)), (
681 "DCP with hybrid KV cache layouts only supports "
682 "full-attention and Mamba groups, got: "
683 f"{type(g.kv_cache_spec).__name__}."
684 )
685 # Fine-grained hash hits require Mamba "align" and compatible cache
686 # managers in every group. TP needs hashing finer than the Mamba block;
687 # DCP accepts equality because it scales the effective full-attention
688 # block instead.
689 has_partial_mamba_group = any(
690 isinstance(g.kv_cache_spec, MambaSpec)
691 and g.kv_cache_spec.mamba_cache_mode == "align"
692 and (
693 (dcp_world_size == 1 and g.kv_cache_spec.block_size > hash_block_size)
694 or (
695 dcp_world_size > 1 and g.kv_cache_spec.block_size >= hash_block_size
696 )
697 )
698 for g in kv_cache_config.kv_cache_groups
699 )
700 self.enable_partial_hash_hits = (
701 allow_partial_hash_hits and has_partial_mamba_group
702 )
703 if self.enable_partial_hash_hits:
704 unsupported_partial_hit_managers = {
705 type(manager).__name__
706 for manager, group in zip(
707 self.single_type_managers, kv_cache_config.kv_cache_groups
708 )
709 if group.kv_cache_spec.prefix_cacheable
710 and not manager.supports_fine_grained_hash_lookup
711 and manager.block_size != hash_block_size
712 }
713 if unsupported_partial_hit_managers:
714 self.enable_partial_hash_hits = False
715 logger.warning_once(
716 "Disabling fine-grained prefix-cache hits because these KV "
717 "cache managers require block-aligned lookups: %s.",
718 ", ".join(sorted(unsupported_partial_hit_managers)),
719 )
720 cache_hit_alignment_tokens = self._cache_hit_alignment_tokens
721 for manager in self.single_type_managers:
722 manager.cache_hit_alignment_tokens = cache_hit_alignment_tokens
723 self.verify_and_split_kv_cache_groups()
724
725 @property
726 def _cache_hit_alignment_tokens(self) -> int:
727 # Fine-grained partial hits may return hash-block-aligned lengths;
728 # otherwise it must stay scheduler-block-aligned.
729 return (
730 self.hash_block_size
731 if self.enable_partial_hash_hits
732 else self.scheduler_block_size
733 )

735 def verify_and_split_kv_cache_groups(self) -> None:
736 """
737 Groups KV cache groups by their spec type for efficient batch processing
738 during cache hit lookup.
739 """
740 self.attention_groups: list[SpecGroup] = []
741 for i, g in enumerate(self.kv_cache_config.kv_cache_groups):
742 # Skip groups that opt out of prefix caching (e.g. GLM-5.3-Flash
743 # kpool tail): their blocks are per-request scratch, never
744 # shareable, so they must not participate in hit lookup (their
745 # manager-level hooks already no-op). Their slot in the per-group
746 # hit tuple stays empty.
747 if not g.kv_cache_spec.prefix_cacheable:
748 continue
749 manager_cls = self.single_type_managers[i].__class__
750 spec = g.kv_cache_spec
751 use_eagle = i in self.eagle_group_ids
752
753 # Try to find an existing group with the same spec
754 for idx, group in enumerate(self.attention_groups):
755 if (
756 group.spec == spec
757 and group.manager_cls is manager_cls
758 and self.single_type_managers[group.group_ids[0]].block_pool
759 is self.single_type_managers[i].block_pool
760 ):
761 group.group_ids.append(i)
762 if use_eagle and not group.use_eagle:
763 self.attention_groups[idx] = group._replace(use_eagle=True)
764 break
765 else:
766 self.attention_groups.append(
767 SpecGroup(spec, [i], manager_cls, use_eagle)
768 )
769
770 assert self.attention_groups, (
771 "HybridKVCacheCoordinator requires at least one cacheable group."
772 )
773
774 # Put full attention first: its efficient left-to-right scan provides
775 # a tighter initial bound, reducing work for subsequent groups.
776 self.attention_groups.sort(
777 key=lambda g: not isinstance(g.spec, FullAttentionSpec)
778 )
779
780 # Dense reference group for per-group lookups (None when the model
781 # has no full-attention layers): full attention is downward-closed,
782 # so any group reporting a longer per-group hit implies the union of
783 # per-group hits is not consistent at a single boundary (#46453).
784 first = self.attention_groups[0]
785 self.full_attention_group_id: int | None = (
786 first.group_ids[0] if isinstance(first.spec, FullAttentionSpec) else None
787 )
788
789 # Propagate the eagle bit to each manager (default to ``use_eagle=False``).
790 for group in self.attention_groups:
791 if group.use_eagle:
792 for gid in group.group_ids:
793 self.single_type_managers[gid].use_eagle = True
794
795 def _align_cacheable(self, num_tokens: int) -> int:
796 """Largest prefix of ``num_tokens`` a future cache hit could match.
797
798 Hits are ``scheduler_block_size``-aligned (see
799 ``find_longest_cache_hit``) unless fine-grained partial hash hits are
800 enabled, in which case no rounding applies -- rounding even to
801 ``hash_block_size`` would re-register a privatized Mamba tail.
802 """
803 if self.enable_partial_hash_hits:
804 return num_tokens
805 return round_down(num_tokens, self.scheduler_block_size)
806
807 def cache_blocks(self, request: Request, num_computed_tokens: int) -> None:
808 cached_num_computed_tokens = self._align_cacheable(num_computed_tokens)
809 boundaries = self.get_replay_boundaries(request)
810 for manager in self.single_type_managers:
811 if not manager.enable_caching:
812 continue
813 num_tokens_to_cache = cached_num_computed_tokens
814 # EAGLE groups match one block past each aligned boundary and drop
815 # it, so make that lookahead block eligible to be cached.
816 if manager.use_eagle and cached_num_computed_tokens > 0:
817 # Only cache tokens with finalized KV. The last
818 # num_reprefillable_tokens tokens can be re-prefilled during
819 # multi-module MTP.
820 num_finalized_computed_tokens = max(
821 0, num_computed_tokens - self.num_reprefillable_tokens
822 )
823 cached_num_finalized_computed_tokens = self._align_cacheable(
824 num_finalized_computed_tokens
825 )
826 num_tokens_to_cache = min(
827 num_finalized_computed_tokens,
828 cached_num_finalized_computed_tokens + manager.block_size,
829 )
830 # The manager already knows the fine hit granularity
831 # (``scheduler_block_size``); retention is passed separately so it
832 # can keep both the coarse segment tails and the fine replay
833 # boundary (which needs the fine value).
834 manager.cache_blocks(
835 request,
836 num_tokens_to_cache,
837 retention_interval=self.retention_interval,
838 replay_boundaries=boundaries,
839 )

841 def find_longest_cache_hit(
842 self,
843 block_hashes: list[BlockHash],
844 max_cache_hit_length: int,
845 ) -> tuple[tuple[list[KVCacheBlock], ...], int, int]:
846 """
847 Find the longest cache hit using an iterative fixed-point algorithm.
848
849 Each attention type either accepts the current candidate length or
850 reduces it. If any type reduces the length, restart checks over all
851 types. This converges because length monotonically decreases and is
852 bounded below by 0.
853
854 Args:
855 block_hashes: The block hashes of the request.
856 max_cache_hit_length: The maximum length of the cache hit.
857
858 Returns:
859 A tuple containing:
860 - A tuple of the cache hit blocks for each single type manager.
861 - The number of tokens of the reconciled (combined) cache hit.
862 - ``num_uncached_common_prefix_tokens``: a shared prefix that a
863 sparse-retention group has not cached yet (0 unless hybrid).
864 """
865
866 num_groups = len(self.kv_cache_config.kv_cache_groups)
867 hit_length = max_cache_hit_length
868 longest_hit_length = 0
869 hit_blocks_by_group: list[list[KVCacheBlock] | None] = [None] * num_groups
870 hit_length_by_group: list[int] = [0] * num_groups
871
872 # Simple hybrid (1 full attn + 1 other): one iteration suffices.
873 # Full attn is always first if it exists.
874 is_simple_hybrid = len(self.attention_groups) == 2 and isinstance(
875 self.attention_groups[0].spec, FullAttentionSpec
876 )
877
878 # Attention-group indices whose EAGLE drop is verified at the current
879 # ``curr_hit_length``. Each eagle group applies the drop at most once
880 # per candidate length (see issue #32802).
881 eagle_verified: set[int] = set()
882
883 while True:
884 curr_hit_length = hit_length
885
886 for idx, (spec, group_ids, manager_cls, use_eagle) in enumerate(
887 self.attention_groups
888 ):
889 first_group_id = group_ids[0]
890 # DCP/PCP shard each block's KV across ranks, so the manager's
891 # effective block size may exceed the spec's.
892 group_block_size = self.single_type_managers[first_group_id].block_size
893 cached_blocks = hit_blocks_by_group[first_group_id]
894 if isinstance(spec, FullAttentionSpec) and cached_blocks is not None:
895 # Full attention is downward-closed: we only need to look
896 # up cached blocks once; on subsequent iterations just trim
897 # to the (reduced) current hit length.
898 curr_hit_length = min(
899 curr_hit_length, hit_length_by_group[first_group_id]
900 )
901 continue
902
903 drop_eagle_block = use_eagle and idx not in eagle_verified
904
905 _max_length = curr_hit_length
906 # Eagle matches one extra drop unit (one hash unit for
907 # fine-grained managers, else one cache block) and then drops
908 # it, landing back at the candidate length. No margin for
909 # mamba: its finder never drops (draft models have no mamba
910 # layers), so the hit would grow past the candidate.
911 if drop_eagle_block and not isinstance(spec, MambaSpec):
912 eagle_margin = (
913 self.hash_block_size
914 if self.enable_partial_hash_hits
915 and manager_cls.supports_fine_grained_hash_lookup
916 and group_block_size > self.hash_block_size
917 else group_block_size
918 )
919 _max_length = min(
920 curr_hit_length + eagle_margin, max_cache_hit_length
921 )
922 hit_blocks, _new_hit_length = manager_cls.find_longest_cache_hit(
923 block_hashes=block_hashes,
924 max_length=_max_length,
925 kv_cache_group_ids=group_ids,
926 block_pool=self.single_type_managers[first_group_id].block_pool,
927 kv_cache_spec=spec,
928 drop_eagle_block=drop_eagle_block,
929 alignment_tokens=self._cache_hit_alignment_tokens,
930 dcp_world_size=self.single_type_managers[
931 first_group_id
932 ].dcp_world_size,
933 pcp_world_size=self.single_type_managers[
934 first_group_id
935 ].pcp_world_size,
936 )
937 if drop_eagle_block:
938 eagle_verified.add(idx)
939 elif _new_hit_length < curr_hit_length:
940 # length shrunk; invalidate previous eagle verifications
941 eagle_verified.clear()
942 curr_hit_length = _new_hit_length
943 for group_id, blocks in zip(group_ids, hit_blocks):
944 hit_blocks_by_group[group_id] = blocks
945 hit_length_by_group[group_id] = _new_hit_length
946
947 longest_hit_length = max(longest_hit_length, curr_hit_length)
948
949 if curr_hit_length >= hit_length:
950 break
951 hit_length = curr_hit_length
952 if is_simple_hybrid:
953 break
954
955 # Truncate every full-attention group (target and draft) blocks
956 # to final hit_length.
957 for group in self.attention_groups:
958 if not isinstance(group.spec, FullAttentionSpec):
959 continue
960 manager = self.single_type_managers[group.group_ids[0]]
961 if manager.retains_longer_hit:
962 continue
963 group_block_size = manager.block_size
964 num_blocks = cdiv(hit_length, group_block_size)
965 for group_id in group.group_ids:
966 if (blks := hit_blocks_by_group[group_id]) is not None:
967 del blks[num_blocks:]
968 hit_length_by_group[group_id] = hit_length
969
970 # Uncached shared prefix detection: if any attn. group cached a longer
971 # prefix than the reconciled hit, it is an uncached common prefix across
972 # requests that a sparse-retention group hasn't cached yet.
973 num_uncached_common_prefix_tokens = longest_hit_length - hit_length
974 cache_hit_blocks = tuple(
975 blocks if blocks is not None else [] for blocks in hit_blocks_by_group
976 )
977 return cache_hit_blocks, hit_length, num_uncached_common_prefix_tokens
978
979 def find_longest_cache_hit_per_group(
980 self,
981 block_hashes: list[BlockHash],
982 max_cache_hit_length: int,
983 ) -> tuple[tuple[list[KVCacheBlock], ...], tuple[int, ...]]:
984 """Like find_longest_cache_hit but evaluates each group independently.
985
986 Returns:
987 (blocks_per_group, hit_lengths_per_group)
988 """
989
990 num_groups = len(self.kv_cache_config.kv_cache_groups)
991 hit_blocks: list[list[KVCacheBlock]] = [[] for _ in range(num_groups)]
992 hit_lengths: list[int] = [0] * num_groups
993
994 for spec, group_ids, manager_cls, use_eagle in self.attention_groups:
995 manager = self.single_type_managers[group_ids[0]]
996 blocks, group_hit = manager_cls.find_longest_cache_hit(
997 block_hashes=block_hashes,
998 max_length=max_cache_hit_length,
999 kv_cache_group_ids=group_ids,
1000 block_pool=manager.block_pool,
1001 kv_cache_spec=spec,
1002 drop_eagle_block=use_eagle,
1003 alignment_tokens=self._cache_hit_alignment_tokens,
1004 dcp_world_size=manager.dcp_world_size,
1005 pcp_world_size=manager.pcp_world_size,
1006 )
1007 for gid, blks in zip(group_ids, blocks):
1008 hit_blocks[gid] = blks
1009 hit_lengths[gid] = group_hit
1010
1011 return tuple(hit_blocks), tuple(hit_lengths)
