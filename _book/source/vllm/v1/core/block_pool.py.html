33class BlockHashToBlockMap:
34 """
35 Cache of blocks that are used for prefix caching. It caches blocks
36 from hash directly to a block or multiple blocks
37 (i.e. {block_hash: KVCacheBlocks})
38 - Mostly block_hash maps to a single KVCacheBlock, and KVCacheBlocks
39 would simply be a KVCacheBlock.
40 - Otherwise, KVCacheBlocks is a dict from {block_id: KVCacheBlock}
41
42 A cached block is a full block with a block hash that can be used
43 for prefix caching.
44 The cached block may be used by running requests or in the
45 free_block_queue that could potentially be evicted.
46
47 NOTE #1: We currently don't de-duplicate the blocks in the cache,
48 meaning that if a block becomes full and is cached, we don't check
49 if there is already an identical block in the cache. This is because
50 we want to make sure the allocated block IDs won't change so that
51 block tables are append-only.
52 NOTE #2: The union type is introduced in order to reduce GC costs
53 from the inner dict.
54 """
55
56 def __init__(self):
57 self._cache: dict[
58 BlockHashWithGroupId, KVCacheBlock | dict[int, KVCacheBlock]
59 ] = {}
60
61 def get_one_block(self, key: BlockHashWithGroupId) -> KVCacheBlock | None:
62 """
63 Gets any block with the given block hash key.
64 """
65 blocks = self._cache.get(key)
66 if blocks is not None:
67 if isinstance(blocks, KVCacheBlock):
68 return blocks
69 if isinstance(blocks, dict):
70 return next(iter(blocks.values()))
71 self._unexpected_blocks_type(blocks)
72 return None
73
74 def contain(self, key: BlockHashWithGroupId, block_id: int) -> bool:
75 """
76 Checks whether the key maps to the given block ID.
77 """
78 blocks = self._cache.get(key)
79 if blocks is None:
80 return False
81 if isinstance(blocks, KVCacheBlock):
82 return blocks.block_id == block_id
83 if isinstance(blocks, dict):
84 return block_id in blocks
85 self._unexpected_blocks_type(blocks)
86 return False
87
88 def insert(self, key: BlockHashWithGroupId, block: KVCacheBlock) -> None:
89 """
90 Inserts the KVCacheBlock to the cache
91 """
92 blocks = self._cache.get(key)
93 if blocks is None:
94 # When key is not found, attach a single block to the key
95 self._cache[key] = block
96 elif isinstance(blocks, KVCacheBlock):
97 # If there's a block with the same key, merge the original block
98 # and the new block into a dict
99 self._cache[key] = {blocks.block_id: blocks, block.block_id: block}
100 elif isinstance(blocks, dict):
101 # If it's already a dict, simply insert the block
102 blocks[block.block_id] = block
103 else:
104 self._unexpected_blocks_type(blocks)
105
106 def pop(self, key: BlockHashWithGroupId, block_id: int) -> KVCacheBlock | None:
107 """
108 Checks if block_hash exists and pop block_id from the cache
109 """
110 blocks = self._cache.pop(key, None)
111 if blocks is None:
112 # block_hash not found in the cache
113 return None
114 # TODO(Jialin): If key is found, block_id should always present
115 # in blocks. We currently keep the original behaviour for safety.
116 #
117 # Will add block_id == blocks.block_id assertion and
118 # use del blocks[block_id] instead as followup.
119 if isinstance(blocks, KVCacheBlock):
120 if blocks.block_id == block_id:
121 return blocks
122 # If the single block ID doesn't match, we should put the
123 # block back (it should happen rarely)
124 self._cache[key] = blocks
125 return None
126 if isinstance(blocks, dict):
127 # Try to pop block_id from the block dict, and if dict still
128 # contain blocks, put back to the cache.
129 block = blocks.pop(block_id, None)
130 if len(blocks) > 0:
131 self._cache[key] = blocks
132 return block
133 self._unexpected_blocks_type(blocks)
134 return None
135
136 def __len__(self) -> int:
137 return len(self._cache)
138
139 def _unexpected_blocks_type(self, blocks: Any) -> None:
140 raise AssertionError(f"Invalid KV cache block type {type(blocks)}")

143class BlockPool:
144 """BlockPool that manages KVCacheBlocks.
145 It provides methods to allocate, free and cache the kv cache blocks. The
146 free_block_queue stores the free blocks in eviction order to enable
147 allocation, free, and cache eviction. The cached_block_hash_to_block
148 maps between block hash and cached block to support finding cached blocks
149 by their block hash.
150
151 Args:
152 num_gpu_blocks: The number of blocks in the pool.
153 enable_caching: Whether to enable prefix caching.
154 hash_block_size: The block size of which the block hashes are computed.
155 The actual block size usually equals hash_block_size, but in cases
156 where different KV cache groups have different block sizes, the
157 actual block size can be a multiple of hash_block_size.
158 enable_kv_cache_events: Whether to enable kv cache events.
159 metrics_collector: Optional metrics collector for tracking block residency.
160 medium: Storage medium reported in KV cache events.
161 """
162
163 def __init__(
164 self,
165 num_gpu_blocks: int,
166 enable_caching: bool,
167 hash_block_size: int,
168 enable_kv_cache_events: bool = False,
169 metrics_collector: KVCacheMetricsCollector | None = None,
170 medium: str = MEDIUM_GPU,
171 ):
172 assert isinstance(num_gpu_blocks, int) and num_gpu_blocks > 0
173 self.num_gpu_blocks = num_gpu_blocks
174 self.medium = medium
175 self.enable_caching = enable_caching
176 self.hash_block_size = hash_block_size
177 # All kv-cache blocks.
178 self.blocks: list[KVCacheBlock] = [
179 KVCacheBlock(idx, pool=self) for idx in range(num_gpu_blocks)
180 ]
181 # Free block queue that constructs and manipulates a doubly linked
182 # list of free blocks (including eviction candidates when caching is
183 # enabled).
184 self.free_block_queue = FreeKVCacheBlockQueue(self.blocks)
185
186 # Cache for block lookup
187 self.cached_block_hash_to_block: BlockHashToBlockMap = BlockHashToBlockMap()
188 self.cached_block_hashes_by_block: dict[int, set[BlockHashWithGroupId]] = {}
189
190 # To represent a placeholder block with block_id=0.
191 # The ref_cnt of null_block is not maintained, needs special care to
192 # avoid freeing it.
193 self.null_block = self.free_block_queue.popleft()
194 self.null_block.is_null = True
195
196 self.enable_kv_cache_events = enable_kv_cache_events
197 self.kv_event_queue: list[KVCacheEvent] = []
198
199 self.metrics_collector = metrics_collector
200 # Callbacks for blocks released with ``unpin_blocks`` whose contents
201 # are still being read until the pool reuses them.
202 self._reuse_watchers: dict[int, Callable[[KVCacheBlock], None]] = {}
203
204 def get_cached_block(
205 self, block_hash: BlockHash, kv_cache_group_ids: list[int]
206 ) -> list[KVCacheBlock] | None:
207 """Get the cached block by the block hash for each group in
208 `kv_cache_group_ids`, or None if cache miss for any group.
209 If there are duplicated blocks, we return the first block in the cache.
210
211 Args:
212 block_hash: The hash value of the block.
213 kv_cache_group_ids: The ids of the KV cache groups.
214
215 Returns:
216 The cached blocks if exists, or None.
217 """
218 cached_blocks = []
219 for group_id in kv_cache_group_ids:
220 block_hash_with_group_id = make_block_hash_with_group_id(
221 block_hash, group_id
222 )
223 block = self.cached_block_hash_to_block.get_one_block(
224 block_hash_with_group_id
225 )
226 if not block:
227 return None
228 cached_blocks.append(block)
229 return cached_blocks

231 def cache_full_blocks(
232 self,
233 request: Request,
234 blocks: list[KVCacheBlock],
235 num_cached_blocks: int,
236 num_full_blocks: int,
237 block_size: int,
238 kv_cache_group_id: int,
239 block_mask: list[bool] | None = None,
240 ) -> None:
241 """Cache a list of full blocks for prefix caching.
242 This function takes a list of blocks that will have their block hash
243 metadata to be updated and cached. Given a request, it updates the
244 metadata for each block and caching it in the
245 `cached_block_hash_to_block`.
246 The block hashes values are computed by the Request object immediately
247 when it is created and when new tokens are appended.
248
249 Args:
250 request: The request to cache the blocks.
251 blocks: All blocks in the request.
252 num_cached_blocks: The number of blocks that are already cached.
253 num_full_blocks: The number of blocks that are full and should
254 be cached after this function.
255 block_size: Number of tokens in each block.
256 kv_cache_group_id: The id of the KV cache group.
257 block_mask: Optional mask aligned with
258 ``blocks[num_cached_blocks:num_full_blocks]``. When provided,
259 blocks where the mask is False are skipped (treated like null
260 blocks). Used by groups whose ``find_longest_cache_hit`` only
261 consults a subset of blocks (e.g. SWA tail-window), so blocks
262 that can never serve a hit stay out of the prefix-cache hash
263 map.
264 """
265 if num_cached_blocks >= num_full_blocks:
266 return
267 new_full_blocks = blocks[num_cached_blocks:num_full_blocks]
268 assert block_mask is None or len(block_mask) == len(new_full_blocks)
269 block_hashes = resolve_block_hashes(
270 request.block_hashes, self.hash_block_size, block_size
271 )
272
273 new_block_hashes = block_hashes[num_cached_blocks:]
274 new_hashes: list[ExternalBlockHash] | None = (
275 [] if self.enable_kv_cache_events else None
276 )
277 for i, blk in enumerate(new_full_blocks):
278 # Some blocks may be null or masked out when enabling sparse attention
279 # like sliding window attention, or Mamba models with prefix-caching
280 # in align mode. We skip null blocks here.
281 if blk.is_null or (block_mask is not None and not block_mask[i]):
282 continue
283 block_hash = new_block_hashes[i]
284 num_hash_tokens = (num_cached_blocks + i + 1) * block_size
285
286 # Update and added the full block to the cache.
287 block_hash_with_group_id = make_block_hash_with_group_id(
288 block_hash, kv_cache_group_id
289 )
290 if blk.block_hash is not None:
291 # The only valid case where a "new full block" already has a
292 # hash is partial->full promotion of the same cache block.
293 assert (
294 blk.block_hash_num_tokens is not None
295 and blk.block_hash_num_tokens < num_hash_tokens
296 )
297 removed_hashes = self._remove_cached_block_hashes(blk)
298 self._emit_block_removed_events(removed_hashes)
299 self._insert_block_hash(
300 block_hash_with_group_id,
301 blk,
302 num_tokens=num_hash_tokens,
303 )
304 if new_hashes is not None:
305 new_hashes.append(maybe_convert_block_hash(block_hash))
306
307 if self.enable_kv_cache_events:
308 if num_cached_blocks == 0:
309 parent_block_hash: ExternalBlockHash | None = None
310 else:
311 parent_block_hash = maybe_convert_block_hash(
312 block_hashes[num_cached_blocks - 1]
313 )
314
315 # Calculate token range for the blocks being cached
316 start_token_idx = num_cached_blocks * block_size
317 end_token_idx = num_full_blocks * block_size
318
319 # Generate extra keys for each block individually.
320 # Each block may have different extra_keys (e.g., different MM
321 # features, or cache_salt only for the first block).
322 # Skip null/masked-out blocks to match the length of new_hashes.
323 extra_keys_list: list[tuple[Any, ...] | None] = []
324 curr_mm_idx = 0
325 for i in range(num_cached_blocks, num_full_blocks):
326 if blocks[i].is_null:
327 continue
328 if block_mask is not None and not block_mask[i - num_cached_blocks]:
329 continue
330 block_start = i * block_size
331 block_end = block_start + block_size
332 extra_keys, curr_mm_idx = generate_block_hash_extra_keys(
333 request, block_start, block_end, curr_mm_idx
334 )
335 extra_keys_list.append(extra_keys)
336
337 self.kv_event_queue.append(
338 self._build_block_stored_event(
339 request,
340 block_hashes=new_hashes,
341 parent_block_hash=parent_block_hash,
342 start_token_idx=start_token_idx,
343 end_token_idx=end_token_idx,
344 block_size=block_size,
345 kv_cache_group_id=kv_cache_group_id,
346 extra_keys_list=extra_keys_list,
347 )
348 )
349
350 def _build_block_stored_event(
351 self,
352 request: Request,
353 block_hashes: list[ExternalBlockHash] | None,
354 parent_block_hash: ExternalBlockHash | None,
355 start_token_idx: int,
356 end_token_idx: int,
357 block_size: int,
358 kv_cache_group_id: int,
359 extra_keys_list: list[tuple[Any, ...] | None],
360 ) -> BlockStored:
361 """Build a ``BlockStored`` KV event for ``request``.
362
363 Shared by ``cache_full_blocks`` (newly cached blocks) and
364 ``emit_cached_block_events`` (prefix-cache-reused blocks) so both emit
365 identical event shapes for downstream consumers.
366 """
367 return BlockStored(
368 block_hashes=block_hashes,
369 parent_block_hash=parent_block_hash,
370 token_ids=request.all_token_ids[start_token_idx:end_token_idx],
371 block_size=block_size,
372 lora_id=request.lora_request.adapter_id if request.lora_request else None,
373 medium=self.medium,
374 lora_name=request.lora_request.name if request.lora_request else None,
375 extra_keys=extra_keys_list if extra_keys_list else None,
376 group_idx=kv_cache_group_id,
377 session_id=request.session_id,
378 )

380 def emit_cached_block_events(
381 self,
382 request: Request,
383 num_cached_blocks: int,
384 block_size: int,
385 kv_cache_group_id: int,
386 ) -> None:
387 """Generate BlockStored events for blocks reused from prefix cache.
388
389 Unlike cache_full_blocks(), this does NOT modify block state —
390 the blocks are already cached. It only generates events so that
391 external consumers (e.g. gateway) can learn about reused blocks.
392
393 Args:
394 request: The request whose prefix cache blocks were reused.
395 num_cached_blocks: Number of blocks that were cache hits.
396 block_size: Number of tokens per block.
397 kv_cache_group_id: The KV cache group ID.
398 """
399 if not self.enable_kv_cache_events or num_cached_blocks == 0:
400 return
401
402 block_hashes = resolve_block_hashes(
403 request.block_hashes, self.hash_block_size, block_size
404 )
405
406 # Collect external hashes and extra_keys for cached blocks.
407 cached_hashes: list[ExternalBlockHash] = []
408 extra_keys_list: list[tuple[Any, ...] | None] = []
409 curr_mm_idx = 0
410 for i in range(num_cached_blocks):
411 block_start = i * block_size
412 block_end = block_start + block_size
413 cached_hashes.append(maybe_convert_block_hash(block_hashes[i]))
414 extra_keys, curr_mm_idx = generate_block_hash_extra_keys(
415 request, block_start, block_end, curr_mm_idx
416 )
417 extra_keys_list.append(extra_keys)
418
419 if not cached_hashes:
420 return
421
422 # Prefix-cache hits always form a contiguous prefix starting at block 0,
423 # so the first (and thus the whole group's) parent block hash is None.
424 parent_block_hash: ExternalBlockHash | None = None
425 start_token_idx = 0
426 end_token_idx = num_cached_blocks * block_size
427
428 logger.debug(
429 "EmitCachedBlock event: block_size=%d, "
430 "num_cached_blocks=%d, parent_block_hash=%s, "
431 "token_ids_len=%d, group_idx=%s",
432 block_size,
433 num_cached_blocks,
434 parent_block_hash,
435 len(request.all_token_ids[start_token_idx:end_token_idx]),
436 kv_cache_group_id,
437 )
438
439 self.kv_event_queue.append(
440 self._build_block_stored_event(
441 request,
442 block_hashes=cached_hashes,
443 parent_block_hash=parent_block_hash,
444 start_token_idx=start_token_idx,
445 end_token_idx=end_token_idx,
446 block_size=block_size,
447 kv_cache_group_id=kv_cache_group_id,
448 extra_keys_list=extra_keys_list,
449 )
450 )

452 def cache_partial_block(
453 self,
454 request: Request,
455 block: KVCacheBlock,
456 num_tokens: int,
457 kv_cache_group_id: int,
458 block_size: int,
459 replace_existing_hashes: bool = False,
460 ) -> BlockHashWithGroupId | None:
461 """Register a partial prefix-cache entry for an existing block.
462
463 Prefix-cache keys normally identify full cache blocks. A partial entry
464 makes an existing cache block reachable from a fine-grained prefix
465 boundary inside that block without allocating or copying a new
466 ``KVCacheBlock``.
467
468 The partial entry is lookup metadata owned by ``block``. If ``block``
469 has no primary hash, the key becomes its primary hash. If the block
470 already has a primary hash, the partial entry is tracked in
471 ``cached_block_hashes_by_block`` so eviction, reset, and promotion can
472 remove every hash key that points to the block.
473
474 Args:
475 request: Request whose token IDs and block hashes define the
476 partial entry.
477 block: Existing cache block to make reachable from the partial
478 prefix boundary.
479 num_tokens: Prefix length represented by the partial entry. It
480 must be a positive multiple of ``self.hash_block_size`` and
481 cannot exceed the request's computed block hashes.
482 kv_cache_group_id: KV cache group that owns the partial entry.
483 block_size: Cache block size for the owning group. The partial
484 entry hash itself is always the prefix-chain hash at
485 ``num_tokens``; ``block_size`` is used to assert that the
486 entry is partial within the owning cache block.
487 replace_existing_hashes: Whether the block contents were replaced
488 and all existing cache entries must be removed before the new
489 entry is registered.
490
491 Returns:
492 The hash key with group ID if a partial entry can be registered;
493 otherwise ``None`` for null blocks.
494 """
495 if block.is_null:
496 return None
497
498 assert block_size % self.hash_block_size == 0
499 assert replace_existing_hashes or (
500 block_size > self.hash_block_size and num_tokens % block_size != 0
501 )
502 block_hash = self._get_partial_block_hash(request, num_tokens)
503 num_hash_blocks = num_tokens // self.hash_block_size
504 block_hash_with_group_id = make_block_hash_with_group_id(
505 block_hash, kv_cache_group_id
506 )
507 already_cached = block.block_hash == block_hash_with_group_id or (
508 self.cached_block_hash_to_block.contain(
509 block_hash_with_group_id, block.block_id
510 )
511 )
512 if replace_existing_hashes:
513 removed_hashes = self._remove_cached_block_hashes(block)
514 self._emit_block_removed_events(removed_hashes)
515 already_cached = False
516 elif (
517 not already_cached
518 and block.block_hash is not None
519 and block.block_hash_num_tokens is not None
520 and block.block_hash_num_tokens < num_hash_blocks * self.hash_block_size
521 ):
522 removed_hashes = self._remove_cached_block_hashes(block)
523 self._emit_block_removed_events(removed_hashes)
524 self._insert_block_hash(
525 block_hash_with_group_id,
526 block,
527 num_tokens=num_hash_blocks * self.hash_block_size,
528 )
529 if self.enable_kv_cache_events and not already_cached:
530 parent_hash, block_start = self._get_partial_block_parent_hash_and_start(
531 request, num_tokens
532 )
533 parent_block_hash = (
534 maybe_convert_block_hash(parent_hash)
535 if parent_hash is not None
536 else None
537 )
538 block_end = num_tokens
539 curr_mm_idx = -1 if block_start > 0 else 0
540 extra_keys, _ = generate_block_hash_extra_keys(
541 request, block_start, block_end, curr_mm_idx
542 )
543 self.kv_event_queue.append(
544 BlockStored(
545 block_hashes=[maybe_convert_block_hash(block_hash)],
546 parent_block_hash=parent_block_hash,
547 token_ids=request.all_token_ids[block_start:block_end],
548 block_size=block_end - block_start,
549 lora_id=request.lora_request.adapter_id
550 if request.lora_request
551 else None,
552 medium=self.medium,
553 lora_name=request.lora_request.name
554 if request.lora_request
555 else None,
556 extra_keys=[extra_keys],
557 group_idx=kv_cache_group_id,
558 session_id=request.session_id,
559 )
560 )
561 return block_hash_with_group_id
562
563 def _get_partial_block_hash(
564 self,
565 request: Request,
566 num_tokens: int,
567 ) -> BlockHash:
568 assert num_tokens % self.hash_block_size == 0
569 num_hash_blocks = num_tokens // self.hash_block_size
570 assert 0 < num_hash_blocks <= len(request.block_hashes)
571
572 # Each hash_block_size hash chains over its full prefix, so the partial
573 # entry for any group block size is the hash at that prefix boundary.
574 return request.block_hashes[num_hash_blocks - 1]
575
576 def _get_partial_block_parent_hash_and_start(
577 self,
578 request: Request,
579 num_tokens: int,
580 ) -> tuple[BlockHash | None, int]:
581 num_hash_blocks = num_tokens // self.hash_block_size
582 parent_hash = (
583 request.block_hashes[num_hash_blocks - 2] if num_hash_blocks > 1 else None
584 )
585 block_start = (num_hash_blocks - 1) * self.hash_block_size
586 return parent_hash, block_start
587
588 def _remove_cached_block_hashes(
589 self,
590 block: KVCacheBlock,
591 ) -> list[BlockHashWithGroupId]:
592 block_hashes: list[BlockHashWithGroupId] = []
593 if block.block_hash is not None:
594 block_hashes.append(block.block_hash)
595 block_hashes.extend(self.cached_block_hashes_by_block.pop(block.block_id, ()))
596 if not block_hashes:
597 return []
598
599 removed_hashes: list[BlockHashWithGroupId] = []
600 for block_hash in block_hashes:
601 if (
602 self.cached_block_hash_to_block.pop(block_hash, block.block_id)
603 is not None
604 ):
605 removed_hashes.append(block_hash)
606 block.reset_hash()
607 return removed_hashes
608
609 def _emit_block_removed_events(
610 self,
611 block_hashes: list[BlockHashWithGroupId],
612 ) -> None:
613 if not self.enable_kv_cache_events:
614 return
615 for block_hash in block_hashes:
616 self.kv_event_queue.append(
617 BlockRemoved(
618 block_hashes=[maybe_convert_block_hash(get_block_hash(block_hash))],
619 medium=self.medium,
620 group_idx=get_group_id(block_hash),
621 )
622 )
623
624 def _insert_block_hash(
625 self,
626 block_hash_with_group_id: BlockHashWithGroupId,
627 block: KVCacheBlock,
628 num_tokens: int | None,
629 ) -> None:
630 if block.block_hash == block_hash_with_group_id:
631 return
632
633 if self.cached_block_hash_to_block.contain(
634 block_hash_with_group_id, block.block_id
635 ):
636 return
637
638 if block.block_hash is None:
639 block.set_block_hash(block_hash_with_group_id, num_tokens=num_tokens)
640 else:
641 self.cached_block_hashes_by_block.setdefault(block.block_id, set()).add(
642 block_hash_with_group_id
643 )
644 self.cached_block_hash_to_block.insert(block_hash_with_group_id, block)

646 def move_block_hashes(
647 self,
648 src_block: KVCacheBlock,
649 dst_block: KVCacheBlock,
650 ) -> None:
651 """Re-point ``src_block``'s prefix-cache entries to ``dst_block``.
652
653 Used when the request owning ``src_block`` keeps writing into it
654 : the prefix cache holds a private copy (``dst_block``)
655 under the same hashes instead. Entries stay live; no events emitted.
656 """
657 assert dst_block.block_hash is None
658 assert dst_block.block_id not in self.cached_block_hashes_by_block
659 num_tokens = src_block.block_hash_num_tokens
660 for block_hash in self._remove_cached_block_hashes(src_block):
661 # `num_tokens` only applies to the first (primary) insertion.
662 self._insert_block_hash(block_hash, dst_block, num_tokens=num_tokens)
663
664 def get_new_blocks(self, num_blocks: int) -> list[KVCacheBlock]:
665 """Get new blocks from the free block pool.
666
667 Note that we do not check block cache in this function.
668
669 Args:
670 num_blocks: The number of blocks to allocate.
671
672 Returns:
673 A list of new block.
674 """
675 if num_blocks > self.get_num_free_blocks():
676 raise ValueError(f"Cannot get {num_blocks} free blocks from the pool")
677
678 ret: list[KVCacheBlock] = self.free_block_queue.popleft_n(num_blocks)
679
680 if self._reuse_watchers:
681 self._notify_reuse(ret)
682
683 # In order to only iterate the list once, we duplicated code a bit
684 if self.enable_caching:
685 for block in ret:
686 self._maybe_evict_cached_block(block)
687 assert block.ref_cnt == 0
688 block.ref_cnt += 1
689 if self.metrics_collector:
690 self.metrics_collector.on_block_allocated(block)
691 else:
692 for block in ret:
693 assert block.ref_cnt == 0
694 block.ref_cnt += 1
695 if self.metrics_collector:
696 self.metrics_collector.on_block_allocated(block)
697 return ret
698
699 def _notify_reuse(self, blocks: Iterable[KVCacheBlock]) -> None:
700 for block in blocks:
701 watcher = self._reuse_watchers.pop(block.block_id, None)
702 if watcher is not None:
703 watcher(block)
704
705 def unpin_blocks(
706 self,
707 blocks: Iterable[KVCacheBlock],
708 on_reuse: Callable[[KVCacheBlock], None],
709 ) -> None:
710 """Release references to blocks that stay readable until reused.
711
712 The blocks become last-resort eviction candidates regardless of prefix
713 caching: they join the tail of the free queue and count as free.
714 ``on_reuse`` fires when ``get_new_blocks`` hands a block out (or the
715 cache is reset), so the caller can stop reading it.
716 """
717 released: list[KVCacheBlock] = []
718 for block in blocks:
719 assert block.ref_cnt > 0 and not block.is_null
720 block.ref_cnt -= 1
721 self._reuse_watchers[block.block_id] = on_reuse
722 if block.ref_cnt == 0:
723 released.append(block)
724 self.free_block_queue.append_n(released)
725
726 def _maybe_evict_cached_block(self, block: KVCacheBlock) -> bool:
727 """
728 If a block is cached in `cached_block_hash_to_block`, we reset its hash
729 metadata and evict it from the cache.
730
731 Args:
732 block: The block to evict.
733
734 Returns:
735 True if the block is evicted, False otherwise.
736 """
737 # Clean up metrics tracking first to prevent leaks
738 if self.metrics_collector:
739 self.metrics_collector.on_block_evicted(block)
740
741 evicted_hashes = self._remove_cached_block_hashes(block)
742 if not evicted_hashes:
743 # The block doesn't have hash, eviction is not needed
744 return False
745
746 self._emit_block_removed_events(evicted_hashes)
747 return True
748
749 def touch(self, blocks: Sequence[KVCacheBlock]) -> None:
750 """Touch a block increases its reference count by 1, and may remove
751 the block from the free queue. This is used when a block is hit by
752 another request with the same prefix.
753
754 Args:
755 blocks: A list of blocks to touch.
756 """
757 for block in blocks:
758 # ref_cnt=0 means this block is in the free list (i.e. eviction
759 # candidate), so remove it.
760 if block.ref_cnt == 0 and not block.is_null:
761 self.free_block_queue.remove(block)
762 block.ref_cnt += 1
763 if self.metrics_collector:
764 self.metrics_collector.on_block_accessed(block)
765
766 def is_block_writable(self, block: KVCacheBlock) -> bool:
767 """Return whether a block can be mutated by its sole owner."""
768 return not block.is_null and block.ref_cnt == 1 and block.block_hash is None
769
770 def free_blocks(self, ordered_blocks: Iterable[KVCacheBlock]) -> None:
771 """Free a list of blocks. The blocks should be ordered by their
772 eviction priority, where the first block will be evicted first.
773
774 Args:
775 ordered_blocks: A list of blocks to free ordered by their eviction
776 priority.
777 """
778 # Identify blocks with hash (LRU cache) and without it (never match APC)
779 blocks_to_evict_last = []
780 blocks_to_evict_first = []
781 other_pools: dict[BlockPool, list[KVCacheBlock]] = {}
782 for block in ordered_blocks:
783 if block.pool is not None and block.pool is not self:
784 other_pools.setdefault(block.pool, []).append(block)
785 continue
786 block.ref_cnt -= 1
787 if block.ref_cnt == 0 and not block.is_null:
788 if block.block_hash is None or not self.enable_caching:
789 # LIFO reuse of non-cached blocks for better GPU locality.
790 blocks_to_evict_first.append(block)
791 else:
792 # FIFO reuse of cached blocks for LRU eviction behavior.
793 blocks_to_evict_last.append(block)
794
795 # Blocks to reuse first are prepended to the front of the free queue.
796 self.free_block_queue.prepend_n(blocks_to_evict_first)
797 # Blocks to reuse last are appended to the end of the free queue.
798 self.free_block_queue.append_n(blocks_to_evict_last)
799 for pool, blocks in other_pools.items():
800 pool.free_blocks(blocks)
801
802 def evict_blocks(self, block_ids: set[int]) -> None:
803 """evict blocks from the prefix cache by their block IDs.
804
805 only evicts blocks that are currently cached (have a hash). blocks
806 with ref_cnt > 0 are not freed from the block pool, only evicted
807 from the prefix cache hash table.
808
809 Args:
810 block_ids: Set of block IDs to evict from cache.
811 """
812 for block_id in block_ids:
813 assert block_id < len(self.blocks), (
814 f"Invalid block_id {block_id} >= {len(self.blocks)}. "
815 f"This indicates a bug in the KV connector - workers should "
816 f"only report block IDs that were allocated by the scheduler."
817 )
818 block = self.blocks[block_id]
819 self._maybe_evict_cached_block(block)
820
821 def reset_prefix_cache(self) -> bool:
822 """Reset prefix cache. This function may be used in RLHF
823 flows to invalid prefix caching after the weights are updated,
824 or used for resetting prefix caching status for benchmarking.
825
826 Returns:
827 bool: True if the prefix cache is successfully reset,
828 False otherwise.
829 """
830 num_used_blocks = self.num_gpu_blocks - self.get_num_free_blocks()
831 if num_used_blocks != 1: # The null block is always marked as used
832 logger.warning(
833 "Failed to reset prefix cache because some "
834 "blocks (%d) are not freed yet",
835 num_used_blocks - 1,
836 )
837 return False
838
839 # Remove all hashes so that no new blocks will hit.
840 self.cached_block_hash_to_block = BlockHashToBlockMap()
841 self.cached_block_hashes_by_block.clear()
842 if self._reuse_watchers:
843 self._notify_reuse(
844 [self.blocks[block_id] for block_id in list(self._reuse_watchers)]
845 )
846
847 # Remove all hashes from all blocks.
848 for block in self.blocks:
849 block.reset_hash()
850
851 if self.metrics_collector:
852 self.metrics_collector.reset()
853
854 logger.info("Successfully reset prefix cache")
855
856 if self.enable_kv_cache_events:
857 self.kv_event_queue.append(AllBlocksCleared())
858
859 return True
860
861 def get_num_free_blocks(self) -> int:
862 """Get the number of free blocks in the pool.
863
864 Returns:
865 The number of free blocks.
866 """
867 return self.free_block_queue.num_free_blocks
868
869 def get_usage(self) -> float:
870 """Get the KV cache usage.
871
872 Returns:
873 The KV cache usage (between 0.0 and 1.0).
874 """
875
876 # Subtract 1 to account for null block.
877 total_gpu_blocks = self.num_gpu_blocks - 1
878 if not total_gpu_blocks:
879 return 0
880 return 1.0 - (self.get_num_free_blocks() / total_gpu_blocks)

882 def take_events(self) -> list[KVCacheEvent]:
883 """Atomically takes all events and clears the queue.
884
885 Returns:
886 A list of KV cache events.
887 """
888 if not self.enable_kv_cache_events:
889 return []
890 events = self.kv_event_queue
891 self.kv_event_queue = []
