34@dataclass
35class KVCacheBlocks:
36 """
37 The allocation result of KVCacheManager, work as the interface between
38 Scheduler and KVCacheManager, to hide KVCacheManager's internal data
39 structure from the Scheduler.
40 """
41
42 blocks: tuple[Sequence[KVCacheBlock], ...]
43 """
44 `blocks[i][j]` refers to the i-th kv_cache_group
45 and the j-th block of tokens.We don't use block of
46 tokens as the outer dimension because it assumes all
47 kv_cache_groups have the same number of blocks, which is true for now but
48 will be broken if we want to give different block_size to different
49 kv_cache_groups in the future.
50
51 Each single type KVCacheBlocks could be represented as:
52 - list[KVCacheBlock] for more than one KVCacheBlock
53 - an empty tuple for requests without KVCacheBlock
54 (a precomputed KVCacheBlocks is in KVCacheManager to avoid GC overhead)
55 """
56
57 def __add__(self, other: "KVCacheBlocks") -> "KVCacheBlocks":
58 """Adds two KVCacheBlocks instances."""
59 return KVCacheBlocks(
60 tuple(
61 list(itertools.chain(blk1, blk2))
62 for blk1, blk2 in zip(self.blocks, other.blocks)
63 )
64 )

82 def get_block_ids(
83 self,
84 allow_none: bool = False,
85 *,
86 group_ids: Sequence[int] | None = None,
87 ) -> tuple[list[int], ...] | None:
88 """
89 Converts the KVCacheBlocks instance to block_ids.
90
91 Args:
92 allow_none: Return None when every selected group is empty.
93 group_ids: KV cache groups to include. Includes all groups by default.
94
95 Returns:
96 tuple[list[int], ...]: A tuple of lists where:
97 - the outer tuple corresponds to KV cache groups
98 - each inner list contains the block_ids of the blocks in that
99 group
100 """
101 groups = (
102 self.blocks
103 if group_ids is None
104 else tuple(self.blocks[group_id] for group_id in group_ids)
105 )
106 if allow_none and all(len(group) == 0 for group in groups):
107 return None
108 return tuple([blk.block_id for blk in group] for group in groups)
109
110 def get_unhashed_block_ids(self) -> list[int]:
111 """Get block_ids of unhashed blocks from KVCacheBlocks instance."""
112 assert len(self.blocks) == 1, "Only one group is supported"
113 return [block.block_id for block in self.blocks[0] if block.block_hash is None]
114
115 def get_unhashed_block_ids_all_groups(self) -> list[list[int]]:
116 """Get block_ids of unhashed blocks from KVCacheBlocks instance."""
117 # Skip padding blocks.
118 return [
119 [
120 block.block_id
121 for block in group
122 if block.block_hash is None and not block.is_null
123 ]
124 for group in self.blocks
125 ]
126
127 def new_empty(self) -> "KVCacheBlocks":
128 """
129 Creates a new KVCacheBlocks instance with no blocks.
130 """
131 return KVCacheBlocks(tuple(() for _ in range(len(self.blocks))))

134class KVCacheManager:
135 def __init__(
136 self,
137 kv_cache_config: KVCacheConfig,
138 max_model_len: int,
139 scheduler_block_size: int,
140 hash_block_size: int,
141 max_in_flight_tokens: int | None = None,
142 enable_caching: bool = True,
143 use_eagle: bool = False,
144 num_prefill_lookahead: int = 0,
145 log_stats: bool = False,
146 enable_kv_cache_events: bool = False,
147 dcp_world_size: int = 1,
148 pcp_world_size: int = 1,
149 metrics_collector: KVCacheMetricsCollector | None = None,
150 watermark: float = 0.0,
151 enable_mamba_fine_grained_prefix_cache: bool = False,
152 ) -> None:
153 self.max_model_len = max_model_len
154 # When unset, fall back to `max_model_len` so the recycling-aware cap
155 # collapses to the prior (uncapped) admission behavior. The scheduler
156 # always supplies the real value at runtime.
157 if max_in_flight_tokens is None:
158 max_in_flight_tokens = max_model_len
159
160 self.enable_caching = enable_caching
161 self.enable_kv_cache_events = enable_kv_cache_events
162 self.use_eagle = use_eagle
163 self.log_stats = log_stats
164 self.metrics_collector = metrics_collector
165 # FIXME: make prefix cache stats conditional on log_stats. We still need
166 # this comment because when the log stats is enabled there are still
167 # potential configs we could expose in the future.
168 self.prefix_cache_stats = PrefixCacheStats() if log_stats else None
169
170 self.coordinator = get_kv_cache_coordinator(
171 kv_cache_config=kv_cache_config,
172 max_model_len=self.max_model_len,
173 max_in_flight_tokens=max_in_flight_tokens,
174 use_eagle=self.use_eagle,
175 enable_caching=self.enable_caching,
176 enable_kv_cache_events=enable_kv_cache_events,
177 dcp_world_size=dcp_world_size,
178 pcp_world_size=pcp_world_size,
179 scheduler_block_size=scheduler_block_size,
180 hash_block_size=hash_block_size,
181 metrics_collector=self.metrics_collector,
182 num_prefill_lookahead=num_prefill_lookahead,
183 )
184 # One predicate, read by both sides of the feature, so the scheduler
185 # cannot end a chunk at a junction the manager would refuse -- a refused
186 # junction costs a forward pass and displaces the block-boundary stop.
187 # Multi-module MTP is excluded because ``cache_blocks`` then hands the
188 # manager ``num_computed - num_reprefillable`` rather than the chunk end.
189 self.mamba_fine_grained_prefix_cache = (
190 enable_mamba_fine_grained_prefix_cache
191 and bool(self.coordinator.eagle_group_ids)
192 and self.coordinator.enable_partial_hash_hits
193 and self.coordinator.num_reprefillable_tokens == 0
194 )
195 if self.mamba_fine_grained_prefix_cache:
196 for manager in self.coordinator.single_type_managers:
197 if isinstance(manager, MambaManager):
198 manager.fine_grained_prefix_cache = True
199 self.num_kv_cache_groups = len(kv_cache_config.kv_cache_groups)
200 self.block_pool = self.coordinator.block_pool
201 self.retained_hit_group_ids = tuple(
202 manager.kv_cache_group_id
203 for manager in self.coordinator.single_type_managers
204 if manager.retains_longer_hit
205 )
206 self.kv_cache_config = kv_cache_config
207
208 # Watermark: minimum number of KV cache blocks to keep free when
209 # admitting waiting/preempted requests, to avoid frequent preemptions.
210 assert watermark >= 0.0, "watermark must be non-negative"
211 self.watermark_blocks = int(watermark * kv_cache_config.num_blocks)
212 self.kv_cache_event_metadata = tuple(
213 (
214 get_kv_cache_spec_kind(group.kv_cache_spec).value,
215 get_kv_cache_spec_sliding_window(group.kv_cache_spec),
216 )
217 for group in kv_cache_config.kv_cache_groups
218 )
219
220 # Pre-constructed KVCacheBlocks with no blocks, callers should use this
221 # via create_kv_cache_blocks instead of creating new ones to avoid GC
222 # overhead.
223 #
224 # We use nested tuples to ensure the empty KVCacheBlocks is immutable.
225 self.empty_kv_cache_blocks = KVCacheBlocks(
226 tuple(() for _ in range(self.num_kv_cache_groups))
227 )

265 def get_computed_blocks(self, request: Request) -> tuple[KVCacheBlocks, int, int]:
266 """Get the computed (cached) blocks for the request.
267 Note that the computed blocks must be full.
268
269 Args:
270 request: The request to get the computed blocks.
271
272 Returns:
273 A tuple containing:
274 - A list of blocks that are computed for the request.
275 - The number of computed tokens.
276 - ``shared_prefix_boundary``: the block-aligned token position of
277 a shared prefix that a sparse-retention group (Mamba / sliding
278 window) has not cached yet (Marconi-style APC), or 0 if none.
279 Pinned so sparse prefix-cache retention does not drop
280 the junction and defeat cross-request reuse.
281 """
282 # We skip finding the prefix cache hit when prefix caching is
283 # disabled or the request is marked as skipping kv cache read
284 # (which happens when the request requires prompt logprobs
285 # or calls a pooling model with all pooling).
286 if not self.prefix_cache_lookup_enabled(request):
287 return self.empty_kv_cache_blocks, 0, 0
288
289 # NOTE: When all tokens hit the cache, we must recompute the last token
290 # to obtain logits. Thus, set max_cache_hit_length to prompt_length - 1.
291 # This can trigger recomputation of an entire block, rather than just
292 # the single last token, because allocate_slots() requires
293 # num_computed_tokens to be block-size aligned. Removing this limitation
294 # could slightly improve performance in the future.
295 max_cache_hit_length = request.num_tokens - 1
296 computed_blocks, num_new_computed_tokens, num_uncached = (
297 self.coordinator.find_longest_cache_hit(
298 request.block_hashes, max_cache_hit_length
299 )
300 )
301
302 # When kv_cache_report_mode is "full", emit BlockStored events
303 # for the reused prefix cache blocks so that external consumers
304 # (e.g. gateway) can learn about them.
305 if (
306 num_new_computed_tokens > 0
307 and self.enable_kv_cache_events
308 and getattr(request, "kv_cache_report_mode", "incremental") == "full"
309 ):
310 self.coordinator.emit_cached_block_events(request, computed_blocks)
311
312 # The junction to pin is where the lagging sparse-retention group stops
313 # (``num_new_computed_tokens``) plus the uncached shared prefix -- i.e.
314 # the longest single-group hit. Sub-block gaps are left to the mask,
315 # which floors to the alignment boundary (a no-op there).
316 shared_prefix_boundary = (
317 num_new_computed_tokens + num_uncached if num_uncached else 0
318 )
319
320 blocks = self.create_kv_cache_blocks(computed_blocks)
321 return blocks, num_new_computed_tokens, shared_prefix_boundary

323 def get_computed_blocks_for_connector(
324 self, request: Request
325 ) -> tuple[KVCacheBlocks, int, int, bool]:
326 """Local prefix-cache lookup for a request scheduled with a KV connector.
327
328 Hybrid (Mamba + full-attention) models can have per-group prefix hits
329 diverge under block pressure: the full-attention tail may be evicted
330 while a deeper Mamba state survives, or vice versa. Report the
331 full-attention hit as the local prefix - the connector transfers the
332 remaining suffix and the Mamba state is transferred unconditionally by
333 nixl's ``_apply_prefix_caching`` - and flag when that hit ran deeper
334 than a lagging group. Such a hit only has a valid Mamba state at its
335 boundary if the connector supplies it, so the caller must fall back to
336 ``get_computed_blocks`` to reconcile when no external tokens are found.
337
338 Non-hybrid models and already-convergent hits use ``get_computed_blocks``.
339
340 Returns:
341 The ``get_computed_blocks`` triple (blocks, number of local computed
342 tokens, shared-prefix boundary) plus ``hit_diverged``.
343 """
344 coordinator = self.coordinator
345 if not (
346 self.kv_cache_config.has_mamba_layers
347 and isinstance(coordinator, HybridKVCacheCoordinator)
348 and coordinator.full_attention_group_id is not None
349 ):
350 return *self.get_computed_blocks(request), False
351
352 if not self.prefix_cache_lookup_enabled(request):
353 return self.empty_kv_cache_blocks, 0, 0, False
354
355 fa_group_id = coordinator.full_attention_group_id
356 computed, per_group_hits = coordinator.find_longest_cache_hit_per_group(
357 request.block_hashes, request.num_tokens - 1
358 )
359 if any(hit > per_group_hits[fa_group_id] for hit in per_group_hits):
360 # A lagging group hit deeper than full attention means its
361 # full-attention blocks were evicted; use the reconciled boundary
362 # that every group agrees on.
363 return *self.get_computed_blocks(request), False
364
365 num_local = per_group_hits[fa_group_id]
366 blocks = self.create_kv_cache_blocks(computed)
367 # Per-group lookups do not detect an uncached shared prefix (boundary 0).
368 return blocks, num_local, 0, min(per_group_hits) < num_local

370 def allocate_slots(
371 self,
372 request: Request,
373 num_new_tokens: int,
374 num_new_computed_tokens: int = 0,
375 new_computed_blocks: KVCacheBlocks | None = None,
376 num_lookahead_tokens: int = 0,
377 num_external_computed_tokens: int = 0,
378 delay_cache_blocks: bool = False,
379 num_encoder_tokens: int = 0,
380 full_sequence_must_fit: bool = False,
381 reserved_blocks: int = 0,
382 has_scheduled_reqs: bool = True,
383 ) -> KVCacheBlocks | None:
384 """Add slots for a request with new tokens to append.
385
386 Args:
387 request: The request to allocate slots.
388 num_new_tokens: The number of new tokens to be allocated and computed.
389 num_new_computed_tokens: The number of new computed tokens just
390 hitting the prefix caching, excluding external tokens.
391 new_computed_blocks: The cached blocks for the above new computed
392 tokens, grouped as a tuple by kv cache groups.
393 num_lookahead_tokens: The number of speculative tokens to allocate.
394 This is used by spec decode proposers with kv-cache such
395 as eagle.
396 num_external_computed_tokens: The number of tokens that their
397 KV caches are not cached by vLLM but cached by the connector.
398 delay_cache_blocks: Whether to skip caching the blocks. This is
399 used by P/D when allocating blocks used in a KV transfer
400 which will complete in a future step.
401 num_encoder_tokens: The number of encoder tokens to allocate for
402 cross-attention in encoder-decoder models(e.g., Whisper).
403 For decoder-only models, this should be 0.
404 full_sequence_must_fit: Only allocate blocks if the KV cache has enough
405 free blocks to hold the full sequence, accounting for prefix cache hits
406 and sliding window. Used as an admission gate to prevent over-admitting
407 requests when chunked prefill would otherwise only check the first chunk
408 reserved_blocks: Number of free blocks that must be left available for
409 other in-flight sequences to complete. The actual allocation is only
410 made if it fits within (free blocks - reserved_blocks). Used to gate
411 async KV-connector loads so their initial allocation cannot consume
412 blocks an already in-flight (prefilling) sequence is relying on.
413 has_scheduled_reqs: Whether any requests are already scheduled to run
414 this step, controls whether watermark is applied.
415
416 Blocks layout:
417 ```
418 ----------------------------------------------------------------------
419 | < comp > | < new_comp > | < ext_comp > | < new > | < lookahead > |
420 ----------------------------------------------------------------------
421 | < to be computed > |
422 ----------------------------------------------------------------------
423 | < to be allocated > |
424 ----------------------------------------------------------------------
425 | < to be cached (roughly, |
426 | details below)> |
427 ----------------------------------------------------------------------
428 | Prefix-cached tokens from either vLLM |
429 | or connector. Can be safely removed if |
430 | they are outside sliding window. |
431 ----------------------------------------------------------------------
432 | < cached by vLLM > | not cached by |
433 | vLLM, but |
434 | ref_cnt | ref_cnt not | cached by |
435 | increased| increased yet| connector |
436 ----------------------------------------------------------------------
437 ```
438
439 Abbrivations:
440
441 ```
442 comp = request.num_computed_tokens
443 new_comp = num_new_computed_tokens
444 = len(new_computed_blocks) * block_size
445 ext_comp = num_external_computed_tokens, cached by the connector
446 new = num_new_tokens, including unverified draft tokens
447 lookahead = num_lookahead_tokens
448 ```
449
450 NOTE: for new tokens which include both verified and unverified draft
451 tokens, we only cache the verified tokens (by capping the number at
452 `request.num_tokens`).
453
454 The allocation has three stages:
455 - Free unnecessary blocks in `comp` and check
456 if we have sufficient free blocks (return None if not).
457 - Handle prefix tokens (`comp + new_comp + ext_comp`):
458 - Free unnecessary blocks (e.g. outside sliding window)
459 - Allocate new blocks for `ext_comp` tokens inside
460 sliding window
461 - Allocate new blocks for tokens to be computed (`new + lookahead`)
462
463 Returns:
464 A list of new allocated blocks.
465 """
466 # When loading KV data asynchronously, we may have zero new tokens to
467 # compute while still allocating slots for externally computed tokens.
468 if num_new_tokens == 0 and num_external_computed_tokens == 0:
469 raise ValueError(
470 "num_new_tokens must be greater than 0 when there are no "
471 "external computed tokens"
472 )
473
474 if new_computed_blocks is not None:
475 new_computed_block_list = new_computed_blocks.blocks
476 if self.retained_hit_group_ids:
477 # Only retain the prefix covered by the local and external hits.
478 reused = num_new_computed_tokens + num_external_computed_tokens
479 groups = list(new_computed_block_list)
480 for group_id in self.retained_hit_group_ids:
481 manager = self.coordinator.single_type_managers[group_id]
482 groups[group_id] = groups[group_id][
483 : cdiv(reused, manager.block_size)
484 ]
485 new_computed_block_list = tuple(groups)
486 else:
487 new_computed_block_list = self.empty_kv_cache_blocks.blocks
488
489 # The number of computed tokens is the number of computed tokens plus
490 # the new prefix caching hits
491 num_local_computed_tokens = (
492 request.num_computed_tokens + num_new_computed_tokens
493 )
494 total_computed_tokens = min(
495 num_local_computed_tokens + num_external_computed_tokens,
496 self.max_model_len,
497 )
498
499 watermark_blocks = 0
500 # The watermark is applied to waiting/preempted requests only, and only
501 # when there's at least one request already scheduled.
502 if has_scheduled_reqs and request.status in (
503 RequestStatus.WAITING,
504 RequestStatus.PREEMPTED,
505 ):
506 watermark_blocks = self.watermark_blocks
507
508 if full_sequence_must_fit:
509 # First check and fail if the full request sequence won't fit.
510 full_num_tokens = min(request.num_tokens, self.max_model_len)
511
512 num_blocks_to_allocate = self.coordinator.get_num_blocks_to_allocate(
513 request_id=request.request_id,
514 num_tokens=full_num_tokens,
515 new_computed_blocks=new_computed_block_list,
516 num_encoder_tokens=num_encoder_tokens,
517 total_computed_tokens=total_computed_tokens,
518 num_local_computed_tokens=num_local_computed_tokens,
519 num_tokens_main_model=full_num_tokens,
520 apply_admission_cap=True,
521 )
522 required_blocks = num_blocks_to_allocate + watermark_blocks
523 if required_blocks > self.block_pool.get_num_free_blocks():
524 return None
525
526 num_tokens_main_model = total_computed_tokens + num_new_tokens
527 num_tokens_need_slot = min(
528 num_tokens_main_model + num_lookahead_tokens, self.max_model_len
529 )
530
531 # Free the blocks that are skipped during the attention computation
532 # (e.g., tokens outside the sliding window).
533 # We can do this even if we cannot schedule this request due to
534 # insufficient free blocks.
535 # Should call this function before allocating new blocks to reduce
536 # the number of evicted blocks.
537 # Free on the processed-token basis: in-flight steps' attention windows
538 # still read blocks below the optimistic boundary, and rejected spec
539 # tokens can roll it back.
540 self.coordinator.remove_skipped_blocks(
541 request.request_id,
542 max(0, total_computed_tokens - request.num_in_flight_tokens),
543 num_prompt_tokens=request.num_prompt_tokens,
544 )
545
546 num_blocks_to_allocate = self.coordinator.get_num_blocks_to_allocate(
547 request_id=request.request_id,
548 num_tokens=num_tokens_need_slot,
549 new_computed_blocks=new_computed_block_list,
550 num_encoder_tokens=num_encoder_tokens,
551 total_computed_tokens=num_local_computed_tokens
552 + num_external_computed_tokens,
553 num_local_computed_tokens=num_local_computed_tokens,
554 num_tokens_main_model=num_tokens_main_model,
555 )
556
557 # Keep `reserved_blocks` free for other in-flight sequences, and an
558 # additional watermark of headroom for waiting/preempted admissions.
559 available_blocks = self.block_pool.get_num_free_blocks() - reserved_blocks
560 required_blocks = num_blocks_to_allocate + watermark_blocks
561 if required_blocks > available_blocks:
562 # Cannot allocate new blocks
563 return None
564
565 if (
566 new_computed_block_list is not self.empty_kv_cache_blocks.blocks
567 or num_external_computed_tokens > 0
568 ):
569 # Append the new computed blocks to the request blocks until now to
570 # avoid the case where the new blocks cannot be allocated.
571 self.coordinator.allocate_new_computed_blocks(
572 request_id=request.request_id,
573 new_computed_blocks=new_computed_block_list,
574 num_local_computed_tokens=num_local_computed_tokens,
575 num_external_computed_tokens=num_external_computed_tokens,
576 )
577
578 new_blocks = self.coordinator.allocate_new_blocks(
579 request.request_id,
580 num_tokens_need_slot,
581 num_tokens_main_model,
582 num_encoder_tokens,
583 )
584
585 # P/D: delay caching blocks if we have to recv from
586 # remote. Update state for locally cached blocks.
587 if delay_cache_blocks:
588 return self.create_kv_cache_blocks(new_blocks)
589
590 # NOTE(woosuk): We want to commit (cache) up to num_local_computed_tokens
591 # + num_external_computed_tokens + num_new_tokens, but must exclude
592 # "non-committable" tokens (e.g., draft tokens that could be rejected).
593 # Therefore, we cap the number at `request.num_tokens`, ensuring only
594 # "finalized" tokens are cached.
595 num_tokens_to_cache = min(
596 total_computed_tokens + num_new_tokens,
597 request.num_tokens,
598 )
599 self.coordinator.cache_blocks(request, num_tokens_to_cache)
600
601 return self.create_kv_cache_blocks(new_blocks)

613 def remove_skipped_blocks(
614 self,
615 request_id: str,
616 processed_computed_tokens: int,
617 num_prompt_tokens: int | None = None,
618 ) -> None:
619 """Remove the blocks that are no longer needed from `blocks` and replace
620 the removed blocks with null_block.
621
622 Args:
623 request_id: The request ID.
624 processed_computed_tokens: Computed-token prefix length covering
625 fully processed and committed tokens only (safe to free).
626 num_prompt_tokens: Optional prompt length for R-SWA gap eviction.
627 """

669 def get_num_common_prefix_blocks(self, running_request_id: str) -> list[int]:
670 """Calculate the number of common prefix blocks for each kv cache group.
671
672 The function selects a running request and iterates through its blocks.
673 A block is considered a common prefix block if ALL requests with
674 allocated KV cache share it (i.e., ref_cnt equals the number of entries
675 in req_to_blocks).
676
677 NOTE(woosuk): The number of requests with allocated KV cache is **greater
678 than or equal to** the number of requests scheduled in the current step.
679 This is because having allocated KV cache only indicates that:
680 1. The request has not yet finished, and
681 2. The request holds its blocks unfreed.
682
683 While all scheduled requests must have allocated KV cache, the inverse
684 is not necessarily true. There may be requests with allocated KV cache
685 that are not scheduled in the current step.
686
687 This can result in an edge case where the number of common prefix blocks
688 is 0, even though all scheduled requests share a common prefix. This
689 occurs because there may be unscheduled requests that do not share the
690 common prefix. Currently, this case cannot be easily detected, so the
691 function returns 0 in such cases.
692
693 Args:
694 running_request_id: The request ID of any running request, used to
695 identify the common prefix blocks.
696
697 Returns:
698 list[int]: The number of common prefix blocks for each kv cache
699 group.
700 """
701 return self.coordinator.get_num_common_prefix_blocks(running_request_id)

703 def take_events(self) -> list[KVCacheEvent]:
704 """Take the KV cache events from the block pool.
705
706 Returns:
707 A list of KV cache events.
708 """
709 events = self.block_pool.take_events()
710 for event in events:
711 if not isinstance(event, BlockStored):
712 continue
713 if event.group_idx is None:
714 continue
715 if event.group_idx < 0 or event.group_idx >= len(
716 self.kv_cache_event_metadata
717 ):
718 logger.warning(
719 "Group index `%s` not in KV cache metadata", event.group_idx
720 )
721 continue
722 # Annotate here so BlockPool can keep emitting structural cache
723 # events without owning semantic KV cache spec metadata.
724 kind, sliding_window = self.kv_cache_event_metadata[event.group_idx]
725 event.kv_cache_spec_kind = kind
726 event.kv_cache_spec_sliding_window = sliding_window
727 return events

737 def get_block_ids_for_computed_tokens(
738 self,
739 request_id: str,
740 num_computed_tokens: int,
741 ) -> tuple[list[int], ...]:
742 """Get block ids covering the request's computed tokens."""
743 block_ids = self.get_block_ids(request_id)
744 clipped_block_ids: list[list[int]] = []
745 for group, ids in zip(self.kv_cache_config.kv_cache_groups, block_ids):
746 spec = group.kv_cache_spec
747 if not isinstance(spec, AttentionSpec) or isinstance(
748 spec, (CrossAttentionSpec, EncoderOnlyAttentionSpec)
749 ):
750 clipped_block_ids.append(ids)
751 continue
752
753 num_valid_blocks = cdiv(num_computed_tokens, spec.block_size)
754 clipped_block_ids.append(ids[:num_valid_blocks])
755 return tuple(clipped_block_ids)

757 def estimate_cached_tokens(self, request: Request) -> int:
758 """Estimate the number of tokens cached by the request."""
759 cached_tokens: int | None = None
760 for group, blocks in zip(
761 self.kv_cache_config.kv_cache_groups,
762 self.get_blocks(request.request_id).blocks,
763 ):
764 if not group.kv_cache_spec.prefix_cacheable:
765 continue
766 if isinstance(
767 group.kv_cache_spec,
768 (CrossAttentionSpec, EncoderOnlyAttentionSpec),
769 ):
770 # Cross-attention and encoder-only groups are not prefix cached.
771 continue
772
773 group_cached_tokens = 0
774 for block in blocks:
775 group_cached_tokens = max(
776 group_cached_tokens,
777 block.block_hash_num_tokens or 0,
778 )
779
780 cached_tokens = (
781 group_cached_tokens
782 if cached_tokens is None
783 else min(cached_tokens, group_cached_tokens)
784 )
785
786 return cached_tokens or 0

805 def truncate_computed_blocks(
806 self, blocks: KVCacheBlocks, num_computed_tokens: int
807 ) -> KVCacheBlocks:
808 """Return a lookup-result view truncated at an aligned token endpoint.
809
810 An external hit can supply the final Mamba state even when the local
811 Mamba group ends before this endpoint. Other groups must cover it.
812 Pure slicing: refcounts are untouched and ``blocks`` is not mutated.
813 """
814 truncated: list[list[KVCacheBlock]] = []
815 for group_blocks, manager, group in zip(
816 blocks.blocks,
817 self.coordinator.single_type_managers,
818 self.kv_cache_config.kv_cache_groups,
819 strict=True,
820 ):
821 if not group.kv_cache_spec.prefix_cacheable:
822 # Scratch groups (e.g. the CSA compressor ring) hold a fixed
823 # block covering no token range, so a lookup result never
824 # carries computed blocks for them and their block size does
825 # not divide the endpoint.
826 assert not group_blocks
827 truncated.append([])
828 continue
829 assert num_computed_tokens % manager.block_size == 0
830 num_blocks = num_computed_tokens // manager.block_size
831 if isinstance(group.kv_cache_spec, MambaSpec):
832 num_blocks = min(num_blocks, len(group_blocks))
833 else:
834 assert num_blocks <= len(group_blocks)
835 truncated.append(list(group_blocks[:num_blocks]))
836 return self.create_kv_cache_blocks(tuple(truncated))

845 def get_zeroing_block_ids_in_range(
846 self, request_id: str, start_token: int, end_token: int
847 ) -> list[int]:

859 def record_blocks_for_zeroing(self, request_id: str, start_token: int) -> None:
860 """Re-record the request's blocks from start_token onwards for
861 zeroing, e.g. blocks a failed async KV load left unwritten.

871 mgr.new_block_ids.extend(blk.block_id for blk in blocks[start_idx:])

873 def take_kv_cache_block_copies(
874 self,
875 ) -> tuple[list[KVCacheBlockCopy], list[KVCacheBlock]]:

890 def take_boundary_state_offloads(
891 self,
892 ) -> dict[str, list[tuple[int, int, int]]]:
893 """Drain this step's boundary-state hand-offs for a KV connector.
894
895 Returns ``{request_id: [(group_id, block_id, boundary_tokens), ...]}``
896 for mamba "align" boundary states: the
897 request's committed boundary-state snapshots and, on a sub-block partial
898 hit, the CoW copy of its last-prompt-boundary state. Only mamba "align"
899 groups contribute; empty otherwise. A connector reads the referenced
900 blocks — never resolving them positionally — and offloads them so a
901 later request can hit that prefix.
902 """
903 offloads: dict[str, list[tuple[int, int, int]]] = {}
904 for mgr in self.coordinator.single_type_managers:
905 for (
906 req_id,
907 group_id,
908 block,
909 boundary_tokens,
910 ) in mgr.take_pending_boundary_state_offloads():
911 offloads.setdefault(req_id, []).append(
912 (group_id, block.block_id, boundary_tokens)
913 )
914 return offloads

916 def finalize_partial_tail_offloads(
917 self, request: Request
918 ) -> list[tuple[int, int, int]]:
919 """Consume safe producer partial tails when a request finishes.
920
921 A mamba align table block is a valid boundary source only if no later
922 token was forwarded. The connector pins and queues the exact table
923 block before request cleanup, then releases the pin when every worker
924 reports the store job complete.
925 """
926 offloads: list[tuple[int, int, int]] = []
927 for mgr in self.coordinator.single_type_managers:
928 finalized = mgr.finalize_partial_tail_offload(
929 request.request_id,
930 request.num_computed_tokens,
931 request.num_in_flight_tokens,
932 )
933 if finalized is None:
934 continue
935 group_id, block, boundary_tokens = finalized
936 offloads.append((group_id, block.block_id, boundary_tokens))
937 return offloads
