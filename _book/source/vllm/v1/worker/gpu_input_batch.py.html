65 def __post_init__(self):
66 self.num_prompt_tokens = length_from_prompt_token_ids_or_embeds(
67 self.prompt_token_ids, self.prompt_embeds

69
70 if self.pooling_params is not None:

73 @property
74 def num_tokens(self) -> int:
75 return self.num_prompt_tokens + len(self.output_token_ids)
76
77 def get_token_id(self, idx: int) -> int:
78 if idx < self.num_prompt_tokens:
79 if self.prompt_token_ids is None:
80 raise ValueError(
81 f"Tried to access token index {idx}, but that token was "
82 "provided via prompt_embeds, and its ID is unknown."
83 )
84 return self.prompt_token_ids[idx]
85 if idx - self.num_prompt_tokens < len(self.output_token_ids):
86 return self.output_token_ids[idx - self.num_prompt_tokens]
87 return -1

322 def _register_add_request(self, request: "CachedRequestState") -> int:
323 """Track add-request operations for logits processors.
324 Not applicable to pooling models.
325 """
326
327 # Fill the next empty index if there is one.
328 if (new_req_index := self.batch_update_builder.pop_removed()) is None:
329 # Append to end otherwise.
330 new_req_index = self.num_reqs
331
332 assert new_req_index < self.max_num_reqs
333 self.batch_update_builder.batch_changed = True
334 if request.sampling_params:
335 # Detailed added request metadata is only required for non-pooling
336 # models, to support logitsprocs.
337 self.batch_update_builder.added.append(
338 (
339 new_req_index,
340 request.sampling_params,
341 request.prompt_token_ids,
342 request.output_token_ids,
343 )
344 )
345
346 return new_req_index

348 def add_request(
349 self,
350 request: "CachedRequestState",
351 ) -> int:
352 req_index = self._register_add_request(request)
353
354 req_id = request.req_id
355 if req_index == len(self._req_ids):
356 self._req_ids.append(req_id)
357 self.req_output_token_ids.append(request.output_token_ids)
358 self.spec_token_ids.append([])
359 else:
360 self._req_ids[req_index] = req_id
361 self.req_output_token_ids[req_index] = request.output_token_ids
362 self.spec_token_ids[req_index].clear()
363
364 self.req_id_to_index[req_id] = req_index
365
366 # Copy the prompt token ids and output token ids.
367 num_prompt_tokens = length_from_prompt_token_ids_or_embeds(
368 request.prompt_token_ids, request.prompt_embeds
369 )
370 self.num_prompt_tokens[req_index] = num_prompt_tokens
371 start_idx = num_prompt_tokens
372 end_idx = start_idx + len(request.output_token_ids)
373 if request.prompt_token_ids is not None:
374 self.token_ids_cpu[req_index, :num_prompt_tokens] = request.prompt_token_ids
375 if request.prompt_is_token_ids is not None:
376 self.is_token_ids[req_index, :num_prompt_tokens] = (
377 request.prompt_is_token_ids
378 )
379 else:
380 self.is_token_ids[req_index, :num_prompt_tokens] = True
381 else:
382 self.is_token_ids[req_index, :num_prompt_tokens] = False
383 if request.prompt_embeds is not None:
384 self.req_prompt_embeds[req_index] = request.prompt_embeds
385 self.token_ids_cpu[req_index, start_idx:end_idx] = request.output_token_ids
386 self.is_token_ids[req_index, start_idx:end_idx] = True
387 # Number of tokens without spec decode tokens.
388 self.num_tokens_no_spec[req_index] = request.num_tokens
389
390 if self.use_replayssm:
391 # Ring origin = full context at (re)admission (prompt + any resumed
392 # output), so a resumed request re-anchors past the prompt.
393 self.replayssm_decode_base[req_index] = request.num_tokens
394
395 self.num_computed_tokens_cpu[req_index] = request.num_computed_tokens
396 self.block_table.add_row(request.block_ids, req_index)
397
398 if sampling_params := request.sampling_params:
399 if sampling_params.sampling_type == SamplingType.GREEDY:
400 # Should avoid division by zero later when apply_temperature.
401 self.temperature_cpu[req_index] = 0.0
402 self.greedy_reqs.add(req_id)
403 else:
404 self.temperature_cpu[req_index] = sampling_params.temperature
405 self.random_reqs.add(req_id)
406
407 self.top_p_cpu[req_index] = sampling_params.top_p
408 if sampling_params.top_p < 1:
409 self.top_p_reqs.add(req_id)
410 top_k = sampling_params.top_k
411 if 0 < top_k < self.vocab_size:
412 self.top_k_reqs.add(req_id)
413 else:
414 top_k = self.vocab_size
415 self.top_k_cpu[req_index] = top_k
416 self.frequency_penalties_cpu[req_index] = sampling_params.frequency_penalty
417 if sampling_params.frequency_penalty != 0.0:
418 self.frequency_penalties_reqs.add(req_id)
419 self.presence_penalties_cpu[req_index] = sampling_params.presence_penalty
420 if sampling_params.presence_penalty != 0.0:
421 self.presence_penalties_reqs.add(req_id)
422 self.repetition_penalties_cpu[req_index] = (
423 sampling_params.repetition_penalty
424 )
425 if sampling_params.repetition_penalty != 1.0:
426 self.repetition_penalties_reqs.add(req_id)
427
428 # NOTE(woosuk): self.generators should not include the requests that
429 # do not have their own generator.
430 if request.generator is not None:
431 self.generators[req_index] = request.generator
432
433 if sampling_params.logprobs is not None:
434 self.num_logprobs[req_id] = (
435 self.vocab_size
436 if sampling_params.logprobs == -1
437 else sampling_params.logprobs
438 )
439
440 # Store specific token IDs to compute logprobs for (more efficient)
441 if sampling_params.logprob_token_ids is not None:
442 self.logprob_token_ids[req_id] = sampling_params.logprob_token_ids
443
444 if sampling_params.allowed_token_ids:
445 self.has_allowed_token_ids.add(req_id)
446 if self.allowed_token_ids_mask_cpu_tensor is None:
447 # Lazy allocation for this tensor, which can be large.
448 # False means we don't fill with -inf.
449 self.allowed_token_ids_mask = torch.zeros(
450 self.max_num_reqs,
451 self.vocab_size,
452 dtype=torch.bool,
453 device=self.device,
454 )
455 self.allowed_token_ids_mask_cpu_tensor = torch.zeros(
456 self.max_num_reqs,
457 self.vocab_size,
458 dtype=torch.bool,
459 device="cpu",
460 )
461 self.allowed_token_ids_mask_cpu_tensor[req_index] = True
462 # False means we don't fill with -inf.
463 self.allowed_token_ids_mask_cpu_tensor[req_index][
464 sampling_params.allowed_token_ids
465 ] = False
466
467 if sampling_params.bad_words_token_ids:
468 self.bad_words_token_ids[req_index] = (
469 sampling_params.bad_words_token_ids
470 )
471 elif pooling_params := request.pooling_params:
472 pooling_states = request.pooling_states
473 assert pooling_states is not None
474
475 self.pooling_params[req_id] = pooling_params
476 self.pooling_states[req_id] = pooling_states
477 self.logits_processing_needs_token_ids[req_index] = False
478 else:
479 raise NotImplementedError("Unrecognized request type")
480
481 # Speculative decoding: by default 1 token is generated.
482 self.num_accepted_tokens_cpu[req_index] = 1
483
484 # Add request lora ID
485 if request.lora_request:
486 lora_id = request.lora_request.lora_int_id
487 if lora_id not in self.lora_id_to_request_ids:
488 self.lora_id_to_request_ids[lora_id] = set()
489
490 self.request_lora_mapping[req_index] = lora_id
491 self.lora_id_to_request_ids[lora_id].add(request.req_id)
492 self.lora_id_to_lora_request[lora_id] = request.lora_request
493 else:
494 # No LoRA
495 self.request_lora_mapping[req_index] = 0
496
497 return req_index

499 def update_req_spec_token_ids(
500 self, request: CachedRequestState, scheduled_spec_tokens: dict[str, list[int]]
501 ) -> None:
502 req_id = request.req_id
503 req_index = self.req_id_to_index[req_id]
504 cur_spec_token_ids = self.spec_token_ids[req_index]
505 # When speculative decoding is used with structured output,
506 # the scheduler can drop draft tokens that do not
507 # conform to the schema. This can result in
508 # scheduler_output.scheduled_spec_decode_tokens being empty,
509 # even when speculative decoding is enabled.
510 cur_spec_token_ids.clear()
511 spec_token_ids = scheduled_spec_tokens.get(req_id, ())
512 num_spec_tokens = len(spec_token_ids)
513 request.prev_num_draft_len = num_spec_tokens
514 if not spec_token_ids:
515 return
516
517 # For async scheduling, token_ids_cpu assigned from
518 # spec_token_ids are placeholders and will be overwritten in
519 # _prepare_input_ids.
520 start_index = self.num_tokens_no_spec[req_index]
521 end_token_index = start_index + num_spec_tokens
522 self.token_ids_cpu[req_index, start_index:end_token_index] = spec_token_ids
523 self.is_token_ids[req_index, start_index:end_token_index] = True
524 cur_spec_token_ids.extend(spec_token_ids)

526 def remove_request(self, req_id: str) -> int | None:
527 """This method must always be followed by a call to condense().
528
529 Args:
530 req_id: request to remove
531
532 Returns:
533 Removed request index, or `None` if `req_id` not recognized
534 """
535
536 req_index = self.req_id_to_index.pop(req_id, None)
537 if req_index is None:
538 return None
539
540 self.batch_update_builder.removed_append(req_index)
541 self._req_ids[req_index] = None
542 self.req_output_token_ids[req_index] = None
543 self.spec_token_ids[req_index].clear()
544 self.block_table.clear_row(req_index)
545
546 # LoRA
547 lora_id = self.request_lora_mapping[req_index]
548 if lora_id != 0:
549 lora_req_ids = self.lora_id_to_request_ids[lora_id]
550 lora_req_ids.discard(req_id)
551 if not lora_req_ids:
552 del self.lora_id_to_request_ids[lora_id]
553 del self.lora_id_to_lora_request[lora_id]
554 self.request_lora_mapping[req_index] = 0
555
556 if self.is_pooling_model:
557 self.pooling_params.pop(req_id, None)
558 self.pooling_states.pop(req_id, None)
559 return req_index
560
561 self.greedy_reqs.discard(req_id)
562 self.random_reqs.discard(req_id)
563 self.top_p_reqs.discard(req_id)
564 self.top_k_reqs.discard(req_id)
565 self.frequency_penalties_reqs.discard(req_id)
566 self.presence_penalties_reqs.discard(req_id)
567 self.repetition_penalties_reqs.discard(req_id)
568 self.generators.pop(req_index, None)
569 self.num_logprobs.pop(req_id, None)
570 self.logprob_token_ids.pop(req_id, None)
571 if self.prev_req_id_to_index is not None:
572 self.prev_req_id_to_index.pop(req_id, None)
573
574 self.has_allowed_token_ids.discard(req_id)
575 if self.allowed_token_ids_mask_cpu_tensor is not None:
576 # False means we don't fill with -inf.
577 self.allowed_token_ids_mask_cpu_tensor[req_index].fill_(False)
578 self.bad_words_token_ids.pop(req_index, None)
579 self.thinking_token_budget_reqs.discard(req_id)
580 return req_index

582 def swap_states(self, i1: int, i2: int) -> None:
583 old_id_i1 = self._req_ids[i1]
584 old_id_i2 = self._req_ids[i2]
585 # Only swap the active token prefix for each request. Copying full
586 # max_model_len rows is expensive and unnecessary during reordering.
587 i1_active_token_count = self._get_active_token_count(i1)
588 i2_active_token_count = self._get_active_token_count(i2)
589 max_active_token_count = max(i1_active_token_count, i2_active_token_count)
590
591 self._req_ids[i1], self._req_ids[i2] = self._req_ids[i2], self._req_ids[i1] # noqa
592 self.req_output_token_ids[i1], self.req_output_token_ids[i2] = (
593 self.req_output_token_ids[i2],
594 self.req_output_token_ids[i1],
595 )
596 self.spec_token_ids[i1], self.spec_token_ids[i2] = (
597 self.spec_token_ids[i2],
598 self.spec_token_ids[i1],
599 )
600 assert old_id_i1 is not None and old_id_i2 is not None
601 self.req_id_to_index[old_id_i1], self.req_id_to_index[old_id_i2] = (
602 self.req_id_to_index[old_id_i2],
603 self.req_id_to_index[old_id_i1],
604 )
605 self.num_tokens_no_spec[i1], self.num_tokens_no_spec[i2] = (
606 self.num_tokens_no_spec[i2],
607 self.num_tokens_no_spec[i1],
608 )
609 self.num_prompt_tokens[i1], self.num_prompt_tokens[i2] = (
610 self.num_prompt_tokens[i2],
611 self.num_prompt_tokens[i1],
612 )
613 if self.use_replayssm:
614 self.replayssm_decode_base[i1], self.replayssm_decode_base[i2] = (
615 self.replayssm_decode_base[i2],
616 self.replayssm_decode_base[i1],
617 )
618 self.num_computed_tokens_cpu[i1], self.num_computed_tokens_cpu[i2] = (
619 self.num_computed_tokens_cpu[i2],
620 self.num_computed_tokens_cpu[i1],
621 )
622
623 # NOTE: the following is unsafe
624 # self.token_ids_cpu[i1, ...], self.token_ids_cpu[i2, ...], =\
625 # self.token_ids_cpu[i2, ...], self.token_ids_cpu[i1, ...]
626 # instead, we need to temporarily copy the data for one of the indices
627 tmp_token_ids = self.token_ids_cpu[i1, :max_active_token_count].copy()
628 self.token_ids_cpu[i1, :max_active_token_count] = self.token_ids_cpu[
629 i2, :max_active_token_count
630 ]
631 self.token_ids_cpu[i2, :max_active_token_count] = tmp_token_ids
632
633 self.is_token_ids[[i1, i2], :max_active_token_count] = self.is_token_ids[
634 [i2, i1], :max_active_token_count
635 ]
636
637 # Swap prompt embeddings if they exist
638 embeds_i1 = self.req_prompt_embeds.get(i1)
639 embeds_i2 = self.req_prompt_embeds.get(i2)
640 if embeds_i1 is not None:
641 self.req_prompt_embeds[i2] = embeds_i1
642 else:
643 self.req_prompt_embeds.pop(i2, None)
644 if embeds_i2 is not None:
645 self.req_prompt_embeds[i1] = embeds_i2
646 else:
647 self.req_prompt_embeds.pop(i1, None)
648
649 self.block_table.swap_row(i1, i2)
650
651 self.request_lora_mapping[i1], self.request_lora_mapping[i2] = (
652 self.request_lora_mapping[i2],
653 self.request_lora_mapping[i1],
654 )
655
656 if self.is_pooling_model:
657 # Sampling and logits parameters don't apply to pooling models.
658 return
659
660 # For autoregressive models, track detailed request reordering info
661 # to support logitsprocs.
662 self.batch_update_builder.moved.append((i1, i2, MoveDirectionality.SWAP))
663
664 self.temperature_cpu[i1], self.temperature_cpu[i2] = (
665 self.temperature_cpu[i2],
666 self.temperature_cpu[i1],
667 )
668 self.top_p_cpu[i1], self.top_p_cpu[i2] = self.top_p_cpu[i2], self.top_p_cpu[i1]
669 self.top_k_cpu[i1], self.top_k_cpu[i2] = self.top_k_cpu[i2], self.top_k_cpu[i1]
670 self.frequency_penalties_cpu[i1], self.frequency_penalties_cpu[i2] = (
671 self.frequency_penalties_cpu[i2],
672 self.frequency_penalties_cpu[i1],
673 )
674 self.presence_penalties_cpu[i1], self.presence_penalties_cpu[i2] = (
675 self.presence_penalties_cpu[i2],
676 self.presence_penalties_cpu[i1],
677 )
678 self.repetition_penalties_cpu[i1], self.repetition_penalties_cpu[i2] = (
679 self.repetition_penalties_cpu[i2],
680 self.repetition_penalties_cpu[i1],
681 )
682 self.num_accepted_tokens_cpu[i1], self.num_accepted_tokens_cpu[i2] = (
683 self.num_accepted_tokens_cpu[i2],
684 self.num_accepted_tokens_cpu[i1],
685 )
686
687 swap_dict_values(self.generators, i1, i2)
688 swap_dict_values(self.bad_words_token_ids, i1, i2)
689
690 if self.allowed_token_ids_mask_cpu_tensor is not None:
691 (
692 self.allowed_token_ids_mask_cpu_tensor[i1],
693 self.allowed_token_ids_mask_cpu_tensor[i2],
694 ) = (
695 self.allowed_token_ids_mask_cpu_tensor[i2],
696 self.allowed_token_ids_mask_cpu_tensor[i1],
697 )

699 def _get_active_token_count(self, req_index: int) -> int:
700 return int(self.num_tokens_no_spec[req_index]) + len(
701 self.spec_token_ids[req_index]

704 def condense(self) -> None:
705 """Slide non-empty requests down into lower, empty indices.
706
707 Any consecutive empty indices at the very end of the list are not
708 filled.
709
710 Returns:
711 swaps: list of (from,to) swap tuples for moved requests
712 empty_req_indices: indices not filled by condensation
713 """
714 num_reqs = self.num_reqs
715
716 if not (empty_req_indices := self.batch_update_builder.removed):
717 # All removed requests were replaced by added requests, or else no
718 # requests were removed at all. No condense() needed
719 return
720 if num_reqs == 0:
721 # The batched states are empty.
722 self._req_ids.clear()
723 self.req_output_token_ids.clear()
724 self.spec_token_ids.clear()
725 return
726
727 # NOTE(woosuk): This function assumes that the empty_req_indices
728 # is sorted in descending order.
729 last_req_index = num_reqs + len(empty_req_indices) - 1
730 while empty_req_indices:
731 # Find the largest non-empty index.
732 while last_req_index in empty_req_indices:
733 last_req_index -= 1
734
735 # Find the smallest empty index.
736 empty_index = self.batch_update_builder.peek_removed()
737 assert empty_index is not None
738 if empty_index >= last_req_index:
739 break
740
741 # Move active request down into empty request
742 # index.
743 self.batch_update_builder.pop_removed()
744 req_id = self._req_ids[last_req_index]
745 output_token_ids = self.req_output_token_ids[last_req_index]
746 assert req_id is not None
747 self._req_ids[empty_index] = req_id
748 self._req_ids[last_req_index] = None
749 self.req_output_token_ids[empty_index] = output_token_ids
750 self.req_output_token_ids[last_req_index] = None
751 self.req_id_to_index[req_id] = empty_index
752
753 num_tokens = self._get_active_token_count(last_req_index)
754
755 (self.spec_token_ids[last_req_index], self.spec_token_ids[empty_index]) = (
756 self.spec_token_ids[empty_index],
757 self.spec_token_ids[last_req_index],
758 )
759 self.spec_token_ids[last_req_index].clear()
760
761 self.token_ids_cpu[empty_index, :num_tokens] = self.token_ids_cpu[
762 last_req_index, :num_tokens
763 ]
764 self.is_token_ids[empty_index, :num_tokens] = self.is_token_ids[
765 last_req_index, :num_tokens
766 ]
767 if last_req_index in self.req_prompt_embeds:
768 self.req_prompt_embeds[empty_index] = self.req_prompt_embeds.pop(
769 last_req_index
770 )
771 self.num_tokens_no_spec[empty_index] = self.num_tokens_no_spec[
772 last_req_index
773 ]
774 self.num_prompt_tokens[empty_index] = self.num_prompt_tokens[last_req_index]
775 if self.use_replayssm:
776 self.replayssm_decode_base[empty_index] = self.replayssm_decode_base[
777 last_req_index
778 ]
779 self.num_computed_tokens_cpu[empty_index] = self.num_computed_tokens_cpu[
780 last_req_index
781 ]
782 self.block_table.move_row(last_req_index, empty_index)
783
784 self.request_lora_mapping[empty_index] = self.request_lora_mapping[
785 last_req_index
786 ]
787
788 if self.is_pooling_model:
789 last_req_index -= 1
790 # Sampling state not used by pooling models.
791 continue
792
793 # Autoregressive models require detailed tracking of condense
794 # operations to support logitsprocs
795 self.batch_update_builder.moved.append(
796 (last_req_index, empty_index, MoveDirectionality.UNIDIRECTIONAL)
797 )
798
799 self.temperature_cpu[empty_index] = self.temperature_cpu[last_req_index]
800 self.top_p_cpu[empty_index] = self.top_p_cpu[last_req_index]
801 self.top_k_cpu[empty_index] = self.top_k_cpu[last_req_index]
802 self.frequency_penalties_cpu[empty_index] = self.frequency_penalties_cpu[
803 last_req_index
804 ]
805 self.presence_penalties_cpu[empty_index] = self.presence_penalties_cpu[
806 last_req_index
807 ]
808 self.repetition_penalties_cpu[empty_index] = self.repetition_penalties_cpu[
809 last_req_index
810 ]
811 self.num_accepted_tokens_cpu[empty_index] = self.num_accepted_tokens_cpu[
812 last_req_index
813 ]
814 generator = self.generators.pop(last_req_index, None)
815 if generator is not None:
816 self.generators[empty_index] = generator
817
818 # TODO convert these to LogitsProcessors
819 if self.allowed_token_ids_mask_cpu_tensor is not None:
820 self.allowed_token_ids_mask_cpu_tensor[empty_index] = (
821 self.allowed_token_ids_mask_cpu_tensor[last_req_index]
822 )
823
824 bad_words_token_ids = self.bad_words_token_ids.pop(last_req_index, None)
825 if bad_words_token_ids is not None:
826 self.bad_words_token_ids[empty_index] = bad_words_token_ids
827
828 # Decrement last_req_index since it is now empty.
829 last_req_index -= 1
830
831 # Trim lists to the batch size.
832 del self._req_ids[num_reqs:]
833 del self.req_output_token_ids[num_reqs:]
834 del self.spec_token_ids[num_reqs:]

836 def refresh_metadata(self):
837 """Apply any batch updates to sampling metadata."""
838
839 if self.is_pooling_model:
840 batch_changed = self.batch_update_builder.reset()
841 if batch_changed:
842 self.sampling_metadata = self._make_sampling_metadata()
843 return
844
845 # For non-pooling models - generate and apply logitsprocs update;
846 # reset batch update tracking.
847 # Update sampling metadata if batch state is changed.
848 batch_update = self.batch_update_builder.get_and_reset(self.num_reqs)
849 if self.thinking_budget_state_holder is not None and batch_update:
850 self.thinking_budget_state_holder.sync_batch(batch_update)
851 for logit_proc in self.logitsprocs.all:
852 logit_proc.update_state(batch_update)
853 if batch_update:
854 self.sampling_metadata = self._make_sampling_metadata()

856 def _make_sampling_metadata(self) -> SamplingMetadata:
857 num_reqs = self.num_reqs
858 if not self.all_greedy:
859 temperature = copy_slice(
860 self.temperature_cpu_tensor, self.temperature, num_reqs
861 )
862 else:
863 temperature = None
864 if not self.no_top_p:
865 copy_slice(self.top_p_cpu_tensor, self.top_p, num_reqs)
866 if not self.no_top_k:
867 copy_slice(self.top_k_cpu_tensor, self.top_k, num_reqs)
868
869 if not self.no_penalties:
870 # Since syncing these tensors is expensive only copy them
871 # if necessary i.e. if there are requests which require
872 # penalties to be applied during sampling.
873 copy_slice(
874 self.frequency_penalties_cpu_tensor, self.frequency_penalties, num_reqs
875 )
876 copy_slice(
877 self.presence_penalties_cpu_tensor, self.presence_penalties, num_reqs
878 )
879 copy_slice(
880 self.repetition_penalties_cpu_tensor,
881 self.repetition_penalties,
882 num_reqs,
883 )
884
885 needs_prompt_token_ids = (
886 not self.no_penalties
887 or self.logits_processing_needs_token_ids[:num_reqs].any()
888 )
889 # The device prompt tokens are used only for applying penalties or
890 # pooling methods that explicitly request GPU token IDs.
891 # Hence copy these tensors only when there are requests which
892 # need penalties/step_pooler to be applied.
893 prompt_token_ids_cpu = (
894 self._make_prompt_token_ids_cpu_tensor() if needs_prompt_token_ids else None
895 )
896 prompt_token_ids = (
897 prompt_token_ids_cpu.to(device=self.device, non_blocking=True)
898 if prompt_token_ids_cpu is not None
899 else None
900 )
901
902 # Only set output_token_ids if required by the current requests'
903 # sampling parameters.
904 holder = self.thinking_budget_state_holder
905 thinking_budget_tracks_reqs = (
906 holder is not None and holder.has_tracked_requests()
907 )
908 needs_output_token_ids = (
909 not self.no_penalties
910 or bool(self.bad_words_token_ids)
911 or self.logitsprocs_need_output_token_ids
912 or thinking_budget_tracks_reqs
913 )
914 output_token_ids = (
915 cast(list[list[int]], self.req_output_token_ids)
916 if needs_output_token_ids
917 else []
918 )
919
920 allowed_token_ids_mask: torch.Tensor | None = None
921 if not self.no_allowed_token_ids:
922 assert self.allowed_token_ids_mask is not None
923 copy_slice(
924 self.allowed_token_ids_mask_cpu_tensor,
925 self.allowed_token_ids_mask,
926 num_reqs,
927 )
928 allowed_token_ids_mask = self.allowed_token_ids_mask[:num_reqs]
929
930 # Build per-request logprob_token_ids mapping: req_index -> token_ids
931 logprob_token_ids_by_index: dict[int, list[int]] | None = None
932 if self.logprob_token_ids:
933 logprob_token_ids_by_index = {}
934 for req_id, token_ids in self.logprob_token_ids.items():
935 if req_id in self.req_id_to_index:
936 req_index = self.req_id_to_index[req_id]
937 logprob_token_ids_by_index[req_index] = token_ids
938
939 return SamplingMetadata(
940 temperature=temperature,
941 all_greedy=self.all_greedy,
942 all_random=self.all_random,
943 top_p=None if self.no_top_p else self.top_p[:num_reqs],
944 top_k=None if self.no_top_k else self.top_k[:num_reqs],
945 generators=self.generators,
946 max_num_logprobs=self.max_num_logprobs,
947 logprob_token_ids=logprob_token_ids_by_index,
948 prompt_token_ids=prompt_token_ids,
949 frequency_penalties=self.frequency_penalties[:num_reqs],
950 presence_penalties=self.presence_penalties[:num_reqs],
951 repetition_penalties=self.repetition_penalties[:num_reqs],
952 output_token_ids=output_token_ids,
953 spec_token_ids=self.spec_token_ids,
954 no_penalties=self.no_penalties,
955 allowed_token_ids_mask=allowed_token_ids_mask,
956 bad_words_token_ids=self.bad_words_token_ids,
957 logitsprocs=self.logitsprocs,
958 thinking_budget_state_holder=self.thinking_budget_state_holder,
959 )

1043 def update_async_output_token_ids(self) -> None:
1044 """
1045 In async scheduling case, update output_token_ids in sampling metadata
1046 from prior steps sampled token ids once they've finished copying to CPU.
1047 This is called right before they are needed by the logits processors.
1048 """
1049 output_token_ids = self.sampling_metadata.output_token_ids
1050 if self.sampled_token_ids_cpu is None or not output_token_ids:
1051 # Output token ids not needed or not async scheduling.
1052 return
1053
1054 assert self.prev_req_id_to_index is not None
1055 sampled_token_ids = None
1056 for index, req_id in enumerate(self.req_ids):
1057 prev_index = self.prev_req_id_to_index.get(req_id)
1058 if prev_index is None:
1059 continue
1060 req_output_token_ids = output_token_ids[index]
1061 if not req_output_token_ids or req_output_token_ids[-1] != -1:
1062 # Final output id is not a placeholder, some tokens must have
1063 # been discarded after a kv-load failure.
1064 continue
1065 if sampled_token_ids is None:
1066 assert self.async_copy_ready_event is not None
1067 self.async_copy_ready_event.synchronize()
1068 sampled_token_ids = self.sampled_token_ids_cpu.tolist()
1069 # Replace placeholder token id(s) with actual sampled id(s).
1070 new_ids: list[int] = sampled_token_ids[prev_index]
1071 if not new_ids:
1072 continue
1073 num_sampled_ids = len(new_ids) if new_ids[-1] != -1 else new_ids.index(-1)
1074 # Also account for case where there may be a smaller number of
1075 # output placeholders (tokens can be discarded after kv-load
1076 # failure) or a larger number (async spec decode adds optimistic
1077 # placeholders that may exceed the actual acceptance count).
1078 first_placeholder = len(req_output_token_ids)
1079 while (
1080 first_placeholder > 0
1081 and req_output_token_ids[first_placeholder - 1] == -1
1082 ):
1083 first_placeholder -= 1
1084 num_placeholders = len(req_output_token_ids) - first_placeholder
1085 num_to_replace = min(num_sampled_ids, num_placeholders)
1086 del new_ids[num_to_replace:]
1087 req_output_token_ids[first_placeholder:] = new_ids
1088 # ^ Implicitly resizes to (first_placeholder + num_to_replace)
1089
1090 def update_async_spec_token_ids(self, draft_token_ids: list[list[int]]) -> None:
1091 """
1092 In async scheduling case, update spec_token_ids in sampling metadata with
1093 real draft token ids from prior step. This is called right before they are
1094 needed by the rejection sampler for penalty/bad_words computation.
1095 """
1096 if not draft_token_ids or not self.prev_req_id_to_index:
1097 return
1098
1099 if (spec_token_ids := self.sampling_metadata.spec_token_ids) is not None:
1100 for req_id, spec_ids in zip(self.req_ids, spec_token_ids):
1101 if spec_ids:
1102 prev_index = self.prev_req_id_to_index.get(req_id)
1103 if prev_index is not None:
1104 draft_ids = draft_token_ids[prev_index]
1105 if draft_ids:
1106 del draft_ids[len(spec_ids) :]
1107 spec_ids.clear()
1108 spec_ids.extend(draft_ids)
