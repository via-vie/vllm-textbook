211class WorkerWrapperBase:
212 """
213 This class represents one process in an executor/engine. It is responsible
214 for lazily initializing the worker and handling the worker's lifecycle.
215 We first instantiate the WorkerWrapper, which remembers the worker module
216 and class name. Then, when we call `update_environment_variables`, and the
217 real initialization happens in `init_worker`.
218 """
219
220 def __init__(
221 self,
222 rpc_rank: int = 0,
223 global_rank: int | None = None,
224 ) -> None:
225 """
226 Initialize the worker wrapper with the given vllm_config and rpc_rank.
227 Note: rpc_rank is the rank of the worker in the executor. In most cases,
228 it is also the rank of the worker in the distributed group. However,
229 when multiple executors work together, they can be different.
230 e.g. in the case of SPMD-style offline inference with TP=2,
231 users can launch 2 engines/executors, each with only 1 worker.
232 All workers have rpc_rank=0, but they have different ranks in the TP
233 group.
234 """
235 self.rpc_rank: int = rpc_rank
236 self.global_rank: int = self.rpc_rank if global_rank is None else global_rank
237
238 # Initialized after init_worker is called
239 self.worker: WorkerBase
240 self.vllm_config: VllmConfig
241
242 def shutdown(self) -> None:
243 if self.worker is not None:
244 self.worker.shutdown()
245
246 def update_environment_variables(
247 self,
248 envs_list: list[dict[str, str]],
249 ) -> None:
250 envs = envs_list[self.rpc_rank]
251 update_environment_variables(envs)
252
253 @instrument(span_name="Worker init")
254 def init_worker(self, all_kwargs: list[dict[str, Any]]) -> None:
255 """
256 Here we inject some common logic before initializing the worker.
257 Arguments are passed to the worker class constructor.
258 """
259 kwargs = all_kwargs[self.rpc_rank]
260
261 vllm_config: VllmConfig | None = kwargs.get("vllm_config")
262 assert vllm_config is not None, (
263 "vllm_config is required to initialize the worker"
264 )
265 self.vllm_config = vllm_config
266
267 vllm_config.enable_trace_function_call_for_thread()
268
269 from vllm.plugins import load_general_plugins
270
271 load_general_plugins()
272
273 parallel_config = vllm_config.parallel_config
274 if isinstance(parallel_config.worker_cls, str):
275 worker_class: type[WorkerBase] = resolve_obj_by_qualname(
276 parallel_config.worker_cls
277 )
278 else:
279 raise ValueError(
280 "passing worker_cls is no longer supported. "
281 "Please pass keep the class in a separate module "
282 "and pass the qualified name of the class as a string."
283 )
284
285 if parallel_config.worker_extension_cls:
286 worker_extension_cls = resolve_obj_by_qualname(
287 parallel_config.worker_extension_cls
288 )
289 extended_calls = []
290 if worker_extension_cls not in worker_class.__bases__:
291 # check any conflicts between worker and worker_extension_cls
292 for attr in dir(worker_extension_cls):
293 if attr.startswith("__"):
294 continue
295 assert not hasattr(worker_class, attr), (
296 f"Worker class {worker_class} already has an attribute"
297 f" {attr}, which conflicts with the worker"
298 f" extension class {worker_extension_cls}."
299 )
300 if callable(getattr(worker_extension_cls, attr)):
301 extended_calls.append(attr)
302 # dynamically inherit the worker extension class
303 worker_class.__bases__ = worker_class.__bases__ + (
304 worker_extension_cls,
305 )
306 logger.info(
307 "Injected %s into %s for extended collective_rpc calls %s",
308 worker_extension_cls,
309 worker_class,
310 extended_calls,
311 )
312
313 assigned_physical_gpu_ids = kwargs.pop("assigned_physical_gpu_ids", None)
314 if assigned_physical_gpu_ids is not None:
315 vllm_config.parallel_config.assigned_physical_gpu_ids = (
316 assigned_physical_gpu_ids
317 )
318
319 shared_worker_lock = kwargs.pop("shared_worker_lock", None)
320 if shared_worker_lock is None:
321 msg = (
322 "Missing `shared_worker_lock` argument from executor. "
323 "This argument is needed for mm_processor_cache_type='shm'."
324 )
325
326 mm_config = vllm_config.model_config.multimodal_config
327 if mm_config and mm_config.mm_processor_cache_type == "shm":
328 raise ValueError(msg)
329 else:
330 logger.warning_once(msg)
331
332 self.mm_receiver_cache = None
333 else:
334 self.mm_receiver_cache = (
335 MULTIMODAL_REGISTRY.worker_receiver_cache_from_config(
336 vllm_config,
337 shared_worker_lock,
338 )
339 )
340
341 with set_current_vllm_config(self.vllm_config):
342 # To make vLLM config available during worker initialization
343 self.worker = worker_class(**kwargs)
344
345 def initialize_from_config(self, kv_cache_configs: list[Any]) -> None:
346 kv_cache_config = kv_cache_configs[self.global_rank]
347 assert self.vllm_config is not None
348 with set_current_vllm_config(self.vllm_config):
349 self.worker.initialize_from_config(kv_cache_config) # type: ignore
350
351 def init_device(self):
352 assert self.vllm_config is not None
353 with set_current_vllm_config(self.vllm_config):
354 # To make vLLM config available during device initialization
355 self.worker.init_device() # type: ignore
356
357 def __getattr__(self, attr: str):
358 return getattr(self.worker, attr)
359
360 def _apply_mm_cache(self, scheduler_output: SchedulerOutput) -> None:
361 mm_cache = self.mm_receiver_cache
362 if mm_cache is None:
363 return
364
365 for req_data in scheduler_output.scheduled_new_reqs:
366 req_data.mm_features = mm_cache.get_and_update_features(
367 req_data.mm_features
368 )
369
370 def execute_model(
371 self, scheduler_output: SchedulerOutput
372 ) -> ModelRunnerOutput | AsyncModelRunnerOutput | None:
373 self._apply_mm_cache(scheduler_output)
374
375 return self.worker.execute_model(scheduler_output)
376
377 def reset_mm_cache(self) -> None:
378 mm_receiver_cache = self.mm_receiver_cache
379 if mm_receiver_cache is not None:
380 mm_receiver_cache.clear_cache()
381
382 self.worker.reset_mm_cache()
