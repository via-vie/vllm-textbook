43@dataclass
44class InputBatch:
45 # batch_idx -> req_id
46 req_ids: list[str]
47 num_reqs: int
48 num_reqs_after_padding: int
49
50 # batch_idx -> req_state_idx
51 idx_mapping: torch.Tensor
52 idx_mapping_np: np.ndarray
53 # Identical to idx_mapping except for spec decoding.
54 expanded_idx_mapping: torch.Tensor
55 # [total_num_logits] position within request for each logit
56 expanded_local_pos: torch.Tensor
57
58 # [num_reqs]
59 # batch_idx -> num_scheduled_tokens, (upper bound when using adaptive verification)
60 num_scheduled_tokens: np.ndarray
61 # number of tokens in the batch,
62 # may be < sum(num_scheduled_tokens) when using adaptive verification
63 num_tokens: int
64 num_tokens_after_padding: int
65 # Sum of draft tokens scheduled across requests.
66 num_draft_tokens: int
67 # [num_reqs] number of draft tokens scheduled for each request, if any.
68 num_draft_tokens_per_req: np.ndarray | None
69
70 # [num_reqs + 1]
71 query_start_loc: torch.Tensor
72 query_start_loc_np: np.ndarray
73 # [num_reqs]
74 seq_lens: torch.Tensor
75 # [num_reqs] CPU upper bound on seq_lens (see CommonAttentionMetadata).
76 seq_lens_cpu_upper_bound: torch.Tensor
77 # [num_reqs]
78 dcp_local_seq_lens: torch.Tensor | None
79 # [num_reqs]
80 num_computed_tokens_np: np.ndarray
81 # [num_reqs]
82 prefill_len_np: np.ndarray
83 # [num_reqs]
84 num_computed_prefill_tokens_np: np.ndarray
85 # [num_reqs] CPU bool array == (num_computed_prefill_tokens_np < prefill_len_np).
86 is_prefilling_np: np.ndarray
87 # == np.any(is_prefilling_np)
88 has_prefill: bool
89
90 # [num_tokens_after_padding]
91 input_ids: torch.Tensor
92 # [num_tokens_after_padding]
93 positions: torch.Tensor
94 # [num_tokens_after_padding]
95 is_padding: torch.Tensor
96
97 # [total_num_logits]
98 logits_indices: torch.Tensor
99 # [num_reqs + 1]
100 cu_num_logits: torch.Tensor
101 cu_num_logits_np: np.ndarray
102
103 # Whether any requests in batch use structured output.
104 has_structured_output_reqs: bool
105
106 # [num_reqs] per-request prompt length, only populated for R-SWA.
107 prompt_lens: torch.Tensor | None
108
109 # Longest query the batch may contain. Set when a cudagraph descriptor promises
110 # a query length this batch's own split does not reach, so attention metadata
111 # stays valid for every replay the graph serves.
112 max_query_len: int | None = None
113
114 # Arms the KV-sharing fast prefill path for this step. Absent for dummy
115 # (cudagraph capture) batches, which run the KV-sharing layers in full.
116 fast_prefill: "FastPrefillBatchMetadata | None" = None
117
118 # [num_reqs] set only under PCP+DCP (see CommonAttentionMetadata).
119 dcp_local_seq_lens_cpu_upper_bound: torch.Tensor | None = None

121 @classmethod
122 def make_dummy(
123 cls,
124 num_reqs: int,
125 num_tokens: int,
126 input_buffers: InputBuffers,
127 max_query_len: int | None = None,
128 ) -> "InputBatch":
129 assert 0 < num_reqs <= num_tokens
130 device = input_buffers.device
131
132 req_ids = [f"req_{i}_{random_uuid()}" for i in range(num_reqs)]
133 idx_mapping_np = np.arange(num_reqs, dtype=np.intp)
134 idx_mapping = torch.arange(num_reqs, dtype=torch.int64, device=device)
135 expanded_idx_mapping = idx_mapping
136 expanded_local_pos = torch.zeros(num_reqs, dtype=torch.int32, device=device)
137
138 # Distribute the remainder evenly so that no dummy request exceeds
139 # ceil(num_tokens / num_reqs) <= max_model_len tokens. Varlen graphs
140 # accept any split with non-empty slots, so this shape works for them
141 # too; attention metadata is built from the promised max_query_len.
142 base_tokens = num_tokens // num_reqs
143 num_extra = num_tokens % num_reqs
144 assert max_query_len is None or base_tokens + (num_extra > 0) <= max_query_len
145 num_scheduled_tokens = np.full(num_reqs, base_tokens, dtype=np.int32)
146 if num_extra > 0:
147 num_scheduled_tokens[-num_extra:] += 1
148 assert int(num_scheduled_tokens.sum()) == num_tokens
149
150 # seq_len equals to query_len
151 input_buffers.seq_lens[: num_reqs - num_extra] = base_tokens
152 input_buffers.seq_lens[num_reqs - num_extra : num_reqs] = base_tokens + 1
153 # Pad for full CUDA graph mode.
154 input_buffers.seq_lens[num_reqs:] = 0
155 seq_lens = input_buffers.seq_lens[:num_reqs]
156
157 query_start_loc_np = np.empty(num_reqs + 1, dtype=np.int32)
158 query_start_loc_np[0] = 0
159 np.cumsum(num_scheduled_tokens, out=query_start_loc_np[1:])
160 input_buffers.query_start_loc[:1] = 0
161 torch.cumsum(
162 seq_lens, dim=0, out=input_buffers.query_start_loc[1 : num_reqs + 1]
163 )
164 # Pad for full CUDA graph mode.
165 input_buffers.query_start_loc[num_reqs + 1 :] = num_tokens
166 query_start_loc = input_buffers.query_start_loc[: num_reqs + 1]
167
168 input_ids = input_buffers.input_ids[:num_tokens].zero_()
169 positions = input_buffers.positions[:num_tokens].zero_()
170
171 input_buffers.is_padding[:num_tokens].fill_(True)
172 is_padding = input_buffers.is_padding[:num_tokens]
173
174 logits_indices = query_start_loc[1:] - 1
175 cu_num_logits = torch.arange(num_reqs + 1, device=device, dtype=torch.int32)
176 cu_num_logits_np = np.arange(num_reqs + 1, dtype=np.int32)
177 # Copy so set_dummy_context can add context in place without touching
178 # num_scheduled_tokens.
179 seq_lens_cpu_upper_bound = torch.from_numpy(num_scheduled_tokens.copy())
180 return cls(
181 req_ids=req_ids,
182 num_reqs=num_reqs,
183 num_reqs_after_padding=num_reqs,
184 idx_mapping=idx_mapping,
185 idx_mapping_np=idx_mapping_np,
186 expanded_idx_mapping=expanded_idx_mapping,
187 expanded_local_pos=expanded_local_pos,
188 num_scheduled_tokens=num_scheduled_tokens,
189 num_tokens=num_tokens,
190 num_tokens_after_padding=num_tokens,
191 num_draft_tokens=0,
192 num_draft_tokens_per_req=None,
193 query_start_loc=query_start_loc,
194 query_start_loc_np=query_start_loc_np,
195 seq_lens=seq_lens,
196 seq_lens_cpu_upper_bound=seq_lens_cpu_upper_bound,
197 dcp_local_seq_lens=None,
198 num_computed_tokens_np=np.zeros(num_reqs, dtype=np.int32),
199 prefill_len_np=np.zeros(num_reqs, dtype=np.int32),
200 num_computed_prefill_tokens_np=np.zeros(num_reqs, dtype=np.int32),
201 is_prefilling_np=np.zeros(num_reqs, dtype=np.bool_),
202 has_prefill=False,
203 input_ids=input_ids,
204 positions=positions,
205 is_padding=is_padding,
206 logits_indices=logits_indices,
207 cu_num_logits=cu_num_logits,
208 cu_num_logits_np=cu_num_logits_np,
209 has_structured_output_reqs=False,
210 prompt_lens=None,
211 max_query_len=max_query_len,
212 )

215def set_dummy_context(
216 input_batch: InputBatch,
217 block_tables: "BlockTables",
218 context_len: int,
219 num_kv_blocks: int,
220 max_model_len: int,
221 input_block_tables: Sequence[torch.Tensor] | None = None,
222) -> None:
223 """Give each dummy request context_len of context, used when profiling step cost."""
224 if input_block_tables is None:
225 input_block_tables = block_tables.input_block_tables
226 if not input_block_tables:
227 # Attention-free models have no KV context to fabricate.
228 return
229 num_reqs = input_batch.num_reqs
230 query_len = input_batch.max_query_len or int(input_batch.num_scheduled_tokens.max())
231 context_len = max(min(context_len, max_model_len - query_len), 0)
232 if not context_len:
233 return
234
235 # Decode-like shape: each request continues after context_len
236 # already-computed tokens.
237 input_batch.seq_lens += context_len
238 input_batch.seq_lens_cpu_upper_bound += context_len
239 input_batch.num_computed_tokens_np.fill(context_len)
240 input_batch.num_computed_prefill_tokens_np.fill(context_len)
241 local_pos = np.arange(input_batch.num_tokens, dtype=np.int64) - np.repeat(
242 input_batch.query_start_loc_np[:-1], input_batch.num_scheduled_tokens
243 )
244 input_batch.positions.copy_(torch.from_numpy(local_pos + context_len))
245
246 seq_len = context_len + query_len
247 for block_table, block_size, bpk in zip(
248 input_block_tables,
249 block_tables.kernel_block_sizes,
250 block_tables.blocks_per_kv_block,
251 ):
252 num_blocks = min(cdiv(seq_len, block_size), block_table.shape[1])
253 # Spans are disjoint until the pool runs out, then they wrap and share
254 # blocks: profiling only needs the reads to be realistic, not distinct.
255 block_ids = torch.arange(
256 num_reqs * num_blocks, dtype=block_table.dtype, device=block_table.device
257 ) % (num_kv_blocks * bpk)
258 block_table[:num_reqs, :num_blocks] = block_ids.view(num_reqs, num_blocks)

261@triton.jit
262def _prepare_prefill_inputs_kernel(
263 input_ids_ptr,
264 next_prefill_tokens_ptr,
265 next_prefill_tokens_stride,
266 num_lookahead,
267 idx_mapping_ptr,
268 query_start_loc_ptr,
269 all_token_ids_ptr,
270 all_token_ids_stride,
271 prefill_lens_ptr,
272 num_computed_tokens_ptr,
273 BLOCK_SIZE: tl.constexpr,
274 LOOKAHEAD_BLOCK: tl.constexpr,
275):
276 batch_idx = tl.program_id(0)
277 req_state_idx = tl.load(idx_mapping_ptr + batch_idx)
278 prefill_len = tl.load(prefill_lens_ptr + req_state_idx)
279 num_computed = tl.load(num_computed_tokens_ptr + req_state_idx)
280 if num_computed >= prefill_len:
281 # Not prefill.
282 return
283
284 query_start = tl.load(query_start_loc_ptr + batch_idx)
285 query_end = tl.load(query_start_loc_ptr + batch_idx + 1)
286 query_len = query_end - query_start
287
288 request_ptr = all_token_ids_ptr + req_state_idx * all_token_ids_stride
289 for i in range(0, query_len, BLOCK_SIZE):
290 block = i + tl.arange(0, BLOCK_SIZE)
291 mask = block < query_len
292 tokens = tl.load(request_ptr + num_computed + block, mask=mask)
293 tl.store(input_ids_ptr + query_start + block, tokens, mask=mask)
294
295 # Store the next num_lookahead prefill tokens.
296 lookahead = tl.arange(0, LOOKAHEAD_BLOCK)
297 pos = num_computed + query_len + lookahead
298 in_lookahead = lookahead < num_lookahead
299 tokens = tl.load(
300 request_ptr + pos, mask=in_lookahead & (pos < prefill_len), other=0
301 )
302 tl.store(
303 next_prefill_tokens_ptr
304 + lookahead * next_prefill_tokens_stride
305 + req_state_idx,
306 tokens,
307 mask=in_lookahead,
308 )

338@triton.jit
339def _prepare_pos_seq_lens_kernel(
340 pos_ptr,
341 seq_lens_ptr,
342 idx_mapping_ptr,
343 query_start_loc_ptr,
344 num_computed_tokens_ptr,
345 max_num_reqs,
346 BLOCK_SIZE: tl.constexpr,
347):
348 req_id = tl.program_id(0)
349 num_reqs = tl.num_programs(0) - 1
350 if req_id == num_reqs:
351 # Pad unused seq_lens as 0 for full CUDA graphs.
352 for i in tl.range(num_reqs, max_num_reqs, BLOCK_SIZE):
353 block = i + tl.arange(0, BLOCK_SIZE)
354 mask = block < max_num_reqs
355 tl.store(seq_lens_ptr + block, 0, mask=mask)
356 return
357
358 req_state_idx = tl.load(idx_mapping_ptr + req_id)
359 num_computed_tokens = tl.load(num_computed_tokens_ptr + req_state_idx)
360
361 start = tl.load(query_start_loc_ptr + req_id)
362 end = tl.load(query_start_loc_ptr + req_id + 1)
363 query_len = end - start
364
365 seq_len = num_computed_tokens + query_len
366 tl.store(seq_lens_ptr + req_id, seq_len)
367
368 for i in tl.range(0, query_len, BLOCK_SIZE):
369 block = i + tl.arange(0, BLOCK_SIZE)
370 mask = block < query_len
371 pos = num_computed_tokens + block
372 tl.store(pos_ptr + start + block, pos, mask=mask)

396@triton.jit
397def _combine_sampled_and_draft_tokens_kernel(
398 input_ids_ptr,
399 idx_mapping_ptr,
400 last_sampled_tokens_ptr,
401 query_start_loc_ptr,
402 seq_lens_ptr,
403 prefill_len_ptr,
404 draft_tokens_ptr,
405 draft_tokens_stride,
406 cu_num_logits_ptr,
407 logits_indices_ptr,
408 BLOCK_SIZE: tl.constexpr,
409 NUM_NEW_SAMPLED_TOKENS: tl.constexpr = 1,
410):
411 batch_idx = tl.program_id(0)
412 req_state_idx = tl.load(idx_mapping_ptr + batch_idx)
413
414 # Get the number of logits and draft tokens.
415 cu_num_logits_start = tl.load(cu_num_logits_ptr + batch_idx)
416 cu_num_logits_end = tl.load(cu_num_logits_ptr + batch_idx + 1)
417 num_logits = cu_num_logits_end - cu_num_logits_start
418 num_draft_tokens = num_logits - NUM_NEW_SAMPLED_TOKENS
419
420 # Compute the logits indices.
421 block = tl.arange(0, BLOCK_SIZE)
422 query_end = tl.load(query_start_loc_ptr + batch_idx + 1)
423 logits_start = query_end - num_logits
424 tl.store(
425 logits_indices_ptr + cu_num_logits_start + block,
426 logits_start + block,
427 mask=block < num_logits,
428 )
429
430 seq_len = tl.load(seq_lens_ptr + batch_idx)
431 prefill_len = tl.load(prefill_len_ptr + req_state_idx)
432 if seq_len <= prefill_len:
433 # Handling prefill tokens. No sampled or draft tokens.
434 return
435
436 # Keep prompt-tail slots intact; only rewrite generated-token slots.
437 first_logit_seq_pos = seq_len - num_logits
438 if NUM_NEW_SAMPLED_TOKENS > 0 and first_logit_seq_pos >= prefill_len:
439 # Write the last sampled token ID to input_ids.
440 last_token_id = tl.load(last_sampled_tokens_ptr + req_state_idx)
441 tl.store(input_ids_ptr + logits_start, last_token_id)
442
443 # Write the draft tokens (if any) to input_ids.
444 if num_draft_tokens > 0:
445 mask = block < num_draft_tokens
446 draft_tokens = tl.load(
447 draft_tokens_ptr + req_state_idx * draft_tokens_stride + block,
448 mask=mask,
449 )
450 tl.store(
451 input_ids_ptr + query_end - num_draft_tokens + block,
452 draft_tokens,
453 mask=mask,
454 )

457def combine_sampled_and_draft_tokens(
458 input_ids: torch.Tensor,
459 idx_mapping: torch.Tensor,
460 last_sampled_tokens: torch.Tensor,
461 query_start_loc: torch.Tensor,
462 seq_lens: torch.Tensor,
463 prefill_len: torch.Tensor,
464 draft_tokens: torch.Tensor,
465 cu_num_logits: torch.Tensor,
466 num_logits: int,
467 num_new_sampled_tokens: int = 1, # excl accepted draft tokens, a.k.a bonus tokens
468) -> torch.Tensor:
469 assert num_new_sampled_tokens in (0, 1), (
470 f"num_new_sampled_tokens must be 0 or 1, got {num_new_sampled_tokens}"
471 )
472 # use idx_mapping.shape[0] for actual request count
473 num_reqs = idx_mapping.shape[0]
474 num_speculative_steps = draft_tokens.shape[-1]
475
476 logits_indices = torch.empty(
477 num_logits,
478 dtype=torch.int64,
479 device=input_ids.device,
480 )
481 _combine_sampled_and_draft_tokens_kernel[(num_reqs,)](
482 input_ids,
483 idx_mapping,
484 last_sampled_tokens,
485 query_start_loc,
486 seq_lens,
487 prefill_len,
488 draft_tokens,
489 draft_tokens.stride(0),
490 cu_num_logits,
491 logits_indices,
492 NUM_NEW_SAMPLED_TOKENS=num_new_sampled_tokens,
493 # NOTE(woosuk): Add num_new_sampled_tokens to ensure the block covers the
494 # last sampled token in addition to all draft tokens.
495 BLOCK_SIZE=triton.next_power_of_2(
496 num_speculative_steps + num_new_sampled_tokens
497 ),
498 )
499 return logits_indices

502@triton.jit
503def _get_num_sampled_and_rejected_kernel(
504 num_sampled_ptr,
505 num_rejected_ptr,
506 seq_lens_ptr,
507 cu_num_logits_ptr,
508 idx_mapping_ptr,
509 prefill_len_ptr,
510):
511 batch_idx = tl.program_id(0)
512 req_state_idx = tl.load(idx_mapping_ptr + batch_idx)
513
514 seq_len = tl.load(seq_lens_ptr + batch_idx)
515 prefill_len = tl.load(prefill_len_ptr + req_state_idx)
516 is_chunked_prefilling = seq_len < prefill_len
517
518 num_sampled = tl.load(num_sampled_ptr + batch_idx)
519 num_sampled = tl.where(is_chunked_prefilling, 0, num_sampled)
520 tl.store(num_sampled_ptr + batch_idx, num_sampled)
521
522 logits_start = tl.load(cu_num_logits_ptr + batch_idx)
523 logits_end = tl.load(cu_num_logits_ptr + batch_idx + 1)
524 num_logits = logits_end - logits_start
525
526 num_rejected = num_logits - num_sampled
527 num_rejected = tl.where(is_chunked_prefilling, 0, num_rejected)
528 tl.store(num_rejected_ptr + batch_idx, num_rejected)

551@triton.jit
552def _post_update_kernel(
553 idx_mapping_ptr,
554 num_computed_tokens_ptr,
555 last_sampled_tokens_ptr,
556 output_bin_counts_ptr,
557 output_bin_counts_stride,
558 sampled_tokens_ptr,
559 sampled_tokens_stride,
560 num_sampled_ptr,
561 num_rejected_ptr,
562 query_start_loc_ptr,
563 all_token_ids_ptr,
564 all_token_ids_stride,
565 total_len_ptr,
566):
567 req_id = tl.program_id(0)
568 req_state_idx = tl.load(idx_mapping_ptr + req_id)
569 if req_state_idx < 0:
570 # Filter rows with negative index entries.
571 return
572
573 total_len = tl.load(total_len_ptr + req_state_idx)
574 num_sampled = tl.load(num_sampled_ptr + req_id)
575 if num_sampled > 0:
576 token_id = tl.load(
577 sampled_tokens_ptr + req_id * sampled_tokens_stride + num_sampled - 1
578 )
579 tl.store(last_sampled_tokens_ptr + req_state_idx, token_id)
580 tl.store(total_len_ptr + req_state_idx, total_len + num_sampled)
581
582 for i in range(num_sampled):
583 token_id = tl.load(sampled_tokens_ptr + req_id * sampled_tokens_stride + i)
584 tl.store(
585 all_token_ids_ptr + req_state_idx * all_token_ids_stride + total_len + i,
586 token_id,
587 )
588
589 if output_bin_counts_ptr is not None:
590 token_ptr = (
591 output_bin_counts_ptr
592 + req_state_idx * output_bin_counts_stride
593 + token_id
594 )
595 count = tl.load(token_ptr)
596 tl.store(token_ptr, count + 1)
597
598 if query_start_loc_ptr is None:
599 query_len = 0
600 else:
601 query_start = tl.load(query_start_loc_ptr + req_id)
602 query_end = tl.load(query_start_loc_ptr + req_id + 1)
603 query_len = query_end - query_start
604 num_rejected = tl.load(num_rejected_ptr + req_id)
605
606 computed_delta = query_len - num_rejected
607 if computed_delta != 0:
608 num_computed = tl.load(num_computed_tokens_ptr + req_state_idx)
609 tl.store(num_computed_tokens_ptr + req_state_idx, num_computed + computed_delta)

685@triton.jit
686def _expand_idx_mapping_kernel(
687 idx_mapping_ptr,
688 expanded_idx_mapping_ptr,
689 expanded_local_pos_ptr,
690 cu_num_logits_ptr,
691 BLOCK_SIZE: tl.constexpr,
692):
693 req_idx = tl.program_id(0)
694 start_idx = tl.load(cu_num_logits_ptr + req_idx)
695 end_idx = tl.load(cu_num_logits_ptr + req_idx + 1)
696 num_tokens = end_idx - start_idx
697
698 block = tl.arange(0, BLOCK_SIZE)
699 mask = block < num_tokens
700 req_state_idx = tl.load(idx_mapping_ptr + req_idx)
701 tl.store(expanded_idx_mapping_ptr + start_idx + block, req_state_idx, mask=mask)
702 tl.store(expanded_local_pos_ptr + start_idx + block, block, mask=mask)

705def expand_idx_mapping(
706 idx_mapping: torch.Tensor,
707 total_num_logits: int,
708 cu_num_logits: torch.Tensor,
709 max_expand_len: int,
710) -> tuple[torch.Tensor, torch.Tensor]:
711 num_reqs = idx_mapping.shape[0]
712 expanded_idx_mapping = idx_mapping.new_empty(total_num_logits)
713 expanded_local_pos = torch.empty(
714 total_num_logits, dtype=torch.int32, device=idx_mapping.device
715 )
716 _expand_idx_mapping_kernel[(num_reqs,)](
717 idx_mapping,
718 expanded_idx_mapping,
719 expanded_local_pos,
720 cu_num_logits,
721 BLOCK_SIZE=triton.next_power_of_2(max_expand_len),
722 )
723 return expanded_idx_mapping, expanded_local_pos
