48class StepTimingCollector:
49 """Times the steps run inside ``collect``; record calls no-op outside it.
50
51 Every step gets its own events, so the steps queue back-to-back and the
52 block resolves them all behind one sync on the way out.
53 """
54
55 def __init__(self):
56 self._collecting = False
57 self._step: StepTimingEvents | None = None
58 self._batch = (False, 0, 0)
59 self._timed: list[tuple[StepTimingEvents, tuple[bool, int, int]]] = []
60
61 @contextlib.contextmanager
62 def collect(self) -> Iterator[list[StepTimingSample]]:
63 """Time every step run in this block.
64
65 The yielded list holds one sample per timed step once the block exits;
66 it stays empty inside the block, where the timings are still on device.
67 """
68 samples: list[StepTimingSample] = []
69 self._collecting = True
70 try:
71 yield samples
72 finally:
73 self._collecting = False
74 timed, self._timed, self._step = self._timed, [], None
75 if not timed:
76 return
77 # Same stream, issue order: once the last step is done, so are the rest.
78 timed[-1][0].drafter_end.synchronize()
79 samples.extend(
80 StepTimingSample(
81 events.forward_start.elapsed_time(events.forward_end),
82 events.drafter_start.elapsed_time(events.drafter_end),
83 num_target_tokens,
84 num_reqs,
85 full_cudagraph,
86 )
87 for events, (full_cudagraph, num_target_tokens, num_reqs) in timed
88 )
89
90 def record_batch(self, input_batch: "InputBatch", full_cudagraph: bool) -> None:
91 """Costs from different execution modes must not share a cost curve."""
92 self._batch = (full_cudagraph, input_batch.num_tokens, input_batch.num_reqs)
93
94 def forward_start(self) -> None:
95 if self._collecting:
96 self._step = StepTimingEvents()
97 self._step.forward_start.record()
98
99 def forward_end(self) -> None:
100 if self._step is not None:
101 self._step.forward_end.record()
102
103 def drafter_start(self) -> None:
104 if self._step is not None:
105 self._step.drafter_start.record()
106
107 def drafter_end(self) -> None:
108 """Ends the step: only steps that reach here have a draft cost."""
109 if self._step is not None:
110 self._step.drafter_end.record()
111 self._timed.append((self._step, self._batch))
112 self._step = None

115class AsyncOutput(AsyncModelRunnerOutput):
116 def __init__(
117 self,
118 model_runner_output: ModelRunnerOutput,
119 sampler_output: SamplerOutput,
120 num_sampled_tokens: torch.Tensor,
121 main_stream: torch.cuda.Stream,
122 copy_stream: torch.cuda.Stream,
123 check_ep_fault: bool = False,
124 routed_experts: RoutedExpertsTensors | None = None,
125 ):
126 # NOTE(woosuk): We must retain references to the GPU tensors,
127 # as the copy operations are performed on a different CUDA stream than
128 # the one where the tensors were created.
129 self.model_runner_output = model_runner_output
130 self.sampler_output = sampler_output
131 self.num_sampled_tokens = num_sampled_tokens
132 self.routed_experts = routed_experts
133 # Blocking (sleep) event to avoid busy-polling the CUDA driver lock.
134 self.copy_event = torch.cuda.Event(blocking=True)
135 self._has_fault: torch.Tensor | None = None
136
137 with stream(copy_stream, main_stream):
138 copy_stream.wait_stream(main_stream)
139
140 self.sampled_token_ids = async_copy_to_np(sampler_output.sampled_token_ids)
141 self.logprobs_tensors: LogprobsTensors | None = None
142 if sampler_output.logprobs_tensors is not None:
143 self.logprobs_tensors = (
144 sampler_output.logprobs_tensors.to_cpu_nonblocking()
145 )
146 self.num_nans: np.ndarray | None = None
147 if sampler_output.num_nans is not None:
148 self.num_nans = async_copy_to_np(sampler_output.num_nans)
149 self.num_sampled_tokens_np = async_copy_to_np(num_sampled_tokens)
150 self.sampling_mask_tensors: SamplingMaskTensors | None = None
151 if sampler_output.sampling_mask_tensors is not None:
152 self.sampling_mask_tensors = (
153 sampler_output.sampling_mask_tensors.to_cpu_nonblocking()
154 )
155 self.routed_experts_cpu: RoutedExpertsTensors | None = None
156 if routed_experts is not None:
157 self.routed_experts_cpu = routed_experts.to_cpu_nonblocking()
158 self.prompt_logprobs_dict = {
159 k: v.to_cpu_nonblocking() if v is not None else None
160 for k, v in self.model_runner_output.prompt_logprobs_dict.items()
161 }
162 if check_ep_fault:
163 has_fault = get_ep_all2all_manager().query_fault()
164 self._has_fault = has_fault.to("cpu", non_blocking=True)
165 self.copy_event.record(copy_stream)
166
167 def get_output(self) -> ModelRunnerOutput:
168 self.copy_event.synchronize()
169
170 # NOTE(woosuk): The following code is to ensure compatibility with
171 # the existing model runner.
172 # Going forward, we should keep the data structures as NumPy arrays
173 # rather than Python lists.
174 sampled_token_ids: list[list[int]] = self.sampled_token_ids.tolist()
175 num_sampled_tokens: list[int] = self.num_sampled_tokens_np.tolist()
176 for token_ids, num_tokens in zip(sampled_token_ids, num_sampled_tokens):
177 del token_ids[num_tokens:]
178 self.model_runner_output.sampled_token_ids = sampled_token_ids
179
180 if self.sampling_mask_tensors is not None:
181 self.model_runner_output.sampling_masks = (
182 self.sampling_mask_tensors.tolists()
183 )
184
185 if self.num_nans is not None:
186 self.model_runner_output.num_nans_in_logits = dict(
187 zip(self.model_runner_output.req_ids, self.num_nans.tolist())
188 )
189 if envs.VLLM_RAISE_ON_LOGIT_NANS:
190 raise_if_nan_logits(self.model_runner_output.num_nans_in_logits)
191
192 if self.logprobs_tensors is not None:
193 self.model_runner_output.logprobs = self.logprobs_tensors.tolists()
194 self.model_runner_output.prompt_logprobs_dict = self.prompt_logprobs_dict
195 if self.routed_experts_cpu is not None:
196 self.model_runner_output.routed_experts = self.routed_experts_cpu.tolists()
197
198 if self._has_fault is not None and self._has_fault.item():
199 mask = get_ep_all2all_manager().query_active_mask()
200 raise RuntimeError(
201 "Fault detected in EP all2all communication: "
202 "one or more ranks timed out during dispatch/combine. "
203 f"Mask: {mask.cpu().tolist()}"
204 )
205
206 return self.model_runner_output

209class AsyncPoolingOutput(AsyncModelRunnerOutput):
210 def __init__(
211 self,
212 model_runner_output: ModelRunnerOutput,
213 pooler_output: PoolerOutput,
214 finished_mask: list[bool],
215 main_stream: torch.cuda.Stream,
216 copy_stream: torch.cuda.Stream,
217 ):
218 self.model_runner_output = model_runner_output
219 self.pooler_output = pooler_output
220 # Blocking (sleep) event to avoid busy-polling the CUDA driver lock.
221 self.copy_event = torch.cuda.Event(blocking=True)
222
223 with stream(copy_stream, main_stream):
224 copy_stream.wait_stream(main_stream)
225 if isinstance(self.pooler_output, torch.Tensor) and all(finished_mask):
226 self.pooler_output_cpu: PoolerOutput = self.pooler_output.to(
227 "cpu", non_blocking=True
228 )
229 else:
230 outputs = (
231 self.pooler_output.unbind()
232 if isinstance(self.pooler_output, torch.Tensor)
233 else self.pooler_output
234 )
235 self.pooler_output_cpu = [
236 None
237 if output is None or not is_finished
238 else output.to("cpu", non_blocking=True)
239 for output, is_finished in zip(outputs, finished_mask, strict=True)
240 ]
241 self.copy_event.record(copy_stream)
242
243 def get_output(self) -> ModelRunnerOutput:
244 if isinstance(self.pooler_output_cpu, torch.Tensor):
245 pooler_output = list(self.pooler_output_cpu.unbind(dim=0))
246 else:
247 pooler_output = self.pooler_output_cpu
248 self.copy_event.synchronize()
249 self.model_runner_output.pooler_output = pooler_output
250 return self.model_runner_output

257@contextlib.contextmanager
258def stream(to_stream: torch.cuda.Stream, from_stream: torch.cuda.Stream):
259 """Lightweight version of torch.cuda.stream() context manager which
260 avoids current_stream and device lookups.
261 """
262 try:
263 torch.cuda.set_stream(to_stream)
264 yield
265 finally:
266 torch.cuda.set_stream(from_stream)
