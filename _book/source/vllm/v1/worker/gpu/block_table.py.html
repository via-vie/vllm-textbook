17class BlockTables:
18 def __init__(
19 self,
20 block_sizes: list[int],
21 max_num_reqs: int,
22 max_num_batched_tokens: int,
23 max_num_blocks_per_group: list[int],
24 device: torch.device,
25 kernel_block_sizes: list[int],
26 cp_size: int = 1,
27 cp_rank: int = 0,
28 cp_interleave: int = 1,
29 slot_mapping_enabled: list[bool] | None = None,
30 ):
31 self.block_sizes = block_sizes
32 self.kernel_block_sizes = kernel_block_sizes
33 self.max_num_reqs = max_num_reqs
34 self.max_num_batched_tokens = max_num_batched_tokens
35 self.device = device
36
37 self.cp_size = cp_size
38 self.cp_rank = cp_rank
39 self.cp_interleave = cp_interleave
40
41 self.num_kv_cache_groups = len(self.block_sizes)
42 assert len(max_num_blocks_per_group) == self.num_kv_cache_groups
43 if slot_mapping_enabled is None:
44 slot_mapping_enabled = [True] * self.num_kv_cache_groups
45 assert len(slot_mapping_enabled) == self.num_kv_cache_groups
46 self._slot_mapping_enabled = slot_mapping_enabled
47
48 self.blocks_per_kv_block = [
49 bs // kbs for bs, kbs in zip(block_sizes, kernel_block_sizes)
50 ]
51
52 # num_kv_cache_groups x [max_num_reqs, max_num_blocks]
53 self.block_tables: list[StagedWriteTensor] = []
54 for i in range(self.num_kv_cache_groups):
55 max_num_blocks = max_num_blocks_per_group[i] * self.blocks_per_kv_block[i]
56 block_table = StagedWriteTensor(
57 (self.max_num_reqs, max_num_blocks), dtype=torch.int32, device=device
58 )
59 self.block_tables.append(block_table)
60
61 self.num_blocks = UvaBackedTensor(
62 (self.num_kv_cache_groups, self.max_num_reqs),
63 dtype=torch.int32,
64 )
65 self.fused_writer: FusedStagedWriter | None = None
66 if self.num_kv_cache_groups > 1:
67 # Only the multi-group path uses the fused writer.
68 self.fused_writer = FusedStagedWriter(
69 self.device, self.num_kv_cache_groups * self.max_num_reqs
70 )
71
72 # Block tables used for model's forward pass.
73 # num_kv_cache_groups x [max_num_reqs, max_num_blocks]
74 self.input_block_tables: list[torch.Tensor] = [
75 torch.zeros_like(b.gpu) for b in self.block_tables
76 ]
77
78 self.slot_mappings = torch.zeros(
79 self.num_kv_cache_groups,
80 self.max_num_batched_tokens,
81 dtype=torch.int64,
82 device=self.device,
83 )
84
85 self.init_block_table_layout_tensors()
86
87 def _make_ptr_tensor(self, x: Iterable[torch.Tensor]) -> torch.Tensor:
88 # NOTE(woosuk): Use uint64 instead of int64 to cover all possible addresses.
89 return torch.tensor(
90 [t.data_ptr() for t in x], dtype=torch.uint64, device=self.device
91 )
92
93 def init_block_table_layout_tensors(self) -> None:
94 self.block_table_ptrs = self._make_ptr_tensor(
95 [b.gpu for b in self.block_tables]
96 )
97 self.block_table_strides = torch.tensor(
98 [b.gpu.stride(0) for b in self.block_tables],
99 dtype=torch.int64,
100 device=self.device,
101 )
102 self.block_sizes_tensor = torch.tensor(
103 self.block_sizes, dtype=torch.int32, device=self.device
104 )
105 self.kernel_block_sizes_tensor = torch.tensor(
106 self.kernel_block_sizes, dtype=torch.int32, device=self.device
107 )
108 self.slot_mapping_enabled = torch.tensor(
109 self._slot_mapping_enabled, dtype=torch.bool, device=self.device
110 )
111 self.input_block_table_ptrs = self._make_ptr_tensor(self.input_block_tables)
112
113 def append_block_ids(
114 self,
115 req_index: int,
116 new_block_ids: tuple[list[int], ...],
117 overwrite: bool,
118 ) -> None:
119 for i in range(self.num_kv_cache_groups):
120 start = self.num_blocks.np[i, req_index] if not overwrite else 0
121 block_ids = new_block_ids[i]
122 bpk = self.blocks_per_kv_block[i]
123 if bpk > 1:
124 block_ids = [b * bpk + k for b in block_ids for k in range(bpk)]
125 end = start + len(block_ids)
126 row_capacity = self.block_tables[i].gpu.shape[1]
127 if end > row_capacity:
128 raise RuntimeError(
129 f"Block table write for request {req_index}, group {i} exceeds "
130 f"row capacity ({end} > {row_capacity})"
131 )
132 self.block_tables[i].stage_write(req_index, start, block_ids)
133 self.num_blocks.np[i, req_index] = end
134
135 def apply_staged_writes(self) -> None:
136 if self.num_kv_cache_groups == 0:
137 return
138 if self.num_kv_cache_groups == 1:
139 # Single group: write directly, skipping the per-write group lookup.
140 self.block_tables[0].apply_write()
141 elif self.num_kv_cache_groups > 1:
142 # Multiple groups: apply all block tables with one fused kernel.
143 assert self.fused_writer is not None
144 self.fused_writer.apply(
145 self.block_tables, self.block_table_ptrs, self.block_table_strides
146 )
147 self.num_blocks.copy_to_uva()
148
149 def gather_block_tables(
150 self,
151 idx_mapping: torch.Tensor,
152 num_reqs_padded: int,
153 out: tuple[torch.Tensor, ...] | None = None,
154 out_ptrs: torch.Tensor | None = None,
155 ) -> tuple[torch.Tensor, ...]:
156 if self.num_kv_cache_groups == 0:
157 return ()
158 if out is None:
159 out = tuple(self.input_block_tables)
160 out_ptrs = self.input_block_table_ptrs
161 else:
162 assert out_ptrs is not None
163 assert len(out) == self.num_kv_cache_groups
164 num_reqs = idx_mapping.shape[0]
165 # Launch kernel with num_reqs_padded to fuse zeroing of padded rows.
166 _gather_block_tables_kernel[(self.num_kv_cache_groups, num_reqs_padded)](
167 idx_mapping,
168 self.block_table_ptrs,
169 out_ptrs,
170 self.block_table_strides,
171 self.num_blocks.gpu,
172 self.num_blocks.gpu.stride(0),
173 num_reqs,
174 BLOCK_SIZE=1024, # type: ignore
175 )
176 return tuple(bt[:num_reqs_padded] for bt in out)
177
178 def get_dummy_block_tables(self, num_reqs: int) -> tuple[torch.Tensor, ...]:
179 # NOTE(woosuk): The output may be used for CUDA graph capture.
180 # Therefore, this method must return the persistent tensor
181 # with the same memory address as that used during the model's forward pass,
182 # rather than allocating a new tensor.
183 #
184 # Zero the rows so dummy runs write mamba state to the reserved null
185 # block rather than through the previous real step's (stale) block
186 # ids, which may point at blocks since freed and reallocated.
187 return tuple(
188 block_table[:num_reqs].zero_() for block_table in self.input_block_tables
189 )
190
191 def compute_slot_mappings(
192 self,
193 idx_mapping: torch.Tensor,
194 query_start_loc: torch.Tensor,
195 positions: torch.Tensor,
196 num_tokens_padded: int,
197 out: torch.Tensor | None = None,
198 ) -> torch.Tensor:
199 if self.num_kv_cache_groups == 0:
200 return (self.slot_mappings if out is None else out)[:, :num_tokens_padded]
201 num_reqs = idx_mapping.shape[0]
202 num_groups = self.num_kv_cache_groups
203 slot_mappings = self.slot_mappings if out is None else out
204 _compute_slot_mappings_kernel[(num_groups, num_reqs + 1)](
205 slot_mappings.shape[1],
206 idx_mapping,
207 query_start_loc,
208 positions,
209 self.block_table_ptrs,
210 self.block_table_strides,
211 self.block_sizes_tensor,
212 self.kernel_block_sizes_tensor,
213 self.slot_mapping_enabled,
214 slot_mappings,
215 slot_mappings.stride(0),
216 self.cp_rank,
217 CP_SIZE=self.cp_size,
218 CP_INTERLEAVE=self.cp_interleave,
219 PAD_ID=PAD_SLOT_ID,
220 TRITON_BLOCK_SIZE=1024, # type: ignore
221 )
222 return slot_mappings[:, :num_tokens_padded]
223
224 def get_dummy_slot_mappings(self, num_tokens: int) -> torch.Tensor:
225 # Fill the entire slot_mappings tensor, not just the first `num_tokens` entries.
226 # This is because the padding logic is complex and kernels may access beyond
227 # the requested range.
228 self.slot_mappings.fill_(PAD_SLOT_ID)
229 # NOTE(woosuk): The output may be used for CUDA graph capture.
230 # Therefore, this method must return the persistent tensor
231 # with the same memory address as that used during the model's forward pass,
232 # rather than allocating a new tensor.
233 return self.slot_mappings[:, :num_tokens]

236@triton.jit(do_not_specialize=["num_reqs"])
237def _gather_block_tables_kernel(
238 batch_idx_to_req_idx, # [batch_size]
239 src_block_table_ptrs, # [num_kv_cache_groups]
240 dst_block_table_ptrs, # [num_kv_cache_groups]
241 block_table_strides, # [num_kv_cache_groups]
242 num_blocks_ptr, # [num_kv_cache_groups, max_num_reqs]
243 num_blocks_stride,
244 num_reqs, # actual number of requests (for padding)
245 BLOCK_SIZE: tl.constexpr,
246):
247 # kv cache group id
248 group_id = tl.program_id(0)
249 batch_idx = tl.program_id(1)
250
251 stride = tl.load(block_table_strides + group_id)
252 max_num_blocks = stride # stride equals max_num_blocks for this group.
253 dst_block_table_ptr = _load_ptr(dst_block_table_ptrs + group_id, tl.int32)
254 dst_row_ptr = dst_block_table_ptr + batch_idx * stride
255
256 if batch_idx >= num_reqs:
257 # Zero out padded rows.
258 for i in tl.range(0, max_num_blocks, BLOCK_SIZE):
259 offset = i + tl.arange(0, BLOCK_SIZE)
260 tl.store(dst_row_ptr + offset, 0, mask=offset < max_num_blocks)
261 return
262
263 req_idx = tl.load(batch_idx_to_req_idx + batch_idx)
264 group_num_blocks_ptr = num_blocks_ptr + group_id * num_blocks_stride
265 num_blocks = tl.load(group_num_blocks_ptr + req_idx)
266
267 src_block_table_ptr = _load_ptr(src_block_table_ptrs + group_id, tl.int32)
268 src_row_ptr = src_block_table_ptr + req_idx * stride
269
270 for i in tl.range(0, num_blocks, BLOCK_SIZE):
271 offset = i + tl.arange(0, BLOCK_SIZE)
272 block_ids = tl.load(src_row_ptr + offset, mask=offset < num_blocks)
273 tl.store(dst_row_ptr + offset, block_ids, mask=offset < num_blocks)

276@triton.jit
277def _compute_slot_mappings_kernel(
278 max_num_tokens,
279 idx_mapping, # [num_reqs]
280 query_start_loc, # [num_reqs + 1]
281 pos, # [num_tokens]
282 block_table_ptrs, # [num_kv_cache_groups]
283 block_table_strides, # [num_kv_cache_groups]
284 block_sizes, # [num_kv_cache_groups]
285 kernel_block_sizes, # [num_kv_cache_groups]
286 slot_mapping_enabled, # [num_kv_cache_groups]
287 slot_mappings_ptr, # [num_kv_cache_groups, max_num_tokens]
288 slot_mappings_stride,
289 cp_rank,
290 CP_SIZE: tl.constexpr,
291 CP_INTERLEAVE: tl.constexpr,
292 PAD_ID: tl.constexpr,
293 TRITON_BLOCK_SIZE: tl.constexpr,
294):
295 # kv cache group id
296 group_id = tl.program_id(0)
297 batch_idx = tl.program_id(1)
298 slot_mapping_ptr = slot_mappings_ptr + group_id * slot_mappings_stride
299
300 if batch_idx == tl.num_programs(1) - 1:
301 # Pad remaining slots to -1. This is needed for CUDA graphs.
302 # Start from actual token count (not padded) to cover the gap
303 # between actual tokens and padded tokens that can contain stale
304 # valid slot IDs from previous chunks during chunked prefill.
305 actual_num_tokens = tl.load(query_start_loc + batch_idx)
306 for i in range(actual_num_tokens, max_num_tokens, TRITON_BLOCK_SIZE):
307 offset = i + tl.arange(0, TRITON_BLOCK_SIZE)
308 tl.store(slot_mapping_ptr + offset, PAD_ID, mask=offset < max_num_tokens)
309 return
310
311 block_table_ptr = _load_ptr(block_table_ptrs + group_id, tl.int32)
312 block_table_stride = tl.load(block_table_strides + group_id)
313 kv_block_size = tl.load(block_sizes + group_id)
314 kernel_block_size = tl.load(kernel_block_sizes + group_id)
315 mapping_enabled = tl.load(slot_mapping_enabled + group_id)
316
317 req_state_idx = tl.load(idx_mapping + batch_idx)
318 start_idx = tl.load(query_start_loc + batch_idx)
319 end_idx = tl.load(query_start_loc + batch_idx + 1)
320 for i in range(start_idx, end_idx, TRITON_BLOCK_SIZE):
321 offset = i + tl.arange(0, TRITON_BLOCK_SIZE)
322 positions = tl.load(pos + offset, mask=offset < end_idx, other=0)
323
324 if CP_SIZE == 1:
325 # Common case: Context parallelism is not used.
326 local_positions = positions
327 is_local = True
328 else:
329 # Context parallelism is used.
330 virtual_block_size = kv_block_size * CP_SIZE
331 virtual_block_indices = positions // virtual_block_size
332 virtual_block_offsets = positions % virtual_block_size
333 is_local = virtual_block_offsets // CP_INTERLEAVE % CP_SIZE == cp_rank
334 rounds = virtual_block_offsets // (CP_INTERLEAVE * CP_SIZE)
335 remainder = virtual_block_offsets % CP_INTERLEAVE
336 local_offsets = rounds * CP_INTERLEAVE + remainder
337 local_positions = virtual_block_indices * kv_block_size + local_offsets
338
339 block_indices = tl.where(
340 mapping_enabled, local_positions // kernel_block_size, 0
341 )
342 block_offsets = local_positions % kernel_block_size
343 block_numbers = tl.load(
344 block_table_ptr + req_state_idx * block_table_stride + block_indices,
345 mask=is_local,
346 other=0,
347 )
348 slot_ids = block_numbers * kernel_block_size + block_offsets
349 if CP_SIZE != 1:
350 slot_ids = tl.where(is_local, slot_ids, PAD_ID)
351
352 slot_ids = tl.where(mapping_enabled, slot_ids, PAD_ID)
353 tl.store(slot_mapping_ptr + offset, slot_ids, mask=offset < end_idx)
