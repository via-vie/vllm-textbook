62@dataclass(frozen=True)
63class BatchExecutionDescriptor:
64 """Describes the shape of the batch and CG mode to run; this is used to make shape
65 matches between the capture and runtime."""
66
67 cg_mode: CUDAGraphMode
68 num_tokens: int
69 num_reqs: int | None # None means no request padding is needed (PIECEWISE graphs)
70 uniform_token_count: int | None = None
71 # Upper bound on per-request query length. Varlen decode graphs leave
72 # uniform_token_count unset, so this is what keeps a prefill batch out of one.
73 max_query_len: int | None = None
74 num_active_loras: int = 0
75 # Number of microbatches the batch is split into (DBO). 1 means no splitting.
76 num_ubatches: int = 1

102def _is_compatible(
103 desc: BatchExecutionDescriptor,
104 num_reqs: int,
105 num_tokens: int,
106 uniform_token_count: int | None,
107 num_active_loras: int,
108 max_query_len: int | None,
109 num_ubatches: int,
110) -> bool:
111 # desc.uniform_token_count=None (PIECEWISE) can handle any uniform_token_count
112 # desc.num_reqs=None means no request padding needed (PIECEWISE)
113 # desc.max_query_len=None means the graph does not constrain query length; a
114 # caller that does not track max_query_len must not match one that does
115 # A graph captured for N microbatches can only serve a batch split N ways.
116 return (
117 (
118 desc.uniform_token_count is None
119 or desc.uniform_token_count == uniform_token_count
120 )
121 and (
122 desc.max_query_len is None
123 or (max_query_len is not None and desc.max_query_len >= max_query_len)
124 )
125 and (desc.num_reqs is None or desc.num_reqs >= num_reqs)
126 and desc.num_tokens >= num_tokens
127 and desc.num_active_loras == num_active_loras
128 and desc.num_ubatches == num_ubatches
129 )

195 def _build_lora_dispatch_map(self) -> tuple[dict[int, int], int]:
196 """Precompute actual num_active_loras -> effective captured case.
197
198 Mirrors the num_tokens candidate expansion in ``_init_candidates``:
199 every possible active-LoRA count is mapped ahead of time to the
200 smallest captured case that can serve it, so ``dispatch`` is a plain
201 dict lookup instead of a per-call bisect.
202 """
203 captured_with_lora = sorted(c for c in self.lora_capture_cases if c > 0)
204 if not captured_with_lora:
205 return {}, 0
206 dispatch_map: dict[int, int] = {}
207 case_idx = 0
208 for n in range(1, captured_with_lora[-1] + 1):
209 while captured_with_lora[case_idx] < n:
210 case_idx += 1
211 dispatch_map[n] = captured_with_lora[case_idx]
212 return dispatch_map, captured_with_lora[-1]

214 def _resolve_effective_loras(self, num_active_loras: int) -> int:
215 """Map an actual active-LoRA count to its captured graph case."""
216 if num_active_loras <= 0 or not self._lora_dispatch_map:
217 return num_active_loras
218 # Counts above the largest captured case clamp to it.
219 return self._lora_dispatch_map.get(num_active_loras, self._max_lora_case)

221 def _maybe_ubatch_twin(
222 self, desc: BatchExecutionDescriptor
223 ) -> BatchExecutionDescriptor | None:
224 """Return a microbatched capture candidate when eligible.
225
226 Uniform query lengths preserve the captured request split. Use the DP
227 dispatch thresholds so all ranks generate the same candidates.
228 """
229 if self.ubatch_runner is None or desc.cg_mode != CUDAGraphMode.FULL:
230 return None
231 if desc.num_reqs is None:
232 return None
233 uniform_token_count, remainder = divmod(desc.num_tokens, desc.num_reqs)
234 if remainder or desc.uniform_token_count not in (None, uniform_token_count):
235 return None
236 parallel_config = self.vllm_config.parallel_config
237 num_ubatches = get_num_ubatches(parallel_config)
238 if desc.num_tokens < num_ubatches:
239 return None
240 if not check_ubatch_thresholds(
241 parallel_config, desc.num_tokens, uniform_decode=True
242 ):
243 return None
244 return replace(
245 desc, num_ubatches=num_ubatches, uniform_token_count=uniform_token_count
246 )

248 def _init_candidates(self) -> None:
249 """Build priority-ordered candidate lists for each token count."""
250 capture_sizes = self.compilation_config.cudagraph_capture_sizes
251 if not (self.cudagraph_mode and capture_sizes):
252 return
253
254 capture_sizes = sorted(capture_sizes)
255 max_decode_tokens = self.max_num_reqs * self.decode_query_len
256 decode_mode = self.cudagraph_mode.decode_mode()
257 mixed_mode = self.cudagraph_mode.mixed_mode()
258 separate_decode_routine = self.cudagraph_mode.separate_routine()
259 max_cg_capture_size = self.compilation_config.max_cudagraph_capture_size
260
261 descs_by_mode: defaultdict[CUDAGraphMode, list[BatchExecutionDescriptor]] = (
262 defaultdict(list)
263 )
264
265 # When using Dynamic SD, num_speculative_tokens is the max number of
266 # draft tokens. The scheduler might use a smaller number so we need
267 # to capture graphs for all possible values during decode.
268 speculative_config = self.vllm_config.speculative_config
269 if (
270 speculative_config
271 and speculative_config.uses_dynamic_speculative_decoding()
272 ):
273 # decode_query_len = num_speculative_steps + num_new_sampled_tokens
274 # _per_step. Recover num_new_sampled_tokens_per_step
275 # from the values the manager already has.
276 num_new_sampled_tokens_per_step = (
277 self.decode_query_len - self.vllm_config.num_speculative_tokens
278 )
279 dense_schedule = build_dynamic_sd_schedule_lookup(
280 speculative_config.num_speculative_tokens_per_batch_size,
281 vllm_max_batch_size=self.max_num_reqs,
282 vllm_num_speculative_tokens=self.vllm_config.num_speculative_tokens,
283 )
284 decode_query_lens = sorted(
285 {
286 num_spec + num_new_sampled_tokens_per_step
287 for num_spec in dense_schedule[1:]
288 }
289 )
290 else:
291 decode_query_lens = [self.decode_query_len]
292
293 capture_varlen_decode = (
294 separate_decode_routine and bool(decode_mode) and self.varlen_decode
295 )
296 for num_tokens, num_active_loras in product(
297 capture_sizes, self.lora_capture_cases
298 ):
299 # Varlen decode graphs take any mix of 1..decode_query_len tokens per
300 # request, worst case 1 token per request (or max_num_reqs)
301 if capture_varlen_decode and num_tokens <= max_decode_tokens:
302 desc = BatchExecutionDescriptor(
303 cg_mode=decode_mode,
304 num_tokens=num_tokens,
305 num_reqs=min(num_tokens, self.max_num_reqs),
306 max_query_len=self.decode_query_len,
307 num_active_loras=num_active_loras,
308 )
309 descs_by_mode[decode_mode].append(desc)
310 # Capture uniform decode specfifc graphs if required
311 # (i.e. separate decode routine)
312 elif separate_decode_routine and decode_mode and not self.varlen_decode:
313 for decode_query_len in decode_query_lens:
314 rounded_num_tokens = round_up(num_tokens, decode_query_len)
315 rounded_num_reqs = rounded_num_tokens // decode_query_len
316
317 if (
318 rounded_num_tokens > max_decode_tokens
319 or rounded_num_tokens > max_cg_capture_size
320 or rounded_num_reqs > self.max_num_reqs
321 ):
322 continue
323
324 desc = BatchExecutionDescriptor(
325 cg_mode=decode_mode,
326 num_tokens=rounded_num_tokens,
327 num_reqs=rounded_num_reqs,
328 uniform_token_count=decode_query_len,
329 num_active_loras=num_active_loras,
330 )
331
332 # avoid duplicate graphs
333 if desc not in descs_by_mode[decode_mode]:
334 descs_by_mode[decode_mode].append(desc)
335
336 ubatch_desc = self._maybe_ubatch_twin(desc)
337 if ubatch_desc is not None and (
338 ubatch_desc not in descs_by_mode[decode_mode]
339 ):
340 descs_by_mode[decode_mode].append(ubatch_desc)
341
342 # recoverSSM cannot capture a dummy query wider than its workspace.
343 if mixed_mode and (
344 not self.vllm_config.cache_config.use_kda_recoverssm
345 or num_tokens <= max_decode_tokens
346 ):
347 # for PIECEWISE graphs there is no limit on requests when replaying
348 # i.e. no request padding is needed, so we leave it as None.
349 # For breakable PW graphs, break-point kernels read the real batch
350 # from the forward context; in-graph kernels handle the token padding
351 # themselves from the padded slot_mapping (rows with slot == -1).
352 num_reqs = None
353 if mixed_mode == CUDAGraphMode.FULL:
354 num_reqs = min(num_tokens, self.max_num_reqs)
355 desc = BatchExecutionDescriptor(
356 cg_mode=mixed_mode,
357 num_tokens=num_tokens,
358 num_reqs=num_reqs,
359 num_active_loras=num_active_loras,
360 )
361 descs_by_mode[mixed_mode].append(desc)
362
363 ubatch_desc = self._maybe_ubatch_twin(desc)
364 if ubatch_desc is not None:
365 descs_by_mode[mixed_mode].append(ubatch_desc)
366
367 for mode, descs in descs_by_mode.items():
368 descs.sort(key=lambda d: d.num_tokens, reverse=True)
369 self._capture_descs[mode] = descs
370
371 for mode in (CUDAGraphMode.FULL, CUDAGraphMode.PIECEWISE):
372 mode_descs = tuple(reversed(descs_by_mode.get(mode, [])))
373 for num_active_loras in self.lora_capture_cases:
374 lora_descs = [
375 d for d in mode_descs if d.num_active_loras == num_active_loras
376 ]
377 current_range_start = 0
378 # Dynamic speculative decoding can produce multiple graphs with the same
379 # num_tokens. Group them so each graph covers the same candidate range.
380 for num_tokens, group in groupby(lora_descs, lambda d: d.num_tokens):
381 matching = list(group)
382 for i in range(current_range_start, num_tokens + 1):
383 key = (i, num_active_loras)
384 self._candidates.setdefault(key, []).extend(matching)
385 current_range_start = num_tokens + 1
386
387 def needs_capture(self) -> bool:
388 return len(self._capture_descs) > 0

390 def _capture_stream(self, desc: BatchExecutionDescriptor) -> torch.cuda.Stream:
391 """Capture on the stream used by the microbatch threads."""
392 if desc.num_ubatches > 1:
393 assert self.ubatch_runner is not None
394 return self.ubatch_runner.capture_stream
395 return current_stream()

397 @torch.inference_mode()
398 def capture(
399 self,
400 create_forward_fn: CreateForwardFn,
401 progress_bar_desc: str = "Capturing CUDA graphs",
402 ) -> None:
403 """Capture CUDA graphs.
404
405 Args:
406 create_forward_fn: Factory that prepares inputs (OUTSIDE graph) and
407 returns a forward_fn. For FULL and breakable PIECEWISE modes,
408 it is invoked once with warmup=True and again with warmup=False
409 because attention backends may mutate or lazily initialize
410 metadata during warmup.
411 """
412 with graph_capture(device=self.device), ExitStack() as stack:
413 if self.ubatch_runner is not None:
414 # Join parked threads on failure to avoid blocking later captures.
415 stack.callback(self.ubatch_runner.abort_pending_run)
416 # Capture in order: PIECEWISE first, then FULL. PIECEWISE has larger
417 # activations so FULL activations should fit in already allocated
418 # buffers in the graph pool.
419 for mode in [CUDAGraphMode.PIECEWISE, CUDAGraphMode.FULL]:
420 if mode not in self._capture_descs:
421 continue
422
423 descs = self._capture_descs[mode]
424 if (
425 mode == CUDAGraphMode.FULL
426 and self._max_full_descs_to_capture is not None
427 ):
428 # Profiling only: capture a sample of the largest FULL
429 # graphs; the total cost is extrapolated from their
430 # per-graph memory deltas.
431 descs = descs[: self._max_full_descs_to_capture]
432 if is_global_first_rank():
433 descs = tqdm(descs, desc=f"{progress_bar_desc} ({mode.name})")
434 for desc in descs:
435 # Prepare inputs and get forward function
436 forward_fn = create_forward_fn(desc, warmup=True)
437
438 # Warmup
439 forward_fn(CUDAGraphMode.NONE)
440
441 # Capture
442 logger.debug(
443 "CG Capture: mode=%s, batch_desc=%s",
444 desc.cg_mode.name,
445 desc,
446 )
447 if (
448 desc.cg_mode == CUDAGraphMode.PIECEWISE
449 and not self.use_breakable_cg
450 ):
451 forward_fn(CUDAGraphMode.PIECEWISE)
452 else:
453 # Capture with fresh attention state.
454 forward_fn = create_forward_fn(desc, warmup=False)
455 if desc.cg_mode == CUDAGraphMode.PIECEWISE:
456 forward_fn(CUDAGraphMode.PIECEWISE)
457 continue
458 assert desc not in self.graphs, (
459 f"Graph already captured for {desc}"
460 )
461 graph = torch.cuda.CUDAGraph()
462 # Sync offloader's copy stream before capture.
463 # Ensure any pre-capture prefetches from offloader are complete.
464 get_offloader().sync_prev_onload()
465 if self.pool is not None:
466 set_graph_pool_id(self.pool)
467 else:
468 set_graph_pool_id(current_platform.graph_pool_handle())
469 if self._capture_mem_samples is not None:
470 torch.accelerator.synchronize()
471 free_before = torch.accelerator.get_memory_info()[0]
472 with torch.cuda.graph(
473 graph, self.pool, stream=self._capture_stream(desc)
474 ):
475 forward_fn(CUDAGraphMode.NONE)
476 # Join offloader's copy stream after forward to avoid
477 # unjoined stream error. The last layer's start_prefetch
478 # forks copy_stream, but wait_prefetch only happens in
479 # the next forward pass.
480 get_offloader().join_after_forward()
481 if self._capture_mem_samples is not None:
482 torch.accelerator.synchronize()
483 free_after = torch.accelerator.get_memory_info()[0]
484 self._capture_mem_samples.append(free_before - free_after)
485 self.graphs[desc] = graph
486 compilation_counter.num_cudagraph_captured += 1
487
488 self._graphs_captured = True
489
490 def captured_token_counts(self) -> list[int]:
491 """Sorted token counts with a captured graph, ignoring LoRA variants."""
492 return sorted(
493 {desc.num_tokens for desc in self.graphs if desc.num_active_loras == 0}
494 )

496 def dispatch(
497 self,
498 num_reqs: int,
499 num_tokens: int,
500 uniform_token_count: int | None,
501 num_active_loras: int,
502 max_query_len: int | None = None,
503 num_ubatches: int = 1,
504 ) -> BatchExecutionDescriptor:
505 """Find matching cudagraph descriptor from priority-ordered candidates."""
506
507 effective_loras = self._resolve_effective_loras(num_active_loras)
508 key = (num_tokens, effective_loras)
509 if self._graphs_captured and num_tokens > 0 and key in self._candidates:
510 for desc in self._candidates[key]:
511 if _is_compatible(
512 desc,
513 num_reqs,
514 num_tokens,
515 uniform_token_count,
516 effective_loras,
517 max_query_len,
518 num_ubatches,
519 ):
520 return desc
521 return BatchExecutionDescriptor(
522 cg_mode=CUDAGraphMode.NONE,
523 num_tokens=num_tokens,
524 num_reqs=num_reqs,
525 num_active_loras=effective_loras,
526 num_ubatches=num_ubatches,
527 )

529 def run_fullgraph(self, desc: BatchExecutionDescriptor):
530 """Replay a captured FULL cudagraph."""
531 assert desc.cg_mode == CUDAGraphMode.FULL, (
532 f"Expected FULL mode, got {desc.cg_mode}"
533 )
534 assert desc in self.graphs, f"No cudagraph for {desc}"
535 # Sync offloader before replay - needed when transitioning from
536 # eager/piecewise to full cudagraph (e.g., prefill → decode).
537 # The previous eager iteration's start_prefetch may have queued
538 # H2D copies on copy_stream that the graph's captured events
539 # cannot see. Without this, replay could overwrite static buffers
540 # while those copies are still in flight.
541 get_offloader().sync_prev_onload()
542 self.graphs[desc].replay()

603 self.init_breakable_cg_runner(model)
604
605 if self.cudagraph_mode.has_piecewise_cudagraphs() and not (
606 self.use_breakable_cg or has_compiled_submodule(model)
607 ):
608 raise RuntimeError(
609 f"{type(model).__name__}: piecewise CUDA graphs "
610 f"(cudagraph_mode={self.cudagraph_mode.name}) unavailable, "
611 "model is not torch-compiled and breakable CUDA graph is off. "
612 "Set VLLM_USE_BREAKABLE_CUDAGRAPH=1 or cudagraph_mode=NONE/FULL."
613 )
614
615 def store_capture_output(num_tokens: int, model_output: Any) -> None:
616 """Copy outputs to persistent buffers, allocating on first use."""
617 if self.is_last_pp_rank:
618 # Last PP rank (common case).
619 if self.use_aux_hidden_state_outputs:
620 hidden_states, aux_hidden_states = model_output
621 else:
622 hidden_states = model_output
623 aux_hidden_states = []
624 if self.hidden_states is None:
625 self.hidden_states = torch.empty_like(hidden_states)
626 self.hidden_states[:num_tokens] = hidden_states
627 if self.use_aux_hidden_state_outputs and not self.aux_hidden_states:
628 self.aux_hidden_states = [
629 torch.empty_like(x) for x in aux_hidden_states
630 ]
631 for i, aux in enumerate(aux_hidden_states):
632 self.aux_hidden_states[i][:num_tokens] = aux
633 else:
634 # Non-last PP rank.
635 assert isinstance(model_output, IntermediateTensors)
636 intermediate_tensors = model_output
637 if self.intermediate_tensors is None:
638 self.intermediate_tensors = IntermediateTensors.empty_like(
639 intermediate_tensors
640 )
641 for k, v in intermediate_tensors.tensors.items():
642 self.intermediate_tensors[k][:num_tokens] = v

644 def create_forward_fn(
645 desc: BatchExecutionDescriptor,
646 warmup: bool,
647 ) -> Callable[[CUDAGraphMode], None]:
648 num_tokens = desc.num_tokens
649 num_reqs = desc.num_reqs or min(num_tokens, self.max_num_reqs)
650
651 # Set LoRA state before capture so kernels see correct adapters.
652 if lora_capture_hook is not None:
653 lora_capture_hook(desc.num_active_loras, num_reqs, num_tokens)
654
655 num_tokens_across_dp = (
656 torch.full((self.dp_size,), num_tokens, dtype=torch.int32, device="cpu")
657 if self.dp_size > 1
658 else None
659 )
660
661 model_inputs = {
662 "input_ids": input_buffers.input_ids[:num_tokens],
663 "positions": input_buffers.positions[:num_tokens],
664 **model_state.prepare_dummy_inputs(num_reqs, num_tokens),
665 }
666 if not self.is_first_pp_rank:
667 # Update for non-first PP ranks.
668 model_inputs["input_ids"] = None
669 model_inputs["inputs_embeds"] = None
670 assert intermediate_tensors is not None
671 model_inputs["intermediate_tensors"] = intermediate_tensors[:num_tokens]
672
673 if desc.num_ubatches > 1:
674 # Prepare and park threads before capture; finish runs inside it.
675 assert self.ubatch_runner is not None
676 ubatch_state = self.ubatch_runner.prepare(
677 InputBatch.make_dummy(num_reqs, num_tokens, input_buffers),
678 block_tables.get_dummy_block_tables(num_reqs),
679 block_tables.get_dummy_slot_mappings(num_tokens),
680 cg_mode=CUDAGraphMode.FULL,
681 for_capture=True,
682 )
683 # Capture with dummy rows marked as padding.
684 input_buffers.is_padding.fill_(True)
685 finish = self.ubatch_runner.begin_capturable_run(
686 model, model_inputs, ubatch_state, for_capture=True
687 )
688
689 def ubatch_forward_fn(cg_mode: CUDAGraphMode) -> None:
690 assert cg_mode != CUDAGraphMode.PIECEWISE, (
691 "DBO does not support PIECEWISE cudagraphs"
692 )
693 store_capture_output(num_tokens, finish())
694
695 return ubatch_forward_fn
696
697 attn_metadata, slot_mappings = prepare_inputs_to_capture(
698 num_reqs,
699 num_tokens,
700 model_state,
701 input_buffers,
702 block_tables,
703 attn_groups,
704 kv_cache_config,
705 full_cudagraph=desc.cg_mode == CUDAGraphMode.FULL,
706 max_query_len=desc.max_query_len,
707 pcp_manager=pcp_manager,
708 )
709
710 # Capture with dummy rows marked as padding.
711 input_buffers.is_padding.fill_(True)
712
713 def forward_fn(cg_mode: CUDAGraphMode) -> None:
714 batch_descriptor = None
715 if cg_mode == CUDAGraphMode.PIECEWISE:
716 batch_descriptor = BatchDescriptor(
717 num_tokens=num_tokens,
718 has_lora=has_lora,
719 num_active_loras=desc.num_active_loras,
720 )
721 with set_forward_context(
722 attn_metadata,
723 self.vllm_config,
724 num_tokens=num_tokens,
725 cudagraph_runtime_mode=cg_mode,
726 num_tokens_across_dp=num_tokens_across_dp,
727 slot_mapping=slot_mappings,
728 batch_descriptor=batch_descriptor,
729 is_padding=input_buffers.is_padding[:num_tokens],
730 ):
731 if cg_mode == CUDAGraphMode.PIECEWISE:
732 # PIECEWISE graph (compiled PW or breakable, chosen inside
733 # run_pw_graph).
734 model_output = self.run_pw_graph(model, model_inputs)
735 else:
736 model_output = model(**model_inputs)
737
738 if cg_mode == CUDAGraphMode.PIECEWISE:
739 # PW CUDA graph (compiled or breakable) internally handles the
740 # model outputs. No need to keep track of the hidden states.
741 return None
742
743 store_capture_output(num_tokens, model_output)
744
745 return forward_fn
746
747 super().capture(create_forward_fn, progress_bar_desc)

749 def run_fullgraph(
750 self, desc: BatchExecutionDescriptor
751 ) -> torch.Tensor | tuple[torch.Tensor, list[torch.Tensor]] | IntermediateTensors:
752 """Replay a captured FULL cudagraph and return hidden states."""
753 super().run_fullgraph(desc)
754 if not self.is_last_pp_rank:
755 assert self.intermediate_tensors is not None
756 return self.intermediate_tensors[: desc.num_tokens]
757
758 assert self.hidden_states is not None
759 hidden_states = self.hidden_states[: desc.num_tokens]
760 if not self.use_aux_hidden_state_outputs:
761 return hidden_states
762 return hidden_states, [x[: desc.num_tokens] for x in self.aux_hidden_states]

765def prepare_inputs_to_capture(
766 num_reqs: int,
767 num_tokens: int,
768 model_state: ModelState,
769 input_buffers: InputBuffers,
770 block_tables: BlockTables,
771 attn_groups: list[list[AttentionGroup]],
772 kv_cache_config: KVCacheConfig,
773 full_cudagraph: bool,
774 max_query_len: int | None = None,
775 pcp_manager: "PCPManager | None" = None,
776) -> AttentionState:
777 input_batch = InputBatch.make_dummy(
778 num_reqs, num_tokens, input_buffers, max_query_len=max_query_len
779 )
780 if pcp_manager is not None:
781 input_batch = pcp_manager.prepare_inputs_to_capture(input_batch)
782
783 block_table_provider = pcp_manager or block_tables
784 input_block_tables = block_table_provider.get_dummy_block_tables(num_reqs)
785 slot_mappings = block_table_provider.get_dummy_slot_mappings(num_tokens)
786 slot_mappings_by_layer = build_slot_mappings_by_layer(
787 slot_mappings, kv_cache_config
788 )
789
790 input_batch.dcp_local_seq_lens = maybe_prepare_dcp_local_seq_lens(
791 input_buffers.dcp_local_seq_lens,
792 input_batch.seq_lens,
793 input_batch.num_reqs,
794 block_tables.cp_size,
795 block_tables.cp_rank,
796 block_tables.cp_interleave,
797 num_reqs_padded=input_batch.num_reqs_after_padding,
798 )
799
800 # NOTE(woosuk): Attention metadata is required not just by standard attention
801 # kernels, but also by specialized attention-like operations (e.g., Inkling's sconv,
802 # DSV4 compressor), which maintain their own states and require special metadata
803 # such as block tables.
804 # During CUDA graph capture:
805 # - For FULL CUDA graphs: We set for_capture=True so that both attention and
806 # attention-like ops produce capturable metadata compatible with CUDA graphs.
807 # - For PIECEWISE CUDA graphs: We still build attention metadata, but set
808 # for_capture=False. This is because:
809 # * Attention-like ops (such as sconv or DSV4 compressor) may not be used as
810 # breakpoints in PIECEWISE CUDA graphs, so we must generate their attention
811 # metadata so they can execute and be captured during graph capture.
812 # * Standard attention ops that are treated as breakpoints will be executed
813 # eagerly at capture time (not included in the graph itself), and for these,
814 # setting for_capture=False is essential. Some attention backends
815 # (like linear attention) cannot generate capturable metadata for prefill,
816 # so for_capture=False ensures they execute without issue.
817 # * We assume that attention-like operations intended for capture will still
818 # produce capturable metadata, even when for_capture=False. While this
819 # assumption is brittle, it currently works in practice.
820 # In summary: We always generate attention metadata for both FULL and PIECEWISE
821 # CUDA graphs, setting for_capture=True for FULL graphs, and for_capture=False
822 # for PIECEWISE graphs, to ensure correct execution and capture.
823 attn_metadata = model_state.prepare_attn(
824 input_batch,
825 CUDAGraphMode.NONE,
826 input_block_tables,
827 slot_mappings,
828 attn_groups,
829 kv_cache_config,
830 for_capture=full_cudagraph,
831 )
832 return AttentionState(attn_metadata, slot_mappings_by_layer)

835# ---------------------------------------------------------------------------
836# CUDA graph memory profiling
837# ---------------------------------------------------------------------------
838
839# Number of FULL graphs captured during profiling; the total FULL capture
840# cost is extrapolated from this sample to avoid a second full capture.
841_FULL_GRAPH_PROFILING_SAMPLES = 2
842# Floor for the extrapolated per-graph cost (driver overhead per graph).
843_MIN_PER_GRAPH_BYTES = 1 << 20
844
845
846@torch.inference_mode()
847def profile_cudagraph_memory(runner: "GPUModelRunner") -> int:
848 """Estimate the GPU memory needed for CUDA graph capture.
849
850 Called during memory profiling, *before* the real KV cache is allocated,
851 so that ``Worker.determine_available_memory`` can reserve headroom for
852 graph capture. Bootstraps a minimal KV cache, runs ``capture_model()``
853 once, then releases everything so the real init/capture path starts clean.
854
855 FULL graphs bake in KV cache pointers, so only the largest few are
856 captured (into a throwaway pool) and their total cost is extrapolated.
857 PIECEWISE, encoder and speculator graphs are measured in full. All
858 profiling captures are discarded afterwards: replaying graphs recorded
859 against the throwaway profiling state is unsafe (e.g. inductor graph
860 partition reclaims the storages of earlier cudagraph recordings once the
861 real capture records new ones, leading to use-after-free crashes).
862 """
863 if runner.compilation_config.cudagraph_mode == CUDAGraphMode.NONE:
864 return 0
865
866 gc.collect()
867 torch.accelerator.empty_cache()
868
869 # Run the whole profiling phase against a throwaway CUDA graph pool by
870 # pointing the global graph pool singleton at it: objects that bind the
871 # pool lazily during profiling (speculator cudagraph managers, breakable
872 # runners created mid-capture) then land on the throwaway pool too. Pools
873 # bound before profiling (piecewise wrappers) are swapped explicitly in
874 # the inner block. Profiling graphs captured into the persistent global
875 # pool and then discarded would drop its use_count to 0, tripping the c10
876 # allocator's create_or_incref_pool assert when the real capture reuses
877 # that pool ("use_count > 0 INTERNAL ASSERT FAILED").
878 platform_cls = type(current_platform)
879 saved_global_pool = platform_cls._global_graph_pool
880 throwaway_pool = current_platform.graph_pool_handle()
881 platform_cls._global_graph_pool = throwaway_pool
882
883 try:
884 with set_current_vllm_config(runner.vllm_config):
885 _init_minimal_kv_cache_for_profiling(runner)
886
887 manager = runner.cudagraph_manager
888 assert manager is not None
889
890 # Don't count profiling captures; the real capture_model() runs later.
891 saved_num_cudagraph_captured = compilation_counter.num_cudagraph_captured
892 saved_capture_triggers = compilation_counter.num_gpu_runner_capture_triggers
893 all_wrappers: list[Any] = []
894 original_pools: dict[int, Any] = {}
895 speculator = getattr(runner, "speculator", None)
896 spec_manager_names: list[str] = []
897 try:
898 if not manager.needs_capture():
899 return 0
900 manager.pool = throwaway_pool
901 if manager.use_breakable_cg:
902 # Create the breakable runner before the wrapper pool swap so
903 # its pool is covered as well.
904 manager.init_breakable_cg_runner(runner.model)
905 all_wrappers = list(CUDAGraphWrapper._all_instances) + list(
906 BreakableCUDAGraphWrapper._all_instances
907 )
908 for wrapper in all_wrappers:
909 original_pools[id(wrapper)] = wrapper.graph_pool
910 wrapper.graph_pool = throwaway_pool
911 if speculator is not None:
912 spec_manager_names = [
913 name
914 for name, value in vars(speculator).items()
915 if isinstance(value, CudaGraphManager)
916 ]
917 manager._max_full_descs_to_capture = _FULL_GRAPH_PROFILING_SAMPLES
918 mem_samples: list[int] = []
919 manager._capture_mem_samples = mem_samples
920
921 measured = int(runner.capture_model(profile_only=True))
922
923 # The measured delta covers PIECEWISE, encoder and speculator graphs
924 # plus the sampled FULL graphs; swap the sampled FULL cost for the
925 # extrapolated total. FULL and PIECEWISE share one pool here just as
926 # they share the global pool at runtime, so the overlap is not
927 # double-counted.
928 num_full_graphs = len(manager._capture_descs.get(CUDAGraphMode.FULL, []))
929 full_estimate = _extrapolate_full_graph_memory(mem_samples, num_full_graphs)
930 return max(measured - sum(mem_samples) + full_estimate, 0)
931 finally:
932 compilation_counter.num_cudagraph_captured = saved_num_cudagraph_captured
933 compilation_counter.num_gpu_runner_capture_triggers = saved_capture_triggers
934 CUDAGraphWrapper.clear_all_graphs()
935 BreakableCUDAGraphWrapper.clear_all_graphs()
936 for wrapper in all_wrappers:
937 if id(wrapper) in original_pools:
938 wrapper.graph_pool = original_pools[id(wrapper)]
939 # Drop the speculator's cudagraph managers; the real
940 # initialize_kv_cache re-creates them. Their profiling graphs
941 # release the throwaway pool here rather than after the real init.
942 for name in spec_manager_names:
943 setattr(speculator, name, None)
944 # Drop local references before teardown detaches the runner's
945 # manager and flushes the allocator.
946 del manager
947 _teardown_profiling_state(runner)
948 finally:
949 platform_cls._global_graph_pool = saved_global_pool

952def _extrapolate_full_graph_memory(mem_samples: list[int], total_graphs: int) -> int:
953 """Extrapolate the total FULL capture cost from samples of the largest
954 graphs. The first capture allocates the pool baseline; later graphs mostly
955 reuse it, so the second sample is taken as the per-graph cost."""
956 if not mem_samples:
957 return 0
958 first_capture = mem_samples[0]
959 per_graph = max(mem_samples[1], _MIN_PER_GRAPH_BYTES) if len(mem_samples) > 1 else 0
960 return first_capture + (total_graphs - 1) * per_graph

964 """Allocate the smallest KV cache that still lets every graph be captured."""
965 from vllm.v1.core.kv_cache_utils import (
966 get_kv_cache_config_from_groups,

968 )

973 min_blocks = (

975 or 1
976 )

990def _teardown_profiling_state(runner: "GPUModelRunner") -> None:
991 """Release the profiling KV cache and captured graphs while keeping model
992 weights, so the real ``initialize_kv_cache`` starts from a clean slate."""
993 torch.accelerator.synchronize()
994 if hasattr(runner.model_state, "_mamba_ctx"):
995 runner.model_state._mamba_ctx = None
996 # Invalidate the align-mode Mamba group metadata cached from the
997 # profiling KVCacheConfig: the real (e.g. PP-projected) config may
998 # place Mamba layers into a different group layout, so it must be
999 # re-derived from the real config.
1000 if hasattr(runner.model_state, "_mamba_group_ids"):
1001 runner.model_state._mamba_group_ids = []
1002 if hasattr(runner.model_state, "_mamba_spec"):
1003 runner.model_state._mamba_spec = None
1004 if hasattr(runner, "kv_caches"):
1005 runner.kv_caches.clear()
1006 if hasattr(runner, "attn_groups"):
1007 runner.attn_groups.clear()
1008 if hasattr(runner, "kv_cache_config"):
1009 del runner.kv_cache_config
1010 # Dropping the manager releases the profiling graphs and throwaway pool.
1011 runner.cudagraph_manager = None
1012 # Release encoder graphs captured during profiling; the real
1013 # capture_model() re-captures them.
1014 if runner.model_state.supports_mm_inputs:
1015 runner.model_state.encoder_runner.clear()
1016 # Detach profiling KV tensors held by attention layers. The layers live
1017 # in the static forward context for compiled models.
1018 layers: Iterable[Any] = runner.compilation_config.static_forward_context.values()
1019 if (model := getattr(runner, "model", None)) is not None:
1020 layers = itertools.chain(layers, model.modules())
1021 clear_layer_kv_caches(layers)
1022 release_hisparse_profiling_cache(runner.compilation_config.static_forward_context)
1023 runner.cache_config.num_gpu_blocks = None
1024 runner.maybe_remove_all_loras(runner.lora_config)
1025 gc.collect()
1026 torch.accelerator.empty_cache()
