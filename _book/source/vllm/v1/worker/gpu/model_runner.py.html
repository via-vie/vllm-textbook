3"""
4NOTE: Coding style guide for this file:
5This model runner is shared by all models: text and multimodal, generative
6and embedding, public and private. As a result, this file must only contain
7code that is common to every model. Model-specific behavior belongs in the
8appropriate model-specific files.
9
10In other words:
11* Be paranoid about changing this file. It should remain stable.
12* Be even more paranoid about adding new lines. It should remain minimal.
13
14Even for shared features (for example, different parallelism modes), keep the
15complexity out of this path. The less common the feature, the more it should be
16hidden. Prefer utility functions defined elsewhere and call them from here,
17instead of embedding feature-specific logic directly.
18"""

388 def load_model(self, load_dummy_weights: bool = False, *args, **kwargs) -> None:
389 time_before_load = time.perf_counter()
390 if load_dummy_weights:
391 self.load_config.load_format = "dummy"
392 self.eplb.prepare_load()
393 eplb_models_added = False
394 with DeviceMemoryProfiler() as m:
395 model_loader = get_model_loader(self.vllm_config.load_config)
396 logger.info_once("Loading model from scratch...")
397
398 # Capture warmup providers selected while constructing the model.
399 with self.jit_warmup_registry.activate():
400 self.model = model_loader.load_model(
401 vllm_config=self.vllm_config,
402 model_config=self.vllm_config.model_config,
403 )
404 if self.lora_config:
405 self.model = self.load_lora_model(
406 self.model, self.vllm_config, self.device
407 )
408
409 if self.use_aux_hidden_state_outputs:
410 assert self.speculative_config is not None
411 set_eagle3_aux_hidden_state_layers(self.model, self.speculative_config)
412 if self.use_pp:
413 assert self.speculative_config.method is not None
414 verify_supports_aux_hidden_states_over_pp(
415 self.model, self.speculative_config.method
416 )
417 assert self.pp_handler is not None
418 self.pp_handler.configure_aux_hidden_state_relay(self.model)
419 if isinstance(self.speculator, DraftModelSpeculator):
420 with use_workspace_lane(self._draft_workspace_lane):
421 self.speculator.load_model(self.model)
422 eplb_models_added = self.eplb.maybe_register_speculator(
423 self.speculator, self.speculative_config, load_dummy_weights
424 )
425 time_after_load = time.perf_counter()
426
427 self.model_memory_usage = m.consumed_memory
428 logger.info(
429 "Model loading took %s GiB memory and %.6f seconds",
430 format_gib(m.consumed_memory),
431 time_after_load - time_before_load,
432 )
433
434 # Initialize the components that require the model.
435 self.model_state = init_model_state(
436 self.vllm_config, self.model, self.encoder_cache, self.device
437 )
438
439 self.decode_query_len = (
440 self.num_speculative_steps
441 + self.model_state.num_new_sampled_tokens_per_step
442 )
443
444 if self.parallel_config.enable_batch_sharded_sampling:
445 if hasattr(self.model, "compute_logits_local"):
446 self.batch_sharder = BatchSharder(
447 max_num_reqs=self.max_num_reqs,
448 max_num_logits_per_req=self.decode_query_len,
449 device=self.device,
450 )
451 logger.info("Batch-sharded sampling enabled.")
452 else:
453 logger.warning_once(
454 "Disabling batch-sharded sampling: %s does not implement "
455 "compute_logits_local",
456 type(self.model).__name__,
457 )
458
459 # Initialize samplers. Model states may override via custom_sampler().
460 if self.is_last_pp_rank and not self.is_pooling_model:
461 sampler_kwargs: dict[str, Any] = {
462 "max_num_reqs": self.max_num_reqs,
463 "vocab_size": self.vocab_size,
464 "device": self.device,
465 "req_states": self.req_states,
466 "logprobs_mode": self.model_config.logprobs_mode,
467 "num_speculative_tokens": self.decode_query_len,
468 "use_fp64_gumbel": self.model_config.use_fp64_gumbel,
469 "enable_trace_replay": self.model_config.enable_trace_replay,
470 "reasoning_config": self.vllm_config.reasoning_config,
471 "return_sampling_mask": self.model_config.return_sampling_mask,
472 }
473 if self.vllm_config.watermark_config is None:
474 self.sampler = Sampler(**sampler_kwargs)
475 else:
476 wm_config = self.vllm_config.watermark_config
477 watermarker = create_watermarker(wm_config)
478 if self.speculative_config is not None:
479 watermarker = create_speculative_target_watermarker(watermarker)
480 self.sampler = GPUWatermarkSampler(
481 watermarker,
482 deduplicate_contexts=wm_config.deduplicate_contexts,
483 deduplicate_contexts_max_history=(
484 wm_config.deduplicate_contexts_max_history
485 ),
486 **sampler_kwargs,
487 )
488 custom = self.model_state.custom_sampler(self.sampler)
489
490 if custom:
491 self.vllm_config._check_watermarking_unsupported(custom_sampler=True)
492 self.sampler, self.rejection_sampler = custom
493 elif self.speculative_config is not None:
494 self.rejection_sampler = RejectionSampler(
495 self.sampler,
496 self.speculative_config,
497 self.device,
498 watermark_key=speculative_target_watermark_key(
499 self.vllm_config.watermark_config
500 ),
501 )
502 self.prompt_logprobs_worker = PromptLogprobsWorker(
503 self.max_num_reqs,
504 logprobs_mode=self.model_config.logprobs_mode,
505 )
506 self.structured_outputs_worker = StructuredOutputsWorker(
507 max_num_logits=self.max_num_reqs * self.decode_query_len,
508 vocab_size=self.vocab_size,
509 device=self.device,
510 mask_stride=self.decode_query_len,
511 num_bonus_tokens=self.model_state.num_new_sampled_tokens_per_step,
512 )
513
514 if self.is_pooling_model and self.is_last_pp_rank:
515 self.pooling_runner = PoolingRunner(self.model, self.vllm_config)
516 eplb_models_added |= self.eplb.maybe_register_model(
517 self.model,
518 self.model_config,
519 load_dummy_weights,
520 )
521 self.eplb.maybe_start_async_loop(eplb_models_added)
522
523 if not self.is_first_pp_rank:
524 # For non-first PP ranks, create intermediate tensors sized
525 # for the max capture size so they can be sliced per batch.
526 # Save as persistent member so runtime can copy received data
527 # into the same addresses that the CUDA graphs captured.
528 self.intermediate_tensors = self.model.make_empty_intermediate_tensors(
529 batch_size=self.max_num_tokens,
530 dtype=self.model_config.dtype,
531 device=self.device,
532 )
533
534 get_offloader().post_init()

1513 self,
1514 hidden_states: torch.Tensor,

1516 grammar_output: GrammarOutput | None,

1518 shard_metadata = None
