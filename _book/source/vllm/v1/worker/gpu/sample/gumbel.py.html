7# Smallest positive value produced by Triton's fp32 `tl.rand`. Used to clamp
8# zero draws before the flipped Gumbel transform below.
9#
10# Triton requires globals accessed from `@triton.jit` functions to be wrapped
11# in `tl.constexpr(...)`. We can only do that when Triton is actually
12# available — on the CPU worker path `tl` is a placeholder whose `constexpr`
13# attribute is `None`, and `tl.constexpr(...)` would crash at import time.
14_TL_RAND_MIN = tl.constexpr(4.6566127342e-10) if HAS_TRITON else 4.6566127342e-10
15
16# Offset salt keeping the draft's Gumbel noise disjoint from the target's.
17# Verification is a probability-ratio test, not a Gumbel coupling, so a proposal
18# and the residual it is resampled from must not share a noise vector.
19# Positions are int64 and never approach 2**30, so the streams cannot collide.
20_DRAFT_NOISE_SALT = tl.constexpr(1 << 30) if HAS_TRITON else (1 << 30)

23@triton.jit
24def _temperature_kernel(
25 logits_ptr,
26 logits_stride,
27 expanded_idx_mapping_ptr,
28 temperature_ptr,
29 vocab_size,
30 BLOCK_SIZE: tl.constexpr,
31):
32 token_idx = tl.program_id(0).to(tl.int64)
33 req_state_idx = tl.load(expanded_idx_mapping_ptr + token_idx)
34 temperature = tl.load(temperature_ptr + req_state_idx).to(tl.float32)
35 if temperature == 0.0 or temperature == 1.0:
36 # Early return to avoid loading logits.
37 return
38
39 block_idx = tl.program_id(1)
40 block = block_idx * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
41 mask = block < vocab_size
42
43 logits = tl.load(logits_ptr + token_idx * logits_stride + block, mask=mask)
44 logits = logits.to(tl.float32)
45 logits = logits / temperature
46 tl.store(logits_ptr + token_idx * logits_stride + block, logits, mask=mask)
47

49def apply_temperature(
50 logits: torch.Tensor,
51 expanded_idx_mapping: torch.Tensor,
52 temperature: torch.Tensor,
53) -> None:
54 num_tokens, vocab_size = logits.shape
55 BLOCK_SIZE = 8192
56 num_blocks = triton.cdiv(vocab_size, BLOCK_SIZE)
57 _temperature_kernel[(num_tokens, num_blocks)](
58 logits,
59 logits.stride(0),
60 expanded_idx_mapping,
61 temperature,
62 vocab_size,
63 BLOCK_SIZE=BLOCK_SIZE,
64 )

67@triton.jit
68def tl_rand64(seed, offset, includes_zero: tl.constexpr):
69 lo, hi, _, _ = tl.randint4x(seed, offset)
70 lo = lo.to(tl.uint32, bitcast=True).to(tl.uint64)
71 hi = hi.to(tl.uint32, bitcast=True).to(tl.uint64)
72 r = (hi << 32) | lo
73
74 # 1 / 2**64
75 scale = 5.421010862427522170037e-20
76 u = r.to(tl.float64) * scale
77 if not includes_zero:
78 u = tl.maximum(u, 2.2250738585072014e-308) # float64 tiny
79 return u

82@triton.jit
83def tl_rand32(seed, offset, includes_zero: tl.constexpr):
84 u = tl.rand(seed, offset)
85 if not includes_zero:
86 u = tl.maximum(u, _TL_RAND_MIN)
87 return u

90@triton.jit
91def gumbel_noised_argmax(
92 logits,
93 keys,
94 mask,
95 seed,
96 pos,
97 temp,
98 IS_DRAFTING: tl.constexpr,
99 USE_FP64: tl.constexpr,
100 APPLY_TEMPERATURE: tl.constexpr = True,
101):
102 """Argmax of logits under Gumbel-max sampling, or plain argmax at temp 0.
103
104 `keys` indexes the noise, so the same token draws the same noise wherever it
105 appears; `pos` and `seed` place the draw in the request's stream, which is
106 what lets a draft and its verification agree.
107 """
108 if temp != 0.0 and APPLY_TEMPERATURE:
109 # Match the behavior of _temperature_kernel: if that kernel uses
110 # tl.div_rn, this must too.
111 logits = logits / temp
112
113 # fp32 is the default reduction dtype; fp64 is ~1/32-1/64x the throughput
114 # on H100/Ada/Blackwell and empirically indistinguishable for Gumbel-max.
115 if USE_FP64:
116 logits = logits.to(tl.float64)
117 if temp != 0.0:
118 if IS_DRAFTING:
119 pos = pos + _DRAFT_NOISE_SALT
120 gumbel_seed = tl.randint(seed, pos)
121 if USE_FP64:
122 u = tl_rand64(gumbel_seed, keys, includes_zero=False)
123 gumbel_noise = -tl.log(-tl.log(u))
124 else:
125 u = tl_rand32(gumbel_seed, keys, includes_zero=False)
126 # log1p keeps the winning tail at u -> 0, where fp32 resolves it.
127 gumbel_noise = -tl.log(-tldevice.log1p(-u))
128 logits = tl.where(mask, logits + gumbel_noise, float("-inf"))
129
130 return tl.max(logits, axis=0, return_indices=True)

133@triton.jit
134def gumbel_block_argmax(
135 logits,
136 block,
137 mask,
138 token_idx,
139 expanded_idx_mapping_ptr,
140 temp_ptr,
141 seeds_ptr,
142 pos_ptr,
143 # [max_num_reqs, num_cols, vocab_size]
144 logits_cache_ptr,
145 logits_cache_stride_0,
146 logits_cache_stride_1,
147 logits_cache_col_ptr,
148 vocab_size,
149 IS_DRAFTING: tl.constexpr,
150 APPLY_TEMPERATURE: tl.constexpr,
151 USE_FP64: tl.constexpr,
152 PER_TOKEN_COL: tl.constexpr = False,
153):
154 req_state_idx = tl.load(expanded_idx_mapping_ptr + token_idx).to(tl.int64)
155 is_valid_req = req_state_idx >= 0
156 temp = tl.load(temp_ptr + req_state_idx, mask=is_valid_req, other=0.0).to(
157 tl.float32
158 )
159 if logits_cache_ptr is not None:
160 # Store the logits *before* temperature. Dividing first would produce a
161 # value that is generally not representable in the cache's dtype, forcing
162 # it to be fp32. Consumers (the rejection sampler) divide by the same
163 # temperature on load, which reproduces the value used below bitwise.
164 if PER_TOKEN_COL:
165 col = tl.load(logits_cache_col_ptr + token_idx)
166 else:
167 col = tl.load(logits_cache_col_ptr)
168 tl.store(
169 logits_cache_ptr
170 + req_state_idx * logits_cache_stride_0
171 + col * logits_cache_stride_1
172 + block,
173 logits,
174 mask=mask & is_valid_req,
175 )
176
177 seed = tl.load(seeds_ptr + req_state_idx, mask=is_valid_req, other=0)
178 pos = tl.load(pos_ptr + token_idx)
179 return gumbel_noised_argmax(
180 logits,
181 block,
182 mask,
183 seed,
184 pos,
185 temp,
186 IS_DRAFTING=IS_DRAFTING,
187 USE_FP64=USE_FP64,
188 APPLY_TEMPERATURE=APPLY_TEMPERATURE,
189 )

192@triton.jit
193def _gumbel_sample_kernel(
194 local_argmax_ptr,
195 local_argmax_stride,
196 local_max_ptr,
197 local_max_stride,
198 # [max_num_reqs, num_cols, vocab_size]
199 logits_cache_ptr,
200 logits_cache_stride_0,
201 logits_cache_stride_1,
202 logits_cache_col_ptr,
203 logits_ptr,
204 logits_stride,
205 expanded_idx_mapping_ptr,
206 seeds_ptr,
207 pos_ptr,
208 temp_ptr,
209 vocab_size,
210 BLOCK_SIZE: tl.constexpr,
211 IS_DRAFTING: tl.constexpr,
212 APPLY_TEMPERATURE: tl.constexpr,
213 USE_FP64: tl.constexpr,
214 PER_TOKEN_COL: tl.constexpr,
215):
216 token_idx = tl.program_id(0).to(tl.int64)
217 block_idx = tl.program_id(1)
218 block = block_idx * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
219 mask = block < vocab_size
220 logits = tl.load(
221 logits_ptr + token_idx * logits_stride + block,
222 mask=mask,
223 other=float("-inf"),
224 )
225 logits = logits.to(tl.float32)
226
227 value, idx = gumbel_block_argmax(
228 logits,
229 block,
230 mask,
231 token_idx,
232 expanded_idx_mapping_ptr,
233 temp_ptr,
234 seeds_ptr,
235 pos_ptr,
236 logits_cache_ptr,
237 logits_cache_stride_0,
238 logits_cache_stride_1,
239 logits_cache_col_ptr,
240 vocab_size,
241 IS_DRAFTING=IS_DRAFTING,
242 APPLY_TEMPERATURE=APPLY_TEMPERATURE,
243 USE_FP64=USE_FP64,
244 PER_TOKEN_COL=PER_TOKEN_COL,
245 )
246 token_id = block_idx * BLOCK_SIZE + idx
247 tl.store(local_argmax_ptr + token_idx * local_argmax_stride + block_idx, token_id)
248 tl.store(local_max_ptr + token_idx * local_max_stride + block_idx, value)

251def gumbel_sample(
252 logits: torch.Tensor, # [num_tokens, vocab_size]
253 expanded_idx_mapping: torch.Tensor, # [num_tokens]
254 temperature: torch.Tensor, # [max_num_reqs]
255 seed: torch.Tensor, # [max_num_reqs]
256 pos: torch.Tensor, # [num_tokens]
257 apply_temperature: bool,
258 is_drafting: bool,
259 logits_cache: torch.Tensor | None = None, # [max_num_reqs, num_cols, vocab_size]
260 logits_cache_col: torch.Tensor | None = None, # scalar or [num_tokens]
261 use_fp64: bool = False,
262) -> torch.Tensor:
263 # Enforce contiguity on non-strided input tensors
264 expanded_idx_mapping = expanded_idx_mapping.contiguous()
265 pos = pos.contiguous()
266 if logits_cache_col is not None:
267 logits_cache_col = logits_cache_col.contiguous()
268 num_tokens, vocab_size = logits.shape
269 if logits_cache is not None:
270 assert logits_cache.size(-1) >= vocab_size, (
271 f"draft logits cache vocab dim ({logits_cache.size(-1)}) is narrower "
272 f"than the sampled logits ({vocab_size}). Cached logits would be "
273 "truncated."
274 )
275 BLOCK_SIZE = 1024
276 num_blocks = triton.cdiv(vocab_size, BLOCK_SIZE)
277 local_argmax = logits.new_empty(num_tokens, num_blocks, dtype=torch.int64)
278 local_max_dtype = torch.float64 if use_fp64 else torch.float32
279 local_max = logits.new_empty(num_tokens, num_blocks, dtype=local_max_dtype)
280 per_token_col = logits_cache_col is not None and logits_cache_col.dim() > 0
281 _gumbel_sample_kernel[(num_tokens, num_blocks)](
282 local_argmax,
283 local_argmax.stride(0),
284 local_max,
285 local_max.stride(0),
286 logits_cache,
287 logits_cache.stride(0) if logits_cache is not None else 0,
288 logits_cache.stride(1) if logits_cache is not None else 0,
289 logits_cache_col,
290 logits,
291 logits.stride(0),
292 expanded_idx_mapping,
293 seed,
294 pos,
295 temperature,
296 vocab_size,
297 BLOCK_SIZE=BLOCK_SIZE,
298 IS_DRAFTING=is_drafting,
299 APPLY_TEMPERATURE=apply_temperature,
300 USE_FP64=use_fp64,
301 PER_TOKEN_COL=per_token_col,
302 )
303 # NOTE(woosuk): Use int64 for later indexing.
304 max_block_idx = local_max.argmax(dim=-1, keepdim=True)
305 sampled = local_argmax.gather(dim=-1, index=max_block_idx).view(-1)
306 return sampled
