25@triton.jit
26def _compact_sampling_mask_kernel(
27 logits_ptr,
28 logits_row_stride,
29 logits_col_stride,
30 num_sampled_tokens_ptr,
31 token_ids_ptr,
32 token_ids_row_stride,
33 packed_mask_ptr,
34 packed_mask_row_stride,
35 counts_ptr,
36 vocab_size,
37 max_num_kept,
38 BLOCK_SIZE: tl.constexpr,
39):
40 """Per row: first ``max_num_kept`` finite-logit ids, the full count, the bitmask."""
41 req_idx = tl.program_id(0)
42 is_active = tl.load(num_sampled_tokens_ptr + req_idx) > 0
43 count = tl.zeros((), dtype=tl.int32)
44
45 for start_idx in range(0, vocab_size, BLOCK_SIZE):
46 offsets = start_idx + tl.arange(0, BLOCK_SIZE)
47 logits = tl.load(
48 logits_ptr + req_idx * logits_row_stride + offsets * logits_col_stride,
49 mask=offsets < vocab_size,
50 other=-float("inf"),
51 )
52 keep = (logits > -float("inf")) & (logits < float("inf")) & is_active
53 keep_i32 = keep.to(tl.int32)
54 pos = count + tl.cumsum(keep_i32, axis=0) - keep_i32
55 tl.store(
56 token_ids_ptr + req_idx * token_ids_row_stride + pos,
57 offsets.to(tl.int32),
58 mask=keep & (pos < max_num_kept),
59 )
60 count += tl.sum(keep_i32)
61
62 bits = tl.reshape(keep_i32, (BLOCK_SIZE // 8, 8)) << tl.arange(0, 8)[None, :]
63 byte_offsets = start_idx // 8 + tl.arange(0, BLOCK_SIZE // 8)
64 tl.store(
65 packed_mask_ptr + req_idx * packed_mask_row_stride + byte_offsets,
66 tl.sum(bits, axis=1).to(tl.uint8),
67 mask=byte_offsets < tl.cdiv(vocab_size, 8),
68 )
69
70 tl.store(counts_ptr + req_idx, count)

73# Bounds the [num_reqs, width] int32 buffer; wider rows use the bitmask instead.
74MAX_COMPACT_SUPPORT = 2048
75
76
77class SamplingMaskTensors(NamedTuple):
78 """Device-side masks pending async D2H: compact ids, plus the bitmask as
79 the exact fallback for rows wider than ``max_num_kept``."""
80
81 # [num_requests, max_num_kept]
82 token_ids: torch.Tensor
83 # [num_requests, ceil(vocab_size / 8)]
84 packed_mask: torch.Tensor
85 # [num_requests]
86 counts: torch.Tensor
87 vocab_size: int
88
89 @classmethod
90 def from_logits(
91 cls,
92 logits: torch.Tensor,
93 num_sampled_tokens: torch.Tensor,
94 max_num_kept: int,
95 ) -> SamplingMaskTensors:
96 """Capture the finite-logit support of every row with a sampled token."""
97 num_reqs, vocab_size = logits.shape
98 max_num_kept = min(max_num_kept, vocab_size, MAX_COMPACT_SUPPORT)
99 device = logits.device
100
101 token_ids = torch.empty(
102 (num_reqs, max_num_kept), dtype=torch.int32, device=device
103 )
104 packed_mask = torch.empty(
105 (num_reqs, (vocab_size + 7) // 8), dtype=torch.uint8, device=device
106 )
107 counts = torch.empty(num_reqs, dtype=torch.int32, device=device)
108 _compact_sampling_mask_kernel[(num_reqs,)](
109 logits,
110 logits.stride(0),
111 logits.stride(1),
112 num_sampled_tokens,
113 token_ids,
114 token_ids.stride(0),
115 packed_mask,
116 packed_mask.stride(0),
117 counts,
118 vocab_size,
119 max_num_kept,
120 BLOCK_SIZE=8192,
121 )
122 return cls(token_ids, packed_mask, counts, vocab_size)
123
124 def to_cpu_nonblocking(self) -> SamplingMaskTensors:
125 if self.token_ids.device.type == "cpu":
126 return self
127 return SamplingMaskTensors(
128 self.token_ids.to("cpu", non_blocking=True),
129 self.packed_mask.to("cpu", non_blocking=True),
130 self.counts.to("cpu", non_blocking=True),
131 self.vocab_size,
132 )
133
134 def tolists(self) -> SamplingMaskLists:
135 """CSR over all requests; rows without a sampled token are empty."""
136 counts = self.counts.cpu().numpy()
137 token_ids = self.token_ids.cpu().numpy()
138 packed_mask = self.packed_mask.cpu().numpy()
139 width = token_ids.shape[1]
140
141 def support(row: int) -> np.ndarray:
142 if counts[row] <= width:
143 return token_ids[row, : counts[row]]
144 # Wider than the compact row (ties or a huge top_k): use the bitmask.
145 bits = np.unpackbits(
146 packed_mask[row], count=self.vocab_size, bitorder="little"
147 )
148 return np.flatnonzero(bits).astype(np.int32, copy=False)
149
150 supports = [support(row) for row in range(len(counts))]
151 offsets = np.zeros(len(supports) + 1, dtype=np.int64)
152 np.cumsum([len(s) for s in supports], out=offsets[1:])
153 return SamplingMaskLists(np.concatenate(supports), offsets)
