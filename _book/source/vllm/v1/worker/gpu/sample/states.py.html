3import numpy as np
4import torch
5
6from vllm.sampling_params import SamplingParams
7from vllm.v1.sample.ops.topk_topp_sampler import apply_top_k_top_p
8from vllm.v1.worker.gpu.buffer_utils import UvaBackedTensor
9from vllm.v1.worker.gpu.sample.gumbel import apply_temperature
10from vllm.v1.worker.gpu.sample.min_p import apply_min_p
11
12NO_LOGPROBS = -1
13_NP_INT64_MIN = np.iinfo(np.int64).min
14_NP_INT64_MAX = np.iinfo(np.int64).max
15
16
17class SamplingStates:
18 def __init__(self, max_num_reqs: int, vocab_size: int):
19 self.max_num_reqs = max_num_reqs
20 self.vocab_size = vocab_size
21
22 self.temperature = UvaBackedTensor(max_num_reqs, dtype=torch.float32)
23 self.top_k = UvaBackedTensor(max_num_reqs, dtype=torch.int32)
24 self.top_p = UvaBackedTensor(max_num_reqs, dtype=torch.float32)
25 self.min_p = UvaBackedTensor(max_num_reqs, dtype=torch.float32)
26 self.seeds = UvaBackedTensor(max_num_reqs, dtype=torch.int64)
27 # Tracks whether `seed` was set explicitly by the user, so callers
28 # can fall back from RNG paths that don't honor per-request seeds.
29 self.seeds_set = np.zeros(max_num_reqs, dtype=bool)
30
31 # Initialize top_k and top_p manually because 0 is an invalid value for them.
32 self.top_k.np.fill(self.vocab_size)
33 self.top_k.copy_to_uva()
34 self.top_p.np.fill(1.0)
35 self.top_p.copy_to_uva()
36
37 self.num_logprobs = np.empty(self.max_num_reqs, dtype=np.int32)
38 # -1 means no logprobs are requested.
39 self.num_logprobs.fill(NO_LOGPROBS)
40
41 def add_request(self, req_idx: int, sampling_params: SamplingParams) -> None:
42 self.temperature.np[req_idx] = sampling_params.temperature
43 self.top_p.np[req_idx] = sampling_params.top_p
44 top_k = sampling_params.top_k
45 if top_k <= 0 or top_k > self.vocab_size:
46 top_k = self.vocab_size
47 self.top_k.np[req_idx] = top_k
48 self.min_p.np[req_idx] = sampling_params.min_p
49
50 seed = sampling_params.seed
51 self.seeds_set[req_idx] = seed is not None
52 if seed is None:
53 seed = np.random.randint(_NP_INT64_MIN, _NP_INT64_MAX)
54 self.seeds.np[req_idx] = seed
55
56 num_logprobs = sampling_params.logprobs
57 if num_logprobs is None:
58 num_logprobs = NO_LOGPROBS
59 elif num_logprobs == -1:
60 num_logprobs = self.vocab_size
61 self.num_logprobs[req_idx] = num_logprobs
62
63 def apply_staged_writes(self) -> None:
64 self.temperature.copy_to_uva()
65 self.top_p.copy_to_uva()
66 self.top_k.copy_to_uva()
67 self.min_p.copy_to_uva()
68 self.seeds.copy_to_uva()
69
70 def apply_temperature(
71 self,
72 logits: torch.Tensor,
73 expanded_idx_mapping: torch.Tensor,
74 idx_mapping_np: np.ndarray,
75 ) -> None:
76 temp_np = self.temperature.np[idx_mapping_np]
77 if np.all((temp_np == 0.0) | (temp_np == 1.0)):
78 # No request requires temperature. Skip the kernel launch.
79 return
80
81 apply_temperature(logits, expanded_idx_mapping, self.temperature.gpu)
82
83 def apply_min_p(
84 self,
85 logits: torch.Tensor,
86 expanded_idx_mapping: torch.Tensor,
87 idx_mapping_np: np.ndarray,
88 ) -> None:
89 if np.all(self.min_p.np[idx_mapping_np] == 0.0):
90 # No request uses min_p. Skip the kernel launch.
91 return
92 apply_min_p(logits, expanded_idx_mapping, self.min_p.gpu)
93
94 def get_top_k_top_p(
95 self, expanded_idx_mapping: torch.Tensor, idx_mapping_np: np.ndarray
96 ) -> tuple[torch.Tensor | None, torch.Tensor | None]:
97 do_top_k = np.any(self.top_k.np[idx_mapping_np] != self.vocab_size)
98 do_top_p = np.any(self.top_p.np[idx_mapping_np] != 1.0)
99 top_k = self.top_k.gpu[expanded_idx_mapping] if do_top_k else None
100 top_p = self.top_p.gpu[expanded_idx_mapping] if do_top_p else None
101 return top_k, top_p
102
103 def apply_top_k_top_p(
104 self,
105 logits: torch.Tensor,
106 expanded_idx_mapping: torch.Tensor,
107 idx_mapping_np: np.ndarray,
108 ) -> torch.Tensor:
109 top_k, top_p = self.get_top_k_top_p(expanded_idx_mapping, idx_mapping_np)
110 if top_k is None and top_p is None:
111 return logits
112 return apply_top_k_top_p(logits, top_k, top_p)
113
114 def any_greedy(self, idx_mapping_np: np.ndarray) -> bool:
115 return bool(np.any(self.temperature.np[idx_mapping_np] == 0.0))
116
117 def any_explicit_seed(self, idx_mapping_np: np.ndarray) -> bool:
118 return bool(np.any(self.seeds_set[idx_mapping_np]))
119
120 def max_num_logprobs(self, idx_mapping_np: np.ndarray) -> int:
121 return int(np.max(self.num_logprobs[idx_mapping_np]))
