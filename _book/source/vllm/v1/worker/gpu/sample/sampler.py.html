33class Sampler:
34 def __init__(
35 self,
36 max_num_reqs: int,
37 vocab_size: int,
38 device: torch.device,
39 req_states: RequestState,
40 logprobs_mode: LogprobsMode = "raw_logprobs",
41 num_speculative_tokens: int = 1,
42 use_fp64_gumbel: bool = False,
43 enable_trace_replay: bool = False,
44 reasoning_config: ReasoningConfig | None = None,
45 return_sampling_mask: bool = False,
46 ):
47 self.logprobs_mode = logprobs_mode
48 self.compute_nans = envs.VLLM_COMPUTE_NANS_IN_LOGITS # False by default.
49 self.use_fp64_gumbel = use_fp64_gumbel
50
51 self.req_states = req_states
52 self.sampling_states = SamplingStates(max_num_reqs, vocab_size)
53 self.penalties_state = PenaltiesState(req_states)
54 self.logit_bias_state = LogitBiasState(max_num_reqs, device)
55 self.bad_words_state = BadWordsState(req_states)
56 self.logprob_token_ids_state = LogprobTokenIdsState(max_num_reqs, device)
57 self.thinking_budget_state = ThinkingBudgetState(req_states, reasoning_config)
58 self.trace_replay_state = (
59 TraceReplayState(req_states) if enable_trace_replay else None
60 )
61 self.needs_logits_processing = np.zeros(max_num_reqs, dtype=bool)
62 self.num_speculative_tokens = num_speculative_tokens
63 self.return_sampling_mask = return_sampling_mask
64 self.use_flashinfer = (
65 not return_sampling_mask and flashinfer_sampler_supported()
66 )

68 def add_request(
69 self, req_idx: int, prompt_len: int, sampling_params: SamplingParams
70 ) -> None:
71 self.sampling_states.add_request(req_idx, sampling_params)
72 self.penalties_state.add_request(req_idx, sampling_params)
73 self.logit_bias_state.add_request(req_idx, prompt_len, sampling_params)
74 self.bad_words_state.add_request(req_idx, sampling_params)
75 self.logprob_token_ids_state.add_request(req_idx, sampling_params)
76 self.thinking_budget_state.add_request(req_idx, sampling_params)
77 if self.trace_replay_state is not None:
78 self.trace_replay_state.add_request(req_idx, sampling_params)
79
80 states = self.sampling_states
81 temperature = states.temperature.np[req_idx]
82 self.needs_logits_processing[req_idx] = (
83 self.logit_bias_state.use_logit_bias[req_idx]
84 or self.penalties_state.use_penalty[req_idx]
85 or self.bad_words_state.num_bad_words.np[req_idx] > 0
86 or (
87 self.thinking_budget_state.enabled
88 and self.thinking_budget_state.use_thinking_budget[req_idx]
89 )
90 or (temperature != 0.0 and temperature != 1.0)
91 or states.min_p.np[req_idx] != 0.0
92 or states.top_k.np[req_idx] != states.vocab_size
93 or states.top_p.np[req_idx] != 1.0
94 )
95
96 def apply_staged_writes(self) -> None:
97 self.sampling_states.apply_staged_writes()
98 self.penalties_state.apply_staged_writes()
99 self.logit_bias_state.apply_staged_writes()
100 self.bad_words_state.apply_staged_writes()
101 self.logprob_token_ids_state.apply_staged_writes()
102 self.thinking_budget_state.apply_staged_writes()
103 if self.trace_replay_state is not None:
104 self.trace_replay_state.apply_staged_writes()

106 def get_logprobs_dims(
107 self, idx_mapping_np: np.ndarray, include_token_ids: bool = True
108 ) -> tuple[int, int] | None:
109 """(num_logprobs, max_per_req_token_ids) for the given requests, or
110 None when none of them want logprobs."""
111 max_num_logprobs = self.sampling_states.max_num_logprobs(idx_mapping_np)
112 max_token_ids = (
113 self.logprob_token_ids_state.max_num_token_ids(idx_mapping_np)
114 if include_token_ids
115 else 0
116 )
117 if max_num_logprobs == NO_LOGPROBS and max_token_ids == 0:
118 return None
119 num_logprobs = max_num_logprobs if max_num_logprobs != NO_LOGPROBS else 0
120 return num_logprobs, max_token_ids
121
122 def __call__(
123 self,
124 logits: torch.Tensor,
125 input_batch: InputBatch,
126 ) -> SamplerOutput:
127 expanded_idx_mapping = input_batch.expanded_idx_mapping
128 idx_mapping = input_batch.idx_mapping
129 idx_mapping_np = input_batch.idx_mapping_np
130 cu_num_logits_np = input_batch.cu_num_logits_np
131 expanded_local_pos = input_batch.expanded_local_pos
132 pos = input_batch.positions[input_batch.logits_indices]
133 input_ids = input_batch.input_ids[input_batch.logits_indices]
134
135 # NOTE(woosuk): We intentionally compute num_nans before sampling to make clear
136 # that num_nans is computed before applying penalties and temperature.
137 num_nans = get_num_nans(logits) if self.compute_nans else None
138
139 logprobs_dims = self.get_logprobs_dims(idx_mapping_np)
140
141 sampled, processed_logits = self.sample(
142 logits,
143 expanded_idx_mapping,
144 idx_mapping,
145 idx_mapping_np,
146 pos,
147 input_ids,
148 expanded_local_pos,
149 return_logprobs=logprobs_dims is not None,
150 )
151
152 if self.trace_replay_state is not None:
153 # Overwrite sampled tokens with the replay trace up-front so that
154 # computed logprobs reflect the real distribution of the forced token.
155 self.trace_replay_state.apply_trace(sampled, idx_mapping)
156
157 if logprobs_dims is not None:
158 num_logprobs, max_per_req_token_ids = logprobs_dims
159 if self.logprobs_mode in PROCESSED_LOGPROBS_MODES:
160 logits = processed_logits
161 expanded_logits = logits.shape[0] != idx_mapping_np.shape[0]
162 cu_num_logits = cu_num_logits_np.tolist() if expanded_logits else None
163 logprobs_tensors = compute_topk_scores(
164 logits,
165 num_logprobs,
166 sampled,
167 cu_num_logits,
168 logprob_token_ids_state=self.logprob_token_ids_state,
169 expanded_idx_mapping=input_batch.expanded_idx_mapping,
170 max_per_req_token_ids=max_per_req_token_ids,
171 logits_mode=self.logprobs_mode in ("raw_logits", "processed_logits"),
172 )
173 else:
174 logprobs_tensors = None
175
176 # 1 sampled token per request, except chunked-prefill requests
177 # (seq_len < prefill_len) which aren't done prefilling and produce no
178 # output token. num_rejected is always 0 here (one logit per request).
179 num_sampled, num_rejected = get_num_sampled_and_rejected(
180 input_batch.seq_lens.new_ones(input_batch.num_reqs),
181 input_batch.seq_lens,
182 input_batch.cu_num_logits,
183 input_batch.idx_mapping,
184 self.req_states.prefill_len.gpu,
185 )
186
187 sampling_mask_tensors = None
188 if self.return_sampling_mask:
189 # Size by the validated top_k batch max; wider supports use the bitmask.
190 max_num_kept = int(np.max(self.sampling_states.top_k.np[idx_mapping_np]))
191 sampling_mask_tensors = SamplingMaskTensors.from_logits(
192 processed_logits, num_sampled, max_num_kept
193 )
194
195 # These are GPU tensors.
196 sampler_output = SamplerOutput(
197 # The sampled tokens are expanded to 2D tensor with shape
198 # [num_requests, 1], where each row represents one generated
199 # token per request.
200 sampled_token_ids=sampled.view(-1, 1),
201 logprobs_tensors=logprobs_tensors,
202 num_nans=num_nans,
203 num_sampled=num_sampled,
204 num_rejected=num_rejected,
205 sampling_mask_tensors=sampling_mask_tensors,
206 )
207 return sampler_output
208
209 def apply_sampling_params(
210 self,
211 logits: torch.Tensor,
212 expanded_idx_mapping: torch.Tensor,
213 idx_mapping: torch.Tensor,
214 idx_mapping_np: np.ndarray,
215 pos: torch.Tensor,
216 input_ids: torch.Tensor,
217 expanded_local_pos: torch.Tensor,
218 skip_top_k_top_p: bool = False,
219 ) -> torch.Tensor:
220 if not np.any(self.needs_logits_processing[idx_mapping_np]):
221 return logits
222
223 # Copy logits to a new FP32 tensor.
224 logits = torch.empty_like(logits, dtype=torch.float32).copy_(logits)
225
226 # Apply logit bias (e.g., allowed_token_ids, min_tokens) in place.
227 self.logit_bias_state.apply_logit_bias(
228 logits, expanded_idx_mapping, idx_mapping_np, pos
229 )
230
231 # Apply penalties in place.
232 self.penalties_state.apply_penalties(
233 logits,
234 expanded_idx_mapping,
235 idx_mapping_np,
236 input_ids,
237 expanded_local_pos,
238 )
239
240 # Apply bad words masking in place.
241 self.bad_words_state.apply_bad_words(
242 logits,
243 expanded_idx_mapping,
244 idx_mapping_np,
245 input_ids,
246 expanded_local_pos,
247 )
248
249 # Force the reasoning end marker once a request's thinking budget is
250 # reached; applied before temperature so the forced token is always kept.
251 self.thinking_budget_state.apply(
252 logits,
253 expanded_idx_mapping,
254 idx_mapping,
255 idx_mapping_np,
256 input_ids,
257 expanded_local_pos,
258 )
259
260 # Apply temperature in place.
261 self.sampling_states.apply_temperature(
262 logits, expanded_idx_mapping, idx_mapping_np
263 )
264
265 # Apply min_p in place.
266 self.sampling_states.apply_min_p(logits, expanded_idx_mapping, idx_mapping_np)
267
268 if skip_top_k_top_p:
269 return logits
270
271 # Apply top_k and/or top_p. This might or might not return a new tensor.
272 return self.sampling_states.apply_top_k_top_p(
273 logits, expanded_idx_mapping, idx_mapping_np
274 )

276 def sample(
277 self,
278 logits: torch.Tensor,
279 expanded_idx_mapping: torch.Tensor,
280 idx_mapping: torch.Tensor,
281 idx_mapping_np: np.ndarray,
282 pos: torch.Tensor,
283 input_ids: torch.Tensor,
284 expanded_local_pos: torch.Tensor,
285 return_logprobs: bool = False,
286 ) -> tuple[torch.Tensor, torch.Tensor]:
287 processed_logits = self.apply_sampling_params(
288 logits,
289 expanded_idx_mapping,
290 idx_mapping,
291 idx_mapping_np,
292 pos,
293 input_ids,
294 expanded_local_pos,
295 skip_top_k_top_p=True,
296 )
297 top_k, top_p = self.sampling_states.get_top_k_top_p(
298 expanded_idx_mapping, idx_mapping_np
299 )
300 use_flashinfer = self.use_flashinfer and not (
301 # Don't use FI sampler if no requests use top_k/top_p, if there are
302 # any greedy requests or per-request seeds, or if post-processed
303 # logprobs need to be returned for any requests.
304 (top_k is None and top_p is None)
305 or (return_logprobs and self.logprobs_mode in PROCESSED_LOGPROBS_MODES)
306 or self.sampling_states.any_greedy(idx_mapping_np)
307 or self.sampling_states.any_explicit_seed(idx_mapping_np)
308 )
309
310 return self._sample_random(
311 processed_logits,
312 expanded_idx_mapping,
313 idx_mapping_np,
314 pos,
315 top_k,
316 top_p,
317 use_flashinfer,
318 )

320 def _sample_random(
321 self,
322 processed_logits: torch.Tensor,
323 expanded_idx_mapping: torch.Tensor,
324 idx_mapping_np: np.ndarray,
325 pos: torch.Tensor,
326 top_k: torch.Tensor | None,
327 top_p: torch.Tensor | None,
328 use_flashinfer: bool,
329 ) -> tuple[torch.Tensor, torch.Tensor]:
330 if use_flashinfer:
331 sampled = flashinfer_sample(processed_logits, top_k, top_p).to(torch.int64)
332 else:
333 processed_logits = apply_top_k_top_p(processed_logits, top_k, top_p)
334 sampled = gumbel_sample(
335 processed_logits,
336 expanded_idx_mapping,
337 self.sampling_states.temperature.gpu,
338 self.sampling_states.seeds.gpu,
339 pos,
340 apply_temperature=False,
341 is_drafting=False,
342 use_fp64=self.use_fp64_gumbel,
343 )
344 return sampled, processed_logits
