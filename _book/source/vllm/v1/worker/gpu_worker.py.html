116def maybe_rocm_profiling_fallback(profile_result: MemoryProfilingResult) -> int | None:
117 """Memory in bytes to size the KV cache from when profiling measured a
118 release on ROCm, or None to keep what profiling measured.
119
120 The measurement is the drop in free device memory across startup, so it comes
121 out negative when anything else on the device released memory meanwhile. Torch
122 reserved is process local, so it stands in as a lower bound instead of refusing
123 to start. Kept to ROCm, where the AMD CI groups hit this, following #45490.
124 """
125 if profile_result.total_consumed >= 0 or not current_platform.is_rocm():
126 return None
127
128 baseline = profile_result.before_create
129 after_profile = profile_result.after_profile
130 torch_reserved = max(after_profile.torch_memory - baseline.torch_memory, 0)
131 logger.warning(
132 "Free memory grew by %s GiB during profiling (initial %s GiB, current "
133 "%s GiB). Sizing the KV cache from the %s GiB torch reserved instead of "
134 "the profiling difference.",
135 format_gib(-profile_result.total_consumed),
136 format_gib(baseline.free_memory),
137 format_gib(after_profile.free_memory),
138 format_gib(torch_reserved),
139 )
140 return torch_reserved

153 def __init__(
154 self,
155 tensors: dict[str, torch.Tensor],
156 comm_handles: list[Handle] | None = None,
157 comm_postprocess: list[Callable[[], None]] | None = None,
158 ) -> None:
159 super().__init__(tensors)
160 self._comm_handles = comm_handles
161 self._comm_postprocess = comm_postprocess
162 self._comm_waited = False
163
164 def wait_for_comm(self) -> None:
165 if self._comm_waited:
166 return
167 if self._comm_handles:
168 for handle in self._comm_handles:
169 handle.wait()
170 if self._comm_postprocess:
171 for fn in self._comm_postprocess:
172 fn()
173 self._comm_waited = True
174
175 def __getattribute__(self, name: str):
176 # ensure `.tensors` is ready before use
177 if name == "tensors" and not object.__getattribute__(self, "_comm_waited"):
178 object.__getattribute__(self, "wait_for_comm")()
179 return object.__getattribute__(self, name)

245 def sleep(self, level: int = 1) -> None:
246 torch.accelerator.synchronize()
247 free_bytes_before_sleep = torch.accelerator.get_memory_info()[0]
248
249 # Save the buffers before level 2 sleep
250 if level == 2:
251 model = self.model_runner.model
252 self._sleep_saved_buffers = {
253 name: buffer.cpu().clone() for name, buffer in model.named_buffers()
254 }
255 draft = self.get_draft_model()
256 if draft is not None:
257 self._sleep_saved_draft_buffers = {
258 name: buffer.cpu().clone() for name, buffer in draft.named_buffers()
259 }
260
261 self.sleep_mode_backend.suspend(level)
262 if self.vllm_config.model_config.enable_nccl_comm_suspend:
263 suspend_device_comms()
264
265 torch.accelerator.synchronize()
266 deadline = time.monotonic() + (5.0 if current_platform.is_rocm() else 0)
267 while True:
268 free_bytes_after_sleep, total = torch.accelerator.get_memory_info()
269 freed_bytes = free_bytes_after_sleep - free_bytes_before_sleep
270 if freed_bytes >= 0 or time.monotonic() >= deadline:
271 break
272 time.sleep(0.1)
273
274 used_bytes = total - free_bytes_after_sleep
275 assert freed_bytes >= 0, "Memory usage increased after sleeping."
276 logger.info(
277 "Sleep mode freed %s GiB memory, %s GiB memory is still in use.",
278 format_gib(freed_bytes),
279 format_gib(used_bytes),
280 )

282 def wake_up(self, tags: list[str] | None = None) -> None:
283 self.sleep_mode_backend.resume(tags)
284 if self.vllm_config.model_config.enable_nccl_comm_suspend:
285 resume_device_comms()
286
287 # Restore the buffers after level 2 sleep
288 wake_weights = tags is None or "weights" in tags
289 if wake_weights and len(self._sleep_saved_buffers):
290 model = self.model_runner.model
291 for name, buffer in model.named_buffers():
292 if name in self._sleep_saved_buffers:
293 buffer.data.copy_(self._sleep_saved_buffers[name].data)
294 self._sleep_saved_buffers = {}
295
296 if wake_weights and len(self._sleep_saved_draft_buffers):
297 draft = self.get_draft_model()
298 if draft is not None:
299 for name, buffer in draft.named_buffers():
300 if name in self._sleep_saved_draft_buffers:
301 buffer.data.copy_(self._sleep_saved_draft_buffers[name].data)
302 self._sleep_saved_draft_buffers = {}
303
304 self.synchronize_device()

335 @contextmanager
336 def _scoped_allocator_max_split(self, max_split_size_mb: int):
337 """Temporarily set max_split_size_mb to reduce allocator fragmentation at the
338 cost of more cudaMalloc calls (negligible in practice). Restores the original
339 value on exit."""
340 if not current_platform.is_cuda():
341 yield
342 return
343
344 conf = os.environ.get("PYTORCH_CUDA_ALLOC_CONF", "")
345 match = re.search(r"max_split_size_mb:(\d+)", conf)
346 original_value = match.group(1) if match else None
347
348 torch._C._accelerator_setAllocatorSettings(
349 f"max_split_size_mb:{max_split_size_mb}"
350 )
351 try:
352 yield
353 finally:
354 # PyTorch defaults to SIZE_MAX (no limit).
355 _SIZE_MAX_MB = (2**64 - 1) // (1024 * 1024)
356 restore = original_value if original_value else str(_SIZE_MAX_MB)
357 torch._C._accelerator_setAllocatorSettings(f"max_split_size_mb:{restore}")

359 @instrument(span_name="Init device")
360 def init_device(self):
361 if self.device_config.device_type == "cuda":
362 # This env var set by Ray causes exceptions with graph building.
363 os.environ.pop("NCCL_ASYNC_ERROR_HANDLING", None)
364 parallel_config = self.parallel_config
365 if (
366 parallel_config.distributed_executor_backend
367 not in ("ray", "external_launcher")
368 and parallel_config.data_parallel_backend != "ray"
369 and parallel_config.nnodes_within_dp == 1
370 ):
371 # Use local DP rank if available, otherwise use global DP rank.
372 dp_local_rank = self.parallel_config.data_parallel_rank_local
373 if dp_local_rank is None:
374 dp_local_rank = self.parallel_config.data_parallel_index
375
376 tp_pp_world_size = (
377 self.parallel_config.pipeline_parallel_size
378 * self.parallel_config.tensor_parallel_size
379 )
380
381 # DP_LOCAL_RANK * TP_PP_WORLD_SIZE + TP_LOCAL_RANK
382 self.local_rank += dp_local_rank * tp_pp_world_size
383
384 # Publish the logical-to-physical mapping for topology queries
385 # such as NIC affinity and P2P checks.
386 assigned_physical_gpu_ids = parallel_config.assigned_physical_gpu_ids
387 if assigned_physical_gpu_ids is not None:
388 from vllm.platforms.interface import set_assigned_physical_gpu_ids
389
390 set_assigned_physical_gpu_ids(assigned_physical_gpu_ids)
391 assert self.local_rank < len(assigned_physical_gpu_ids), (
392 f"local_rank {self.local_rank} is out of bounds for "
393 f"assigned_physical_gpu_ids {assigned_physical_gpu_ids}"
394 )
395 # NOTE(patch pr45026): local_world_size is derived from
396 # parallel_config.nnodes, which is only set for the "mp"
397 # multi-node backend. With the "ray"/"external_launcher"
398 # backends nnodes stays 1, so local_world_size collapses to
399 # the full world_size and this check wrongly fires on
400 # cross-node deployments. assigned_physical_gpu_ids is already
401 # per-node and the local_rank bound above fully validates the
402 # mapping for these backends, so skip the check for them.
403 if parallel_config.distributed_executor_backend not in (
404 "ray",
405 "external_launcher",
406 ):
407 assert self.parallel_config.local_world_size <= len(
408 assigned_physical_gpu_ids
409 ), (
410 f"local_world_size ({self.parallel_config.local_world_size})"
411 " exceeds assigned_physical_gpu_ids count "
412 f"({len(assigned_physical_gpu_ids)})"
413 )
414 else:
415 assert self.local_rank < torch.accelerator.device_count(), (
416 f"DP adjusted local rank {self.local_rank} is out of "
417 f"bounds for {torch.accelerator.device_count()} devices."
418 )
419
420 visible_device_index = (
421 current_platform.logical_device_id_to_visible_device_id(self.local_rank)
422 )
423 self.device = torch.device(f"cuda:{visible_device_index}")
424 torch.accelerator.set_device_index(self.device)
425
426 current_platform.check_if_supports_dtype(self.model_config.dtype)

527 @torch.inference_mode()
528 def determine_available_memory(self) -> int:
529 """Profiles the peak memory usage of the model to determine how much
530 memory can be used for KV cache without OOMs.
531
532 The engine will first conduct a profiling of the existing memory usage.
533 Then, it calculates the free memory that can be used for KV cache in
534 bytes.
535
536 Tip:
537 You may limit the usage of GPU memory
538 by adjusting the `gpu_memory_utilization` parameter.
539 """
540 maybe_apply_startup_plan(self)
541
542 if kv_cache_memory_bytes := self.cache_config.kv_cache_memory_bytes:
543 # still need a profile run which compiles the model for
544 # max_num_batched_tokens
545 self.model_runner.profile_run()
546
547 msg = (
548 f"Initial free memory {format_gib(self.init_snapshot.free_memory)} "
549 f"GiB, reserved {format_gib(kv_cache_memory_bytes)} GiB memory for "
550 "KV Cache as specified by kv_cache_memory_bytes config and "
551 "skipped memory profiling. This does not respect the "
552 "gpu_memory_utilization config. Only use kv_cache_memory_bytes "
553 "config when you want manual control of KV cache memory "
554 "size. If OOM'ed, check the difference of initial free "
555 "memory between the current run and the previous run "
556 "where kv_cache_memory_bytes is suggested and update it "
557 "correspondingly."
558 )
559 logger.info(msg)
560 return reserve_mm_ipc_gpu_memory(
561 kv_cache_memory_bytes,
562 self.model_config.multimodal_config,
563 getattr(self.parallel_config, "_api_process_count", 1),
564 )
565
566 # Execute a forward pass with dummy inputs to profile the memory usage
567 # of the model.
568 with memory_profiling(
569 self.init_snapshot,
570 weights_memory=int(self.model_runner.model_memory_usage),
571 ) as profile_result:
572 self.model_runner.profile_run()
573
574 # Profile CUDA graph memory if graphs will be captured.
575 # ROCm is included: #44825 moved the profiler to
576 # torch.accelerator.get_memory_info (reliable on ROCm, as used by
577 # the AMD-CI mem tests), and graph_pool_handle resolves to the same
578 # torch.cuda handle the live capture path already uses on ROCm.
579 # XPU stays excluded (see #39977).
580 cudagraph_memory_estimate = 0
581 if (
582 current_platform.is_cuda_alike()
583 and self.vllm_config.compilation_config.cudagraph_mode != CUDAGraphMode.NONE
584 ):
585 cudagraph_memory_estimate = self.model_runner.profile_cudagraph_memory()
586
587 # Respect the opt-in flag as originally designed.
588 cudagraph_memory_estimate_applied = (
589 cudagraph_memory_estimate
590 if envs.VLLM_MEMORY_PROFILER_ESTIMATE_CUDAGRAPHS
591 else 0
592 )
593
594 init_free_memory = self.init_snapshot.free_memory
595 free_gpu_memory = profile_result.after_profile.free_memory
596 rocm_fallback = maybe_rocm_profiling_fallback(profile_result)
597 if rocm_fallback is None:
598 # NOTE(woosuk): Here we assume that the other processes using the same
599 # GPU did not change their memory usage during the profiling.
600 assert init_free_memory >= free_gpu_memory, (
601 "Error in memory profiling. "
602 f"Initial free memory {format_gib(init_free_memory)} GiB, "
603 f"current free memory {format_gib(free_gpu_memory)} GiB. "
604 "This happens when other processes sharing the same container "
605 "release GPU memory while vLLM is profiling during initialization. "
606 "To fix this, ensure consistent GPU memory allocation or "
607 "isolate vLLM in its own container."
608 )
609 else:
610 profile_result.total_consumed = rocm_fallback
611 profile_result.non_kv_cache_memory = (
612 profile_result.total_consumed + profile_result.transient_peak_headroom
613 )
614
615 self.total_consumed = profile_result.total_consumed
616 self.peak_activation_memory = (
617 profile_result.transient_peak_headroom + cudagraph_memory_estimate_applied
618 )
619 self.cudagraph_memory_estimate = cudagraph_memory_estimate
620
621 self.available_kv_cache_memory_bytes = (
622 self.requested_memory
623 - profile_result.non_kv_cache_memory
624 - cudagraph_memory_estimate_applied
625 )
626
627 unrequested_memory = self.init_snapshot.free_memory - self.requested_memory
628 logger.debug(
629 "Initial free memory: %s GiB; Requested memory: %f (util), %s GiB",
630 format_gib(self.init_snapshot.free_memory),
631 self.cache_config.gpu_memory_utilization,
632 format_gib(self.requested_memory),
633 )
634 logger.debug(
635 "Free memory after profiling: %s GiB (total), %s GiB (within requested)",
636 format_gib(free_gpu_memory),
637 format_gib(free_gpu_memory - unrequested_memory),
638 )
639 logger.debug(profile_result)
640 logger.info_once(
641 "Available KV cache memory: %s GiB",
642 format_gib(self.available_kv_cache_memory_bytes),
643 )
644
645 if cudagraph_memory_estimate > 0:
646 total_mem = self.init_snapshot.total_memory
647 current_util = self.cache_config.gpu_memory_utilization
648 cg_util_delta = cudagraph_memory_estimate / total_mem
649 if envs.VLLM_MEMORY_PROFILER_ESTIMATE_CUDAGRAPHS:
650 equiv_util = round(current_util - cg_util_delta, 4)
651 suggested_util = min(
652 round(current_util + cg_util_delta, 4),
653 1.0,
654 )
655 logger.info(
656 "CUDA graph memory profiling is enabled (default since "
657 "v0.21.0). The current --gpu-memory-utilization=%.4f is "
658 "equivalent to --gpu-memory-utilization=%.4f without "
659 "CUDA graph memory profiling. To maintain the same "
660 "effective KV cache size as before, increase "
661 "--gpu-memory-utilization to %.4f. To disable, set "
662 "VLLM_MEMORY_PROFILER_ESTIMATE_CUDAGRAPHS=0.",
663 current_util,
664 equiv_util,
665 suggested_util,
666 )
667 else:
668 suggested_util = min(
669 round(current_util + cg_util_delta, 4),
670 1.0,
671 )
672 logger.warning(
673 "CUDA graph memory profiling is disabled "
674 "(VLLM_MEMORY_PROFILER_ESTIMATE_CUDAGRAPHS=0). "
675 "Without it, CUDA graph memory is not accounted for "
676 "during KV cache allocation, which may require lowering "
677 "--gpu-memory-utilization to avoid OOM. Consider "
678 "re-enabling it (the default as of v0.21.0) and increasing "
679 "--gpu-memory-utilization from %.4f to %.4f.",
680 current_util,
681 suggested_util,
682 )

685 int(self.available_kv_cache_memory_bytes),
686 self.model_config.multimodal_config,
687 getattr(self.parallel_config, "_api_process_count", 1),

772 @instrument(span_name="Warmup (GPU)")
773 def compile_or_warm_up_model(self) -> CompilationTimes:
774 warmup_sizes: list[int] = []
775
776 if self.vllm_config.compilation_config.mode == CompilationMode.VLLM_COMPILE:
777 # warm up sizes that are not in cudagraph capture sizes,
778 # but users still want to compile for better performance,
779 # e.g. for the max-num-batched token size in chunked prefill.
780 compile_sizes = self.vllm_config.compilation_config.compile_sizes
781 warmup_sizes = compile_sizes.copy() if compile_sizes is not None else [] # type: ignore[assignment]
782 cg_capture_sizes: list[int] = []
783
784 if self.vllm_config.compilation_config.cudagraph_mode != CUDAGraphMode.NONE:
785 cg_sizes = self.vllm_config.compilation_config.cudagraph_capture_sizes
786 cg_capture_sizes = [] if cg_sizes is None else cg_sizes
787 warmup_sizes = [x for x in warmup_sizes if x not in cg_capture_sizes]
788
789 compile_ranges = self.vllm_config.compilation_config.get_compile_ranges()
790 # For each compile_range, if none of the batch sizes
791 # in warmup_sizes or cudagraph_capture_sizes are in the range,
792 # add the end of the range to ensure compilation/warmup.
793 all_sizes = set(cg_capture_sizes)
794 all_sizes.update([x for x in warmup_sizes if isinstance(x, int)])
795 for compile_range in compile_ranges:
796 if not any(x in compile_range for x in all_sizes):
797 warmup_sizes.append(compile_range.end)
798
799 # We skip EPLB here since we don't want to record dummy metrics
800 for size in sorted(warmup_sizes, reverse=True):
801 logger.info("Compile and warming up model for size %d", size)
802 self.model_runner._dummy_run(size, skip_eplb=True, remove_lora=False)
803 self.model_runner.maybe_remove_all_loras(self.model_runner.lora_config)
804
805 # Warmup and tune the kernels used during model execution before
806 # cuda graph capture.
807 kernel_warmup(self)
808
809 if self.use_v2_model_runner:
810 # A workspace resize after capture frees what the graphs point at.
811 warmup_kernels(self.model_runner, self.execute_model, self.sample_tokens)
812
813 cuda_graph_memory_bytes = 0
814 if not self.model_config.enforce_eager:
815 with self._get_cudagraph_capture_context():
816 cuda_graph_memory_bytes = self.model_runner.capture_model()
817
818 # Compare actual vs estimated CUDA graph memory (if we did profiling)
819 if (
820 hasattr(self, "cudagraph_memory_estimate")
821 and self.cudagraph_memory_estimate > 0
822 ):
823 GiB = lambda b: round(b / GiB_bytes, 2)
824 diff = abs(cuda_graph_memory_bytes - self.cudagraph_memory_estimate)
825 logger.info(
826 "CUDA graph pool memory: %s GiB (actual), %s GiB (estimated), "
827 "difference: %s GiB (%.1f%%).",
828 GiB(cuda_graph_memory_bytes),
829 GiB(self.cudagraph_memory_estimate),
830 GiB(diff),
831 100 * diff / max(cuda_graph_memory_bytes, 1),
832 )
833
834 if self.cache_config.kv_cache_memory_bytes is None and hasattr(
835 self, "peak_activation_memory"
836 ):
837 # Suggests optimal kv cache memory size if we rely on
838 # memory_profiling to guess the kv cache memory size which
839 # provides peak_activation_memory and a few other memory
840 # consumption. `memory_profiling` does not consider
841 # CUDAGraph memory size and may not utilize all gpu memory.
842 # Users may want fine-grained control to specify kv cache
843 # memory size.
844
845 # empirically observed that the memory profiling may
846 # slightly underestimate the memory consumption.
847 # So leave a small buffer (=150MiB) to avoid OOM.
848 redundancy_buffer_memory = 150 * (1 << 20)
849
850 non_kv_cache_memory = (
851 self.total_consumed
852 + self.peak_activation_memory
853 + cuda_graph_memory_bytes
854 )
855 kv_cache_memory_bytes_to_gpu_limit = (
856 self.init_snapshot.free_memory
857 - non_kv_cache_memory
858 - redundancy_buffer_memory
859 )
860 kv_cache_memory_bytes_to_requested_limit = (
861 int(self.requested_memory)
862 - non_kv_cache_memory
863 - redundancy_buffer_memory
864 )
865
866 msg = (
867 f"Free memory on device "
868 f"({format_gib(self.init_snapshot.free_memory)}/"
869 f"{format_gib(self.init_snapshot.total_memory)} GiB) on startup. "
870 f"Desired GPU memory utilization is "
871 f"({self.cache_config.gpu_memory_utilization}, "
872 f"{format_gib(self.requested_memory)} GiB). "
873 f"Actual usage is {format_gib(self.total_consumed)} "
874 f"GiB for consumed memory (weights + non-torch), "
875 f"{format_gib(self.peak_activation_memory)} GiB "
876 f"for peak activation, and {format_gib(cuda_graph_memory_bytes)} "
877 f"GiB for CUDAGraph memory. Replace gpu_memory_utilization "
878 f"config with `--kv-cache-memory="
879 f"{kv_cache_memory_bytes_to_requested_limit}` "
880 f"({format_gib(kv_cache_memory_bytes_to_requested_limit)} GiB) to fit "
881 f"into requested memory, or `--kv-cache-memory="
882 f"{kv_cache_memory_bytes_to_gpu_limit}` "
883 f"({format_gib(kv_cache_memory_bytes_to_gpu_limit)} GiB) to fully "
884 f"utilize gpu memory. Current kv cache memory in use is "
885 f"{format_gib(self.available_kv_cache_memory_bytes)} GiB."
886 )
887
888 logger.info(msg)
889
890 maybe_save_startup_plan(self, kv_cache_memory_bytes_to_requested_limit)
891
892 if not self.use_v2_model_runner and get_pp_group().is_last_rank:
893 # V1: Warm up sampler and preallocate memory buffer for logits and other
894 # sampling related tensors of max possible shape to avoid memory
895 # fragmentation issue.
896 # NOTE: This is called after `capture_model` on purpose to prevent
897 # memory buffers from being cleared by `torch.accelerator.empty_cache`.
898 max_num_reqs = min(
899 self.scheduler_config.max_num_seqs,
900 self.scheduler_config.max_num_batched_tokens,
901 )
902
903 # We skip EPLB here since we don't want to record dummy metrics
904 hidden_states, last_hidden_states = self.model_runner._dummy_run(
905 num_tokens=max_num_reqs,
906 skip_eplb=True,
907 cudagraph_runtime_mode=CUDAGraphMode.NONE,
908 )
909 if self.model_runner.is_pooling_model:
910 self.model_runner._dummy_pooler_run(hidden_states)
911 else:
912 self.model_runner._dummy_sampler_run(hidden_states=last_hidden_states)
913
914 # Reset the seed to ensure that the random state is not affected by
915 # the model initialization and profiling.
916 set_random_seed(self.model_config.seed)
917
918 # Eagerly trigger inductor's once-per-process lazy inits during
919 # warmup (rather than on a later compile cache-miss at runtime).
920 c_config = self.compilation_config
921 if c_config.mode != CompilationMode.NONE and c_config.backend == "inductor":
922 from vllm.compilation.compiler_interface import (
923 trigger_inductor_lazy_init,
924 )
925
926 trigger_inductor_lazy_init(self.device)
927
928 # All warmup is done — start monitoring for unexpected JIT
929 # compilations that would cause latency spikes during inference.
930 from vllm.utils.jit_monitor import activate as activate_jit_monitor
931
932 activate_jit_monitor(
933 mode=self.observability_config.jit_monitor_mode,
934 verbose=self.observability_config.jit_monitor_verbose,
935 )
936
937 # Freeze the worker heap so the GC won't scan static objects
938 # (model weights, KV caches, CUDA graphs) during inference.
939 freeze_gc_heap()
940 maybe_attach_gc_debug_callback()
941
942 # Warmup / first-compile is done — activate the `VLLM_GPU_SYNC_CHECK`
943 # gate so subsequent `execute_model` / `sample_tokens` calls enforce it.
944 enable_gpu_sync_check()
945
946 # Startup is done; steady-state serving gets no benefit from torch
947 # intra-op parallelism.
948 set_torch_threads_for_runtime()
949
950 return CompilationTimes(
951 language_model=self.compilation_config.compilation_time,
952 encoder=self.compilation_config.encoder_compilation_time,
953 )

1027 def annotate_profile(self, scheduler_output):
1028 # add trace annotation so that we can easily distinguish
1029 # context/generation request numbers in each iteration.
1030 # A context request is a request that has not yet generated any tokens
1031 if not self.profiler:
1032 return nullcontext()
1033
1034 self.profiler.step()
1035 if not self.profiler.is_running:
1036 return nullcontext()
1037
1038 iteration_details = compute_iteration_details(scheduler_output)
1039
1040 if self.vllm_config.profiler_config.detailed_trace_annotation:
1041 # Compute roofline-model metrics per request, split by phase
1042 # (context vs generation). These help estimate compute and
1043 # memory intensity from the trace.
1044 #
1045 # Per-request quantities:
1046 # query_len = number of scheduled (new) tokens for this request
1047 # seq_len = total sequence length (computed + scheduled tokens)
1048 #
1049 # Aggregated across requests in each phase
1050 # (ctx_=context, gen_=generation):
1051 # seq_len_sum = sum of seq_len (total KV length)
1052 # qq_compute = sum of query_len*query_len
1053 # (proxy for QK^T compute cost)
1054 # qk_compute = sum of query_len*seq_len
1055 # (proxy for QK^T compute cost for decode and
1056 # chunked prefill)
1057 # total_scheduled_tokens = scheduled tokens across all requests
1058 ctx_seq_len_sum = 0
1059 ctx_qq_compute = 0
1060 ctx_qk_compute = 0
1061 gen_seq_len_sum = 0
1062 gen_qq_compute = 0
1063 gen_qk_compute = 0
1064 total_scheduled_tokens = 0
1065
1066 # Build a map of req_id -> num_computed_tokens for all requests
1067 new_req_ids = {
1068 new_req.req_id for new_req in scheduler_output.scheduled_new_reqs
1069 }
1070 num_computed_tokens_ids = {
1071 new_req.req_id: new_req.num_computed_tokens
1072 for new_req in scheduler_output.scheduled_new_reqs
1073 }
1074 for req_id, num_computed_tokens in zip(
1075 scheduler_output.scheduled_cached_reqs.req_ids,
1076 scheduler_output.scheduled_cached_reqs.num_computed_tokens,
1077 ):
1078 num_computed_tokens_ids[req_id] = num_computed_tokens
1079
1080 # Accumulate per-phase metrics
1081 for req_id, num_tokens in scheduler_output.num_scheduled_tokens.items():
1082 query_len = num_tokens
1083 total_scheduled_tokens += query_len
1084 seq_len = num_computed_tokens_ids.get(req_id, 0) + query_len
1085 if (
1086 scheduler_output.scheduled_cached_reqs.is_context_phase(req_id)
1087 or req_id in new_req_ids
1088 ):
1089 ctx_seq_len_sum += seq_len
1090 ctx_qq_compute += query_len * query_len
1091 ctx_qk_compute += query_len * seq_len
1092 else:
1093 gen_seq_len_sum += seq_len
1094 gen_qq_compute += query_len * query_len
1095 gen_qk_compute += query_len * seq_len
1096 annotation = "".join(
1097 [
1098 "execute_",
1099 str(total_scheduled_tokens),
1100 "_context_",
1101 str(iteration_details.num_ctx_requests),
1102 "(sq",
1103 str(iteration_details.num_ctx_tokens),
1104 "sk",
1105 str(ctx_seq_len_sum),
1106 "sqsq",
1107 str(ctx_qq_compute),
1108 "sqsk",
1109 str(ctx_qk_compute),
1110 ")_generation_",
1111 str(iteration_details.num_generation_requests),
1112 "(sq",
1113 str(iteration_details.num_generation_tokens),
1114 "sk",
1115 str(gen_seq_len_sum),
1116 "sqsq",
1117 str(gen_qq_compute),
1118 "sqsk",
1119 str(gen_qk_compute),
1120 ")",
1121 ]
1122 )
1123 else:
1124 annotation = "".join(
1125 [
1126 "execute_context_",
1127 str(iteration_details.num_ctx_requests),
1128 "(",
1129 str(iteration_details.num_ctx_tokens),
1130 ")",
1131 "_generation_",
1132 str(iteration_details.num_generation_requests),
1133 "(",
1134 str(iteration_details.num_generation_tokens),
1135 ")",
1136 ]
1137 )
1138 return self.profiler.annotate_context_manager(annotation)

1147 @torch.inference_mode()
1148 @with_gpu_sync_check
1149 def execute_model(
1150 self, scheduler_output: "SchedulerOutput"
1151 ) -> ModelRunnerOutput | AsyncModelRunnerOutput | None:
1152 # Wait for the previous step's sends so this forward pass cannot
1153 # overwrite buffers they are still reading.
1154 if self._pp_send_work:
1155 for handle in self._pp_send_work:
1156 handle.wait()
1157 self._pp_send_work = []
1158
1159 intermediate_tensors = None
1160 forward_pass = scheduler_output.total_num_scheduled_tokens > 0
1161 num_scheduled_tokens = scheduler_output.total_num_scheduled_tokens
1162 all_gather_tensors = {}
1163 compilation_config = self.vllm_config.compilation_config
1164 parallel_config = self.vllm_config.parallel_config
1165
1166 if (
1167 parallel_config.pipeline_parallel_size > 1
1168 and compilation_config.pass_config.enable_sp
1169 and forward_pass
1170 ):
1171 # currently only supported by V1 GPUModelRunner
1172 assert not self.use_v2_model_runner
1173 num_scheduled_tokens_np = np.array(
1174 list(scheduler_output.num_scheduled_tokens.values()),
1175 dtype=np.int32,
1176 )
1177 # TODO(lucas): This is pretty gross; ideally we should only ever call
1178 # `_determine_batch_execution_and_padding` once (will get called again
1179 # in `execute_model`) but this requires a larger refactor of PP.
1180 _, batch_desc, _, _, _ = (
1181 self.model_runner._determine_batch_execution_and_padding(
1182 num_tokens=num_scheduled_tokens,
1183 num_reqs=len(num_scheduled_tokens_np),
1184 num_scheduled_tokens_np=num_scheduled_tokens_np,
1185 max_num_scheduled_tokens=num_scheduled_tokens_np.max(),
1186 use_cascade_attn=False, # TODO(lucas): Handle cascade attention
1187 )
1188 )
1189 all_gather_tensors = {
1190 "residual": not is_residual_scattered_for_sp(
1191 self.vllm_config, batch_desc.num_tokens
1192 )
1193 }
1194
1195 if forward_pass and not get_pp_group().is_first_rank:
1196 tensor_dict, comm_handles, comm_postprocess = (
1197 get_pp_group().irecv_tensor_dict(
1198 all_gather_group=get_tp_group(),
1199 all_gather_tensors=all_gather_tensors,
1200 )
1201 )
1202 assert tensor_dict is not None
1203 intermediate_tensors = AsyncIntermediateTensors(
1204 tensor_dict,
1205 comm_handles=comm_handles,
1206 comm_postprocess=comm_postprocess,
1207 )
1208
1209 with self.annotate_profile(scheduler_output):
1210 output = self.model_runner.execute_model(
1211 scheduler_output, intermediate_tensors
1212 )
1213 if (
1214 self.use_v2_model_runner
1215 and self.model_runner.is_pooling_model
1216 and output is None
1217 ):
1218 output = self.model_runner.pool() # type: ignore
1219 if isinstance(
1220 output, ModelRunnerOutput | AsyncModelRunnerOutput | NoneType
1221 ):
1222 return output
1223
1224 assert isinstance(output, IntermediateTensors)
1225 parallel_config = self.vllm_config.parallel_config
1226 assert (
1227 parallel_config.distributed_executor_backend != "external_launcher"
1228 and not get_pp_group().is_last_rank
1229 )
1230
1231 # Non-blocking send of the intermediate tensors. The metadata handle
1232 # is reaped lazily by the GroupCoordinator; the device handles are
1233 # waited at the top of the next step.
1234 handles = get_pp_group().isend_tensor_dict(
1235 output.tensors,
1236 all_gather_group=get_tp_group(),
1237 all_gather_tensors=all_gather_tensors,
1238 )
1239 self._pp_send_work = handles[1:]
1240
1241 if self.use_v2_model_runner and self.model_runner.is_pooling_model:
1242 return self.model_runner.pool() # type: ignore
1243 return None

1248 def profile(self, is_start: bool = True, profile_prefix: str | None = None):
1249 # Check if profiling is enabled
1250 if self.profiler_config is None or self.profiler_config.profiler is None:
1251 raise RuntimeError(
1252 "Profiling is not enabled. Please set --profiler-config to enable "
1253 "profiling. Example: "
1254 "'--profiler-config.profiler=torch --profiler-config.torch_profiler_dir"
1255 "=YOUR_DIR_PATH_TO_DUMP_TRACE'"
1256 )
1257
1258 if is_start:
1259 profiler_type = self.profiler_config.profiler
1260 # Generate the trace name by combining prefix with comprehensive rank suffix
1261 from vllm.distributed.utils import get_worker_rank_suffix
1262
1263 rank_suffix = get_worker_rank_suffix(global_rank=self.rank)
1264
1265 # Build the full trace name
1266 if profile_prefix:
1267 trace_name = f"{profile_prefix}_{rank_suffix}"
1268 else:
1269 trace_name = rank_suffix
1270
1271 if profiler_type == "proton" and self.profiler is not None:
1272 self.profiler.set_output_name(trace_name)
1273
1274 # Create the profiler wrapper only on the first start call
1275 if self.profiler is None:
1276 if profiler_type == "torch":
1277 self.profiler = TorchProfilerWrapper(
1278 self.profiler_config,
1279 worker_name=trace_name,
1280 local_rank=self.local_rank,
1281 activities=["CPU", "CUDA"],
1282 )
1283 logger.debug(
1284 "Starting torch profiler with trace name: %s", trace_name
1285 )
1286 elif profiler_type == "cuda":
1287 self.profiler = CudaProfilerWrapper(self.profiler_config)
1288 logger.debug("Starting CUDA profiler")
1289 elif profiler_type == "proton":
1290 self.profiler = ProtonProfilerWrapper(
1291 self.profiler_config, worker_name=trace_name
1292 )
1293 logger.debug(
1294 "Starting Proton profiler with trace name: %s", trace_name
1295 )
1296 else:
1297 # Config validation should prevent this code being reached
1298 raise ValueError(
1299 f"Invalid profiler value of {self.profiler_config.profiler}"
1300 )
1301
1302 self.profiler.start()
1303 else:
1304 if self.profiler is None:
1305 logger.warning("Profiler was not started, nothing to stop.")
1306 return
1307 try:
1308 self.profiler.stop()
1309 finally:
1310 if self.profiler_config.profiler == "proton" and not (
1311 self.profiler.has_cuda_graph_session
1312 ):
1313 # Proton output names are fixed when the wrapper is constructed.
1314 # Recreate it so the next profile_prefix is honored.
1315 self.profiler = None

1399 def _start_weight_update(self, is_draft: bool = False) -> None:
1400 self._check_weight_transfer_engine()
1401 assert self.weight_transfer_engine is not None
1402
1403 if is_draft and not self.weight_transfer_engine.supports_draft_weight_update:
1404 raise RuntimeError(
1405 f"{type(self.weight_transfer_engine).__name__} does not support "
1406 "draft model weight updates."
1407 )
1408
1409 if self._weight_update_active:
1410 raise RuntimeError(
1411 "start_weight_update called while a weight update is already "
1412 "active. Call finish_weight_update first."
1413 )
1414
1415 try:
1416 if is_draft:
1417 self._set_draft_weight_update_target()
1418 self.weight_transfer_engine.start_weight_update()
1419 except BaseException:
1420 self.weight_transfer_engine.reset_weight_update_target()
1421 raise
1422 self._weight_update_active = True
1423 self._weight_update_is_draft = is_draft

1425 def update_weights(self, update_info: dict | list[dict]) -> None:
1426 """
1427 Receive one weight update chunk from the trainer.
1428
1429 start_weight_update must be called before update_weights and
1430 finish_weight_update must be called after all chunks have been sent.
1431 Every chunk loads into whichever model the session's start_weight_update
1432 / start_draft_weight_update call selected.
1433
1434 Args:
1435 update_info: Backend-specific update info, or a list indexed by
1436 global worker rank across data parallel replicas.
1437 """
1438 self._check_weight_transfer_engine()
1439 assert self.weight_transfer_engine is not None
1440
1441 if not self._weight_update_active:
1442 raise RuntimeError(
1443 "start_weight_update must be called before update_weights."
1444 )
1445
1446 with set_current_vllm_config(self.vllm_config):
1447 try:
1448 if isinstance(update_info, list):
1449 parallel_config = self.vllm_config.parallel_config
1450 local_update_info = update_info[
1451 parallel_config.data_parallel_rank * parallel_config.world_size
1452 + self.rank
1453 ]
1454 else:
1455 local_update_info = update_info
1456 self.weight_transfer_engine.update_weights(local_update_info)
1457 except BaseException:
1458 self._weight_update_active = False
1459 self.weight_transfer_engine.reset_weight_update_target()
1460 raise
1461
1462 def finish_weight_update(self) -> None:
1463 """Finish the current weight update session."""
1464 self._check_weight_transfer_engine()
1465 assert self.weight_transfer_engine is not None
1466
1467 if not self._weight_update_active:
1468 raise RuntimeError(
1469 "finish_weight_update called without a matching start_weight_update."
1470 )
1471
1472 with set_current_vllm_config(self.vllm_config):
1473 self.weight_transfer_engine.finish_weight_update()
1474 self.weight_transfer_engine.reset_weight_update_target()
1475 self._weight_update_active = False
1476
1477 # Weight transfer bypasses GPUModelRunner.reload_weights().
1478 if not self._weight_update_is_draft:
1479 self.model_runner.reset_lora_state()
