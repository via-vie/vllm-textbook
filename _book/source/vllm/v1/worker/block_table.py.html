29def get_block_table_width(
30 max_num_blocks: int,
31 block_size: int,
32 kernel_block_size: int | None = None,
33 *,
34 token_alignment: int | None = 128,
35) -> int:
36 """Return the width after optional alignment and virtual block splitting."""
37 if kernel_block_size is None:
38 kernel_block_size = block_size
39 if block_size % kernel_block_size != 0:
40 raise ValueError(
41 f"kernel_block_size {kernel_block_size} must divide "
42 f"block_size {block_size} evenly"
43 )
44 if token_alignment is not None:
45 if token_alignment <= 0:
46 raise ValueError("token_alignment must be positive")
47 block_alignment = token_alignment // math.gcd(token_alignment, block_size)
48 max_num_blocks = cdiv(max_num_blocks, block_alignment) * block_alignment
49 return max_num_blocks * block_size // kernel_block_size

57class BlockTable:
58 def __init__(
59 self,
60 block_size: int,
61 max_num_reqs: int,
62 max_num_blocks_per_req: int,
63 max_num_batched_tokens: int,
64 pin_memory: bool,
65 device: torch.device,
66 kernel_block_size: int,
67 cp_kv_cache_interleave_size: int,
68 slot_mapping_mode: SlotMappingMode = SlotMappingMode.TOKEN_TO_KV_SLOT,
69 ):
70 """
71 Args:
72 block_size: Block size used for KV cache memory allocation
73 max_num_reqs: Maximum number of concurrent requests supported.
74 max_num_blocks_per_req: Maximum number of blocks per request.
75 max_num_batched_tokens: Maximum number of tokens in a batch.
76 pin_memory: Whether to pin memory for faster GPU transfers.
77 device: Target device for the block table.
78 kernel_block_size: The block_size of underlying attention kernel.
79 Will be the same as `block_size` if `block_size` is supported
80 by the attention kernel.
81 slot_mapping_mode: How this cache group maps scheduled tokens to
82 cache slots. Mamba-like state caches do not use token slot
83 mappings and should use SlotMappingMode.NONE.
84 """
85 self.max_num_reqs = max_num_reqs
86 self.max_num_batched_tokens = max_num_batched_tokens
87 self.pin_memory = pin_memory
88 self.device = device
89 self.kv_cache_block_size = block_size
90
91 if kernel_block_size == block_size:
92 # Standard case: allocation and computation use same block size
93 # No block splitting needed, direct mapping
94 self.block_size = block_size
95 self.blocks_per_kv_block = 1
96 self.use_hybrid_blocks = False
97 else:
98 # Hybrid case: allocation block size differs from kernel block size
99 # Memory blocks are subdivided to match kernel requirements
100 # Example: 32-token memory blocks with 16-token kernel blocks
101 # → Each memory block corresponds to 2 kernel blocks
102 if block_size % kernel_block_size != 0:
103 raise ValueError(
104 f"kernel_block_size {kernel_block_size} must divide "
105 f"kv_manager_block_size size {block_size} evenly"
106 )
107
108 self.block_size = kernel_block_size
109 self.blocks_per_kv_block = block_size // kernel_block_size
110 self.use_hybrid_blocks = True
111
112 self.max_num_blocks_per_req = max_num_blocks_per_req * self.blocks_per_kv_block
113
114 self.block_table = self._make_buffer(
115 self.max_num_reqs, self.max_num_blocks_per_req, dtype=torch.int32
116 )
117 self.num_blocks_per_row = np.zeros(max_num_reqs, dtype=np.int32)
118
119 self.slot_mapping = self._make_buffer(
120 self.max_num_batched_tokens, dtype=torch.int64
121 )
122
123 if self.use_hybrid_blocks:
124 self._kernel_block_arange = np.arange(0, self.blocks_per_kv_block).reshape(
125 1, -1
126 )
127 else:
128 self._kernel_block_arange = None
129
130 try:
131 self.pcp_world_size = get_pcp_group().world_size
132 self.pcp_rank = get_pcp_group().rank_in_group
133 except AssertionError:
134 # PCP might not be initialized in testing
135 self.pcp_world_size = 1
136 self.pcp_rank = 0
137 try:
138 self.dcp_world_size = get_dcp_group().world_size
139 self.dcp_rank = get_dcp_group().rank_in_group
140 except AssertionError:
141 # DCP might not be initialized in testing
142 self.dcp_world_size = 1
143 self.dcp_rank = 0
144 self.cp_kv_cache_interleave_size = cp_kv_cache_interleave_size
145 self.slot_mapping_mode = slot_mapping_mode
146 if self.slot_mapping_mode == SlotMappingMode.TOKEN_TO_KV_SLOT:
147 _COMPUTE_SLOT_MAPPING_KERNEL.register_warmup(
148 kv_cache_block_size=self.kv_cache_block_size,
149 blocks_per_kv_block=self.blocks_per_kv_block,
150 total_cp_world_size=self.dcp_world_size,
151 total_cp_rank=self.dcp_rank,
152 cp_kv_cache_interleave_size=self.cp_kv_cache_interleave_size,
153 block_table_stride=self.block_table.gpu.stride(0),
154 block_size=self.block_size,
155 )
156
157 def append_row(
158 self,
159 block_ids: list[int],
160 row_idx: int,
161 ) -> None:
162 if not block_ids:
163 return
164
165 if self.use_hybrid_blocks:
166 block_ids = self.map_to_kernel_blocks(
167 np.array(block_ids), self.blocks_per_kv_block, self._kernel_block_arange
168 )
169
170 num_blocks = len(block_ids)
171 start = self.num_blocks_per_row[row_idx]
172 self.num_blocks_per_row[row_idx] += num_blocks
173 self.block_table.np[row_idx, start : start + num_blocks] = block_ids
174
175 def add_row(self, block_ids: list[int], row_idx: int) -> None:
176 self.num_blocks_per_row[row_idx] = 0
177 self.append_row(block_ids, row_idx)
178
179 def clear_row(self, row_idx: int) -> None:
180 num_blocks = self.num_blocks_per_row[row_idx]
181 if num_blocks > 0:
182 self.block_table.np[row_idx, :num_blocks] = 0
183 self.num_blocks_per_row[row_idx] = 0
184
185 def move_row(self, src: int, tgt: int) -> None:
186 num_blocks = self.num_blocks_per_row[src]
187 block_table_np = self.block_table.np
188 block_table_np[tgt, :num_blocks] = block_table_np[src, :num_blocks]
189 self.num_blocks_per_row[tgt] = num_blocks
190 # Clear the vacated source row: dummy-run batches dereference stale
191 # rows as mamba state slots and write state in place there, possibly
192 # after the blocks have been freed and reallocated.
193 block_table_np[src, :num_blocks] = 0
194 self.num_blocks_per_row[src] = 0
195
196 def swap_row(self, src: int, tgt: int) -> None:
197 src_tgt, tgt_src = [src, tgt], [tgt, src]
198 self.num_blocks_per_row[src_tgt] = self.num_blocks_per_row[tgt_src]
199 self.block_table.np[src_tgt] = self.block_table.np[tgt_src]
200
201 def compute_slot_mapping(
202 self,
203 num_reqs: int,
204 query_start_loc: torch.Tensor,
205 positions: torch.Tensor,
206 ) -> None:
207 num_tokens = positions.shape[0]
208 if self.slot_mapping_mode == SlotMappingMode.NONE:
209 # Mamba/GDN groups consume the block table as recurrent state
210 # indices and do not use per-token slot mappings.
211 return
212 assert self.slot_mapping_mode == SlotMappingMode.TOKEN_TO_KV_SLOT
213
214 _COMPUTE_SLOT_MAPPING_KERNEL(
215 num_reqs,
216 num_tokens,
217 self.max_num_batched_tokens,
218 query_start_loc,
219 positions,
220 self.block_table.gpu,
221 self.block_table.gpu.stride(0),
222 self.block_size,
223 self.slot_mapping.gpu,
224 self.kv_cache_block_size,
225 self.blocks_per_kv_block,
226 self.dcp_world_size,
227 self.dcp_rank,
228 self.cp_kv_cache_interleave_size,
229 )
230
231 def commit_block_table(self, num_reqs: int) -> None:
232 self.block_table.copy_to_gpu(num_reqs)
233
234 def clear(self) -> None:
235 self.block_table.gpu.fill_(0)
236 self.block_table.cpu.fill_(0)

238 @staticmethod
239 def map_to_kernel_blocks(
240 kv_manager_block_ids: np.ndarray,
241 blocks_per_kv_block: int,
242 kernel_block_arange: np.ndarray,
243 ) -> np.ndarray:
244 """Convert kv_manager_block_id IDs to kernel block IDs.
245
246 Example:
247 # kv_manager_block_ids: 32 tokens,
248 # Kernel block size: 16 tokens
249 # blocks_per_kv_block = 2
250 >>> kv_manager_block_ids = np.array([0, 1, 2])
251 >>> Result: [0, 1, 2, 3, 4, 5]
252
253 # Each kv_manager_block_id maps to 2 kernel block id:
254 # kv_manager_block_id 0 → kernel block id [0, 1]
255 # kv_manager_block_id 1 → kernel block id [2, 3]
256 # kv_manager_block_id 2 → kernel block id [4, 5]
257 """
258 if blocks_per_kv_block == 1:
259 return kv_manager_block_ids
260
261 kernel_block_ids = (
262 kv_manager_block_ids.reshape(-1, 1) * blocks_per_kv_block
263 + kernel_block_arange
264 )
265
266 return kernel_block_ids.reshape(-1)

268 def get_device_tensor(self, num_reqs: int) -> torch.Tensor:
269 """Returns the device tensor of the block table."""
270 return self.block_table.gpu[:num_reqs]
271
272 def get_cpu_tensor(self) -> torch.Tensor:
273 """Returns the CPU tensor of the block table."""
274 return self.block_table.cpu
275
276 def get_numpy_array(self) -> np.ndarray:
277 """Returns the numpy array of the block table."""
278 return self.block_table.np

288class MultiGroupBlockTable:
289 """The BlockTables for each KV cache group."""
290
291 def __init__(
292 self,
293 max_num_reqs: int,
294 max_num_batched_tokens: int,
295 pin_memory: bool,
296 device: torch.device,
297 block_sizes: list[int],
298 kernel_block_sizes: list[int],
299 max_num_blocks: list[int],
300 cp_kv_cache_interleave_size: int = 1,
301 slot_mapping_modes: list[SlotMappingMode] | None = None,
302 ) -> None:
303 if len(kernel_block_sizes) != len(block_sizes):
304 raise ValueError(
305 f"kernel_block_sizes length ({len(kernel_block_sizes)}) "
306 f"must match block_sizes length ({len(block_sizes)})"
307 )
308 if slot_mapping_modes is None:
309 slot_mapping_modes = [SlotMappingMode.TOKEN_TO_KV_SLOT] * len(block_sizes)
310 if len(slot_mapping_modes) != len(block_sizes):
311 raise ValueError(
312 f"slot_mapping_modes length ({len(slot_mapping_modes)}) "
313 f"must match block_sizes length ({len(block_sizes)})"
314 )
315
316 if len(max_num_blocks) != len(block_sizes):
317 raise ValueError(
318 f"max_num_blocks length ({len(max_num_blocks)}) "
319 f"must match block_sizes length ({len(block_sizes)})"
320 )
321
322 max_num_blocks = [
323 (
324 get_block_table_width(n, block_size, token_alignment=None)
325 if slot_mapping_mode == SlotMappingMode.NONE
326 else get_block_table_width(n, block_size)
327 )
328 for n, block_size, slot_mapping_mode in zip(
329 max_num_blocks, block_sizes, slot_mapping_modes
330 )
331 ]
332
333 self.block_tables = [
334 BlockTable(
335 block_size,
336 max_num_reqs,
337 max_num_blocks_per_req,
338 max_num_batched_tokens,
339 pin_memory,
340 device,
341 kernel_block_size,
342 cp_kv_cache_interleave_size,
343 slot_mapping_mode=slot_mapping_mode,
344 )
345 for (
346 block_size,
347 kernel_block_size,
348 max_num_blocks_per_req,
349 slot_mapping_mode,
350 ) in zip(
351 block_sizes, kernel_block_sizes, max_num_blocks, slot_mapping_modes
352 )
353 ]
354
355 def append_row(self, block_ids: tuple[list[int], ...], row_idx: int) -> None:
356 for i, block_table in enumerate(self.block_tables):
357 block_table.append_row(block_ids[i], row_idx)
358
359 def add_row(self, block_ids: tuple[list[int], ...], row_idx: int) -> None:
360 for i, block_table in enumerate(self.block_tables):
361 block_table.add_row(block_ids[i], row_idx)
362
363 def clear_row(self, row_idx: int) -> None:
364 for block_table in self.block_tables:
365 block_table.clear_row(row_idx)
366
367 def move_row(self, src: int, tgt: int) -> None:
368 for block_table in self.block_tables:
369 block_table.move_row(src, tgt)
370
371 def swap_row(self, src: int, tgt: int) -> None:
372 for block_table in self.block_tables:
373 block_table.swap_row(src, tgt)
374
375 def compute_slot_mapping(
376 self,
377 num_reqs: int,
378 query_start_loc: torch.Tensor,
379 positions: torch.Tensor,
380 ) -> None:
381 for block_table in self.block_tables:
382 block_table.compute_slot_mapping(num_reqs, query_start_loc, positions)
383
384 def commit_block_table(self, num_reqs: int) -> None:
385 for block_table in self.block_tables:
386 block_table.commit_block_table(num_reqs)
387
388 def clear(self) -> None:
389 for block_table in self.block_tables:
390 block_table.clear()
391
392 def __getitem__(self, idx: int) -> "BlockTable":
393 """Returns the BlockTable for the i-th KV cache group."""
394 return self.block_tables[idx]

397class ComputeSlotMappingKernel(
398 VllmTritonJitKernel["ComputeSlotMappingKernel.CompileKey"]
399):
400 triton_block_size = 1024

402 @dataclass(frozen=True)
403 class CompileKey:
404 kv_cache_block_size: int
405 blocks_per_kv_block: int
406 total_cp_world_size: int
407 total_cp_rank: int
408 cp_kv_cache_interleave_size: int
409 block_table_stride: int
410 block_size: int
411
412 @staticmethod
413 @triton.jit(do_not_specialize=["num_tokens", "max_num_tokens"])
414 def kernel(
415 num_tokens,
416 max_num_tokens,
417 query_start_loc_ptr, # [num_reqs + 1], int32
418 positions_ptr, # [num_tokens], int64
419 block_table_ptr, # [max_num_reqs, max_num_blocks_per_req], int32 (flat)
420 block_table_stride, # max_num_blocks_per_req
421 block_size,
422 slot_mapping_ptr, # [max_num_tokens], int64
423 KV_CACHE_BLOCK_SIZE: tl.constexpr,
424 BLOCKS_PER_KV_BLOCK: tl.constexpr,
425 TOTAL_CP_WORLD_SIZE: tl.constexpr,
426 TOTAL_CP_RANK: tl.constexpr,
427 CP_KV_CACHE_INTERLEAVE_SIZE: tl.constexpr,
428 PAD_ID: tl.constexpr,
429 BLOCK_SIZE: tl.constexpr,
430 ):
431 req_idx = tl.program_id(0)
432
433 if req_idx == tl.num_programs(0) - 1:
434 # Pad remaining slots for CUDA graph compatibility.
435 for i in range(num_tokens, max_num_tokens, BLOCK_SIZE):
436 offsets = i + tl.arange(0, BLOCK_SIZE)
437 tl.store(
438 slot_mapping_ptr + offsets,
439 PAD_ID,
440 mask=offsets < max_num_tokens,
441 )
442 return
443
444 start_idx = tl.load(query_start_loc_ptr + req_idx).to(tl.int64)
445 end_idx = tl.load(query_start_loc_ptr + req_idx + 1).to(tl.int64)
446
447 virtual_block_size = KV_CACHE_BLOCK_SIZE * TOTAL_CP_WORLD_SIZE
448 row_offset = req_idx * block_table_stride
449 for i in range(start_idx, end_idx, BLOCK_SIZE):
450 offsets = i + tl.arange(0, BLOCK_SIZE)
451 mask = offsets < end_idx
452 pos = tl.load(positions_ptr + offsets, mask=mask, other=0)
453 virtual_block_indices = pos // virtual_block_size
454 virtual_block_offsets = pos - virtual_block_indices * virtual_block_size
455 is_local = (
456 virtual_block_offsets // CP_KV_CACHE_INTERLEAVE_SIZE
457 ) % TOTAL_CP_WORLD_SIZE == TOTAL_CP_RANK
458 local_block_offsets = (
459 virtual_block_offsets
460 // (TOTAL_CP_WORLD_SIZE * CP_KV_CACHE_INTERLEAVE_SIZE)
461 ) * CP_KV_CACHE_INTERLEAVE_SIZE + (
462 virtual_block_offsets % CP_KV_CACHE_INTERLEAVE_SIZE
463 )
464
465 block_indices = (
466 virtual_block_indices * BLOCKS_PER_KV_BLOCK
467 + local_block_offsets // block_size
468 )
469 block_numbers = tl.load(
470 block_table_ptr + row_offset + block_indices,
471 mask=mask & is_local,
472 other=0,
473 ).to(tl.int64)
474 slot_offsets = local_block_offsets % block_size
475 slot_ids = block_numbers * block_size + slot_offsets
476 slot_ids = tl.where(is_local, slot_ids, PAD_ID)
477 tl.store(slot_mapping_ptr + offsets, slot_ids, mask=mask)
478
