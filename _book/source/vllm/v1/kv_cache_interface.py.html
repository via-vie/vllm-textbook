39class KVQuantMode(IntEnum):
40 """KV cache quantization mode.
41
42 Used by attention backends and kernels to dispatch quantization logic
43 without string matching on ``kv_cache_dtype``.
44 """
45
46 NONE = 0
47 FP8_PER_TENSOR = 1 # per-tensor scales (current fp8 path)
48 INT8_PER_TOKEN_HEAD = 2 # per-token-head dynamic scales for int8
49 FP8_PER_TOKEN_HEAD = 3 # per-token-head dynamic scales for fp8
50 INT4_PER_TOKEN_HEAD = 4 # packed 2×int4/byte, RHT + asymmetric zp
51 NVFP4 = 5 # packed fp4 data + fp8 block scales
52 # Hadamard-rotated Lloyd-Max quant, packed K+V per slot.
53 TURBOQUANT_K8V4 = 6
54 TURBOQUANT_4BIT_NC = 7
55 TURBOQUANT_K3V4_NC = 8
56 TURBOQUANT_3BIT_NC = 9
57 NVFP4_DS_MLA = 10 # opaque-bytes NVFP4 DS-MLA layouts (FlashMLA sparse)
58
59 @property
60 def is_per_token_head(self) -> bool:
61 """True for any per-token-head quantization mode."""
62 return self in (
63 KVQuantMode.INT8_PER_TOKEN_HEAD,
64 KVQuantMode.FP8_PER_TOKEN_HEAD,
65 KVQuantMode.INT4_PER_TOKEN_HEAD,
66 )
67
68 @property
69 def is_nvfp4(self) -> bool:
70 """True for NVFP4 packed quantization mode."""
71 return self == KVQuantMode.NVFP4
72
73 @property
74 def is_turboquant(self) -> bool:
75 """True for any turboquant quantization mode."""
76 return self in (
77 KVQuantMode.TURBOQUANT_K8V4,
78 KVQuantMode.TURBOQUANT_4BIT_NC,
79 KVQuantMode.TURBOQUANT_K3V4_NC,
80 KVQuantMode.TURBOQUANT_3BIT_NC,
81 )

84def get_kv_quant_mode(kv_cache_dtype: str) -> KVQuantMode:
85 """Map a ``kv_cache_dtype`` string to a :class:`KVQuantMode`."""
86 if kv_cache_dtype == "int4_per_token_head":
87 return KVQuantMode.INT4_PER_TOKEN_HEAD
88 if kv_cache_dtype == "int8_per_token_head":
89 return KVQuantMode.INT8_PER_TOKEN_HEAD
90 if kv_cache_dtype == "fp8_per_token_head":
91 return KVQuantMode.FP8_PER_TOKEN_HEAD
92 # Must precede the ``nvfp4`` prefix test below, which would otherwise match.
93 if kv_cache_dtype == "nvfp4_ds_mla":
94 # Page size is keyed on cache_dtype_str in the MLA specs, not
95 # nvfp4_kv_cache_full_dim.
96 return KVQuantMode.NVFP4_DS_MLA
97 if kv_cache_dtype.startswith("nvfp4"):
98 return KVQuantMode.NVFP4
99 if isinstance(kv_cache_dtype, str) and kv_cache_dtype.startswith("turboquant_"):
100 return KVQuantMode[kv_cache_dtype.upper()]
101 if isinstance(kv_cache_dtype, str) and kv_cache_dtype.startswith("fp8"):
102 return KVQuantMode.FP8_PER_TENSOR
103 return KVQuantMode.NONE

106def is_quantized_kv_cache(kv_cache_dtype: str) -> bool:
107 return get_kv_quant_mode(kv_cache_dtype) != KVQuantMode.NONE

110def replace_as(
111 spec: KVCacheSpec,
112 target_cls: type[_SpecT],
113 *,
114 drop: Collection[str] = (),
115 **changes,
116) -> _SpecT:
117 """``dataclasses.replace``, but rebuilding *spec* as *target_cls*
118 e.g. ``SlidingWindowSpec`` -> ``FullAttentionSpec``
119
120 Every field of *spec* must exist on *target_cls* unless named in *drop*;
121 fields only *target_cls* has keep their default values.
122 """
123 kwargs = {
124 f.name: getattr(spec, f.name)
125 for f in fields(spec)
126 if f.init and f.name not in drop
127 }
128 kwargs.update(changes)
129 return target_cls(**kwargs)

277def group_kernel_blocks(cache: torch.Tensor, num_blocks: int) -> torch.Tensor:
278 """View a kernel-block-granular layer cache with manager blocks as dim 0.
279
280 Kernel block splitting subdivides each manager block into uniformly strided
281 kernel blocks, so grouping is a pure view: ``(num_blocks * ratio, ...)``
282 """
283 if cache.shape[0] == num_blocks:
284 return cache
285 assert cache.shape[0] % num_blocks == 0
286 return cache.unflatten(0, (num_blocks, -1))

289def compute_layer_kv_cache_shape_bytes(
290 spec: KVCacheSpec,
291 num_blocks: int,
292 kernel_block_size: int | None = None,
293) -> tuple[int, ...]:
294 """Return the 4D logical shape ``(B, H, N, C)`` where C is in bytes."""
295 bs = kernel_block_size if kernel_block_size is not None else spec.block_size
296 assert spec.block_size % bs == 0, (
297 f"Kernel block size {bs} must divide KV cache block size {spec.block_size}."
298 )
299 blocks_per_page = spec.block_size // bs
300 return (
301 num_blocks * blocks_per_page,
302 spec.num_heads,
303 spec.get_num_kernel_states(bs),
304 spec.state_content_size_bytes,
305 )

308def compute_layout_strides(
309 spec: KVCacheSpec,
310 num_blocks: int,
311 num_layers: int,
312 layout: KVCacheLayout,
313 kernel_block_size: int | None = None,
314 fixed_strides: tuple[int | None, ...] = (None,) * 5,
315) -> tuple[int, ...]:
316 """Byte strides in logical ``[L, B, H, N, C]`` axis order."""
317 assert len(fixed_strides) == 5
318 assert all(stride is None or stride > 0 for stride in fixed_strides)
319 shape = (
320 num_layers,
321 *compute_layer_kv_cache_shape_bytes(spec, num_blocks, kernel_block_size),
322 )
323 order = layout.stride_order
324 padded_page_size = getattr(spec, "page_size_padded", None)
325 if padded_page_size is not None:
326 assert kernel_block_size is None or kernel_block_size == spec.block_size, (
327 "Padded KV pages do not support kernel block splitting."
328 )
329 page_grid_end = max(order.index(_DIM_L), order.index(_DIM_B)) + 1
330 page_grid_shape = tuple(shape[dim] for dim in order[:page_grid_end])
331 assert prod(page_grid_shape) == num_layers * num_blocks, (
332 "Page padding requires dimensions outside the page tail to be L, B, "
333 f"or singleton; got {layout.name} with shape {shape}."
334 )
335
336 strides = [0] * 5
337 current_stride = 1
338 for physical_idx, dim in reversed(tuple(enumerate(order))):
339 if padded_page_size is not None and physical_idx == page_grid_end - 1:
340 current_stride = max(current_stride, padded_page_size)
341 assert current_stride % padded_page_size == 0
342 strides[dim] = fixed_strides[dim] or current_stride
343 current_stride = strides[dim] * shape[dim]
344 return tuple(strides)

347def create_kv_cache_views(
348 raw: torch.Tensor,
349 spec: KVCacheSpec,
350 num_blocks: int,
351 layout: KVCacheLayout,
352 kv_cache_tensor: KVCacheTensor,
353 kernel_block_size: int | None = None,
354) -> list[torch.Tensor]:
355 """View a flat int8 buffer as one 4D ``[B, H, N, C]`` view per layer.
356
357 Block ``b`` of layer ``l`` starts at the tensor offset plus its layer and
358 block stride contributions.
359 """
360 num_layers = len(kv_cache_tensor.layers)
361 layer_stride = kv_cache_tensor.layer_stride
362 block_stride = kv_cache_tensor.block_stride
363 shape_bytes = compute_layer_kv_cache_shape_bytes(
364 spec, num_blocks, kernel_block_size
365 )
366 ratio = shape_bytes[0] // num_blocks
367 if ratio > 1:
368 # Kernel blocks subdivide a manager block into `ratio` equal pieces, so
369 # they sit a constant stride apart only if a block is one dense page: no
370 # padding at its end, and no other layer's page before the next block.
371 dense_page_size = prod(compute_layer_kv_cache_shape_bytes(spec, 1)[1:])
372 if block_stride != dense_page_size:
373 raise ValueError(
374 f"The resolved KV cache layout ({layout.name}) does not store "
375 "blocks as dense, unpadded pages (block stride "
376 f"{block_stride} != page {dense_page_size}), so a manager "
377 f"block cannot be split into {ratio} kernel blocks of "
378 f"{kernel_block_size} tokens. Reduce --block-size to "
379 f"{kernel_block_size} or set VLLM_KV_CACHE_LAYOUT to a "
380 "layer-compact layout (e.g. LBNHC)."
381 )
382 assert block_stride % ratio == 0, (
383 f"Block stride {block_stride} must divide into {ratio} equal kernel blocks."
384 )
385 block_stride //= ratio
386
387 logical_shape = (num_layers, *shape_bytes)
388 strides = compute_layout_strides(
389 spec,
390 num_blocks,
391 num_layers,
392 layout,
393 kernel_block_size,
394 fixed_strides=(layer_stride, block_stride, None, None, None),
395 )
396 dtype = getattr(spec, "dtype", None)
397
398 view_5d = torch.as_strided(
399 raw,
400 size=logical_shape,
401 stride=strides,
402 storage_offset=raw.storage_offset() + kv_cache_tensor.offset,
403 )
404
405 views = []
406 for layer_idx in range(num_layers):
407 cache_logical = view_5d[layer_idx]
408 if dtype is not None:
409 cache_logical = cache_logical.view(dtype)
410 views.append(cache_logical)
411 return views

478@dataclass(frozen=True, kw_only=True)
479class AttentionSpec(KVCacheSpec):
480 num_kv_heads: int
481 head_size: int
482 dtype: torch.dtype
483 head_size_v: int = None # type: ignore[assignment]
484 kv_quant_mode: KVQuantMode = KVQuantMode.NONE
485 page_size_padded: int | None = None
486 num_head_slots: int | None = None
487 """H of the logical ``[B, H, N, C]`` page when packing diverges from one
488 slot per KV head. None means one slot per KV head. Published by the backend.
489 """
490 state_content_bytes: int | None = None
491 """C in bytes when packed; None means dense K/V content."""
492 tokens_per_state: int | Fraction = 1
493 """Tokens covered by one stored state. Ints > 1 compress multiple tokens
494 into one state (DSv4 sparse MLA); fractions < 1 store multiple states per
495 token (Whisper block pooling: ``Fraction(1, block_pool_size)``)."""
496
497 def __post_init__(self):
498 if self.head_size_v is None:
499 object.__setattr__(self, "head_size_v", self.head_size)
500
501 @property
502 def num_heads(self) -> int:
503 if self.num_head_slots is not None:
504 return self.num_head_slots
505 return self.num_kv_heads
506
507 @property
508 def state_content_size_bytes(self) -> int:
509 """Bytes per (head slot, stored state) cell of the page."""
510 if self.state_content_bytes is not None:
511 return self.state_content_bytes
512 return (self.head_size + self.head_size_v) * get_dtype_size(self.dtype)
513
514 @property
515 def unpadded_page_size_bytes(self) -> int:
516 return self.num_heads * self.num_states * self.state_content_size_bytes
517
518 @property
519 def page_size_bytes(self) -> int:
520 if self.page_size_padded is not None:
521 assert self.page_size_padded >= self.unpadded_page_size_bytes
522 return self.page_size_padded
523 return self.unpadded_page_size_bytes
524
525 @property
526 def real_page_size_bytes(self) -> int:
527 """Alias of ``unpadded_page_size_bytes``
528 TODO(lucas): follow up with TPU backend to see if we can remove this property.
529 """
530 return self.unpadded_page_size_bytes
531
532 def max_num_blocks_per_req(self, vllm_config: VllmConfig, max_len: int) -> int:
533 parallel_config = vllm_config.parallel_config
534 kv_shard_count = parallel_config.decode_context_parallel_size
535 return cdiv(max_len, self.block_size * kv_shard_count)

538@dataclass(frozen=True, kw_only=True)
539class FullAttentionSpec(AttentionSpec):
540 """
541 When hybrid allocator is disabled and the model contains both full
542 attention layers and sliding window attention layers, sliding
543 window attention are regarded as full attention in KV cache manager
544 (blocks are allocated for all tokens), while computed as sliding window
545 attention in model runner.
546 In this case, we use FullAttentionSpec and record the sliding window size.
547 """
548
549 sliding_window: int | None = None
550 """
551 Default to None for not using sliding window attention.
552 """
553 attention_chunk_size: int | None = None
554
555 non_causal: bool = False
556 """
557 Whether the layer attends non-causally (e.g. Prefix LM). Carried on the
558 spec so the engine core, which collects specs from all workers before the
559 scheduler is built, can adjust scheduling policy (chunked prefill / prefix
560 caching) regardless of tensor-parallel layout. It does not affect the KV
561 cache layout itself.
562 """
563
564 def max_memory_usage_bytes(self, vllm_config: VllmConfig) -> int:
565 max_model_len = vllm_config.model_config.max_model_len
566 dcp_world_size = vllm_config.parallel_config.decode_context_parallel_size
567 if dcp_world_size > 1:
568 max_model_len = cdiv(max_model_len, dcp_world_size)
569 return cdiv(max_model_len, self.block_size) * self.page_size_bytes

571 @classmethod
572 def merge_window_sizes(cls, window_sizes: set[int]) -> int | None:
573 if len(window_sizes) == 0:
574 return None
575 elif len(window_sizes) == 1:
576 return window_sizes.pop()
577 else:
578 raise ValueError(
579 "All attention layers in the same KV cache group must have the "

583 @classmethod
584 def merge(cls, specs: list[Self]) -> Self:
585 """
586 Merge a list of FullAttentionSpec objects into a single
587 FullAttentionSpec object.
588 """
589 assert all(isinstance(spec, FullAttentionSpec) for spec in specs), (
590 "All attention layers in the same KV cache group must be FullAttentionSpec."
591 )
592
593 sliding_window = set(
594 spec.sliding_window for spec in specs if spec.sliding_window is not None
595 )
596 attention_chunk_size = set(
597 spec.attention_chunk_size
598 for spec in specs
599 if spec.attention_chunk_size is not None
600 )
601 assert not any(isinstance(spec, MLAAttentionSpec) for spec in specs), (
602 "MLAAttentionSpec should be merged in MLAAttentionSpec.merge"
603 )
604 merged_spec = cls(
605 block_size=specs[0].block_size,
606 num_kv_heads=specs[0].num_kv_heads,
607 head_size=specs[0].head_size,
608 head_size_v=specs[0].head_size_v,
609 dtype=specs[0].dtype,
610 kv_quant_mode=specs[0].kv_quant_mode,
611 page_size_padded=specs[0].page_size_padded,
612 num_head_slots=specs[0].num_head_slots,
613 state_content_bytes=specs[0].state_content_bytes,
614 tokens_per_state=specs[0].tokens_per_state,
615 sliding_window=cls.merge_window_sizes(sliding_window),
616 attention_chunk_size=cls.merge_window_sizes(attention_chunk_size),
617 # If any layer in the group is non-causal, treat the group as
618 # non-causal so the engine core disables incompatible scheduling.
619 non_causal=any(spec.non_causal for spec in specs),
620 )
621 for spec in specs:
622 for f in fields(AttentionSpec):
623 assert getattr(spec, f.name) == getattr(merged_spec, f.name), (
624 "All attention layers in the same KV cache group must have "
625 "the same attention spec."
626 )
627 assert (merged_spec.sliding_window is not None) + (
628 merged_spec.attention_chunk_size is not None
629 ) <= 1, (
630 "Model with both sliding window layers and chunked local attention "
631 "layers is not supported."
632 )
633 return merged_spec

636def _apply_alignment_padding(spec: MLAAttentionSpec | SlidingWindowMLASpec):
637 if spec.alignment is None:

806@dataclass(frozen=True, kw_only=True)
807class SlidingWindowSpec(AttentionSpec):
808 sliding_window: int
809 # The trailing edge of the window is extended by ``extra_retained_tokens``
810 # so that those extra trailing tokens' blocks are retained (but not
811 # attended). This is needed for multi-module spec decoding which can
812 # re-prefill the last num_spec_prefill_tokens - 1 tokens from the end
813 # of the sequence, and thus needs to delay freeing/caching of blocks.
814 extra_retained_tokens: int = 0
815
816 def max_admission_blocks_per_request(
817 self, max_in_flight_tokens: int, max_model_len: int
818 ) -> int:
819 """Per-request admission cap, in blocks.
820
821 Single source of truth for both startup pool sizing
822 (`max_memory_usage_bytes`) and the runtime admission gate. Per-request
823 real-held blocks plateau at this bound because
824 `SlidingWindowManager.remove_skipped_blocks` runs from `allocate_slots`
825 before each chunk's `get_num_blocks_to_allocate`.
826
827 `max_in_flight_tokens` is the max tokens scheduled but not yet settled
828 (one batch per concurrent step); see `VllmConfig.max_in_flight_tokens`.
829 """
830 # During chunked prefill, we hold KV for the last `sliding_window-1`
831 # computed tokens plus the in-flight tokens (frees happen on the
832 # processed-token basis); never more than `max_model_len`. An additional
833 # `extra_retained_tokens` trailing tokens are kept alive below the
834 # window for multi-module spec decoding, and must be accounted here too.
835 num_tokens = min(
836 self.sliding_window - 1 + self.extra_retained_tokens + max_in_flight_tokens,
837 max_model_len,
838 )
839 # +1 because the sliding window may not start from the beginning of
840 # the block. E.g. block size 4 and num_token 4 needs two blocks
841 # [XXCD][EF] to store the 6-token window [CDEF].
842 return cdiv(num_tokens, self.block_size) + 1
843

846 "DCP not support sliding window."

848 max_blocks = self.max_admission_blocks_per_request(
849 max_in_flight_tokens=vllm_config.max_in_flight_tokens,
850 max_model_len=vllm_config.model_config.max_model_len,
851 )
852 return max_blocks * self.page_size_bytes

855 self, kv_cache_specs: dict[str, KVCacheSpec]

857 return all(
858 isinstance(spec, SlidingWindowSpec)

1074 num_tokens: int,
1075 hash_block_size: int,
1076 drop_eagle_block: bool,
1077) -> int:
1078 """Return the reusable Mamba checkpoint boundary for a prefill."""
1079 checkpoint_position = (num_tokens - 1) // hash_block_size * hash_block_size

1085def is_mamba_prefill_checkpoint_valid(
1086 query_start: int,
1087 query_end: int,
1088 checkpoint_position: int,
1089 hash_block_size: int,
1090 mamba_block_size: int,
1091 checkpoint_alignment: int | None,
1092) -> bool:
1093 """Whether a backend can export the checkpoint in this query."""
1094 if checkpoint_alignment is None:
1095 return False
1096 assert checkpoint_alignment > 0
1097
1098 initial_state_col = (query_start - 1) // mamba_block_size
1099 checkpoint_col = cdiv(query_end, mamba_block_size) - 2
1100 return (
1101 query_start % hash_block_size == 0
1102 and checkpoint_col > initial_state_col
1103 and query_start + hash_block_size <= checkpoint_position
1104 and query_start < checkpoint_position < query_end
1105 and (checkpoint_position - query_start) % checkpoint_alignment == 0

1240 """
1241 block_sizes = set(spec.block_size for spec in kv_cache_specs.values())
1242 if len(block_sizes) > 1:
1243 # Different block sizes, not uniform.
1244 return False
1245 first_spec = next(iter(kv_cache_specs.values()))
1246 return first_spec.is_uniform_with_collection(kv_cache_specs)

1290def is_full_attention_spec(kv_cache_spec: KVCacheSpec) -> bool:
1291 """Whether a KV cache group spec is (or wraps) full attention.
1292
1293 ``UniformTypeKVCacheSpecs`` is not itself a ``FullAttentionSpec``, so a bare
1294 isinstance check misses groups that carry the wrapper -- DeepSeek-V4's MLA
1295 layers, or any model taking the ``UniformTypeKVCacheSpecs.from_specs`` path.
1296
1297 Every layer must be full attention: a group holding a recycling
1298 (sliding-window) layer has no stable slot layout, so callers that key data
1299 by slot cannot use it.
1300 """
1301 layer_specs = iter_layer_specs(kv_cache_spec)
1302 return len(layer_specs) > 0 and all(
1303 isinstance(spec, FullAttentionSpec) for spec in layer_specs
1304 )

1308 if isinstance(kv_cache_spec, UniformTypeKVCacheSpecs):
1309 inner_kinds = {
1310 get_kv_cache_spec_kind(spec)
1311 for spec in kv_cache_spec.kv_cache_specs.values()
1312 }
1313 if len(inner_kinds) == 1:
1314 return next(iter(inner_kinds))
1315 # A group is only formed when all members share one registered
1316 # uniform_type_base_spec, so UNKNOWN would discard what the merge
1317 # already established.
1318 base_specs = {

1320 for spec in kv_cache_spec.kv_cache_specs.values()

1322 if len(base_specs) == 1 and next(iter(base_specs)) is FullAttentionSpec:
1323 return KVCacheSpecKind.FULL_ATTENTION
1324 return KVCacheSpecKind.UNKNOWN
1325 # Keep subclass checks before base classes so specialized specs keep their
1326 # more precise kind.
1327 if isinstance(kv_cache_spec, SlidingWindowMLASpec):
1328 return KVCacheSpecKind.SLIDING_WINDOW_MLA
1329 if isinstance(kv_cache_spec, MLAAttentionSpec):
1330 return KVCacheSpecKind.MLA_ATTENTION
1331 if isinstance(kv_cache_spec, SinkFullAttentionSpec):
1332 return KVCacheSpecKind.SINK_FULL_ATTENTION
1333 if isinstance(kv_cache_spec, FullAttentionSpec):
1334 return KVCacheSpecKind.FULL_ATTENTION
1335 if isinstance(kv_cache_spec, ChunkedLocalAttentionSpec):
1336 return KVCacheSpecKind.CHUNKED_LOCAL_ATTENTION
1337 if isinstance(kv_cache_spec, SlidingWindowSpec):
1338 return KVCacheSpecKind.SLIDING_WINDOW
1339 if isinstance(kv_cache_spec, MambaSpec):
1340 return KVCacheSpecKind.MAMBA
1341 if isinstance(kv_cache_spec, EncoderOnlyAttentionSpec):
1342 return KVCacheSpecKind.ENCODER_ONLY_ATTENTION

1360@dataclass
1361class KVCacheTensor:
1362 """
1363 A class for specifying how the workers should initialize the KV cache.
1364
1365 Placement of a set of same-shaped layers in the KV cache allocation.
1366 Layer ``layers[l]``'s page for block ``b`` starts at
1367 ``offset + l * layer_stride + b * block_stride`` bytes into the backing
1368 allocation of ``size`` bytes. Layer-outermost layouts give each layer a
1369 contiguous region (``layer_stride = page * num_blocks``,
1370 ``block_stride = page``); block-outermost layouts make each block a
1371 block of all layers' pages (``layer_stride = page``, ``block_stride`` =
1372 the packed block). Tensors whose address ranges overlap
1373 alias the same bytes: cache groups overlay each other, which is sound
1374 because a block ID is owned by one group at a time.
1375 """
1376
1377 size: int # total size of the backing allocation in bytes
1378 layers: list[str] # layer names in L order
1379 layer_stride: int
1380 block_stride: int
1381 offset: int = 0 # byte offset of layers[0]'s block 0
1382 host_resident: bool = False
