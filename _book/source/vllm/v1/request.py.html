32@dataclass
33class StreamingUpdate:
34 """Lightweight data for streaming session continuation.
35
36 Contains only the fields needed to update an existing streaming session
37 with new input data.
38 """
39
40 mm_features: list[MultiModalFeatureSpec] | None
41 prompt_token_ids: list[int] | None
42 max_tokens: int
43 arrival_time: float
44 sampling_params: SamplingParams | None

49 return None
50 return cls(
51 mm_features=request.mm_features,
52 prompt_token_ids=request.prompt_token_ids,
53 max_tokens=request.max_tokens,
54 arrival_time=request.arrival_time,

59class Request:
60 def __init__(
61 self,
62 request_id: str,
63 prompt_token_ids: list[int] | None,
64 sampling_params: SamplingParams | None,
65 pooling_params: PoolingParams | None,
66 client_index: int = 0,
67 arrival_time: float | None = None,
68 prompt_embeds: torch.Tensor | None = None,
69 prompt_is_token_ids: list[bool] | None = None,
70 mm_features: list[MultiModalFeatureSpec] | None = None,
71 lora_request: "LoRARequest | None" = None,
72 cache_salt: str | None = None,
73 priority: int = 0,
74 trace_headers: Mapping[str, str] | None = None,
75 block_hasher: Callable[["Request"], list["BlockHash"]] | None = None,
76 resumable: bool = False,
77 session_id: str | None = None,
78 reasoning_ended: bool | None = None,
79 reasoning_parser_kwargs: dict[str, Any] | None = None,
80 abort_immediately: bool = False,
81 ) -> None:
82 self.request_id = request_id
83 self.client_index = client_index
84 self.priority = priority
85 self.sampling_params = sampling_params
86 self.pooling_params = pooling_params
87 self.lora_request = lora_request
88 self.structured_output_request = StructuredOutputRequest.from_sampling_params(
89 sampling_params
90 )
91 if self.structured_output_request is not None:
92 self.structured_output_request.reasoning_ended = reasoning_ended
93 self.structured_output_request.reasoning_parser_kwargs = (
94 reasoning_parser_kwargs
95 )
96 self.arrival_time = arrival_time if arrival_time is not None else time.time()
97
98 self.status = RequestStatus.WAITING
99 self.events: list[EngineCoreEvent] = []
100 self.stop_reason: int | str | None = None
101
102 # P/D: Connector-specific KV transfer parameters.
103 self.kv_transfer_params: dict[str, Any] | None = None
104 # E/P/D: Connector-specific encoder-cache transfer parameters.
105 self.ec_transfer_params: dict[str, Any] | None = None
106
107 if pooling_params is not None:
108 # Pooling models.
109 self.max_tokens = 1
110 elif sampling_params is not None:
111 # Generative models.
112 assert sampling_params.max_tokens is not None
113 self.max_tokens = sampling_params.max_tokens
114 if self.structured_output_request is not None:
115 self.status = RequestStatus.WAITING_FOR_STRUCTURED_OUTPUT_GRAMMAR
116
117 if sampling_params.extra_args is not None:
118 self.kv_transfer_params = sampling_params.extra_args.get(
119 "kv_transfer_params"
120 )
121 self.ec_transfer_params = sampling_params.extra_args.get(
122 "ec_transfer_params"
123 )
124 self.kv_cache_report_mode = sampling_params.extra_args.get(
125 "kv_cache_report_mode", "incremental"
126 )
127 else:
128 self.kv_cache_report_mode = "incremental"
129 else:
130 raise ValueError("sampling_params and pooling_params can't both be unset")
131
132 self.prompt_token_ids = prompt_token_ids
133 self.prompt_embeds = prompt_embeds
134 # Per-position mask used in mixed-mode (chat completion with
135 # prompt_embeds). `None` except when both `prompt_token_ids` and
136 # `prompt_embeds` are set and their positions are interleaved.
137 self.prompt_is_token_ids = prompt_is_token_ids
138 # Cache per-block prompt-embed hashes to avoid rehashing the same
139 # tensor slices when generating extra keys.
140 self._prompt_embeds_per_block_hashes: dict[tuple[int, int], bytes] = {}
141 self.num_prompt_tokens = length_from_prompt_token_ids_or_embeds(
142 prompt_token_ids, prompt_embeds
143 )
144 self._output_token_ids: list[int] = []
145 if self.prompt_token_ids is None:
146 self._all_token_ids: list[int] = [0] * self.num_prompt_tokens
147 elif self.prompt_is_token_ids is None:
148 self._all_token_ids = self.prompt_token_ids.copy()
149 else:
150 # Mixed-mode prompt: positions covered by prompt_embeds hold a sentinel
151 # special token id that may lie outside the embedding. Zero them, matching
152 # the no-token-ids case above, so embedding gathers over these placeholder
153 # ids stay in bounds; the actual inputs come from prompt_embeds.
154 self._all_token_ids = [
155 t if is_tok else 0
156 for t, is_tok in zip(self.prompt_token_ids, self.prompt_is_token_ids)
157 ]
158
159 # Used in async scheduling.
160 self.num_output_placeholders = 0
161 # Tokens of output in flight when the request was preempted: delivered
162 # on return, but must not mutate the reset counters.
163 self.num_stale_output_tokens = 0
164 # Drop the stale output instead, for same-step preempt + resume
165 # (reset_prefix_cache).
166 self.drop_stale_output = False
167
168 # Tokens of steps whose output is not yet processed (async scheduling
169 # and PP run ahead of the GPU); `num_computed_tokens` counts them
170 # optimistically.
171 self.num_in_flight_tokens = 0
172
173 # V2+PP+async: Enforces `pp_size` cadence between same-request decode steps
174 # so the worker's broadcast slot ring stays consistent.
175 self.next_decode_eligible_step = 0
176
177 # Seq of the most recent step this request was scheduled in; fences
178 # deferred block freeing (see Scheduler._free_request_blocks).
179 self.last_sched_seq = 0
180
181 self.spec_token_ids: list[int] = []
182 self.num_computed_tokens = 0
183 self.cache_salt: str | None = cache_salt
184
185 # Multi-modal related
186 self.mm_features = mm_features or []
187
188 # Read-only views
189 # Prevent directly appending to these lists since
190 # they should also be updated simultaneously.
191 self.output_token_ids = ConstantList(self._output_token_ids)
192 self.all_token_ids = ConstantList(self._all_token_ids)
193 # trace_headers
194 self.trace_headers = trace_headers
195 self.session_id = session_id
196
197 # True if this request is scheduled as a non-final prefill chunk.
198 self.is_prefill_chunk = False
199
200 # Block-aligned token position of a proven shared prefix worth pinning
201 # in the (sparse) prefix cache; 0 means none. Set at admission for
202 # hybrid/Mamba models when a shared prefix is detected (Marconi-style).
203 self.shared_prefix_boundary = 0
204
205 # The number of NaNs in logits. A value greater than 0
206 # indicates that the output is corrupted
207 self.num_nans_in_logits = 0
208
209 # The number of times this request has been preempted by the scheduler.
210 self.num_preemptions = 0
211
212 self.prefill_stats: PrefillStats | None = PrefillStats()
213
214 # Per-request speculative-decoding acceptance accumulator. Populated by
215 # the scheduler when --per-request-spec-decode-metrics is set (eagerly on
216 # add_request, then observed each verify step); stays None otherwise.
217 self.spec_decode_metrics: RequestSpecDecodeMetrics | None = None
218
219 self.block_hashes: list[BlockHash] = []
220 # Store the block hasher without binding self to avoid creating a
221 # reference cycle (Request -> partial -> Request) that prevents
222 # immediate garbage collection via reference counting.
223 self._block_hasher: Callable[[Request], list[BlockHash]] | None = block_hasher
224 self.update_block_hashes()
225
226 self.skip_reading_prefix_cache = self.get_skip_reading_prefix_cache()
227
228 # Used for streaming
229 self.resumable = resumable
230 # None entry in the queue means finished.
231 self.streaming_queue: deque[StreamingUpdate | None] | None = None
232
233 # If True, request should be aborted immediately after being added to
234 # the scheduler so the connector's request_finished hook runs.
235 self.abort_immediately = abort_immediately

265 def append_output_token_ids(
266 self,
267 token_ids: int | list[int],
268 ) -> None:
269 if isinstance(token_ids, int):
270 self._output_token_ids.append(token_ids)
271 self._all_token_ids.append(token_ids)

273 self._output_token_ids.extend(token_ids)
274 self._all_token_ids.extend(token_ids)

276 self.update_block_hashes()

278 def update_block_hashes(self) -> None:
279 """Compute block hashes for any new full blocks and append them."""
280 if self._block_hasher is not None:
281 self.block_hashes.extend(self._block_hasher(self))

350 def __lt__(self, other: "Request") -> bool:
351 """
352 Compare two requests based on priority, arrival time, and request ID.
353 Used in priority scheduling.
354 """
355 if self.priority != other.priority:
356 return self.priority < other.priority
357 if self.arrival_time != other.arrival_time:
358 return self.arrival_time < other.arrival_time
359 if self.request_id != other.request_id:
360 return self.request_id < other.request_id
361 return id(self) < id(other)

364class RequestStatus(enum.IntEnum):
365 """Status of a request."""
366
367 WAITING = enum.auto()
368 WAITING_FOR_STRUCTURED_OUTPUT_GRAMMAR = enum.auto()
369 WAITING_FOR_REMOTE_KVS = enum.auto()
370 WAITING_FOR_STREAMING_REQ = enum.auto()
371 RUNNING = enum.auto()
372 PREEMPTED = enum.auto()
373 # Note: anything after PREEMPTED will be considered
374 # as a finished status.
375 FINISHED_STOPPED = enum.auto()
376 FINISHED_LENGTH_CAPPED = enum.auto()
377 FINISHED_ABORTED = enum.auto()
378 FINISHED_IGNORED = enum.auto()
379 FINISHED_ERROR = enum.auto()
380 FINISHED_REPETITION = enum.auto()
381
382 def __str__(self) -> str:
383 return self.name
384
385 @staticmethod
386 def is_finished(status: "RequestStatus") -> bool:
387 return status > RequestStatus.PREEMPTED
388
389 @staticmethod
390 def get_finished_reason(status: "RequestStatus") -> FinishReason | None:
391 return _FINISHED_REASON_MAP.get(status)
392
393
394# Mapping of finished statuses to their finish reasons.
395# NOTE: The ignored requests are the requests whose prompt lengths
396# are longer than the model's length cap. Therefore, the stop
397# reason should also be "length" as in OpenAI API.
398_FINISHED_REASON_MAP = {
399 RequestStatus.FINISHED_STOPPED: FinishReason.STOP,
400 RequestStatus.FINISHED_LENGTH_CAPPED: FinishReason.LENGTH,
401 RequestStatus.FINISHED_ABORTED: FinishReason.ABORT,
402 RequestStatus.FINISHED_IGNORED: FinishReason.LENGTH,
403 RequestStatus.FINISHED_ERROR: FinishReason.ERROR,
404 RequestStatus.WAITING_FOR_STREAMING_REQ: FinishReason.STOP,
405 RequestStatus.FINISHED_REPETITION: FinishReason.REPETITION,
406}
