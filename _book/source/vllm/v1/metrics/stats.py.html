35class CachingMetrics:
36 """Metrics for caching with a hit rate of the most recent N requests.
37 Args:
38 interval: The number of the most recent requests to aggregate.
39 Defaults to 1000.
40 """
41
42 def __init__(self, max_recent_requests: int = 1000) -> None:
43 super().__init__()
44
45 self.max_recent_requests = max_recent_requests
46 # The current aggregated values.
47 self.aggregated_requests = 0
48 self.aggregated_query_total = 0
49 self.aggregated_query_hit = 0
50
51 # A deque of (requests, queries, hits) for the most recent requests.
52 self.query_queue = deque[tuple[int, int, int]]()
53
54 def observe(self, stats: BaseCacheStats):
55 """Observe the prefix caching for a set of requests.
56
57 This function is called with information gathered when new requests
58 are being scheduled and are looking for computed blocks.
59
60 When there are more than `max_recent_requests` requests, the oldest set
61 of requests are removed from the metrics.
62
63 Args:
64 stats: The prefix cache stats.
65 """
66 # reset_prefix_cache was invoked before the current update.
67 # Reset the metrics before aggregating the current stats.
68 if stats.reset:
69 self.reset()
70
71 # DO NOT appending empty stats to avoid helpful info get kicked out
72 # due to sliding window.
73 if stats.requests == 0:
74 return
75
76 # Update the metrics.
77 self.query_queue.append((stats.requests, stats.queries, stats.hits))
78 self.aggregated_requests += stats.requests
79 self.aggregated_query_total += stats.queries
80 self.aggregated_query_hit += stats.hits
81
82 # Remove the oldest stats until number of requests does not exceed
83 # the limit.
84 # NOTE: We preserve the latest added stats regardless.
85 while (
86 len(self.query_queue) > 1
87 and self.aggregated_requests > self.max_recent_requests
88 ):
89 old_requests, old_queries, old_hits = self.query_queue.popleft()
90 self.aggregated_requests -= old_requests
91 self.aggregated_query_total -= old_queries
92 self.aggregated_query_hit -= old_hits
93
94 def reset(self):
95 """Reset the metrics."""
96 self.aggregated_requests = 0
97 self.aggregated_query_total = 0
98 self.aggregated_query_hit = 0
99 self.query_queue.clear()
100
101 @property
102 def empty(self) -> bool:
103 """Return true if no requests have been observed."""
104 return self.aggregated_requests == 0
105
106 @property
107 def hit_rate(self) -> float:
108 """Calculate the hit rate for the past N requests."""
109 if self.aggregated_query_total == 0:
110 return 0.0
111 return self.aggregated_query_hit / self.aggregated_query_total

260@dataclass
261class PrefillStats:
262 """Breakdown of a scheduled prefill computation.
263
264 Fields:
265 num_prompt_tokens: Total number of tokens to be prefilled.
266 num_computed_tokens: Tokens to be prefilled locally (actual compute work).
267 num_cached_tokens: Tokens to be prefilled without actual compute work.
268 num_local_cached_tokens: Tokens to be prefilled from local prefix cache.
269 num_external_cached_tokens: Tokens to be prefilled from external KV transfer.
270 num_cache_creation_tokens: Tokens computed and written to the prefix cache.
271 """
272
273 num_prompt_tokens: int = 0
274 num_computed_tokens: int = 0
275 num_cached_tokens: int = 0
276 num_local_cached_tokens: int = 0
277 num_external_cached_tokens: int = 0
278 num_cache_creation_tokens: int = 0
279
280 def set(
281 self,
282 num_prompt_tokens: int,
283 num_local_cached_tokens: int,
284 num_external_cached_tokens: int,
285 ):
286 num_cached_tokens = num_local_cached_tokens + num_external_cached_tokens
287 assert num_cached_tokens <= num_prompt_tokens
288
289 self.num_prompt_tokens = num_prompt_tokens
290 self.num_computed_tokens = num_prompt_tokens - num_cached_tokens
291 self.num_cached_tokens = num_cached_tokens
292 self.num_local_cached_tokens = num_local_cached_tokens
293 self.num_external_cached_tokens = num_external_cached_tokens
294
295 def finalize(self, num_cached_tokens: int) -> None:
296 assert num_cached_tokens >= 0
297 self.num_cache_creation_tokens = max(
298 0, min(num_cached_tokens, self.num_prompt_tokens) - self.num_cached_tokens
299 )

302@dataclass
303class RequestSpecDecodeMetrics:
304 """Per-output-sequence speculative-decoding statistics accumulator.
305
306 Accumulates, over one sequence's verify steps, a histogram of accepted
307 draft-token counts (``j``, draft-only) and the total number of proposed
308 draft tokens. When ``detailed`` is requested it also records the ordered
309 per-step accepted/proposed sequences (``summary`` omits them). Tracked per
310 engine ``Request`` (one per sampled sequence, so ``n > 1`` yields one per
311 child), surfaced via ``EngineCoreOutput`` and the response
312 ``metrics.speculative_decoding`` for single-sequence requests (see
313 ``to_dict``).
314
315 Fields:
316 num_spec_tokens: Configured ``num_speculative_tokens`` (the max ``k``);
317 also the histogram's upper bound.
318 histogram: Dense counts indexed by accepted draft tokens ``j``
319 (length ``num_spec_tokens + 1``).
320 num_draft_tokens: Total proposed draft tokens, after the
321 grammar-invalidated (``num_invalid_spec_tokens``) adjustment.
322 per_step_accepted: Ordered accepted-draft count per verify step
323 (``detailed`` only; empty otherwise).
324 per_step_drafted: Ordered proposed-draft count per verify step
325 (``detailed`` only; empty otherwise).
326 """
327
328 num_spec_tokens: int
329 histogram: list[int] = field(default_factory=list)
330 num_draft_tokens: int = 0
331 per_step_accepted: list[int] = field(default_factory=list)
332 per_step_drafted: list[int] = field(default_factory=list)
333
334 @classmethod
335 def new(cls, num_spec_tokens: int) -> "RequestSpecDecodeMetrics":
336 return cls(
337 num_spec_tokens=num_spec_tokens,
338 histogram=[0] * (num_spec_tokens + 1),
339 )
340
341 def observe(
342 self, num_draft_tokens: int, num_accepted: int, detailed: bool = False
343 ) -> None:
344 self.histogram[num_accepted] += 1
345 self.num_draft_tokens += num_draft_tokens
346 if detailed:
347 self.per_step_accepted.append(num_accepted)
348 self.per_step_drafted.append(num_draft_tokens)
349
350 def to_dict(self) -> dict[str, Any]:
351 """Payload matching ``SpeculativeDecodingMetrics`` for the response.
352
353 ``acceptance_histogram`` is a dense list indexed by accepted draft count
354 ``j`` (length ``num_spec_tokens + 1``). ``mean_acceptance_length``
355 includes the bonus token (``j + 1``); ``draft_acceptance_rate`` is
356 draft-only, full precision. Per-step arrays are included only when
357 populated (``detailed`` level).
358 """
359 num_spec_steps = sum(self.histogram)
360 num_accepted = sum(j * count for j, count in enumerate(self.histogram))
361 mean_al = 1.0 + num_accepted / num_spec_steps if num_spec_steps else 1.0
362 rate = num_accepted / self.num_draft_tokens if self.num_draft_tokens else 0.0
363 result: dict[str, Any] = {
364 "mean_acceptance_length": mean_al,
365 "draft_acceptance_rate": rate,
366 "acceptance_histogram": list(self.histogram),
367 "num_spec_steps": num_spec_steps,
368 "num_accepted_draft_tokens": num_accepted,
369 "num_draft_tokens": self.num_draft_tokens,
370 "num_spec_tokens": self.num_spec_tokens,
371 }
372 if self.per_step_accepted:
373 result["per_step_accepted"] = self.per_step_accepted
374 result["per_step_drafted"] = self.per_step_drafted
375 return result

378@dataclass
379class PromptTokenStats:
380 """Breakdown of prompt tokens by source.
381
382 Fields:
383 computed: Tokens prefilled locally (actual compute work).
384 local_cache_hit: Tokens from local prefix cache.
385 external_kv_transfer: Tokens from external KV transfer.
386 cached_tokens: Tokens skipped during prefill (from scheduler).
387 total: Total prompt tokens.
388
389 Invariants:
390 computed + local_cache_hit + external_kv_transfer = total
391 local_cache_hit + external_kv_transfer = cached_tokens
392 """
393
394 ALL_SOURCES: tuple[str, ...] = (
395 "local_compute",
396 "local_cache_hit",
397 "external_kv_transfer",
398 )
399
400 computed: int = 0
401 local_cache_hit: int = 0
402 external_kv_transfer: int = 0
403 cached_tokens: int = 0
404 total: int = 0
405
406 def update_from_output(self, prefill_stats: PrefillStats) -> None:
407 """Update stats from a prefill output."""
408 self.computed += prefill_stats.num_computed_tokens
409 self.cached_tokens += prefill_stats.num_cached_tokens
410 self.total += prefill_stats.num_prompt_tokens
411
412 self.local_cache_hit += prefill_stats.num_local_cached_tokens
413 self.external_kv_transfer += prefill_stats.num_external_cached_tokens
414
415 def get_by_source(self, source: str) -> int:
416 """Get token count by source label."""
417 source_map = {
418 "local_compute": self.computed,
419 "local_cache_hit": self.local_cache_hit,
420 "external_kv_transfer": self.external_kv_transfer,
421 }
422 if source not in source_map:
423 raise ValueError(f"Unknown source: {source}")
424 return source_map[source]

427class IterationStats:
428 """Stats associated with a single set of EngineCoreOutputs."""
429
430 def __init__(self):
431 self.iteration_timestamp = time.time()
432 self.num_generation_tokens = 0
433 self.prompt_token_stats = PromptTokenStats()
434 self.num_preempted_reqs = 0
435 self.finished_requests: list[FinishedRequestStats] = []
436 self.max_num_generation_tokens_iter: list[int] = []
437 self.n_params_iter: list[int] = []
438 self.time_to_first_tokens_iter: list[float] = []
439 self.inter_token_latencies_iter: list[float] = []
440 self.num_corrupted_reqs: int = 0
441
442 def __repr__(self) -> str:
443 field_to_value_str = ", ".join(f"{k}={v}" for k, v in vars(self).items())
444 return f"{self.__class__.__name__}({field_to_value_str})"
445
446 @property
447 def num_prompt_tokens(self) -> int:
448 """Total prompt tokens (for backward compatibility)."""
449 return self.prompt_token_stats.total
450
451 def _time_since(self, start: float) -> float:
452 """Calculate an interval relative to this iteration's timestamp."""
453 return self.iteration_timestamp - start
454
455 def update_from_output(
456 self,
457 output: "EngineCoreOutput",
458 engine_core_timestamp: float,
459 is_prefilling: bool,
460 req_stats: RequestStateStats,
461 lora_states: "LoRARequestStates",
462 lora_name: str | None,
463 ):
464 num_new_generation_tokens = len(output.new_token_ids)
465
466 self.num_generation_tokens += num_new_generation_tokens
467 if is_prefilling:
468 if output.prefill_stats is not None:
469 self.prompt_token_stats.update_from_output(output.prefill_stats)
470
471 first_token_latency = self._time_since(req_stats.arrival_time)
472 self.time_to_first_tokens_iter.append(first_token_latency)
473 req_stats.first_token_latency = first_token_latency
474
475 req_stats.num_generation_tokens += num_new_generation_tokens
476
477 # Track if this request is corrupted (only check once per request)
478 # Early exit if already marked as corrupted to avoid redundant checks
479 if (
480 envs.VLLM_COMPUTE_NANS_IN_LOGITS
481 and not req_stats.is_corrupted
482 and output.num_nans_in_logits > 0
483 ):
484 req_stats.is_corrupted = True
485
486 # Process request-level engine core events
487 if output.events is not None:
488 self.update_from_events(
489 output.request_id,
490 output.events,
491 is_prefilling,
492 req_stats,
493 lora_states,
494 lora_name,
495 )
496
497 # Process the batch-level "new tokens" engine core event
498 if is_prefilling:
499 req_stats.first_token_ts = engine_core_timestamp
500 else:
501 itl = engine_core_timestamp - req_stats.last_token_ts
502 self.inter_token_latencies_iter.append(itl)
503
504 req_stats.last_token_ts = engine_core_timestamp
505
506 def update_from_events(
507 self,
508 req_id: str,
509 events: list["EngineCoreEvent"],
510 is_prefilling: bool,
511 req_stats: RequestStateStats,
512 lora_states: "LoRARequestStates",
513 lora_name: str | None,
514 ):
515 # Avoid circular dependency
516 from vllm.v1.engine import EngineCoreEventType
517
518 for event in events:
519 if event.type == EngineCoreEventType.QUEUED:
520 req_stats.queued_ts = event.timestamp
521 lora_states.request_waiting(req_id, lora_name)
522 elif event.type == EngineCoreEventType.SCHEDULED:
523 if req_stats.scheduled_ts == 0.0: # ignore preemptions
524 req_stats.scheduled_ts = event.timestamp
525 lora_states.request_running(req_id, lora_name)
526 elif event.type == EngineCoreEventType.PREEMPTED:
527 self.num_preempted_reqs += 1
528 req_stats.num_preemptions += 1
529 lora_states.request_waiting(req_id, lora_name)
530
531 def update_from_finished_request(
532 self,
533 finish_reason: "FinishReason",
534 request_id: str,
535 num_prompt_tokens: int,
536 max_tokens_param: int | None,
537 req_stats: RequestStateStats,
538 num_cached_tokens: int = 0,
539 ):
540 e2e_latency = self._time_since(req_stats.arrival_time)
541
542 # Queued interval is from first QUEUED event to first SCHEDULED
543 queued_time = req_stats.scheduled_ts - req_stats.queued_ts
544
545 # Prefill interval is from first SCHEDULED to first NEW_TOKEN
546 # Any preemptions during prefill is included in the interval
547 prefill_time = req_stats.first_token_ts - req_stats.scheduled_ts
548
549 # Decode interval is from first NEW_TOKEN to last NEW_TOKEN
550 # Any preemptions during decode are included
551 decode_time = req_stats.last_token_ts - req_stats.first_token_ts
552
553 # Inference interval is from first SCHEDULED to last NEW_TOKEN
554 # Any preemptions during prefill or decode are included
555 inference_time = req_stats.last_token_ts - req_stats.scheduled_ts
556
557 # Do not count the token generated by the prefill phase
558 mean_time_per_output_token = (
559 decode_time / (req_stats.num_generation_tokens - 1)
560 if req_stats.num_generation_tokens - 1 > 0
561 else 0
562 )
563
564 finished_req = FinishedRequestStats(
565 finish_reason=finish_reason,
566 request_id=request_id,
567 e2e_latency=e2e_latency,
568 num_prompt_tokens=num_prompt_tokens,
569 num_preemptions=req_stats.num_preemptions,
570 num_generation_tokens=req_stats.num_generation_tokens,
571 max_tokens_param=max_tokens_param,
572 queued_time=queued_time,
573 prefill_time=prefill_time,
574 inference_time=inference_time,
575 decode_time=decode_time,
576 mean_time_per_output_token=mean_time_per_output_token,
577 is_corrupted=req_stats.is_corrupted,
578 num_cached_tokens=num_cached_tokens,
579 )
580 self.finished_requests.append(finished_req)
581
582 # Count corrupted requests when they finish (only once per request)
583 if req_stats.is_corrupted:
584 self.num_corrupted_reqs += 1

587class LoRAStats:
588 """Tracks waiting and running request IDs for a single LoRA."""
589
590 def __init__(self):
591 self.waiting: set[str] = set()
592 self.running: set[str] = set()
593
594 def update(self, req_id: str, waiting: bool, running: bool):
595 assert not (waiting and running)
596 if waiting:
597 self.waiting.add(req_id)
598 else:
599 self.waiting.discard(req_id)
600
601 if running:
602 self.running.add(req_id)
603 else:
604 self.running.discard(req_id)
605
606 @property
607 def empty(self) -> bool:
608 return not (self.waiting or self.running)

611class LoRARequestStates:
612 """A per-LoRA count of running and waiting requests."""
613
614 def __init__(self, log_stats: bool = False):
615 self.log_stats = log_stats
616 self.requests: defaultdict[str, LoRAStats] = defaultdict(LoRAStats)
617
618 def _request_update(
619 self, req_id: str, lora_name: str | None, waiting: bool, running: bool
620 ):
621 if not self.log_stats or lora_name is None:
622 return
623
624 lora_stats = self.requests[lora_name]
625 lora_stats.update(req_id, waiting, running)
626 if lora_stats.empty:
627 del self.requests[lora_name]
628
629 def request_waiting(self, req_id: str, lora_name: str | None):
630 self._request_update(req_id, lora_name, waiting=True, running=False)
631
632 def request_running(self, req_id: str, lora_name: str | None):
633 self._request_update(req_id, lora_name, waiting=False, running=True)
634
635 def request_finished(self, req_id: str, lora_name: str | None):
636 self._request_update(req_id, lora_name, waiting=False, running=False)
637
638 def update_scheduler_stats(self, scheduler_stats: SchedulerStats | None):
639 if not self.log_stats or scheduler_stats is None:
640 return
641 for lora_name, stats in self.requests.items():
642 scheduler_stats.waiting_lora_adapters[lora_name] = len(stats.waiting)
643 scheduler_stats.running_lora_adapters[lora_name] = len(stats.running)
