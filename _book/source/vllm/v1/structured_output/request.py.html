3import dataclasses
4import functools
5import json
6from concurrent.futures import Future
7from concurrent.futures._base import TimeoutError
8from typing import TYPE_CHECKING, Any, cast
9
10from vllm.sampling_params import SamplingParams, StructuredOutputsParams
11from vllm.v1.structured_output.backend_types import (
12 StructuredOutputGrammar,
13 StructuredOutputKey,
14 StructuredOutputOptions,
15)
16
17if TYPE_CHECKING:
18 from vllm.reasoning import ReasoningParser
19
20
21@dataclasses.dataclass
22class StructuredOutputRequest:
23 params: StructuredOutputsParams
24 _grammar: (
25 Future[StructuredOutputGrammar] | StructuredOutputGrammar | Exception | None
26 ) = None
27 reasoning_ended: bool | None = None
28 reasoning_parser_kwargs: dict[str, Any] | None = None
29 # Cached per request; do not share reasoning parsers across requests because
30 # their behavior can depend on reasoning_parser_kwargs.
31 reasoner: "ReasoningParser | None" = None
32
33 @staticmethod
34 def from_sampling_params(
35 sampling_params: SamplingParams | None,
36 ) -> "StructuredOutputRequest | None":
37 if sampling_params is None:
38 return None
39 params = sampling_params.structured_outputs
40 if not params or params.all_constraints_none():
41 return None
42 return StructuredOutputRequest(params=params)
43
44 def _check_grammar_completion(self) -> bool:
45 if isinstance(self._grammar, Future):
46 try:
47 # We will check whether the future is ready within 100 us
48 self._grammar = self._grammar.result(timeout=0.0001)
49 except TimeoutError:
50 return False
51 except Exception as e:
52 self._grammar = e
53 return True
54
55 @property
56 def is_grammar_ready(self) -> bool:
57 return self._check_grammar_completion()
58
59 @property
60 def grammar(self) -> StructuredOutputGrammar | Exception | None:
61 if not self._check_grammar_completion():
62 return None
63 return cast(StructuredOutputGrammar | Exception | None, self._grammar)
64
65 @grammar.setter
66 def grammar(
67 self, grammar: StructuredOutputGrammar | Future[StructuredOutputGrammar]
68 ) -> None:
69 self._grammar = grammar
70
71 @functools.cached_property
72 def structured_output_key(self) -> StructuredOutputKey:
73 return get_structured_output_key(self.params)
74
75
76def get_structured_output_key(params: StructuredOutputsParams) -> StructuredOutputKey:
77 if params.json is not None:
78 if not isinstance(params.json, str):
79 json_str = json.dumps(params.json)
80 else:
81 json_str = params.json
82 return StructuredOutputOptions.JSON, json_str
83 if params.json_object:
84 return StructuredOutputOptions.JSON_OBJECT, ""
85 if params.regex is not None:
86 return StructuredOutputOptions.REGEX, params.regex
87 if params.choice is not None:
88 if not isinstance(params.choice, str):
89 json_str = json.dumps(params.choice)
90 else:
91 json_str = params.choice
92 return StructuredOutputOptions.CHOICE, json_str
93 if params.grammar is not None:
94 return StructuredOutputOptions.GRAMMAR, params.grammar
95 if params.structural_tag is not None:
96 return StructuredOutputOptions.STRUCTURAL_TAG, params.structural_tag
97 raise ValueError("No valid structured output parameter found")
