46def get_engine_process_shutdown_timeout(
47 request_timeout: float | None,
48 process_timeout: float | None,
49) -> float | None:
50 """Return the EngineCore process-manager shutdown timeout.
51
52 ``VllmConfig.shutdown_timeout`` controls how long in-flight requests may
53 drain. A value of zero therefore tells EngineCore to abort requests as soon
54 as it receives SIGTERM. The parent process manager still needs a separate
55 window in which the EngineCore can release device resources before it is
56 force-killed. ROCm teardown can take longer than the generic best-effort
57 window, and force-killing during teardown can leave VRAM resident.
58
59 ``process_timeout`` may be a remaining budget computed by an outer process
60 manager. Keep it unchanged unless both values are zero: a zero remaining
61 budget for a positive request timeout must not receive a fresh grace period
62 because EngineCore relies on that deadline to enforce request draining.
63 """
64 if request_timeout == 0 and process_timeout == 0 and current_platform.is_rocm():
65 return ROCM_ENGINE_PROCESS_SHUTDOWN_TIMEOUT_S
66 return process_timeout

127def _node_ip_from_resources(node_resources: dict) -> str | None:
128 """Return the node IP encoded in a Ray per-node resource dict, or None.
129
130 Ray advertises each node's IP as a ``node:<ip>`` resource key. The head node
131 also carries ``node:__internal_head__``, and placement groups add
132 ``..._group_...`` keys; both are ignored.
133 """
134 for key in node_resources:
135 if (
136 key.startswith("node:")
137 and key != "node:__internal_head__"
138 and "_group_" not in key
139 ):
140 return key.split(":", 1)[1]
141 return None

144class CoreEngineProcManager:
145 """
146 Utility class to handle creation, readiness, and shutdown
147 of background processes used by the AsyncLLM and LLMEngine.
148 """
149
150 def __init__(
151 self,
152 local_engine_count: int,
153 start_index: int,
154 local_start_index: int,
155 vllm_config: VllmConfig,
156 local_client: bool,
157 handshake_address: str,
158 executor_class: type[Executor],
159 log_stats: bool,
160 client_handshake_address: str | None = None,
161 tensor_queue: Queue | None = None,
162 ):
163 self._request_shutdown_timeout = vllm_config.shutdown_timeout
164 context = get_mp_context()
165 common_kwargs = {
166 "vllm_config": vllm_config,
167 "local_client": local_client,
168 "handshake_address": handshake_address,
169 "executor_class": executor_class,
170 "log_stats": log_stats,
171 "tensor_queue": tensor_queue,
172 }
173
174 if client_handshake_address:
175 common_kwargs["client_handshake_address"] = client_handshake_address
176
177 is_dp = vllm_config.parallel_config.data_parallel_size > 1
178
179 from vllm.v1.engine.core import EngineCoreProc
180
181 self.processes: list[BaseProcess] = []
182 local_dp_ranks = []
183 for index in range(local_engine_count):
184 local_index = local_start_index + index
185 global_index = start_index + index
186
187 # Start EngineCore in background process.
188 local_dp_ranks.append(local_index)
189 self.processes.append(
190 context.Process(
191 target=EngineCoreProc.run_engine_core,
192 name=f"EngineCore_DP{global_index}" if is_dp else "EngineCore",
193 kwargs=common_kwargs
194 | {"dp_rank": global_index, "local_dp_rank": local_index},
195 )
196 )
197
198 self._finalizer = weakref.finalize(self, shutdown, self.processes)
199 self.manager_stopped = threading.Event()
200 self.failed_proc_name: str | None = None
201
202 # All ranks share this config object: capture the user-provided
203 # --device-ids list before the per-rank shard overwrites it. Mutating
204 # the config before each proc.start() works because the spawn method
205 # pickles process args at start() time, sequentially per rank.
206 user_assigned_gpu_ids = vllm_config.parallel_config.assigned_physical_gpu_ids
207 try:
208 for proc, local_dp_rank in zip(self.processes, local_dp_ranks):
209 # Populate the logical-to-physical GPU mapping in DP for
210 # platforms that cannot rely on
211 # torch.accelerator.set_device_index(), and for Ray.
212 needs_device_env_isolation = not (
213 current_platform.is_cuda_alike() or current_platform.is_xpu()
214 )
215 if is_dp and (
216 needs_device_env_isolation or vllm_config.parallel_config.use_ray
217 ):
218 set_assigned_physical_gpu_ids_for_dp_rank(
219 vllm_config, local_dp_rank, user_assigned_gpu_ids
220 )
221
222 with numa_utils.configure_subprocess(
223 # EngineCore itself does not have a TP/PP-local rank.
224 # When DP is enabled, set_assigned_physical_gpu_ids_for_dp_rank()
225 # populates the logical-to-physical mapping for this DP
226 # shard, so local_rank=0 means "the first local GPU in
227 # this shard". The actual TP/PP worker processes spawned
228 # by the executor are bound separately with their own
229 # local_rank values.
230 vllm_config,
231 local_rank=0,
232 dp_local_rank=local_dp_rank,
233 process_kind="EngineCore",
234 ):
235 proc.start()
236 finally:
237 # Kill other procs if not all are running.
238 if self.finished_procs():
239 self.shutdown()
240
241 def shutdown(self, timeout: float | None = None) -> None:
242 """Shutdown engine core processes with configurable timeout."""
243 self.manager_stopped.set()
244 if self._finalizer.detach() is not None:
245 process_timeout = get_engine_process_shutdown_timeout(
246 self._request_shutdown_timeout, timeout
247 )
248 if process_timeout != timeout:
249 logger.info(
250 "[shutdown] EngineCore process manager: using %ss ROCm "
251 "cleanup grace after immediate request abort",
252 process_timeout,
253 )
254 shutdown(self.processes, timeout=process_timeout)
255
256 def monitor_engine_liveness(self) -> None:
257 """Monitor engine core process liveness."""
258
259 sentinel_to_proc = {proc.sentinel: proc for proc in self.processes}
260 sentinels = set(sentinel_to_proc.keys())
261
262 while sentinels and not self.manager_stopped.is_set():
263 died_sentinels = connection.wait(sentinels, timeout=1)
264
265 for sentinel in died_sentinels:
266 proc = sentinel_to_proc.pop(cast(int, sentinel))
267 exitcode = proc.exitcode
268 if exitcode != 0 and not self.manager_stopped.is_set():
269 self.failed_proc_name = proc.name
270 if died_sentinels:
271 break
272
273 self.shutdown()
274
275 def sentinels(self) -> list:
276 return [proc.sentinel for proc in self.processes]
277
278 def finished_procs(self) -> dict[str, int]:
279 """Returns dict of proc name -> exit code for any finished procs."""
280 return {
281 proc.name: proc.exitcode
282 for proc in self.processes
283 if proc.exitcode is not None
284 }

287class SignalCallback:
288 """Safely trigger a callback from signal handler context via a dedicated thread."""
289
290 def __init__(self, callback: Callable[[], None]):
291 self._callback = callback
292 self._event = threading.Event()
293 self._stopped = False
294 self._thread = threading.Thread(
295 target=self._run,
296 daemon=True,
297 name="signal-callback",
298 )
299 self._thread.start()
300
301 def _run(self):
302 self._event.wait()
303 if not self._stopped:
304 self._callback()
305
306 def trigger(self):
307 self._event.set()
308
309 def stop(self):
310 self._stopped = True
311 self._event.set()

314def set_assigned_physical_gpu_ids_for_dp_rank(
315 vllm_config: VllmConfig,
316 local_dp_rank: int,

328 local_world_size = vllm_config.parallel_config.local_world_size
329 evar = current_platform.device_control_env_var

334 world_size,

337 )
338 vllm_config.parallel_config.assigned_physical_gpu_ids = physical_gpu_ids

341def get_physical_gpu_ids_for_local_dp_rank(
342 device_control_env_var: str,
343 local_dp_rank: int,
344 world_size: int,
345 local_world_size: int | None = None,
346 user_assigned_gpu_ids: list[int] | None = None,
347) -> list[int]:
348 """
349 Returns list of physical GPU IDs for the specified
350 data parallel rank.
351
352 For example, if world_size=2 and local_dp_rank=1, and there are 4 devices,
353 this will return [2, 3] for local_dp_rank=1.
354
355 If user_assigned_gpu_ids is provided (e.g. from --device-ids), this DP
356 rank's shard is sliced from it instead of being derived from the
357 device-control env var.
358 """
359 if local_world_size is None:
360 local_world_size = world_size
361 if user_assigned_gpu_ids is not None:
362 start = local_dp_rank * world_size
363 stop = start + local_world_size
364 if stop > len(user_assigned_gpu_ids):
365 raise ValueError(
366 f"--device-ids provides {len(user_assigned_gpu_ids)} devices, "
367 f"but DP rank {local_dp_rank} needs devices [{start}, {stop})"
368 )
369 return user_assigned_gpu_ids[start:stop]
370 try:
371 return [
372 current_platform.device_id_to_physical_device_id(i)
373 for i in range(
374 local_dp_rank * world_size,
375 local_dp_rank * world_size + local_world_size,
376 )
377 ]
378 except IndexError as e:
379 raise Exception(
380 f"Error computing device indices for "
381 f"{device_control_env_var}: "
382 f"local range: [{local_dp_rank * world_size}, "
383 f"{(local_dp_rank + 1) * world_size}) "
384 "base value: "
385 f'"{os.getenv(device_control_env_var)}"'
386 ) from e

551 @staticmethod
552 def create_dp_placement_groups(
553 vllm_config: VllmConfig,
554 ) -> tuple[list["PlacementGroup"], list[int]]:
555 """
556 Create placement groups for data parallel.
557 """
558
559 import ray
560 from ray._private.state import available_resources_per_node
561
562 logger.info("Creating placement groups for data parallel")
563 dp_master_ip = vllm_config.parallel_config.data_parallel_master_ip
564 dp_size = vllm_config.parallel_config.data_parallel_size
565 dp_size_local = vllm_config.parallel_config.data_parallel_size_local
566
567 available_resources = available_resources_per_node()
568 world_size = vllm_config.parallel_config.world_size
569 placement_groups: list[PlacementGroup] = []
570 local_dp_ranks: list[int] = []
571
572 dp_master_ip_key = f"node:{dp_master_ip}"
573 nodes = sorted(
574 available_resources.values(), key=lambda x: dp_master_ip_key not in x
575 )
576 assert len(nodes) > 0, "No nodes with resources found in Ray cluster."
577 assert dp_master_ip_key in nodes[0], (
578 f"The DP master node (ip: {dp_master_ip}) is missing or dead"
579 )
580
581 # optionally restrict DP placement to a caller-provided node set.
582 requested_node_ips = {
583 ip.strip()
584 for ip in envs.VLLM_RAY_DP_PLACEMENT_NODE_IPS.split(",")
585 if ip.strip()
586 }
587 if requested_node_ips:
588 allowed_node_ips = set(requested_node_ips)
589 # The master node must host the local ranks, so it has to be allowed.
590 if dp_master_ip not in allowed_node_ips:
591 allowed_node_ips.add(dp_master_ip)
592 filtered_nodes = [
593 node_resources
594 for node_resources in nodes
595 if _node_ip_from_resources(node_resources) in allowed_node_ips
596 ]
597 logger.info(
598 "VLLM_RAY_DP_PLACEMENT_NODE_IPS set; restricting DP placement "
599 "from %d to %d node(s): %s",
600 len(nodes),
601 len(filtered_nodes),
602 sorted(allowed_node_ips),
603 )
604 nodes = filtered_nodes
605
606 device_str = current_platform.ray_device_key
607 n_node_devices: list[int] = [
608 int(node_resources[device_str])
609 for node_resources in nodes
610 if device_str in node_resources
611 ]
612 assert n_node_devices, f"No {device_str} found in Ray cluster."
613 max_device_per_node = max(n_node_devices)
614
615 pack_strategy = envs.VLLM_RAY_DP_PACK_STRATEGY
616 _supported_pack_strategies = ("strict", "fill", "span")
617 if pack_strategy not in _supported_pack_strategies:
618 raise ValueError(
619 f"{envs.VLLM_RAY_DP_PACK_STRATEGY} is not supported. "
620 "Make sure to set `VLLM_RAY_DP_PACK_STRATEGY` "
621 f"to one of {_supported_pack_strategies}"
622 )
623
624 all2all_backend = vllm_config.parallel_config.all2all_backend
625 if pack_strategy == "fill" and (
626 all2all_backend == "deepep_high_throughput"
627 or all2all_backend == "deepep_low_latency"
628 ):
629 raise ValueError(
630 "DeepEP kernels require EP ranks [0,7] (same for [8,15], ...) "
631 "to be on the same node, but VLLM_RAY_DP_PACK_STRATEGY=fill "
632 "does not guarantee that. "
633 "Please use VLLM_RAY_DP_PACK_STRATEGY=strict instead."
634 )
635
636 if pack_strategy in ("strict", "fill"):
637 placement_strategy = "STRICT_PACK"
638 else:
639 placement_strategy = "PACK"
640 assert world_size > max_device_per_node, (
641 f"World size {world_size} is smaller than the "
642 "maximum number of devices per node "
643 f"{max_device_per_node}. Make sure to set "
644 "`VLLM_RAY_DP_PACK_STRATEGY` to `strict` or `fill`"
645 )
646
647 # if we need multiple nodes per dp group, we require for now that
648 # available nodes are homogeneous
649 assert set(n_node_devices) == {max_device_per_node}, (
650 f"Nodes are not homogeneous, {nodes}"
651 )
652 assert world_size % max_device_per_node == 0, (
653 f"For multi-node data parallel groups, world_size ({world_size}) must "
654 f"be a multiple of number of devices per node ({max_device_per_node})."
655 )
656 assert len(n_node_devices) * max_device_per_node >= world_size * dp_size, (
657 f"Not enough total available nodes ({len(n_node_devices)}) "
658 f"and devices per node ({max_device_per_node}) "
659 f"to satisfy required world size {world_size} and data parallel size "
660 f"{dp_size}"
661 )

1310 for fd, proc in frontend_process_by_fd.items()
1311 if proc.exitcode is not None
1312 or any(event_fd == fd for event_fd, _ in events)
1313 }
1314 if failed_frontend_procs and not finished:
1315 raise RuntimeError(
1316 "Frontend process failed during engine core initialization. "
1317 "See root cause above. "
1318 f"Failed frontend proc(s): {failed_frontend_procs}"
1319 )
1320 raise RuntimeError(
1321 "Engine core initialization failed. "
1322 "See root cause above. "
1323 f"Failed core proc(s): {finished}"
1324 + (
1325 f", failed frontend proc(s): {failed_frontend_procs}"
1326 if failed_frontend_procs
1327 else ""
1328 )
1329 )
1330
1331 # Receive HELLO and READY messages from the input socket.
1332 eng_identity, ready_msg_bytes = handshake_socket.recv_multipart()
1333 eng_index = int.from_bytes(eng_identity, "little")
1334 engine = next((e for e in core_engines if e.identity == eng_identity), None)
1335 if engine is None:
1336 raise RuntimeError(
1337 f"Message from engine with unexpected data parallel rank: {eng_index}"
1338 )
1339 msg = msgspec.msgpack.decode(ready_msg_bytes)
1340 status, local, headless = msg["status"], msg["local"], msg["headless"]
1341 if local != engine.local:
1342 raise RuntimeError(
1343 f"{status} message from "
1344 f"{'local' if local else 'remote'} "
1345 f"engine {eng_index}, expected it to be "
1346 f"{'local' if engine.local else 'remote'}"
1347 )
1348
1349 # Remote engines must be headless iff we aren't in hybrid dp lb mode.
1350 if not local and headless != remote_should_be_headless:
1351 if headless:
1352 raise RuntimeError(
1353 f"Remote engine {eng_index} must not use "
1354 f"--headless in external or hybrid dp lb "
1355 f"mode"
1356 )
1357 else:
1358 raise RuntimeError(
1359 f"Remote engine {eng_index} must use "
1360 f"--headless unless in external or hybrid "
1361 f"dp lb mode"
1362 )
1363
1364 if status == "HELLO" and engine.state == CoreEngineState.NEW:
1365 # Send init message with DP config info.
1366 init_message = msgspec.msgpack.encode(
1367 EngineHandshakeMetadata(
1368 addresses=launch.addresses,
1369 parallel_config={
1370 k: getattr(parallel_config, k)
1371 for k in (
1372 "data_parallel_master_ip",
1373 "data_parallel_master_port",
1374 "_data_parallel_master_port_list",
1375 "data_parallel_size",
1376 )
1377 }
1378 if coordinated_dp
1379 else {},
1380 )
1381 )
1382 handshake_socket.send_multipart((eng_identity, init_message), copy=False)
1383 conn_pending[0 if local else 1] -= 1
1384 start_pending[0 if local else 1] += 1
1385 engine.state = CoreEngineState.CONNECTED
1386 elif status == "READY" and engine.state == CoreEngineState.CONNECTED:
1387 # Validate config hash consistency across DP workers for MoE models.
1388 if coordinated_dp:
1389 worker_config_hash = msg.get("parallel_config_hash")
1390 expected_hash = parallel_config.compute_hash()
1391 if worker_config_hash != expected_hash:
1392 raise RuntimeError(
1393 f"Configuration mismatch detected for engine "
1394 f"{eng_index}. All DP workers must have identical "
1395 f"configurations for parameters that affect collective "
1396 f"communication (e.g., enable_eplb, "
1397 f"eplb_config.log_balancedness). "
1398 f"Worker hash: {worker_config_hash}, "
1399 f"Expected hash: {expected_hash}. "
1400 f"Please ensure all workers are started with the same "
1401 f"command-line arguments."
1402 )
1403
1404 start_pending[0 if local else 1] -= 1
1405 engine.state = CoreEngineState.READY
1406 else:
1407 raise RuntimeError(
1408 f"Unexpected {status} message for "
1409 f"{'local' if local else 'remote'} engine "
1410 f"{eng_index} in {engine.state} state."
1411 )
1412
1413 logger.debug(
1414 "%s from %s core engine process %s.",
1415 status,
1416 "local" if local else "remote",
1417 eng_index,
1418 )
