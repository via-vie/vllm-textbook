51 def __init__(
52 self,
53 vllm_config: VllmConfig,
54 executor_class: type[Executor],
55 log_stats: bool,
56 aggregate_engine_logging: bool = False,
57 usage_context: UsageContext = UsageContext.ENGINE_CONTEXT,
58 stat_loggers: list[StatLoggerFactory] | None = None,
59 mm_registry: MultiModalRegistry = MULTIMODAL_REGISTRY,
60 multiprocess_mode: bool = False,
61 ) -> None:
62 self.vllm_config = vllm_config
63 self.model_config = vllm_config.model_config
64 self.observability_config = vllm_config.observability_config
65
66 tracing_endpoint = self.observability_config.otlp_traces_endpoint
67 if tracing_endpoint is not None:
68 init_tracer("vllm.llm_engine", tracing_endpoint)
69
70 self.log_stats = log_stats
71
72 parallel_config = vllm_config.parallel_config
73 executor_backend = parallel_config.distributed_executor_backend
74
75 self.external_launcher_dp = (
76 parallel_config.data_parallel_size > 1
77 and executor_backend == "external_launcher"
78 )
79 # important: init dp group before init the engine_core
80 # In the decoupled engine case this is handled in EngineCoreProc.
81 if (
82 not multiprocess_mode
83 and parallel_config.data_parallel_size > 1
84 and not self.external_launcher_dp
85 ):
86 self.dp_group = parallel_config.stateless_init_dp_group()
87 else:
88 self.dp_group = None
89 self.should_execute_dummy_batch = False
90
91 self.renderer = renderer = renderer_from_config(self.vllm_config)
92
93 # Convert EngineInput --> EngineCoreRequest.
94 self.input_processor = InputProcessor(self.vllm_config, renderer)
95
96 # Converts EngineCoreOutputs --> RequestOutput.
97 self.output_processor = OutputProcessor(
98 renderer.tokenizer,
99 log_stats=self.log_stats,
100 stream_interval=self.vllm_config.scheduler_config.stream_interval,
101 tracing_enabled=tracing_endpoint is not None,
102 )
103
104 # EngineCore (gets EngineCoreRequests and gives EngineCoreOutputs)
105 # Hand the renderer to the client. In multiprocess mode the client
106 # starts the MM warmup only after engine-core fork (the why is in
107 # BaseRenderer.start_mm_warmup_in_background); InprocClient takes no
108 # renderer, so MM warmup stays inside renderer.warmup() there.
109 self.engine_core = EngineCoreClient.make_client(
110 multiprocess_mode=multiprocess_mode,
111 asyncio_mode=False,
112 vllm_config=vllm_config,
113 executor_class=executor_class,
114 log_stats=self.log_stats,
115 renderer=renderer,
116 )
117
118 self.logger_manager: StatLoggerManager | None = None
119 if self.log_stats:
120 self.logger_manager = StatLoggerManager(
121 vllm_config=vllm_config,
122 custom_stat_loggers=stat_loggers,
123 enable_default_loggers=log_stats,
124 aggregate_engine_logging=aggregate_engine_logging,
125 )
126 self.logger_manager.log_engine_initialized()
127
128 if not multiprocess_mode:
129 # for v0 compatibility
130 self.model_executor = self.engine_core.engine_core.model_executor # type: ignore
131
132 # Capture the model while reachable so the finalizer can drop the
133 # bytecode hooks pinning it (frees GPU memory on engine deletion).
134 model = self._get_driver_model_for_cleanup()
135 if model is not None:
136 self._finalizer = weakref.finalize(
137 self, LLMEngine._cleanup_instance_caches, weakref.ref(model)
138 )
139
140 if self.external_launcher_dp:
141 # If we use DP in external launcher mode, we reuse the
142 # existing DP group used for data communication.
143 self.dp_group = get_dp_group().cpu_group
144
145 # Don't keep the dummy data in memory
146 self.reset_mm_cache()

223 def add_request(
224 self,
225 request_id: str,
226 prompt: EngineCoreRequest | PromptType | EngineInput,
227 params: SamplingParams | PoolingParams,
228 arrival_time: float | None = None,
229 lora_request: LoRARequest | None = None,
230 tokenization_kwargs: dict[str, Any] | None = None,
231 trace_headers: Mapping[str, str] | None = None,
232 priority: int = 0,
233 session_id: str | None = None,
234 prompt_text: str | None = None,
235 ) -> str:
236 # Validate the request_id type.
237 if not isinstance(request_id, str):
238 raise TypeError(f"request_id must be a string, got {type(request_id)}")
239
240 # Process raw inputs into the request.
241 if isinstance(prompt, EngineCoreRequest):
242 logger.warning_once(
243 "Passing EngineCoreRequest to LLMEngine.generate() and .add_requests() "
244 "is deprecated and will be removed in the future. You should "
245 "instead pass the outputs of Renderer.render_cmpl() or "
246 "Renderer.render_chat()."
247 )
248
249 request = prompt
250 if request_id != request.request_id:
251 logger.warning_once(
252 "LLMEngine.add_request() was passed a request_id parameter that "
253 "does not match the EngineCoreRequest.request_id attribute. The "
254 "latter will be used, and the former will be ignored."
255 )
256 else:
257 request = self.input_processor.process_inputs(
258 request_id,
259 prompt,
260 params,
261 supported_tasks=self.get_supported_tasks(),
262 arrival_time=arrival_time,
263 lora_request=lora_request,
264 tokenization_kwargs=tokenization_kwargs,
265 trace_headers=trace_headers,
266 priority=priority,
267 session_id=session_id,
268 )
269 prompt_text, _, _ = extract_prompt_components(self.model_config, prompt)
270
271 self.input_processor.assign_request_id(request)
272
273 req_id = request.request_id
274
275 # Use cloned params that may have been updated in process_inputs()
276 params = request.params
277
278 n = params.n if isinstance(params, SamplingParams) else 1
279
280 if n == 1:
281 # Make a new RequestState and queue.
282 self.output_processor.add_request(request, prompt_text, None, 0)
283 # Add the request to EngineCore.
284 self.engine_core.add_request(request)
285 return req_id
286
287 # Fan out child requests (for n>1).
288 parent_req = ParentRequest(request)
289 for idx in range(n):
290 request_id, child_params = parent_req.get_child_info(idx)
291 child_request = request if idx == n - 1 else copy(request)
292 child_request.request_id = request_id
293 child_request.sampling_params = child_params
294
295 # Make a new RequestState and queue.
296 self.output_processor.add_request(
297 child_request, prompt_text, parent_req, idx
298 )
299 # Add the request to EngineCore.
300 self.engine_core.add_request(child_request)
301
302 return req_id

304 def step(self) -> list[RequestOutput | PoolingRequestOutput]:
305 if self.should_execute_dummy_batch:
306 self.should_execute_dummy_batch = False
307 self.engine_core.execute_dummy_batch()
308 return []
309
310 # 1) Get EngineCoreOutput from the EngineCore.
311 with record_function_or_nullcontext("llm_engine step: get_output"):
312 outputs = self.engine_core.get_output()
313
314 # 2) Process EngineCoreOutputs.
315 with record_function_or_nullcontext("llm_engine step: process_outputs"):
316 iteration_stats = (
317 IterationStats() if self.log_stats and outputs.outputs else None
318 )
319 processed_outputs = self.output_processor.process_outputs(
320 outputs.outputs,
321 engine_core_timestamp=outputs.timestamp,
322 iteration_stats=iteration_stats,
323 )
324 self.output_processor.update_scheduler_stats(outputs.scheduler_stats)
325
326 # 3) Abort any reqs that finished due to stop strings.
327 with record_function_or_nullcontext("llm_engine step: abort_requests"):
328 self.engine_core.abort_requests(processed_outputs.reqs_to_abort)
329
330 # 4) Record stats
331 with record_function_or_nullcontext("llm_engine step: record_stats"):
332 if self.logger_manager is not None and outputs.scheduler_stats is not None:
333 # Record even when this step produced no request outputs.
334 self.logger_manager.record(
335 scheduler_stats=outputs.scheduler_stats,
336 iteration_stats=iteration_stats,
337 mm_cache_stats=self.renderer.stat_mm_cache(),
338 )
339 if outputs.outputs:
340 self.do_log_stats_with_interval()
341
342 return processed_outputs.request_outputs

454 @staticmethod
455 def _cleanup_instance_caches(model_ref: "weakref.ref[nn.Module]") -> None:
456 """Remove the bytecode hooks that pin the compiled model."""
457 from vllm.compilation.wrapper import TorchCompileWithNoGuardsWrapper
458
459 model = model_ref()
460 if model is None:
461 return
462 for module in model.modules():
463 if isinstance(module, TorchCompileWithNoGuardsWrapper):
464 module.cleanup()
