29@dataclass
30class LogprobsProcessor:
31 # Tokenizer for this request,
32 # None if detokenization is disabled.
33 tokenizer: TokenizerLike | None
34
35 # Logprobs for this request
36 logprobs: SampleLogprobs | None
37 prompt_logprobs: PromptLogprobs | None
38 cumulative_logprob: float | None
39 num_logprobs: int | None
40 num_prompt_logprobs: int | None

42 @classmethod
43 def from_new_request(
44 cls,
45 tokenizer: TokenizerLike | None,
46 request: EngineCoreRequest,
47 ) -> "LogprobsProcessor":
48 sampling_params = request.sampling_params
49 assert sampling_params is not None
50 num_logprobs = sampling_params.num_logprobs
51 num_prompt_logprobs = sampling_params.prompt_logprobs
52 return cls(
53 tokenizer=tokenizer,
54 cumulative_logprob=(None if num_logprobs is None else 0.0),
55 logprobs=(
56 None
57 if num_logprobs is None
58 else create_sample_logprobs(sampling_params.flat_logprobs)
59 ),
60 prompt_logprobs=(
61 None
62 if num_prompt_logprobs is None
63 else create_prompt_logprobs(sampling_params.flat_logprobs)
64 ),
65 num_prompt_logprobs=num_prompt_logprobs,
66 num_logprobs=num_logprobs,
67 )

69 def _update_sample_logprobs(self, logprobs_lists: LogprobsLists) -> None:
70 """Update with sample logprobs from EngineCore.
71
72 Outer lists are only of len > 1 if EngineCore made
73 >1 tokens in prior step (e.g. in spec decoding).
74
75 Args:
76 logprobs_lists: the lists of logprob tokens, logprobs, and ranks.
77
78 """
79
80 assert self.num_logprobs is not None
81 assert self.logprobs is not None
82 assert self.cumulative_logprob is not None
83
84 token_ids_lst, logprobs_lst, ranks_lst, _ = logprobs_lists
85
86 for rank_np, logprobs_np, token_ids_np in zip(
87 ranks_lst, logprobs_lst, token_ids_lst
88 ):
89 rank = rank_np.tolist()
90 logprobs = logprobs_np.tolist()
91 token_ids = token_ids_np.tolist()
92 # Detokenize (non-incrementally).
93 decoded_tokens: list[str] | Iterable[None]
94 if self.tokenizer is None:
95 decoded_tokens = NONES
96 else:
97 decoded_tokens_list = convert_ids_list_to_tokens(
98 self.tokenizer, token_ids
99 )
100 context_token_ids = self._get_sampled_context_ids(self.logprobs)
101 decoded_tokens = self._verify_tokens(
102 decoded_tokens_list=decoded_tokens_list,
103 tokens=token_ids,
104 context_token_ids=context_token_ids,
105 )
106
107 # Sampler puts the sampled logprob in first.
108 sampled_token_logprob = logprobs[0]
109 self.cumulative_logprob += sampled_token_logprob
110
111 # Update with the Logprob container for this pos.
112 append_logprobs_for_next_position(
113 self.logprobs,
114 token_ids,
115 logprobs,
116 decoded_tokens,
117 rank,
118 self.num_logprobs,
119 )

121 def _update_prompt_logprobs(
122 self,
123 prompt_logprobs_tensors: LogprobsTensors,
124 ) -> None:
125 """Update with prompt logprobs from EngineCore.
126
127 Args:
128 prompt_logprobs_tensors: tuple containing the prompt logprobs
129 tensors.
130
131 """
132
133 # Prompt logprobs are enabled.
134 assert self.num_prompt_logprobs is not None
135 assert self.prompt_logprobs is not None
136
137 token_ids, logprobs, ranks, *_ = prompt_logprobs_tensors
138
139 # Recover shapes.
140 num_prompt_tokens, num_logprobs = logprobs.shape
141
142 # Detokenize non-incrementally.
143 # Output is flat: [num_tok, num_lps] -> [num_tok * num_lps]
144 all_decoded_tokens: list[str] | None = (
145 None
146 if self.tokenizer is None
147 else convert_ids_list_to_tokens(
148 self.tokenizer, token_ids.flatten().tolist()
149 )
150 )
151
152 # Pythonize the torch tensors.
153 prompt_token_ranks = ranks.tolist()
154 prompt_logprobs = logprobs.tolist()
155 token_ids_list = token_ids.tolist()
156
157 # Make Logprob for each position.
158 for pos in range(num_prompt_tokens):
159 # Handle flattening and UTF-8 correction per position
160 offset = pos * num_logprobs
161 offset_end = offset + num_logprobs
162
163 decoded_tokens_for_pos: list[str] | Iterable[None]
164 if all_decoded_tokens is None:
165 decoded_tokens_for_pos = NONES
166 else:
167 # Extract decoded tokens for this position
168 decoded_tokens_slice = all_decoded_tokens[offset:offset_end]
169 # Context: preceding prompt tokens accumulated in
170 # self.prompt_logprobs from previous loop iterations.
171 context_token_ids = self._get_sampled_context_ids(self.prompt_logprobs)
172 # Apply UTF-8 correction within this position's token boundaries
173 decoded_tokens_for_pos = self._verify_tokens(
174 decoded_tokens_list=decoded_tokens_slice,
175 tokens=token_ids_list[pos],
176 context_token_ids=context_token_ids,
177 )
178
179 # Update with the Logprob container for this pos.
180 append_logprobs_for_next_position(
181 self.prompt_logprobs,
182 token_ids_list[pos],
183 prompt_logprobs[pos],
184 decoded_tokens_for_pos,
185 prompt_token_ranks[pos],
186 self.num_prompt_logprobs,
187 )

189 def pop_prompt_logprobs(self) -> PromptLogprobs | None:
190 """Pop and return all request prompt logprobs
191
192 The logprobs processor aggregates prompt chunk logprobs
193 over one or more prefill chunks. This method returns
194 all prompt logprobs at once and then forgets them.
195 Ensures correct RequestOutputKind.DELTA semantics
196 wherein all prompt logprobs are returned at once at
197 the end of prefill.
198
199 Returns:
200 None if prompt logprobs are disabled for this request.
201 List of all prompt logprobs, otherwise.
202 """
203 plp = self.prompt_logprobs
204 if plp:
205 self.prompt_logprobs = []
206 return plp

208 @staticmethod
209 def _get_sampled_context_ids(
210 logprobs_source: SampleLogprobs | PromptLogprobs | None,
211 max_context: int = 4,
212 ) -> list[int]:
213 """Extract recent sampled token IDs from a logprobs source.
214
215 The sampled (or prompt) token at each position is the first
216 entry, since it is always inserted first by
217 append_logprobs_for_next_position.
218
219 Args:
220 logprobs_source: The logprobs container to extract from.
221 max_context: Maximum number of preceding tokens to return.
222 4 is sufficient for any UTF-8 multi-byte sequence.
223
224 Returns:
225 List of sampled token IDs, oldest first, most recent last.
226 """
227 if not logprobs_source:
228 return []
229
230 n = len(logprobs_source)
231 start = max(0, n - max_context)
232
233 # Efficient path for FlatLogprobs: access token_ids directly.
234 if isinstance(logprobs_source, FlatLogprobs):
235 return [
236 logprobs_source.token_ids[logprobs_source.start_indices[i]]
237 for i in range(start, n)
238 if logprobs_source.start_indices[i] < logprobs_source.end_indices[i]
239 ]
240
241 # list[dict] path
242 result: list[int] = []
243 for i in range(start, n):
244 entry = logprobs_source[i]
245 if entry is not None:
246 result.append(next(iter(entry)))
247 return result

249 def _correct_decoded_token(
250 self, token_id: int, context_token_ids: list[int]
251 ) -> str:
252 """Correct a decoded token that contains the replacement character.
253
254 When byte-fallback tokenization splits multi-byte UTF-8
255 characters across tokens, individual token decoding produces
256 the replacement character U+FFFD. This method uses preceding
257 sampled tokens as context to reconstruct the correct text.
258
259 Args:
260 token_id: The single token ID to correct.
261 context_token_ids: Preceding sampled token IDs in sequential
262 order (oldest first). These are the actual tokens in
263 the generated sequence, NOT top-k alternatives.
264
265 Returns:
266 The corrected decoded string, or empty string if the byte
267 sequence is genuinely incomplete at this point.
268 """
269 assert self.tokenizer is not None
270
271 max_ctx = min(len(context_token_ids), 4)
272
273 for num_ctx in range(1, max_ctx + 1):
274 context = context_token_ids[-num_ctx:]
275 full_decoded = self.tokenizer.decode(context + [token_id])
276
277 if full_decoded.endswith("�"):
278 continue
279
280 # Find the boundary between "clean" context tokens and
281 # byte-fallback tokens that are part of the same incomplete
282 # sequence. Byte-fallback context tokens returned "" when
283 # they were processed, so their text must be attributed to
284 # this completing token.
285 clean_end = len(context)
286 for j in range(len(context) - 1, -1, -1):
287 if self.tokenizer.decode([context[j]]).endswith("�"):
288 clean_end = j
289 else:
290 break
291
292 # Decode only the clean (non-byte-fallback) prefix.
293 if clean_end > 0:
294 clean_prefix = self.tokenizer.decode(context[:clean_end])
295 else:
296 clean_prefix = ""
297
298 if full_decoded.startswith(clean_prefix):
299 return full_decoded[len(clean_prefix) :]
300
301 # Tokenizer normalization may cause prefix mismatch.
302 # Find the longest common prefix between them.
303 common_len = 0
304 for a, b in zip(clean_prefix, full_decoded):
305 if a != b:
306 break
307 common_len += 1
308 return full_decoded[common_len:]
309
310 return ""

312 def _verify_tokens(
313 self,
314 decoded_tokens_list: list[str],
315 tokens: list[int],
316 context_token_ids: list[int] | None = None,
317 ) -> list[str]:
318 """Verify and correct decoded tokens with replacement characters.
319
320 Args:
321 decoded_tokens_list: Decoded token strings to verify.
322 tokens: Token IDs corresponding to decoded_tokens_list.
323 These are alternatives at the SAME position (e.g.
324 [sampled, top1, top2]), NOT sequential tokens.
325 context_token_ids: Preceding sampled token IDs providing
326 sequential context. If None, extracted from
327 self.logprobs.
328 """
329 if context_token_ids is None:
330 context_token_ids = self._get_sampled_context_ids(self.logprobs)
331
332 corrected_decoded_token_map = dict()
333 for idx, text in enumerate(decoded_tokens_list):
334 if text.endswith("�"):
335 # Replacement char at the end means a potential
336 # unfinished byte sequence from byte-fallback
337 # tokenization. Correct each token independently
338 # using only the sequential context.
339 corrected_decoded_token_map[idx] = self._correct_decoded_token(
340 tokens[idx], context_token_ids
341 )
342
343 for idx, text in corrected_decoded_token_map.items():
344 decoded_tokens_list[idx] = text
345
346 return decoded_tokens_list
