297 def check_admission(self, n: int = 1, request_id: str | None = None) -> None:
298 """Reject the request if it would exceed queue limits.
299
300 Both limits return HTTP 503 (Service Unavailable) so that load
301 balancers and client SDKs retry on a different instance.
302
303 - ``max_num_queued_reqs``: approximate global cap on the number of
304 unfinished requests (waiting + running). A request with ``n > 1``
305 counts as ``n`` slots.
306 - ``max_num_queued_tokens``: TTFT QoS — cap on the total prompt
307 tokens of requests still in prefill.
308
309 Note: ``get_num_queued_tokens`` uses ``prompt_len`` for all prefilling requests.
310 Chunked prefill progress and prefix-cache hits are not subtracted because the
311 scheduler's ``num_computed_tokens`` and ``num_cached_tokens`` are only
312 propagated to the API server after prefill completes. The overestimation is
313 conservative — earlier rejection, preserving TTFT targets.
314
315 Args:
316 n: Number of sequences the request will occupy.
317 request_id: Request id, used for logging only.
318
319 Raises:
320 QueueOverflowError: If ``max_num_queued_reqs`` would be exceeded.
321 MaxQueuedTokensError: If ``max_num_queued_tokens`` would be exceeded.
322 """
323 max_num_reqs = self.scheduler_config.max_num_queued_reqs
324 if max_num_reqs is not None:
325 current_requests = (
326 self.admission_stats.get_num_requests()
327 if self.admission_stats is not None
328 else self.get_num_unfinished_requests()
329 )
330 if current_requests + n > max_num_reqs:
331 logger.info(
332 "Request queue full - rejecting request %s "
333 "(current=%d, n=%d, max=%d).",
334 request_id,
335 current_requests,
336 n,
337 max_num_reqs,
338 )
339 raise QueueOverflowError()
340
341 max_queued_tokens = self.scheduler_config.max_num_queued_tokens
342 if max_queued_tokens is not None:
343 current_tokens = self.get_num_queued_tokens()
344 if current_tokens >= max_queued_tokens:
345 logger.info(
346 "Max queued tokens reached - rejecting request %s "
347 "(current_tokens=%d, max=%d).",
348 request_id,
349 current_tokens,
350 max_queued_tokens,
351 )
352 raise MaxQueuedTokensError()

361 async def add_request(
362 self,
363 request_id: str,
364 prompt: EngineCoreRequest
365 | PromptType
366 | EngineInput
367 | AsyncGenerator[StreamingInput, None],
368 params: SamplingParams | PoolingParams,
369 arrival_time: float | None = None,
370 lora_request: LoRARequest | None = None,
371 tokenization_kwargs: dict[str, Any] | None = None,
372 trace_headers: Mapping[str, str] | None = None,
373 priority: int = 0,
374 data_parallel_rank: int | None = None,
375 session_id: str | None = None,
376 prompt_text: str | None = None,
377 reasoning_ended: bool | None = None,
378 reasoning_parser_kwargs: dict[str, Any] | None = None,
379 ) -> RequestOutputCollector:
380 """Add new request to the AsyncLLM."""
381
382 if self.errored:
383 raise EngineDeadError()
384
385 is_pooling = isinstance(params, PoolingParams)
386
387 if (
388 self.vllm_config.cache_config.kv_sharing_fast_prefill
389 and not is_pooling
390 and params.prompt_logprobs
391 ):
392 raise VLLMValidationError(
393 "--kv-sharing-fast-prefill produces incorrect logprobs for "
394 "prompt tokens, please disable it when the requests need "
395 "prompt logprobs"
396 )
397
398 if isinstance(params, SamplingParams) and params.n > 1:
399 # TODO (NickLucche) Batch check admission check for all n requests
400 self.check_admission(params.n, request_id)
401
402 if isinstance(prompt, AsyncGenerator):
403 if reasoning_ended is not None or reasoning_parser_kwargs is not None:
404 raise NotImplementedError
405
406 # Streaming input case.
407 return await self._add_streaming_input_request(
408 request_id,
409 prompt,
410 params,
411 arrival_time,
412 lora_request,
413 tokenization_kwargs,
414 trace_headers,
415 priority,
416 data_parallel_rank,
417 session_id,
418 )
419
420 # Convert Input --> Request.
421 if isinstance(prompt, EngineCoreRequest):
422 logger.warning_once(
423 "Passing EngineCoreRequest to AsyncLLM.generate() and .add_requests() "
424 "is deprecated and will be removed in the future. You should "
425 "instead pass the outputs of Renderer.render_cmpl() or "
426 "Renderer.render_chat()."
427 )
428
429 request = prompt
430 if request_id != request.request_id:
431 logger.warning_once(
432 "AsyncLLM.add_request() was passed a request_id parameter that "
433 "does not match the EngineCoreRequest.request_id attribute. The "
434 "latter will be used, and the former will be ignored."
435 )
436 else:
437 if isinstance(prompt, dict) and "type" in prompt:
438 # Rendered EngineInput; no blocking preprocessing needed.
439 request = self.input_processor.process_inputs(
440 request_id,
441 prompt,
442 params,
443 supported_tasks=await self.get_supported_tasks(),
444 arrival_time=arrival_time,
445 lora_request=lora_request,
446 tokenization_kwargs=tokenization_kwargs,
447 trace_headers=trace_headers,
448 priority=priority,
449 data_parallel_rank=data_parallel_rank,
450 session_id=session_id,
451 )
452 else:
453 # Raw prompts require tokenization and possibly multimodal
454 # processing, which must not block the event loop.
455 request = await self.input_processor.process_inputs_async(
456 request_id,
457 prompt,
458 params,
459 supported_tasks=await self.get_supported_tasks(),
460 arrival_time=arrival_time,
461 lora_request=lora_request,
462 tokenization_kwargs=tokenization_kwargs,
463 trace_headers=trace_headers,
464 priority=priority,
465 data_parallel_rank=data_parallel_rank,
466 session_id=session_id,
467 )
468 prompt_text, _, _ = extract_prompt_components(self.model_config, prompt)
469
470 if reasoning_ended is not None:
471 request.reasoning_ended = reasoning_ended
472 if reasoning_parser_kwargs is not None:
473 request.reasoning_parser_kwargs = reasoning_parser_kwargs
474
475 self.input_processor.assign_request_id(request)
476
477 # We start the output_handler on the first call to add_request() so
478 # we can call __init__ before the event loop, which enables us
479 # to handle startup failure gracefully in the OpenAI server.
480 self._run_output_handler()
481
482 # Create a new output collector for the request.
483 queue = RequestOutputCollector(params.output_kind, request.request_id)
484
485 # Use cloned params that may have been updated in process_inputs()
486 params = request.params
487
488 if is_pooling or params.n == 1:
489 await self._add_request(request, prompt_text, None, 0, queue)
490 return queue
491
492 parent_params = params
493 assert isinstance(parent_params, SamplingParams)
494
495 # Fan out child requests (for n>1).
496 parent_request = ParentRequest(request)
497 for idx in range(parent_params.n):
498 request_id, child_params = parent_request.get_child_info(idx)
499 child_request = request if idx == parent_params.n - 1 else copy(request)
500 child_request.request_id = request_id
501 child_request.sampling_params = child_params
502 await self._add_request(
503 child_request, prompt_text, parent_request, idx, queue
504 )
505 return queue

507 async def _add_request(
508 self,
509 request: EngineCoreRequest,
510 prompt: str | None,
511 parent_req: ParentRequest | None,
512 index: int,
513 queue: RequestOutputCollector,
514 ):
515 if parent_req is None and not self.output_processor.has_request(
516 request.request_id
517 ):
518 self.check_admission(request_id=request.request_id)
519
520 # Register locally before the first await so concurrent tasks see this request.
521 self.output_processor.add_request(request, prompt, parent_req, index, queue)
522
523 # Add the EngineCoreRequest to EngineCore (separate process).
524 await self.engine_core.add_request_async(request)
525
526 if self.log_requests:
527 logger.info("Added request %s.", request.request_id)

529 async def _add_streaming_input_request(
530 self,
531 request_id: str,
532 input_stream: AsyncGenerator[StreamingInput, None],
533 sampling_params: SamplingParams | PoolingParams,
534 arrival_time: float | None = None,
535 lora_request: LoRARequest | None = None,
536 tokenization_kwargs: dict[str, Any] | None = None,
537 trace_headers: Mapping[str, str] | None = None,
538 priority: int = 0,
539 data_parallel_rank: int | None = None,
540 session_id: str | None = None,
541 ) -> RequestOutputCollector:
542 self._validate_streaming_input_sampling_params(sampling_params)
543
544 inputs = dict(
545 supported_tasks=await self.get_supported_tasks(),
546 arrival_time=arrival_time,
547 lora_request=lora_request,
548 tokenization_kwargs=tokenization_kwargs,
549 trace_headers=trace_headers,
550 priority=priority,
551 data_parallel_rank=data_parallel_rank,
552 session_id=session_id,
553 )
554
555 if not sampling_params.skip_clone:
556 sampling_params = sampling_params.clone()
557 sampling_params.skip_clone = True
558
559 # Create request for validation, also used as the finished signal
560 # once the input stream is closed.
561 final_req = self.input_processor.process_inputs(
562 request_id=request_id,
563 prompt=TokensPrompt(prompt_token_ids=[0]),
564 params=sampling_params,
565 **inputs, # type: ignore[arg-type]
566 )
567 self.input_processor.assign_request_id(final_req)
568 internal_req_id = final_req.request_id
569
570 queue = RequestOutputCollector(sampling_params.output_kind, internal_req_id)
571
572 async def handle_inputs():
573 cancelled = False
574 try:
575 async for input_chunk in input_stream:
576 sp = input_chunk.sampling_params
577 if sp:
578 self._validate_streaming_input_sampling_params(sp)
579 else:
580 sp = sampling_params
581 # TODO(nick): Avoid re-validating reused sampling parameters
582 req = self.input_processor.process_inputs(
583 request_id=internal_req_id,
584 prompt=input_chunk.prompt,
585 params=sp,
586 resumable=True,
587 **inputs, # type: ignore[arg-type]
588 )
589 req.external_req_id = request_id
590 if req.prompt_embeds is not None:
591 raise VLLMValidationError(
592 "prompt_embeds not supported for streaming inputs"
593 )
594 prompt_text, _, _ = extract_prompt_components(
595 self.model_config, input_chunk.prompt
596 )
597 await self._add_request(req, prompt_text, None, 0, queue)
598 except (asyncio.CancelledError, GeneratorExit):
599 cancelled = True
600 except Exception as error:
601 # Wrap in InputStreamError so generate() can propagate it
602 # without wrapping in EngineGenerateError.
603 queue.put(InputStreamError(error))
604 finally:
605 queue._input_stream_task = None
606 if not cancelled:
607 # Send empty final request to indicate that inputs have
608 # finished. Don't send if cancelled (session was aborted).
609 await self._add_request(final_req, None, None, 0, queue)
610
611 # Ensure output handler is running.
612 self._run_output_handler()
613
614 queue._input_stream_task = asyncio.create_task(handle_inputs())
615 return queue

617 @staticmethod
618 def _validate_streaming_input_sampling_params(
619 params: SamplingParams | PoolingParams,
620 ):
621 if (
622 not isinstance(params, SamplingParams)
623 or params.n > 1
624 or params.output_kind == RequestOutputKind.FINAL_ONLY
625 or params.stop
626 ):
627 raise VLLMValidationError(
628 "Input streaming not currently supported "
629 "for pooling models, n > 1, request_kind = FINAL_ONLY "
630 "or with stop strings."
631 )
632
633 # TODO: we should support multiple prompts in one call, as you
634 # can do with LLM.generate. So that for multi-prompt completion
635 # requests we don't need to send multiple messages to core proc,
636 # and so we don't need multiple streams which then get
637 # re-multiplexed in the API server anyhow.
638 async def generate(
639 self,
640 prompt: EngineCoreRequest
641 | PromptType
642 | EngineInput
643 | AsyncGenerator[StreamingInput, None],
644 sampling_params: SamplingParams,
645 request_id: str,
646 *,
647 prompt_text: str | None = None,
648 lora_request: LoRARequest | None = None,
649 tokenization_kwargs: dict[str, Any] | None = None,
650 trace_headers: Mapping[str, str] | None = None,
651 priority: int = 0,
652 data_parallel_rank: int | None = None,
653 session_id: str | None = None,
654 reasoning_ended: bool | None = None,
655 reasoning_parser_kwargs: dict[str, Any] | None = None,
656 ) -> AsyncGenerator[RequestOutput, None]:
657 """
658 Main function called by the API server to kick off a request
659 * 1) Making an AsyncStream corresponding to the Request.
660 * 2) Processing the Input.
661 * 3) Adding the Request to the Detokenizer.
662 * 4) Adding the Request to the EngineCore (separate process).
663
664 A separate output_handler loop runs in a background AsyncIO task,
665 pulling outputs from EngineCore and putting them into the
666 per-request AsyncStream.
667
668 The caller of generate() iterates the returned AsyncGenerator,
669 returning the RequestOutput back to the caller.
670
671 Note:
672 Passing a raw prompt string directly to this method is deprecated.
673 Advanced power-users can manually bypass the raw-prompt fallback
674 path using the Engine's underlying Renderer pipeline:
675
676 >>> from vllm.inputs import parse_model_prompt
677 >>> parsed = parse_model_prompt(self.model_config, "Prompt text")
678 >>> params = self.renderer.default_cmpl_tok_params
679 >>> (engine_input,) = self.renderer.render_cmpl([parsed], params)
680 >>> gen = self.generate(engine_input, sampling_params, request_id)
681 """
682
683 q: RequestOutputCollector | None = None
684 try:
685 q = await self.add_request(
686 request_id,
687 prompt,
688 sampling_params,
689 lora_request=lora_request,
690 tokenization_kwargs=tokenization_kwargs,
691 trace_headers=trace_headers,
692 priority=priority,
693 data_parallel_rank=data_parallel_rank,
694 session_id=session_id,
695 prompt_text=prompt_text,
696 reasoning_ended=reasoning_ended,
697 reasoning_parser_kwargs=reasoning_parser_kwargs,
698 )
699
700 # The output_handler task pushes items into the queue.
701 # This task pulls from the queue and yields to caller.
702 finished = False
703 while not finished:
704 # Note: drain queue without await if possible (avoids
705 # task switching under load which helps performance).
706 out = q.get_nowait() or await q.get()
707
708 # Note: both OutputProcessor and EngineCore handle their
709 # own request cleanup based on finished.
710 assert isinstance(out, RequestOutput)
711 finished = out.finished
712 if out is not STREAM_FINISHED:
713 yield out
714
715 # If the request is disconnected by the client, generate()
716 # is cancelled or the generator is garbage collected. So,
717 # we abort the request if we end up here.
718 except (asyncio.CancelledError, GeneratorExit):
719 if q is not None:
720 await self.abort(q.request_id, internal=True)
721 if self.log_requests:
722 logger.info("Request %s aborted.", request_id)
723 raise
724
725 # Engine is dead. Do not abort since we shut down.
726 except EngineDeadError:
727 if self.log_requests:
728 logger.info("Request %s failed (engine dead).", request_id)
729 raise
730
731 # Request validation error or admission control rejection.
732 except (VLLMClientError, GracefulHTTPError) as e:
733 if self.log_requests:
734 logger.info("Request %s failed (bad request): %s.", request_id, e)
735 raise
736
737 # Error from input stream generator - propagate directly.
738 except InputStreamError as e:
739 if q is not None:
740 await self.abort(q.request_id, internal=True)
741 if self.log_requests:
742 logger.info("Request %s failed (input error): %s.", request_id, e)
743 raise e.cause from e
744
745 # Unexpected error in the generate() task (possibly recoverable).
746 except Exception as e:
747 if q is not None:
748 await self.abort(q.request_id, internal=True)
749 if self.log_requests:
750 try:
751 s = f"{e.__class__.__name__}: {e}"
752 except Exception as e2:
753 s = (
754 f"{e.__class__.__name__}: "
755 "error during printing an exception of class"
756 + e2.__class__.__name__
757 )
758 logger.info("Request %s failed due to %s.", request_id, s)
759 raise EngineGenerateError() from e
760 finally:
761 if q is not None:
762 q.close()

764 def _run_output_handler(self):
765 """Background loop: pulls from EngineCore and pushes to AsyncStreams."""
766
767 if self.output_handler is not None:
768 return
769
770 # Ensure that the task doesn't have a circular ref back to the AsyncLLM
771 # object, or else it won't be garbage collected and cleaned up properly.
772 engine_core = self.engine_core
773 output_processor = self.output_processor
774 log_stats = self.log_stats
775 # We use a mutable list for logger_manager so that it can be updated
776 # during elastic EP scaling (see scale_elastic_ep) without creating
777 # a circular reference via self.
778 self._logger_ref = [self.logger_manager]
779 logger_ref = self._logger_ref
780 renderer = self.renderer
781 # P0 multi-modal sender ("shadow") cache; None for text-only models.
782 mm_processor_cache = renderer.mm_processor_cache
783 chunk_size = envs.VLLM_V1_OUTPUT_PROC_CHUNK_SIZE
784
785 async def output_handler():
786 try:
787 while True:
788 # 1) Pull EngineCoreOutputs from the EngineCore.
789 outputs = await engine_core.get_output_async()
790 num_outputs = len(outputs.outputs)
791
792 iteration_stats = (
793 IterationStats() if (log_stats and num_outputs) else None
794 )
795
796 # Split outputs into chunks of at most
797 # VLLM_V1_OUTPUT_PROC_CHUNK_SIZE, so that we don't block the
798 # event loop for too long.
799 engine_core_outputs = outputs.outputs
800 for start in range(0, num_outputs, chunk_size):
801 end = start + chunk_size
802 outputs_slice = engine_core_outputs[start:end]
803 # 2) Process EngineCoreOutputs.
804 processed_outputs = output_processor.process_outputs(
805 outputs_slice, outputs.timestamp, iteration_stats
806 )
807 # NOTE: RequestOutputs are pushed to their queues.
808 assert not processed_outputs.request_outputs
809
810 # 2b) Recover from P0/P1 cache drift: the engine flags hashes
811 # it couldn't find (mm_cache_miss_hashes); drop them from the
812 # P0 shadow so the client's retry resends the data and
813 # repopulates P1. Hot-path no-op (field is None otherwise).
814 if mm_processor_cache is not None:
815 for eco in outputs_slice:
816 if eco.mm_cache_miss_hashes:
817 for mm_hash in eco.mm_cache_miss_hashes:
818 mm_processor_cache.invalidate(mm_hash)
819
820 # Allow other asyncio tasks to run between chunks
821 if end < num_outputs:
822 await asyncio.sleep(0)
823
824 # 3) Abort any reqs that finished due to stop strings.
825 if processed_outputs.reqs_to_abort:
826 await engine_core.abort_requests_async(
827 processed_outputs.reqs_to_abort
828 )
829
830 output_processor.update_scheduler_stats(outputs.scheduler_stats)
831
832 # 4) Logging.
833 # TODO(rob): make into a coroutine and launch it in
834 # background thread once Prometheus overhead is non-trivial.
835 if logger_ref[0]:
836 logger_ref[0].record(
837 engine_idx=outputs.engine_index,
838 scheduler_stats=outputs.scheduler_stats,
839 iteration_stats=iteration_stats,
840 mm_cache_stats=renderer.stat_mm_cache(),
841 )
842 except Exception as e:
843 logger.exception("AsyncLLM output_handler failed.")
844 output_processor.propagate_error(e)
845
846 self.output_handler = asyncio.create_task(output_handler())

889 async def pause_generation(
890 self,
891 *,
892 mode: PauseMode = "abort",
893 wait_for_inflight_requests: bool | None = None,
894 clear_cache: bool = True,
895 ) -> None:
896 """
897 Pause generation to allow model weight updates.
898
899 All mode handling (abort / wait / keep) and cache clearing is done
900 in the engine. New generation/encoding requests will not be scheduled
901 until resume is called.
902
903 Args:
904 mode: How to handle in-flight requests:
905 - ``"abort"``: Abort all in-flight requests immediately
906 (default).
907 - ``"wait"``: Wait for in-flight requests to complete.
908 - ``"keep"``: Freeze requests in queue; they resume on
909 :meth:`resume_generation`.
910 wait_for_inflight_requests: DEPRECATED: use mode argument.
911 clear_cache: Whether to clear KV cache and prefix cache after
912 draining. Set to ``False`` to preserve cache for faster resume.
913 """
914 if wait_for_inflight_requests:
915 warnings.warn(
916 "The `wait_for_inflight_requests` parameter in "
917 "`AsyncLLM.pause_generation()` is deprecated. "
918 "Please use `mode` argument instead.",
919 DeprecationWarning,
920 stacklevel=2,
921 )
922 mode = "wait"
923 if clear_cache:
924 await self.renderer.clear_mm_cache_async()
925 await self.engine_core.pause_scheduler_async(mode=mode, clear_cache=clear_cache)
926 # Small sleep to help ensure that final outputs from any in-flight requests are
927 # returned prior to this method returning. These outputs come out of the engine
928 # prior to the wait-for-idle completion event, but involve additional async
929 # tasks in output processing.
930 # Note that this is not required for correctness, just more intuitive ordering
931 # of events from caller's pov.
932 await asyncio.sleep(0.02)
933
934 async def resume_generation(self) -> None:
935 """Resume generation after :meth:`pause_generation`."""
936 await self.engine_core.resume_scheduler_async()

942 async def encode(
943 self,
944 prompt: PromptType | EngineInput,

947 lora_request: LoRARequest | None = None,
948 trace_headers: Mapping[str, str] | None = None,
949 priority: int = 0,
950 tokenization_kwargs: dict[str, Any] | None = None,
951 reasoning_ended: bool | None = None,
952 ) -> AsyncGenerator[PoolingRequestOutput, None]:
953 """
954 Main function called by the API server to kick off a request
955 * 1) Making an AsyncStream corresponding to the Request.
956 * 2) Processing the Input.
957 * 3) Adding the Request to the EngineCore (separate process).
958
959 A separate output_handler loop runs in a background AsyncIO task,
960 pulling outputs from EngineCore and putting them into the
961 per-request AsyncStream.
962
963 The caller of generate() iterates the returned AsyncGenerator,
964 returning the RequestOutput back to the caller.
965 """
966
967 q: RequestOutputCollector | None = None
968 try:
969 q = await self.add_request(
970 request_id,
971 prompt,
972 pooling_params,
973 lora_request=lora_request,
974 tokenization_kwargs=tokenization_kwargs,
975 trace_headers=trace_headers,
976 priority=priority,
977 reasoning_ended=reasoning_ended,
978 )
979
980 # The output_handler task pushes items into the queue.
981 # This task pulls from the queue and yields to caller.
982 finished = False
983 while not finished:
984 # Note: drain queue without await if possible (avoids
985 # task switching under load which helps performance).
986 out = q.get_nowait() or await q.get()
987 assert isinstance(out, PoolingRequestOutput)
988 # Note: both OutputProcessor and EngineCore handle their
989 # own request cleanup based on finished.
990 finished = out.finished
991 yield out
992
993 # If the request is disconnected by the client, generate()
994 # is cancelled. So, we abort the request if we end up here.
995 except asyncio.CancelledError:
996 if q is not None:
997 await self.abort(q.request_id, internal=True)
998 if self.log_requests:
999 logger.info("Request %s aborted.", request_id)
1000 raise
1001
1002 # Engine is dead. Do not abort since we shut down.
1003 except EngineDeadError:
1004 if self.log_requests:
1005 logger.info("Request %s failed (engine dead).", request_id)
1006 raise
1007
1008 # Request validation error or admission control rejection.
1009 except (VLLMClientError, GracefulHTTPError):
1010 if self.log_requests:
1011 logger.info("Request %s failed (bad request).", request_id)
1012 raise
1013
1014 # Unexpected error in the generate() task (possibly recoverable).
1015 except Exception as e:
1016 if q is not None:
1017 await self.abort(q.request_id, internal=True)
1018 if self.log_requests:
1019 logger.info("Request %s failed.", request_id)
1020 raise EngineGenerateError() from e
1021 finally:
1022 if q is not None:
1023 q.close()
