253 @instrument(span_name="Prepare model")
254 def _initialize_kv_caches(self, vllm_config: VllmConfig) -> KVCacheConfig:
255 start = time.time()
256
257 # register all kvcache specs in enginecore process.
258 register_all_kvcache_specs(vllm_config)
259
260 # Get all kv cache needed by the model
261 kv_cache_specs = self.model_executor.get_kv_cache_specs()
262
263 # Some layers (e.g. Prefix LM attention) run non-causally and tag their
264 # KV cache spec with ``non_causal=True``. The specs are collected here in
265 # the engine-core process (the same process that builds the scheduler),
266 # so this is the multiproc-safe place to translate that layer-level
267 # signal into a scheduling policy: chunked prefill and prefix caching
268 # both assume causal attention and would corrupt non-causal prefill.
269 if any(
270 getattr(spec, "non_causal", False)
271 for worker_specs in kv_cache_specs
272 for spec in worker_specs.values()
273 ):
274 if vllm_config.scheduler_config.enable_chunked_prefill:
275 logger.info(
276 "Disabling chunked prefill: model has non-causal attention layers."
277 )
278 vllm_config.scheduler_config.enable_chunked_prefill = False
279 if vllm_config.cache_config.enable_prefix_caching:
280 logger.info(
281 "Disabling prefix caching: model has non-causal attention layers."
282 )
283 vllm_config.cache_config.enable_prefix_caching = False
284
285 # Resolve the KV cache layout before memory profiling: workers that
286 # capture full cudagraphs initialize a minimal KV cache during it.
287 # Attention-free models resolve the default so layout reads never precede
288 # resolution.
289 layout = resolve_kv_cache_layout(
290 vllm_config,
291 self.model_executor.get_supported_kv_cache_layouts(),
292 [s for specs in kv_cache_specs for s in specs.values()],
293 )
294 self.model_executor.set_kv_cache_layout(layout.name)
295
296 has_kv_cache = any(kv_cache_spec for kv_cache_spec in kv_cache_specs)
297 if has_kv_cache:
298 if envs.VLLM_ELASTIC_EP_SCALE_UP_LAUNCH:
299 # NOTE(yongji): should already be set
300 # during _eep_scale_up_before_kv_init
301 assert self.available_gpu_memory_for_kv_cache > 0
302 available_gpu_memory = [self.available_gpu_memory_for_kv_cache] * len(
303 kv_cache_specs
304 )
305 else:
306 # Profiles the peak memory usage of the model to determine how
307 # much memory can be allocated for kv cache.
308 available_gpu_memory = self.model_executor.determine_available_memory()
309 self.available_gpu_memory_for_kv_cache = available_gpu_memory[0]
310 else:
311 # Attention free models don't need memory for kv cache
312 available_gpu_memory = [0] * len(kv_cache_specs)
313
314 assert len(kv_cache_specs) == len(available_gpu_memory)
315
316 # Track max_model_len before KV cache config to detect auto-fit changes
317 max_model_len_before = vllm_config.model_config.max_model_len
318
319 kv_cache_configs = get_kv_cache_configs(
320 vllm_config, kv_cache_specs, available_gpu_memory
321 )
322 for kv_cache_config in kv_cache_configs:
323 kv_cache_config.kv_cache_layout = vllm_config.cache_config.kv_cache_layout
324
325 # If auto-fit reduced max_model_len, sync the new value to workers.
326 # This is needed because workers were spawned before memory profiling
327 # and have the original (larger) max_model_len cached.
328 max_model_len_after = vllm_config.model_config.max_model_len
329 if max_model_len_after != max_model_len_before:
330 self.collective_rpc("update_max_model_len", args=(max_model_len_after,))
331
332 scheduler_kv_cache_config = generate_scheduler_kv_cache_config(kv_cache_configs)
333 vllm_config.cache_config.num_gpu_blocks = scheduler_kv_cache_config.num_blocks
334 kv_cache_groups = scheduler_kv_cache_config.kv_cache_groups
335 if kv_cache_groups:
336 # Exclude groups that opt out of prefix caching (e.g. GLM-5.3-Flash
337 # kpool tail, a 1-block/req scratch buffer with block_size=kpool):
338 # their small block_size would otherwise drag the global block_size
339 # below the real allocator block size and desync it from mamba.
340 participating = [
341 g.kv_cache_spec.block_size
342 for g in kv_cache_groups
343 if g.kv_cache_spec.prefix_cacheable
344 ]
345 vllm_config.cache_config.block_size = min(
346 participating
347 if participating
348 else [g.kv_cache_spec.block_size for g in kv_cache_groups]
349 )
350 update_kv_cache_capacity(vllm_config, scheduler_kv_cache_config)
351
352 vllm_config.validate_block_size()
353
354 self.model_executor.initialize_from_config(kv_cache_configs)
355 if not envs.VLLM_ELASTIC_EP_SCALE_UP_LAUNCH:
356 self.model_executor.compile_or_warm_up_model()
357
358 elapsed = time.time() - start
359 compile_time = vllm_config.compilation_config.compilation_time
360 encoder_compile_time = vllm_config.compilation_config.encoder_compilation_time
361 if encoder_compile_time > 0:
362 logger.info_once(
363 "init engine (profile, create kv cache, warmup model) took "
364 "%.2f s (compilation: %.2f s — language_model: %.2f s, "
365 "encoder: %.2f s)",
366 elapsed,
367 compile_time + encoder_compile_time,
368 compile_time,
369 encoder_compile_time,
370 )
371 elif compile_time > 0:
372 logger.info_once(
373 "init engine (profile, create kv cache, warmup model) took "
374 "%.2f s (compilation: %.2f s)",
375 elapsed,
376 compile_time,
377 )
378 else:
379 logger.info_once(
380 "init engine (profile, create kv cache, warmup model) took %.2f s",
381 elapsed,
382 )
383 return scheduler_kv_cache_config

444 def add_request(self, request: Request, request_wave: int = 0):
445 """Add request to the scheduler.
446
447 `request_wave`: indicate which wave of requests this is expected to
448 belong to in DP case
449 """
450 # Validate the request_id type.
451 if not isinstance(request.request_id, str):
452 raise TypeError(
453 f"request_id must be a string, got {type(request.request_id)}"
454 )
455
456 if pooling_params := request.pooling_params:
457 supported_pooling_tasks = [
458 task for task in self.get_supported_tasks() if task in POOLING_TASKS
459 ]
460
461 if pooling_params.task not in supported_pooling_tasks:
462 raise ValueError(
463 f"Unsupported task: {pooling_params.task!r} "
464 f"Supported tasks: {supported_pooling_tasks}"
465 )
466
467 if request.kv_transfer_params is not None and (
468 not self.scheduler.get_kv_connector()
469 ):
470 logger.warning(
471 "Got kv_transfer_params, but no KVConnector found. "
472 "Disabling KVTransfer for this request."
473 )
474
475 if (
476 request.ec_transfer_params is not None
477 and self.scheduler.get_ec_connector() is None
478 ):
479 logger.warning(
480 "Got ec_transfer_params, but no ECConnector found. "
481 "Disabling ECTransfer for this request."
482 )
483
484 self.scheduler.add_request(request)
485 if request.abort_immediately:
486 # Immediately abort so the connector's request_finished hook runs
487 # to free any pre-admission KV-transfer resources.
488 self.abort_requests([request.request_id])

514 @contextmanager
515 def capture_iteration_details(
516 self, scheduler_output: SchedulerOutput | None
517 ) -> Generator[SchedulerIterationDetails | None, None, None]:
518 enable_details = (
519 self.vllm_config.observability_config.enable_logging_iteration_details

526 scheduler_output is not None

529 yield None
530 return

536 iteration_index=iteration_index,
537 num_ctx_requests=0,

540 num_generation_tokens=0,

589 def step(self) -> tuple[dict[int, EngineCoreOutputs], bool]:
590 """Schedule, execute, and make output.
591
592 Returns tuple of outputs and a flag indicating whether the model
593 was executed.
594 """
595
596 # Check for any requests remaining in the scheduler - unfinished,
597 # or finished and not yet removed from the batch.
598 if not self.scheduler.has_requests():
599 return {}, False
600 scheduler_output = self.scheduler.schedule(self._should_throttle_prefills())
601 future = self.model_executor.execute_model(scheduler_output, non_block=True)
602 grammar_output = self.scheduler.get_grammar_bitmask(scheduler_output)
603 with (
604 self.capture_iteration_details(scheduler_output) as iteration_details,
605 self.log_error_detail(scheduler_output),
606 ):
607 model_output = future.result()
608 if model_output is None:
609 model_output = self.model_executor.sample_tokens(grammar_output)
610
611 # Before processing the model output, process any aborts that happened
612 # during the model execution.
613 self._process_aborts_queue()
614 engine_core_outputs = self.scheduler.update_from_output(
615 scheduler_output, model_output
616 )
617 self._attach_iteration_details(engine_core_outputs, iteration_details)
618
619 return engine_core_outputs, scheduler_output.total_num_scheduled_tokens > 0

630 def step_with_batch_queue(
631 self,
632 ) -> tuple[dict[int, EngineCoreOutputs] | None, bool]:
633 """Schedule and execute batches with the batch queue.
634 Note that if nothing to output in this step, None is returned.
635
636 The execution flow is as follows:
637 1. Try to schedule a new batch if the batch queue is not full.
638 If a new batch is scheduled, directly return an empty engine core
639 output. In other words, fulfilling the batch queue has a higher priority
640 than getting model outputs.
641 2. If there is no new scheduled batch, meaning that the batch queue
642 is full or no other requests can be scheduled, we block until the first
643 batch in the job queue is finished.
644 3. Update the scheduler from the output.
645 """
646
647 batch_queue = self.batch_queue
648 assert batch_queue is not None
649
650 # Try to schedule a new batch if the batch queue is not full, but
651 # the scheduler may return an empty batch if all requests are scheduled.
652 # Note that this is not blocking.
653 assert len(batch_queue) < self.batch_queue_size
654
655 model_executed = False
656 deferred_scheduler_output = None
657 if self.scheduler.has_requests():
658 scheduler_output = self.scheduler.schedule(self._should_throttle_prefills())
659 with self.log_error_detail(scheduler_output):
660 exec_future = self.model_executor.execute_model(
661 scheduler_output, non_block=True
662 )
663 if self.is_ec_consumer:
664 model_executed = scheduler_output.total_num_scheduled_tokens > 0
665
666 if self.is_pooling_model or not model_executed:
667 # No sampling required (no requests scheduled).
668 future = cast(Future[ModelRunnerOutput], exec_future)
669 else:
670 if not scheduler_output.pending_structured_output_tokens:
671 # We aren't waiting for any tokens, get any grammar output
672 # and sample immediately.
673 grammar_output = self.scheduler.get_grammar_bitmask(
674 scheduler_output
675 )
676 future = self.model_executor.sample_tokens(
677 grammar_output, non_block=True
678 )
679 else:
680 # We need to defer sampling until we have processed the model output
681 # from the prior step.
682 deferred_scheduler_output = scheduler_output
683
684 if not deferred_scheduler_output:
685 # Add this step's future to the queue.
686 batch_queue.appendleft((future, scheduler_output, exec_future))
687 if len(batch_queue) < self.batch_queue_size and (
688 model_executed or self.scheduler.has_requests()
689 ):
690 # Don't block on next worker response unless the queue is full
691 # or there are no more requests to schedule.
692 return None, model_executed
693
694 elif not batch_queue:
695 # Queue is empty. We should not reach here since this method should
696 # only be called when the scheduler contains requests or the queue
697 # is non-empty.
698 return None, False
699
700 # Block until the next result is available.
701 future, scheduler_output, exec_model_fut = batch_queue.pop()
702 with (
703 self.capture_iteration_details(scheduler_output) as iteration_details,
704 self.log_error_detail(scheduler_output),
705 ):
706 model_output = future.result()
707 if model_output is None:
708 # None from sample_tokens() implies that the original execute_model()
709 # call failed - raise that exception.
710 exec_model_fut.result()
711 raise RuntimeError("unexpected error")
712
713 # Before processing the model output, process any aborts that happened
714 # during the model execution.
715 self._process_aborts_queue()
716 engine_core_outputs = self.scheduler.update_from_output(
717 scheduler_output, model_output
718 )
719 self._attach_iteration_details(engine_core_outputs, iteration_details)
720
721 # NOTE(nick): We can either handle the deferred tasks here or save
722 # in a field and do it immediately once step_with_batch_queue is
723 # re-called. The latter slightly favors TTFT over TPOT/throughput.
724 if deferred_scheduler_output:
725 # When draft tokens are used with structured output, validate them
726 # before computing the grammar bitmask for the deferred request.
727 if self.check_for_draft_tokens:
728 draft_token_ids = self.model_executor.take_draft_token_ids()
729 if draft_token_ids is not None:
730 # Update the draft token ids in the scheduler output to
731 # filter out the invalid spec tokens, which will be padded
732 # with -1 and skipped by the grammar bitmask computation.
733 self.scheduler.update_draft_token_ids_in_output(
734 draft_token_ids, deferred_scheduler_output
735 )
736 # We now have the tokens needed to compute the bitmask for the
737 # deferred request. Get the bitmask and call sample tokens.
738 grammar_output = self.scheduler.get_grammar_bitmask(
739 deferred_scheduler_output
740 )
741 future = self.model_executor.sample_tokens(grammar_output, non_block=True)
742 batch_queue.appendleft((future, deferred_scheduler_output, exec_future))
743
744 return engine_core_outputs, model_executed

841 def pause_scheduler(
842 self, mode: PauseMode = "abort", clear_cache: bool = True
843 ) -> Future | None:
844 """Pause generation; behavior depends on mode.
845
846 All pause modes queue new adds -- "abort" and "keep" skip step();
847 "wait" allows step() so in-flight requests can drain.
848
849 - ``abort``: Set PAUSED_NEW, abort all requests, wait for abort
850 outputs to be sent (when running with output_queue), optionally
851 clear caches, then complete the returned Future.
852 - ``wait``: Set PAUSED_NEW (queue adds, keep stepping); when drained,
853 optionally clear caches, then complete the returned Future.
854 - ``keep``: Set PAUSED_ALL; return a Future that completes when the
855 output queue is empty.
856 """
857 if mode not in get_args(PauseMode):
858 raise ValueError(f"Invalid pause mode: {mode}")
859 if mode == "wait":
860 raise ValueError("'wait' mode can't be used in inproc-engine mode")
861
862 if mode == "abort":
863 self.scheduler.finish_requests(None, RequestStatus.FINISHED_ABORTED)
864
865 pause_state = PauseState.PAUSED_ALL if mode == "keep" else PauseState.PAUSED_NEW
866 self.scheduler.set_pause_state(pause_state)
867 self._finish_pause(clear_cache)
868
869 return None

879 def sleep(self, level: int = 1, mode: PauseMode = "abort") -> None | Future:
880 """Put the engine to sleep at the specified level.
881
882 Args:
883 level: Sleep level.
884 - Level 0: Pause scheduling only. Requests are still accepted
885 but not processed. No GPU memory changes.
886 - Level 1: Offload model weights to CPU, discard KV cache.
887 - Level 2: Discard all GPU memory.
888 mode: Pause mode - how to deal with any existing requests, see
889 documentation of pause_scheduler method.
890 """
891
892 # Pause scheduler before sleeping.
893 clear_prefix_cache = level >= 1
894 pause_future = self.pause_scheduler(mode=mode, clear_cache=clear_prefix_cache)
895 if level < 1:
896 return pause_future
897
898 # Level 1+: Delegate to executor for GPU memory management
899 model_executor = self.model_executor
900 if pause_future is None:
901 model_executor.sleep(level)
902 return None
903
904 future = Future[Any]()
905
906 def pause_complete(f: Future):
907 try:
908 f.result() # propagate any exception
909 future.set_result(model_executor.sleep(level))
910 except Exception as e:
911 future.set_exception(e)
912
913 logger.info("Waiting for in-flight requests to complete before sleeping...")
914 pause_future.add_done_callback(pause_complete)
915 return future

1402 @fault_tolerant_wrapper
1403 def run_busy_loop(self):
1404 """Core busy loop of the EngineCore."""
1405 while self._handle_shutdown():
1406 # 1) Poll the input queue until there is work to do.
1407 self._process_input_queue()
1408 # Publish request counts before and after GPU step to ensure freshness.
1409 self._maybe_publish_request_counts()
1410 # 2) Step the engine core and return the outputs.
1411 self._process_engine_step()
1412 self._maybe_publish_request_counts()
1413
1414 raise SystemExit

1416 def _maybe_publish_request_counts(self):
1417 if not self.publish_dp_lb_stats:
1418 return

1421 counts = self.scheduler.get_request_counts()
1422 if counts != self.last_counts:
1423 self.last_counts = counts

1425 *counts, kv_cache_usage=self.scheduler.get_kv_cache_usage()
1426 )

1429 def _process_input_queue(self):
1430 """Exits when an engine step needs to be performed."""
1431
1432 waited = False
1433 while not self.has_work() and self.is_running():
1434 # Notify callbacks waiting for engine to become idle.
1435 self._notify_idle_state_callbacks()
1436 if self.input_queue.empty():
1437 # Drain aborts queue; all aborts are also processed via input_queue.
1438 with self.aborts_queue.mutex:
1439 self.aborts_queue.queue.clear()
1440 if logger.isEnabledFor(DEBUG):
1441 logger.debug("EngineCore waiting for work.")
1442 waited = True
1443 block = self.process_input_queue_block
1444 try:
1445 req = self.input_queue.get(block=block)
1446 self._handle_client_request(*req)
1447 except queue.Empty:
1448 break
1449 if not block:
1450 break
1451
1452 if waited:
1453 logger.debug("EngineCore loop active.")
1454
1455 # Handle any more client requests.
1456 while not self.input_queue.empty():
1457 req = self.input_queue.get_nowait()
1458 self._handle_client_request(*req)

1460 def _process_engine_step(self) -> bool:
1461 """Called only when there are unfinished local requests."""
1462
1463 # Step the engine core.
1464 outputs, model_executed = self.step_fn()
1465 # Put EngineCoreOutputs into the output queue.
1466 for output in outputs.items() if outputs else ():
1467 self.output_queue.put_nowait(output)
1468 # Post-step hook.
1469 self.post_step(model_executed)
1470
1471 # If no model execution happened but there is still scheduler work
1472 # (e.g. WAITING_FOR_REMOTE_KVS or delayed KV connector frees), yield
1473 # the GIL briefly to allow background transfer threads to make progress.
1474 if not model_executed and self.scheduler.has_requests():
1475 time.sleep(0.001)
1476
1477 return model_executed

1484 def _handle_shutdown(self) -> bool:
1485 # Check if shutdown was requested and handle it
1486 if self.shutdown_state == EngineShutdownState.RUNNING:
1487 return True
1488
1489 if self.shutdown_state == EngineShutdownState.REQUESTED:
1490 shutdown_timeout = self.vllm_config.shutdown_timeout
1491 mode = "abort" if shutdown_timeout == 0 else "drain"
1492
1493 logger.info(
1494 "[shutdown] EngineCore: start mode=%s timeout=%ds",
1495 mode,
1496 shutdown_timeout,
1497 )
1498
1499 if shutdown_timeout == 0:
1500 num_requests = self.scheduler.get_num_unfinished_requests()
1501 if num_requests > 0:
1502 logger.info(
1503 "[shutdown] EngineCore: aborting in-flight requests count=%d",
1504 num_requests,
1505 )
1506 aborted_reqs = self.scheduler.finish_requests(
1507 None, RequestStatus.FINISHED_ABORTED
1508 )
1509 self._send_abort_outputs(aborted_reqs)
1510 else:
1511 num_requests = self.scheduler.get_num_unfinished_requests()
1512 if num_requests > 0:
1513 logger.info(
1514 "[shutdown] EngineCore: draining in-flight requests "
1515 "count=%d timeout=%ds",
1516 num_requests,
1517 shutdown_timeout,
1518 )
1519
1520 self.shutdown_state = EngineShutdownState.SHUTTING_DOWN
1521
1522 # Exit when no work remaining
1523 if not self.has_work():
1524 logger.info(
1525 "[shutdown] EngineCore: request processing complete; "
1526 "starting resource teardown"
1527 )
1528 return False
1529
1530 return True

1532 def _handle_client_request(
1533 self, request_type: EngineCoreRequestType, request: Any
1534 ) -> None:
1535 """Dispatch request from client."""
1536
1537 if request_type == EngineCoreRequestType.WAKEUP:
1538 return
1539 elif request_type == EngineCoreRequestType.ADD:
1540 req, request_wave = request
1541 if self._reject_add_in_shutdown(req):
1542 return
1543 self.add_request(req, request_wave)
1544 elif request_type == EngineCoreRequestType.ABORT:
1545 self.abort_requests(request)
1546 elif request_type == EngineCoreRequestType.UTILITY:
1547 client_idx, call_id, method_name, args = request
1548 if self._reject_utility_in_shutdown(client_idx, call_id, method_name):
1549 return
1550 output = UtilityOutput(call_id)
1551 # Lazily look-up utility method so that failure will be handled/returned.
1552 get_result = lambda: (
1553 (method := getattr(self, method_name))
1554 and method(*self._convert_msgspec_args(method, args))
1555 )
1556 enqueue_output = lambda out: self.output_queue.put_nowait(
1557 (client_idx, EngineCoreOutputs(utility_output=out))
1558 )
1559 self._invoke_utility_method(method_name, get_result, output, enqueue_output)
1560 elif request_type == EngineCoreRequestType.EXECUTOR_FAILED:
1561 raise RuntimeError("Executor failed.")
1562 else:
1563 logger.error(
1564 "Unrecognized input request type encountered: %s", request_type
1565 )

1686 def process_input_sockets(
1687 self,
1688 input_addresses: list[str],
1689 coord_input_address: str | None,
1690 identity: bytes,
1691 ready_event: threading.Event,
1692 ):
1693 """Input socket IO thread."""
1694
1695 # Msgpack serialization decoding with optional tensor IPC receiver.
1696 add_request_decoder = MsgpackDecoder(
1697 EngineCoreRequest, oob_tensor_provider=self.tensor_ipc_receiver
1698 )
1699 generic_decoder = MsgpackDecoder(oob_tensor_provider=self.tensor_ipc_receiver)
1700
1701 with ExitStack() as stack, zmq.Context() as ctx:
1702 input_sockets = [
1703 stack.enter_context(
1704 make_zmq_socket(
1705 ctx, input_address, zmq.DEALER, identity=identity, bind=False
1706 )
1707 )
1708 for input_address in input_addresses
1709 ]
1710 if coord_input_address is None:
1711 coord_socket = None
1712 else:
1713 coord_socket = stack.enter_context(
1714 make_zmq_socket(
1715 ctx,
1716 coord_input_address,
1717 zmq.XSUB,
1718 identity=identity,
1719 bind=False,
1720 )
1721 )
1722 # Send subscription message to coordinator.
1723 coord_socket.send(b"\x01")
1724
1725 # Register sockets with poller.
1726 poller = zmq.Poller()
1727 ready_response = self._make_ready_response()
1728 ready_payload = msgspec.msgpack.encode(ready_response)
1729 for input_socket in input_sockets:
1730 # Send initial message to each input socket - this is required
1731 # before the front-end ROUTER socket can send input messages
1732 # back to us.
1733 input_socket.send(ready_payload)
1734 poller.register(input_socket, zmq.POLLIN)
1735
1736 if coord_socket is not None:
1737 # Wait for ready message from coordinator.
1738 assert coord_socket.recv() == b"READY"
1739 poller.register(coord_socket, zmq.POLLIN)
1740
1741 ready_event.set()
1742 del ready_event
1743 while True:
1744 for input_socket, _ in poller.poll():
1745 # (RequestType, RequestData)
1746 type_frame, *data_frames = input_socket.recv_multipart(copy=False)
1747 # NOTE(yongji): ignore READY message sent by DP coordinator
1748 # that is used to notify newly started engines
1749 if type_frame.buffer == b"READY":
1750 assert input_socket == coord_socket
1751 continue
1752 request_type = EngineCoreRequestType(bytes(type_frame.buffer))
1753
1754 # Deserialize the request data.
1755 request: Any
1756 if request_type == EngineCoreRequestType.ADD:
1757 req: EngineCoreRequest = add_request_decoder.decode(data_frames)
1758 try:
1759 request = self.preprocess_add_request(req)
1760 except MultiModalCacheMissError as e:
1761 # P0/P1 shadow drift -- return a retryable signal (P0
1762 # drops the stale entry, client resends with data).
1763 self._handle_mm_cache_miss(req, e)
1764 continue
1765 except Exception:
1766 self._handle_request_preproc_error(req)
1767 continue
1768 elif request_type == EngineCoreRequestType.UTILITY:
1769 request = generic_decoder.decode(data_frames)
1770 client_idx, call_id, method, args = request
1771 if method == FT_UTILITY_METHOD:
1772 self.ft_sentinel.handle_command(
1773 client_idx, call_id, args[0]
1774 )
1775 continue
1776 else:
1777 request = generic_decoder.decode(data_frames)
1778
1779 if request_type == EngineCoreRequestType.ABORT:
1780 # Aborts are added to *both* queues, allows us to eagerly
1781 # process aborts while also ensuring ordering in the input
1782 # queue to avoid leaking requests. This is ok because
1783 # aborting in the scheduler is idempotent.
1784 self.aborts_queue.put_nowait(request)
1785
1786 # Push to input queue for core busy loop.
1787 self.input_queue.put_nowait((request_type, request))

1789 def process_output_sockets(
1790 self, output_paths: list[str], coord_output_path: str | None, engine_index: int
1791 ):
1792 """Output socket IO thread."""
1793
1794 # Msgpack serialization encoding.
1795 encoder = MsgpackEncoder()
1796 # Send buffers to reuse.
1797 reuse_buffers: list[bytearray] = []
1798 # Payload buffers that can't be reused yet because zmq may still be
1799 # sending them.
1800 # Buffers of the zero-copy tensor/ndarray frames don't need tracking
1801 # here: zmq itself holds a reference to each until it's done with it.
1802 pending = deque[tuple[zmq.MessageTracker, bytearray]]()
1803
1804 # We must set linger to ensure the ENGINE_CORE_DEAD
1805 # message is sent prior to closing the socket.
1806 with ExitStack() as stack, zmq.Context() as ctx:
1807 sockets = [
1808 stack.enter_context(
1809 make_zmq_socket(ctx, output_path, zmq.PUSH, linger=4000)
1810 )
1811 for output_path in output_paths
1812 ]
1813 coord_socket = (
1814 stack.enter_context(
1815 make_zmq_socket(
1816 ctx, coord_output_path, zmq.PUSH, bind=False, linger=4000
1817 )
1818 )
1819 if coord_output_path is not None
1820 else None
1821 )
1822 max_reuse_bufs = len(sockets) + 1
1823
1824 while True:
1825 output = self.output_queue.get()
1826 if output == EngineCoreProc.ENGINE_CORE_DEAD:
1827 for socket in sockets:
1828 socket.send(output)
1829 break
1830 assert not isinstance(output, bytes)
1831 client_index, outputs = output
1832 outputs.engine_index = engine_index
1833
1834 if client_index == -1:
1835 # Don't reuse buffer for coordinator message
1836 # which will be very small.
1837 assert coord_socket is not None
1838 coord_socket.send_multipart(encoder.encode(outputs))
1839 continue
1840
1841 # Reclaim buffers that zmq is finished with.
1842 while pending and pending[-1][0].done:
1843 reclaimed = pending.pop()[1]
1844 if len(reuse_buffers) < max_reuse_bufs:
1845 reuse_buffers.append(reclaimed)
1846
1847 buffer = reuse_buffers.pop() if reuse_buffers else bytearray()
1848 buffers = encoder.encode_into(outputs, buffer)
1849 tracker = self._send_msg_tracking_payload(
1850 sockets[client_index], buffers
1851 )
1852 if not tracker.done:
1853 pending.appendleft((tracker, buffer))
1854 elif len(reuse_buffers) < max_reuse_bufs:
1855 # Limit the number of buffers to reuse.
1856 reuse_buffers.append(buffer)

1876 request.client_index,
1877 EngineCoreOutputs(
1878 engine_index=self.engine_index,

1880 outputs=[
1881 EngineCoreOutput(

1883 new_token_ids=[],

1889 )

1892 @staticmethod
1893 def _send_msg_tracking_payload(
1894 socket: zmq.Socket, buffers: Sequence[bytestr]
1895 ) -> zmq.MessageTracker:
1896 """Send `buffers` as a zero-copy multipart message, returning a tracker
1897 for the *first* frame.
1898
1899 Used instead of `Socket.send_multipart()` because we reuse the buffer
1900 passed to `MsgpackEncoder.encode_into()`: `send_multipart()` returns a
1901 tracker for the last frame only.
1902 """

1904 tracker = socket.send(buffers[0], more_flag, copy=False, track=True)
1905 if more_flag:
1906 socket.send_multipart(buffers[1:], copy=False)
1907 return tracker
