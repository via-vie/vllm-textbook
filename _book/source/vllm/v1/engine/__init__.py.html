47class FinishReason(enum.IntEnum):
48 """
49 Reason a request finished - stop, length, abort, error, or repetition.
50
51 Int rather than Str for more compact serialization.
52
53 stop - a stop string was emitted
54 length - max_tokens was consumed, or max_model_len was reached
55 abort - aborted by client
56 error - retryable request-level internal error (e.g., KV load failure).
57 Invariant: always converted to 500 Internal Server Error.
58 repetition - repetitive token pattern detected (hallucination)
59
60 """
61
62 STOP = 0
63 LENGTH = 1
64 ABORT = 2
65 ERROR = 3
66 REPETITION = 4
67
68 def __str__(self):
69 return FINISH_REASON_STRINGS[self.value]

177class EngineCoreEvent(msgspec.Struct):
178 """A timestamped engine core event associated with a request.
179
180 The timestamp is a monotonic timestamp and is used by the engine
181 frontend to calculate intervals between engine core events. These
182 timestamps should not be compared with timestamps from other processes.
183 """
184
185 type: EngineCoreEventType
186 timestamp: float
187
188 @classmethod
189 def new_event(
190 cls, event_type: EngineCoreEventType, timestamp: float | None = None
191 ) -> "EngineCoreEvent":
192 timestamp = time.monotonic() if timestamp is None else timestamp
193 return cls(event_type, timestamp)
