84 def _validate_params(
85 self,
86 params: SamplingParams | PoolingParams,
87 supported_tasks: tuple[SupportedTask, ...],
88 ) -> None:
89 """Raise `ValueError` if SamplingParams or PoolingParams is not valid."""
90 if isinstance(params, SamplingParams):
91 supported_generation_tasks = [
92 task for task in supported_tasks if task in GENERATION_TASKS
93 ]
94 if not supported_generation_tasks:
95 raise VLLMValidationError("This model does not support generation")
96
97 params.verify(
98 self.model_config,
99 self.speculative_config,
100 self.structured_outputs_config,
101 self.tokenizer,
102 )
103
104 if self.model_config.return_sampling_mask:
105 if params.temperature <= 0:
106 raise ValueError(
107 "sampling distribution replay requires temperature > 0"
108 )
109 if params.top_k <= 0:
110 raise ValueError(
111 "sampling distribution replay requires top_k > 0 to "
112 "bound sampling mask size, reduce transfer overhead, "
113 "and avoid potential OOMs"
114 )
115 if params.thinking_token_budget is not None and (
116 self.vllm_config.reasoning_config is None
117 or not self.vllm_config.reasoning_config.enabled
118 ):
119 raise VLLMValidationError(
120 "thinking_token_budget is set but reasoning_config is "
121 "not configured. Please set --reasoning-parser "
122 "and/or --reasoning-config to use thinking_token_budget."
123 )
124 if (
125 params.trace_decode_token_ids
126 and not self.model_config.enable_trace_replay
127 ):
128 raise VLLMValidationError(
129 "trace_decode_token_ids is set but trace replay is not "
130 "enabled. Start the engine with --enable-trace-replay "
131 "to use it."
132 )
133 elif isinstance(params, PoolingParams):
134 supported_pooling_tasks = [
135 task for task in supported_tasks if task in POOLING_TASKS
136 ]
137 if not supported_pooling_tasks:
138 raise VLLMValidationError("This model does not support pooling")
139
140 if params.task is None:
141 if "token_embed" in supported_pooling_tasks:
142 params.task = "token_embed"
143 elif "token_classify" in supported_pooling_tasks:
144 params.task = "token_classify"
145 elif "plugin" in supported_pooling_tasks:
146 params.task = "plugin"
147
148 if params.task not in supported_pooling_tasks:
149 raise VLLMValidationError(
150 f"Unsupported task: {params.task!r} "
151 f"Supported tasks: {supported_pooling_tasks}"
152 )
153
154 params.verify(self.model_config)
155 else:
156 raise TypeError(
157 f"params must be either SamplingParams or PoolingParams, "
158 f"but got {type(params).__name__}"
159 )

161 def _normalize_trace_replay_params(
162 self, sampling_params: SamplingParams, prompt_len: int
163 ) -> None:
164 """Apply trace replay's generation semantics to request-local params."""
165 trace_token_ids = sampling_params.trace_decode_token_ids
166 assert trace_token_ids
167 assert sampling_params.max_tokens is not None
168
169 max_trace_len = max(self.model_config.max_model_len - prompt_len, 1)
170 trace_token_ids = trace_token_ids[:max_trace_len]
171 sampling_params.trace_decode_token_ids = trace_token_ids
172
173 # Apply this after the generation config so its EOS token cannot stop
174 # replay before the trace is exhausted.
175 sampling_params.max_tokens = min(
176 len(trace_token_ids), sampling_params.max_tokens
177 )
178 sampling_params.min_tokens = 0
179 sampling_params.ignore_eos = True
180 sampling_params._eos_token_id = None
181 sampling_params.stop = []
182 sampling_params.stop_token_ids = []
183 sampling_params._all_stop_token_ids = set()

186 if lora_request is None:
187 return

204 def _get_mm_identifier(
205 self,
206 mm_hash: str,
207 lora_request: LoRARequest | None,
208 ) -> str:
209 """
210 When enable_tower_connector_lora is True, multi-modal embeddings
211 vary depending on the LoRA request. Therefore, the mm_hash must be
212 generated based on the LoRA request to prevent incorrect cache hits.
213 """
214 if (
215 lora_request is None
216 or self.lora_config is None
217 or not self.lora_config.enable_tower_connector_lora
218 ):
219 return mm_hash
220 return f"{lora_request.lora_name}:{mm_hash}"

222 def inject_into_mm_cache(
223 self,
224 mm_hashes: dict[str, list[str]],
225 mm_kwargs: dict[str, list],
226 ) -> None:
227 """Inject pre-processed mm_kwargs into the processor cache.
228
229 Call this when mm_kwargs have already been through the HF processor
230 externally (e.g. by a frontend that transfers pre-processed tensors
231 to the backend). This ensures MM cache hit rate metrics are reported
232 accurately and avoids redundant processing on subsequent requests
233 with the same images.
234
235 Uses ``get_and_update_item()`` with an empty prompt_updates list,
236 since token expansion has already been handled externally.
237 """
238 cache = self.renderer.mm_processor_cache
239 if cache is None:
240 return
241 try:
242 for modality, hashes in mm_hashes.items():
243 items = mm_kwargs.get(modality, [])
244 for i, mm_hash in enumerate(hashes):
245 if i < len(items) and items[i] is not None:
246 # Insert into cache via get_and_update_item.
247 # Use the returned item (may be an address for SHM
248 # cache or the original item for LRU cache).
249 items[i], _ = cache.get_and_update_item(
250 (items[i], []),
251 mm_hash,
252 )
253 # Update cache stats to reflect the externally processed items
254 self.renderer.update_mm_cache_stats()
255 except Exception:
256 logger.warning(
257 "Failed to inject mm_kwargs into processor cache",
258 exc_info=True,
259 )

261 @staticmethod
262 def assign_request_id(request: EngineCoreRequest):
263 """Replace the externally supplied request ID with an internal request ID
264 that adds 8 random characters in order to ensure uniqueness.
265 """
266 if request.external_req_id is not None:
267 raise ValueError(
268 "The external_req_id field should not be set on EngineCoreRequests"
269 " passed to vLLM; use the request_id field."
270 )
271 request.external_req_id = request.request_id
272 if envs.VLLM_DISABLE_REQUEST_ID_RANDOMIZATION:
273 logger.warning_once(
274 "VLLM_DISABLE_REQUEST_ID_RANDOMIZATION is set and will be "
275 "removed in a future release. Duplicate externally-provided "
276 "request IDs may cause failures and/or subtle correctness errors."
277 )
278 else:
279 request.request_id = f"{request.external_req_id}-{random_uuid():.8}"

281 def process_inputs(
282 self,
283 request_id: str,
284 prompt: PromptType | EngineInput,
285 params: SamplingParams | PoolingParams,
286 supported_tasks: tuple[SupportedTask, ...],
287 arrival_time: float | None = None,
288 lora_request: LoRARequest | None = None,
289 tokenization_kwargs: dict[str, Any] | None = None,
290 trace_headers: Mapping[str, str] | None = None,
291 priority: int = 0,
292 data_parallel_rank: int | None = None,
293 resumable: bool = False,
294 session_id: str | None = None,
295 ) -> EngineCoreRequest:
296 self._validate_params(params, supported_tasks)
297 self._validate_lora(lora_request)
298
299 parallel_config = self.vllm_config.parallel_config
300 dp_size = parallel_config.data_parallel_size
301 dp_local_size = parallel_config.data_parallel_size_local
302 num_ranks = dp_local_size if parallel_config.local_engines_only else dp_size
303 if data_parallel_rank is not None and not (0 <= data_parallel_rank < num_ranks):
304 raise VLLMValidationError(
305 f"data_parallel_rank {data_parallel_rank} "
306 f"is out of range [0, {num_ranks})."
307 )
308
309 if isinstance(prompt, dict) and "type" in prompt:
310 if arrival_time is None:
311 arrival_time = prompt.get("arrival_time", time.time()) # type: ignore[assignment]
312
313 engine_input: EngineInput = prompt # type: ignore[assignment]
314 else:
315 logger.warning_once(
316 "Passing raw prompts to InputProcessor is deprecated "
317 "and will be removed in the future. You should instead pass "
318 "the outputs of Renderer.render_cmpl() or Renderer.render_chat()."
319 )
320
321 if arrival_time is None:
322 arrival_time = time.time()
323
324 renderer = self.renderer
325 model_config = self.model_config
326
327 parsed_prompt = parse_model_prompt(model_config, prompt)
328 tok_params = renderer.default_cmpl_tok_params.with_kwargs(
329 **(tokenization_kwargs or {})
330 )
331
332 (engine_input,) = renderer.render_cmpl(
333 [parsed_prompt],
334 tok_params,
335 )
336
337 current_platform.validate_request(engine_input, params)
338
339 encoder_input, decoder_input = split_enc_dec_input(engine_input)
340 self._validate_model_inputs(encoder_input, decoder_input)
341
342 # Mypy can be conservative for TypedDict unions; normalize access.
343 if decoder_input["type"] == "embeds":
344 prompt_embeds = decoder_input["prompt_embeds"]
345 prompt_token_ids = decoder_input.get("prompt_token_ids")
346 prompt_is_token_ids = decoder_input.get("is_token_ids")
347 else:
348 prompt_token_ids = decoder_input["prompt_token_ids"]
349 prompt_embeds = None
350 prompt_is_token_ids = None
351
352 sampling_params = None
353 pooling_params = None
354 if isinstance(params, SamplingParams):
355 # TODO: can we avoid cloning here in multiproc case?
356 sampling_params = params.clone()
357 prompt_len = length_from_prompt_token_ids_or_embeds(
358 prompt_token_ids, prompt_embeds
359 )
360 if not 0 <= sampling_params.routed_experts_prompt_start <= prompt_len:
361 raise VLLMValidationError(
362 f"routed_experts_prompt_start must be between 0 and "
363 f"the prompt length ({prompt_len}), inclusive.",
364 parameter="routed_experts_prompt_start",
365 value=sampling_params.routed_experts_prompt_start,
366 )
367 # If unset max tokens, then generate up to the max_model_len.
368 if sampling_params.max_tokens is None:
369 sampling_params.max_tokens = (
370 self.model_config.max_model_len - prompt_len
371 )
372
373 sampling_params.update_from_generation_config(
374 self.generation_config_fields,
375 self.renderer.get_eos_token_id(),
376 )
377 if self.tokenizer is not None:
378 sampling_params.update_from_tokenizer(self.tokenizer)
379 if sampling_params.trace_decode_token_ids:
380 self._normalize_trace_replay_params(sampling_params, prompt_len)
381 else:
382 pooling_params = params.clone()
383
384 # Multimodal related.
385 mm_features: list[MultiModalFeatureSpec] | None = None
386
387 if decoder_input["type"] == "multimodal":
388 decoder_mm_inputs = decoder_input["mm_kwargs"]
389 decoder_mm_positions = decoder_input["mm_placeholders"]
390 decoder_mm_hashes = decoder_input["mm_hashes"]
391
392 if not all(
393 isinstance(leaf, str) for leaf in json_iter_leaves(decoder_mm_hashes)
394 ):
395 raise ValueError(
396 f"mm_hashes must contain only strings, got: {decoder_mm_hashes}. "
397 "This is likely due to an incorrect custom implementation of "
398 "MultiModalProcessor.apply method."
399 )
400
401 # Merge and flatten multimodal placeholders, hashes and inputs
402 # from dictionaries to lists, and sort them by each item's position
403 # in the input sequence.
404 sorted_mm_idxs = argsort_mm_positions(decoder_mm_positions)
405
406 mm_features = []
407 for modality, idx in sorted_mm_idxs:
408 base_mm_hash = decoder_mm_hashes[modality][idx]
409 mm_features.append(
410 MultiModalFeatureSpec(
411 data=decoder_mm_inputs[modality][idx],
412 modality=modality,
413 identifier=self._get_mm_identifier(
414 base_mm_hash,
415 lora_request,
416 ),
417 mm_position=decoder_mm_positions[modality][idx],
418 mm_hash=base_mm_hash,
419 )
420 )
421
422 return EngineCoreRequest(
423 request_id=request_id,
424 prompt_token_ids=prompt_token_ids,
425 prompt_embeds=prompt_embeds,
426 prompt_is_token_ids=prompt_is_token_ids,
427 mm_features=mm_features,
428 sampling_params=sampling_params,
429 pooling_params=pooling_params,
430 arrival_time=arrival_time,
431 lora_request=lora_request,
432 cache_salt=decoder_input.get("cache_salt"),
433 priority=priority,
434 data_parallel_rank=data_parallel_rank,
435 trace_headers=trace_headers,
436 resumable=resumable,
437 session_id=session_id,
438 )

440 def _validate_prompt_len(
441 self,
442 prompt_len: int,
443 prompt_type: Literal["encoder", "decoder"],
444 ):
445 if self.skip_prompt_length_check and prompt_type == "encoder":
446 return
447
448 if prompt_len == 0 and prompt_type == "decoder":
449 raise VLLMValidationError(f"The {prompt_type} prompt cannot be empty")
450
451 model_config = self.model_config
452 max_prompt_len = (
453 model_config.max_model_len
454 if prompt_type == "decoder"
455 else self.mm_encoder_cache_size
456 )
457 if prompt_len > max_prompt_len:
458 if self.supports_mm_inputs:
459 suggestion = (
460 "Make sure that `max_model_len` is no smaller than the "
461 "number of text tokens plus multimodal tokens. For image "
462 "inputs, the number of image tokens depends on the number "
463 "of images, and possibly their aspect ratios as well."
464 )
465 else:
466 suggestion = (
467 "Make sure that `max_model_len` is no smaller than the "
468 "number of text tokens."
469 )
470
471 raise VLLMValidationError(
472 f"The {prompt_type} prompt (length {prompt_len}) is "
473 f"longer than the maximum model length of {max_prompt_len}. "
474 f"{suggestion}"
475 )
476 elif prompt_len == max_prompt_len and model_config.runner_type == "generate":
477 suggestion = (
478 "Make sure that `max_model_len` is no smaller than the "
479 "number of text tokens (prompt + requested output tokens)."
480 )
481 raise VLLMValidationError(
482 f"The {prompt_type} prompt (length {prompt_len}) plus the number of "
483 f"requested output tokens (at least 1) is longer than the maximum "
484 f"model length of {max_prompt_len}. {suggestion}"
485 )

487 def _validate_model_input(
488 self,
489 prompt_input: SingletonInput,
490 prompt_type: Literal["encoder", "decoder"],
491 ) -> None:
492 model_config = self.model_config
493 tokenizer = self.tokenizer
494
495 prompt_ids = (
496 None
497 if prompt_input["type"] == "embeds"
498 else prompt_input["prompt_token_ids"]
499 )
500 prompt_embeds = (
501 prompt_input["prompt_embeds"] if prompt_input["type"] == "embeds" else None
502 )
503
504 prompt_len = length_from_prompt_token_ids_or_embeds(prompt_ids, prompt_embeds)
505 self._validate_prompt_len(prompt_len, prompt_type)
506
507 if prompt_input["type"] == "multimodal":
508 decoder_mm_positions = prompt_input["mm_placeholders"]
509 for modality, mm_positions in decoder_mm_positions.items():
510 for mm_position in mm_positions:
511 num_embeds = mm_position.get_num_embeds()
512 if num_embeds > self.mm_encoder_cache_size:
513 raise VLLMValidationError(
514 f"The {prompt_type} prompt contains a(n) {modality} item "
515 f"with {num_embeds} embedding tokens, which exceeds the "
516 f"pre-allocated encoder cache size "
517 f"{self.mm_encoder_cache_size}. Please reduce the input "
518 f"size or increase the encoder cache size "
519 f"by setting --limit-mm-per-prompt at startup."
520 )
521
522 if prompt_ids and tokenizer is not None:
523 max_input_id = max(prompt_ids, default=0)
524 min_input_id = min(prompt_ids, default=0)
525
526 # NOTE: tokenizer.max_token_id is the tokenizer’s vocab size while
527 # self.model_config.get_vocab_size() is the model’s vocab size.
528 # For Qwen3 models, the language model has extra tokens that do
529 # not exist in the tokenizer, and vice versa for multimodal
530 # placeholder tokens in some multimodal models.
531 # See https://github.com/QwenLM/Qwen3/issues/29#issuecomment-1933720399 # noqa: E501
532 # and https://github.com/vllm-project/vllm/pull/22471#discussion_r2312251421 # noqa: E501
533
534 # Here we take the max of the two to determine if a token id is
535 # truly out-of-vocabulary.
536 model_vocab_size = model_config.get_vocab_size()
537 # A negative id is out of vocabulary just like an over-large one,
538 # but is not caught by the upper-bound check below. Reject it here
539 # so it is not used as an embedding index downstream. This
540 # validation path is shared by generate, embedding and pooling
541 # requests, so the check covers all three.
542 if min_input_id < 0:
543 raise VLLMValidationError(
544 f"Token id {min_input_id} is out of vocabulary"
545 )
546 if max_input_id > max(tokenizer.max_token_id, model_vocab_size - 1):
547 raise VLLMValidationError(
548 f"Token id {max_input_id} is out of vocabulary"
549 )
550
551 def _validate_model_inputs(
552 self,
553 encoder_input: SingletonInput | None,
554 decoder_input: SingletonInput,
555 ):
556 if encoder_input is not None:
557 self._validate_model_input(encoder_input, prompt_type="encoder")
558
559 self._validate_model_input(decoder_input, prompt_type="decoder")
