79class EngineCoreClient(ABC):
80 """
81 EngineCoreClient: subclasses handle different methods for pushing
82 and pulling from the EngineCore for asyncio / multiprocessing.
83
84 Subclasses:
85 * InprocClient: In process EngineCore (for V0-style LLMEngine use)
86 * SyncMPClient: ZMQ + background proc EngineCore (for LLM)
87 * AsyncMPClient: ZMQ + background proc EngineCore w/ asyncio (for AsyncLLM)
88 """

90 @staticmethod
91 def make_client(
92 multiprocess_mode: bool,
93 asyncio_mode: bool,
94 vllm_config: VllmConfig,
95 executor_class: type[Executor],
96 log_stats: bool,
97 renderer: BaseRenderer | None = None,
98 ) -> "EngineCoreClient":
99 # renderer is passed through to the multiprocess clients, which start
100 # the frontend MM warmup (renderer.start_mm_warmup_in_background) once
101 # the engine-core processes have been forked, so warmup overlaps
102 # engine-core load without a live thread at fork() time. In-process
103 # clients do not take a renderer: the engine core is built in this
104 # process, so there is no other process to overlap with and the MM
105 # warmup runs synchronously inside renderer.warmup() instead.
106 # TODO: support this for debugging purposes.
107 if asyncio_mode and not multiprocess_mode:
108 raise NotImplementedError(
109 "Running EngineCore in asyncio without multiprocessing "
110 "is not currently supported."
111 )
112
113 if multiprocess_mode and asyncio_mode:
114 return EngineCoreClient.make_async_mp_client(
115 vllm_config,
116 executor_class,
117 log_stats,
118 renderer=renderer,
119 )
120
121 if multiprocess_mode and not asyncio_mode:
122 return SyncMPClient(
123 vllm_config,
124 executor_class,
125 log_stats,
126 renderer=renderer,
127 )
128
129 return InprocClient(vllm_config, executor_class, log_stats)

447@dataclass
448class BackgroundResources:
449 """Used as a finalizer for clean shutdown, avoiding
450 circular reference back to the client object."""
451
452 ctx: zmq.Context
453 # If CoreEngineProcManager, it manages local engines;
454 # if CoreEngineActorManager, it manages all engines.
455 engine_manager: CoreEngineProcManager | CoreEngineActorManager | None = None
456 coordinator: DPCoordinator | None = None
457 output_socket: zmq.Socket | zmq.asyncio.Socket | None = None
458 input_socket: zmq.Socket | zmq.asyncio.Socket | None = None
459 first_req_send_socket: zmq.asyncio.Socket | None = None
460 first_req_rcv_socket: zmq.asyncio.Socket | None = None
461 stats_update_socket: zmq.asyncio.Socket | None = None
462 output_queue_task: asyncio.Task | None = None
463 stats_update_task: asyncio.Task | None = None
464 shutdown_path: str | None = None
465
466 # Set if any of the engines are dead. Here so that the output
467 # processing threads can access it without holding a ref to the client.
468 engine_dead: bool = False
469
470 def __call__(self):
471 """Clean up background resources."""
472
473 logger.debug_once("[shutdown] MPClient: background resource cleanup start")
474 self.engine_dead = True
475 if self.engine_manager is not None:
476 self.engine_manager.shutdown(
477 timeout=envs.VLLM_WORKER_SHUTDOWN_TIMEOUT_SECONDS
478 )
479 if self.coordinator is not None:
480 self.coordinator.shutdown()
481
482 if isinstance(self.output_socket, zmq.asyncio.Socket):
483 # Async case.
484 loop = self.output_queue_task._loop if self.output_queue_task else None
485
486 sockets = (
487 self.output_socket,
488 self.input_socket,
489 self.first_req_send_socket,
490 self.first_req_rcv_socket,
491 self.stats_update_socket,
492 )
493
494 tasks = (self.output_queue_task, self.stats_update_task)
495
496 def close_sockets_and_tasks():
497 close_sockets(sockets)
498 for task in tasks:
499 if task is not None and not task.done():
500 with contextlib.suppress(Exception):
501 task.cancel()
502
503 if loop is not None:
504 if in_loop(loop):
505 close_sockets_and_tasks()
506 elif not loop.is_closed():
507 loop.call_soon_threadsafe(close_sockets_and_tasks)
508 else:
509 # Loop has been closed, try to clean up directly.
510 del tasks
511 del close_sockets_and_tasks
512 close_sockets(sockets)
513 del self.output_queue_task
514 del self.stats_update_task
515 else:
516 # Sync case.
517
518 # ZMQ context termination can hang if the sockets
519 # aren't explicitly closed first.
520 close_sockets((self.output_socket, self.input_socket))
521
522 if self.shutdown_path is not None:
523 # We must ensure that the sync output socket is
524 # closed cleanly in its own thread.
525 with self.ctx.socket(zmq.PAIR) as shutdown_sender:
526 shutdown_sender.connect(self.shutdown_path)
527 # Send shutdown signal.
528 shutdown_sender.send(b"")
529
530 logger.debug_once("[shutdown] MPClient: background resource cleanup complete")
531
532 def validate_alive(self, frames: Sequence[zmq.Frame]):
533 if len(frames) == 1 and (frames[0].buffer == EngineCoreProc.ENGINE_CORE_DEAD):
534 self.engine_dead = True
535 raise EngineDeadError()

545class MPClient(EngineCoreClient):
546 """
547 MPClient: base client for multi-proc EngineCore.
548 EngineCore runs in a background process busy loop, getting
549 new EngineCoreRequests and returning EngineCoreOutputs
550
551 * pushes EngineCoreRequests via input_socket
552 * pulls EngineCoreOutputs via output_socket
553
554 * AsyncMPClient subclass for AsyncLLM usage
555 * SyncMPClient subclass for LLM usage
556 """
557
558 def __init__(
559 self,
560 asyncio_mode: bool,
561 vllm_config: VllmConfig,
562 executor_class: type[Executor],
563 log_stats: bool,
564 client_addresses: dict[str, Any] | None = None,
565 renderer: BaseRenderer | None = None,
566 ):
567 self.vllm_config = vllm_config
568 self._renderer: BaseRenderer | None = renderer
569
570 # ZMQ setup.
571 sync_ctx = zmq.Context(io_threads=2)
572 self.ctx = zmq.asyncio.Context(sync_ctx) if asyncio_mode else sync_ctx
573
574 # This will ensure resources created so far are closed
575 # when the client is garbage collected, even if an
576 # exception is raised mid-construction.
577 self.resources = BackgroundResources(ctx=sync_ctx)
578 self._finalizer = weakref.finalize(self, self.resources)
579 success = False
580 try:
581 # State used for data parallel.
582 self.engines_running = False
583 parallel_config = vllm_config.parallel_config
584 # Elastic EP can remove a rank and later add it back with the same
585 # identity. The client input ROUTER needs handover to allow the new
586 # engine to replace the dead connection.
587 enable_input_socket_handover = parallel_config.enable_elastic_ep
588
589 self.stats_update_address: str | None = None
590 tensor_queue: Queue | None = None
591 if client_addresses:
592 # Engines are managed externally to this client.
593 input_address = client_addresses["input_address"]
594 output_address = client_addresses["output_address"]
595 self.stats_update_address = client_addresses.get("stats_update_address")
596 # Tensor queues passed via client_addresses for multi-API-server case
597 tensor_queue = client_addresses.get("tensor_queue")
598 self.input_socket = self.resources.input_socket = make_zmq_socket(
599 self.ctx,
600 input_address,
601 zmq.ROUTER,
602 bind=True,
603 router_handover=enable_input_socket_handover,
604 )
605 self.resources.output_socket = make_zmq_socket(
606 self.ctx, output_address, zmq.PULL
607 )
608
609 # Report bound endpoints back so the parent can forward
610 # them to engines (mirrors the DPCoordinator pattern).
611 actual_address_pipe: Connection | None = client_addresses.get(
612 "actual_address_pipe"
613 )
614 if actual_address_pipe is not None:
615 try:
616 actual_input = self.input_socket.getsockopt(
617 zmq.LAST_ENDPOINT
618 ).decode()
619 actual_output = self.resources.output_socket.getsockopt(
620 zmq.LAST_ENDPOINT
621 ).decode()
622 actual_address_pipe.send(
623 {
624 "input_address": actual_input,
625 "output_address": actual_output,
626 }
627 )
628 finally:
629 actual_address_pipe.close()
630 # Engines are managed externally: this process does not fork
631 # them, so there is no fork race; start the MM warmup now.
632 self._start_mm_warmup()
633 else:
634 # Engines are managed by this client.
635 addresses = get_engine_zmq_addresses(vllm_config)
636 self.input_socket = self.resources.input_socket = make_zmq_socket(
637 self.ctx,
638 addresses.inputs[0],
639 zmq.ROUTER,
640 bind=True,
641 router_handover=enable_input_socket_handover,
642 )
643 self.resources.output_socket = make_zmq_socket(
644 self.ctx, addresses.outputs[0], zmq.PULL
645 )
646
647 # Resolve ``tcp://host:0`` placeholders to bound endpoints
648 # before engines DEALER-connect. No-op for IPC.
649 addresses.inputs[0] = self.input_socket.getsockopt(
650 zmq.LAST_ENDPOINT
651 ).decode()
652 addresses.outputs[0] = self.resources.output_socket.getsockopt(
653 zmq.LAST_ENDPOINT
654 ).decode()
655
656 with launch_core_engines(
657 vllm_config, executor_class, log_stats, addresses
658 ) as engine_launch:
659 self.resources.coordinator = engine_launch.coordinator
660 self.resources.engine_manager = engine_launch.engine_manager
661 coordinator = engine_launch.coordinator
662 addresses = engine_launch.addresses
663 tensor_queue = engine_launch.tensor_queue
664 # Engine-core processes have now all been forked/started
665 # (CoreEngineProcManager.proc.start()). It is now safe to
666 # launch the frontend background MM warmup: it must not
667 # run while fork() is in flight (a live thread holding a
668 # lock would deadlock the forked child), but the
669 # engine-core model load (minutes) that follows is exactly
670 # what the warmup should overlap with.
671 self._start_mm_warmup()
672
673 self.stats_update_address = addresses.frontend_stats_publish_address
674 if coordinator is not None:
675 assert self.stats_update_address == (
676 coordinator.get_stats_publish_address()
677 )
678
679 # Serialization setup with tensor queues for multimodal tensor IPC.
680 tensor_ipc_sender: TensorIpcSender | None = None
681 model_config = getattr(vllm_config, "model_config", None)
682 if model_config is not None and model_config.multimodal_config is not None:
683 mm_tensor_ipc = model_config.multimodal_config.mm_tensor_ipc
684 if mm_tensor_ipc == "torch_shm" and tensor_queue is not None:
685 tensor_ipc_sender = TensorIpcSender(tensor_queue)
686
687 self.encoder = MsgpackEncoder(oob_tensor_consumer=tensor_ipc_sender)
688 self.decoder = MsgpackDecoder(EngineCoreOutputs)
689
690 dp_size = parallel_config.data_parallel_size
691 dp_rank = parallel_config.data_parallel_index
692 dp_local_size = parallel_config.data_parallel_size_local
693 offline_mode = parallel_config.data_parallel_rank_local is not None
694 # Client manages local+remote EngineCores in pure internal LB case.
695 # Client manages local EngineCores in hybrid and external LB case.
696 num_ranks = dp_local_size if parallel_config.local_engines_only else dp_size
697 self.engine_ranks_managed = (
698 [dp_rank] if offline_mode else list(range(dp_rank, dp_rank + num_ranks))
699 )
700 assert parallel_config.data_parallel_size_local <= len(
701 self.engine_ranks_managed
702 )
703

705 self.core_engines: list[EngineIdentity] = [
706 rank.to_bytes(2, "little") for rank in self.engine_ranks_managed
707 ]

709 # Wait for ready messages from each engine on the input socket.
710 identities = set(self.core_engines)
711 sync_input_socket = zmq.Socket.shadow(self.input_socket)
712 while identities:

714 timeout=VLLM_ENGINE_READY_TIMEOUT_S * 1000 # convert to ms

716 raise TimeoutError(
717 f"Timed out waiting for engine core processes to "
718 f"start. This is often caused by slow weight loading "
719 f"for large models. Waited "
720 f"{VLLM_ENGINE_READY_TIMEOUT_S}s (configured by "
721 f"VLLM_ENGINE_READY_TIMEOUT_S). To increase the "
722 f"timeout, set the environment variable: "
723 f"VLLM_ENGINE_READY_TIMEOUT_S=<seconds>"
724 )
725 identity, payload = sync_input_socket.recv_multipart()
726 identities.remove(identity)
727 self._apply_ready_response(payload)
728
729 self.core_engine: EngineIdentity = self.core_engines[0]
730 self.utility_results: dict[int, AnyFuture] = {}
731
732 # Start monitoring engine core processes for unexpected failures
733 self.start_engine_core_monitor()
734
735 success = True
736 finally:
737 if not success:
738 self._finalizer()

1105 assert output_socket is not None

1546 def get_core_engine_for_request(self, request: EngineCoreRequest) -> EngineIdentity:
1547 # Engines are in rank order.
1548 if (eng_index := request.data_parallel_rank) is None and (
1549 eng_index := get_late_interaction_engine_index(
1550 request.pooling_params, len(self.core_engines)
1551 )
1552 ) is None:
1553 current_counts = self.lb_engines
1554 # TODO use P2C alg for larger DP sizes
1555 num_engines = len(current_counts)
1556 min_score: float = sys.maxsize
1557 eng_index = 0
1558 for i in range(num_engines):
1559 # Start from client_index to help with balancing when engines
1560 # are empty.
1561 idx = (self.eng_start_index + i) % num_engines
1562 waiting, running, kv_cache_usage = current_counts[idx]
1563 # Estimate engine load as the greater of the coordinator's
1564 # latest (waiting + running) snapshot and this client's own
1565 # in-flight count (scaled by the number of clients). The
1566 # in-flight floor is exact and can't be erased by a snapshot
1567 # rebind, so a burst spreads round-robin even when snapshots
1568 # race with routing decisions; the snapshot raises the score
1569 # when other clients or stale requests load the engine.
1570 inflight = self.engine_inflight[self.core_engines[idx]]
1571 score: float = max(self.client_count * inflight, waiting + running)
1572 if waiting:
1573 # Waiting requests are penalized in proportion to KV cache
1574 # pressure: a queue on a KV-bound engine drains slowly, so
1575 # new requests should strongly prefer other engines. With
1576 # low KV usage the queue is transient (e.g. mid-burst) and
1577 # the penalty stays off, preserving exact round-robin.
1578 # Ramps from 0 at <=50% usage to 3x waiting at 100%.
1579 score += waiting * 6.0 * max(0.0, kv_cache_usage - 0.5)
1580 if score < min_score:
1581 min_score = score
1582 eng_index = idx
1583 # Increment local waiting count for better balancing between stats
1584 # updates from the coordinator (which happen every 100ms).
1585 current_counts[eng_index][0] += self.client_count
1586 # Rotate the scan start so that ties (equal scores, e.g. right
1587 # after a coordinator stats reset when engines look equally loaded)
1588 # don't systematically favor the same engine. This removes the
1589 # fixed tie-break bias without affecting load-aware decisions when
1590 # scores actually differ.
1591 self.eng_start_index = (self.eng_start_index + 1) % num_engines
1592
1593 chosen_engine = self.core_engines[eng_index]
1594 # Record which engine is chosen for this request, to handle aborts.
1595 self.reqs_in_flight[request.request_id] = chosen_engine
1596 self.engine_inflight[chosen_engine] += 1
1597 return chosen_engine
