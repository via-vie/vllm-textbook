51class RequestOutputCollector:
52 """
53 Collects streamed RequestOutputs per individual request,
54 for hand-off to the consuming asyncio generate task.
55
56 When streaming deltas, RequestOutputs are merged if the
57 producer gets ahead of the consumer.
58 """
59
60 def __init__(self, output_kind: RequestOutputKind, request_id: str):
61 self.aggregate = output_kind == RequestOutputKind.DELTA
62 self.request_id = request_id
63 self.output: RequestOutput | PoolingRequestOutput | Exception | None = None
64 self.ready = asyncio.Event()
65
66 self._input_stream_task: asyncio.Task | None = None
67
68 def put(self, output: RequestOutput | PoolingRequestOutput | Exception) -> None:
69 """Non-blocking put operation."""
70 if self.output is None or isinstance(output, Exception):
71 self.output = output
72 self.ready.set()
73 elif isinstance(self.output, RequestOutput) and isinstance(
74 output, RequestOutput
75 ):
76 # This ensures that request outputs with different request indexes
77 # (if n > 1) do not override each other.
78 self.output.add(output, aggregate=self.aggregate)
79 elif isinstance(self.output, PoolingRequestOutput) and isinstance(
80 output, PoolingRequestOutput
81 ):
82 self.output = output
83
84 async def get(self) -> RequestOutput | PoolingRequestOutput:
85 """Get operation blocks on put event."""
86 while (output := self.output) is None:
87 await self.ready.wait()
88 self.output = None
89 self.ready.clear()
90 if isinstance(output, Exception):
91 raise output
92 return output
93
94 def get_nowait(self) -> RequestOutput | PoolingRequestOutput | None:
95 """Non-blocking get operation."""
96 output = self.output
97 if output is not None:
98 self.output = None
99 self.ready.clear()
100 if isinstance(output, Exception):
101 raise output
102 return output
103
104 def close(self):
105 if self._input_stream_task is not None:
106 self._input_stream_task.cancel()
107 self._input_stream_task = None
108
109 def __del__(self):
110 if (task := self._input_stream_task) is not None:
111 task.get_loop().call_soon_threadsafe(task.cancel)
112 self._input_stream_task = None

286 def make_request_output(
287 self,
288 new_token_ids: list[int],
289 pooling_output: torch.Tensor | None,
290 finish_reason: FinishReason | None,
291 stop_reason: int | str | None,
292 kv_transfer_params: dict[str, Any] | None = None,
293 ec_transfer_params: dict[str, Any] | None = None,
294 ) -> RequestOutput | PoolingRequestOutput | None:
295 finished = finish_reason is not None
296 final_only = self.output_kind == RequestOutputKind.FINAL_ONLY
297
298 if not finished and final_only:
299 # Only the final output is required in FINAL_ONLY mode.
300 return None
301
302 if self.stream_interval > 1:
303 assert self.detokenizer is not None
304
305 # Send output request only when
306 # 1. It has finished, or
307 # 2. It is the first token, or
308 # 3. It has reached the stream interval number of tokens
309 if not (
310 finished
311 or self.sent_tokens_offset == 0
312 or self.detokenizer.num_output_tokens() - self.sent_tokens_offset
313 >= self.stream_interval
314 ):
315 return None
316
317 if self.output_kind == RequestOutputKind.DELTA:
318 # Send tokens from the offset in DELTA mode, otherwise all
319 # tokens are sent.
320 new_token_ids = self.detokenizer.output_token_ids[
321 self.sent_tokens_offset :
322 ]
323 self.sent_tokens_offset = self.detokenizer.num_output_tokens()
324
325 external_req_id = self.external_req_id
326
327 if pooling_output is not None:
328 return self._new_request_output(
329 external_req_id,
330 [self._new_pooling_output(pooling_output)],
331 finished,
332 )
333
334 output = self._new_completion_output(new_token_ids, finish_reason, stop_reason)
335
336 if self.parent_req is None:
337 outputs = [output]
338 else:
339 outputs, finished = self.parent_req.get_outputs(self.request_id, output)
340 if not outputs:
341 return None
342 external_req_id = self.parent_req.external_req_id
343
344 return self._new_request_output(
345 external_req_id,
346 outputs,
347 finished,
348 kv_transfer_params,
349 ec_transfer_params,
350 )

352 def _new_request_output(
353 self,
354 external_req_id: str,
355 outputs: list[CompletionOutput] | list[PoolingOutput],
356 finished: bool,
357 kv_transfer_params: dict[str, Any] | None = None,
358 ec_transfer_params: dict[str, Any] | None = None,
359 ) -> RequestOutput | PoolingRequestOutput:
360 # If prompt embeds were used, put placeholder prompt token ids
361 prompt_token_ids = self.prompt_token_ids
362 if prompt_token_ids is None and self.prompt_embeds is not None:
363 prompt_token_ids = [0] * len(self.prompt_embeds)
364 assert prompt_token_ids is not None
365
366 first_output = outputs[0]
367 if isinstance(first_output, PoolingOutput):
368 assert len(outputs) == 1
369 return PoolingRequestOutput(
370 request_id=external_req_id,
371 outputs=first_output,
372 num_cached_tokens=self.num_cached_tokens,
373 prompt_token_ids=prompt_token_ids,
374 finished=finished,
375 )
376 assert self.logprobs_processor is not None
377 if self.output_kind == RequestOutputKind.DELTA:
378 # Side effect: logprobs processor forgets prompt logprobs
379 prompt_logprobs = self.logprobs_processor.pop_prompt_logprobs()
380 else:
381 prompt_logprobs = self.logprobs_processor.prompt_logprobs
382
383 return RequestOutput(
384 request_id=external_req_id, # request_id is what was provided externally
385 lora_request=self.lora_request,
386 prompt=self.prompt,
387 prompt_token_ids=prompt_token_ids,
388 prompt_logprobs=prompt_logprobs,
389 outputs=cast(list[CompletionOutput], outputs),
390 finished=finished,
391 kv_transfer_params=kv_transfer_params,
392 ec_transfer_params=ec_transfer_params,
393 num_cached_tokens=self.num_cached_tokens,
394 num_cache_creation_tokens=self.num_cache_creation_tokens,
395 metrics=self.stats,
396 )

398 def _new_completion_output(
399 self,
400 token_ids: list[int],
401 finish_reason: FinishReason | None,
402 stop_reason: int | str | None,
403 ) -> CompletionOutput:
404 assert self.detokenizer is not None
405 assert self.logprobs_processor is not None
406 finished = finish_reason is not None
407 delta = self.output_kind == RequestOutputKind.DELTA
408
409 # Prepare text and token_ids, based on delta mode
410 text = self.detokenizer.get_next_output_text(finished, delta)
411 if not delta:
412 token_ids = self.detokenizer.output_token_ids
413
414 # Prepare logprobs, based on delta mode
415 logprobs = self.logprobs_processor.logprobs
416 if delta and logprobs:
417 num_new_tokens = len(token_ids)
418 # Avoid [-0:], which returns the full accumulated history when a
419 # delta contains no new token IDs. [:0] preserves the concrete
420 # list or FlatLogprobs representation while returning no entries.
421 logprobs = logprobs[-num_new_tokens:] if num_new_tokens else logprobs[:0]
422
423 sampling_mask = None
424 if finished and self.sampling_mask_chunks:
425 sampling_mask = SamplingMask(
426 [chunk.token_ids.tolist() for chunk in self.sampling_mask_chunks]
427 )
428
429 # Concatenate routed experts on finish
430 routed_experts = None
431 if finished and self.routed_experts_chunks:
432 routed_experts = np.concatenate(self.routed_experts_chunks, axis=0)
433
434 return CompletionOutput(
435 index=self.request_index,
436 text=text,
437 token_ids=token_ids,
438 routed_experts=routed_experts,
439 sampling_mask=sampling_mask,
440 logprobs=logprobs,
441 cumulative_logprob=self.logprobs_processor.cumulative_logprob,
442 finish_reason=str(finish_reason) if finished else None,
443 stop_reason=stop_reason if finished else None,
444 spec_decode_metrics=self.spec_decode_metrics if finished else None,
445 )

447 def _new_pooling_output(self, pooling_output: torch.Tensor) -> PoolingOutput:
448 return PoolingOutput(data=pooling_output)

500 def abort_requests(self, request_ids: Iterable[str], internal: bool) -> list[str]:
501 """Abort a list of requests.
502
503 The request_ids may be either external request IDs (those passed to
504 InputProcessor.process_inputs()) or internal request IDs (those randomly
505 generated when creating the EngineCoreRequest).
506
507 If an external request ID is provided, and that external request ID
508 was used for multiple requests, all requests associated with that external
509 request ID are aborted.
510
511 In the case of parallel sampling, a request ID may be used to identify
512 a parent request, in which case the associated child requests are aborted
513 also.
514 """
515 internal_req_ids = []
516 for request_id in request_ids:
517 if internal:
518 # Internal ID - this may be a parent request
519 internal_req_ids.append(request_id)
520
521 # Remove internal ID from the external->internal mapping
522 if req_state := self.request_states.get(request_id):
523 external_req_id = req_state.external_req_id
524 internal_ids = self.external_req_ids[external_req_id]
525 internal_ids.remove(request_id)
526 if not internal_ids:
527 del self.external_req_ids[external_req_id]
528 elif internal_ids := self.external_req_ids.pop(request_id, []):
529 # External ID - abort all requests in the external->internal mapping
530 internal_req_ids.extend(internal_ids)
531
532 request_ids_to_abort = []
533 for request_id in internal_req_ids:
534 req_state = self.request_states.pop(request_id, None)
535 if req_state is not None:
536 self.lora_states.request_finished(request_id, req_state.lora_name)
537 request_ids_to_abort.append(request_id)
538 # Produce final abort output.
539 if req_state.queue is not None and (
540 request_output := req_state.make_request_output(
541 new_token_ids=[],
542 # Set pooling_output is not None to
543 # correctly enter the abort pooling branch
544 pooling_output=EMPTY_CPU_TENSOR
545 if req_state.detokenizer is None
546 else None,
547 finish_reason=FinishReason.ABORT,
548 stop_reason=None,
549 kv_transfer_params=None,
550 ec_transfer_params=None,
551 )
552 ):
553 req_state.queue.put(request_output)
554 elif parent := self.parent_requests.get(request_id):
555 # Abort children prior to removing the parent.
556 if parent.child_requests:
557 child_reqs = list(parent.child_requests)
558 child_reqs = self.abort_requests(child_reqs, internal=True)
559 request_ids_to_abort.extend(child_reqs)
560 self.parent_requests.pop(request_id, None)
561 self._update_admission_stats()
562 return request_ids_to_abort

564 def add_request(
565 self,
566 request: EngineCoreRequest,
567 prompt: str | None,
568 parent_req: ParentRequest | None = None,
569 request_index: int = 0,
570 queue: RequestOutputCollector | None = None,
571 ) -> None:
572 request_id = request.request_id
573 req_state = self.request_states.get(request_id)
574 if req_state is not None:
575 self._update_streaming_request_state(req_state, request, prompt)
576 return
577
578 req_state = RequestState.from_new_request(
579 tokenizer=self.tokenizer,
580 request=request,
581 prompt=prompt,
582 parent_req=parent_req,
583 request_index=request_index,
584 queue=queue,
585 log_stats=self.log_stats,
586 stream_interval=self.stream_interval,
587 )
588 self.request_states[request_id] = req_state
589 if parent_req:
590 self.parent_requests[parent_req.request_id] = parent_req
591
592 # Track the external_req_id -> [internal_req_id, ...] mapping
593 self.external_req_ids[req_state.external_req_id].append(request_id)
594 self._update_admission_stats()

596 def _update_streaming_request_state(
597 self, req_state: RequestState, request: EngineCoreRequest, prompt: str | None
598 ) -> None:
599 """Queue a streaming update instead of immediately applying it."""
600 if not request.resumable:
601 # Final request - just mark completion, don't add its dummy tokens.
602 if req_state.input_chunk_queue is None:
603 # Engine already finished - emit final output and clean up.
604 self._finish_request(req_state)
605 if req_state.queue is not None:
606 # Emit a final output with finished=True
607 # to unblock the generate() loop.
608 req_state.queue.put(STREAM_FINISHED)
609 elif req_state.input_chunk_queue:
610 req_state.input_chunk_queue[-1].final = True
611 else:
612 req_state.streaming_input = False
613 return
614
615 update = StreamingUpdate(
616 prompt=prompt,
617 prompt_token_ids=request.prompt_token_ids,
618 arrival_time=request.arrival_time,
619 )
620
621 # Apply request updates now if the last input already completed.
622 if req_state.input_chunk_queue is None:
623 req_state.apply_streaming_update(update)
624 req_state.input_chunk_queue = deque()
625 else:
626 # Queue the streaming update otherwise.
627 req_state.input_chunk_queue.append(update)

629 def process_outputs(
630 self,
631 engine_core_outputs: list[EngineCoreOutput],
632 engine_core_timestamp: float | None = None,
633 iteration_stats: IterationStats | None = None,
634 ) -> OutputProcessorOutput:
635 """
636 Process the EngineCoreOutputs:
637 1) Compute stats for logging
638 2) Detokenize
639 3) Create and handle RequestOutput objects:
640 * If there is a queue (for usage with AsyncLLM),
641 put the RequestOutput objects into the queue for
642 handling by the per-request generate() tasks.
643
644 * If there is no queue (for usage with LLMEngine),
645 return a list of RequestOutput objects.
646
647 NOTE FOR DEVELOPERS
648
649 vLLM V1 minimizes the number of python loops over the full
650 batch to ensure system overheads are minimized. This is the
651 only function that should loop over EngineCoreOutputs.
652
653 If you need to touch every element of the batch, do it from
654 within the loop below.
655 """
656
657 request_outputs: list[RequestOutput | PoolingRequestOutput] = []
658 reqs_to_abort: list[str] = []
659 for engine_core_output in engine_core_outputs:
660 req_id = engine_core_output.request_id
661 req_state = self.request_states.get(req_id)
662 if req_state is None:
663 # Ignore output for already-aborted request.
664 continue
665
666 # 1) Compute stats for this iteration.
667 self._update_stats_from_output(
668 req_state, engine_core_output, engine_core_timestamp, iteration_stats
669 )
670
671 new_token_ids = engine_core_output.new_token_ids
672 pooling_output = engine_core_output.pooling_output
673 finish_reason = engine_core_output.finish_reason
674 stop_reason = engine_core_output.stop_reason
675 kv_transfer_params = engine_core_output.kv_transfer_params
676 ec_transfer_params = engine_core_output.ec_transfer_params
677 if engine_core_output.routed_experts is not None:
678 req_state.routed_experts_chunks.append(
679 engine_core_output.routed_experts
680 )
681
682 if req_state.is_prefilling:
683 if engine_core_output.prefill_stats is not None:
684 req_state.num_cached_tokens = (
685 engine_core_output.prefill_stats.num_cached_tokens
686 )
687 req_state.num_cache_creation_tokens = (
688 engine_core_output.prefill_stats.num_cache_creation_tokens
689 )
690 req_state.is_prefilling = False
691
692 if engine_core_output.spec_decode_metrics is not None:
693 req_state.spec_decode_metrics = engine_core_output.spec_decode_metrics
694
695 if pooling_output is None:
696 assert req_state.detokenizer is not None
697 assert req_state.logprobs_processor is not None
698 if engine_core_output.new_sampling_mask is not None:
699 req_state.sampling_mask_chunks.append(
700 engine_core_output.new_sampling_mask
701 )
702 # 2) Detokenize the token ids into text and perform stop checks.
703 stop_string = req_state.detokenizer.update(
704 new_token_ids, finish_reason == FinishReason.STOP
705 )
706 if stop_string:
707 finish_reason = FinishReason.STOP
708 stop_reason = stop_string
709
710 # 3) Compute sample and prompt logprobs for request,
711 # if required.
712 req_state.logprobs_processor.update_from_output(engine_core_output)
713
714 # 4) Create and handle RequestOutput objects.
715 if request_output := req_state.make_request_output(
716 new_token_ids,
717 pooling_output,
718 finish_reason,
719 stop_reason,
720 kv_transfer_params,
721 ec_transfer_params,
722 ):
723 if req_state.streaming_input:
724 request_output.finished = False
725
726 if req_state.queue is not None:
727 # AsyncLLM: put into queue for handling by generate().
728 req_state.queue.put(request_output)
729 else:
730 # LLMEngine: return list of RequestOutputs.
731 request_outputs.append(request_output)
732
733 # Free completed requests.
734 if finish_reason is not None:
735 if req_state.streaming_input:
736 if req_state.input_chunk_queue:
737 update = req_state.input_chunk_queue.popleft()
738 req_state.apply_streaming_update(update)
739 else:
740 req_state.input_chunk_queue = None
741 else:
742 self._finish_request(req_state)
743 if not engine_core_output.finished:
744 # If req not finished in EngineCore, but Detokenizer
745 # detected stop string, abort needed in EngineCore.
746 reqs_to_abort.append(req_id)
747
748 # Track per-request stats
749 self._update_stats_from_finished(
750 req_state, finish_reason, iteration_stats
751 )
752 if self.tracing_enabled:
753 self.do_tracing(engine_core_output, req_state, iteration_stats)
754
755 return OutputProcessorOutput(
756 request_outputs=request_outputs,
757 reqs_to_abort=reqs_to_abort,
758 )

760 def _finish_request(self, req_state: RequestState) -> None:
761 req_id = req_state.request_id
762 self.request_states.pop(req_id)
763
764 internal_ids = self.external_req_ids[req_state.external_req_id]
765 internal_ids.remove(req_id)
766 if not internal_ids:
767 del self.external_req_ids[req_state.external_req_id]
768
769 # Remove parent request if applicable.
770 parent_req = req_state.parent_req
771 if parent_req and not parent_req.child_requests:
772 self.parent_requests.pop(parent_req.request_id, None)
773 self._update_admission_stats()
774
775 def _update_admission_stats(self) -> None:
776 if self.admission_stats is not None:
777 self.admission_stats.set_num_requests(self.get_num_unfinished_requests())
778
779 def update_scheduler_stats(self, scheduler_stats: SchedulerStats | None):
780 self.lora_states.update_scheduler_stats(scheduler_stats)

782 def do_tracing(
783 self,
784 engine_core_output: EngineCoreOutput,
785 req_state: RequestState,
786 iteration_stats: IterationStats | None,
787 ) -> None:
788 assert req_state.stats is not None
789 assert iteration_stats is not None
790
791 metrics = req_state.stats
792 arrival_time_ns = int(metrics.arrival_time * 1e9)
793 trace_context = extract_trace_context(engine_core_output.trace_headers)
794 prompt_length = length_from_prompt_token_ids_or_embeds(
795 req_state.prompt_token_ids, req_state.prompt_embeds
796 )
797
798 # Calculate timing metrics
799 e2e_time = iteration_stats.iteration_timestamp - metrics.arrival_time
800 queued_time = metrics.scheduled_ts - metrics.queued_ts
801 prefill_time = metrics.first_token_ts - metrics.scheduled_ts
802 decode_time = metrics.last_token_ts - metrics.first_token_ts
803 inference_time = metrics.last_token_ts - metrics.scheduled_ts
804
805 # Build attributes dict
806 attributes: dict[str, Any] = {
807 SpanAttributes.GEN_AI_LATENCY_TIME_TO_FIRST_TOKEN: (
808 metrics.first_token_latency
809 ),
810 SpanAttributes.GEN_AI_LATENCY_E2E: e2e_time,
811 SpanAttributes.GEN_AI_LATENCY_TIME_IN_QUEUE: queued_time,
812 SpanAttributes.GEN_AI_USAGE_PROMPT_TOKENS: prompt_length,
813 SpanAttributes.GEN_AI_USAGE_COMPLETION_TOKENS: (
814 metrics.num_generation_tokens
815 ),
816 SpanAttributes.GEN_AI_LATENCY_TIME_IN_MODEL_PREFILL: prefill_time,
817 SpanAttributes.GEN_AI_LATENCY_TIME_IN_MODEL_DECODE: decode_time,
818 SpanAttributes.GEN_AI_LATENCY_TIME_IN_MODEL_INFERENCE: inference_time,
819 SpanAttributes.GEN_AI_REQUEST_ID: req_state.external_req_id,
820 }
821
822 # Add optional request parameters
823 if req_state.top_p:
824 attributes[SpanAttributes.GEN_AI_REQUEST_TOP_P] = req_state.top_p
825 if req_state.max_tokens_param:
826 attributes[SpanAttributes.GEN_AI_REQUEST_MAX_TOKENS] = (
827 req_state.max_tokens_param
828 )
829 if req_state.temperature:
830 attributes[SpanAttributes.GEN_AI_REQUEST_TEMPERATURE] = (
831 req_state.temperature
832 )
833 if req_state.n:
834 attributes[SpanAttributes.GEN_AI_REQUEST_N] = req_state.n
835
836 instrument_manual(
837 span_name="llm_request",
838 start_time=arrival_time_ns,
839 attributes=attributes,
840 context=trace_context,
841 kind=SpanKind.SERVER,
842 )
