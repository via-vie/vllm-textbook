32 def __init__(self):
33 self.token_ids: list[int] = []
34
35 @property
36 def output_token_ids(self) -> list[int]:
37 return self.token_ids
38
39 def num_output_tokens(self) -> int:
40 return len(self.token_ids)
41
42 def update(self, new_token_ids: list[int], stop_terminated: bool) -> str | None:
43 self.token_ids.extend(new_token_ids)
44 return None
45

49 @classmethod
50 def from_new_request(
51 cls,
52 tokenizer: TokenizerLike | None,
53 request: EngineCoreRequest,
54 ) -> "IncrementalDetokenizer":
55 assert request.sampling_params is not None
56
57 if tokenizer is None:
58 # No tokenizer => skipping detokenization.
59 return IncrementalDetokenizer()
60
61 if USE_FAST_DETOKENIZER and isinstance(tokenizer, TokenizersBackend):
62 # Fast tokenizer => use tokenizers library DecodeStream.
63 return FastIncrementalDetokenizer(tokenizer, request)
64
65 # Fall back to slow python-based incremental detokenization.
66 return SlowIncrementalDetokenizer(tokenizer, request)

69class BaseIncrementalDetokenizer(IncrementalDetokenizer, ABC):
70 def __init__(self, request: EngineCoreRequest):
71 super().__init__()
72
73 # Stop strings
74 params = request.sampling_params
75 assert params is not None
76 if params.stop is None:
77 self.stop = []
78 elif isinstance(params.stop, str):
79 self.stop = [params.stop]
80 else:
81 self.stop = params.stop
82 self.min_tokens = params.min_tokens
83 self.include_stop_str_in_output = params.include_stop_str_in_output
84
85 # Number of chars to hold back when stop strings are to be excluded
86 # from streamed output.
87 if self.stop and not self.include_stop_str_in_output:
88 self.stop_buffer_length = max(len(s) for s in self.stop) - 1
89 else:
90 self.stop_buffer_length = 0
91 self._last_output_text_offset: int = 0
92
93 # Generation data
94 self.output_text = ""
95
96 def update(self, new_token_ids: list[int], stop_terminated: bool) -> str | None:
97 """
98 Update RequestState for the request_id by:
99 1) Detokenize the new token ids incrementally.
100 2) Evaluate stop criteria.
101
102 Return matched stop string or None.
103 """
104 if not new_token_ids:
105 # Skip detokenization if no new token ids.
106 return None
107
108 if stop_terminated and not self.include_stop_str_in_output:
109 # If stop-terminated, exclude last token from detokenization
110 # based on include_stop_str_in_output parameter.
111 skipped_stop_token_id = new_token_ids[-1]
112 new_token_ids = new_token_ids[:-1]
113 else:
114 skipped_stop_token_id = None
115
116 # 1) Detokenize the new token ids incrementally.
117 stop_check_offset = len(self.output_text)
118 for new_token_id in new_token_ids:
119 self.token_ids.append(new_token_id)
120 self.output_text += self.decode_next(new_token_id)
121 # Support min_tokens, see https://github.com/vllm-project/vllm/pull/22014
122 if self.min_tokens and self.num_output_tokens() <= self.min_tokens:
123 stop_check_offset = len(self.output_text)
124
125 if skipped_stop_token_id is not None:
126 # Cleanup after skipping detokenization.
127 self.token_ids.append(skipped_stop_token_id)
128
129 # 2) Evaluate stop strings.
130 stop_string = None
131 if self.stop and self.num_output_tokens() > self.min_tokens:
132 stop = check_stop_strings(
133 output_text=self.output_text,
134 new_char_count=len(self.output_text) - stop_check_offset,
135 stop=self.stop,
136 include_in_output=self.include_stop_str_in_output,
137 )
138 if stop is not None:
139 stop_string, truncate_to = stop
140 if truncate_to != -1:
141 self.output_text = self.output_text[:truncate_to]
142
143 return stop_string
144
145 @abstractmethod
146 def decode_next(self, next_token_id: int) -> str:
147 raise NotImplementedError
148
149 def get_next_output_text(self, finished: bool, delta: bool) -> str:
150 """If delta is True, only new text since the last call to
151 this method is returned"""
152
153 # We return the full output text if the sequence is finished.
154 buffer_length = 0 if finished else self.stop_buffer_length
155 if not delta:
156 if not buffer_length:
157 return self.output_text
158 return self.output_text[:-buffer_length]
159
160 length = len(self.output_text) - buffer_length
161 last_offset = self._last_output_text_offset
162 if last_offset < length:
163 self._last_output_text_offset = length
164 return self.output_text[last_offset:length]
165 return ""

168class FastIncrementalDetokenizer(BaseIncrementalDetokenizer):
169 def __init__(self, tokenizer: TokenizersBackend, request: EngineCoreRequest):
170 super().__init__(request)
171
172 sampling_params = request.sampling_params
173 assert sampling_params is not None
174
175 self.request_id = request.request_id
176 self.skip_special_tokens = sampling_params.skip_special_tokens
177
178 self.tokenizer: Tokenizer = tokenizer._tokenizer
179
180 # Use native prefill to prime the decode stream with prompt tokens.
181 # Look up DecodeStream on the module so backend patches (e.g. the
182 # fastokens shim that replaces ``tokenizers.decoders.DecodeStream``)
183 # are honored regardless of import order.
184 self.stream = tokenizers.decoders.DecodeStream(
185 ids=request.prompt_token_ids,
186 skip_special_tokens=self.skip_special_tokens,
187 )
188
189 self.spaces_between_special_tokens = (
190 sampling_params.skip_special_tokens
191 or sampling_params.spaces_between_special_tokens
192 )
193
194 if not self.spaces_between_special_tokens:
195 # Store dict of added token ids so that we can suppress
196 # the spaces between them.
197 added_token_ids = getattr(self.tokenizer, "added_token_ids", None)
198 if added_token_ids is None:
199 self.tokenizer.added_token_ids = added_token_ids = {
200 tid: tok.content
201 for tid, tok in self.tokenizer.get_added_tokens_decoder().items()
202 }
203
204 if added_token_ids:
205 self.last_special = False
206 self.added_token_ids = added_token_ids
207 else:
208 # No added tokens.
209 self.spaces_between_special_tokens = True
210
211 def decode_next(self, next_token_id: int) -> str:
212 token = self._protected_step(next_token_id)
213
214 if not self.spaces_between_special_tokens:
215 special_token = self.added_token_ids.get(next_token_id)
216 is_special = special_token is not None
217 if is_special and self.last_special:
218 # Return raw token string without any prefixed spaces.
219 token = special_token
220 self.last_special = is_special
221
222 return token or ""
223
224 def _protected_step(self, next_token_id: int) -> str | None:
225 try:
226 token = self.stream.step(self.tokenizer, next_token_id)
227 except (OverflowError, TypeError):
228 # Handle rare observed overflow, still to be diagnosed.
229 # See https://github.com/vllm-project/vllm/issues/21951.
230 logger.exception("Encountered invalid token id: %r", next_token_id)
231 token = None
232 except Exception as e:
233 if not str(e).startswith(INVALID_PREFIX_ERR_MSG):
234 raise e
235 # Recover from edge case where tokenizer can produce non-monotonic,
236 # invalid UTF-8 output, which breaks the internal state of
237 # tokenizers' DecodeStream.
238 # See https://github.com/vllm-project/vllm/issues/17448.
239 logger.warning(
240 "Encountered invalid prefix detokenization error"
241 " for request %s, resetting decode stream.",
242 self.request_id,
243 )
244 self.stream = tokenizers.decoders.DecodeStream(
245 skip_special_tokens=self.skip_special_tokens
246 )
247 token = self.stream.step(self.tokenizer, next_token_id)
248 return token

251class SlowIncrementalDetokenizer(BaseIncrementalDetokenizer):
252 def __init__(self, tokenizer: TokenizerLike, request: EngineCoreRequest):
253 super().__init__(request)
254
255 self.tokenizer = tokenizer
256 params = request.sampling_params
257 assert params is not None
258
259 self.prompt_len = length_from_prompt_token_ids_or_embeds(
260 request.prompt_token_ids, request.prompt_embeds
261 )
262
263 # Metadata for incremental detokenization.
264 if request.prompt_token_ids is not None:
265 self.tokens, self.prefix_offset, self.read_offset = (
266 convert_prompt_ids_to_tokens(
267 tokenizer=tokenizer,
268 prompt_ids=request.prompt_token_ids,
269 skip_special_tokens=params.skip_special_tokens,
270 )
271 )
272 else:
273 # Prompt embedding requests cannot be detokenized, in general.
274 self.tokens = [""] * self.prompt_len
275 self.prefix_offset = 0
276 self.read_offset = 0
277
278 self.token_ids.extend(request.prompt_token_ids or [0] * self.prompt_len)
279
280 self.skip_special_tokens = params.skip_special_tokens
281 self.spaces_between_special_tokens = params.spaces_between_special_tokens
282
283 @property
284 def output_token_ids(self) -> list[int]:
285 if self.prompt_len:
286 return self.token_ids[self.prompt_len :]
287 return self.token_ids
288
289 def num_output_tokens(self) -> int:
290 return len(self.token_ids) - self.prompt_len
291
292 def decode_next(self, next_token_id: int) -> str:
293 new_tokens, decoded_text, prefix_offset, read_offset = detokenize_incrementally(
294 tokenizer=self.tokenizer,
295 all_input_ids=self.token_ids,
296 prev_tokens=self.tokens,
297 prefix_offset=self.prefix_offset,
298 read_offset=self.read_offset,
299 skip_special_tokens=self.skip_special_tokens,
300 spaces_between_special_tokens=self.spaces_between_special_tokens,
301 )
302
303 self.tokens.extend(new_tokens)
304 self.prefix_offset = prefix_offset
305 self.read_offset = read_offset
306
307 return decoded_text

310def check_stop_strings(
311 output_text: str,
312 new_char_count: int,
313 stop: list[str],
314 include_in_output: bool,
315) -> tuple[str, int] | None:
316 """Check if any stop strings are matched and truncate sequence
317 output text accordingly.
318
319 Returns tuple (stop_string, offset) if matched or else None.
320
321 Where stop_string is the matched stop string and offset is the
322 length to which output_text should be truncated, or -1 for no
323 truncation.
324
325 When several stop strings match within the newly generated text (for
326 example when speculative decoding appends multiple tokens in a single
327 step), the stop string that completes earliest in the text is selected,
328 so the result matches appending one token at a time. Ties are broken by
329 stop-list order.
330 """
331 if not new_char_count or not stop:
332 return None
333
334 best_stop_str: str | None = None
335 best_stop_index = 0
336 best_end = sys.maxsize
337 for stop_str in stop:
338 stop_string_len = len(stop_str)
339 # Avoid searching already-searched text.
340 stop_index = output_text.find(stop_str, 1 - new_char_count - stop_string_len)
341 if stop_index == -1:
342 continue
343
344 # Prefer the stop string that completes earliest in the text.
345 end = stop_index + stop_string_len
346 if end < best_end:
347 best_stop_str = stop_str
348 best_stop_index = stop_index
349 best_end = end
350
351 if best_stop_str is None:
352 return None
353
354 if include_in_output:
355 # Truncate to end of stop string.
356 if best_end >= len(output_text):
357 # No truncation required.
358 return best_stop_str, -1
359 return best_stop_str, best_end
360
361 # Truncate the output text to the beginning of the stop string.
362 return best_stop_str, best_stop_index
