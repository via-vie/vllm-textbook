32 # [num_reqs x num_generated_tokens, max_num_logprobs + 1]

40 # decoding where the number of generated tokens may be

45 if self.cu_num_generated_tokens is not None:
46 req_idx = self.cu_num_generated_tokens[req_idx]
47 end_idx = req_idx + num_positions
48 return LogprobsLists(
49 self.logprob_token_ids[req_idx:end_idx],
50 self.logprobs[req_idx:end_idx],
51 self.sampled_token_ranks[req_idx:end_idx],
52 None,
53 )

56class SamplingMaskLists(NamedTuple):
57 """CSR sampling masks; a step slice holds one position (``offsets=None``)."""
58
59 # [num_kept_tokens]
60 token_ids: np.ndarray
61 # [num_positions + 1], or None for a single position
62 offsets: np.ndarray | None = None
63 # Unused with one position per request; kept for the wire layout.
64 cu_num_generated_tokens: list[int] | None = None
65
66 def slice_request(self, req_idx: int, num_positions: int) -> "SamplingMaskLists":
67 assert num_positions == 1 and self.offsets is not None
68 return SamplingMaskLists(
69 self.token_ids[self.offsets[req_idx] : self.offsets[req_idx + 1]]
70 )
71
72 def to_nested_list(self) -> list[list[int]]:
73 token_ids = self.token_ids.tolist()
74 if self.offsets is None:
75 return [token_ids]
76 offsets = self.offsets.tolist()
77 return [token_ids[offsets[i] : offsets[i + 1]] for i in range(len(offsets) - 1)]

80class LogprobsTensors(NamedTuple):
81 # [num_reqs x num_generated_tokens, max_num_logprobs + 1]
82 logprob_token_ids: torch.Tensor
83 # [num_reqs x num_generated_tokens, max_num_logprobs + 1]
84 logprobs: torch.Tensor
85 # [num_reqs x num_generated_tokens]
86 selected_token_ranks: torch.Tensor
87 # [num_reqs + 1]
88 cu_num_generated_tokens: list[int] | None = None
89 # [num_reqs + 1]. Set instead of cu_num_generated_tokens when the
90 # boundaries only exist on device (adaptive verification); rides along on
91 # the async D2H copy.
92 cu_num_generated_tokens_tensor: torch.Tensor | None = None
93
94 def tolists(self, cu_num_generated_tokens: list[int] | None = None):
95 if cu_num_generated_tokens is None:
96 if self.cu_num_generated_tokens_tensor is not None:
97 cu_num_generated_tokens = self.cu_num_generated_tokens_tensor.tolist()
98 else:
99 cu_num_generated_tokens = self.cu_num_generated_tokens
100 return LogprobsLists(
101 self.logprob_token_ids.cpu().numpy(),
102 self.logprobs.cpu().numpy(),
103 self.selected_token_ranks.cpu().numpy(),
104 cu_num_generated_tokens,
105 )
106
107 def to_cpu_nonblocking(self) -> "LogprobsTensors":
108 if self.logprob_token_ids.device.type == "cpu":
109 return self
110 cu_tensor = self.cu_num_generated_tokens_tensor
111 if cu_tensor is not None:
112 cu_tensor = cu_tensor.to("cpu", non_blocking=True)
113 return LogprobsTensors(
114 self.logprob_token_ids.to("cpu", non_blocking=True),
115 self.logprobs.to("cpu", non_blocking=True),
116 self.selected_token_ranks.to("cpu", non_blocking=True),
117 self.cu_num_generated_tokens,
118 cu_tensor,
119 )
120
121 def filter(self, mask: torch.Tensor) -> "LogprobsTensors":
122 """Filter the logprobs tensors with the given bool mask."""
123 assert self.cu_num_generated_tokens is None, (
124 "filter can't be used with cu_num_generated_tokens"
125 )
126 assert self.cu_num_generated_tokens_tensor is None, (
127 "filter can't be used with cu_num_generated_tokens_tensor"
128 )
129 return LogprobsTensors(
130 self.logprob_token_ids[mask],
131 self.logprobs[mask],
132 self.selected_token_ranks[mask],
133 )
134
135 @staticmethod
136 def cat(
137 tensors: Sequence["LogprobsTensors"],
138 cu_num_generated_tokens: list[int] | None = None,
139 ) -> "LogprobsTensors":
140 """Concatenate flattened logprob tensors."""
141 assert tensors
142 assert cu_num_generated_tokens is not None or all(
143 tensor.cu_num_generated_tokens is None for tensor in tensors
144 )
145 if len(tensors) == 1:
146 tensor = tensors[0]
147 if cu_num_generated_tokens is None:
148 return tensor
149 return tensor._replace(cu_num_generated_tokens=cu_num_generated_tokens)
150 # The multi-chunk path rebuilds boundaries from the CPU layout, which
151 # the device-only boundaries of adaptive verification never use.
152 assert all(tensor.cu_num_generated_tokens_tensor is None for tensor in tensors)
153 return LogprobsTensors(
154 logprob_token_ids=torch.cat(
155 [tensor.logprob_token_ids for tensor in tensors]
156 ),
157 logprobs=torch.cat([tensor.logprobs for tensor in tensors]),
158 selected_token_ranks=torch.cat(
159 [tensor.selected_token_ranks for tensor in tensors]
160 ),
161 cu_num_generated_tokens=cu_num_generated_tokens,
162 )
163
164 @staticmethod
165 def empty_cpu(
166 num_positions: int, num_tokens_per_position: int
167 ) -> "LogprobsTensors":
168 """Create empty LogprobsTensors on CPU."""
169
170 logprob_token_ids = torch.empty(
171 (num_positions, num_tokens_per_position),
172 dtype=torch.int32,
173 device="cpu",
174 pin_memory=PIN_MEMORY,
175 )
176 logprobs = logprob_token_ids.new_empty(
177 (num_positions, num_tokens_per_position),
178 dtype=torch.float32,
179 pin_memory=PIN_MEMORY,
180 )
181 selected_token_ranks = logprob_token_ids.new_empty(
182 num_positions, pin_memory=PIN_MEMORY
183 )
184 return LogprobsTensors(
185 logprob_token_ids=logprob_token_ids,
186 logprobs=logprobs,
187 selected_token_ranks=selected_token_ranks,
188 )

191class RoutedExpertsTensors(NamedTuple):
192 """Device-side snapshot of routed experts data, pending async D2H.
193
194 Produced by :class:`GPUModelRunner` at the end of each async-scheduled
195 step. The copy stream waits on the default stream, then issues
196 non-blocking D2H via :meth:`to_cpu_nonblocking` into a pinned CPU
197 buffer; :class:`AsyncGPUModelRunnerOutput.get_output` synchronizes
198 the copy before the scheduler reads it.
199
200 Sliced to ``total_num_scheduled_tokens`` (step-level, across all
201 requests — NOT per-request). Both ``routing_data`` and
202 ``slot_mapping`` must be private clones when sourced from shared
203 capturer / prepare-input buffers, so the next forward pass /
204 ``_prepare_inputs`` on the default stream does not race with a
205 D2H still pending on the copy stream.
206 """
207
208 # (num_scheduled_tokens, num_layers, num_experts_per_tok)
209 routing_data: torch.Tensor
210 # (num_scheduled_tokens,)
211 slot_mapping: torch.Tensor
212
213 def to_cpu_nonblocking(self) -> "RoutedExpertsTensors":
214 """Issue non-blocking D2H on the current stream.
215
216 NOTE: ``non_blocking=True`` only delivers true overlap when the
217 CPU target is pinned. The current fallback here allocates a
218 new pageable CPU tensor per call, which silently degrades to a
219 synchronous copy; acceptable because the sync happens on the
220 dedicated copy stream, not the default stream.
221 """
222 if self.routing_data.device.type == "cpu":
223 return self
224 return RoutedExpertsTensors(
225 self.routing_data.to("cpu", non_blocking=True),
226 self.slot_mapping.to("cpu", non_blocking=True),
227 )
228
229 def tolists(self) -> "RoutedExpertsLists":
230 """Convert to the numpy-backed form consumed by the scheduler.
231
232 ``.cpu()`` is a no-op when the tensor is already on CPU, so this
233 is cheap for the post-D2H case; for raw device tensors it will
234 synchronously block, which is only reached in tests.
235 """
236 return RoutedExpertsLists(
237 self.routing_data.cpu().numpy(),
238 self.slot_mapping.cpu().numpy(),
239 )

437def make_empty_encoder_model_runner_output(
438 scheduler_output: "SchedulerOutput",
439) -> ModelRunnerOutput:
440 """
441 Create a ModelRunnerOutput stub that contains the correct
442 per-request bookkeeping but no generated data yet.
443 """
444 if not scheduler_output.num_scheduled_tokens:
445 return EMPTY_MODEL_RUNNER_OUTPUT
446
447 # Convert to list so we get a deterministic, indexable sequence
448 req_ids: list[str] = list(scheduler_output.num_scheduled_tokens.keys())
449
450 # Give every request its own contiguous index
451 req_id_to_index: dict[str, int] = {rid: idx for idx, rid in enumerate(req_ids)}
452
453 # An encoder instance never samples, so it emits no tokens at all. The
454 # scheduler finishes these requests once their prompt is fully encoded
455 # (see `Scheduler.update_from_output`).
456 sampled_token_ids: list[list[int]] = [[] for _ in req_ids]
457
458 # Pooler outputs are not available yet ⇒ use None placeholders
459 pooler_output: list[torch.Tensor | None] = [None for _ in req_ids]
460
461 return ModelRunnerOutput(
462 req_ids=req_ids,
463 req_id_to_index=req_id_to_index,
464 sampled_token_ids=sampled_token_ids,
465 pooler_output=pooler_output,
466 )
467

469EMPTY_MODEL_RUNNER_OUTPUT = ModelRunnerOutput(req_ids=[], req_id_to_index={})
