39GREEDY_TEMPERATURE: tl.constexpr = 0

41# step. This value is chosen to be large enough to handle typical use cases.

45class RejectionSampler(nn.Module):
46 """
47 The implementation strictly follows the algorithm described in
48 https://arxiv.org/abs/2211.17192.
49 However, we want to clarify the terminology used in the implementation:
50 accepted tokens: tokens that are accepted based on the relationship
51 between the "raw" draft and target probabilities.
52 recovered tokens: tokens that are sampled based on the adjusted probability
53 distribution, which is derived from both the draft and target
54 probabilities.
55 bonus tokens:
56 If all proposed tokens are accepted, the bonus token is added to the
57 end of the sequence. The bonus token is only sampled from the target
58 probabilities. We pass in the bonus tokens instead of sampling them
59 in the rejection sampler to allow for more flexibility in the
60 sampling process. For example, we can use top_p, top_k sampling for
61 bonus tokens, while spec decode does not support these sampling
62 strategies.
63 output tokens:
64 Tokens are finally generated with the rejection sampler.
65 output tokens = accepted tokens + recovered tokens + bonus tokens
66 """

107 def forward(
108 self,
109 metadata: SpecDecodeMetadata,
110 # [num_tokens, vocab_size]
111 draft_probs: torch.Tensor | None,
112 # [num_tokens + batch_size, vocab_size]
113 logits: torch.Tensor,
114 sampling_metadata: SamplingMetadata,
115 ) -> SamplerOutput:
116 """
117 Args:
118 metadata:
119 Metadata for spec decoding.
120 draft_probs (Optional[torch.Tensor]):
121 Probability distribution for the draft tokens. Shape is
122 [num_tokens, vocab_size]. Can be None if probabilities are
123 not provided, which is the case for ngram spec decode.
124 logits (torch.Tensor):
125 Target model's logits probability distribution.
126 Shape is [num_tokens + batch_size, vocab_size]. Here,
127 probabilities from different requests are flattened into a
128 single tensor because this is the shape of the output logits.
129 NOTE: `logits` can be updated in place to save memory.
130 sampling_metadata (vllm.v1.sample.metadata.SamplingMetadata):
131 Additional metadata needed for sampling, such as temperature,
132 top-k/top-p parameters, or other relevant information.
133 Returns:
134 SamplerOutput:
135 Contains the final output token IDs and their logprobs if
136 requested.
137 """
138 assert metadata.max_spec_len <= MAX_SPEC_LEN
139
140 bonus_logits_indices = metadata.bonus_logits_indices
141 target_logits_indices = metadata.target_logits_indices
142
143 # When indexing with a tensor (bonus_logits_indices), PyTorch
144 # creates a new tensor with separate storage from the original
145 # logits tensor. This means any in-place operations on bonus_logits
146 # won't affect the original logits tensor.
147 assert logits is not None
148 bonus_logits = logits[bonus_logits_indices]
149 bonus_sampler_output = self.sampler(
150 logits=bonus_logits,
151 sampling_metadata=replace(
152 sampling_metadata,
153 max_num_logprobs=-1,
154 ),
155 predict_bonus_token=True,
156 # Override the logprobs mode to return logits because they are
157 # needed later to compute the accepted token logprobs.
158 logprobs_mode_override="processed_logits"
159 if self.is_processed_logprobs_mode
160 else "raw_logits",
161 )
162 bonus_token_ids = bonus_sampler_output.sampled_token_ids
163
164 # Just like `bonus_logits`, `target_logits` is a new tensor with
165 # separate storage from the original `logits` tensor. Therefore,
166 # it is safe to update `target_logits` in place.
167 raw_target_logits = logits[target_logits_indices]
168 # Use float32 for the target_logits.
169 raw_target_logits = raw_target_logits.to(torch.float32)
170 target_logits = raw_target_logits
171 if not self.is_processed_logprobs_mode:
172 # Clone raw_target_logits before applying processors to preserve
173 # the original raw logits for logprobs computation, since
174 # apply_logits_processors modifies the tensor in-place.
175 target_logits = target_logits.clone()
176 target_logits = self.apply_logits_processors(
177 target_logits, sampling_metadata, metadata
178 )
179 # [num_tokens, vocab_size]
180 # NOTE(woosuk): `target_logits` can be updated in place inside the
181 # `apply_sampling_constraints` function.
182 target_logits = apply_sampling_constraints(
183 target_logits,
184 metadata.cu_num_draft_tokens,
185 sampling_metadata,
186 )
187
188 output_token_ids = rejection_sample(
189 metadata.draft_token_ids,
190 metadata.num_draft_tokens,
191 metadata.max_spec_len,
192 metadata.cu_num_draft_tokens,
193 draft_probs,
194 target_logits,
195 bonus_token_ids,
196 sampling_metadata,
197 synthetic_mode=self.synthetic_mode,
198 synthetic_conditional_rates=self.synthetic_conditional_rates,
199 use_fp64_gumbel=self.use_fp64_gumbel,
200 )
201
202 logprobs_tensors = None
203 if sampling_metadata.max_num_logprobs is not None:
204 logprobs_tensors = self._get_logprobs_tensors(
205 sampling_metadata.max_num_logprobs,
206 metadata,
207 logits,
208 target_logits if self.is_processed_logprobs_mode else raw_target_logits,
209 bonus_sampler_output.logprobs_tensors.logprobs,
210 output_token_ids,
211 )
212
213 return SamplerOutput(
214 sampled_token_ids=output_token_ids,
215 logprobs_tensors=logprobs_tensors,
216 )

218 def _get_logprobs_tensors(
219 self,
220 max_num_logprobs: int,
221 metadata: SpecDecodeMetadata,
222 logits: torch.Tensor,
223 target_logits: torch.Tensor,
224 bonus_logits: torch.Tensor,
225 sampled_token_ids: torch.Tensor,
226 ) -> LogprobsTensors:
227 cu_num_sampled_tokens = torch.zeros_like(metadata.cu_num_sampled_tokens)
228 cu_num_sampled_tokens[1:] = metadata.cu_num_sampled_tokens[:-1]
229
230 # Collect target and bonus logits.
231 bonus_logits_indices = metadata.bonus_logits_indices
232 target_logits_indices = metadata.target_logits_indices
233 final_logits = torch.zeros_like(logits, dtype=torch.float32)
234 final_logits[target_logits_indices] = target_logits.to(torch.float32)
235 final_logits[bonus_logits_indices] = bonus_logits.to(torch.float32)
236
237 # NOTE: To avoid cpu-gpu synchronization, we now simply compute indices for
238 # all draft tokens, including the rejected ones. The rejected tokens will
239 # be filtered out in the `parse_output`.
240 logit_start_indices = cu_num_sampled_tokens
241 offsets = torch.arange(
242 sampled_token_ids.shape[-1],
243 device=logit_start_indices.device,
244 dtype=logit_start_indices.dtype,
245 )
246 accepted_logit_indices = (
247 logit_start_indices.unsqueeze(1) + offsets.unsqueeze(0)
248 ).flatten()
249 accepted_logit_indices.clamp_(max=final_logits.shape[0] - 1)
250 accepted_tokens = sampled_token_ids.clone().flatten()
251 # we replace rejected token ids with 0 to avoid gather_logprobs error
252 accepted_tokens[accepted_tokens == PLACEHOLDER_TOKEN_ID] = 0
253
254 # Compute logprobs for accepted tokens.
255 accepted_logits = final_logits[accepted_logit_indices]
256 accepted_logprobs = (
257 accepted_logits
258 if self.is_logits_logprobs_mode
259 else self.sampler.compute_logprobs(accepted_logits)
260 )
261 return self.sampler.gather_logprobs(
262 accepted_logprobs,
263 max_num_logprobs,
264 accepted_tokens.to(torch.int64),
265 )

267 @staticmethod
268 def parse_output(
269 output_token_ids: torch.Tensor,
270 vocab_size: int,
271 discard_req_indices: Sequence[int] = (),
272 logprobs_tensors: LogprobsTensors | None = None,
273 ) -> tuple[list[list[int]], LogprobsLists | None]:
274 """Parse the output of the rejection sampler.
275 Args:
276 output_token_ids: The sampled token IDs in shape
277 [batch_size, max_spec_len + 1]. The rejected tokens are
278 replaced with `PLACEHOLDER_TOKEN_ID` by the rejection sampler
279 and will be filtered out in this function.
280 vocab_size: The size of the vocabulary.
281 discard_req_indices: Optional row indices to discard tokens in.
282 logprobs_tensors: Optional logprobs tensors to filter.
283 Returns:
284 A list of lists of token IDs.
285 """
286 output_token_ids_np = output_token_ids.cpu().numpy()
287 # Create mask for valid tokens.
288 valid_mask = (output_token_ids_np != PLACEHOLDER_TOKEN_ID) & (
289 output_token_ids_np < vocab_size
290 )
291 output_logprobs = None
292 if logprobs_tensors is not None:
293 cu_num_tokens = [0] + valid_mask.sum(axis=1).cumsum().tolist()
294 filtered_tensors = logprobs_tensors.filter(valid_mask.flatten())
295 output_logprobs = filtered_tensors.tolists(cu_num_tokens)
296
297 if len(discard_req_indices) > 0:
298 valid_mask[discard_req_indices] = False
299 outputs = [
300 row[valid_mask[i]].tolist() for i, row in enumerate(output_token_ids_np)
301 ]
302 return outputs, output_logprobs

304 def apply_logits_processors(
305 self,
306 logits: torch.Tensor,
307 sampling_metadata: SamplingMetadata,
308 metadata: SpecDecodeMetadata,
309 ) -> torch.Tensor:
310 has_penalties = not sampling_metadata.no_penalties
311 any_penalties_or_bad_words = (
312 sampling_metadata.bad_words_token_ids or has_penalties
313 )
314 output_token_ids = sampling_metadata.output_token_ids
315 if any_penalties_or_bad_words:
316 output_token_ids = self._combine_outputs_with_spec_tokens(
317 output_token_ids,
318 sampling_metadata.spec_token_ids,
319 )
320
321 # Calculate indices of target logits.
322 repeat_indices: torch.Tensor | None = None
323 need_repeat_indices = (
324 sampling_metadata.allowed_token_ids_mask is not None or has_penalties
325 )
326 if need_repeat_indices:
327 num_requests = len(metadata.num_draft_tokens)
328 num_draft_tokens = torch.tensor(metadata.num_draft_tokens, device="cpu")
329 original_indices = torch.arange(num_requests, device="cpu")
330 repeat_indices_cpu = original_indices.repeat_interleave(num_draft_tokens)
331 repeat_indices = async_tensor_h2d(repeat_indices_cpu, logits.device)
332 logits = self.apply_penalties(
333 logits, sampling_metadata, metadata, repeat_indices, output_token_ids
334 )
335
336 # Apply allowed token ids.
337 if sampling_metadata.allowed_token_ids_mask is not None:
338 token_mask = sampling_metadata.allowed_token_ids_mask[repeat_indices]
339 logits.masked_fill_(token_mask, float("-inf"))
340
341 # Apply bad words exclusion.
342 if bad_words_token_ids := sampling_metadata.bad_words_token_ids:
343 apply_bad_words_with_drafts(
344 logits, bad_words_token_ids, output_token_ids, metadata.num_draft_tokens
345 )
346
347 for processor in sampling_metadata.logitsprocs.non_argmax_invariant:
348 if isinstance(processor, MinTokensLogitsProcessor):
349 logits = processor.apply_with_spec_decode(
350 logits, metadata.num_draft_tokens
351 )
352 holder = sampling_metadata.thinking_budget_state_holder
353 if holder is not None and holder.has_tracked_requests():
354 logits = holder.apply_to_logits(
355 logits,
356 predict_bonus_token=False,
357 spec_token_ids=sampling_metadata.spec_token_ids,
358 )
359 return logits
360
361 @staticmethod
362 def apply_penalties(
363 logits: torch.Tensor,
364 sampling_metadata: SamplingMetadata,
365 metadata: SpecDecodeMetadata,
366 repeat_indices: torch.Tensor,
367 output_token_ids: list[list[int]],
368 ) -> torch.Tensor:
369 if sampling_metadata.no_penalties:
370 return logits
371
372 assert sampling_metadata.prompt_token_ids is not None
373
374 prompt_token_ids = sampling_metadata.prompt_token_ids[repeat_indices]
375 presence_penalties = sampling_metadata.presence_penalties[repeat_indices]
376 frequency_penalties = sampling_metadata.frequency_penalties[repeat_indices]
377 repetition_penalties = sampling_metadata.repetition_penalties[repeat_indices]
378
379 logits = apply_all_penalties(
380 logits,
381 prompt_token_ids,
382 presence_penalties,
383 frequency_penalties,
384 repetition_penalties,
385 output_token_ids,
386 )
387 return logits
388
389 @staticmethod
390 def _combine_outputs_with_spec_tokens(
391 output_token_ids: list[list[int]],
392 spec_token_ids: list[list[int]] | None = None,
393 ) -> list[list[int]]:
394 if spec_token_ids is None:
395 return output_token_ids
396
397 result = []
398 for out, spec in zip(output_token_ids, spec_token_ids):
399 if len(spec) == 0:
400 continue
401 result.append(out)
402 for i in range(len(spec) - 1):
403 result.append([*result[-1], spec[i]])
404 return result

407def rejection_sample(
408 # [num_tokens]
409 draft_token_ids: torch.Tensor,
410 # [batch_size]
411 num_draft_tokens: list[int],
412 max_spec_len: int,
413 # [batch_size]
414 cu_num_draft_tokens: torch.Tensor,
415 # [num_tokens, vocab_size]
416 draft_probs: torch.Tensor | None,
417 # [num_tokens, vocab_size]
418 target_logits: torch.Tensor,
419 # [batch_size, 1]
420 bonus_token_ids: torch.Tensor,
421 sampling_metadata: SamplingMetadata,
422 synthetic_mode: bool = False,
423 synthetic_conditional_rates: torch.Tensor | None = None,
424 use_fp64_gumbel: bool = False,
425) -> torch.Tensor:
426 assert draft_token_ids.ndim == 1
427 assert draft_probs is None or draft_probs.ndim == 2
428 assert cu_num_draft_tokens.ndim == 1
429 assert target_logits.ndim == 2
430
431 batch_size = len(num_draft_tokens)
432 num_tokens = draft_token_ids.shape[0]
433 vocab_size = target_logits.shape[-1]
434 device = target_logits.device
435 assert draft_token_ids.is_contiguous()
436 assert draft_probs is None or draft_probs.is_contiguous()
437 assert bonus_token_ids.is_contiguous()
438 assert target_logits.shape == (num_tokens, vocab_size)
439
440 # Create output buffer.
441 output_token_ids = torch.full(
442 (batch_size, max_spec_len + 1),
443 PLACEHOLDER_TOKEN_ID,
444 dtype=torch.int32, # Consistent with SamplerOutput.sampled_token_ids.
445 device=device,
446 )
447
448 if sampling_metadata.all_greedy:
449 is_greedy = None
450 else:
451 is_greedy = sampling_metadata.temperature == GREEDY_TEMPERATURE
452
453 # Generate uniform probabilities before either kernel because synthetic
454 # mode needs them in the greedy kernel too. Skip only when all requests
455 # are greedy *and* synthetic mode is off (the standard fast-path).
456 # [num_tokens]
457 uniform_probs: torch.Tensor | None = None
458 if synthetic_mode or not sampling_metadata.all_greedy:
459 uniform_probs = generate_uniform_probs(
460 num_tokens,
461 num_draft_tokens,
462 sampling_metadata.generators,
463 device,
464 )
465
466 if not sampling_metadata.all_random:
467 # Rejection sampling for greedy sampling requests.
468 target_argmax = target_logits.argmax(dim=-1)
469 _rejection_greedy_sample(
470 output_token_ids,
471 cu_num_draft_tokens,
472 draft_token_ids,
473 target_argmax,
474 bonus_token_ids,
475 is_greedy,
476 max_spec_len,
477 uniform_probs,
478 synthetic_conditional_rates,
479 )
480 if sampling_metadata.all_greedy:
481 return output_token_ids
482
483 # Compute probability distribution from target logits.
484 target_probs = target_logits.softmax(dim=-1, dtype=torch.float32)
485 assert target_probs.is_contiguous()
486
487 # Sample recovered tokens for each position.
488 # [num_tokens]
489 recovered_token_ids = sample_recovered_tokens(
490 max_spec_len,
491 num_draft_tokens,
492 cu_num_draft_tokens,
493 draft_token_ids,
494 draft_probs,
495 target_probs,
496 sampling_metadata,
497 device,
498 use_fp64_gumbel,
499 )
500
501 # Rejection sampling for random sampling requests.
502 assert uniform_probs is not None
503 _rejection_random_sample(
504 output_token_ids,
505 cu_num_draft_tokens,
506 draft_token_ids,
507 draft_probs,
508 target_probs,
509 bonus_token_ids,
510 recovered_token_ids,
511 uniform_probs,
512 is_greedy,
513 max_spec_len,
514 synthetic_conditional_rates,
515 )
516 return output_token_ids

519def apply_sampling_constraints(
520 logits: torch.Tensor, # [num_tokens, vocab_size]
521 cu_num_draft_tokens: torch.Tensor, # [batch_size]
522 sampling_metadata: SamplingMetadata,
523) -> torch.Tensor:
524 """Process logits based on sampling metadata.
525
526 This function applies temperature scaling to the logits,
527 as well as top-k and top-p. For greedy decoding, it returns
528 the original logits.
529
530 Args:
531 logits: Input logits tensor to be processed.
532 cu_num_draft_tokens: Cumulative number of draft tokens.
533 sampling_metadata: Metadata containing sampling parameters such as
534 temperature and whether greedy sampling is used.
535
536 Returns:
537 torch.Tensor: Processed logits if non-greedy sampling is used,
538 otherwise returns the original logits.
539 """
540 assert logits.ndim == 2
541 assert cu_num_draft_tokens.ndim == 1
542 if sampling_metadata.all_greedy:
543 return logits
544
545 num_tokens = logits.shape[0]
546 temperature = expand_batch_to_tokens(
547 sampling_metadata.temperature,
548 cu_num_draft_tokens,
549 num_tokens,
550 replace_from=GREEDY_TEMPERATURE,
551 replace_to=1,
552 )
553 # NOTE(woosuk): Update `logits` in place to avoid allocating a new tensor.
554 logits.div_(temperature.unsqueeze(-1))
555
556 # Get expanded top_k and top_p tensors.
557 top_k = None
558 if sampling_metadata.top_k is not None:
559 top_k = expand_batch_to_tokens(
560 sampling_metadata.top_k,
561 cu_num_draft_tokens,
562 num_tokens,
563 )
564 top_p = None
565 if sampling_metadata.top_p is not None:
566 top_p = expand_batch_to_tokens(
567 sampling_metadata.top_p,
568 cu_num_draft_tokens,
569 num_tokens,
570 )
571
572 # NOTE(woosuk): `apply_top_k_top_p` uses sorting to calculate the mask,
573 # which is slow for large vocab sizes. This may cause performance issues.
574 return apply_top_k_top_p(logits, top_k, top_p)

577def expand_batch_to_tokens(
578 x: torch.Tensor, # [batch_size]
579 cu_num_tokens: torch.Tensor, # [batch_size]
580 num_tokens: int,
581 replace_from: int = 0,
582 replace_to: int = 0,
583) -> torch.Tensor:
584 """Expand [batch_size] tensor to [num_tokens] tensor based on the number of
585 tokens per batch in cu_num_tokens.
586
587 For example, if x = [a, b, c] and cu_num_tokens = [2, 5, 6], then
588 num_tokens = 6, and expanded_x = [a, a, b, b, b, c].
589
590 Args:
591 x: [batch_size] tensor to expand.
592 cu_num_tokens: [batch_size] tensor containing the cumulative number of
593 tokens per batch. Each element represents the total number of
594 tokens up to and including that batch.
595 num_tokens: Total number of tokens.
596 replace_from: int = 0
597 Value to be replaced if it is found in x.
598 replace_to: int = 0
599 Value to replace with when replace_from is found.
600 Returns:
601 expanded_x: [num_tokens] tensor.
602 """
603 batch_size = x.shape[0]
604 assert cu_num_tokens.shape[0] == batch_size
605 expanded_x = x.new_empty(num_tokens)
606 _expand(
607 expanded_x,
608 x,
609 cu_num_tokens,
610 replace_from,
611 replace_to,
612 )
613 return expanded_x

616def generate_uniform_probs(
617 num_tokens: int,
618 num_draft_tokens: list[int],
619 generators: dict[int, torch.Generator],
620 device: torch.device,
621) -> torch.Tensor:
622 """
623 Generates a batch of uniform random samples, with optional seeding
624 if available.
625
626 This method creates a tensor of shape `(num_tokens, )` filled
627 with uniform random values in the range [0, 1). If `generators` is provided,
628 the requests with their own seeds will use the provided `torch.Generator`
629 for reproducibility. The samples for the other requests will be generated
630 without a seed.
631
632 Args:
633 num_tokens: int
634 Total number of tokens.
635 num_draft_tokens: List[List[int]]
636 Number of draft tokens per request.
637 generators: Optional[Dict[int, torch.Generator]]
638 A dictionary mapping indices in the batch to
639 `torch.Generator` objects.
640 device: torch.device
641 The device on which to allocate the tensor.
642 Returns:
643 uniform_rand: torch.Tensor
644 A tensor of shape `(num_tokens, )` containing uniform
645 random values in the range [0, 1).
646 """
647 # NOTE(woosuk): We deliberately use float64 instead of float32 here
648 # because when using float32, there's a non-negligible chance that
649 # uniform_prob is sampled to be exact 0.0 as reported in
650 # https://github.com/pytorch/pytorch/issues/16706. Using float64
651 # mitigates the issue.
652 uniform_probs = torch.rand(
653 (num_tokens,),
654 dtype=torch.float64,
655 device=device,
656 )
657 start_idx = 0
658 for req_idx, n in enumerate(num_draft_tokens):
659 # Do not generate random numbers for requests with no draft tokens.
660 # This can be important for reproducibility.
661 if n == 0:
662 continue
663 end_idx = start_idx + n
664 generator = generators.get(req_idx)
665 if generator is not None:
666 uniform_probs[start_idx:end_idx].uniform_(generator=generator)
667 start_idx = end_idx
668 return uniform_probs

671def sample_recovered_tokens(
672 max_spec_len: int,
673 num_draft_tokens: list[int],
674 # [batch_size]
675 cu_num_draft_tokens: torch.Tensor,
676 # [num_tokens]
677 draft_token_ids: torch.Tensor,
678 # [num_tokens, vocab_size]
679 draft_probs: torch.Tensor | None,
680 # [num_tokens, vocab_size]
681 target_probs: torch.Tensor,
682 sampling_metadata: SamplingMetadata,
683 device: torch.device,
684 use_fp64_gumbel: bool = False,
685) -> torch.Tensor:
686 # NOTE(woosuk): Create only one distribution for each request.
687 batch_size = len(num_draft_tokens)
688 vocab_size = target_probs.shape[-1]
689 q_dtype = torch.float64 if use_fp64_gumbel else torch.float32
690 q = torch.empty(
691 (batch_size, vocab_size),
692 dtype=q_dtype,
693 device=device,
694 )
695 q.exponential_()
696 for i, generator in sampling_metadata.generators.items():
697 # Do not generate random numbers for requests with no draft tokens.
698 # This can be important for reproducibility.
699 if num_draft_tokens[i] > 0:
700 q[i].exponential_(generator=generator)
701
702 inv_q = q.reciprocal()
703
704 recovered_token_ids = torch.empty_like(draft_token_ids)
705 _sample_recovered_tokens(
706 recovered_token_ids,
707 cu_num_draft_tokens,
708 draft_token_ids,
709 draft_probs,
710 target_probs,
711 inv_q,
712 max_spec_len=max_spec_len,
713 )
714 return recovered_token_ids

719def rejection_greedy_sample_kernel(
720 output_token_ids_ptr, # [batch_size, max_spec_len + 1]
721 cu_num_draft_tokens_ptr, # [batch_size]
722 draft_token_ids_ptr, # [num_tokens]
723 target_argmax_ptr, # [num_tokens]
724 bonus_token_ids_ptr, # [batch_size]
725 is_greedy_ptr, # [batch_size] or None
726 max_spec_len,
727 uniform_probs_ptr, # [num_tokens] or None (synthetic mode only)
728 synthetic_conditional_rates_ptr, # [num_speculative_tokens] or None
729 SYNTHETIC_MODE: tl.constexpr,
730):
731 req_idx = tl.program_id(0)
732 # FIXME(woosuk): Because is_greedy_ptr is not None at profiling run,
733 # re-compilation may happen during runtime when is_greedy_ptr is None.
734 is_greedy = True if is_greedy_ptr is None else tl.load(is_greedy_ptr + req_idx)
735 if not is_greedy:
736 # Early exit for non-greedy sampling requests.
737 return
738
739 start_idx = (
740 tl.zeros([], dtype=cu_num_draft_tokens_ptr.dtype.element_ty)
741 if req_idx == 0
742 else tl.load(cu_num_draft_tokens_ptr + req_idx - 1)
743 )
744 end_idx = tl.load(cu_num_draft_tokens_ptr + req_idx)
745 num_draft_tokens = end_idx - start_idx
746
747 rejected = False
748 for pos in range(num_draft_tokens):
749 if not rejected:
750 draft_token_id = tl.load(draft_token_ids_ptr + start_idx + pos)
751 target_argmax_id = tl.load(target_argmax_ptr + start_idx + pos).to(tl.int32)
752 if SYNTHETIC_MODE:
753 uniform_prob = tl.load(uniform_probs_ptr + start_idx + pos)
754 rate = tl.load(synthetic_conditional_rates_ptr + pos)
755 # -1 is used for padded draft token ids that should be rejected.
756 accepted = (uniform_prob < rate) and draft_token_id >= 0
757 token_id = draft_token_id if accepted else target_argmax_id
758 rejected = not accepted
759 else:
760 token_id = target_argmax_id
761 rejected = draft_token_id != target_argmax_id
762 tl.store(
763 output_token_ids_ptr + req_idx * (max_spec_len + 1) + pos,
764 token_id,
765 )
766
767 if not rejected:
768 # If all tokens are accepted, append the bonus token.
769 bonus_token_id = tl.load(bonus_token_ids_ptr + req_idx)
770 tl.store(
771 output_token_ids_ptr + req_idx * (max_spec_len + 1) + num_draft_tokens,
772 bonus_token_id,
773 )

821@triton.jit(do_not_specialize=["max_spec_len"])
822def rejection_random_sample_kernel(
823 output_token_ids_ptr, # [batch_size, max_spec_len + 1]
824 cu_num_draft_tokens_ptr, # [batch_size]
825 draft_token_ids_ptr, # [num_tokens]
826 draft_probs_ptr, # [num_tokens, vocab_size] or None
827 target_probs_ptr, # [num_tokens, vocab_size]
828 bonus_token_ids_ptr, # [batch_size]
829 recovered_token_ids_ptr, # [num_tokens]
830 uniform_probs_ptr, # [num_tokens]
831 is_greedy_ptr, # [batch_size]
832 max_spec_len,
833 vocab_size,
834 synthetic_conditional_rates_ptr, # [num_speculative_tokens] or None
835 NO_DRAFT_PROBS: tl.constexpr,
836 SYNTHETIC_MODE: tl.constexpr,
837):
838 req_idx = tl.program_id(0)
839 is_greedy = tl.load(is_greedy_ptr + req_idx)
840 if is_greedy:
841 # Early exit for greedy sampling requests.
842 return
843
844 start_idx = (
845 tl.zeros([], dtype=cu_num_draft_tokens_ptr.dtype.element_ty)
846 if req_idx == 0
847 else tl.load(cu_num_draft_tokens_ptr + req_idx - 1)
848 )
849 end_idx = tl.load(cu_num_draft_tokens_ptr + req_idx)
850 num_draft_tokens = end_idx - start_idx
851
852 rejected = False
853 for pos in range(num_draft_tokens):
854 if not rejected:
855 draft_token_id = tl.load(draft_token_ids_ptr + start_idx + pos)
856 uniform_prob = tl.load(uniform_probs_ptr + start_idx + pos)
857 if draft_token_id < 0:
858 # -1 is used for padded draft token ids that should be rejected.
859 accepted = False
860 elif SYNTHETIC_MODE:
861 rate = tl.load(synthetic_conditional_rates_ptr + pos)
862 accepted = uniform_prob < rate
863 else:
864 if NO_DRAFT_PROBS:
865 draft_prob = 1
866 else:
867 draft_prob = tl.load(
868 draft_probs_ptr
869 + (start_idx + pos) * vocab_size
870 + draft_token_id
871 )
872 target_prob = tl.load(
873 target_probs_ptr + (start_idx + pos) * vocab_size + draft_token_id
874 )
875 # NOTE(woosuk): While the draft probability should never be 0,
876 # we check it to avoid NaNs. If it happens to be 0, we reject.
877 accepted = draft_prob > 0 and target_prob / draft_prob >= uniform_prob
878 if accepted:
879 token_id = draft_token_id
880 else:
881 rejected = True
882 token_id = tl.load(recovered_token_ids_ptr + start_idx + pos)
883 tl.store(
884 output_token_ids_ptr + req_idx * (max_spec_len + 1) + pos, token_id
885 )
886
887 if not rejected:
888 # If all tokens are accepted, append the bonus token.
889 bonus_token_id = tl.load(bonus_token_ids_ptr + req_idx)
890 tl.store(
891 output_token_ids_ptr + req_idx * (max_spec_len + 1) + num_draft_tokens,
892 bonus_token_id,
893 )

996@triton.jit
997def sample_recovered_tokens_kernel(
998 output_token_ids_ptr, # [num_tokens]
999 cu_num_draft_tokens_ptr, # [batch_size]
1000 draft_token_ids_ptr, # [num_tokens]
1001 draft_probs_ptr, # [num_tokens, vocab_size] or None
1002 target_probs_ptr, # [num_tokens, vocab_size]
1003 inv_q_ptr, # [batch_size, vocab_size]
1004 vocab_size,
1005 BLOCK_SIZE: tl.constexpr,
1006 NO_DRAFT_PROBS: tl.constexpr,
1007 USE_FP64_GUMBEL: tl.constexpr,
1008):
1009 req_idx = tl.program_id(0)
1010 start_idx = (
1011 tl.zeros([], dtype=cu_num_draft_tokens_ptr.dtype.element_ty)
1012 if req_idx == 0
1013 else tl.load(cu_num_draft_tokens_ptr + req_idx - 1)
1014 )
1015 end_idx = tl.load(cu_num_draft_tokens_ptr + req_idx)
1016 num_draft_tokens = end_idx - start_idx
1017
1018 # Early exit for out-of-range positions.
1019 pos = tl.program_id(1)
1020 if pos >= num_draft_tokens:
1021 return
1022
1023 token_idx = start_idx + pos
1024
1025 if NO_DRAFT_PROBS:
1026 draft_token_id = tl.load(draft_token_ids_ptr + token_idx)
1027
1028 if USE_FP64_GUMBEL:
1029 max_val = tl.full((), float("-inf"), tl.float64)
1030 else:
1031 max_val = tl.full((), float("-inf"), tl.float32)
1032 recovered_id = 0
1033 for v in range(0, vocab_size, BLOCK_SIZE):
1034 vocab_offset = v + tl.arange(0, BLOCK_SIZE)
1035 vocab_mask = vocab_offset < vocab_size
1036
1037 if NO_DRAFT_PROBS:
1038 prob = tl.load(
1039 target_probs_ptr + token_idx * vocab_size + vocab_offset,
1040 mask=(vocab_mask & (vocab_offset != draft_token_id)),
1041 other=0.0,
1042 )
1043 else:
1044 draft_prob = tl.load(
1045 draft_probs_ptr + token_idx * vocab_size + vocab_offset,
1046 mask=vocab_mask,
1047 other=0.0,
1048 )
1049 target_prob = tl.load(
1050 target_probs_ptr + token_idx * vocab_size + vocab_offset,
1051 mask=vocab_mask,
1052 other=0.0,
1053 )
1054 prob = tl.maximum(target_prob - draft_prob, 0.0)
1055 # NOTE(woosuk): We don't need `prob = prob / tl.sum(prob)` here because
1056 # `tl.argmax` will select the maximum value.
1057
1058 inv_q = tl.load(
1059 inv_q_ptr + req_idx * vocab_size + vocab_offset,
1060 mask=vocab_mask,
1061 other=0.0,
1062 )
1063
1064 # Local tile reduction.
1065 # Mask out-of-vocabulary entries to -inf so they can never win
1066 # the argmax — prevents producing recovered_id >= vocab_size
1067 # when all valid entries in the last tile have zero probability.
1068 score = prob * inv_q
1069 score = tl.where(vocab_mask, score, float("-inf"))
1070 local_max, local_id = tl.max(score, axis=0, return_indices=True)
1071
1072 if local_max > max_val:
1073 max_val = local_max
1074 recovered_id = v + local_id
1075
1076 recovered_id = tl.minimum(recovered_id, vocab_size - 1)
1077 tl.store(output_token_ids_ptr + token_idx, recovered_id)
