44def flashinfer_sampler_supported() -> bool:
45 """Decide whether FlashInfer's top-p/top-k sampler can be used.
46
47 Returns False (with appropriate logging) when ``VLLM_USE_FLASHINFER_SAMPLER``
48 is 0, when the platform isn't CUDA, when the GPU's compute capability is
49 unsupported. Raises ``RuntimeError`` if the user explicitly opted in
50 via the env var but FlashInfer is unavailable.
51
52 Assumes flashinfer is installed, as guaranteed by ``requirements/cuda.txt``;
53 otherwise importing the FlashInfer backend below raises ``ImportError``.
54
55 Note: callers must additionally ensure ``logprobs_mode`` doesn't require
56 post-top-k/top-p logits/logprobs for any request whose logprobs will be
57 returned in this step, since FlashInfer doesn't expose those.
58 """
59 if not current_platform.is_cuda():
60 return False
61 if not envs.VLLM_USE_FLASHINFER_SAMPLER:
62 logger.info_once(
63 "FlashInfer top-p/top-k sampling disabled via "
64 "VLLM_USE_FLASHINFER_SAMPLER=0."
65 )
66 return False
67 from vllm.v1.attention.backends.flashinfer import FlashInferBackend
68
69 capability = current_platform.get_device_capability()
70 assert capability is not None
71 unsupported_reason: str | None = None
72 if not FlashInferBackend.supports_compute_capability(capability):
73 unsupported_reason = (
74 f"unsupported compute capability {capability.as_version_str()}"
75 )
76
77 if unsupported_reason is None:
78 logger.info_once("Using FlashInfer for top-p & top-k sampling.", scope="global")
79 return True
80 if envs.is_set("VLLM_USE_FLASHINFER_SAMPLER"):
81 raise RuntimeError(
82 f"FlashInfer top-p/top-k sampling unavailable: {unsupported_reason}. "
83 "Unset VLLM_USE_FLASHINFER_SAMPLER=1."
84 )
85 logger.warning_once(
86 "FlashInfer top-p/top-k sampling unavailable: %s; falling back. "
87 "Set VLLM_USE_FLASHINFER_SAMPLER=0 to silence.",
88 unsupported_reason,
89 )
90 return False

93class TopKTopPSampler(nn.Module):
94 """
95 Module that performs optional top-k and top-p filtering followed by
96 weighted random sampling of logits.
97
98 Implementations may update the logits tensor in-place.
99 """
100
101 def __init__(
102 self,
103 logprobs_mode: LogprobsMode = "raw_logprobs",
104 use_fp64_gumbel: bool = False,
105 ) -> None:
106 super().__init__()
107 self.logprobs_mode = logprobs_mode
108 self.use_fp64_gumbel = use_fp64_gumbel
109 if current_platform.is_cuda():
110 # FlashInfer doesn't expose post-top-k/top-p logits/logprobs,
111 # so it can't be used when the configured mode requires them.
112 can_use_flashinfer = (
113 logprobs_mode not in PROCESSED_LOGPROBS_MODES
114 and flashinfer_sampler_supported()
115 )
116 self.forward = (
117 self.forward_cuda if can_use_flashinfer else self.forward_native
118 )
119 elif current_platform.is_cpu():
120 arch = current_platform.get_cpu_architecture()
121 # Fall back to native implementation for POWERPC and RISCV.
122 # On PowerPC argmax produces incorrect output with torch.compile.
123 # PR: https://github.com/vllm-project/vllm/pull/26987
124 if arch in (CpuArchEnum.RISCV, CpuArchEnum.POWERPC):
125 self.forward = self.forward_native
126 else:
127 self.forward = self.forward_cpu
128 elif current_platform.is_xpu():
129 if envs.VLLM_XPU_USE_SAMPLER_KERNEL:
130 self.forward = self.forward_xpu
131 else:
132 self.forward = self.forward_native
133 elif (
134 logprobs_mode not in PROCESSED_LOGPROBS_MODES
135 and rocm_aiter_ops.is_enabled()
136 and not _skip_aiter_sampler_on_gfx1250() # TODO (JPVILLAM): Enable
137 ):
138 self.aiter_ops = None
139 self._aiter_ops_import_failed = False
140 logger.info_once(
141 "Using aiter sampler on ROCm (lazy import, sampling-only)."
142 )
143 self.forward = self.forward_hip
144 else:
145 self.forward = self.forward_native
146
147 # Every accelerator backend can fall back to native sampling at runtime.
148 register_top_k_top_p_warmups()

150 def forward_native(
151 self,
152 logits: torch.Tensor,
153 generators: dict[int, torch.Generator],
154 k: torch.Tensor | None,
155 p: torch.Tensor | None,
156 ) -> tuple[torch.Tensor, torch.Tensor | None]:
157 """
158 PyTorch-native implementation of top-k and top-p sampling.
159
160 The logits tensor may be updated in-place.
161 """
162 logits = apply_top_k_top_p(logits, k, p)
163 logits_to_return = None
164 if self.logprobs_mode == "processed_logits":
165 logits_to_return = logits
166 elif self.logprobs_mode == "processed_logprobs":
167 logits_to_return = logits.log_softmax(dim=-1, dtype=torch.float32)
168 probs = logits.softmax(dim=-1, dtype=torch.float32)
169 return (
170 random_sample(probs, generators, self.use_fp64_gumbel),
171 logits_to_return,
172 )

174 def forward_cuda(
175 self,
176 logits: torch.Tensor,
177 generators: dict[int, torch.Generator],
178 k: torch.Tensor | None,
179 p: torch.Tensor | None,
180 ) -> tuple[torch.Tensor, torch.Tensor | None]:
181 """More optimized implementation for top-k and top-p sampling."""
182 # Fall back to the PyTorch-native path when FlashInfer has nothing
183 # to do (no top-k / top-p filter) or when per-request generators
184 # are present (unsupported by FlashInfer 0.2.3+).
185 if (k is None and p is None) or generators:
186 if generators:
187 logger.debug_once(
188 "FlashInfer 0.2.3+ does not support "
189 "per-request generators. Falling back to "
190 "PyTorch-native implementation."
191 )
192 return self.forward_native(logits, generators, k, p)
193 if self.use_fp64_gumbel:
194 return self.forward_native(logits, generators, k, p)
195 assert self.logprobs_mode not in PROCESSED_LOGPROBS_MODES, (
196 "FlashInfer does not support returning logits/logprobs"
197 )
198 # flashinfer sampling functions expect contiguous logits.
199 # In flex_attn/triton_attn fp32 inference, logits can be non-contiguous
200 # because of slicing operation in logits_processor.
201 return flashinfer_sample(logits.contiguous(), k, p, generators), None

203 def forward_cpu(
204 self,
205 logits: torch.Tensor,
206 generators: dict[int, torch.Generator],
207 k: torch.Tensor | None,
208 p: torch.Tensor | None,
209 ) -> tuple[torch.Tensor, torch.Tensor | None]:
210 """
211 PyTorch-native implementation of top-k and top-p sampling for CPU.
212
213 The logits tensor may be updated in-place.
214 """
215 logits = apply_top_k_top_p(logits, k, p)
216 logits_to_return = None
217 if self.logprobs_mode == "processed_logits":
218 logits_to_return = logits
219 elif self.logprobs_mode == "processed_logprobs":
220 logits_to_return = logits.log_softmax(dim=-1, dtype=torch.float32)
221
222 if not generators and not self.use_fp64_gumbel:
223 return compiled_random_sample(logits), logits_to_return
224
225 probs = logits.softmax(dim=-1, dtype=torch.float32)
226 q = empty_exponential_noise_like(probs, self.use_fp64_gumbel)
227 q.exponential_()
228 for i, generator in generators.items():
229 q[i].exponential_(generator=generator)
230
231 return sample_with_exponential_noise(probs, q), logits_to_return

233 def _init_aiter_ops(self) -> bool:
234 if self._aiter_ops_import_failed:
235 return False
236 try:
237 import aiter.ops.sampling # noqa: F401
238 except ImportError:
239 self._aiter_ops_import_failed = True
240 self.forward = self.forward_native
241 logger.warning_once(
242 "aiter.ops.sampling is not available on ROCm. "
243 "Falling back to PyTorch-native implementation."
244 )
245 return False
246 self.aiter_ops = torch.ops.aiter
247 return True

273 def aiter_sample(
274 self,
275 logits: torch.Tensor,
276 k: torch.Tensor | None,
277 p: torch.Tensor | None,
278 generators: dict[int, torch.Generator],
279 ) -> torch.Tensor:
280 """Sample from logits using aiter ops."""
281 assert self.aiter_ops is not None
282 use_top_k = k is not None
283 use_top_p = p is not None
284 # Joint k+p path
285 if use_top_p and use_top_k:
286 probs = logits.softmax(dim=-1, dtype=torch.float32).contiguous()
287 next_token_ids = self.aiter_ops.top_k_top_p_sampling_from_probs(
288 probs,
289 None,
290 *_to_tensor_scalar_tuple(k),
291 *_to_tensor_scalar_tuple(p),
292 deterministic=True,
293 )
294 return next_token_ids.view(-1)
295 # Top-p only path
296 elif use_top_p:
297 probs = logits.softmax(dim=-1, dtype=torch.float32).contiguous()
298 next_token_ids = self.aiter_ops.top_p_sampling_from_probs(
299 probs, None, *_to_tensor_scalar_tuple(p), deterministic=True
300 )
301 return next_token_ids.view(-1)
302 # Top-k only path
303 elif use_top_k:
304 probs = logits.softmax(dim=-1, dtype=torch.float32).contiguous()
305 renorm_probs = self.aiter_ops.top_k_renorm_probs(
306 probs, *_to_tensor_scalar_tuple(k)
307 )
308 return torch.multinomial(renorm_probs, num_samples=1).view(-1)
309 raise RuntimeError("aiter_sample was called with no active top-k or top-p.")

311 def forward_xpu(
312 self,
313 logits: torch.Tensor,
314 generators: dict[int, torch.Generator],
315 k: torch.Tensor | None,
316 p: torch.Tensor | None,
317 ) -> tuple[torch.Tensor, torch.Tensor | None]:
318 if generators:
319 logger.warning_once(
320 "xpu kernel topk_topp_sampler does not support "
321 "per-request generators. Falling back to "
322 "PyTorch-native implementation."
323 )
324 return self.forward_native(logits, generators, k, p)
325 random_sampled = torch.empty(
326 logits.shape[0], dtype=torch.int64, device=logits.device
327 )
328 logits_to_return = None
329 if self.logprobs_mode in PROCESSED_LOGPROBS_MODES:
330 logits_to_return = torch.empty_like(logits)
331
332 assert len(generators) != logits.shape[0], (
333 "xpu kernel topk_topp_sampler does not support batch-wise generators."
334 )
335 generator = torch.xpu.default_generators[logits.device.index]
336
337 state = generator.get_state()
338 seed, offset = state.view(torch.int64)
339 seeds = torch.tensor(
340 [seed, offset], dtype=torch.int64, device=torch.device("cpu")
341 )
342 # The XPU kernel expects k as int64 (Long), but the input batch
343 # stores top_k as int32. Cast here to avoid dtype mismatch.
344 if k is not None:
345 k = k.to(torch.int64)
346 torch.ops.vllm.xpu_topk_topp_sampler(
347 random_sampled, logits_to_return, logits, k, p, self.logprobs_mode, seeds
348 )
349 # The custom XPU sampler kernel consumes RNG values internally, so advance
350 # the default generator's offset to keep future draws deterministic.
351 # pytorch: offset must be multiple of 4
352 offset = (offset + logits.numel() + 3) // 4 * 4
353 state.view(torch.int64)[1] = offset
354 generator.set_state(state)
355 return random_sampled, logits_to_return

358# Note: this is a workaround for
359# https://github.com/pytorch/pytorch/pull/151218
360@torch.compile(dynamic=True, backend=current_platform.simple_compile_backend)
361def compiled_random_sample(logits: torch.Tensor) -> torch.Tensor:
362 probs = logits.softmax(dim=-1, dtype=torch.float32)
363 q = torch.empty_like(probs)
364 q.exponential_()
365 return probs.div(q).argmax(dim=-1).view(-1)

368def apply_top_k_top_p(
369 logits: torch.Tensor, k: torch.Tensor | None, p: torch.Tensor | None
370) -> torch.Tensor:
371 if p is None and k is None:
372 return logits
373
374 if HAS_TRITON:
375 return apply_top_k_top_p_triton(logits, k, p)

377 is_cpu = current_platform.is_cpu()

381def apply_top_k_top_p_pytorch(
382 logits: torch.Tensor,
383 k: torch.Tensor | None,
384 p: torch.Tensor | None,
385 allow_cpu_sync: bool = False,
386) -> torch.Tensor:
387 """Apply top-k and top-p masks to the logits.
388
389 If a top-p is used, this function will sort the logits tensor,
390 which can be slow for large batches.
391
392 The logits tensor may be updated in-place.
393 """
394 if p is None:
395 if k is None:
396 return logits
397
398 if allow_cpu_sync:
399 # Avoid sorting vocab for top-k only case.
400 return apply_top_k_only(logits, k)
401
402 logits_sort, logits_idx = logits.sort(dim=-1, descending=False)
403
404 if k is not None:
405 # Apply top-k.
406 top_k_mask = logits_sort.size(1) - k.to(torch.long) # shape: B
407 # Get all the top_k values.
408 top_k_mask = logits_sort.gather(1, top_k_mask.unsqueeze(dim=1))
409 top_k_mask = logits_sort < top_k_mask
410 logits_sort.masked_fill_(top_k_mask, -float("inf"))
411
412 if p is not None:
413 # Apply top-p.
414 probs_sort = logits_sort.softmax(dim=-1)
415 probs_sum = torch.cumsum(probs_sort, dim=-1, out=probs_sort)
416 top_p_mask = probs_sum <= 1 - p.unsqueeze(dim=1)
417 # at least one
418 top_p_mask[:, -1] = False
419 logits_sort.masked_fill_(top_p_mask, -float("inf"))
420
421 # Re-sort the probabilities.
422 return logits.scatter_(dim=-1, index=logits_idx, src=logits_sort)

425def apply_top_k_only(logits: torch.Tensor, k: torch.Tensor) -> torch.Tensor:
426 """
427 Apply top-k mask to the logits.
428
429 This implementation doesn't involve sorting the entire vocab.
430 Note however that it involves a GPU->CPU sync which can be detrimental for
431 async scheduling performance.
432
433 The logits tensor may be updated in-place.
434 """
435 no_top_k_mask = k == logits.shape[1]
436 # Set non-top-k rows to 1 so that we can gather.
437 k = k.masked_fill(no_top_k_mask, 1)
438 max_top_k = k.max()
439 # topk.values tensor has shape [batch_size, max_top_k].
440 # Convert top k to 0-based index in range [0, max_top_k).
441 k_index = k.sub_(1).unsqueeze(1)
442 top_k_mask = logits.topk(max_top_k, dim=1).values.gather(1, k_index.long())
443 # Handle non-topk rows.
444 top_k_mask.masked_fill_(no_top_k_mask.unsqueeze(1), -float("inf"))
445 return logits.masked_fill_(logits < top_k_mask, -float("inf"))

448def empty_exponential_noise_like(
449 probs: torch.Tensor, use_fp64_gumbel: bool
450) -> torch.Tensor:
451 dtype = torch.float64 if use_fp64_gumbel else probs.dtype
452 return torch.empty(probs.shape, dtype=dtype, device=probs.device)

455def sample_with_exponential_noise(probs: torch.Tensor, q: torch.Tensor) -> torch.Tensor:
456 if q.dtype == probs.dtype:
457 scores = probs.div_(q)
458 else:
459 scores = q.reciprocal_()
460 scores.mul_(probs)
461 return scores.argmax(dim=-1).view(-1)

464def random_sample(
465 probs: torch.Tensor,
466 generators: dict[int, torch.Generator],
467 use_fp64_gumbel: bool = False,
468) -> torch.Tensor:
469 """Randomly sample from the probabilities.
470
471 We use this function instead of torch.multinomial because torch.multinomial
472 causes CPU-GPU synchronization.
473 """
474 q = empty_exponential_noise_like(probs, use_fp64_gumbel)
475 # NOTE(woosuk): To batch-process the requests without their own seeds,
476 # which is the common case, we first assume that every request does
477 # not have its own seed. Then, we overwrite the values for the requests
478 # that have their own seeds.
479 if len(generators) != probs.shape[0]:
480 q.exponential_()
481 if generators:
482 # TODO(woosuk): This can be slow because we handle each request
483 # one by one. Optimize this.
484 for i, generator in generators.items():
485 q[i].exponential_(generator=generator)
486 return sample_with_exponential_noise(probs, q)

489def flashinfer_sample(
490 logits: torch.Tensor,
491 k: torch.Tensor | None,
492 p: torch.Tensor | None,
493 generators: dict[int, torch.Generator] = {}, # noqa
494) -> torch.Tensor:
495 """Sample from the logits using FlashInfer.
496
497 Statistically, this function is equivalent to the `random_sample` function.
498 However, this function is faster because it avoids sorting the logits tensor
499 via rejection sampling.
500
501 NOTE: The outputs of this function do not necessarily match the outputs of
502 the `random_sample` function. It only guarantees that the outputs are
503 statistically equivalent.
504 """
505 import flashinfer
506
507 assert not (k is None and p is None)
508 if k is None:
509 # Top-p only.
510 probs = logits.softmax(dim=-1, dtype=torch.float32)
511 next_token_ids = flashinfer.sampling.top_p_sampling_from_probs(
512 probs, p, deterministic=True
513 )
514 elif p is None:
515 # Top-k only.
516 probs = logits.softmax(dim=-1, dtype=torch.float32)
517 next_token_ids = flashinfer.sampling.top_k_sampling_from_probs(
518 probs, k, deterministic=True
519 )
520 else:
521 # Both top-k and top-p.
522 next_token_ids = flashinfer.sampling.top_k_top_p_sampling_from_logits(
523 logits, k, p, deterministic=True
524 )
525
526 return next_token_ids.view(-1)
