21class Sampler(nn.Module):
22 """
23 A layer that samples the next tokens from the model's outputs
24 with the following steps in order:
25
26 1. If logprobs are requested:
27 a) If `logprobs_mode` is `raw_logprobs`, compute logprobs
28 as the final logprobs to return.
29 b) If `logprobs_mode` is `raw_logits`, clone the logits
30 as the final logprobs to return.
31 2. Convert logits to float32.
32 3. Apply allowed token ids whitelist.
33 4. Apply bad words exclusion.
34 5. Apply logit processors which are not argmax-invariant,
35 i.e. that can impact greedy sampling.
36 a) Min tokens processor
37 b) Logit bias processor
38 6. Apply penalties
39 a) Repetition penalty
40 b) Frequency penalty
41 c) Presence penalty
42 7. Sample the next tokens. `sample` method performs the following steps:
43 a) If not `all_random`, perform greedy sampling. If `all_greedy`,
44 return the greedily sampled tokens and final logprobs if requested.
45 b) Apply temperature.
46 c) Apply logit processors which are argmax-invariant, by default
47 the min_p processor.
48 d) Apply top_k and/or top_p.
49 e) Sample the next tokens with the probability distribution.
50 f) If `all_random` or temperature >= epsilon (1e-5), return the
51 randomly sampled tokens and final logprobs if requested. Else,
52 return the greedily sampled tokens and logprobs if requested.
53 8. Gather the logprobs of the top `max_num_logprobs` and sampled token
54 (if requested). Note that if the sampled token is within the top
55 `max_num_logprobs`, the logprob will be eventually merged in
56 `LogprobsProcessor` during output processing. Therefore, the
57 final output may contain either `max_num_logprobs + 1` or
58 `max_num_logprobs` logprobs.
59 9. Return the final `SamplerOutput`.
60 """
61
62 def __init__(
63 self,
64 logprobs_mode: LogprobsMode = "raw_logprobs",
65 use_fp64_gumbel: bool = False,
66 ):
67 super().__init__()
68 self.topk_topp_sampler = TopKTopPSampler(logprobs_mode, use_fp64_gumbel)
69 self.pin_memory = PIN_MEMORY
70 self.logprobs_mode = logprobs_mode
71 self.use_fp64_gumbel = use_fp64_gumbel

73 def forward(
74 self,
75 logits: torch.Tensor,
76 sampling_metadata: SamplingMetadata,
77 predict_bonus_token: bool = False,
78 logprobs_mode_override: LogprobsMode | None = None,
79 ) -> SamplerOutput:
80 logprobs_mode = logprobs_mode_override or self.logprobs_mode
81 # NOTE(woosuk): Use the original logits (before any penalties or
82 # temperature scaling) for the top-k logprobs.
83 # This is different from the V0 sampler, which uses the logits that
84 # is used for sampling (after penalties and temperature scaling).
85 num_logprobs = sampling_metadata.max_num_logprobs
86 raw_logprobs: torch.Tensor | None = None
87 if num_logprobs is not None or sampling_metadata.logprob_token_ids:
88 if logprobs_mode == "raw_logprobs":
89 raw_logprobs = self.compute_logprobs(logits)
90 elif logprobs_mode == "raw_logits":
91 if logits.dtype == torch.float32:
92 raw_logprobs = logits.clone()
93 else:
94 raw_logprobs = logits.to(torch.float32)
95
96 # Use float32 for the logits.
97 logits = logits.to(torch.float32)
98
99 logits = self.apply_logits_processors(
100 logits, sampling_metadata, predict_bonus_token
101 )
102 # Sample the next token.
103 sampled, processed_logprobs = self.sample(logits, sampling_metadata)
104 if processed_logprobs is not None:
105 raw_logprobs = processed_logprobs
106 # Convert sampled token ids to int64 (long) type to ensure compatibility
107 # with subsequent operations that may use these values as indices.
108 # This conversion is necessary because FlashInfer sampling operations
109 # return int32 (while PyTorch argmax and topk return int64).
110 sampled = sampled.long()
111
112 # Handle logprob_token_ids if specified (more efficient than full vocab)
113 # This is used by generative_scoring API to get logprobs for specific tokens
114 logprob_token_ids_tensors = None
115 if sampling_metadata.logprob_token_ids:
116 assert raw_logprobs is not None
117 logprob_token_ids_tensors = self.gather_specific_token_logprobs(
118 raw_logprobs, sampling_metadata.logprob_token_ids, sampled
119 )
120
121 if num_logprobs is None:
122 logprobs_tensors = logprob_token_ids_tensors
123 elif num_logprobs == -1:
124 # Return the full unsorted and unranked logprobs.
125 logprobs_tensors = LogprobsTensors(
126 torch.empty(0), raw_logprobs, torch.empty(0)
127 )
128 else:
129 # Gather the logprobs and ranks of the topk and sampled token.
130 logprobs_tensors = self.gather_logprobs(
131 raw_logprobs, num_logprobs, token_ids=sampled
132 )
133
134 # If we have both num_logprobs and logprob_token_ids, prefer
135 # logprob_token_ids as it's more specific
136 if logprob_token_ids_tensors is not None and num_logprobs is not None:
137 logprobs_tensors = logprob_token_ids_tensors
138
139 # Use int32 to reduce the tensor size.
140 sampled = sampled.to(torch.int32)
141
142 # These are GPU tensors.
143 sampler_output = SamplerOutput(
144 # The sampled tokens are expanded to 2D tensor with shape
145 # [num_requests, 1], where each row represents one generated
146 # token per request.
147 sampled_token_ids=sampled.unsqueeze(-1),
148 logprobs_tensors=logprobs_tensors,
149 )
150 return sampler_output

152 def gather_specific_token_logprobs(
153 self,
154 logprobs: torch.Tensor,
155 logprob_token_ids: dict[int, list[int]],
156 sampled: torch.Tensor,
157 ) -> LogprobsTensors | None:
158 """Gather logprobs for specific token IDs requested per request.
159
160 Used by the generative_scoring API to return logprobs for an explicit
161 set of token ids rather than the top-k. Handles heterogeneous token
162 id lists across requests by padding shorter lists to the max length.
163
164 Args:
165 logprobs: [batch_size, vocab_size] tensor of (raw) logprobs to
166 gather from.
167 logprob_token_ids: dict mapping req_index -> list of token IDs
168 sampled: [batch_size] tensor of sampled token IDs
169
170 Returns:
171 LogprobsTensors with logprobs for the specified tokens, or None
172 if no requests have logprob_token_ids.
173 """
174 if not logprob_token_ids:
175 return None
176
177 batch_size = logprobs.shape[0]
178 device = logprobs.device
179
180 # Find max number of tokens across all requests
181 max_num_tokens = max(len(tids) for tids in logprob_token_ids.values())
182 pin = self.pin_memory
183
184 # Build the padded token_ids and valid_mask matrices on pinned CPU,
185 # then upload non-blocking.
186 token_ids_cpu = torch.zeros(
187 batch_size, max_num_tokens + 1, dtype=torch.int64, pin_memory=pin
188 )
189 # Create mask for valid positions (True = valid, False = padded)
190 valid_mask_cpu = torch.zeros(
191 batch_size, max_num_tokens + 1, dtype=torch.bool, pin_memory=pin
192 )
193 valid_mask_cpu[:, 0] = True # Sampled token is always valid
194 for req_idx, token_ids in logprob_token_ids.items():
195 num_tokens = len(token_ids)
196 token_ids_cpu[req_idx, 1 : num_tokens + 1] = torch.as_tensor(
197 token_ids, dtype=torch.int64
198 )
199 valid_mask_cpu[req_idx, 1 : num_tokens + 1] = True
200
201 token_ids_tensor = token_ids_cpu.to(device, non_blocking=True)
202 valid_mask = valid_mask_cpu.to(device, non_blocking=True)
203 # Sampled token in column 0 — fill on-device from the sampled GPU
204 # tensor so we don't need to D2H + re-upload.
205 token_ids_tensor[:, 0] = sampled
206
207 # Gather logprobs at the requested token ids.
208 gathered_logprobs = logprobs.gather(-1, token_ids_tensor)
209
210 # Mask invalid (padded) positions with -inf
211 gathered_logprobs = gathered_logprobs.masked_fill(~valid_mask, float("-inf"))
212
213 # Compute ranks for the sampled token. log_softmax is monotonic w.r.t.
214 # the original logits, so ranks computed from logprobs are equivalent.
215 sampled_logprobs = logprobs.gather(-1, sampled.unsqueeze(-1))
216 # Avoid 0/1 specialization recompile on the batch dimension of the
217 # compiled batched_count_greater_than. See gather_logprobs for context.
218 torch._dynamo.decorators.mark_unbacked(logprobs, 0)
219 torch._dynamo.decorators.mark_unbacked(sampled_logprobs, 0)
220 token_ranks = batched_count_greater_than(logprobs, sampled_logprobs)
221
222 return LogprobsTensors(
223 logprob_token_ids=token_ids_tensor.to(torch.int32),
224 logprobs=gathered_logprobs,
225 selected_token_ranks=token_ranks,
226 )

228 @staticmethod
229 def apply_temperature(
230 logits: torch.Tensor,
231 temp: torch.Tensor,
232 all_random: bool,
233 ) -> torch.Tensor:
234 # Use in-place division to avoid creating a new tensor.
235 # Avoid division by zero if there are greedy requests.
236 if not all_random:
237 temp = torch.where(temp < _SAMPLING_EPS, 1.0, temp)
238 return logits.div_(temp.unsqueeze(dim=1))

240 @staticmethod
241 def greedy_sample(logits: torch.Tensor) -> torch.Tensor:
242 return logits.argmax(dim=-1).view(-1)
243
244 def sample(
245 self,
246 logits: torch.Tensor,
247 sampling_metadata: SamplingMetadata,
248 logprobs_mode_override: LogprobsMode | None = None,
249 ) -> tuple[torch.Tensor, torch.Tensor | None]:
250 """Sample logits based on sampling metadata.
251
252 The various logits processing functions called in this method
253 may update the logits tensor in-place.
254 """
255
256 logprobs_mode = logprobs_mode_override or self.logprobs_mode
257 assert not (sampling_metadata.all_greedy and sampling_metadata.all_random)
258 if sampling_metadata.all_random:
259 greedy_sampled = None
260 else:
261 greedy_sampled = self.greedy_sample(logits)
262 if sampling_metadata.all_greedy:
263 processed_logprobs = None
264 if (
265 sampling_metadata.max_num_logprobs is not None
266 or sampling_metadata.logprob_token_ids
267 ):
268 if logprobs_mode == "processed_logits":
269 processed_logprobs = logits
270 elif logprobs_mode == "processed_logprobs":
271 processed_logprobs = self.compute_logprobs(logits)
272 return greedy_sampled, processed_logprobs
273
274 assert sampling_metadata.temperature is not None
275
276 # Apply temperature.
277 logits = self.apply_temperature(
278 logits, sampling_metadata.temperature, sampling_metadata.all_random
279 )
280
281 # Apply logits processors that only apply to random sampling
282 # (argmax invariant)
283 for processor in sampling_metadata.logitsprocs.argmax_invariant:
284 logits = processor.apply(logits)
285
286 # Apply top_k and/or top_p.
287 random_sampled, processed_logprobs = self.topk_topp_sampler(
288 logits,
289 sampling_metadata.generators,
290 sampling_metadata.top_k,
291 sampling_metadata.top_p,
292 )
293
294 if greedy_sampled is None:
295 return random_sampled, processed_logprobs
296
297 sampled = torch.where(
298 sampling_metadata.temperature < _SAMPLING_EPS,
299 greedy_sampled,
300 random_sampled,
301 out=greedy_sampled, # Reuse tensor
302 )
303 return sampled, processed_logprobs

305 @staticmethod
306 def compute_logprobs(logits: torch.Tensor) -> torch.Tensor:
307 return logits.log_softmax(dim=-1, dtype=torch.float32)
308
309 @staticmethod
310 def gather_logprobs(
311 logprobs: torch.Tensor,
312 num_logprobs: int,
313 token_ids: torch.Tensor,
314 ) -> LogprobsTensors:
315 """
316 Gather logprobs for topk and sampled/prompt token.
317
318 Args:
319 logprobs: (num tokens) x (vocab) tensor
320 num_logprobs: maximum number of logprobs to
321 retain per token
322 token_ids: prompt tokens (if prompt logprobs)
323 or sampled tokens (if sampled
324 logprobs); 1D token ID tensor
325 with (num tokens) elements
326 Must be int64.
327
328 Returns:
329 Top-k int indices tensor, (num tokens) x (num_logprobs + 1)
330 Top-k float logprobs tensor, (num tokens) x (num_logprobs + 1)
331 Sampled token rank tensor, (num tokens)
332 """
333 assert token_ids.dtype == torch.int64
334 # Find the topK values.
335 topk_logprobs, topk_indices = torch.topk(logprobs, num_logprobs, dim=-1)
336
337 # Get with the logprob of the prompt or sampled token.
338 token_ids = token_ids.unsqueeze(-1)
339 token_logprobs = logprobs.gather(-1, token_ids)
340
341 # Compute the ranks of the actual token.
342 # Avoid 0/1 specialization recompile on the batch dimension
343 # of the compiled batched_count_greater_than. mark_unbacked makes
344 # the size fully symbolic so dynamo doesn't specialize when
345 # batch_size transitions from 1 to >=2.
346 with gpu_sync_allowed(first_only=True):
347 torch._dynamo.decorators.mark_unbacked(logprobs, 0)
348 torch._dynamo.decorators.mark_unbacked(token_logprobs, 0)
349 token_ranks = batched_count_greater_than(logprobs, token_logprobs)
350
351 # Concatenate together with the topk.
352 indices = torch.cat((token_ids, topk_indices), dim=1)
353 logprobs = torch.cat((token_logprobs, topk_logprobs), dim=1)
354
355 # Use int32 to reduce the tensor size.
356 indices = indices.to(torch.int32)
357
358 return LogprobsTensors(indices, logprobs, token_ranks)

360 @staticmethod
361 def _combine_outputs_with_spec_tokens(
362 output_token_ids: list[list[int]],
363 spec_token_ids: list[list[int]] | None = None,
364 ) -> list[list[int]]:
365 if spec_token_ids is None:
366 return output_token_ids
367
368 return [
369 [*out, *spec] if spec else out
370 for out, spec in zip(output_token_ids, spec_token_ids)
371 ]

373 def apply_logits_processors(
374 self,
375 logits: torch.Tensor,
376 sampling_metadata: SamplingMetadata,
377 predict_bonus_token: bool,
378 ) -> torch.Tensor:
379 bad_words_token_ids = sampling_metadata.bad_words_token_ids
380 any_penalties_or_bad_words = (
381 bool(bad_words_token_ids) or not sampling_metadata.no_penalties
382 )
383 output_token_ids = sampling_metadata.output_token_ids
384 if predict_bonus_token and any_penalties_or_bad_words:
385 # Combine base outputs with spec tokens when speculative decoding
386 # is enabled.
387 output_token_ids = self._combine_outputs_with_spec_tokens(
388 output_token_ids,
389 sampling_metadata.spec_token_ids,
390 )
391
392 # Apply allowed token ids.
393 if sampling_metadata.allowed_token_ids_mask is not None:
394 logits.masked_fill_(sampling_metadata.allowed_token_ids_mask, float("-inf"))
395
396 # Apply bad words exclusion.
397 if bad_words_token_ids:
398 apply_bad_words(logits, bad_words_token_ids, output_token_ids)
399
400 # Apply logits processors which can impact greedy sampling.
401 for processor in sampling_metadata.logitsprocs.non_argmax_invariant:
402 logits = processor.apply(logits)
403
404 # Apply penalties (e.g., freq_penalties).
405 logits = self.apply_penalties(logits, sampling_metadata, output_token_ids)
406 holder = sampling_metadata.thinking_budget_state_holder
407 if holder is not None and holder.has_tracked_requests():
408 # Committed outputs only; spec drafts live in ``spec_token_ids``.
409 holder.update_state(
410 sampling_metadata.output_token_ids,
411 sampling_metadata.spec_token_ids,
412 repeat_indices=None,
413 )
414 logits = holder.apply_to_logits(
415 logits,
416 predict_bonus_token,
417 sampling_metadata.spec_token_ids,
418 )
419 return logits

421 @staticmethod
422 def apply_penalties(
423 logits: torch.Tensor,
424 sampling_metadata: SamplingMetadata,
425 output_token_ids: list[list[int]],
426 ) -> torch.Tensor:
427 if sampling_metadata.no_penalties:
428 return logits
429
430 assert sampling_metadata.prompt_token_ids is not None
431 return apply_all_penalties(
432 logits,
433 sampling_metadata.prompt_token_ids,
434 sampling_metadata.presence_penalties,
435 sampling_metadata.frequency_penalties,
436 sampling_metadata.repetition_penalties,
437 output_token_ids,
438 )
