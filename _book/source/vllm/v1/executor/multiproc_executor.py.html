78class FutureWrapper(Future):
79 def __init__(
80 self,
81 futures_queue: deque["FutureWrapper"],
82 get_response: Callable[[], Any],
83 aggregate: Callable = lambda x: x,
84 ):
85 self.futures_queue = futures_queue
86 self.get_response = get_response
87 self.aggregate = aggregate
88 super().__init__()
89 self.futures_queue.appendleft(self)
90
91 def result(self, timeout=None):
92 if timeout is not None:
93 raise RuntimeError("timeout not implemented")
94
95 # Drain any futures ahead of us in the queue.
96 while not self.done():
97 future = self.futures_queue.pop()
98 future._wait_for_response()
99 return super().result()
100
101 def _wait_for_response(self):
102 try:
103 response = self.aggregate(self.get_response())
104 with suppress(InvalidStateError):
105 self.set_result(response)
106 except Exception as e:
107 with suppress(InvalidStateError):
108 self.set_exception(e)

118 def _init_executor(self) -> None:
119 # Call self.shutdown at exit to clean up
120 # and ensure workers will be terminated.
121 self._finalizer = weakref.finalize(self, self.shutdown)
122 self.is_failed = False
123 self.failure_callback: FailureCallback | None = None
124
125 tp_size, pp_size, pcp_size = self._get_parallel_sizes()
126 assert self.world_size == tp_size * pp_size * pcp_size, (
127 f"world_size ({self.world_size}) must be equal to the "
128 f"tensor_parallel_size ({tp_size}) x pipeline"
129 f"_parallel_size ({pp_size}) x prefill_context"
130 f"_parallel_size ({pcp_size}). "
131 )
132
133 num_local_procs = self.local_world_size * max(
134 1, self.parallel_config.data_parallel_size_local
135 )
136 set_multiprocessing_worker_envs(num_local_procs)
137
138 if aiter_requires_tcp_store():
139 distributed_init_method = get_distributed_init_method(
140 get_loopback_ip(), get_open_port()
141 )
142 else:
143 distributed_init_method = get_file_store_init_method()
144 self.rpc_broadcast_mq: MessageQueue | None = None
145 scheduler_output_handle: Handle | None = None
146 # Initialize worker and set up message queues for SchedulerOutputs
147 # and ModelRunnerOutputs
148 if self.parallel_config.node_rank_within_dp == 0:
149 # For leader node within each dp rank,
150 # each dp will have its own leader multiproc executor.
151 max_chunk_bytes = envs.VLLM_MQ_MAX_CHUNK_BYTES_MB * 1024 * 1024
152 mq_connect_ip = get_ip()
153 logger.info(
154 "DP group leader: node_rank=%d, node_rank_within_dp=%d, "
155 "master_addr=%s, mq_connect_ip=%s (local), "
156 "world_size=%d, local_world_size=%d",
157 self.parallel_config.node_rank,
158 self.parallel_config.node_rank_within_dp,
159 self.parallel_config.master_addr,
160 mq_connect_ip,
161 self.world_size,
162 self.local_world_size,
163 )
164 self.rpc_broadcast_mq = MessageQueue(
165 self.world_size,
166 self.local_world_size,
167 max_chunk_bytes=max_chunk_bytes,
168 connect_ip=mq_connect_ip,
169 )
170 scheduler_output_handle = self.rpc_broadcast_mq.export_handle()
171 # Create workers
172 context = get_mp_context()
173 shared_worker_lock = context.Lock()
174 unready_workers: list[UnreadyWorkerProcHandle] = []
175 success = False
176 try:
177 global_start_rank = (
178 self.local_world_size * self.parallel_config.node_rank_within_dp
179 )
180 # When using fork, keep track of socket file descriptors that are
181 # inherited by the worker, so that we can close them in subsequent
182 # workers
183 inherited_fds: list[int] | None = (
184 [] if context.get_start_method() == "fork" else None
185 )
186
187 # For CPU backend only, to setup OpenMP threads affinity
188 cpu_omp_manager = OMPProcessManager(self.vllm_config)
189 for local_rank in range(self.local_world_size):
190 global_rank = global_start_rank + local_rank
191 is_driver_worker = self._is_driver_worker(global_rank)
192 with cpu_omp_manager.configure_omp_envs(
193 rank=global_rank, local_rank=local_rank
194 ):
195 unready_worker_handle = WorkerProc.make_worker_process(
196 vllm_config=self.vllm_config,
197 local_rank=local_rank,
198 rank=global_rank,
199 distributed_init_method=distributed_init_method,
200 input_shm_handle=scheduler_output_handle,
201 shared_worker_lock=shared_worker_lock,
202 is_driver_worker=is_driver_worker,
203 inherited_fds=inherited_fds,
204 )
205 unready_workers.append(unready_worker_handle)
206 if inherited_fds is not None:
207 inherited_fds.append(unready_worker_handle.death_writer.fileno())
208 inherited_fds.append(unready_worker_handle.ready_pipe.fileno())
209
210 # Workers must be created before wait_for_ready to avoid
211 # deadlock, since worker.init_device() does a device sync.
212
213 # Wait for all local workers to be ready.
214 self.workers = WorkerProc.wait_for_ready(unready_workers)
215
216 # The workers have inherited their thread count (see
217 # set_multiprocessing_worker_envs); this process only schedules, so
218 # it gets no benefit from torch intra-op parallelism, just CPU
219 # contention with them.
220 set_torch_threads_for_runtime()
221
222 # Start background thread to monitor worker health if not in headless mode.
223 if self.monitor_workers:
224 self.start_worker_monitor()
225
226 self.response_mqs = []
227 # Only leader node have remote response mqs
228 if self.parallel_config.node_rank_within_dp == 0:
229 for rank in range(self.world_size):
230 if rank < self.local_world_size:
231 local_message_queue = self.workers[rank].worker_response_mq
232 assert local_message_queue is not None
233 self.response_mqs.append(local_message_queue)
234 else:
235 remote_message_queue = self.workers[0].peer_worker_response_mqs[
236 rank
237 ]
238 assert remote_message_queue is not None
239 self.response_mqs.append(remote_message_queue)
240
241 # Ensure message queues are ready. Will deadlock if re-ordered
242 # Must be kept consistent with the WorkerProc.
243
244 # Wait for all input mqs to be ready.
245 if self.rpc_broadcast_mq is not None:
246 self.rpc_broadcast_mq.wait_until_ready()
247 # Wait for all remote response mqs to be ready.
248 for response_mq in self.response_mqs:
249 response_mq.wait_until_ready()
250
251 self.futures_queue = deque[FutureWrapper]()
252
253 self._post_init_executor()
254
255 success = True
256 finally:
257 if not success:
258 # Clean up the worker procs if there was a failure.
259 # Close death_writers first to signal workers to exit
260 for uw in unready_workers:
261 if uw.death_writer is not None:
262 uw.death_writer.close()
263 uw.death_writer = None
264 self._ensure_worker_termination([uw.proc for uw in unready_workers])
265
266 self.output_rank = self._get_output_rank()

302 # Monitors worker process liveness. If any die unexpectedly,
303 # logs an error, shuts down the executor and invokes the failure
304 # callback to inform the engine.
305 def monitor_workers():
306 sentinels = [h.proc.sentinel for h in workers]
307 died = multiprocessing.connection.wait(sentinels)
308 _self = self_ref()
309 if not _self or getattr(_self, "shutting_down", False):
310 logger.debug("MultiprocWorkerMonitor: shutdown already initiated")
311 return
312 _self.is_failed = True
313 proc = next(h.proc for h in workers if h.proc.sentinel == died[0])
314 logger.error(
315 "Worker proc %s died unexpectedly (exit code: %s), "
316 "shutting down executor.",
317 proc.name,
318 proc.exitcode,
319 )
320 _self.shutdown()
321 callback = _self.failure_callback
322 if callback is not None:
323 _self.failure_callback = None
324 callback()
325
326 if not inline:
327 Thread(
328 target=monitor_workers, daemon=True, name="MultiprocWorkerMonitor"
329 ).start()
330 return

332 monitor_workers()

375 def collective_rpc( # type: ignore[override]
376 self,
377 method: str | Callable,
378 timeout: float | None = None,
379 args: tuple = (),
380 kwargs: dict | None = None,
381 non_block: bool = False,
382 unique_reply_rank: int | None = None,
383 kv_output_aggregator: KVOutputAggregator | None = None,
384 ec_output_aggregator: ECOutputAggregator | None = None,
385 ) -> Any:
386 """Returns single result if unique_reply_rank and/or an output
387 aggregator is provided, otherwise list."""
388 assert self.rpc_broadcast_mq is not None, (
389 "collective_rpc should not be called on follower node"
390 )
391 if self.is_failed:
392 raise RuntimeError("Executor failed.")
393
394 deadline = None if timeout is None else time.monotonic() + timeout
395 kwargs = kwargs or {}
396
397 aggregators = [a for a in (kv_output_aggregator, ec_output_aggregator) if a]
398 aggregate: Callable[[Any], Any]
399 if aggregators:
400 output_rank = None
401
402 def _aggregate(outputs: Any) -> Any:
403 # Each aggregator merges its own connector's output onto
404 # outputs[output_rank] in place and returns it, so chaining is safe.
405 rank = unique_reply_rank or 0
406 result = outputs[rank]
407 for a in aggregators:
408 result = a.aggregate(outputs, output_rank=rank)
409 return result
410
411 aggregate = _aggregate
412 else:
413 output_rank = unique_reply_rank
414 aggregate = lambda x: x
415
416 if isinstance(method, str):
417 send_method = method
418 else:
419 send_method = cloudpickle.dumps(method, protocol=pickle.HIGHEST_PROTOCOL)
420 self.rpc_broadcast_mq.enqueue((send_method, args, kwargs, output_rank))
421
422 response_mqs: Sequence[MessageQueue] = self.response_mqs
423 if output_rank is not None:
424 response_mqs = (response_mqs[output_rank],)
425
426 def get_response():
427 responses = []
428 for mq in response_mqs:
429 dequeue_timeout = (
430 None if deadline is None else max(0.0, deadline - time.monotonic())
431 )
432 try:
433 status, result = mq.dequeue(timeout=dequeue_timeout)
434 except TimeoutError as e:
435 raise TimeoutError(f"RPC call to {method} timed out.") from e
436 if status != WorkerProc.ResponseStatus.SUCCESS:
437 raise RuntimeError(
438 f"Worker failed with error '{result}', please check the"
439 " stack trace above for the root cause"
440 )
441 responses.append(result)
442 return responses[0] if output_rank is not None else responses
443
444 future = FutureWrapper(
445 self.futures_queue, get_response=get_response, aggregate=aggregate
446 )
447
448 return future if non_block else future.result()

450 @staticmethod
451 def _ensure_worker_termination(worker_procs: list[BaseProcess]):
452 """Ensure that all worker processes are terminated. Assumes workers have
453 received termination requests. Waits for processing, then sends
454 termination and kill signals if needed."""
455
456 def wait_for_termination(procs, timeout):
457 if not time:
458 # If we are in late stage shutdown, the interpreter may replace
459 # `time` with `None`.
460 return all(not proc.is_alive() for proc in procs)
461 start_time = time.time()
462 while time.time() - start_time < timeout:
463 if all(not proc.is_alive() for proc in procs):
464 return True
465 time.sleep(0.1)
466 return False
467
468 active_procs = lambda: [proc for proc in worker_procs if proc.is_alive()]
469 initial_count = len(active_procs())
470
471 # Give processes time to clean themselves up properly first
472 logger.info(
473 "[shutdown] Executor: waiting for worker exit count=%d",
474 initial_count,
475 )
476 if wait_for_termination(
477 active_procs(), timeout=envs.VLLM_WORKER_SHUTDOWN_TIMEOUT_SECONDS
478 ):
479 logger.info_once("[shutdown] Executor: all workers exited gracefully")
480 return
481
482 # Send SIGTERM if still running
483 remaining = active_procs()
484 logger.warning(
485 "[shutdown] Executor: workers still running after grace period; "
486 "sending SIGTERM count=%d",
487 len(remaining),
488 )
489 for p in remaining:
490 p.terminate()
491 if not wait_for_termination(active_procs(), 4):
492 # Send SIGKILL if still running
493 remaining = active_procs()
494 logger.warning(
495 "[shutdown] Executor: workers still running after SIGTERM; "
496 "sending SIGKILL count=%d",
497 len(remaining),
498 )
499 for p in remaining:
500 p.kill()

502 def shutdown(self):
503 """Properly shut down the executor and its workers"""
504 if not getattr(self, "shutting_down", False):
505 worker_count = len(getattr(self, "workers", None) or [])
506 logger.debug(
507 "[shutdown] Executor: start worker_count=%d",
508 worker_count,
509 )
510 self.shutting_down = True
511
512 # Make sure all the worker processes are terminated first.
513 if workers := getattr(self, "workers", None):
514 for w in workers:
515 # Close death_writer to signal child processes to exit
516 if w.death_writer is not None:
517 w.death_writer.close()
518 w.death_writer = None
519 self._ensure_worker_termination([w.proc for w in workers])
520
521 for w in workers:
522 # Shutdown response queues
523 if w.worker_response_mq is not None:
524 w.worker_response_mq.shutdown()
525 w.worker_response_mq = None
526
527 if rpc_broadcast_mq := getattr(self, "rpc_broadcast_mq", None):
528 rpc_broadcast_mq.shutdown()
529 self.rpc_broadcast_mq = None
530 if response_mqs := getattr(self, "response_mqs", None):
531 for mq in response_mqs:
532 mq.shutdown()
533 self.response_mqs = []
534
535 logger.debug_once("[shutdown] Executor: complete")
536

541 def _get_output_rank(self) -> int:
542 # Only returns ModelRunnerOutput from TP rank=0 and PP rank=-1
543 # (the first TP worker of the last PP stage).
544 # Example:
545 # Assuming TP=8, PP=4, then the world_size=32
546 # 0-7, PP rank 0
547 # 8-15, PP rank 1
548 # 16-23, PP rank 2
549 # 24-31, PP rank 3
550 # so world_size - tp_size = 32 - 8 = 24 should be PP rank = -1 (i.e. 3)
551 return (
552 self.world_size
553 - self.parallel_config.tensor_parallel_size
554 * self.parallel_config.prefill_context_parallel_size
555 )

778 @staticmethod
779 def wait_for_ready(
780 unready_proc_handles: list[UnreadyWorkerProcHandle],
781 ) -> list[WorkerProcHandle]:
782 e = Exception(
783 "WorkerProc initialization failed due to an exception in a "
784 "background process. See stack trace for root cause."
785 )
786
787 pipes = {handle.ready_pipe: handle for handle in unready_proc_handles}
788 ready_proc_handles: list[WorkerProcHandle | None] = [None] * len(
789 unready_proc_handles
790 )
791 while pipes:
792 ready = multiprocessing.connection.wait(pipes.keys())
793 for pipe in ready:
794 assert isinstance(pipe, Connection)
795 try:
796 # Wait until the WorkerProc is ready.
797 unready_proc_handle = pipes.pop(pipe)
798 response: dict[str, Any] = pipe.recv()
799 if response["status"] != "READY":
800 raise e
801
802 idx = unready_proc_handle.rank % len(ready_proc_handles)
803 ready_proc_handles[idx] = WorkerProc.wait_for_response_handle_ready(
804 response, unready_proc_handle
805 )
806 except EOFError:
807 e.__suppress_context__ = True
808 raise e from None
809
810 finally:
811 # Close connection.
812 pipe.close()

852 @staticmethod
853 def worker_main(*args, **kwargs):
854 """Worker initialization and execution loops.
855 This runs a background process"""
856
857 # Signal handler used for graceful termination.
858 # SystemExit exception is only raised once to allow this and worker
859 # processes to terminate without error
860 shutdown_requested = threading.Event()
861
862 def signal_handler(signum, frame):
863 nonlocal shutdown_requested
864 if not shutdown_requested.is_set():
865 shutdown_requested.set()
866 logger.debug(
867 "WorkerProc handling signal %d, raising SystemExit", signum
868 )
869 raise SystemExit()
870
871 # Either SIGTERM or SIGINT will terminate the worker
872 signal.signal(signal.SIGTERM, signal_handler)
873 signal.signal(signal.SIGINT, signal_handler)
874
875 # Publish the logical-to-physical mapping early so topology helpers
876 # work before init_device (needed by set_worker_net_device below).
877 assigned_physical_gpu_ids = kwargs[
878 "vllm_config"
879 ].parallel_config.assigned_physical_gpu_ids
880 if assigned_physical_gpu_ids is not None:
881 from vllm.platforms.interface import set_assigned_physical_gpu_ids
882
883 set_assigned_physical_gpu_ids(assigned_physical_gpu_ids)
884
885 # Set net device env vars for the worker if VLLM_GPU_NIC_PCIE_MAPPING is set
886 set_worker_net_device(kwargs.get("local_rank", 0), kwargs["vllm_config"])
887
888 worker = None
889 ready_writer = kwargs.pop("ready_pipe")
890 death_pipe = kwargs.pop("death_pipe", None)
891
892 # Close inherited pipes from parent (incl. other worker pipes)
893 # Explicitly passing in existing pipes and closing them makes the pipe
894 # behave when using fork. Otherwise, a hidden reference to the pipes
895 # exist in the child process and prevents EOF closure.
896 for fd in kwargs.pop("inherited_fds", []):
897 try:
898 os.close(fd)
899 except Exception as e:
900 logger.warning("Error closing inherited connection: %s: %s", type(e), e)
901
902 try:
903 # Initialize tracer
904 rank = kwargs.get("rank", 0)
905 maybe_init_worker_tracer(
906 instrumenting_module_name="vllm.worker",
907 process_kind="worker",
908 process_name=f"Worker_{rank}",
909 )
910
911 worker = WorkerProc(*args, **kwargs)
912 assert worker.worker_response_mq is not None
913 if kwargs["vllm_config"].parallel_config.numa_bind:
914 numa_utils.log_current_affinity_state(f"Worker_{worker.rank}")
915
916 worker.monitor_death_pipe(death_pipe, shutdown_requested)
917
918 # Send READY once we know everything is loaded
919 ready_writer.send(
920 {
921 "status": WorkerProc.READY_STR,
922 "handle": worker.worker_response_mq.export_handle(),
923 "peer_response_handles": worker.peer_response_handles,
924 }
925 )
926
927 # Ensure message queues are ready. Will deadlock if re-ordered.
928 # Must be kept consistent with the Executor
929 if worker.rpc_broadcast_mq is not None:
930 worker.rpc_broadcast_mq.wait_until_ready()
931 worker.worker_response_mq.wait_until_ready()
932 ready_writer.close()
933 ready_writer = None
934
935 worker.worker_busy_loop()
936
937 except Exception:
938 # NOTE: if an Exception arises in busy_loop, we send
939 # a FAILURE message over the MQ RPC to notify the Executor,
940 # which triggers system shutdown.
941 # TODO(rob): handle case where the MQ itself breaks.
942
943 if ready_writer is not None:
944 logger.exception("WorkerProc failed to start.")
945 elif shutdown_requested.is_set():
946 logger.debug_once(
947 "[shutdown] WorkerProc: exiting after shutdown request"
948 )
949 else:
950 logger.exception("WorkerProc failed.")
951
952 # The parent sends a SIGTERM to all worker processes if
953 # any worker dies. Set this value so we don't re-throw
954 # SystemExit() to avoid zmq exceptions in __del__.
955 shutdown_requested.set()
956
957 except SystemExit as e:
958 # SystemExit is raised on SIGTERM or SIGKILL, which usually indicates that
959 # the graceful shutdown process did not succeed
960 if shutdown_requested.is_set():
961 logger.debug_once(
962 "[shutdown] WorkerProc: terminated by shutdown signal"
963 )
964 else:
965 logger.warning("WorkerProc was terminated")
966 # SystemExit must never be ignored
967 raise e
968
969 finally:
970 if ready_writer is not None:
971 ready_writer.close()
972 if death_pipe is not None:
973 death_pipe.close()
974 # Clean up once worker exits busy loop
975 if worker is not None:
976 worker.shutdown()

982 def enqueue_output(self, output: Any):
983 """Prepares output from the worker and enqueues it to the
984 worker_response_mq. If the output is an Exception, it is
985 converted to a FAILURE response.
986 """
987 if isinstance(output, AsyncModelRunnerOutput):
988 try:
989 output = output.get_output()
990 except Exception as e:
991 logger.exception("Error getting async model runner output")
992 output = e
993
994 if isinstance(output, Exception):
995 result = (WorkerProc.ResponseStatus.FAILURE, str(output))
996 else:
997 result = (WorkerProc.ResponseStatus.SUCCESS, output)
998 if (response_mq := self.worker_response_mq) is not None:
999 response_mq.enqueue(result)

1001 def handle_output(self, output: Any):
1002 """Handles output from the worker. If async scheduling is enabled,
1003 it is passed to the async_output_busy_loop thread. Otherwise, it is
1004 enqueued directly to the worker_response_mq.

1007 self.async_output_queue.put(output)

1011 def async_output_busy_loop(self):
1012 """Entrypoint for the thread which handles outputs asynchronously."""
1013
1014 # set device to the worker device for the thread.
1015 # a thread will not inherit the context of the main thread.
1016 # when calling any cuda runtime functions, it will implicitly
1017 # create a new cuda context on device 0, consuming extra memory.
1018 # here we set the device to the worker device for the thread,
1019 # enforcing the context to be the same as the main thread.
1020 from vllm.platforms import current_platform
1021
1022 if hasattr(self.worker, "device"):
1023 current_platform.set_device(self.worker.device)
1024
1025 while True:
1026 output = self.async_output_queue.get()
1027 self.enqueue_output(output)

1029 def worker_busy_loop(self):
1030 """Main busy loop for Multiprocessing Workers"""
1031 assert self.rpc_broadcast_mq is not None
1032 while True:
1033 self._execute_worker_rpc(self.rpc_broadcast_mq.dequeue(indefinite=True))

1035 def _execute_worker_rpc(
1036 self,
1037 rpc_request: tuple[str | bytes, tuple[Any, ...], dict[str, Any], int | None],
1038 ) -> None:
1039 """Execute one RPC in a separate frame from the dequeue loop."""
1040 method, args, kwargs, output_rank = rpc_request
1041 try:
1042 if isinstance(method, str):
1043 func = getattr(self.worker, method)
1044 elif isinstance(method, bytes):
1045 func = partial(cloudpickle.loads(method), self.worker)
1046
1047 output = func(*args, **kwargs)
1048
1049 if output_rank is None or self.rank == output_rank:
1050 self.handle_output(output)
1051 except Exception as e:
1052 # Notes have been introduced in python 3.11
1053 if hasattr(e, "add_note"):
1054 e.add_note(traceback.format_exc())
1055 logger.exception("WorkerProc hit an exception.")
1056 # enqueue_output converts the exception to a FAILURE response
1057 # containing its string representation before transport.
1058 if output_rank is None or self.rank == output_rank:
1059 self.handle_output(e)

1099 """Set up environment variables that should be used when there are workers
1100 in a multiprocessing environment. This should be called by the parent
1101 process before worker processes are created"""
1102

1109 # must not set its own: `torch.set_num_threads()` spawns the thread pool

1120 # a forked child whose parent had run a parallel region deadlocks, whereas
1121 # one whose parent merely sized the pool does not.
1122 torch.set_num_threads(num_threads)

1124 "Set OMP_NUM_THREADS=%d for %d worker process(es).",
1125 num_threads,
