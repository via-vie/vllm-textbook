38class Executor(ABC):
39 """Abstract base class for vLLM executors."
40
41 An executor is responsible for executing the model on one device,
42 or it can be a distributed executor that can execute the model on multiple devices.
43 """
44
45 uses_ray: bool = False # whether the executor uses Ray for orchestration.
46 supports_pp: bool = False # whether the executor supports PP
47
48 @staticmethod
49 def get_class(vllm_config: VllmConfig) -> type["Executor"]:
50 executor_class: type[Executor]
51 parallel_config = vllm_config.parallel_config
52 distributed_executor_backend = parallel_config.distributed_executor_backend
53 # distributed_executor_backend must be set in VllmConfig.__post_init__
54 if isinstance(distributed_executor_backend, type):
55 if not issubclass(distributed_executor_backend, Executor):
56 raise TypeError(
57 "distributed_executor_backend must be a subclass of "
58 f"Executor. Got {distributed_executor_backend}."
59 )
60 executor_class = distributed_executor_backend
61 elif distributed_executor_backend == "ray":
62 if envs.VLLM_USE_RAY_V2_EXECUTOR_BACKEND:
63 from vllm.v1.executor.ray_executor_v2 import RayExecutorV2
64
65 executor_class = RayExecutorV2
66 else:
67 from vllm.v1.executor.ray_executor import RayDistributedExecutor
68
69 executor_class = RayDistributedExecutor
70 elif distributed_executor_backend == "mp":
71 from vllm.v1.executor.multiproc_executor import MultiprocExecutor
72
73 executor_class = MultiprocExecutor
74 elif distributed_executor_backend == "uni":
75 from vllm.v1.executor.uniproc_executor import UniProcExecutor
76
77 executor_class = UniProcExecutor
78 elif distributed_executor_backend == "external_launcher":
79 # TODO: make v1 scheduling deterministic
80 # to support external launcher
81 executor_class = ExecutorWithExternalLauncher
82 elif isinstance(distributed_executor_backend, str):
83 executor_class = resolve_obj_by_qualname(distributed_executor_backend)
84 if not issubclass(executor_class, Executor):
85 raise TypeError(
86 "distributed_executor_backend must be a subclass of "
87 f"Executor. Got {executor_class}."
88 )
89 else:
90 raise ValueError(
91 f"Unknown distributed executor backend: {distributed_executor_backend}"
92 )
93 return executor_class

162 @overload
163 def collective_rpc(
164 self,
165 method: str | Callable[[WorkerBase], _R],
166 timeout: float | None = None,
167 args: tuple = (),
168 kwargs: dict | None = None,
169 non_block: Literal[False] = False,
170 ) -> list[_R]:
171 """
172 Execute an RPC call on all workers.
173
174 Args:
175 method: Name of the worker method to execute, or a callable that
176 is serialized and sent to all workers to execute.
177
178 If the method is a callable, it should accept an additional
179 `self` argument, in addition to the arguments passed in `args`
180 and `kwargs`. The `self` argument will be the worker object.
181 timeout: Maximum time in seconds to wait for execution. Raises a
182 [`TimeoutError`][] on timeout. `None` means wait indefinitely.
183 args: Positional arguments to pass to the worker method.
184 kwargs: Keyword arguments to pass to the worker method.
185 non_block: If `True`, returns a list of Futures instead of waiting
186 for the results.
187
188 Returns:
189 A list containing the results from each worker.
190
191 Note:
192 It is recommended to use this API to only pass control messages,
193 and set up data-plane communication to pass data.
194 """
195 pass

337 def sleep(self, level: int = 1):
338 if self.is_sleeping:
339 logger.warning("Executor is already sleeping.")
340 return
341 time_before_sleep = time.perf_counter()
342 self.collective_rpc("sleep", kwargs=dict(level=level))
343 time_after_sleep = time.perf_counter()
344 self.sleeping_tags = {"weights", "kv_cache"}
345 self.is_sleeping = True
346 logger.info(
347 "It took %.6f seconds to fall asleep.", time_after_sleep - time_before_sleep
348 )
349
350 def wake_up(self, tags: list[str] | None = None):
351 if not self.is_sleeping:
352 logger.warning("Executor is not sleeping.")
353 return
354 if tags:
355 for tag in tags:
356 if tag not in self.sleeping_tags:
357 logger.warning(
358 "Tag %s is not in sleeping tags %s", tag, self.sleeping_tags
359 )
360 return
361 time_before_wakeup = time.perf_counter()
362 self.collective_rpc("wake_up", kwargs=dict(tags=tags))
363 time_after_wakeup = time.perf_counter()
364 logger.info(
365 "It took %.6f seconds to wake up tags %s.",
366 time_after_wakeup - time_before_wakeup,
367 tags if tags is not None else self.sleeping_tags,
368 )
369 if tags:
370 for tag in tags:
371 self.sleeping_tags.remove(tag)
372 else:
373 self.sleeping_tags.clear()
374 if not self.sleeping_tags:
375 self.is_sleeping = False
