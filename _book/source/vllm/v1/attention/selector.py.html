64def get_attn_spec_kind(
65 use_mla: bool,
66 has_sliding_window: bool,
67 attn_type: str,
68) -> "KVCacheSpecKind":
69 """Derive the KV-cache group kind a layer belongs to from its signals.
70
71 Mirrors ``get_kv_cache_spec_kind`` (which derives the kind from the
72 produced ``KVCacheSpec``) so users can target groups by kind when
73 setting ``AttentionConfig.backend_per_kind``.
74
75 ``SINK_FULL_ATTENTION`` is intentionally not derived here: it is produced
76 only by the ``StaticSinkAttention`` layer, whereas a plain ``Attention``
77 layer with attention sinks (e.g. gpt-oss) still yields a
78 ``FullAttentionSpec``/``SlidingWindowSpec``. Sinks therefore do not change
79 the kind.
80
81 Args:
82 use_mla: Whether the layer uses multi-head latent attention.
83 has_sliding_window: Whether the layer applies a sliding window.
84 attn_type: The layer's ``AttentionType``.
85
86 Returns:
87 The ``KVCacheSpecKind`` the layer maps to.
88 """
89 from vllm.v1.kv_cache_interface import KVCacheSpecKind
90
91 if attn_type == AttentionType.ENCODER_ONLY:
92 return KVCacheSpecKind.ENCODER_ONLY_ATTENTION
93 if attn_type == AttentionType.ENCODER_DECODER:
94 return KVCacheSpecKind.CROSS_ATTENTION
95 if use_mla:
96 if has_sliding_window:
97 return KVCacheSpecKind.SLIDING_WINDOW_MLA
98 return KVCacheSpecKind.MLA_ATTENTION
99 if has_sliding_window:
100 return KVCacheSpecKind.SLIDING_WINDOW
101 return KVCacheSpecKind.FULL_ATTENTION

104def get_attn_backend(
105 head_size: int,
106 dtype: torch.dtype,
107 kv_cache_dtype: str | None,
108 use_mla: bool = False,
109 has_sink: bool = False,
110 use_sparse: bool = False,
111 use_mm_prefix: bool = False,
112 use_per_head_quant_scales: bool = False,
113 attn_type: str | None = None,
114 num_heads: int | None = None,
115 has_sliding_window: bool = False,
116) -> type[AttentionBackend]:
117 """Selects which attention backend to use and lazily imports it."""
118
119 if kv_cache_dtype is not None:
120 valid_cache_dtypes = get_args(CacheDType)
121 assert kv_cache_dtype in valid_cache_dtypes, (
122 f"Invalid kv_cache_dtype: {kv_cache_dtype}. "
123 f"Valid values are: {valid_cache_dtypes}"
124 )
125
126 from vllm.config import get_current_vllm_config
127
128 vllm_config = get_current_vllm_config()
129
130 cache_config = vllm_config.cache_config
131 block_size: int | None
132 if cache_config is not None and cache_config.user_specified_block_size:
133 block_size = cache_config.block_size
134 else:
135 block_size = None
136
137 kv_transfer_config = vllm_config.kv_transfer_config
138 use_kv_connector = (
139 kv_transfer_config is not None and kv_transfer_config.is_kv_transfer_instance
140 )
141
142 speculative_config = vllm_config.speculative_config
143 use_adaptive_verification = (
144 speculative_config is not None
145 and speculative_config.enable_adaptive_verification
146 )
147 if use_adaptive_verification:
148 from vllm.compilation.backends import model_tag
149
150 # The drafter always runs full-length blocks; only the verifier sees
151 # the trimmed, device-decided query lengths.
152 use_adaptive_verification = model_tag != "dspark_head"
153
154 attn_type = attn_type or AttentionType.DECODER
155 attn_selector_config = AttentionSelectorConfig(
156 head_size=head_size,
157 dtype=dtype,
158 kv_cache_dtype=cast(CacheDType | None, kv_cache_dtype),
159 block_size=block_size,
160 use_mla=use_mla,
161 has_sink=has_sink,
162 use_sparse=use_sparse,
163 use_mm_prefix=use_mm_prefix,
164 use_per_head_quant_scales=use_per_head_quant_scales,
165 attn_type=attn_type,
166 has_sliding_window=has_sliding_window,
167 use_non_causal=vllm_config.attention_config.use_non_causal,
168 use_batch_invariant=envs.VLLM_BATCH_INVARIANT,
169 use_kv_connector=use_kv_connector,
170 use_pcp=vllm_config.parallel_config.prefill_context_parallel_size > 1,
171 use_adaptive_verification=use_adaptive_verification,
172 use_dcp=vllm_config.parallel_config.decode_context_parallel_size > 1,
173 use_rswa=(
174 vllm_config.model_config is not None
175 and vllm_config.model_config.rswa_window is not None
176 ),
177 )
178
179 # A per-KV-group override (keyed by KVCacheSpecKind) takes precedence over
180 # the global backend; kinds not present in the map fall back to it.
181 attention_config = vllm_config.attention_config
182 backend = attention_config.backend
183 if attention_config.backend_per_kind:
184 kind = get_attn_spec_kind(
185 use_mla=use_mla,
186 has_sliding_window=has_sliding_window,
187 attn_type=attn_type,
188 )
189 backend = attention_config.backend_per_kind.get(kind.value, backend)
190
191 # The KV cache layout is resolved across all of the model's backends at once
192 # in get_kv_cache_spec(); a single selection cannot see its peers.
193 return _cached_get_attn_backend(
194 backend=backend,
195 attn_selector_config=attn_selector_config,
196 num_heads=num_heads,
197 )
198
199
200@cache
201def _cached_get_attn_backend(
202 backend,
203 attn_selector_config: AttentionSelectorConfig,
204 num_heads: int | None = None,
205) -> type[AttentionBackend]:
206 from vllm.platforms import current_platform
207
208 attention_cls = current_platform.get_attn_backend_cls(
209 backend,
210 attn_selector_config=attn_selector_config,
211 num_heads=num_heads,
212 )
213 if not attention_cls:
214 raise ValueError(
215 f"Invalid attention backend for {current_platform.device_name}"
216 )
217 backend = resolve_obj_by_qualname(attention_cls)
218 return backend

221def get_mamba_attn_backend(
222 mamba_type: MambaAttentionBackendEnum,
223) -> type[AttentionBackend]:
224 """Select which mamba attention backend to use and lazily import it."""
225 return _cached_get_mamba_attn_backend(mamba_type)

228@cache
229def _cached_get_mamba_attn_backend(
230 mamba_type: MambaAttentionBackendEnum,
231) -> type[AttentionBackend]:
232 assert mamba_type and isinstance(mamba_type, MambaAttentionBackendEnum)
233
234 mamba_attn_backend = mamba_type.get_class()
235 if envs.VLLM_BATCH_INVARIANT and not mamba_attn_backend.supports_batch_invariance():
236 raise RuntimeError(
237 "VLLM batch_invariant mode is not supported for "
238 f"{mamba_attn_backend.get_name()}."
239 )
240 return mamba_attn_backend
