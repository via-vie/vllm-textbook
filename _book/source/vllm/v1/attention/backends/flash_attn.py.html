529def _maybe_symmetrize_window(
530 window: tuple[int, int] | None,
531 causal: bool | torch.Tensor,
532) -> tuple[int, int] | None:
533 """Make a causal sliding window ``(w, 0)`` symmetric ``(w, w)`` when attention
534 is non-causal, so bidirectional queries attend in both directions. Leaves
535 full-attention ``(-1, -1)`` and already-symmetric windows untouched.
536 """
537 non_causal = isinstance(causal, torch.Tensor) or causal is False
538 if window is not None and window[0] >= 0 and window[1] == 0 and non_causal:
539 return (window[0], window[0])
540 return window

1180 Args:

1182 key: shape = [num_tokens, num_kv_heads, head_size]
1183 value: shape = [num_tokens, num_kv_heads, head_size]

1208 # IMPORTANT!
1209 # NOTE(woosuk): With piece-wise CUDA graphs, this method is executed in
1210 # eager-mode PyTorch. Thus, we need to be careful about any CPU overhead
1211 # in this method. For example, `view` and `slice` (or `[:n]`) operations
1212 # are surprisingly slow even in the case they do not invoke any GPU ops.
1213 # Minimize the PyTorch ops in this method as much as possible.

1215 # performance to make sure it does not introduce any overhead.

1722@functools.cache
1723def _make_mm_prefix_mask_mod(
1724 sliding_window: int = 0,
1725 sliding_window_left: int | None = None,
1726):
1727 """Build a CuTE-DSL mask_mod implementing
1728 ``(causal AND sliding_window) OR mm_prefix``.
1729
1730 Cached so identical ``(sliding_window, sliding_window_left)`` reuse the
1731 same function object. FA4's ``hash_callable`` mixes ``repr()`` of closure
1732 cells into the compile key; the nested ``_load_q_range`` would otherwise
1733 get a new address each call and force a full JIT recompile every forward.
1734
1735 The FA4 kernel passes *local* ``q_idx`` (0-based within the current
1736 prefill chunk) while ``kv_idx`` is absolute (0-based over the full
1737 KV cache). We recover the absolute Q position via
1738 ``q_abs = q_idx + seqlen_k - seqlen_q`` (the context-length offset)
1739 so that causal, sliding-window, and mm_prefix range comparisons all
1740 use consistent absolute positions. This matches the Triton
1741 reference path (``compute_kv_seq_mask``).
1742
1743 ``aux_tensors[0]`` holds the absolute ``[start, end]`` bounds of the
1744 mm_prefix range containing each scheduled query token (``(-1, -1)`` when
1745 none), and ``aux_tensors[1]`` is ``cu_seqlens_q``, used to turn the local
1746 ``q_idx`` into a packed row index. Because mm_prefix ranges never overlap,
1747 ``r_start <= kv_idx <= r_end`` is exactly "query and key share a range", so
1748 no key-side lookup is needed and ``kv_idx`` is never used as an index.
1749 The ``(-1, -1)`` sentinel falls out for free: ``kv_idx <= -1`` is false for
1750 every valid key.
1751
1752 ``sliding_window_left`` enforces the sliding window on the causal
1753 term (None = full causal, no window). ``sliding_window`` clamps the
1754 bidirectional block to the window (0 = unclamped; >0 = Gemma4 local
1755 layers via ``mm_prefix_clamp_sliding_window``).
1756 """
1757 import cutlass
1758 import cutlass.cute as cute
1759 from cutlass import Int32 # type: ignore[attr-defined]
1760
1761 from vllm.vllm_flash_attn.cute.utils import ( # type: ignore[import-untyped]
1762 scalar_to_ssa,
1763 ssa_to_scalar,
1764 )
1765
1766 @cute.jit
1767 def _load_q_range(q_idx, seqlen_info, aux_tensors, batch_idx):
1768 """Load the mm_prefix range bounds for this query row.
1769
1770 Both loads depend only on the query index, so they hoist out of the
1771 unrolled per-element mask loop. ``ssa_to_scalar`` reads lane 0, so one
1772 call must not span query rows (hence the ``__vec_size__`` pin below).
1773 Clamping keeps ``token_idx`` in bounds for partial tiles and padded
1774 (``seqlen_q == 0``) rows; neither produces output.
1775 """
1776 q_ranges = aux_tensors[0]
1777 cu_seqlens_q = aux_tensors[1]
1778 b = batch_idx[0]
1779 q_local = cutlass.min(ssa_to_scalar(q_idx), seqlen_info.seqlen_q - Int32(1))

1831def _make_rswa_mask_mod():
1832 """Build a CuTE-DSL mask_mod for Reference Sliding Window Attention (R-SWA).
1833
1834 FA4 varlen + paged-KV convention (verified from cute/mask.py apply_mask):
1835 q_idx = LOCAL query-token offset (0 .. seqlen_q - 1) within this sequence.
1836 kv_idx = LOCAL KV-token position (0 .. seqlen_k - 1) within this sequence.
1837
1838 To recover the ABSOLUTE token position (needed for causal and the sliding
1839 window distance), use the standard offset:
1840 abs_q = q_idx + (seqlen_k - seqlen_q)
1841
1842 R-SWA keep condition:
1843 abs_q >= kv_idx (causal: KV at or before the query)
1844 AND (kv_idx < prefix_len (global prefix is always visible)
1845 OR abs_q - kv_idx < window) (generated tokens: sliding window)
1846
1847 aux_tensors[0]: prefix_lens [num_reqs] int32 — per-request prefill length.
1848 aux_tensors[1]: rswa_window [1] int32 — decode sliding window size.
1849
1850 use_fast_sampling=True lets FA4 skip fully-masked KV blocks (gap blocks)
1851 without loading their data.
1852 """
1853 import cutlass.cute as cute
1854 from cutlass import Int32 # type: ignore[attr-defined]
1855
1856 from vllm.vllm_flash_attn.cute.utils import ( # type: ignore[import-untyped]
1857 scalar_to_ssa,
1858 )
1859
1860 @cute.jit
1861 def rswa_mask_mod(
1862 batch_idx: cute.TensorSSA,
1863 head_idx: cute.TensorSSA,
1864 q_idx: cute.TensorSSA,
1865 kv_idx: cute.TensorSSA,
1866 seqlen_info,
1867 aux_tensors,
1868 ):
1869 b = batch_idx[0]
1870 prefix_len = scalar_to_ssa(aux_tensors[0][b], Int32)
1871 window = scalar_to_ssa(aux_tensors[1][0], Int32)
1872 # Convert local q offset to absolute token position.
1873 offset = scalar_to_ssa(seqlen_info.seqlen_k - seqlen_info.seqlen_q, Int32)
1874 abs_q = q_idx + offset
1875 causal = kv_idx <= abs_q
1876 in_prefix = kv_idx < prefix_len
1877 in_window = (abs_q - kv_idx) < window
1878 return causal & (in_prefix | in_window)
1879
1880 rswa_mask_mod.use_fast_sampling = True
1881 return rswa_mask_mod

1884def use_cascade_attention(
1885 common_prefix_len: int,
1886 query_lens: np.ndarray,
1887 num_query_heads: int,
1888 num_kv_heads: int,
1889 use_alibi: bool,
1890 use_sliding_window: bool,
1891 use_local_attention: bool,
1892 num_sms: int,
1893 dcp_world_size: int,
1894) -> bool:
1895 """Decide whether to use cascade attention.
1896
1897 This function 1) checks whether cascade attention is supported with the
1898 given configuration, and 2) heuristically decides whether using cascade
1899 attention can improve performance.
1900 """
1901 # Too short common prefix. Probably not worth using cascade attention.
1902 # We use an arbitrary threshold of 256 tokens. TODO: Tune this threshold.
1903 # NOTE(woosuk): This is the common case. We should return False as soon as
1904 # possible to avoid any unnecessary computation.
1905 if common_prefix_len < 256:
1906 return False
1907 # Cascade attention is currently not supported with these variants.
1908 if use_alibi or use_sliding_window or use_local_attention:
1909 return False
1910 # Too few queries. Probably not worth using cascade attention.
1911 # We use an arbitrary threshold of 8 queries. TODO: Tune this threshold.
1912 num_reqs = len(query_lens)
1913 if num_reqs < 8:
1914 return False
1915 # disable cascade attention for DCP
1916 if dcp_world_size > 1:
1917 return False
1918
1919 # Heuristics to decide whether using cascade attention is beneficial.
1920 # 1. When FlashDecoding is not used for normal attention, cascade attention
1921 # is likely to be faster since it saves memory bandwidth.
1922 num_queries_per_kv = num_query_heads // num_kv_heads
1923 # The criteria for using FlashDecoding can be found in the following link:
1924 # https://github.com/vllm-project/flash-attention/blob/96266b1111111f3d11aabefaf3bacbab6a89d03c/csrc/flash_attn/flash_api.cpp#L535
1925 use_flash_decoding = (
1926 num_queries_per_kv > 1
1927 and not use_sliding_window
1928 and not use_alibi
1929 and np.all(query_lens == 1)
1930 )
1931 if not use_flash_decoding:
1932 # Use cascade attention.
1933 return True
1934
1935 # 2. When FlashDecoding is used for normal attention, it is not clear
1936 # whether cascade attention is beneficial, because FlashDecoding can
1937 # launch more CTAs than cascade attention.
1938 # We use a simple performance model to compare the two methods.
1939 # NOTE(woosuk): The performance model is very rough and may not be
1940 # accurate.
1941 num_tokens = num_reqs
1942 # NOTE(woosuk): These are default tile sizes. flash-attn might use
1943 # different tile sizes (e.g., 64 or 256) depending on the configuration.
1944 q_tile_size = 128
1945 kv_tile_size = 128
1946 num_prefix_tiles = cdiv(common_prefix_len, kv_tile_size)
1947
1948 cascade_ctas = num_query_heads * cdiv(num_tokens, q_tile_size)
1949 cascade_waves = cdiv(cascade_ctas, num_sms)
1950 cascade_time = cascade_waves * num_prefix_tiles
1951
1952 flash_decoding_ctas = (
1953 num_reqs * num_kv_heads * cdiv(num_queries_per_kv, q_tile_size)
1954 )
1955 flash_decoding_ctas *= num_prefix_tiles
1956 flash_decoding_time = cdiv(flash_decoding_ctas, num_sms)
1957
1958 # Use cascade attention if it is faster than FlashDecoding.
1959 return cascade_time < flash_decoding_time
