384@dataclass
385class CommonAttentionMetadata:
386 """
387 Per-batch attention metadata, shared across layers and backends.
388 AttentionMetadataBuilder instances use it to construct per-layer metadata.
389
390 For many of the tensors we keep both GPU and CPU versions.
391 """
392
393 query_start_loc: torch.Tensor
394 query_start_loc_cpu: torch.Tensor
395 """(batch_size + 1,), the start location of each request in query Tensor"""
396
397 seq_lens: torch.Tensor
398 """(batch_size,), the number of computed tokens for each request"""
399
400 num_reqs: int
401 """Number of requests"""
402 # TODO(lucas): rename to num_tokens since it may be padded and this is misleading
403 num_actual_tokens: int
404 """Total number of tokens in batch"""
405 max_query_len: int
406 """Longest query in batch"""
407 max_seq_len: int
408 """Longest context length (may be an upper bound)"""
409
410 block_table_tensor: torch.Tensor
411 slot_mapping: torch.Tensor
412
413 causal: bool | torch.Tensor = True
414
415 # Needed by FastPrefillAttentionBuilder
416 logits_indices_padded: torch.Tensor | None = None
417 num_logits_indices: int | None = None
418 max_logits_per_req: int | None = None
419
420 # Needed by CrossAttentionBuilder
421 encoder_seq_lens: torch.Tensor | None = None
422 encoder_seq_lens_cpu: np.ndarray | None = None
423
424 dcp_local_seq_lens: torch.Tensor | None = None
425 """Sequence lengths of the local rank in decode context parallelism world"""
426
427 dcp_local_seq_lens_cpu_upper_bound: torch.Tensor | None = None
428 """(batch_size,) CPU upper bound on dcp_local_seq_lens. Under PCP+DCP it
429 holds, on every row, the largest shard any DCP rank has of the row's
430 whole request, identical on every PCP rank, so the sparse backends pad
431 their KV gather to it."""
432
433 positions: torch.Tensor | None = None
434 """(num_actual_tokens,) token positions. Optional; set when the caller
435 has positions available so that builders can pre-compute position-dependent
436 sparse metadata for DeepSeek V4 C128A layers."""
437
438 is_prefilling: torch.Tensor | None = None
439 """(batch_size,) bool tensor: True if request is still in prefill phase
440 (num_computed_tokens < num_prompt_tokens). Used by some backends to
441 distinguish actual decodes from short extends."""
442
443 seq_lens_cpu_upper_bound: torch.Tensor | None = None
444 """(batch_size,) CPU upper bound on seq_lens. Precise for prefill rows
445 and for all rows outside async spec decode; optimistic for async-spec
446 decode rows (assumes every draft was accepted). Not safe for kernels
447 that need exact per-row context lengths on decode rows."""
448
449 req_idx: np.ndarray | None = None
450 """(batch_size,) index of each row's request in the runner's request
451 table. Rows of one request are adjacent, so equal neighbours are PCP
452 chunks sharing one KV context."""
453
454 mm_req_doc_ranges: dict[int, list[tuple[int, int]]] | None = None
455 """PrefixLM bidirectional ranges for multimodal tokens. Maps
456 request index to list of (start, end) token position ranges
457 where bidirectional attention should apply. None for text-only
458 batches or non-PrefixLM models. A request's ranges must not overlap."""
459
460 rswa_prefix_lens: torch.Tensor | None = None
461 """(batch_size,) per-request prefix length (prompt/image token count) for
462 Reference Sliding Window Attention (R-SWA). Tokens with logical index below
463 this stay globally visible; later (generated) tokens additionally see a
464 fixed sliding window. None disables R-SWA. The attention backend copies this
465 into its own persistent buffer and reads ``rswa_window`` from model config."""
466
467 replayssm_decode_base_cpu: torch.Tensor | None = None
468 """(batch_size,) CPU ring origin for Mamba2 ReplaySSM decode: num_computed
469 at the current decode run's last full-state write. write_pos counts from
470 here, so a preemption-resumed request re-anchors past the prompt boundary."""
471
472 _num_computed_tokens_cache: torch.Tensor | None = None
473 _token_to_req_indices_cache: torch.Tensor | None = None
474
475 def batch_size(self) -> int:
476 return self.seq_lens.shape[0]
477
478 def naive_query_lens(self) -> torch.Tensor:
479 """Naive because it assumes that query ends where the next query starts."""
480 return self.query_start_loc[1:] - self.query_start_loc[:-1]
481
482 def replace(self, **kwargs) -> "CommonAttentionMetadata":
483 return replace(self, **kwargs)
484
485 def compute_num_computed_tokens(self) -> torch.Tensor:
486 """Compute num_computed_tokens on device (seq_lens - query_lens)."""
487 if self._num_computed_tokens_cache is None:
488 query_lens = self.query_start_loc[1:] - self.query_start_loc[:-1]
489 self._num_computed_tokens_cache = self.seq_lens - query_lens
490 return self._num_computed_tokens_cache
491
492 def token_to_req_indices(self, buffer: torch.Tensor) -> torch.Tensor:
493 """Build or reuse the per-token request index mapping."""
494 num_tokens = self.num_actual_tokens
495 if self._token_to_req_indices_cache is not None:
496 assert self._token_to_req_indices_cache.device == buffer.device
497 assert self._token_to_req_indices_cache.dtype == torch.int32
498 assert self._token_to_req_indices_cache.shape[0] >= num_tokens
499 return self._token_to_req_indices_cache[:num_tokens]
500
501 # Built from the device query_start_loc: adaptive verification decides the
502 # per-request draft split on device, so the CPU copy carries the right total
503 # but not the right per-request boundaries. Padding requests have a query
504 # length of zero and are skipped by the device boundary search.
505 num_mapped_tokens = int(self.query_start_loc_cpu[-1])
506 from vllm.v1.attention.ops.metadata import _token_request_mapping_kernel
507
508 num_output_tokens = max(num_mapped_tokens, num_tokens)
509 assert buffer.shape[0] >= num_output_tokens
510 _token_request_mapping_kernel[((num_output_tokens + 255) // 256,)](
511 self.query_start_loc,
512 buffer,
513 self.query_start_loc.shape[0] - 1,
514 num_mapped_tokens,
515 num_output_tokens,
516 num_warps=4,
517 )
518 self._token_to_req_indices_cache = buffer[:num_output_tokens]
519 return self._token_to_req_indices_cache[:num_tokens]
520
521 # TODO(lucas): remove once we have FULL-CG spec-decode support
522 def unpadded(
523 self, num_actual_tokens: int, num_actual_reqs: int
524 ) -> "CommonAttentionMetadata":
525 maybe_slice_reqs = lambda x: x[:num_actual_reqs] if x is not None else None
526 return CommonAttentionMetadata(
527 query_start_loc=self.query_start_loc[: num_actual_reqs + 1],
528 query_start_loc_cpu=self.query_start_loc_cpu[: num_actual_reqs + 1],
529 seq_lens=self.seq_lens[:num_actual_reqs],
530 num_reqs=num_actual_reqs,
531 num_actual_tokens=num_actual_tokens,
532 max_query_len=self.max_query_len,
533 max_seq_len=self.max_seq_len,
534 block_table_tensor=self.block_table_tensor[:num_actual_reqs],
535 slot_mapping=self.slot_mapping[:num_actual_tokens],
536 causal=self.causal[:num_actual_reqs]
537 if isinstance(self.causal, torch.Tensor)
538 else self.causal,
539 logits_indices_padded=self.logits_indices_padded,
540 num_logits_indices=self.num_logits_indices,
541 max_logits_per_req=self.max_logits_per_req,
542 encoder_seq_lens=maybe_slice_reqs(self.encoder_seq_lens),
543 encoder_seq_lens_cpu=maybe_slice_reqs(self.encoder_seq_lens_cpu),
544 dcp_local_seq_lens=maybe_slice_reqs(self.dcp_local_seq_lens),
545 dcp_local_seq_lens_cpu_upper_bound=maybe_slice_reqs(
546 self.dcp_local_seq_lens_cpu_upper_bound
547 ),
548 seq_lens_cpu_upper_bound=maybe_slice_reqs(self.seq_lens_cpu_upper_bound),
549 is_prefilling=maybe_slice_reqs(self.is_prefilling),
550 req_idx=maybe_slice_reqs(self.req_idx),
551 rswa_prefix_lens=maybe_slice_reqs(self.rswa_prefix_lens),
552 replayssm_decode_base_cpu=maybe_slice_reqs(self.replayssm_decode_base_cpu),
553 )

766class AttentionImplBase(ABC, Generic[T]):
767 """Base class for attention implementations.
768
769 Contains common attributes and initialization logic shared by both
770 standard AttentionImpl and MLAAttentionImpl. Does not define a forward
771 method - subclasses define their own forward interfaces.
772 """
773
774 # Whether this impl uses a sparse (top-k) attention path. Used by MLA to
775 # route between the dense-MHA prefill and sparse-MQA paths.
776 is_sparse: ClassVar[bool] = False
777
778 # Whether this impl provides a dense-MHA prefill path (forward_mha). Sparse
779 # impls without one run the top-k MQA path for all requests.
780 supports_dense_mha_prefill: ClassVar[bool] = True
781
782 # Required attributes that all impls should have
783 num_heads: int
784 head_size: int
785 scale: float
786
787 # Whether the attention impl can return the softmax lse for decode.
788 # Some features like decode context parallelism require the softmax lse.
789 can_return_lse_for_decode: bool = False
790
791 # Base of the logarithm used by this backend when returning softmax lse.
792 # True => natural log (lse = ln(sum(exp(qk))))
793 # -- e.g. Triton MLA, FlashAttention, FlashMLA, Cutlass MLA
794 # False => base 2 (lse = log2(sum(exp(qk))))
795 # -- e.g. FlashInfer trtllm-gen MLA
796 # The DCP combine kernel (cp_lse_ag_out_rs / dcp_a2a_lse_reduce in
797 # vllm/v1/attention/ops/dcp.py) branches on this via its IS_BASE_E
798 # constexpr; getting it wrong silently corrupts the cross-shard
799 # softmax denominator.
800 lse_base_on_e: bool = True
801
802 # Whether the attention impl supports Prefill Context Parallelism.
803 supports_pcp: bool = False
804 # Whether the attention impl supports Decode Context Parallelism.
805 supports_dcp: bool = False
806 # Whether the attention impl(or ops) supports MTP
807 # when cp_kv_cache_interleave_size > 1
808 supports_mtp_with_cp_non_trivial_interleave_size: bool = False
809
810 # some attention backends might not always want to return lse
811 # even if they can return lse (for efficiency reasons)
812 need_to_return_lse_for_decode: bool = False
813
814 # Whether this attention implementation supports pre-quantized query input.
815 # When True, the attention layer will quantize queries before passing them
816 # to this backend, allowing torch.compile to fuse the quantization with
817 # previous operations. This is typically supported when using FP8 KV cache
818 # with compatible attention kernels (e.g., TRT-LLM).
819 # Subclasses should set this in __init__.
820 # TODO add support to more backends:
821 # https://github.com/vllm-project/vllm/issues/25584
822 supports_quant_query_input: bool = False
823
824 dcp_world_size: int
825 dcp_rank: int
826
827 pcp_world_size: int
828 pcp_rank: int
829
830 total_cp_world_size: int
831 total_cp_rank: int
832
833 def __new__(cls, *args, **kwargs):
834 # use __new__ so that all subclasses will call this
835 self = super().__new__(cls)
836 try:
837 from vllm.distributed.parallel_state import get_dcp_group
838
839 self.dcp_world_size = get_dcp_group().world_size
840 self.dcp_rank = get_dcp_group().rank_in_group
841 except AssertionError:
842 # DCP might not be initialized in testing
843 self.dcp_world_size = 1
844 self.dcp_rank = 0
845 try:
846 from vllm.distributed.parallel_state import get_pcp_group
847
848 self.pcp_world_size = get_pcp_group().world_size
849 self.pcp_rank = get_pcp_group().rank_in_group
850 except AssertionError:
851 self.pcp_world_size = 1
852 self.pcp_rank = 0
853 self.total_cp_world_size = self.dcp_world_size
854 self.total_cp_rank = self.dcp_rank
855
856 self.need_to_return_lse_for_decode = (
857 self.dcp_world_size > 1 and self.can_return_lse_for_decode
858 )
859 return self
