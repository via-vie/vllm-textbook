15class PagedAttention:
16 @staticmethod
17 def split_kv_cache(
18 kv_cache: torch.Tensor,
19 num_kv_heads: int,
20 head_size: int,
21 ) -> tuple[torch.Tensor, torch.Tensor]:
22 x = 16 // kv_cache.element_size()
23 num_blocks = kv_cache.shape[1]
24
25 key_cache = kv_cache[0]
26 key_cache = key_cache.view(num_blocks, num_kv_heads, head_size // x, -1, x)
27 value_cache = kv_cache[1]
28 value_cache = value_cache.view(num_blocks, num_kv_heads, head_size, -1)
29 return key_cache, value_cache
