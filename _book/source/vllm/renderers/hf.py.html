79# Cache of `tokenizer -> prompt_embeds placeholder token ID`. Keyed by the
80# tokenizer object (not `id(tokenizer)`) so a fresh tokenizer landing at a
81# recycled memory address can't pick up a stale tid. Entries evict atomically
82# with the tokenizer's garbage-collection.
83_PROMPT_EMBEDS_PLACEHOLDER_TOKEN_ID_CACHE: Final[
84 weakref.WeakKeyDictionary[HfTokenizer, int]
85] = weakref.WeakKeyDictionary()
86_PROMPT_EMBEDS_PLACEHOLDER_TOKEN_ID_ERROR: Final[str] = (
87 "Expected {token!r} to tokenize to exactly 1 token, got {num_ids} ({ids!r})."
88)
89_PROMPT_EMBEDS_PLACEHOLDER_SPAN_MISMATCH_ERROR: Final[str] = (
90 "Expected {expected} prompt_embeds placeholder spans in the "
91 "tokenized prompt, found {actual}."
92)
93_MISSING_PROMPT_TOKEN_IDS_ERROR: Final[str] = (
94 "Expected prompt_token_ids in rendered prompt when prompt_embeds "
95 "are present. This indicates the chat template was invoked with "
96 "tokenize=False."
97)
98_TOKENIZE_OVERRIDE_WARNING: Final[str] = (
99 "Overriding `tokenize=False` to `True` because `prompt_embeds` "
100 "post-processing requires tokenized IDs."
101)

104def _ensure_prompt_embeds_placeholder_token(tokenizer: HfTokenizer) -> int:
105 """Register `PROMPT_EMBEDS_PLACEHOLDER_TOKEN` as a special token and return
106 its token ID."""
107 cached = _PROMPT_EMBEDS_PLACEHOLDER_TOKEN_ID_CACHE.get(tokenizer)
108 if cached is not None:
109 return cached
110
111 tokenizer.add_special_tokens(
112 {"additional_special_tokens": [PROMPT_EMBEDS_PLACEHOLDER_TOKEN]}
113 )
114
115 ids = tokenizer.encode(PROMPT_EMBEDS_PLACEHOLDER_TOKEN, add_special_tokens=False)
116 if len(ids) != 1:
117 raise RuntimeError(
118 _PROMPT_EMBEDS_PLACEHOLDER_TOKEN_ID_ERROR.format(
119 token=PROMPT_EMBEDS_PLACEHOLDER_TOKEN,
120 num_ids=len(ids),
121 ids=ids,
122 )
123 )
124
125 token_id = ids[0]
126 _PROMPT_EMBEDS_PLACEHOLDER_TOKEN_ID_CACHE[tokenizer] = token_id
127 return token_id

130def _build_prompt_embeds_updates(
131 prompt_embeds_tensors: Sequence[torch.Tensor],
132 placeholder_token_id: int,
133) -> MultiModalPromptUpdates:
134 """Build `MultiModalPromptUpdates` for `prompt_embeds` expansion.
135
136 Each tensor produces a `PromptReplacement` that maps
137 `[placeholder_token_id]` -> `[placeholder_token_id] x N`
138 (where `N = tensor.shape[0]`).
139 """
140 updates: list[Sequence[ResolvedPromptUpdate]] = []
141 for i, tensor in enumerate(prompt_embeds_tensors):
142 update = PromptReplacement(
143 modality="prompt_embeds",
144 target=[placeholder_token_id],
145 replacement=[placeholder_token_id] * tensor.shape[0],
146 )
147 updates.append([update.resolve(item_idx=i)])
148 return {"prompt_embeds": updates}

151def _expand_prompt_embeds_placeholders(
152 token_ids: list[int],
153 mm_prompt_updates: MultiModalPromptUpdates,
154) -> list[int]:
155 """Expand each 1-token `prompt_embeds` sentinel into an N-token span.
156
157 Uses `apply_token_matches`. Each single placeholder token in
158 `token_ids` is replaced with a consecutive span of
159 `tensor.shape[0]` copies, following tensors in order.
160 """
161 expanded, _ = apply_token_matches(token_ids, mm_prompt_updates)
162 return expanded

165def _build_prompt_embeds_positions(
166 token_ids: list[int],
167 num_tensors: int,
168 mm_prompt_updates: MultiModalPromptUpdates,
169) -> list[tuple[int, int]]:
170 """Locate each prompt_embeds placeholder span in `token_ids`.
171
172 Expects `token_ids` to already contain expanded N-token spans.
173 Returns `[(start_idx, length), ...]` aligned with the tensors.
174 """
175 placeholders = find_mm_placeholders(token_ids, mm_prompt_updates)
176 features = placeholders.get("prompt_embeds", [])
177
178 if len(features) != num_tensors:
179 raise ValueError(
180 _PROMPT_EMBEDS_PLACEHOLDER_SPAN_MISMATCH_ERROR.format(
181 expected=num_tensors,
182 actual=len(features),
183 )
184 )
185
186 return [(f.start_idx, f.length) for f in features]

189def _build_mixed_prompt_embeds(
190 token_ids: list[int],
191 prompt_embeds_tensors: Sequence[torch.Tensor],
192 positions: list[tuple[int, int]],
193) -> tuple[torch.Tensor, list[bool]]:
194 """Build the full-length `prompt_embeds` tensor and the `is_token_ids`
195 mask aligned to `token_ids`."""
196 total_len = len(token_ids)
197 hidden_size = prompt_embeds_tensors[0].shape[1]
198 dtype = prompt_embeds_tensors[0].dtype
199
200 full_embeds = torch.zeros(total_len, hidden_size, dtype=dtype)
201 is_token_ids = torch.ones(total_len, dtype=torch.bool)
202
203 for (start, length), tensor in zip(positions, prompt_embeds_tensors, strict=True):
204 full_embeds[start : start + length] = tensor
205 is_token_ids[start : start + length] = False
206
207 return full_embeds, is_token_ids.tolist()

211"""
212Used in `_try_get_processor_chat_template` to avoid calling
213`cached_get_processor` again if the processor fails to be loaded.

219def _try_get_processor_chat_template(
220 tokenizer: HfTokenizer,
221 *,
222 revision: str | None,
223 code_revision: str | None,
224 trust_remote_code: bool,
225) -> str | None:
226 cache_key = (
227 tokenizer.name_or_path,
228 revision,
229 code_revision,
230 trust_remote_code,
231 )
232 if cache_key in _PROCESSOR_CHAT_TEMPLATES:
233 return _PROCESSOR_CHAT_TEMPLATES[cache_key]
234
235 from transformers import ProcessorMixin, PythonBackend, TokenizersBackend
236
237 try:
238 processor = cached_get_processor(
239 tokenizer.name_or_path,
240 processor_cls=(PythonBackend, TokenizersBackend, ProcessorMixin),
241 revision=revision,
242 code_revision=code_revision,
243 trust_remote_code=trust_remote_code,
244 )
245 if (
246 isinstance(processor, ProcessorMixin)
247 and hasattr(processor, "chat_template")
248 and (chat_template := processor.chat_template) is not None
249 ):
250 _PROCESSOR_CHAT_TEMPLATES[cache_key] = chat_template
251 return chat_template
252 except Exception:
253 logger.debug(
254 "Failed to load AutoProcessor chat template for %s",
255 tokenizer.name_or_path,
256 exc_info=True,
257 )
258
259 _PROCESSOR_CHAT_TEMPLATES[cache_key] = None
260 return None

263def resolve_chat_template(
264 tokenizer: HfTokenizer,
265 chat_template: str | None,
266 tools: list[dict[str, Any]] | None,
267 *,
268 model_config: ModelConfig,
269) -> str | None:
270 # 1st priority: The given chat template
271 if chat_template is not None:
272 # Resolve template names (e.g. "tool_use") to actual Jinja content
273 # so that downstream kwargs detection can parse template variables.
274 return tokenizer.get_chat_template(chat_template, tools=tools)
275
276 # 2nd priority: AutoProcessor chat template, unless tool calling is enabled
277 if tools is None:
278 chat_template = _try_get_processor_chat_template(
279 tokenizer,
280 revision=model_config.revision,
281 code_revision=model_config.code_revision,
282 trust_remote_code=model_config.trust_remote_code,
283 )
284 if chat_template is not None:
285 return chat_template
286
287 # 3rd priority: AutoTokenizer chat template
288 try:
289 return tokenizer.get_chat_template(chat_template, tools=tools)
290 except Exception:
291 logger.debug(
292 "Failed to load AutoTokenizer chat template for %s",
293 tokenizer.name_or_path,
294 exc_info=True,
295 )
296
297 # 4th priority: Predefined fallbacks
298 path = get_chat_template_fallback_path(
299 model_type=model_config.hf_config.model_type,
300 tokenizer_name_or_path=tokenizer.name_or_path,
301 )
302 if path is not None:
303 logger.info_once(
304 "Loading chat template fallback for %s as there isn't one "
305 "defined on HF Hub.",
306 tokenizer.name_or_path,
307 )
308 chat_template = load_chat_template(path)
309 else:
310 logger.debug_once(
311 "There is no chat template fallback for %s", tokenizer.name_or_path
312 )
313
314 return chat_template

317def _is_var_access(node: jinja2.nodes.Node, varname: str) -> bool:
318 if isinstance(node, jinja2.nodes.Name):

321 return False

324def _is_attr_access(node: jinja2.nodes.Node, varname: str, key: str) -> bool:
325 if isinstance(node, jinja2.nodes.Getitem):
326 return (
327 _is_var_access(node.node, varname)
328 and isinstance(node.arg, jinja2.nodes.Const)
329 and node.arg.value == key
330 )
331
332 if isinstance(node, jinja2.nodes.Getattr):
333 return _is_var_access(node.node, varname) and node.attr == key
334
335 return False

338def _is_var_or_elems_access(
339 node: jinja2.nodes.Node,
340 varname: str,
341 key: str | None = None,
342) -> bool:
343 if isinstance(node, jinja2.nodes.Filter):
344 return node.node is not None and _is_var_or_elems_access(
345 node.node, varname, key
346 )
347 if isinstance(node, jinja2.nodes.Test):
348 return _is_var_or_elems_access(node.node, varname, key)
349
350 if isinstance(node, jinja2.nodes.Getitem) and isinstance(
351 node.arg, jinja2.nodes.Slice
352 ):
353 return _is_var_or_elems_access(node.node, varname, key)
354
355 return _is_attr_access(node, varname, key) if key else _is_var_access(node, varname)

358def _iter_nodes_assign_var_or_elems(root: jinja2.nodes.Node, varname: str):
359 # Global variable that is implicitly defined at the root
360 yield root, varname
361
362 # Iterative BFS
363 related_varnames = deque([varname])
364 while related_varnames:
365 related_varname = related_varnames.popleft()
366
367 for assign_ast in root.find_all(jinja2.nodes.Assign):
368 lhs = assign_ast.target
369 rhs = assign_ast.node
370
371 if _is_var_or_elems_access(rhs, related_varname):
372 assert isinstance(lhs, jinja2.nodes.Name)
373 yield assign_ast, lhs.name
374
375 # Avoid infinite looping for self-assignment
376 if lhs.name != related_varname:
377 related_varnames.append(lhs.name)
378

380# NOTE: The proper way to handle this is to build a CFG so that we can handle
381# the scope in which each variable is defined, but that is too complicated
382def _iter_nodes_assign_messages_item(root: jinja2.nodes.Node):
383 messages_varnames = [
384 varname for _, varname in _iter_nodes_assign_var_or_elems(root, "messages")
385 ]
386
387 # Search for {%- for message in messages -%} loops
388 for loop_ast in root.find_all(jinja2.nodes.For):
389 loop_iter = loop_ast.iter
390 loop_target = loop_ast.target
391
392 for varname in messages_varnames:
393 if _is_var_or_elems_access(loop_iter, varname):
394 assert isinstance(loop_target, jinja2.nodes.Name)
395 yield loop_ast, loop_target.name
396 break

399def _iter_nodes_assign_content_item(root: jinja2.nodes.Node):
400 """Yield loops that iterate over message content or macro-bound content."""
401 message_varnames = [
402 varname for _, varname in _iter_nodes_assign_messages_item(root)
403 ]
404
405 # Track macro parameters that receive message.content as an argument.
406 # Some templates pass message.content through a macro parameter whose
407 # name is not literally "content".
408 macro_content_params_by_loop: dict[int, set[str]] = {}
409 loops_in_macros: set[int] = set()
410 for macro_node in root.find_all(jinja2.nodes.Macro):
411 macro_param_names = {arg.name for arg in macro_node.args}
412 macro_content_params: set[str] = set()
413 for call_node in root.find_all(jinja2.nodes.Call):
414 if (
415 isinstance(call_node.node, jinja2.nodes.Name)
416 and call_node.node.name == macro_node.name
417 ):
418 for i, arg in enumerate(call_node.args):
419 if i < len(macro_node.args) and any(
420 _is_var_or_elems_access(arg, varname, "content")
421 for varname in message_varnames
422 ):
423 macro_content_params.add(macro_node.args[i].name)
424 for kwarg in call_node.kwargs:
425 if (
426 isinstance(kwarg, jinja2.nodes.Keyword)
427 and kwarg.key in macro_param_names
428 and any(
429 _is_var_or_elems_access(kwarg.value, varname, "content")
430 for varname in message_varnames
431 )
432 ):
433 macro_content_params.add(kwarg.key)
434
435 for loop_ast in macro_node.find_all(jinja2.nodes.For):
436 loops_in_macros.add(id(loop_ast))
437 if macro_content_params:
438 macro_content_params_by_loop[id(loop_ast)] = macro_content_params
439
440 # Search for {%- for content in message['content'] -%} loops
441 # or {%- for item in content -%} loops
442 for loop_ast in root.find_all(jinja2.nodes.For):
443 loop_iter = loop_ast.iter
444 loop_target = loop_ast.target
445
446 for varname in message_varnames:
447 if _is_var_or_elems_access(loop_iter, varname, "content"):
448 assert isinstance(loop_target, jinja2.nodes.Name)
449 yield loop_ast, loop_target.name
450 break
451 else:
452 macro_content_params_for_loop = macro_content_params_by_loop.get(
453 id(loop_ast)
454 )
455 if (
456 isinstance(loop_iter, jinja2.nodes.Name)
457 and macro_content_params_for_loop is not None
458 and loop_iter.name in macro_content_params_for_loop
459 ):
460 assert isinstance(loop_target, jinja2.nodes.Name)
461 yield loop_ast, loop_target.name
462 continue
463
464 if (
465 id(loop_ast) not in loops_in_macros
466 and isinstance(loop_iter, jinja2.nodes.Name)
467 and loop_iter.name == "content"
468 ):
469 assert isinstance(loop_target, jinja2.nodes.Name)
470 yield loop_ast, loop_target.name

525def _consolidate_system_messages(
526 conversation: list[ConversationMessage],
527) -> list[ConversationMessage]:
528 """Merge all system messages into one at position 0.
529
530 Some chat templates (e.g. Qwen 3.6) require the system message to be the
531 very first message. After developer-to-system conversion, system messages
532 may appear at non-first positions; this merges them into a single message.
533 """
534 system_contents: list[str] = []
535 non_system: list[ConversationMessage] = []
536 needs_consolidation = False
537 for i, msg in enumerate(conversation):
538 if msg["role"] == "system":
539 if i > 0 or system_contents:
540 needs_consolidation = True
541 content = msg.get("content", "")
542 if isinstance(content, list):
543 parts = []
544 for part in content:
545 if isinstance(part, dict) and "text" in part:
546 parts.append(part["text"])
547 elif isinstance(part, str):
548 parts.append(part)
549 content = "\n".join(parts)
550 if content:
551 system_contents.append(content)
552 else:
553 non_system.append(msg)
554
555 if not needs_consolidation:
556 return conversation
557
558 merged: ConversationMessage = {
559 "role": "system",
560 "content": "\n\n".join(system_contents),
561 }
562 return [merged, *non_system]

572the chat template on every request.

574`lru_cache` cannot be used because `tools` is a list and `ModelConfig` defines
575`__eq__` without `__hash__`, so the key holds the fields `resolve_chat_template`
576selects the template by. Only the presence of `tools` is part of it: it decides
577whether the processor template is tried, and its contents never reach the AST.

584 tokenizer: HfTokenizer,
585 *,
586 model_config: ModelConfig,
587) -> ChatTemplateContentFormat:

589 chat_template,
590 tools is None,
591 tokenizer.name_or_path,
592 model_config.revision,
593 model_config.code_revision,
594 model_config.trust_remote_code,
595 )
596 if (cached_format := _CONTENT_FORMATS.get(cache_key)) is not None:

620 _CONTENT_FORMATS.clear()

689def _get_hf_base_chat_template_params() -> frozenset[str]:
690 from transformers import PythonBackend
691
692 # Get standard parameters from HuggingFace's base tokenizer class.
693 # This dynamically extracts parameters from PythonBackend's
694 # apply_chat_template method, ensuring compatibility with tokenizers
695 # that use **kwargs to receive standard parameters.
696
697 # Read signature from HF's base class - the single source of truth
698 base_sig = inspect.signature(PythonBackend.apply_chat_template)
699
700 # Exclude VAR_KEYWORD (**kwargs) and VAR_POSITIONAL (*args) placeholders
701 return frozenset(
702 p.name
703 for p in base_sig.parameters.values()
704 if p.kind
705 not in (inspect.Parameter.VAR_KEYWORD, inspect.Parameter.VAR_POSITIONAL)
706 )

709def resolve_chat_template_kwargs(
710 tokenizer: HfTokenizer,
711 chat_template: str,
712 chat_template_kwargs: dict[str, Any],
713 raise_on_unexpected: bool = True,
714) -> dict[str, Any]:
715 # We exclude chat_template from kwargs here, because
716 # chat template has been already resolved at this stage
717 unexpected_vars = {"chat_template", "tokenize"}
718 if raise_on_unexpected and (
719 unexpected_in_kwargs := unexpected_vars & chat_template_kwargs.keys()
720 ):
721 raise ValueError(
722 "Found unexpected chat template kwargs from request: "
723 f"{unexpected_in_kwargs}"
724 )
725
726 fn_kw = {
727 k
728 for k in chat_template_kwargs
729 if supports_kw(tokenizer.apply_chat_template, k, allow_var_kwargs=False)
730 }
731 template_vars = _cached_resolve_chat_template_kwargs(chat_template)
732
733 # Allow standard HF parameters even if tokenizer uses **kwargs to receive them
734 hf_base_params = _get_hf_base_chat_template_params()
735
736 accept_vars = (fn_kw | template_vars | hf_base_params) - unexpected_vars
737 return {k: v for k, v in chat_template_kwargs.items() if k in accept_vars}

740def _template_error_reason(exc: BaseException) -> str:
741 # Extract the most specific reason from a chat template error chain.
742 seen: set[int] = set()
743 current: BaseException | None = exc
744 while current is not None and id(current) not in seen:
745 seen.add(id(current))
746 if isinstance(current, jinja2.TemplateError):
747 return str(current)
748 current = current.__cause__ or current.__context__
749 return str(exc)

787def safe_apply_chat_template(
788 model_config: ModelConfig,
789 tokenizer: HfTokenizer,
790 conversation: list[ConversationMessage],
791 *,
792 tools: list[dict[str, Any]] | None = None,
793 chat_template: str | None = None,
794 tokenize: bool = True,
795 return_assistant_tokens_mask: bool = False,
796 **kwargs,
797) -> str | list[int] | tuple[list[int], list[int] | None]:
798 chat_template = resolve_chat_template(
799 tokenizer,
800 chat_template=chat_template,
801 tools=tools,
802 model_config=model_config,
803 )
804 if chat_template is None:
805 raise ChatTemplateResolutionError(
806 "As of transformers v4.44, default chat template is no longer "
807 "allowed, so you must provide a chat template if the tokenizer "
808 "does not define one."
809 )
810 if any(
811 msg["role"] == "developer" for msg in conversation
812 ) and not _detect_developer_role_support(chat_template):
813 conversation = _convert_developer_to_system(conversation)
814 conversation = _consolidate_system_messages(conversation)
815 logger.info_once(
816 "Chat template does not support the 'developer' message role. "
817 "Converting developer messages to 'system' role.",
818 )
819 resolved_kwargs = resolve_chat_template_kwargs(
820 tokenizer=tokenizer,
821 chat_template=chat_template,
822 chat_template_kwargs=kwargs,
823 )
824
825 # assistant_tokens_mask requires tokenized output — force tokenize=True.
826 if return_assistant_tokens_mask:
827 tokenize = True
828
829 # When return_assistant_tokens_mask is requested and the template supports it,
830 # request assistant_tokens_mask via return_dict.
831 # Check for the actual Jinja tag, not just the word "generation"
832 # (which also appears in add_generation_prompt).
833 if return_assistant_tokens_mask and "{% generation %}" in chat_template:
834 resolved_kwargs["return_assistant_tokens_mask"] = True
835 resolved_kwargs["return_dict"] = True
836 resolved_kwargs.pop("tokenize", None)
837 try:
838 result = tokenizer.apply_chat_template(
839 conversation=conversation, # type: ignore[arg-type]
840 tools=tools, # type: ignore[arg-type]
841 chat_template=chat_template,
842 tokenize=True,
843 **resolved_kwargs,
844 )
845 except (TypeError, ValueError) as exc:
846 logger.warning(
847 "apply_chat_template failed for assistant_tokens_mask: %s", exc
848 )
849 else:
850 if isinstance(result, Mapping):
851 token_ids = list(result.get("input_ids", []))
852 mask_raw = result.get("assistant_masks")
853 mask = list(mask_raw) if mask_raw is not None else None
854 return token_ids, mask
855 return list(result), None
856
857 # transformers v5 changed the default of `return_dict` to True, which
858 # makes `apply_chat_template(tokenize=True)` return a `BatchEncoding`
859 # instead of `list[int]`. Force `return_dict=False` so downstream code
860 # that expects a flat token list (e.g. `parse_dec_only_prompt`) works
861 # consistently across v4 and v5.
862 if tokenize and "return_dict" not in resolved_kwargs:
863 resolved_kwargs["return_dict"] = False
864
865 try:
866 plain = tokenizer.apply_chat_template(
867 conversation=conversation, # type: ignore[arg-type]
868 tools=tools, # type: ignore[arg-type]
869 chat_template=chat_template,
870 tokenize=tokenize,
871 **resolved_kwargs,
872 )
873 except Exception as e:
874 # Chat templates reject invalid user input (e.g. an unsupported
875 # `reasoning_effort` value) by raising from within the template.
876 # Surface those as a 400 Bad Request carrying the template's own
877 # reason (which typically lists the supported values) instead of a
878 # 500 or any generic upstream wrapper message.
879 logger.warning("Chat template rejected the request: %s", e)
880 raise VLLMValidationError(_template_error_reason(e)) from e
881
882 if return_assistant_tokens_mask:
883 assert isinstance(plain, list), f"Expected list[int], got {type(plain)}"
884 return plain, None
885 return plain

923def build_video_prompts_from_mm_data(
924 mm_data: MultiModalDataDict,
925) -> list[str]:
926 """Build video prompts from vision_chunk data.
927
928 Collects prompts from video chunks and groups them by video_idx.
929
930 Args:
931 mm_data: Processed multimodal data with vision_chunk items
932
933 Returns:
934 List of video prompts, one per video.
935 """
936 vision_chunks = mm_data.get("vision_chunk")
937 if vision_chunks is None:
938 return []
939
940 # Group chunks by video_idx
941 video_prompts_dict: dict[int, list[str]] = defaultdict(list)
942
943 for item in vision_chunks:
944 # vision_chunk items are always dicts (VisionChunkImage/VisionChunkVideo)
945 assert isinstance(item, dict)
946 if item.get("type") == "video_chunk":
947 video_idx = item.get("video_idx", 0)
948 prompt = item.get("prompt", "")
949 video_prompts_dict[video_idx].append(prompt)
950
951 # Build prompts in video order
952 video_prompts = [
953 "".join(video_prompts_dict[video_idx])
954 for video_idx in sorted(video_prompts_dict.keys())
955 ]
956
957 return video_prompts

960def replace_vision_chunk_video_placeholder(
961 prompt_raw: str | list[int],
962 mm_data: MultiModalDataDict,
963 video_placeholder: str | None,
964) -> str | list[int]:
965 # get video placeholder, replace it with runtime video-chunk prompts
966 if video_placeholder and isinstance(prompt_raw, str):
967 video_prompts = build_video_prompts_from_mm_data(mm_data)
968
969 # replace in order
970 prompt_raw_parts = prompt_raw.split(video_placeholder)
971 if len(prompt_raw_parts) == len(video_prompts) + 1:
972 prompt_raw = "".join(
973 itertools.chain.from_iterable(zip(prompt_raw_parts, video_prompts))
974 )
975 prompt_raw += prompt_raw_parts[-1]
976 else:
977 logger.warning(
978 "Number of video placeholders (%d) does not match "
979 "number of videos (%d) in the request.",
980 len(prompt_raw_parts) - 1,
981 len(video_prompts),
982 )
983 return prompt_raw

1260 @override
1261 def _process_tokens(
1262 self,
1263 prompt: TokensPrompt,
1264 *,
1265 skip_mm_cache: bool = False,
1266 ) -> TokensInput | MultiModalInput:
1267 """Pre-expand `prompt_embeds` sentinels before delegating to the MM
1268 processor, then attach `prompt_embeds` modality data to the result.
1269
1270 Mixed mode only: the `_prompt_embeds` stash is set by
1271 `render_messages` when `prompt_embeds` co-exist with other MM data
1272 (images, audio, …). We expand each 1-token sentinel to an N-token
1273 span *before* calling `super()._process_tokens()` so the MM
1274 processor records all placeholder offsets in the final (post-expansion)
1275 coordinate space, no offset shifting needed afterwards.
1276 """
1277 assistant_tokens_mask = cast(dict, prompt).pop("_assistant_tokens_mask", None)
1278 prompt_embeds_info = cast(dict, prompt).pop("_prompt_embeds", None)
1279 if prompt_embeds_info is not None:
1280 tensors, placeholder_token_id = prompt_embeds_info
1281 mm_updates = _build_prompt_embeds_updates(tensors, placeholder_token_id)
1282 cast(dict, prompt)["prompt_token_ids"] = _expand_prompt_embeds_placeholders(
1283 list(prompt["prompt_token_ids"]), mm_updates
1284 )
1285 engine_input = super()._process_tokens(prompt, skip_mm_cache=skip_mm_cache)
1286 if prompt_embeds_info is not None:
1287 tensors, _ = prompt_embeds_info
1288 self._apply_prompt_embeds_to_engine_input(
1289 cast(MultiModalInput, engine_input),
1290 tensors,
1291 mm_updates,
1292 )
1293 if assistant_tokens_mask is not None:
1294 engine_input["assistant_tokens_mask"] = assistant_tokens_mask
1295 return engine_input

1327 @staticmethod
1328 def _apply_prompt_embeds_to_prompt(
1329 prompt: DictPrompt,
1330 prompt_embeds_tensors: list[torch.Tensor],
1331 placeholder_token_id: int,
1332 ) -> None:
1333 """Mutate `prompt` from `TokensPrompt` to `EmbedsPrompt` shape.
1334
1335 Pure `prompt_embeds` path only (no other MM modalities). Expands
1336 each `<prompt_embeds>` sentinel token into an N-token span and builds
1337 the full-length `prompt_embeds` tensor + `prompt_is_token_ids` mask
1338 that the engine's `enable_prompt_embeds` worker branch consumes.
1339 """
1340 token_ids = cast(list[int] | None, prompt.get("prompt_token_ids"))
1341 if token_ids is None:
1342 raise RuntimeError(_MISSING_PROMPT_TOKEN_IDS_ERROR)
1343
1344 embeds_orig_positions: list[int] = [
1345 i for i, tok in enumerate(token_ids) if tok == placeholder_token_id
1346 ]
1347 if len(embeds_orig_positions) != len(prompt_embeds_tensors):
1348 raise ValueError(
1349 f"Expected {len(prompt_embeds_tensors)} prompt_embeds "
1350 f"placeholder tokens in the rendered prompt, found "
1351 f"{len(embeds_orig_positions)}."
1352 )
1353
1354 mm_updates = _build_prompt_embeds_updates(
1355 prompt_embeds_tensors, placeholder_token_id
1356 )
1357 expanded = _expand_prompt_embeds_placeholders(token_ids, mm_updates)
1358 positions = _build_prompt_embeds_positions(
1359 expanded, len(prompt_embeds_tensors), mm_updates
1360 )
1361
1362 embeds_prompt = cast(EmbedsPrompt, prompt)
1363 embeds_prompt["prompt_token_ids"] = expanded
1364 full_embeds, is_token_ids_mask = _build_mixed_prompt_embeds(
1365 expanded, prompt_embeds_tensors, positions
1366 )
1367 embeds_prompt["prompt_embeds"] = full_embeds
1368 embeds_prompt["prompt_is_token_ids"] = is_token_ids_mask
1369
1370 def _apply_prompt_embeds_to_engine_input(
1371 self,
1372 engine_input: MultiModalInput,
1373 prompt_embeds_tensors: list[torch.Tensor],
1374 mm_updates: MultiModalPromptUpdates,
1375 ) -> None:
1376 """Augment `engine_input` in-place with a `prompt_embeds` modality.
1377
1378 Mixed mode: called after `_process_multimodal` has already run on the
1379 pre-expanded token IDs (expansion was done in `_process_tokens` before
1380 calling `super()`). Locates the already-expanded `prompt_embeds` spans
1381 and adds `prompt_embeds` entries to `mm_kwargs`, `mm_hashes`, and
1382 `mm_placeholders`.
1383 """
1384 # token_ids already contain the pre-expanded N-token spans.
1385 token_ids = list(engine_input["prompt_token_ids"])
1386
1387 positions = _build_prompt_embeds_positions(
1388 token_ids, len(prompt_embeds_tensors), mm_updates
1389 )
1390
1391 pe_kwargs_items: list[MultiModalKwargsItem] = []
1392 pe_hashes: list[str] = []
1393 pe_placeholders: list[PlaceholderRange] = []
1394 mm_config = self.model_config.get_multimodal_config()
1395 for tensor, (start, length) in zip(
1396 prompt_embeds_tensors, positions, strict=True
1397 ):
1398 pe_kwargs_items.append(
1399 MultiModalKwargsItem(
1400 {
1401 "embedding": MultiModalFieldElem(
1402 data=tensor,
1403 field=MultiModalSharedField(batch_size=1),
1404 )
1405 }
1406 )
1407 )
1408 pe_hashes.append(
1409 MultiModalHasher.hash_kwargs(
1410 mm_config.mm_hasher_algorithm, prompt_embeds=tensor
1411 )
1412 )
1413 # `is_embed=None` matches the existing image_embeds-style
1414 # "no encoder, just splice the tensor directly" semantics.
1415 pe_placeholders.append(
1416 PlaceholderRange(offset=start, length=length, is_embed=None)
1417 )
1418
1419 cast(
1420 MultiModalKwargsItems[MultiModalKwargsItem | None],
1421 engine_input["mm_kwargs"],
1422 )["prompt_embeds"] = pe_kwargs_items
1423 engine_input["mm_hashes"] = {
1424 **engine_input["mm_hashes"],
1425 "prompt_embeds": pe_hashes,
1426 }
1427 cast(dict, engine_input["mm_placeholders"])["prompt_embeds"] = pe_placeholders
