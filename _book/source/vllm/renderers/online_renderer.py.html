84def _extract_allowed_tools_from_mcp_requests(
85 tools: list[Tool],
86) -> dict[str, list[str] | None]:
87 allowed_tools_map: dict[str, list[str] | None] = {}
88 for tool in tools:
89 if not isinstance(tool, Mcp):
90 continue
91
92 allowed_tools = None
93 if isinstance(tool.allowed_tools, list):
94 allowed_tools = tool.allowed_tools
95 elif tool.allowed_tools is not None:
96 allowed_tools = tool.allowed_tools.tool_names
97 if allowed_tools is not None and "*" in allowed_tools:
98 allowed_tools = None
99 allowed_tools_map[tool.server_label] = allowed_tools
100 return allowed_tools_map

103def _reused_prompt_token_ids(request: Any) -> list[int] | None:
104 """Pop prompt token ids forwarded for decode-side reuse, if any.
105
106 Disaggregated serving carries the prefill stage's ids in
107 ``kv_transfer_params`` so the decode stage can skip re-tokenizing. Removing
108 the key keeps the id list out of the engine's sampling metadata.
109 """
110 kv = getattr(request, "kv_transfer_params", None)
111 if not isinstance(kv, dict):
112 return None
113 return kv.pop("prompt_token_ids", None) or None

170 async def render_chat(
171 self,
172 request: ChatCompletionRequest,
173 *,
174 skip_mm_cache: bool = False,
175 ) -> tuple[list[ConversationMessage], list[EngineInput]] | ErrorResponse:
176 """Core preprocessing logic for chat requests (no model/engine check).
177
178 Called directly by render_chat_request and delegated to by
179 OpenAIServingChat.render_chat_request after its engine-aware checks.
180
181 Decode-side token reuse (ids forwarded in ``kv_transfer_params``) is
182 handled deeper, in ``preprocess_chat`` / ``_make_request_with_harmony``,
183 so it skips only templating and tokenization while tool-choice
184 validation and ``adjust_request`` still run and the output is
185 detokenized (text-out).
186 """
187 tokenizer = self.renderer.tokenizer
188
189 tool_parser = self.parser.tool_parser_cls if self.parser is not None else None
190
191 if is_mistral_tokenizer(tokenizer):
192 # because of issues with pydantic we need to potentially
193 # re-serialize the tool_calls field of the request
194 _mt.maybe_serialize_tool_calls(request) # type: ignore[arg-type]
195 _mt.truncate_tool_call_ids(request) # type: ignore[arg-type]
196 _mt.validate_request_params(request)
197
198 # Check if tool parsing is unavailable (common condition)
199 tool_parsing_unavailable = (
200 tool_parser is None
201 and not is_mistral_tokenizer(tokenizer)
202 and not self.use_harmony
203 )
204
205 # Validate tool_choice when tool parsing is required but unavailable
206 if tool_parsing_unavailable and request.tool_choice not in (
207 None,
208 "none",
209 ):
210 if request.tool_choice == "auto" and not self.enable_auto_tools:
211 # for hf tokenizers, "auto" tools requires
212 # --enable-auto-tool-choice and --tool-call-parser
213 return self.create_error_response(
214 '"auto" tool choice requires '
215 "--enable-auto-tool-choice and --tool-call-parser to be set"
216 )
217 elif request.tool_choice != "auto":
218 # "required" or named tool requires tool parser
219 if isinstance(request.tool_choice, ChatCompletionNamedToolChoiceParam):
220 tool_choice_desc = f'function "{request.tool_choice.function.name}"'
221 else:
222 tool_choice_desc = f'"{request.tool_choice}"'
223 return self.create_error_response(
224 f"tool_choice={tool_choice_desc} requires "
225 "--tool-call-parser to be set"
226 )
227
228 if request.tools is None or (
229 request.tool_choice == "none" and self.exclude_tools_when_tool_choice_none
230 ):
231 tool_dicts = None
232 else:
233 tool_dicts = [tool.model_dump() for tool in request.tools]
234
235 if not self.use_harmony:
236 # Common case.
237 error_check_ret = self.validate_chat_template(
238 request_chat_template=request.chat_template,
239 chat_template_kwargs=request.chat_template_kwargs,
240 trust_request_chat_template=self.trust_request_chat_template,
241 )
242 if error_check_ret is not None:
243 return error_check_ret
244
245 conversation, engine_inputs = await self.preprocess_chat(
246 request,
247 request.messages,
248 default_template=self.chat_template,
249 default_template_content_format=self.chat_template_content_format,
250 default_template_kwargs=self.default_chat_template_kwargs,
251 tool_dicts=tool_dicts,
252 parser=self.parser,
253 skip_mm_cache=skip_mm_cache,
254 )
255 else:
256 # For GPT-OSS.
257 if self.parser is not None:
258 # HarmonyParser doesn't need chat_template_kwargs
259 # TODO: Unify adjust_request() call with non-harmony branch
260 self.parser(
261 self.renderer.get_tokenizer(),
262 request.tools,
263 model_config=self.model_config,
264 ).adjust_request(request=request)
265
266 should_include_tools = tool_dicts is not None
267 conversation, engine_inputs = self._make_request_with_harmony(
268 request, should_include_tools
269 )
270
271 return conversation, engine_inputs

273 async def render_responses(
274 self,
275 request: ResponsesRequest,
276 *,
277 previous_messages: ResponsesPreviousMessages | None = None,
278 previous_response_outputs: list[ResponseOutputItem] | None = None,
279 tool_server: "ToolServer | None" = None,
280 skip_mm_cache: bool = False,
281 ) -> ResponsesRenderResult | ErrorResponse:
282 """Render a Responses request using only explicitly supplied history."""
283 template_error = self.validate_chat_template(
284 request_chat_template=None,
285 chat_template_kwargs=request.chat_template_kwargs,
286 trust_request_chat_template=self.trust_request_chat_template,
287 )
288 if template_error is not None:
289 return template_error
290
291 if self.use_harmony:
292 return self._render_responses_with_harmony(
293 request,
294 previous_messages=previous_messages,
295 previous_response_outputs=previous_response_outputs,
296 tool_server=tool_server,
297 )
298
299 if previous_messages is not None and any(
300 isinstance(message, OpenAIMessage) for message in previous_messages
301 ):
302 return self.create_error_response(
303 "Non-Harmony Responses history must use chat messages.",
304 err_type="invalid_request_error",
305 param="previous_response_id",
306 )
307
308 tool_dicts = construct_tool_dicts(
309 request.tools,
310 request.tool_choice,
311 exclude_tools_when_tool_choice_none=(
312 self.exclude_tools_when_tool_choice_none
313 ),
314 )
315 messages = construct_input_messages(
316 request_instructions=request.instructions,
317 request_input=request.input,
318 prev_msg=list(previous_messages) if previous_messages is not None else None,
319 prev_response_output=(
320 list(previous_response_outputs)
321 if previous_response_outputs is not None
322 else None
323 ),
324 )
325 chat_template_kwargs = (
326 request.build_chat_params(
327 self.chat_template,
328 self.chat_template_content_format,
329 )
330 .with_defaults(self.default_chat_template_kwargs)
331 .chat_template_kwargs
332 )
333 _, engine_inputs = await self.preprocess_chat(
334 request,
335 messages,
336 default_template=self.chat_template,
337 default_template_content_format=self.chat_template_content_format,
338 default_template_kwargs=chat_template_kwargs,
339 tool_dicts=tool_dicts,
340 parser=self.parser,
341 skip_mm_cache=skip_mm_cache,
342 )
343 return self._responses_render_result(messages, engine_inputs)

345 def _render_responses_with_harmony(
346 self,
347 request: ResponsesRequest,
348 *,
349 previous_messages: ResponsesPreviousMessages | None,
350 previous_response_outputs: list[ResponseOutputItem] | None,
351 tool_server: "ToolServer | None",
352 ) -> ResponsesRenderResult | ErrorResponse:
353 if self.parser is not None:
354 # HarmonyParser doesn't need chat_template_kwargs
355 # TODO: Unify adjust_request() call with non-harmony branch
356 self.parser(
357 self.renderer.get_tokenizer(),
358 request.tools,
359 model_config=self.model_config,
360 ).adjust_request(request=request)
361
362 if previous_messages is not None and any(
363 not isinstance(message, OpenAIMessage) for message in previous_messages
364 ):
365 return self.create_error_response(
366 "Harmony Responses history must use Harmony messages.",
367 err_type="invalid_request_error",
368 param="previous_response_id",
369 )
370
371 messages: list[OpenAIMessage] = []
372 request_input = request.input
373 if previous_messages is None:
374 tool_types = extract_tool_types(request.tools)
375 with_custom_tools = has_custom_tools(tool_types)
376 instructions = request.instructions
377 if instructions is None and isinstance(request_input, list):
378 instructions, request_input = extract_instructions_from_messages(
379 request_input
380 )
381 descriptions = self._get_harmony_builtin_tool_descriptions(
382 request.tools,
383 tool_types,
384 tool_server,
385 )
386 messages.extend(
387 build_harmony_preamble(
388 instructions=instructions,
389 tools=request.tools if with_custom_tools else None,
390 reasoning_effort=(
391 request.reasoning.effort if request.reasoning else None
392 ),
393 with_custom_tools=with_custom_tools,
394 **descriptions,
395 )
396 )
397 messages.extend(construct_harmony_previous_input_messages(request))
398 else:
399 messages.extend(previous_messages)
400
401 previous_outputs = list(previous_response_outputs or ())
402 try:
403 if isinstance(request_input, str):
404 if request_input or not request.previous_input_messages:
405 messages.append(get_user_message(request_input))
406 else:
407 for response_message in request_input:
408 new_message = response_input_to_harmony(
409 response_message,
410 previous_outputs,
411 )
412 if new_message is not None:
413 messages.append(new_message)
414 if isinstance(response_message, ResponseFunctionToolCall):
415 previous_outputs.append(response_message)
416 except (ValueError, VLLMValidationError) as exc:
417 return self.create_error_response(
418 str(exc),
419 err_type="invalid_request_error",
420 param="input",
421 )
422
423 return ResponsesRenderResult(
424 messages=messages,
425 engine_input=self.render_responses_harmony_messages(
426 messages,
427 cache_salt=request.cache_salt,
428 tok_params=request.build_tok_params(self.model_config),
429 ),
430 )

432 def _get_harmony_builtin_tool_descriptions(
433 self,
434 tools: list[Tool],
435 tool_types: set[str],
436 tool_server: "ToolServer | None",
437 ) -> dict[str, ToolNamespaceConfig | None]:
438 allowed_tools = _extract_allowed_tools_from_mcp_requests(tools)
439 descriptions: dict[str, ToolNamespaceConfig | None] = {}
440 for server_name, request_name in BUILTIN_TOOL_TO_MCP_SERVER_LABEL.items():
441 description = (
442 tool_server.get_tool_description(
443 server_name,
444 allowed_tools.get(request_name),
445 )
446 if request_name in tool_types
447 and tool_server is not None
448 and tool_server.has_tool(server_name)
449 else None
450 )
451 descriptions[f"{server_name}_description"] = description
452 return descriptions

454 def render_responses_harmony_messages(
455 self,
456 messages: list[OpenAIMessage],
457 *,
458 cache_salt: str | None,
459 tok_params: TokenizeParams | None = None,
460 ) -> EngineInput:
461 arrival_time = time.time()
462 prompt = TokensPrompt(prompt_token_ids=render_for_completion(messages))
463 if tok_params is not None:
464 tok_params.apply_post_tokenization(
465 self.renderer.tokenizer,
466 prompt,
467 )
468 engine_input = tokens_input(prompt["prompt_token_ids"], cache_salt=cache_salt)
469 engine_input["arrival_time"] = arrival_time
470 return engine_input

486 def _make_request_with_harmony(
487 self,
488 request: ChatCompletionRequest,
489 should_include_tools: bool = True,
490 ):
491 """Build Harmony (GPT-OSS) messages and engine prompt from a chat request."""
492 reuse_ids = _reused_prompt_token_ids(request)
493 if reuse_ids:
494 # Decode-side token reuse: feed the forwarded ids straight to the
495 # engine. Harmony has no adjust_request hook to preserve.
496 engine_input = tokens_input(reuse_ids, cache_salt=request.cache_salt)
497 return [], [engine_input]
498
499 messages: list[OpenAIMessage] = []
500
501 # because of issues with pydantic we need to potentially
502 # re-serialize the tool_calls field of the request
503 # for more info: see comment in `maybe_serialize_tool_calls`
504 _mt.maybe_serialize_tool_calls(request) # type: ignore[arg-type]
505
506 chat_messages = list(request.messages)
507 instructions, chat_messages = extract_instructions_from_messages(chat_messages)
508
509 # Add system message.
510 # NOTE: In Chat Completion API, browsing is enabled by default
511 # if the model supports it. TODO: Support browsing.
512 assert not self.supports_browsing
513 assert not self.supports_code_interpreter
514 if (reasoning_effort := request.reasoning_effort) == "none":
515 raise VLLMValidationError(
516 f"Harmony does not support {reasoning_effort=}",
517 parameter="reasoning_effort",
518 )
519 tools = request.tools if should_include_tools else None
520 messages.extend(
521 build_harmony_preamble(
522 instructions=instructions,
523 tools=tools, # type: ignore[arg-type]
524 reasoning_effort=reasoning_effort,
525 with_custom_tools=should_include_tools,
526 )
527 )
528
529 # Add remaining conversation messages.
530 messages.extend(parse_chat_inputs_to_harmony_messages(chat_messages))
531
532 # Render prompt token ids.
533 prompt_token_ids = render_for_completion(messages)
534 engine_input = tokens_input(prompt_token_ids, cache_salt=request.cache_salt)
535
536 return messages, [engine_input]

539 self,
540 request: CompletionRequest,
541 *,
542 skip_mm_cache: bool = False,
543 ) -> list[EngineInput] | ErrorResponse:
544 """Core preprocessing logic for completion requests (no model/engine check).
545
546 Called directly by render_completion_request and delegated to by

559 )
560

562 request,
563 prompt_input=request.prompt,
564 prompt_embeds=request.prompt_embeds,
565 skip_mm_cache=skip_mm_cache,
566 )
567
568 return engine_inputs

648 async def preprocess_chat(
649 self,
650 request: Any,
651 messages: list[Any],
652 default_template: str | None,
653 default_template_content_format: ChatTemplateContentFormatOption,
654 default_template_kwargs: dict[str, Any] | None,
655 tool_dicts: list[dict[str, Any]] | None = None,
656 parser: type[Parser] | None = None,
657 *,
658 skip_mm_cache: bool = False,
659 ) -> tuple[list[ConversationMessage], list[EngineInput]]:
660 """Copied from GenerateBaseServing._preprocess_chat."""
661 renderer = self.renderer
662 mm_config = self.model_config.multimodal_config
663
664 default_template_kwargs = merge_kwargs(
665 default_template_kwargs,
666 dict(
667 tools=tool_dicts,
668 tokenize=(
669 is_mistral_tokenizer(renderer.tokenizer)
670 or self.model_config.enable_prompt_embeds
671 ),
672 ),
673 )
674
675 tok_params = request.build_tok_params(self.model_config)
676 chat_params = request.build_chat_params(
677 default_template, default_template_content_format
678 ).with_defaults(
679 default_template_kwargs,
680 default_media_io_kwargs=(mm_config.media_io_kwargs if mm_config else None),
681 default_mm_processor_kwargs=getattr(request, "mm_processor_kwargs", None),
682 )
683
684 reuse_ids = _reused_prompt_token_ids(request)
685 if reuse_ids:
686 # Decode-side token reuse: feed the forwarded ids straight to the
687 # engine, skipping templating and tokenization. ``messages`` are not
688 # tokenized, so conversation is empty. The adjust_request tail below
689 # still runs.
690 conversation: list[ConversationMessage] = []
691 engine_input = tokens_input(
692 reuse_ids, cache_salt=getattr(request, "cache_salt", None)
693 )
694 else:
695 (conversation,), (engine_input,) = await renderer.render_chat_async(
696 [messages],
697 chat_params,
698 tok_params,
699 prompt_extras={
700 k: v
701 for k in ("mm_processor_kwargs", "cache_salt")
702 if (v := getattr(request, k, None)) is not None
703 },
704 skip_mm_cache=skip_mm_cache,
705 )
706
707 # tool parsing is done only if a tool_parser has been set and if
708 # tool_choice is not "none" (if tool_choice is "none" but a tool_parser
709 # is set, we want to prevent parsing a tool_call hallucinated by the LLM
710 #
711 # Exception: Mistral grammar-capable tokenizers always call
712 # adjust_request — even for tool_choice="none" — so that the grammar

719 tool_parser is not None
720 and is_mistral_tool_parser(tool_parser)
721 and is_mistral_tokenizer(tokenizer)

723 )
724 should_adjust_request = (
725 parser.reasoning_parser_cls is not None
726 or tool_choice != "none"
727 or is_mistral_grammar_eligible
728 )
729 if should_adjust_request:
730 if not isinstance(request, ChatCompletionRequest | ResponsesRequest):
731 msg = (
732 "Tool usage is only supported "
733 "for Chat Completions API or Responses API requests, "
734 f"but got {type(request).__name__}"
735 )

737 request = parser(
738 tokenizer,
739 request.tools,
740 model_config=self.model_config,
741 chat_template_kwargs=chat_params.chat_template_kwargs,
742 ).adjust_request(
743 request=request,
744 )
