88 # Background multimodal warmup task (a future returned by
89 # _mm_executor) so it overlaps engine-core initialization while
90 # staying serialized with the serving path through the same
91 # single-worker executor. None when no background warmup has been
92 # launched (or it has already been joined). Routing the warmup through
93 # _mm_executor (max_workers=1) guarantees it can never run
94 # concurrently with a serving request inside the numba workqueue
95 # parallel region, which would otherwise hit numba's
96 # "Concurrent access has been detected" fatal abort (see
97 # vllm/utils/jit_monitor.py, which forces the workqueue layer).
98 # Joined by the first of reset_mm_cache / warmup / shutdown so the
99 # mm_processor_cache is also never touched concurrently by the warmup

104 # warmup already ran even after reset_mm_cache joined it first, and

263 def warmup_mm(self) -> None:
264 # Run only the multimodal warmup (mm + readonly mm), serially.
265 # Idempotent: runs at most once. A background warmup
266 # (start_mm_warmup_in_background) may already have run it, and
267 # reset_mm_cache may have joined that background warmup task before
268 # warmup() reaches this point — the _mm_warmup_done flag survives the
269 # join (the future pointer does not), so warmup() can still tell it
270 # already ran.
271 if self._mm_warmup_done:
272 return
273 try:
274 # prevent MM processor hangs
275 with set_default_torch_num_threads(1):
276 if self.mm_processor:
277 try:
278 logger.debug("Warming up multi-modal processing...")
279 self._warmup_mm_processor(
280 self.mm_processor,
281 log_prefix="Multi-modal",
282 )
283 except Exception:
284 logger.warning("Multi-modal warmup failed")
285 finally:
286 self.clear_mm_cache()
287
288 if self._readonly_mm_processor is not None:
289 try:
290 logger.debug("Warming up readonly multi-modal processing...")
291 self._warmup_mm_processor(
292 self._readonly_mm_processor,
293 log_prefix="Readonly multi-modal",
294 )
295 except Exception:
296 logger.warning("Readonly multi-modal warmup failed")
297 finally:
298 self._clear_processor_cache(self._readonly_mm_processor)
299 finally:
300 self._mm_warmup_done = True

302 def start_mm_warmup_in_background(self) -> None:
303 # Launch the multimodal warmup as a task on the single-worker
304 # _mm_executor so it overlaps with engine-core initialization
305 # (model loading) while staying serialized with the serving path.
306 #
307 # Why the same executor as serving (not a separate Thread):
308 # _process_multimodal runs on _mm_executor (max_workers=1) and
309 # enters the numba workqueue parallel region (e.g. Kimi K2.5 vision
310 # preprocessing via kimi_k25_vision_fused). A separate warmup Thread
311 # would run concurrently with that single serving worker, so both
312 # threads would enter the numba workqueue parallel region at the same
313 # time and trip numba's "Concurrent access has been detected" ->
314 # "Fatal Python error: Aborted" (see vllm/utils/jit_monitor.py, which
315 # forces the non-threadsafe workqueue layer). Submitting the warmup to
316 # the same single-worker executor makes warmup and serving
317 # physically incapable of overlapping on the numba threading layer,
318 # while still overlapping at the process level with the (forked)
319 # engine-core init in the main thread.
320 #
321 # The warmup task is joined by the first of: reset_mm_cache, warmup, or
322 # shutdown. This also guarantees the frontend mm_processor_cache —
323 # which is not designed for concurrent access (its get_and_update must
324 # stay ordered with the engine-core cache) — is never touched
325 # concurrently by the warmup and the serving path.
326 if self.mm_processor is None and self._readonly_mm_processor is None:
327 return # text-only model: nothing to warm up
328 if self._mm_warmup_future is not None:
329 return # already launched
330 self._mm_warmup_future = self._mm_executor.submit(self.warmup_mm)
331
332 def _join_mm_warmup(self) -> None:
333 # Wait for the background MM warmup task to finish, if one was started.
334 # Called from the main / event-loop thread (reset_mm_cache, warmup,
335 # shutdown) — never from inside the _mm_executor worker — so
336 # waiting on the future cannot deadlock.
337 future = getattr(self, "_mm_warmup_future", None)
338 if future is not None:
339 future.result()
340 self._mm_warmup_future = None

342 def warmup(self, chat_params: ChatParams) -> None:
343 """
344 Warm up this renderer to avoid first-request latency.
345
346 For chat requests:
347 - Jinja2 template compilation
348
349 If a background multimodal warmup was launched (to overlap
350 engine-core init) it is joined first so it is not still touching the
351 mm_processor_cache. warmup_mm is idempotent, so the call below is a
352 no-op if the background warmup — or a prior reset_mm_cache that joined
353 it — already ran it.
354 """
355 from vllm.entrypoints.chat_utils import ChatTemplateResolutionError
356
357 # prevent MM processor hangs
358 with set_default_torch_num_threads(1):
359 try:
360 logger.debug("Warming up chat template processing...")
361 start_time = time.perf_counter()
362
363 self.render_chat([[{"role": "user", "content": "warmup"}]], chat_params)
364
365 elapsed = time.perf_counter() - start_time
366 logger.debug("Chat template warmup completed in %.3fs", elapsed)
367 except ChatTemplateResolutionError:
368 logger.debug("This model does not support chat template.")
369 except Exception:
370 logger.warning("Chat template warmup failed", exc_info=True)
371
372 if self._mm_warmup_future is not None:
373 # Join a background MM warmup launched to overlap engine-core init.
374 self._join_mm_warmup()
375 # Idempotent: no-op if the background warmup (or a prior reset_mm_cache)
376 # already ran the MM warmup.
377 self.warmup_mm()
378
379 async def clear_mm_cache_async(self) -> None:
380 """Serialize clear_mm_cache through the multimodal executor to avoid
381 races with concurrent process_inputs on the mm_processor_cache."""
382 await self._clear_mm_cache_async()

384 def shutdown(self) -> None:
385 # Wait for any background MM warmup to finish first: it runs on the
386 # _mm_executor and touches the mm_processor_cache, so it must be
387 # quiescent before _resources.close() tears down the executors and
388 # the cache (both are registered as resources in __init__).
389 self._join_mm_warmup()
390 self._resources.close()

540 @staticmethod
541 def _build_tokens_prompt(
542 token_ids: Sequence[int],
543 prompt: "TextPrompt",
544 *,
545 offset_mapping: Sequence[tuple[int, int]] | None = None,
546 ) -> "TokensPrompt":
547 """Build a TokensPrompt from already-extracted token ids.
548
549 ``offset_mapping`` is the per-token ``(start, end)`` sequence from
550 a BatchEncoding; pass it only when offsets were requested, and it
551 is attached as ``prompt_token_offsets``.
552 """
553 if offset_mapping is not None:
554 return TokensPrompt(
555 prompt_token_ids=list(token_ids),
556 prompt_token_offsets=[(int(s), int(e)) for s, e in offset_mapping],
557 **prompt,
558 )
559 return TokensPrompt(prompt_token_ids=list(token_ids), **prompt)
560
561 @staticmethod
562 def _apply_prompt_char_offset(
563 prompt: TokensPrompt, char_offset: int
564 ) -> TokensPrompt:
565 """Map token offsets back to the original prompt after a text pre-trim."""
566 if char_offset == 0:
567 return prompt
568
569 offsets = prompt.get("prompt_token_offsets")
570 if offsets is not None:
571 prompt["prompt_token_offsets"] = [
572 # Fast tokenizers use (0, 0) for special tokens, which are
573 # not character spans in the source prompt.
574 (start, end)
575 if (start, end) == (0, 0)
576 else (start + char_offset, end + char_offset)
577 for start, end in offsets
578 ]
579
580 return prompt

582 def _tokenize_prompt(
583 self,
584 prompt: TextPrompt,
585 params: TokenizeParams,
586 ) -> TokensPrompt:
587 tokenizer = self.get_tokenizer()
588 want_offsets = self._wants_offsets(prompt, params)
589 kwargs = params.get_encode_kwargs()
590 if want_offsets:
591 kwargs = {**kwargs, "return_offsets_mapping": True}
592 encoding = tokenizer(prompt["prompt"], **kwargs)
593 return self._build_tokens_prompt(
594 encoding["input_ids"],
595 prompt,
596 offset_mapping=encoding["offset_mapping"] if want_offsets else None,
597 )

781 # Step 4: Convert to engine inputs
782 def _validate_mm_uuids(
783 self,
784 mm_data: MultiModalDataDict,
785 mm_data_items: MultiModalDataItems,
786 mm_uuid_items: MultiModalUUIDItems,
787 ) -> None:
788 # NOTE: Keys corresponding to `None` in `mm_data` don't appear in
789 # `mm_data_items`
790 modalities = mm_data.keys() | mm_uuid_items.keys()
791
792 for modality in modalities:
793 data_items = mm_data_items.get(modality)
794 uuid_items = mm_uuid_items.get(modality)
795
796 if data_items is None:
797 if uuid_items is None:
798 raise ValueError(
799 f"multi_modal_data[{modality!r}] is empty but "
800 f"multi_modal_uuids[{modality!r}] is missing."
801 )
802
803 elif uuid_items is not None:
804 if len(data_items) != len(uuid_items):
805 raise ValueError(
806 f"If given, multi_modal_uuids[{modality!r}] must have "
807 f"same length as multi_modal_data[{modality!r}], but "
808 f"got {len(uuid_items)} vs {len(data_items)}."
809 )
810
811 for i, item in enumerate(data_items):
812 if item is None and uuid_items[i] is None:
813 raise ValueError(
814 f"multi_modal_data[{modality!r}][{i}] is empty but "
815 f"multi_modal_uuids[{modality!r}][{i}] is missing."
816 )
817
818 def _process_mm_uuids(
819 self,
820 mm_data: MultiModalDataDict,
821 mm_data_items: MultiModalDataItems,
822 mm_uuid_items: MultiModalUUIDItems,
823 mm_req_id: str,
824 ) -> MultiModalUUIDItems:
825 model_config = self.model_config
826
827 # Skip content hashing when both caches are disabled, but preserve
828 # explicit UUIDs used to identify embeddings across EC instances.
829 if (
830 model_config.multimodal_config
831 and model_config.multimodal_config.mm_processor_cache_gb == 0
832 and not self.config.cache_config.enable_prefix_caching
833 ):
834 mm_uuid_items = dict(mm_uuid_items)
835 for modality, data_count in mm_data_items.get_all_counts().items():
836 uuids = mm_uuid_items.get(modality)
837 if uuids is None:
838 uuids = [None] * data_count
839 mm_uuid_items[modality] = [
840 uuid if uuid is not None else f"{mm_req_id}-{modality}-{i}"
841 for i, uuid in enumerate(uuids)
842 ]
843
844 self._validate_mm_uuids(mm_data, mm_data_items, mm_uuid_items)
845
846 return mm_uuid_items

848 def _process_multimodal(
849 self,
850 prompt: list[int],
851 mm_data: MultiModalDataDict,
852 mm_uuids: MultiModalUUIDDict | None,
853 mm_processor_kwargs: Mapping[str, object] | None,
854 media_io_kwargs: Mapping[str, Mapping[str, object]] | None = None,
855 *,
856 skip_mm_cache: bool = False,
857 ) -> "MultiModalInput":
858 if skip_mm_cache and self._readonly_mm_processor is not None:
859 mm_processor = self._readonly_mm_processor
860 else:
861 mm_processor = self.get_mm_processor()
862
863 mm_req_id = f"renderer{self.api_process_rank}-mm-{self._mm_req_counter.inc(1)}"
864
865 mm_data_items = mm_processor.info.parse_mm_data(mm_data)
866 mm_uuid_items = parse_mm_uuids(mm_uuids)
867
868 mm_uuid_items = self._process_mm_uuids(
869 mm_data, mm_data_items, mm_uuid_items, mm_req_id
870 )
871
872 mm_processor_inputs = MMProcessorInputs(
873 prompt,
874 mm_data_items,
875 mm_uuid_items,
876 hf_processor_mm_kwargs=mm_processor_kwargs or {},
877 media_io_kwargs=media_io_kwargs or {},
878 )
879 mm_timing_ctx = self._mm_timing_registry.get(mm_req_id)
880
881 with set_default_torch_num_threads():
882 mm_inputs = mm_processor.apply(mm_processor_inputs, mm_timing_ctx)
883
884 self.update_mm_cache_stats()
885
886 return mm_inputs

888 def _process_tokens(
889 self,
890 prompt: TokensPrompt,
891 *,
892 skip_mm_cache: bool = False,
893 ) -> TokensInput | MultiModalInput:
894 """Process token inputs, with multimodal preprocessing offloaded
895 to the shared thread pool in the async variant.
896 """
897 prompt_token_ids = prompt["prompt_token_ids"]
898
899 engine_input: TokensInput | MultiModalInput
900 if multi_modal_data := prompt.get("multi_modal_data"):
901 engine_input = self._process_multimodal(
902 prompt_token_ids,
903 multi_modal_data,
904 mm_processor_kwargs=prompt.get("mm_processor_kwargs"),
905 mm_uuids=prompt.get("multi_modal_uuids"),
906 media_io_kwargs=prompt.get("media_io_kwargs"),
907 skip_mm_cache=skip_mm_cache,
908 )
909 else:
910 engine_input = tokens_input(prompt_token_ids)
911
912 if prompt_text := prompt.get("prompt"):
913 engine_input["prompt"] = prompt_text
914 if cache_salt := prompt.get("cache_salt"):
915 engine_input["cache_salt"] = cache_salt
916 # Narrow the union — `prompt_token_offsets` is only on TokensInput.
917 if engine_input["type"] == "token" and (
918 (offsets := prompt.get("prompt_token_offsets")) is not None
919 ):
920 engine_input["prompt_token_offsets"] = offsets
921
922 return engine_input

924 def _process_embeds(self, prompt: EmbedsPrompt) -> EmbedsInput:
925 if not self.model_config.enable_prompt_embeds:
926 raise ValueError(
927 "You must set `--enable-prompt-embeds` to input `prompt_embeds`."
928 )
929
930 prompt_embeds = prompt["prompt_embeds"]
931
932 # prompt_embeds must be (seq_len, hidden_size), but if the user
933 # passes in a batch of size 1, i.e. (1, seq_len, hidden_size),
934 # we can unambiguously process the intent by squeezing the batch
935 # dimension.
936 if prompt_embeds.ndim == 3:
937 prompt_embeds = prompt_embeds.squeeze(dim=0)
938
939 if prompt_embeds.ndim != 2:
940 raise ValueError("prompt_embeds must be of shape (seq_len, hidden_size).")
941
942 # Tensors must be on CPU for serialization between processes
943 # in the MsgpackEncoder. Casting to CPU here ensures that there is no
944 # hidden device transfer in the critical path of generation.
945 prompt_embeds = prompt_embeds.cpu()
946
947 return embeds_input(
948 prompt_embeds=prompt_embeds,
949 cache_salt=prompt.get("cache_salt"),
950 prompt_token_ids=prompt.get("prompt_token_ids"),
951 is_token_ids=prompt.get("prompt_is_token_ids"),
952 )
