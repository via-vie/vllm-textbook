40 async def _check_model(
41 self,
42 request: AnyRequest | AnyPoolingRequest,
43 ) -> ErrorResponse | None:
44 error_response = None
45
46 if self._is_model_supported(request.model):
47 return None
48 if request.model in self.models.lora_requests:
49 return None
50 if (
51 envs.VLLM_ALLOW_RUNTIME_LORA_UPDATING
52 and request.model
53 and (load_result := await self.models.resolve_lora(request.model))
54 ):
55 if isinstance(load_result, LoRARequest):
56 return None
57 if (
58 isinstance(load_result, ErrorResponse)
59 and load_result.error.code == HTTPStatus.BAD_REQUEST.value
60 ):
61 error_response = load_result
62
63 return error_response or self.create_error_response(
64 message=f"The model `{request.model}` does not exist.",
65 err_type="NotFoundError",
66 status_code=HTTPStatus.NOT_FOUND,
67 param="model",
68 )

70 def _is_model_supported(self, model_name: str | None) -> bool:
71 if not model_name:
72 return True
73 if envs.VLLM_SKIP_MODEL_NAME_VALIDATION:
74 return True
75 return self.models.is_base_model(model_name)

128 def _get_message_types(self, request: AnyRequest | AnyPoolingRequest) -> set[str]:
129 """Retrieve the set of types from message content dicts up
130 until `_`; we use this to match potential multimodal data
131 with default per modality loras.
132 """
133 message_types: set[str] = set()
134
135 if not hasattr(request, "messages"):
136 return message_types
137
138 messages = request.messages
139 if messages is None or isinstance(messages, (str, bytes)):
140 return message_types
141
142 for message in messages:
143 if (
144 isinstance(message, dict)
145 and "content" in message
146 and isinstance(message["content"], list)
147 ):
148 for content_dict in message["content"]:
149 if "type" in content_dict:
150 message_types.add(content_dict["type"].split("_")[0])
151 return message_types
152
153 def _get_active_default_mm_loras(
154 self, request: AnyRequest | AnyPoolingRequest
155 ) -> LoRARequest | None:
156 """Determine if there are any active default multimodal loras."""
157 # TODO: Currently this is only enabled for chat completions
158 # to be better aligned with only being enabled for .generate
159 # when run offline. It would be nice to support additional
160 # tasks types in the future.
161 message_types = self._get_message_types(request)
162 default_mm_loras = set()
163
164 for lora in self.models.lora_requests.values():
165 # Best effort match for default multimodal lora adapters;
166 # There is probably a better way to do this, but currently
167 # this matches against the set of 'types' in any content lists
168 # up until '_', e.g., to match audio_url -> audio
169 if lora.lora_name in message_types:
170 default_mm_loras.add(lora)
171
172 # Currently only support default modality specific loras if
173 # we have exactly one lora matched on the request.
174 if len(default_mm_loras) == 1:
175 return default_mm_loras.pop()
176 return None
177
178 def _maybe_get_adapters(
179 self,
180 request: AnyRequest | AnyPoolingRequest,
181 supports_default_mm_loras: bool = False,
182 ) -> LoRARequest | None:
183 if request.model in self.models.lora_requests:
184 return self.models.lora_requests[request.model]
185
186 # Currently only support default modality specific loras
187 # if we have exactly one lora matched on the request.
188 if supports_default_mm_loras:
189 default_mm_lora = self._get_active_default_mm_loras(request)
190 if default_mm_lora is not None:
191 return default_mm_lora
192
193 if self._is_model_supported(request.model):
194 return None
195
196 # if _check_model has been called earlier, this will be unreachable
197 raise VLLMNotFoundError(f"The model `{request.model}` does not exist.")
