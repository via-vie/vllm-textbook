50def build_per_request_timing_metrics(
51 metrics: RequestStateStats | None,
52 num_generation_tokens: int,
53) -> PerRequestMetrics:
54 """Build per-request timing metrics from ``RequestStateStats``.
55
56 ``generation_time_ms`` is the decode interval only (first output token to
57 last output token); it excludes both queue wait and prefill/TTFT.
58 ``tokens_per_second`` is overall output throughput: all generated tokens
59 over the inference interval (scheduling to last output token), so it counts
60 the prefill/TTFT phase and is not simply the reciprocal of ``mean_itl_ms``.
61 Each field is left ``None`` when the timestamps it depends on are
62 unavailable.
63 """
64 if metrics is None:
65 return PerRequestMetrics()
66
67 queued_ts = metrics.queued_ts
68 scheduled_ts = metrics.scheduled_ts
69 first_token_ts = metrics.first_token_ts
70 last_token_ts = metrics.last_token_ts
71
72 time_to_first_token_ms: float | None = None
73 generation_time_ms: float | None = None
74 queue_time_ms: float | None = None
75 mean_itl_ms: float | None = None
76 tokens_per_second: float | None = None
77
78 if scheduled_ts > 0 and first_token_ts > 0:
79 time_to_first_token_ms = (first_token_ts - scheduled_ts) * 1000
80
81 if first_token_ts > 0 and last_token_ts > 0:
82 generation_time_ms = (last_token_ts - first_token_ts) * 1000
83
84 if queued_ts > 0 and scheduled_ts > 0:
85 queue_time_ms = (scheduled_ts - queued_ts) * 1000
86
87 if first_token_ts > 0 and last_token_ts > 0 and num_generation_tokens > 1:
88 decode_time = last_token_ts - first_token_ts
89 mean_itl_ms = decode_time / (num_generation_tokens - 1) * 1000
90
91 if scheduled_ts > 0 and last_token_ts > 0:
92 inference_time_ms = (last_token_ts - scheduled_ts) * 1000
93 if inference_time_ms > 0:
94 tokens_per_second = num_generation_tokens / inference_time_ms * 1000
95
96 return PerRequestMetrics(
97 time_to_first_token_ms=time_to_first_token_ms,
98 generation_time_ms=generation_time_ms,
99 queue_time_ms=queue_time_ms,
100 mean_itl_ms=mean_itl_ms,
101 tokens_per_second=tokens_per_second,
102 )

105def build_spec_decoding_metrics(
106 final_res: "RequestOutput | None",
107) -> SpeculativeDecodingMetrics | None:
108 """Build per-request spec-decode acceptance metrics from the single output
109 sequence, or ``None`` when unavailable (metrics disabled, or no sequence
110 yet).
111
112 Only meaningful for single-sequence requests; callers suppress it for n>1.
113 """
114 if final_res is None or not final_res.outputs:
115 return None
116 metrics = final_res.outputs[0].spec_decode_metrics
117 if metrics is None:
118 return None
119 return SpeculativeDecodingMetrics(**metrics.to_dict())

289 async def _with_kv_transfer_rejection_cleanup(
290 self,
291 awaitable: Awaitable[_T],
292 request: ChatCompletionRequest | CompletionRequest | ResponsesRequest,
293 raw_request: Request | None,
294 ) -> _T:
295 """Wrap a `create_*` coroutine so that, if it raises or returns an
296 ErrorResponse (i.e. the request never reached the engine), the KV
297 connector is notified to free any pinned remote-prefill blocks."""
298 kv_transfer_params = self.has_kv_connector and request.kv_transfer_params
299 if not kv_transfer_params or not kv_transfer_params.get("do_remote_prefill"):
300 return await awaitable
301
302 notify = True
303 try:
304 result = await awaitable
305 if not isinstance(result, ErrorResponse):
306 notify = False
307 return result
308 finally:
309 if notify:
310 try:
311 await self.engine_client.notify_kv_transfer_request_rejected(
312 request.request_id,
313 kv_transfer_params,
314 data_parallel_rank=self._get_data_parallel_rank(raw_request),
315 )
316 except Exception:
317 logger.warning(
318 "Failed to notify KV connector about rejected request %s",
319 request.request_id,
320 exc_info=True,
321 )

348def resolve_token_id_placeholder(
349 token: str, tokenizer: TokenizerLike
350) -> tuple[str, list[int] | None]:
351 """Decode a 'token_id:N' placeholder back to a token string and UTF-8 bytes.
352
353 Returns (token, None) unchanged if token is not a placeholder.
354 This is the inverse of format_token_id_placeholder / _get_decoded_token
355 when return_as_token_id=True.
356 """
357 suffix = token.removeprefix("token_id:")
358 if suffix == token:
359 return token, None
360 try:
361 token_id = int(suffix)
362 except ValueError:
363 return token, None
364 token_repr = tokenizer.convert_ids_to_tokens([token_id])[0]
365 if token_repr is None:
366 logger.warning_once(
367 "resolve_token_id_placeholder: token_id %d has no vocab entry; "
368 "substituting empty string",
369 token_id,
370 )
371 return "", None
372 token_str = tokenizer.convert_tokens_to_string([token_repr])
373 return token_str, list(token_str.encode("utf-8", errors="replace"))

376def clamp_prompt_logprobs(
377 prompt_logprobs: PromptLogprobs | None,
378) -> PromptLogprobs | None:
379 if prompt_logprobs is None:
380 return prompt_logprobs
381
382 for logprob_dict in prompt_logprobs:
383 if logprob_dict is None:
384 continue
385 for logprob_values in logprob_dict.values():
386 if logprob_values.logprob == float("-inf"):
387 logprob_values.logprob = -9999.0
388 return prompt_logprobs
