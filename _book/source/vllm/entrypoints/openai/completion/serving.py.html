88 async def render_completion_request(
89 self,
90 request: CompletionRequest,
91 ) -> list[EngineInput] | ErrorResponse:
92 """
93 Validate the model and preprocess a completion request.
94
95 Delegates preprocessing logic to OnlineRenderer, adding the
96 engine-aware checks (LoRA model validation, engine health).
97
98 Returns:
99 A list of engine_inputs on success, or an ErrorResponse on failure.
100 """
101 error_check_ret = await self._check_model(request)

110 self,
111 request: CompletionRequest,
112 raw_request: Request | None = None,

114 """Completion API similar to OpenAI's API.
115
116 See https://platform.openai.com/docs/api-reference/completions/create

119 NOTE: Currently we do not support the following feature:

127 async def _create_completion(
128 self,
129 request: CompletionRequest,
130 raw_request: Request | None = None,
131 ) -> AsyncGenerator[str, None] | CompletionResponse | ErrorResponse:
132 if request.stream and request.use_beam_search:
133 return self.create_error_response(
134 "Streaming is not currently supported with beam search"
135 )
136
137 result = await self.render_completion_request(request)
138 if isinstance(result, ErrorResponse):
139 return result
140
141 engine_inputs = result
142
143 request_id = f"cmpl-{self._base_request_id(raw_request, request.request_id)}"
144 created_time = int(time.time())
145
146 request_metadata = RequestResponseMetadata(request_id=request_id)
147 if raw_request:
148 raw_request.state.request_metadata = request_metadata
149
150 lora_request = self._maybe_get_adapters(request)
151
152 # Extract data_parallel_rank from header (router can inject it)
153 data_parallel_rank = self._get_data_parallel_rank(raw_request)
154
155 # Schedule the request and get the result generator.
156 max_model_len = self.model_config.max_model_len
157 generators: list[AsyncGenerator[RequestOutput, None]] = []
158 for i, engine_input in enumerate(engine_inputs):
159 max_tokens = get_max_tokens(
160 max_model_len,
161 request.max_tokens,
162 self._extract_prompt_len(engine_input),
163 self.default_sampling_params,
164 self.override_max_tokens,
165 truncate_prompt_tokens=request.truncate_prompt_tokens,
166 )
167
168 sampling_params: SamplingParams | BeamSearchParams
169 if request.use_beam_search:
170 sampling_params = request.to_beam_search_params(
171 max_tokens, self.default_sampling_params
172 )
173 else:
174 sampling_params = request.to_sampling_params(
175 max_tokens,
176 self.default_sampling_params,
177 )
178
179 request_id_item = f"{request_id}-{i}"
180
181 self._log_inputs(
182 request_id_item,
183 engine_input,
184 params=sampling_params,
185 lora_request=lora_request,
186 )
187
188 trace_headers = (
189 None
190 if raw_request is None
191 else await self._get_trace_headers(raw_request.headers)
192 )
193 session_id = self._get_session_id(request, raw_request)
194
195 if isinstance(sampling_params, BeamSearchParams):
196 generator = self.beam_search(
197 prompt=engine_input,
198 request_id=request_id,
199 params=sampling_params,
200 lora_request=lora_request,
201 trace_headers=trace_headers,
202 session_id=session_id,
203 )
204 else:
205 generator = self.engine_client.generate(
206 engine_input,
207 sampling_params,
208 request_id_item,
209 lora_request=lora_request,
210 trace_headers=trace_headers,
211 priority=self._get_priority(request, raw_request),
212 data_parallel_rank=data_parallel_rank,
213 session_id=session_id,
214 )
215
216 generators.append(generator)
217
218 result_generator = merge_async_iterators(*generators)
219
220 model_name = self.models.model_name(lora_request)
221 num_prompts = len(engine_inputs)
222
223 # Streaming response
224 tokenizer = self.renderer.tokenizer
225
226 if request.stream:
227 return self.completion_stream_generator(
228 request,
229 engine_inputs,
230 result_generator,
231 request_id,
232 created_time,
233 model_name,
234 num_prompts=num_prompts,
235 tokenizer=tokenizer,
236 request_metadata=request_metadata,
237 )
238
239 # Non-streaming response
240 final_res_batch: list[RequestOutput | None] = [None] * num_prompts
241 try:
242 async for i, res in result_generator:
243 final_res_batch[i] = res
244
245 for i, final_res in enumerate(final_res_batch):
246 assert final_res is not None
247
248 # The output should contain the input text
249 # We did not pass it into vLLM engine to avoid being redundant
250 # with the inputs token IDs
251 if final_res.prompt is None:
252 final_res.prompt = self._extract_prompt_text(engine_inputs[i])
253
254 final_res_batch_checked = cast(list[RequestOutput], final_res_batch)
255
256 response = self.request_output_to_completion_response(
257 final_res_batch_checked,
258 request,
259 request_id,
260 created_time,
261 model_name,
262 tokenizer,
263 request_metadata,
264 )
265 except asyncio.CancelledError:
266 return self.create_error_response("Client disconnected")
267
268 # When user requests streaming but we don't stream, we still need to
269 # return a streaming response with a single event.
270 if request.stream:
271 response_json = response.model_dump_json()
272
273 async def fake_stream_generator() -> AsyncGenerator[str, None]:
274 yield f"data: {response_json}\n\n"
275 yield "data: [DONE]\n\n"
276
277 return fake_stream_generator()
278
279 return response

281 async def completion_stream_generator(
282 self,
283 request: CompletionRequest,
284 engine_inputs: list[EngineInput],
285 result_generator: AsyncIterator[tuple[int, RequestOutput]],
286 request_id: str,
287 created_time: int,
288 model_name: str,
289 num_prompts: int,
290 tokenizer: TokenizerLike | None,
291 request_metadata: RequestResponseMetadata,
292 ) -> AsyncGenerator[str, None]:
293 num_choices = 1 if request.n is None else request.n
294 previous_text_lens = [0] * num_choices * num_prompts
295 previous_num_tokens = [0] * num_choices * num_prompts
296 has_echoed = [False] * num_choices * num_prompts
297 num_prompt_tokens = [0] * num_prompts
298 num_cached_tokens = None
299 first_iteration = True
300
301 stream_options = request.stream_options
302 include_usage, include_continuous_usage = should_include_usage(
303 stream_options, self.enable_force_include_usage
304 )
305
306 last_res: RequestOutput | None = None
307 try:
308 async for prompt_idx, res in result_generator:
309 last_res = res
310 prompt_token_ids = res.prompt_token_ids
311 prompt_logprobs = res.prompt_logprobs
312
313 if first_iteration:
314 num_cached_tokens = res.num_cached_tokens
315 first_iteration = False
316
317 prompt_text = res.prompt
318 if prompt_text is None:
319 engine_input = engine_inputs[prompt_idx]
320 prompt_text = self._extract_prompt_text(engine_input)
321
322 # Prompt details are excluded from later streamed outputs
323 if prompt_token_ids is not None:
324 num_prompt_tokens[prompt_idx] = len(prompt_token_ids)
325
326 delta_token_ids: GenericSequence[int]
327 out_logprobs: GenericSequence[dict[int, Logprob] | None] | None
328
329 for output in res.outputs:
330 i = output.index + prompt_idx * num_choices
331
332 # Useful when request.return_token_ids is True
333 # Returning prompt token IDs shares the same logic
334 # with the echo implementation.
335 prompt_token_ids_to_return: list[int] | None = None
336
337 assert request.max_tokens is not None
338 if request.echo and not has_echoed[i]:
339 assert prompt_token_ids is not None
340 if request.return_token_ids:
341 prompt_text = ""
342 assert prompt_text is not None
343 if request.max_tokens == 0:
344 # only return the prompt
345 delta_text = prompt_text
346 delta_token_ids = prompt_token_ids
347 out_logprobs = prompt_logprobs
348 else:
349 # echo the prompt and first token
350 delta_text = prompt_text + output.text
351 delta_token_ids = [
352 *prompt_token_ids,
353 *output.token_ids,
354 ]
355 out_logprobs = [
356 *(prompt_logprobs or []),
357 *(output.logprobs or []),
358 ]
359 prompt_token_ids_to_return = prompt_token_ids
360 has_echoed[i] = True
361 else:
362 # return just the delta
363 delta_text = output.text
364 delta_token_ids = output.token_ids
365 out_logprobs = output.logprobs
366
367 # has_echoed[i] is reused here to indicate whether
368 # we have already returned the prompt token IDs.
369 if not has_echoed[i] and request.return_token_ids:
370 prompt_token_ids_to_return = prompt_token_ids
371 has_echoed[i] = True
372
373 if (
374 not delta_text
375 and not delta_token_ids
376 and not previous_num_tokens[i]
377 ):
378 # Chunked prefill case, don't return empty chunks
379 continue
380
381 if request.logprobs is not None:
382 assert out_logprobs is not None, "Did not output logprobs"
383 logprobs = self._create_completion_logprobs(
384 token_ids=delta_token_ids,
385 top_logprobs=out_logprobs,
386 num_output_top_logprobs=request.logprobs,
387 tokenizer=tokenizer,
388 logprob_token_ids=request.logprob_token_ids,
389 initial_text_offset=previous_text_lens[i],
390 return_as_token_id=request.return_tokens_as_token_ids,
391 )
392 else:
393 logprobs = None
394
395 previous_text_lens[i] += len(delta_text)
396 previous_num_tokens[i] += len(output.token_ids)
397 finish_reason = output.finish_reason
398 stop_reason = output.stop_reason
399
400 self._raise_if_error(finish_reason, request_id)
401
402 chunk = CompletionStreamResponse(
403 id=request_id,
404 object="text_completion",
405 created=created_time,
406 model=model_name,
407 choices=[
408 CompletionResponseStreamChoice(
409 index=i,
410 text=delta_text,
411 logprobs=logprobs,
412 finish_reason=finish_reason,
413 stop_reason=stop_reason,
414 prompt_token_ids=prompt_token_ids_to_return,
415 token_ids=(
416 as_list(output.token_ids)
417 if request.return_token_ids
418 else None
419 ),
420 )
421 ],
422 )
423 # Stamp on terminal chunk only when no trailing usage chunk
424 # will follow (that one is the true final message).
425 if (
426 not include_usage
427 and self.system_fingerprint is not None
428 and finish_reason is not None
429 ):
430 chunk.system_fingerprint = self.system_fingerprint
431 if include_continuous_usage:
432 prompt_tokens = num_prompt_tokens[prompt_idx]
433 completion_tokens = previous_num_tokens[i]
434 chunk.usage = UsageInfo(
435 prompt_tokens=prompt_tokens,
436 completion_tokens=completion_tokens,
437 total_tokens=prompt_tokens + completion_tokens,
438 )
439
440 response_json = chunk.model_dump_json(exclude_unset=True)
441 yield f"data: {response_json}\n\n"
442
443 total_prompt_tokens = sum(num_prompt_tokens)
444 total_completion_tokens = sum(previous_num_tokens)
445 final_usage_info = UsageInfo(
446 prompt_tokens=total_prompt_tokens,
447 completion_tokens=total_completion_tokens,
448 total_tokens=total_prompt_tokens + total_completion_tokens,
449 )
450
451 if self.enable_prompt_tokens_details and num_cached_tokens is not None:
452 final_usage_info.prompt_tokens_details = PromptTokenUsageInfo(
453 cached_tokens=num_cached_tokens
454 )
455
456 if include_usage:
457 # In streaming, metrics ride on this final usage chunk, which is
458 # only emitted when usage reporting is enabled (i.e.
459 # ``stream_options.include_usage=true`` or
460 # ``--enable-force-include-usage``).
461 stream_per_request_metrics: PerRequestMetrics | None = None
462 # See note in request_output_to_completion_response: suppress when
463 # not attributable to one stream (multi-prompt or n>1).
464 if num_prompts == 1 and (request.n or 1) == 1:
465 if self.enable_per_request_metrics:
466 last_metrics = (
467 last_res.metrics if last_res is not None else None
468 )
469 stream_per_request_metrics = build_per_request_timing_metrics(
470 last_metrics, total_completion_tokens
471 )
472 spec_stats = build_spec_decoding_metrics(last_res)
473 if spec_stats is not None:
474 if stream_per_request_metrics is None:
475 stream_per_request_metrics = PerRequestMetrics()
476 stream_per_request_metrics.speculative_decoding = spec_stats
477
478 final_usage_chunk = CompletionStreamResponse(
479 id=request_id,
480 created=created_time,
481 model=model_name,
482 choices=[],
483 usage=final_usage_info,
484 system_fingerprint=self.system_fingerprint,
485 metrics=stream_per_request_metrics,
486 )
487 final_usage_data = final_usage_chunk.model_dump_json(
488 exclude_unset=False, exclude_none=True
489 )
490 yield f"data: {final_usage_data}\n\n"
491
492 # report to FastAPI middleware aggregate usage across all choices
493 request_metadata.final_usage_info = final_usage_info
494
495 except GenerationError as e:
496 yield f"data: {self._convert_generation_error_to_streaming_response(e)}\n\n"
497 except Exception as e:
498 logger.exception("Error in completion stream generator.")
499 data = self.create_streaming_error_response(e)
500 yield f"data: {data}\n\n"
501 yield "data: [DONE]\n\n"

503 def request_output_to_completion_response(
504 self,
505 final_res_batch: list[RequestOutput],
506 request: CompletionRequest,
507 request_id: str,
508 created_time: int,
509 model_name: str,
510 tokenizer: TokenizerLike | None,
511 request_metadata: RequestResponseMetadata,
512 ) -> CompletionResponse:
513 choices: list[CompletionResponseChoice] = []
514 num_prompt_tokens = 0
515 num_generated_tokens = 0
516 kv_transfer_params = None
517 ec_transfer_params = None
518 last_final_res = None
519 for final_res in final_res_batch:
520 last_final_res = final_res
521 prompt_token_ids = final_res.prompt_token_ids
522 assert prompt_token_ids is not None
523 prompt_logprobs = clamp_prompt_logprobs(final_res.prompt_logprobs)
524 prompt_text = final_res.prompt
525
526 token_ids: GenericSequence[int]
527 out_logprobs: GenericSequence[dict[int, Logprob] | None] | None
528
529 for output in final_res.outputs:
530 self._raise_if_error(output.finish_reason, request_id)
531
532 assert request.max_tokens is not None
533 if request.echo:
534 if request.return_token_ids:
535 prompt_text = ""
536 assert prompt_text is not None
537 if request.max_tokens == 0:
538 token_ids = prompt_token_ids
539 out_logprobs = prompt_logprobs
540 output_text = prompt_text
541 else:
542 token_ids = [*prompt_token_ids, *output.token_ids]
543
544 if request.logprobs is None:
545 out_logprobs = None
546 else:
547 assert prompt_logprobs is not None
548 assert output.logprobs is not None
549 out_logprobs = [
550 *prompt_logprobs,
551 *output.logprobs,
552 ]
553
554 output_text = prompt_text + output.text
555 else:
556 token_ids = output.token_ids
557 out_logprobs = output.logprobs
558 output_text = output.text
559
560 if request.logprobs is not None:
561 assert out_logprobs is not None, "Did not output logprobs"
562 logprobs = self._create_completion_logprobs(
563 token_ids=token_ids,
564 top_logprobs=out_logprobs,
565 tokenizer=tokenizer,
566 num_output_top_logprobs=request.logprobs,
567 logprob_token_ids=request.logprob_token_ids,
568 return_as_token_id=request.return_tokens_as_token_ids,
569 )
570 else:
571 logprobs = None
572
573 routed_experts_b64 = (
574 numpy2base64(output.routed_experts)
575 if output.routed_experts is not None
576 else None
577 )
578
579 choice_data = CompletionResponseChoice(
580 index=len(choices),
581 text=output_text,
582 logprobs=logprobs,
583 finish_reason=output.finish_reason,
584 stop_reason=output.stop_reason,
585 prompt_logprobs=final_res.prompt_logprobs,
586 prompt_token_ids=(
587 prompt_token_ids if request.return_token_ids else None
588 ),
589 token_ids=(
590 as_list(output.token_ids) if request.return_token_ids else None
591 ),
592 routed_experts=routed_experts_b64,
593 )
594 choices.append(choice_data)
595
596 num_generated_tokens += len(output.token_ids)
597
598 num_prompt_tokens += len(prompt_token_ids)
599
600 usage = UsageInfo(
601 prompt_tokens=num_prompt_tokens,
602 completion_tokens=num_generated_tokens,
603 total_tokens=num_prompt_tokens + num_generated_tokens,
604 )
605
606 if (
607 self.enable_prompt_tokens_details
608 and last_final_res
609 and last_final_res.num_cached_tokens is not None
610 ):
611 usage.prompt_tokens_details = PromptTokenUsageInfo(
612 cached_tokens=last_final_res.num_cached_tokens
613 )
614
615 request_metadata.final_usage_info = usage
616
617 per_request_metrics: PerRequestMetrics | None = None
618 # Per-request metrics (timing + spec-decode acceptance) describe a single
619 # generation stream, so suppress them when they cannot be attributed to
620 # one: multiple prompts (timestamps span prompts) or n>1 (stats belong to
621 # one of the n sequences).
622 if len(final_res_batch) == 1 and (request.n or 1) == 1:
623 if self.enable_per_request_metrics:
624 last_metrics = (
625 last_final_res.metrics if last_final_res is not None else None
626 )
627 per_request_metrics = build_per_request_timing_metrics(
628 last_metrics, num_generated_tokens
629 )
630 spec_stats = build_spec_decoding_metrics(last_final_res)
631 if spec_stats is not None:
632 if per_request_metrics is None:
633 per_request_metrics = PerRequestMetrics()
634 per_request_metrics.speculative_decoding = spec_stats
635
636 if final_res_batch:
637 kv_transfer_params = final_res_batch[0].kv_transfer_params
638 ec_transfer_params = final_res_batch[0].ec_transfer_params
639
640 return CompletionResponse(
641 id=request_id,
642 created=created_time,
643 model=model_name,
644 choices=choices,
645 usage=usage,
646 system_fingerprint=self.system_fingerprint,
647 kv_transfer_params=kv_transfer_params,
648 ec_transfer_params=ec_transfer_params,
649 metrics=per_request_metrics,
650 )

652 def _create_completion_logprobs(
653 self,
654 token_ids: GenericSequence[int],
655 top_logprobs: GenericSequence[dict[int, Logprob] | None],
656 num_output_top_logprobs: int,
657 tokenizer: TokenizerLike | None,
658 logprob_token_ids: list[int] | None = None,
659 initial_text_offset: int = 0,
660 return_as_token_id: bool | None = None,
661 ) -> CompletionLogProbs:
662 """Create logprobs for OpenAI Completion API."""
663 out_text_offset: list[int] = []
664 out_token_logprobs: list[float | None] = []
665 out_tokens: list[str] = []
666 out_top_logprobs: list[dict[str, float] | None] = []
667
668 last_token_len = 0
669
670 should_return_as_token_id = (
671 return_as_token_id
672 if return_as_token_id is not None
673 else self.return_tokens_as_token_ids
674 )
675 for i, token_id in enumerate(token_ids):
676 step_top_logprobs = top_logprobs[i]
677 if step_top_logprobs is None:
678 if should_return_as_token_id:
679 token = format_token_id_placeholder(token_id)
680 else:
681 if tokenizer is None:
682 raise VLLMValidationError(
683 "Unable to get tokenizer because "
684 "`skip_tokenizer_init=True`",
685 parameter="skip_tokenizer_init",
686 value=True,
687 )
688
689 token = tokenizer.decode(token_id)
690
691 out_tokens.append(token)
692 out_token_logprobs.append(None)
693 out_top_logprobs.append(None)
694 else:
695 step_token = step_top_logprobs[token_id]
696
697 token = self._get_decoded_token(
698 step_token,
699 token_id,
700 tokenizer,
701 return_as_token_id=should_return_as_token_id,
702 )
703 token_logprob = max(step_token.logprob, -9999.0)
704
705 out_tokens.append(token)
706 out_token_logprobs.append(token_logprob)
707
708 # makes sure to add the top num_output_top_logprobs + 1
709 # logprobs, as defined in the openai API
710 # (cf. https://github.com/openai/openai-openapi/blob/
711 # 893ba52242dbd5387a97b96444ee1c742cfce9bd/openapi.yaml#L7153)
712 out_top_logprobs.append(
713 {
714 # Convert float("-inf") to the
715 # JSON-serializable float that OpenAI uses
716 self._get_decoded_token(
717 top_lp[1],
718 top_lp[0],
719 tokenizer,
720 return_as_token_id=should_return_as_token_id,
