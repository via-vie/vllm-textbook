73def _get_mm_token_counts(engine_input: EngineInput) -> dict[str, int]:
74 """Sum per-modality placeholder tokens from ``mm_placeholders``.
75
76 Keyed by modality name; ``PlaceholderRange.length`` is the placeholder's
77 prompt token span, so each sum matches the placeholder tokens already
78 counted in ``usage.prompt_tokens``.
79 """
80 mm_placeholders = cast(
81 MultiModalPlaceholders | None, engine_input.get("mm_placeholders")
82 )
83 return {
84 modality: sum(p.length for p in ranges)
85 for modality, ranges in (mm_placeholders or {}).items()
86 if ranges
87 }

201 def _engine_chat_template_kwargs(
202 self, chat_template_kwargs: dict[str, Any]
203 ) -> dict[str, Any]:
204 """Subclass hook to narrow ``chat_template_kwargs`` for the engine.
205
206 The same dict is used twice: to build the API-server-side parser
207 instance, and to populate
208 ``EngineCoreRequest.reasoning_parser_kwargs`` for the engine-core
209 side. The latter crosses ZMQ as msgpack, so a handler that stashes
210 request-scoped state only the API-server-side parser needs (values
211 msgpack can't encode, or payloads not worth shipping) can drop
212 those entries here without affecting the in-process parser.
213
214 Must not mutate the argument -- the caller still needs the full
215 dict. The default forwards it unchanged.
216 """
217 return chat_template_kwargs

258 async def _create_chat_completion(
259 self,
260 request: ChatCompletionRequest,
261 raw_request: Request | None = None,
262 ) -> AsyncGenerator[str, None] | ChatCompletionResponse | ErrorResponse:
263 # Streaming response
264 tokenizer = self.renderer.tokenizer
265 assert tokenizer is not None
266 chat_template_kwargs = self._effective_chat_template_kwargs(request)
267 parser: Parser | None = None
268 if self.parser_cls is not None:
269 parser = self.parser_cls(
270 tokenizer,
271 request.tools,
272 chat_template_kwargs=chat_template_kwargs,
273 model_config=self.model_config,
274 )
275 result = await self.render_chat_request(request)
276 if isinstance(result, ErrorResponse):
277 return result
278
279 conversation, engine_inputs = result
280
281 request_id = (
282 f"chatcmpl-{self._base_request_id(raw_request, request.request_id)}"
283 )
284
285 request_metadata = RequestResponseMetadata(request_id=request_id)
286 if raw_request:
287 raw_request.state.request_metadata = request_metadata
288
289 lora_request = self._maybe_get_adapters(request, supports_default_mm_loras=True)
290
291 model_name = self.models.model_name(lora_request)
292
293 # Extract data_parallel_rank from header (router can inject it)
294 data_parallel_rank = self._get_data_parallel_rank(raw_request)
295
296 # Schedule the request and get the result generator.
297 max_model_len = self.model_config.max_model_len
298 generators: list[AsyncGenerator[RequestOutput, None]] = []
299 mm_token_counts: dict[str, int] | None = None
300 for i, engine_input in enumerate(engine_inputs):
301 prompt_token_ids = self._extract_prompt_components(engine_input).token_ids
302 mm_token_counts = _get_mm_token_counts(engine_input)
303
304 # If we are creating sub requests for multiple prompts, ensure that they
305 # have unique request ids.
306 sub_request_id = (
307 request_id if len(engine_inputs) == 1 else f"{request_id}_{i}"
308 )
309
310 max_tokens = get_max_tokens(
311 max_model_len,
312 request.max_completion_tokens
313 if request.max_completion_tokens is not None
314 else request.max_tokens,
315 self._extract_prompt_len(engine_input),
316 self.default_sampling_params,
317 self.override_max_tokens,
318 truncate_prompt_tokens=request.truncate_prompt_tokens,
319 )
320
321 sampling_params: SamplingParams | BeamSearchParams
322 if request.use_beam_search:
323 sampling_params = request.to_beam_search_params(
324 max_tokens, self.default_sampling_params
325 )
326 else:
327 sampling_params = request.to_sampling_params(
328 max_tokens,
329 self.default_sampling_params,
330 )
331
332 self._log_inputs(
333 sub_request_id,
334 engine_input,
335 params=sampling_params,
336 lora_request=lora_request,
337 )
338
339 trace_headers = (
340 None
341 if raw_request is None
342 else await self._get_trace_headers(raw_request.headers)
343 )
344 session_id = self._get_session_id(request, raw_request)
345
346 if isinstance(sampling_params, BeamSearchParams):
347 generator = self.beam_search(
348 prompt=engine_input,
349 request_id=sub_request_id,
350 params=sampling_params,
351 lora_request=lora_request,
352 trace_headers=trace_headers,
353 session_id=session_id,
354 )
355 else:
356 if not request.include_reasoning:
357 reasoning_ended = True
358 elif request._grammar_from_parser:
359 # The Mistral grammar already includes an optional
360 # `think?` rule that handles both reasoning and
361 # non-reasoning outputs.
362 reasoning_ended = True
363 elif parser is not None and parser.reasoning_parser is not None:
364 reasoning_ended = parser.is_reasoning_end(prompt_token_ids or [])
365 else:
366 reasoning_ended = None
367
368 generator = self.engine_client.generate(
369 engine_input,
370 sampling_params,
371 sub_request_id,
372 lora_request=lora_request,
373 trace_headers=trace_headers,
374 priority=self._get_priority(request, raw_request),
375 data_parallel_rank=data_parallel_rank,
376 session_id=session_id,
377 reasoning_ended=reasoning_ended,
378 reasoning_parser_kwargs={
379 "chat_template_kwargs": self._engine_chat_template_kwargs(
380 chat_template_kwargs
381 ),
382 }
383 if parser is not None and parser.reasoning_parser is not None
384 else None,
385 )
386
387 generators.append(generator)
388
389 assert len(generators) == 1
390 (result_generator,) = generators
391
392 if request.stream:
393 return self.chat_completion_stream_generator(
394 request,
395 result_generator,
396 request_id,
397 model_name,
398 conversation,
399 tokenizer,
400 request_metadata,
401 chat_template_kwargs=chat_template_kwargs,
402 mm_token_counts=mm_token_counts,
403 )
404
405 return await self.chat_completion_full_generator(
406 request,
407 result_generator,
408 request_id,
409 model_name,
410 conversation,
411 tokenizer,
412 request_metadata,
413 parser=parser,
414 mm_token_counts=mm_token_counts,
415 )

450 async def chat_completion_stream_generator(
451 self,
452 request: ChatCompletionRequest,
453 result_generator: AsyncIterator[RequestOutput],
454 request_id: str,
455 model_name: str,
456 conversation: list[ConversationMessage],
457 tokenizer: TokenizerLike,
458 request_metadata: RequestResponseMetadata,
459 chat_template_kwargs: dict[str, Any] | None = None,
460 mm_token_counts: dict[str, int] | None = None,
461 ) -> AsyncGenerator[str, None]:
462 created_time = int(time.time())
463 chunk_object_type: Final = "chat.completion.chunk"
464 first_iteration = True
465
466 # Send response for each token for each request.n (index)
467 num_choices = 1 if request.n is None else request.n
468 previous_num_tokens = [0] * num_choices
469 # TODO: Remove once all reasoning parsers use the Parser Engine.
470 generated_token_ids: list[list[int]] = [[] for _ in range(num_choices)]
471 previous_reasoning_tokens = [0] * num_choices
472 finish_reason_sent = [False] * num_choices
473 num_prompt_tokens = 0
474 num_cached_tokens = None
475 num_cache_creation_tokens = None
476 tools_streamed = [False] * num_choices
477
478 if isinstance(request.tool_choice, ChatCompletionNamedToolChoiceParam):
479 tool_choice_function_name = request.tool_choice.function.name
480 else:
481 tool_choice_function_name = None
482
483 previous_texts = [""] * num_choices
484
485 try:
486 if self.parser_cls is not None:
487 if tokenizer is None:
488 raise ValueError(
489 "Tokenizer not available when `skip_tokenizer_init=True`"
490 )
491 parsers: list[Parser | None] = [
492 self.parser_cls(
493 tokenizer,
494 request.tools,
495 chat_template_kwargs=chat_template_kwargs,
496 model_config=self.model_config,
497 )
498 for _ in range(num_choices)
499 ]
500 else:
501 parsers = [None] * num_choices
502 except Exception as e:
503 logger.exception("Error in parser creation.")
504 data = self.create_streaming_error_response(e)
505 yield f"data: {data}\n\n"
506 yield "data: [DONE]\n\n"
507 return
508
509 stream_options = request.stream_options
510 include_usage, include_continuous_usage = should_include_usage(
511 stream_options, self.enable_force_include_usage
512 )
513
514 last_res: RequestOutput | None = None
515 try:
516 async for res in result_generator:
517 last_res = res
518 if res.prompt_token_ids is not None:
519 num_prompt_tokens = len(res.prompt_token_ids)
520 if res.encoder_prompt_token_ids is not None:
521 num_prompt_tokens += len(res.encoder_prompt_token_ids)
522
523 # We need to do it here, because if there are exceptions in
524 # the result_generator, it needs to be sent as the FIRST
525 # response (by the try...catch).
526 if first_iteration:
527 num_cached_tokens = res.num_cached_tokens
528 num_cache_creation_tokens = res.num_cache_creation_tokens
529 # Send first response for each request.n (index) with
530 # the role
531 role = self.get_chat_request_role(request)
532
533 # ``res.prompt`` is the rendered chat-templated prompt
534 prompt_text = res.prompt if request.return_prompt_text else None
535
536 # NOTE num_choices defaults to 1 so this usually executes
537 # once per request
538 for i in range(num_choices):
539 choice_data = ChatCompletionResponseStreamChoice(
540 index=i,
541 delta=DeltaMessage(
542 role=role,
543 content="",
544 ),
545 logprobs=None,
546 finish_reason=None,
547 )
548
549 # return prompt_token_ids at the first chunk ever
550 chunk = ChatCompletionStreamResponse(
551 id=request_id,
552 object=chunk_object_type,
553 created=created_time,
554 choices=[choice_data],
555 model=model_name,
556 prompt_token_ids=(
557 res.prompt_token_ids
558 if request.return_token_ids
559 else None
560 ),
561 prompt_text=prompt_text,
562 )
563
564 # if continuous usage stats are requested, add it
565 if include_continuous_usage:
566 chunk.usage = UsageInfo(
567 prompt_tokens=num_prompt_tokens,
568 completion_tokens=0,
569 total_tokens=num_prompt_tokens,
570 completion_tokens_details=(
571 _make_completion_tokens_details(0)
572 if self._include_reasoning_tokens_details
573 else None
574 ),
575 )
576
577 data = chunk.model_dump_json(exclude_unset=True)
578 yield f"data: {data}\n\n"
579
580 # Send response to echo the input portion of the
581 # last message
582 if request.echo:
583 last_msg_content: str | list[dict[str, str]] = ""
584 if (
585 conversation
586 and "content" in conversation[-1]
587 and conversation[-1].get("role") == role
588 ):
589 last_msg_content = conversation[-1]["content"] or ""
590
591 if last_msg_content:
592 for i in range(num_choices):
593 choice_data = ChatCompletionResponseStreamChoice(
594 index=i,
595 delta=DeltaMessage(content=last_msg_content),
596 logprobs=None,
597 finish_reason=None,
598 )
599 chunk = ChatCompletionStreamResponse(
600 id=request_id,
601 object=chunk_object_type,
602 created=created_time,
603 choices=[choice_data],
604 model=model_name,
605 )
606 if include_continuous_usage:
607 chunk.usage = UsageInfo(
608 prompt_tokens=num_prompt_tokens,
609 completion_tokens=0,
610 total_tokens=num_prompt_tokens,
611 completion_tokens_details=(
612 _make_completion_tokens_details(0)
613 if self._include_reasoning_tokens_details
614 else None
615 ),
616 )
617
618 data = chunk.model_dump_json(exclude_unset=True)
619 yield f"data: {data}\n\n"
620 first_iteration = False
621
622 for output in res.outputs:
623 i = output.index
624 parser = parsers[i]
625 if finish_reason_sent[i]:
626 continue
627
628 if request.logprobs and (
629 request.top_logprobs is not None or request.logprob_token_ids
630 ):
631 assert output.logprobs is not None, "Did not output logprobs"
632 logprobs = self._create_chat_logprobs(
633 token_ids=output.token_ids,
634 top_logprobs=output.logprobs,
635 tokenizer=tokenizer,
636 num_output_top_logprobs=request.top_logprobs,
637 logprob_token_ids=request.logprob_token_ids,
638 return_as_token_id=request.return_tokens_as_token_ids,
639 )
640 else:
641 logprobs = None
642
643 delta_text = output.text
644
645 if (
646 not delta_text
647 and not output.token_ids
648 and not previous_num_tokens[i]
649 ):
650 # Chunked prefill case, don't return empty chunks
651 continue
652
653 delta_message: DeltaMessage | None
654
655 if parser is not None:
656 delta_message = parser.parse_delta(
657 delta_text=delta_text,
658 delta_token_ids=as_list(output.token_ids),
659 request=request,
660 prompt_token_ids=res.prompt_token_ids,
661 finished=output.finish_reason is not None,
662 )
663 if delta_message is not None and delta_message.tool_calls:
664 tools_streamed[i] = True
665
666 # handle streaming just a content delta (no parsers)
667 else:
668 delta_message = DeltaMessage(content=delta_text)
669
670 previous_texts[i] += delta_text
671
672 # set the previous values for the next iteration
673 previous_num_tokens[i] += len(output.token_ids)
674 if parser is not None:
675 generated_token_ids[i].extend(output.token_ids)
676 previous_reasoning_tokens[i] = parser.count_reasoning_tokens(
677 tuple(generated_token_ids[i])
678 )
679
680 # if the message delta is None (e.g. because it was a
681 # "control token" for tool calls or the parser otherwise
682 # wasn't ready to send a token, then
683 # get the next token without streaming a chunk
684 # When reasoning is hidden, suppress per-token
685 # metadata (logprobs, token_ids) on every chunk to
686 # prevent leaking reasoning tokens through decoded
687 # token text in logprob entries or raw token IDs.
688 hide_stream_metadata = (
689 not request.include_reasoning and parser is not None
690 )
691 if hide_stream_metadata:
692 logprobs = None
693
694 if delta_message is None:
695 # NOTE: If return_token_ids is enabled, we still need to
696 # send a chunk with token_ids even if delta_message is None
697 # to ensure all tokens are included in the response
698 if output.finish_reason is None and (
699 not request.return_token_ids or hide_stream_metadata
700 ):
701 continue
702 delta_message = DeltaMessage()
703
704 # Log streaming delta if output logging is enabled
705 if self.enable_log_outputs and self.request_logger:
706 delta_content_parts = []
707 if delta_message.content:
708 delta_content_parts.append(delta_message.content)
709 if delta_message.reasoning:
710 reasoning = delta_message.reasoning
711 delta_content_parts.append(f"[reasoning: {reasoning}]")
712 if delta_message.tool_calls:
713 tool_args = "".join(
714 tc.function.arguments
715 for tc in delta_message.tool_calls
716 if tc.function and tc.function.arguments
717 )
718 if tool_args:
719 delta_content_parts.append(f"[tool_calls: {tool_args}]")
720
721 if delta_content_parts and self.enable_log_deltas:
722 delta_content = " ".join(delta_content_parts)
723 self.request_logger.log_outputs(
724 request_id=request_id,
725 outputs=delta_content,
726 output_token_ids=as_list(output.token_ids),
727 finish_reason=output.finish_reason,
728 is_streaming=True,
729 delta=True,
730 )
731
732 include_token_ids = (
733 request.return_token_ids and not hide_stream_metadata
734 )
735
736 if output.finish_reason is None:
737 # Send token-by-token response for each request.n
738 choice_data = ChatCompletionResponseStreamChoice(
739 index=i,
740 delta=delta_message,
741 logprobs=logprobs,
742 finish_reason=None,
743 token_ids=(
744 as_list(output.token_ids) if include_token_ids else None
745 ),
746 )
747
748 # if the model is finished generating
749 else:
750 # check for error finish reason and abort streaming
751 # finish_reason='error' indicates a retryable error
752 self._raise_if_error(output.finish_reason, request_id)
753
754 # Send the finish response for each request.n only once
755 # In OpenAI's API, when a tool is called, the
756 # finish_reason is:
757 # "tool_calls" for "auto" or "required" tool calls,
758 # and "stop" for named tool calls.
759 if tools_streamed[i] and not tool_choice_function_name:
760 finish_reason_ = "tool_calls"
761 else:
762 finish_reason_ = (
763 output.finish_reason if output.finish_reason else "stop"
764 )
765 choice_data = ChatCompletionResponseStreamChoice(
766 index=i,
767 delta=delta_message,
768 logprobs=logprobs,
769 finish_reason=finish_reason_,
770 stop_reason=output.stop_reason,
771 token_ids=(
772 as_list(output.token_ids) if include_token_ids else None
773 ),
774 )
775
776 finish_reason_sent[i] = True
777
778 choice_data = maybe_filter_parallel_tool_calls(choice_data, request)
779 chunk = ChatCompletionStreamResponse(
780 id=request_id,
781 object=chunk_object_type,
782 created=created_time,
783 choices=[choice_data],
784 model=model_name,
785 )
786 # Stamp the fingerprint on terminal chunks only (those with
787 # finish_reason set). When ``include_usage`` is on, the
788 # trailing usage chunk below overrides this as the true
789 # final message.
790 if (
791 not include_usage
792 and self.system_fingerprint is not None
793 and choice_data.finish_reason is not None
794 ):
795 chunk.system_fingerprint = self.system_fingerprint
796
797 # handle usage stats if requested & if continuous
798 if include_continuous_usage:
799 completion_tokens = previous_num_tokens[i]
800 chunk.usage = UsageInfo(
801 prompt_tokens=num_prompt_tokens,
802 completion_tokens=completion_tokens,
803 total_tokens=num_prompt_tokens + completion_tokens,
804 completion_tokens_details=(
805 _make_completion_tokens_details(
806 previous_reasoning_tokens[i]
807 )
808 if self._include_reasoning_tokens_details
809 else None
810 ),
811 )
812
813 data = chunk.model_dump_json(exclude_unset=True)
814 yield f"data: {data}\n\n"
815
816 # once the final token is handled, if stream_options.include_usage
817 # is sent, send the usage
818 if include_usage:
819 completion_tokens = sum(previous_num_tokens)
820 final_usage = UsageInfo(
821 prompt_tokens=num_prompt_tokens,
822 completion_tokens=completion_tokens,
823 total_tokens=num_prompt_tokens + completion_tokens,
824 completion_tokens_details=_make_completion_tokens_details(
825 sum(previous_reasoning_tokens)
826 )
827 if self._include_reasoning_tokens_details
828 else None,
829 )
830 final_usage.prompt_tokens_details = _make_prompt_tokens_details(
831 self.enable_prompt_tokens_details,
832 num_cached_tokens,
833 num_cache_creation_tokens,
834 mm_token_counts,
835 )
836
837 # In streaming, metrics ride on this final usage chunk, which is
838 # only emitted when usage reporting is enabled (i.e.
839 # ``stream_options.include_usage=true`` or
840 # ``--enable-force-include-usage``).
841 stream_per_request_metrics: PerRequestMetrics | None = None
842 # See note in chat_completion_full_generator: suppress for n>1.
843 if (request.n or 1) == 1:
844 if self.enable_per_request_metrics:
845 last_metrics = (
846 last_res.metrics if last_res is not None else None
847 )
848 stream_per_request_metrics = build_per_request_timing_metrics(
849 last_metrics, completion_tokens
850 )
851 spec_stats = build_spec_decoding_metrics(last_res)
852 if spec_stats is not None:
853 if stream_per_request_metrics is None:
854 stream_per_request_metrics = PerRequestMetrics()
855 stream_per_request_metrics.speculative_decoding = spec_stats
856
857 final_usage_chunk = ChatCompletionStreamResponse(
858 id=request_id,
859 object=chunk_object_type,
860 created=created_time,
861 choices=[],
862 model=model_name,
863 usage=final_usage,
864 system_fingerprint=self.system_fingerprint,
865 metrics=stream_per_request_metrics,
866 )
867 final_usage_data = final_usage_chunk.model_dump_json(
868 exclude_unset=True, exclude_none=True
869 )
870 yield f"data: {final_usage_data}\n\n"
871
872 # report to FastAPI middleware aggregate usage across all choices
873 num_completion_tokens = sum(previous_num_tokens)
874 request_metadata.final_usage_info = UsageInfo(
875 prompt_tokens=num_prompt_tokens,
876 completion_tokens=num_completion_tokens,
877 total_tokens=num_prompt_tokens + num_completion_tokens,
878 completion_tokens_details=_make_completion_tokens_details(
879 sum(previous_reasoning_tokens)
880 )
881 if self._include_reasoning_tokens_details
882 else None,
883 )
884
885 # Log complete streaming response if output logging is enabled
886 if self.enable_log_outputs and self.request_logger:
887 # Log the complete response for each choice
888 for i in range(num_choices):
889 full_text = (
890 previous_texts[i]
891 if previous_texts and i < len(previous_texts)
892 else f"<streaming_complete: {previous_num_tokens[i]} tokens>"
893 )
894 self.request_logger.log_outputs(
895 request_id=request_id,
896 outputs=full_text,
897 output_token_ids=None, # Consider also logging all token IDs
898 finish_reason="streaming_complete",
899 is_streaming=True,
900 delta=False,
901 )
902
903 except GenerationError as e:
904 yield f"data: {self._convert_generation_error_to_streaming_response(e)}\n\n"
905 except Exception as e:
906 logger.exception("Error in chat completion stream generator.")
907 data = self.create_streaming_error_response(e)
908 yield f"data: {data}\n\n"
909 # Send the final done message after all response.n are finished
910 yield "data: [DONE]\n\n"

912 async def chat_completion_full_generator(
913 self,
914 request: ChatCompletionRequest,
915 result_generator: AsyncIterator[RequestOutput],
916 request_id: str,
917 model_name: str,
918 conversation: list[ConversationMessage],
919 tokenizer: TokenizerLike,
920 request_metadata: RequestResponseMetadata,
921 parser: Parser | None = None,
922 mm_token_counts: dict[str, int] | None = None,
923 ) -> ErrorResponse | ChatCompletionResponse:
924 created_time = int(time.time())
925 final_res: RequestOutput | None = None
926
927 try:
928 async for res in result_generator:
929 final_res = res
930 except asyncio.CancelledError:
931 return self.create_error_response("Client disconnected")
932
933 if final_res is None:
934 return self.create_error_response(
935 "No output received from the engine.",
936 err_type="InternalServerError",
937 status_code=HTTPStatus.INTERNAL_SERVER_ERROR,
938 )
939
940 choices: list[ChatCompletionResponseChoice] = []
941 total_reasoning_tokens = 0
942
943 role = self.get_chat_request_role(request)
944 tool_parser_cls = (
945 self.parser_cls.tool_parser_cls if self.parser_cls is not None else None
946 )
947 for output in final_res.outputs:
948 # check for error finish reason and raise GenerationError
949 # finish_reason='error' indicates a retryable request-level internal error
950 self._raise_if_error(output.finish_reason, request_id)
951 token_ids = output.token_ids
952 out_logprobs = output.logprobs
953
954 if request.logprobs and (
955 request.top_logprobs is not None or request.logprob_token_ids
956 ):
957 assert out_logprobs is not None, "Did not output logprobs"
958 logprobs = self._create_chat_logprobs(
959 token_ids=token_ids,
960 top_logprobs=out_logprobs,
961 num_output_top_logprobs=request.top_logprobs,
962 logprob_token_ids=request.logprob_token_ids,
963 tokenizer=tokenizer,
964 return_as_token_id=request.return_tokens_as_token_ids,
965 )
966 else:
967 logprobs = None
968
969 if parser is not None:
970 reasoning, content, tool_calls = parser.parse(
971 output.text,
972 request,
973 enable_auto_tools=self.enable_auto_tools,
974 model_output_token_ids=token_ids,
975 )
976 suppress_metadata = not request.include_reasoning and parser is not None
977 if not request.include_reasoning:
978 reasoning = None
979 total_reasoning_tokens += parser.count_reasoning_tokens(token_ids)
980 if suppress_metadata:
981 logprobs = None
982 else:
983 reasoning = None
984 content = output.text
985 tool_calls = []
986 suppress_metadata = False
987
988 auto_tools_called = False
989 is_named_tool_choice = (
990 request.tool_choice is not None
991 and type(request.tool_choice) is ChatCompletionNamedToolChoiceParam
992 )
993 is_required_tool_choice = request.tool_choice == "required"
994
995 # All six construction sites route through ``self._create_chat_message``
996 # so subclasses can swap in a specialized :class:`ChatMessage`
997 # (e.g. the Cohere v2 handler's ``CohereChatMessage``) without
998 # having to duplicate this branch logic.
999 if (not self.enable_auto_tools or not tool_parser_cls) and (
1000 not is_named_tool_choice and not is_required_tool_choice
1001 ):
1002 message = self._create_chat_message(
1003 role=role, reasoning=reasoning, content=content
1004 )
1005
1006 elif is_named_tool_choice or is_required_tool_choice:
1007 message = self._create_chat_message(
1008 role=role,
1009 reasoning=reasoning,
1010 content=content or "",
1011 tool_calls=[
1012 ToolCall(id=tc.id or make_tool_call_id(), function=tc)
1013 for tc in (tool_calls or [])
1014 ],
1015 )
1016
1017 # if the request doesn't use tool choice
1018 # OR specifies to not use a tool
1019 elif not request.tool_choice or request.tool_choice == "none":
1020 message = self._create_chat_message(
1021 role=role, reasoning=reasoning, content=content
1022 )
1023
1024 # handle when there are tools and tool choice is auto
1025 elif (
1026 request.tools
1027 and (request.tool_choice == "auto" or request.tool_choice is None)
1028 and self.enable_auto_tools
1029 and tool_parser_cls
1030 ):
1031 auto_tools_called = tool_calls is not None and len(tool_calls) > 0
1032 if tool_calls:
1033 message = self._create_chat_message(
1034 role=role,
1035 reasoning=reasoning,
1036 content=content,
1037 tool_calls=[
1038 ToolCall(id=tc.id or make_tool_call_id(), function=tc)
1039 for tc in tool_calls
1040 ],
1041 )
1042
1043 else:
1044 message = self._create_chat_message(
1045 role=role,
1046 reasoning=reasoning,
1047 content=content,
1048 )
1049
1050 # undetermined case that is still important to handle
1051 else:
1052 logger.error(
1053 "Error in chat_completion_full_generator - cannot determine"
1054 " if tools should be extracted. Returning a standard chat "
1055 "completion."
1056 )
1057 message = self._create_chat_message(
1058 role=role, reasoning=reasoning, content=content
1059 )
1060
1061 # Subclass hook: enrich the constructed message with any
1062 # parser-side extras that don't fit through the plain
1063 # ``(reasoning, content, tool_calls)`` tuple. Base is a no-op;
1064 # citation-aware handlers use this to surface grounding
1065 # metadata cached on the reasoning parser.
1066 message = self._finalize_response_message(message, parser=parser)
1067
1068 # In OpenAI's API, when a tool is called, the finish_reason is:
1069 # "tool_calls" for "auto" or "required" tool calls,
1070 # and "stop" for named tool calls.
1071 is_finish_reason_tool_calls = auto_tools_called or (
1072 request.tool_choice
1073 and request.tool_choice == "required"
1074 and output.finish_reason == "stop"
1075 )
1076
1077 routed_experts_b64 = (
1078 numpy2base64(output.routed_experts)
1079 if output.routed_experts is not None
1080 else None
1081 )
1082
1083 choice_data = ChatCompletionResponseChoice(
1084 index=output.index,
1085 message=message,
1086 logprobs=logprobs,
1087 finish_reason="tool_calls"
1088 if is_finish_reason_tool_calls
1089 else output.finish_reason
1090 if output.finish_reason
1091 else "stop",
1092 stop_reason=output.stop_reason,
1093 token_ids=(
1094 as_list(output.token_ids)
1095 if request.return_token_ids and not suppress_metadata
1096 else None
1097 ),
1098 routed_experts=routed_experts_b64,
1099 )
1100 choice_data = maybe_filter_parallel_tool_calls(choice_data, request)
1101
1102 choices.append(choice_data)
1103
1104 if request.echo:
1105 last_msg_content: str | list[dict[str, str]] = ""
1106 if (
1107 conversation
1108 and "content" in conversation[-1]
1109 and conversation[-1].get("role") == role
1110 ):
1111 last_msg_content = conversation[-1]["content"] or ""
1112 if isinstance(last_msg_content, list):
1113 last_msg_content = "\n".join(msg["text"] for msg in last_msg_content)
1114
1115 for choice in choices:
1116 full_message = last_msg_content + (choice.message.content or "")
1117 choice.message.content = full_message
1118
1119 assert final_res.prompt_token_ids is not None
1120 num_prompt_tokens = len(final_res.prompt_token_ids)
1121 if final_res.encoder_prompt_token_ids is not None:
1122 num_prompt_tokens += len(final_res.encoder_prompt_token_ids)
1123 num_generated_tokens = sum(
1124 len(output.token_ids) for output in final_res.outputs
1125 )
1126 usage = UsageInfo(
1127 prompt_tokens=num_prompt_tokens,
1128 completion_tokens=num_generated_tokens,
1129 total_tokens=num_prompt_tokens + num_generated_tokens,
1130 completion_tokens_details=_make_completion_tokens_details(
1131 total_reasoning_tokens
1132 )
1133 if self._include_reasoning_tokens_details
1134 else None,
1135 )
1136 usage.prompt_tokens_details = _make_prompt_tokens_details(
1137 self.enable_prompt_tokens_details,
1138 final_res.num_cached_tokens,
1139 final_res.num_cache_creation_tokens,
1140 mm_token_counts,
1141 )
1142
1143 request_metadata.final_usage_info = usage
1144
1145 per_request_metrics: PerRequestMetrics | None = None
1146 # Per-request metrics (timing + spec-decode acceptance) describe a single
1147 # generation stream. For n>1 the stats belong to only one of the n
1148 # sequences, so they cannot be attributed to the request; suppress.
1149 if (request.n or 1) == 1:
1150 if self.enable_per_request_metrics:
1151 per_request_metrics = build_per_request_timing_metrics(
1152 final_res.metrics, num_generated_tokens
1153 )
1154 spec_stats = build_spec_decoding_metrics(final_res)
1155 if spec_stats is not None:
1156 if per_request_metrics is None:
1157 per_request_metrics = PerRequestMetrics()
1158 per_request_metrics.speculative_decoding = spec_stats
1159
1160 # ``final_res.prompt`` is the rendered chat-templated prompt text
1161 prompt_text = final_res.prompt if request.return_prompt_text else None
1162
1163 response = ChatCompletionResponse(
1164 id=request_id,
1165 created=created_time,
1166 model=model_name,
1167 choices=choices,
1168 usage=usage,
1169 system_fingerprint=self.system_fingerprint,
1170 prompt_logprobs=clamp_prompt_logprobs(final_res.prompt_logprobs),
1171 prompt_token_ids=(
1172 final_res.prompt_token_ids if request.return_token_ids else None
1173 ),
1174 prompt_text=prompt_text,
1175 kv_transfer_params=final_res.kv_transfer_params,
1176 ec_transfer_params=final_res.ec_transfer_params,
1177 metrics=per_request_metrics,
1178 )
1179
1180 # Log complete response if output logging is enabled
1181 if self.enable_log_outputs and self.request_logger:
1182 for choice in choices:
1183 output_text = ""
1184 if choice.message.content:
1185 output_text = choice.message.content
1186 elif choice.message.tool_calls:
1187 # For tool calls, log the function name and arguments
1188 tool_call_descriptions = []
1189 for tc in choice.message.tool_calls: # type: ignore
1190 function_call: FunctionCall = tc.function # type: ignore
1191 tool_call_descriptions.append(
1192 f"{function_call.name}({function_call.arguments})"
1193 )
1194 tool_calls_str = ", ".join(tool_call_descriptions)
1195 output_text = f"[tool_calls: {tool_calls_str}]"
1196
1197 if output_text:
1198 # Get the corresponding output token IDs
1199 output_token_ids = None
1200 if choice.index < len(final_res.outputs):
1201 output_token_ids = final_res.outputs[choice.index].token_ids
1202
1203 self.request_logger.log_outputs(
1204 request_id=request_id,
1205 outputs=output_text,
1206 output_token_ids=output_token_ids,
1207 finish_reason=choice.finish_reason,
1208 is_streaming=False,
1209 delta=False,
1210 )
1211
1212 return response

1214 def _get_top_logprobs(
1215 self,
1216 logprobs: dict[int, Logprob],
1217 top_logprobs: int | None,
1218 tokenizer: TokenizerLike | None,
1219 should_return_as_token_id: bool,
1220 return_all: bool = False,
1221 ) -> list[ChatCompletionLogProb]:
1222 return [
1223 ChatCompletionLogProb(
1224 token=(
1225 token := self._get_decoded_token(
1226 p[1],
1227 p[0],
1228 tokenizer,
1229 return_as_token_id=should_return_as_token_id,
1230 )
1231 ),
1232 logprob=max(p[1].logprob, -9999.0),
1233 bytes=list(token.encode("utf-8", errors="replace")),
1234 )
1235 for i, p in enumerate(logprobs.items())
1236 if return_all
1237 or top_logprobs == -1
1238 or (top_logprobs is not None and i < top_logprobs)
1239 ]

1241 def _create_chat_logprobs(
1242 self,
1243 token_ids: GenericSequence[int],
1244 top_logprobs: GenericSequence[dict[int, Logprob] | None],
1245 tokenizer: TokenizerLike | None,
1246 num_output_top_logprobs: int | None = None,
1247 logprob_token_ids: list[int] | None = None,
1248 return_as_token_id: bool | None = None,
1249 ) -> ChatCompletionLogProbs:
1250 """Create OpenAI-style logprobs."""
1251 logprobs_content: list[ChatCompletionLogProbsContent] = []
1252
1253 should_return_as_token_id = (
1254 return_as_token_id
1255 if return_as_token_id is not None
1256 else self.return_tokens_as_token_ids
1257 )
1258 for i, token_id in enumerate(token_ids):
1259 step_top_logprobs = top_logprobs[i]
1260 if step_top_logprobs is None or step_top_logprobs.get(token_id) is None:
1261 if should_return_as_token_id:
1262 token = format_token_id_placeholder(token_id)
1263 else:
1264 if tokenizer is None:
1265 raise ValueError(
1266 "Unable to get tokenizer because `skip_tokenizer_init=True`"
1267 )
1268
1269 token = tokenizer.decode(token_id)
1270
1271 logprobs_content.append(
1272 ChatCompletionLogProbsContent(
1273 token=token,
1274 bytes=list(token.encode("utf-8", errors="replace")),
1275 )
1276 )
1277 else:
1278 step_token = step_top_logprobs[token_id]
1279 step_decoded = step_token.decoded_token
1280
1281 logprobs_content.append(
1282 ChatCompletionLogProbsContent(
1283 token=self._get_decoded_token(
1284 step_token,
1285 token_id,
1286 tokenizer,
1287 should_return_as_token_id,
1288 ),
1289 logprob=max(step_token.logprob, -9999.0),
1290 bytes=(
1291 None
1292 if step_decoded is None
1293 else list(step_decoded.encode("utf-8", errors="replace"))
1294 ),
1295 top_logprobs=self._get_top_logprobs(
1296 step_top_logprobs,
1297 num_output_top_logprobs,
1298 tokenizer,
1299 should_return_as_token_id,
1300 return_all=bool(logprob_token_ids),
1301 ),
1302 )
1303 )
1304
1305 return ChatCompletionLogProbs(content=logprobs_content)
