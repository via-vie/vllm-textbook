451def _detect_field(
452 tensors: list[torch.Tensor],
453 mm_processor: BaseMultiModalProcessor,
454):
455 first_item = tensors[0]
456 hidden_size = mm_processor.info.ctx.model_config.get_inputs_embeds_size()
457
458 if (
459 len(tensors) == 1
460 and first_item.ndim == 3
461 and first_item.shape[0] == 1
462 and first_item.shape[-1] == hidden_size
463 ):
464 logger.warning(
465 "Batched multi-modal embedding inputs are deprecated for Chat API. "
466 "Please pass a separate content part for each multi-modal item."
467 )
468 return _BatchedSingleItemField(batch_size=1)
469
470 first_shape = first_item.shape
471 if all(t.shape == first_shape for t in tensors):
472 return MultiModalBatchedField()
473
474 size_per_item = [len(tensor) for tensor in tensors]
475 slice_idxs = [0, *accumulate(size_per_item)]
476 slices = [
477 (slice(slice_idxs[i], slice_idxs[i + 1]),) for i in range(len(size_per_item))
478 ]
479 return MultiModalFlatField(slices=slices)

482def _merge_embeds(
483 data_items: list[dict[str, "torch.Tensor"]],
484 mm_processor: BaseMultiModalProcessor,
485):
486 if not data_items:
487 return {}
488
489 first_keys = set(data_items[0].keys())
490 if any(set(item.keys()) != first_keys for item in data_items[1:]):
491 raise VLLMValidationError(
492 "All dictionaries in the list of embeddings must have the same keys."
493 )
494
495 fields = {
496 key: _detect_field([item[key] for item in data_items], mm_processor)
497 for key in first_keys
498 }
499 data_merged = {
500 key: field._reduce_data([item[key] for item in data_items], pin_memory=False)
501 for key, field in fields.items()
502 }
503
504 try:
505 # TODO: Support per-request mm_processor_kwargs
506 parsed_configs = mm_processor._get_mm_fields_config(
507 transformers.BatchFeature(data_merged),
508 {},
509 )
510 parsed_fields = {key: parsed_configs[key].field for key in first_keys}
511 keys_to_update = [
512 key
513 for key in first_keys
514 if (
515 fields[key] != parsed_fields[key]
516 and not isinstance(fields[key], _BatchedSingleItemField)
517 )
518 ]
519
520 for key in keys_to_update:
521 data_merged[key] = parsed_fields[key]._reduce_data(
522 [item[key] for item in data_items], pin_memory=False
523 )
524 except Exception:
525 logger.exception(
526 "Error when parsing merged embeddings. "
527 "Falling back to auto-detected fields."
528 )
529
530 return data_merged

542def _parse_metadata_array(key: str, value: list, metadata_fields: set[str]):
543 if key not in metadata_fields:
544 raise VLLMValidationError(f"JSON arrays are only supported for metadata: {key}")
545 if not is_list_of_numbers(value):
546 raise VLLMValidationError(f"Metadata {key} must be a finite numeric array.")
547 dtype = torch.float64 if any(isinstance(v, float) for v in value) else torch.int64

551 raise VLLMValidationError(f"Invalid metadata array: {key}") from error

555 modality: str,
556 data_items: list[Any],
557 mm_processor: BaseMultiModalProcessor,
558):
559 if len(data_items) == 0:
560 return data_items

653 def add(self, modality: ModalityStr, item: _T) -> str | None:
654 """
655 Add a multi-modal item to the current prompt and returns the
656 placeholder string to use, if any.
657
658 An optional uuid can be added which serves as a unique identifier of the
659 media.
660
661 Note:
662 `prompt_embeds` bypass MM-processor validation because they are
663 pre-computed embeddings that do not go through any HF processor, encoder,
664 or model-specific placeholder logic. The corresponding placeholder string is
665 managed by the parser via `_add_placeholder`, so we return None here.
666 """
667 add_info = self._validate_add(modality)
668 if add_info is None:
669 self._items_by_modality["prompt_embeds"].append(item)
670 return None
671
672 input_modality, original_modality, use_vision_chunk, num_items = add_info
673
674 # Track original modality for vision_chunk items
675 if use_vision_chunk:
676 self._items_by_modality[input_modality].append(item) # type: ignore
677 self._modality_order["vision_chunk"].append(original_modality)
678 else:
679 self._items_by_modality[original_modality].append(item)
680
681 return self.model_cls.get_placeholder_str(modality, num_items)

683 def _validate_add(self, modality: ModalityStr) -> tuple[str, str, bool, int] | None:
684 """Validate that one more item of the modality can be tracked."""
685 if modality == "prompt_embeds":
686 return None
687
688 input_modality = modality.replace("_embeds", "")
689 original_modality = modality
690 use_vision_chunk = (
691 self.use_unified_vision_chunk_modality
692 and original_modality in ["video", "image"]
693 )
694
695 # If use_unified_vision_chunk_modality is enabled,
696 # map image/video to vision_chunk
697 if use_vision_chunk:
698 # To avoid validation fail
699 # because models with use_unified_vision_chunk_modality=True
700 # will only accept vision_chunk modality.
701 input_modality = "vision_chunk"
702 num_items = len(self._items_by_modality[input_modality]) + 1
703 else:
704 num_items = len(self._items_by_modality[original_modality]) + 1
705
706 mm_config = self.model_config.multimodal_config
707 if (
708 mm_config is not None
709 and mm_config.enable_mm_embeds
710 and mm_config.get_limit_per_prompt(input_modality) == 0
711 and original_modality.endswith("_embeds")
712 ):
713 # Skip validation: embeddings bypass limit when enable_mm_embeds=True
714 pass
715 else:
716 self.mm_processor.info.validate_num_items(input_modality, num_items)
717
718 return input_modality, original_modality, use_vision_chunk, num_items

727def _resolve_vision_chunk_items(
728 vision_chunk_items: list[tuple[object, str | None]],
729 mm_processor: BaseMultiModalProcessor,
730 vision_chunks_modality_order: list[str],
731):
732 # Process vision_chunk items - extract from (data, modality) tuples
733 # and convert to VisionChunk types with proper UUID handling
734 vision_chunks_uuids = [uuid for data, uuid in vision_chunk_items]
735
736 assert len(vision_chunk_items) == len(vision_chunks_modality_order), (
737 f"vision_chunk items ({len(vision_chunk_items)}) and "
738 f"modality_order ({len(vision_chunks_modality_order)}) must have same length"
739 )
740
741 processed_chunks: list[VisionChunk] = []
742 video_idx = 0
743 for inner_modality, (data, uuid) in zip(
744 vision_chunks_modality_order, vision_chunk_items
745 ):
746 if inner_modality == "image":
747 # Cast data to proper type for image
748 # Use .media (PIL.Image) directly to avoid redundant
749 # bytes→PIL conversion in media_processor
750 if hasattr(data, "media"):
751 image_data = data.media # type: ignore[union-attr]
752 processed_chunks.append(
753 VisionChunkImage(type="image", image=image_data, uuid=uuid)
754 )
755 else:
756 processed_chunks.append(data) # type: ignore[arg-type]
757 elif inner_modality == "video":
758 # For video, we may need to split into chunks
759 # if processor supports it
760 # For now, just wrap as a video chunk placeholder
761 if hasattr(mm_processor, "split_video_chunks") and data is not None:
762 try:
763 video_uuid = uuid or random_uuid()
764 # video await result is (video_data, video_meta) tuple
765 if isinstance(data, tuple) and len(data) >= 1:
766 video_data = data[0]
767 else:
768 video_data = data
769 video_chunks = mm_processor.split_video_chunks(video_data)

789def _resolve_items(
790 items_by_modality: dict[str, list[tuple[object, str | None]]],
791 mm_processor: BaseMultiModalProcessor | None,
792 modality_order: dict[str, list[str]],
793) -> tuple[MultiModalDataDict, MultiModalUUIDDict]:
794 """
795 Materialize the tracker's per-modality items into `mm_data` / `mm_uuids`.
796
797 Note:
798 `mm_processor` is `None` for text-only models (no registered HF
799 processor) whose only modality is `prompt_embeds`. Every other
800 modality requires a processor, enforced by the guard below.
801 """
802 if "image" in items_by_modality and "image_embeds" in items_by_modality:
803 raise VLLMValidationError(
804 "Mixing raw image and embedding inputs is not allowed",
805 parameter="image_embeds",
806 )
807 if "audio" in items_by_modality and "audio_embeds" in items_by_modality:
808 raise VLLMValidationError(
809 "Mixing raw audio and embedding inputs is not allowed",
810 parameter="audio_embeds",
811 )
812 if "video" in items_by_modality and "video_embeds" in items_by_modality:
813 raise VLLMValidationError(
814 "Mixing raw video and embedding inputs is not allowed",
815 parameter="video_embeds",
816 )
817 # `prompt_embeds` bypasses HF MM processors. Every other modality requires one.
818 processor_modalities = items_by_modality.keys() - {"prompt_embeds"}
819 if processor_modalities and mm_processor is None:
820 raise RuntimeError(
821 _REQUIRE_MM_PROCESSOR_ERROR.format(modality=processor_modalities)
822 )
823
824 mm_data = {}
825 mm_uuids = {}
826 if "image_embeds" in items_by_modality:
827 assert mm_processor is not None
828 mm_data["image"] = _get_embeds_data(
829 "image",
830 [data for data, uuid in items_by_modality["image_embeds"]],
831 mm_processor,
832 )
833 mm_uuids["image"] = [uuid for data, uuid in items_by_modality["image_embeds"]]
834 if "image" in items_by_modality:
835 mm_data["image"] = [data for data, uuid in items_by_modality["image"]]
836 mm_uuids["image"] = [uuid for data, uuid in items_by_modality["image"]]
837 if "audio_embeds" in items_by_modality:
838 assert mm_processor is not None
839 mm_data["audio"] = _get_embeds_data(
840 "audio",
841 [data for data, uuid in items_by_modality["audio_embeds"]],
842 mm_processor,
843 )
844 mm_uuids["audio"] = [uuid for data, uuid in items_by_modality["audio_embeds"]]
845 if "audio" in items_by_modality:
846 mm_data["audio"] = [data for data, uuid in items_by_modality["audio"]]
847 mm_uuids["audio"] = [uuid for data, uuid in items_by_modality["audio"]]
848 if "video" in items_by_modality:
849 mm_data["video"] = [data for data, uuid in items_by_modality["video"]]
850 mm_uuids["video"] = [uuid for data, uuid in items_by_modality["video"]]
851 if "video_embeds" in items_by_modality:
852 assert mm_processor is not None
853 mm_data["video"] = _get_embeds_data(
854 "video",
855 [data for data, uuid in items_by_modality["video_embeds"]],
856 mm_processor,
857 )
858 mm_uuids["video"] = [uuid for data, uuid in items_by_modality["video_embeds"]]
859 if "vision_chunk" in items_by_modality:
860 assert mm_processor is not None
861 # Process vision_chunk items - extract from (data, modality) tuples
862 # and convert to VisionChunk types with proper UUID handling
863 processed_chunks, vision_chunk_uuids = _resolve_vision_chunk_items(
864 items_by_modality["vision_chunk"],
865 mm_processor,
866 modality_order.get("vision_chunk", []),
867 )
868 mm_data["vision_chunk"] = processed_chunks
869 mm_uuids["vision_chunk"] = vision_chunk_uuids
870 if "prompt_embeds" in items_by_modality:
871 mm_data["prompt_embeds"] = [
872 data for data, _uuid in items_by_modality["prompt_embeds"]
873 ]
874
875 return mm_data, mm_uuids

903class AsyncMultiModalItemTracker(BaseMultiModalItemTracker[_AsyncMultiModalItem]):
904 async def resolve_items(
905 self,
906 ) -> tuple[MultiModalDataDict | None, MultiModalUUIDDict | None]:
907 if not self._items_by_modality:
908 return None, None
909
910 # Fetch all modalities together. Each tracked item is already an
911 # independent awaitable, and the async connector offloads blocking
912 # decode work, so waiting for one modality before starting the next
913 # needlessly adds their latency.
914 # Keep the original group and item order when rebuilding the result.
915 item_groups = list(self._items_by_modality.items())
916 items = [item for _, group in item_groups for item in group]
917 results = await asyncio.gather(
918 *(item() for item in items), return_exceptions=True
919 )
920 for result in results:
921 if isinstance(result, BaseException):
922 # Gathering with return_exceptions=True lets every task finish
923 # (or itself fail) before we raise, instead of abandoning
924 # still-in-flight fetches (real network/thread-pool work) the
925 # moment the first one fails.
926 raise result
927
928 resolved_items_by_modality: dict[str, list[Any]] = {}
929 result_idx = 0
930 for modality, group in item_groups:
931 next_result_idx = result_idx + len(group)
932 resolved_items_by_modality[modality] = results[result_idx:next_result_idx]
933 result_idx = next_result_idx
934
935 mm_processor = (
936 self.mm_processor if self._model_config.is_multimodal_model else None
937 )
938 return _resolve_items(
939 resolved_items_by_modality,
940 mm_processor,
941 self._modality_order,
942 )
