58 def _resolve_mm_lora(
59 self,
60 prompt: EngineInput,
61 lora_request: LoRARequest | None,
62 ) -> LoRARequest | None:
63 if prompt["type"] != "multimodal":
64 return lora_request
65
66 lora_config = self.llm_engine.vllm_config.lora_config
67 default_mm_loras = None if lora_config is None else lora_config.default_mm_loras
68 if not default_mm_loras:
69 return lora_request
70
71 prompt_modalities = prompt["mm_placeholders"].keys()
72 intersection = set(prompt_modalities).intersection(default_mm_loras.keys())
73 if not intersection:
74 return lora_request
75
76 if len(intersection) > 1:
77 # TODO: Would be nice to be able to have multiple loras per prompt
78 logger.warning(
79 "Multiple modality specific loras were registered and would be "
80 "used by a single prompt consuming several modalities; "
81 "currently we only support one lora per request; as such, "
82 "lora(s) registered with modalities: %s will be skipped",
83 intersection,
84 )
85 return lora_request
86
87 # Build the LoRA request; the ID of the default mm lora is the
88 # index of the modality name sorted alphabetically + 1.
89 modality_name = intersection.pop()
90 modality_lora_path = default_mm_loras[modality_name]
91 modality_lora_id = sorted(default_mm_loras).index(modality_name) + 1
92
93 # If we have a collision, warn if there is a collision,
94 # but always send the explicitly provided request.
95 if lora_request:
96 if lora_request.lora_int_id != modality_lora_id:
97 logger.warning(
98 "A modality with a registered lora and a lora_request "
99 "with a different ID were provided; falling back to the "
100 "lora_request as we only apply one LoRARequest per prompt"
101 )
102 return lora_request
103
104 return LoRARequest(
105 modality_name,
106 modality_lora_id,
107 modality_lora_path,
108 )

110 def _preprocess_cmpl(
111 self,
112 prompts: Sequence[PromptType],
113 tokenization_kwargs: dict[str, Any] | None = None,
114 mm_processor_kwargs: dict[str, Any] | None = None,
115 ) -> Sequence[EngineInput]:
116 """
117 Convert prompt inputs from LLM APIs (other than [LLM.chat][]) into
118 a format that can be passed to `_add_request`.
119
120 Refer to [LLM.generate][] for a complete description of the arguments.
121
122 Returns:
123 A list of `EngineInput` objects ready to be passed into LLMEngine.
124 """
125 renderer = self.renderer
126 model_config = self.model_config
127
128 parsed_prompts = [
129 parse_model_prompt(model_config, prompt) for prompt in prompts
130 ]
131 tok_params = renderer.default_cmpl_tok_params.with_kwargs(
132 **(tokenization_kwargs or {})
133 )
134 prompt_extras = (
135 None
136 if mm_processor_kwargs is None
137 else {"mm_processor_kwargs": mm_processor_kwargs}
138 )
139
140 return renderer.render_cmpl(
141 parsed_prompts,
142 tok_params,
143 prompt_extras=prompt_extras,
144 )

159 def _preprocess_chat(
160 self,
161 conversations: Sequence[list[ChatCompletionMessageParam]],
162 chat_template: str | None = None,
163 chat_template_content_format: ChatTemplateContentFormatOption = "auto",
164 chat_template_kwargs: dict[str, Any] | None = None,
165 add_generation_prompt: bool = True,
166 continue_final_message: bool = False,
167 tools: list[dict[str, Any]] | None = None,
168 tokenization_kwargs: dict[str, Any] | None = None,
169 mm_processor_kwargs: dict[str, Any] | None = None,
170 ) -> Sequence[EngineInput]:
171 """
172 Convert a list of conversations into prompts so that they can then
173 be used as input for other LLM APIs.
174
175 Refer to [LLM.chat][] for a complete description of the arguments.
176
177 Returns:
178 A list of `EngineInput` objects ready to be passed into LLMEngine.
179 """
180 renderer = self.renderer
181
182 chat_params = ChatParams(
183 chat_template=chat_template,
184 chat_template_content_format=chat_template_content_format,
185 chat_template_kwargs=merge_kwargs(
186 chat_template_kwargs,
187 dict(
188 add_generation_prompt=add_generation_prompt,
189 continue_final_message=continue_final_message,
190 tools=tools,
191 tokenize=(
192 is_mistral_tokenizer(renderer.tokenizer)
193 or self.model_config.enable_prompt_embeds
194 ),
195 ),
196 ),
197 mm_processor_kwargs=mm_processor_kwargs,
198 )
199 # The chat template is responsible for emitting BOS/EOS, so do not let
200 # the tokenizer add them again unless the caller asks for it. This
201 # matches the online chat API (`ChatCompletionRequest.add_special_tokens`
202 # defaults to `False`) and avoids a double BOS for multimodal models,
203 # whose processor default is `add_special_tokens=True` (#55197).
204 tokenization_kwargs = {
205 "add_special_tokens": False,
206 **(tokenization_kwargs or {}),
207 }
208 tok_params = renderer.default_chat_tok_params.with_kwargs(**tokenization_kwargs)
209 prompt_extras = (
210 None
211 if mm_processor_kwargs is None
212 else {"mm_processor_kwargs": mm_processor_kwargs}
213 )
214
215 _, engine_inputs = renderer.render_chat(
216 conversations,
217 chat_params,
218 tok_params,
219 prompt_extras=prompt_extras,
220 )
221
222 return engine_inputs

251 self,
252 params: _P | Sequence[_P],
253 num_requests: int,

455 def _adjust_params_for_parsing(
456 self, params: Sequence[SamplingParams | PoolingParams]
457 ) -> None:
458 """Set ``skip_special_tokens=False`` when the model encodes
459 structured output syntax as special tokens.
460
461 Models like Gemma4 register thinking delimiters
462 (``<|channel>``/``<channel|>``) and tool call tokens
463 (``<|tool_call>``/``<tool_call|>``/``<|"|>``) as special tokens.
464 The default ``skip_special_tokens=True`` strips them from
465 ``output.text``, breaking parsing of both reasoning blocks and
466 tool calls.
467
468 This is a no-op for models whose structured tokens are regular
469 text tokens (e.g. DeepSeek's ``<think>``/``</think>``).
470 """
471 # The offline API currently lacks a unified rendering pipeline.
472 # Until the planned Renderer refactor is complete, we hardcode
473 # this token preservation logic specifically for Gemma4 models
474 # to avoid regressions on other models.
475 hf_config = getattr(self.model_config, "hf_config", None)
476 architectures = getattr(hf_config, "architectures", [])
477
478 if any("Gemma4" in arch for arch in architectures):
479 tokenizer = self.renderer.get_tokenizer()
480 vocab = tokenizer.get_vocab()
481 special_ids = set(getattr(tokenizer, "all_special_ids", []))
482
483 # Tokens used for thinking delimiters and tool call syntax
484 # that some models (Gemma4) register as special tokens.
485 structured_tokens = (
486 "<|channel>",
487 "<channel|>", # thinking delimiters
488 "<|tool_call>",
489 "<tool_call|>", # tool call delimiters
490 '<|"|>', # string quoting in tool args
491 )
492 needs_special = any(
493 vocab.get(tok) in special_ids
494 for tok in structured_tokens
495 if tok in vocab
496 )
497 if needs_special:
498 for sp in params:
499 if isinstance(sp, SamplingParams) and sp.skip_special_tokens:
500 sp.skip_special_tokens = False

502 def _render_and_run_requests(
503 self,
504 prompts: Iterable[EngineInput],
505 params: Sequence[SamplingParams | PoolingParams],
506 output_type: type[_O],
507 *,
508 lora_requests: Sequence[LoRARequest | None] | None = None,
509 priorities: Sequence[int] | None = None,
510 use_tqdm: bool | Callable[..., tqdm] = True,
511 ):
512 if isinstance(prompts, (list, tuple)):
513 logger.warning_once(
514 "Rendering all prompts before adding them to the engine "
515 "is less efficient than performing both on the same prompt "
516 "before processing the next prompt. You should instead pass "
517 "a generator that renders one prompt per iteration, as that allows "
518 "engine execution to begin for the first prompt while processing "
519 "the next prompt."
520 )
521
522 self._render_and_add_requests(
523 prompts=prompts,
524 params=params,
525 lora_requests=lora_requests,
526 priorities=priorities,
527 )
528
529 return self._run_engine(output_type, use_tqdm=use_tqdm)

531 def _render_and_add_requests(
532 self,
533 prompts: Iterable[EngineInput],
534 params: Sequence[SamplingParams | PoolingParams],
535 *,
536 lora_requests: Sequence[LoRARequest | None] | None = None,
537 priorities: Sequence[int] | None = None,
538 ) -> list[str]:
539 added_request_ids: list[str] = []
540
541 try:
542 for i, prompt in enumerate(prompts):
543 request_id = self._add_request(
544 prompt,
545 params[i],
546 lora_request=self._resolve_mm_lora(
547 prompt,
548 None if lora_requests is None else lora_requests[i],
549 ),
550 priority=0 if priorities is None else priorities[i],
551 )
552 added_request_ids.append(request_id)
553 except Exception as e:
554 if added_request_ids:
555 self.llm_engine.abort_request(added_request_ids, internal=True)
556 raise e
557
558 return added_request_ids

560 def _add_request(
561 self,
562 prompt: EngineInput,
563 params: SamplingParams | PoolingParams,
564 lora_request: LoRARequest | None = None,
565 priority: int = 0,
566 ) -> str:
567 if isinstance(params, SamplingParams):
568 # We only care about the final output
569 params.output_kind = RequestOutputKind.FINAL_ONLY
570
571 request_id = str(next(self.request_counter))
572
573 return self.llm_engine.add_request(
574 request_id,
575 prompt,
576 params,
577 lora_request=lora_request,
578 priority=priority,
579 )

581 def _run_engine(
582 self,
583 output_type: type[_O] | tuple[type[_O], ...],
584 *,
585 use_tqdm: bool | Callable[..., tqdm] = True,
586 ) -> list[_O]:
587 # Initialize tqdm.
588 if use_tqdm:
589 num_requests = self.llm_engine.get_num_unfinished_requests()
590 tqdm_func = use_tqdm if callable(use_tqdm) else tqdm
591 pbar = tqdm_func(
592 total=num_requests,
593 desc="Processed prompts",
594 dynamic_ncols=True,
595 postfix=(f"est. speed input: {0:.2f} toks/s, output: {0:.2f} toks/s"),
596 )
597
598 # Run the engine.
599 outputs: list[_O] = []
600 total_in_toks = 0
601 total_out_toks = 0
602 while self.llm_engine.has_unfinished_requests():
603 step_outputs = self.llm_engine.step()
604 for output in step_outputs:
605 assert isinstance(output, output_type)
606 if output.finished:
607 outputs.append(output) # type: ignore[arg-type]
608 if use_tqdm:
609 if isinstance(output, RequestOutput):
610 # Calculate tokens only for RequestOutput
611 n = len(output.outputs)
612 assert output.prompt_token_ids is not None
613 total_in_toks += len(output.prompt_token_ids) * n
614 in_spd = total_in_toks / pbar.format_dict["elapsed"]
615 total_out_toks += sum(
616 len(stp.token_ids) for stp in output.outputs
617 )
618 out_spd = total_out_toks / pbar.format_dict["elapsed"]
619 pbar.postfix = (
620 f"est. speed input: {in_spd:.2f} toks/s, "
621 f"output: {out_spd:.2f} toks/s"
622 )
623 pbar.update(n)
624 else:
625 pbar.update(1)
626 if pbar.n == num_requests:
627 pbar.refresh()
628
629 if use_tqdm:
630 pbar.close()
631 # Sort the outputs by request ID.
632 # This is necessary because some requests may be finished earlier than
633 # its previous requests.
634 return sorted(outputs, key=lambda x: int(x.request_id))
