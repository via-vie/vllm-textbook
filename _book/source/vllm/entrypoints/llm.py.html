67class LLM(BeamSearchOfflineMixin, PoolingOfflineMixin, OfflineInferenceMixin):
68 """An LLM for generating texts from given prompts and sampling parameters.
69
70 This class includes a tokenizer, a language model (possibly distributed
71 across multiple GPUs), and GPU memory space allocated for intermediate
72 states (aka KV cache). Given a batch of prompts and sampling parameters,
73 this class generates texts from the model, using an intelligent batching
74 mechanism and efficient memory management.
75
76 Args:
77 model: The name or path of a HuggingFace Transformers model.
78 tokenizer: The name or path of a HuggingFace Transformers tokenizer.
79 tokenizer_mode: The tokenizer mode. "auto" will use the fast tokenizer
80 if available, and "slow" will always use the slow tokenizer.
81 skip_tokenizer_init: If true, skip initialization of tokenizer and
82 detokenizer. Expect valid prompt_token_ids and None for prompt
83 from the input.
84 trust_remote_code: Trust remote code (e.g., from HuggingFace) when
85 downloading the model and tokenizer.
86 allowed_local_media_path: Allowing API requests to read local images
87 or videos from directories specified by the server file system.
88 This is a security risk. Should only be enabled in trusted
89 environments.
90 allowed_media_domains: If set, only media URLs that belong to this
91 domain can be used for multi-modal inputs.
92 tensor_parallel_size: The number of GPUs to use for distributed
93 execution with tensor parallelism.
94 dtype: The data type for the model weights and activations. Currently,
95 we support `float32`, `float16`, and `bfloat16`. If `auto`, we use
96 the `dtype` attribute of the Transformers model's config. However,
97 if the `dtype` in the config is `float32`, we will use `float16` instead.
98 quantization: The method used to quantize the model weights. Currently,
99 we support "awq", "gptq", and "fp8" (experimental).
100 If None, we first check the `quantization_config` attribute in the
101 model config file. If that is None, we assume the model weights are
102 not quantized and use `dtype` to determine the data type of
103 the weights.
104 revision: The specific model version to use. It can be a branch name,
105 a tag name, or a commit id.
106 tokenizer_revision: The specific tokenizer version to use. It can be a
107 branch name, a tag name, or a commit id.
108 chat_template: The chat template to apply.
109 seed: The seed to initialize the random number generator for sampling.
110 gpu_memory_utilization: The ratio (between 0 and 1) of GPU memory to
111 reserve for the model weights, activations, and KV cache. Higher
112 values will increase the KV cache size and thus improve the model's
113 throughput. However, if the value is too high, it may cause out-of-
114 memory (OOM) errors.
115 kv_cache_memory_bytes: Size of KV Cache per GPU in bytes. By default,
116 this is set to None and vllm can automatically infer the kv cache
117 size based on gpu_memory_utilization. However, users may want to
118 manually specify the kv cache memory size. kv_cache_memory_bytes
119 allows more fine-grain control of how much memory gets used when
120 compared with using gpu_memory_utilization. Note that
121 kv_cache_memory_bytes (when not-None) ignores
122 gpu_memory_utilization
123 cpu_offload_gb: The size (GiB) of CPU memory to use for offloading
124 the model weights. This virtually increases the GPU memory space
125 you can use to hold the model weights, at the cost of CPU-GPU data
126 transfer for every forward pass.
127 offload_group_size: Prefetch offloading: Group every N layers
128 together. Offload last `offload_num_in_group` layers of each group.
129 Default is 0 (disabled).
130 offload_num_in_group: Prefetch offloading: Number of layers to
131 offload per group. Default is 1.
132 offload_prefetch_step: Prefetch offloading: Number of layers to
133 prefetch ahead. Higher values hide more latency but use more GPU
134 memory. Default is 1.
135 offload_params: Prefetch offloading: Set of parameter name segments
136 to selectively offload. Only parameters whose names contain one of
137 these segments will be offloaded (e.g., {"gate_up_proj", "down_proj"}
138 for MLP weights, or {"w13_weight", "w2_weight"} for MoE expert
139 weights). If None or empty, all parameters are offloaded.
140 enforce_eager: Whether to enforce eager execution. If True, we will
141 disable CUDA graph and always execute the model in eager mode.
142 If False, we will use CUDA graph and eager execution in hybrid.
143 enable_return_routed_experts: Whether to return routed experts.
144 disable_custom_all_reduce: See
145 [ParallelConfig][vllm.config.ParallelConfig].
146 hf_token: The token to use as HTTP bearer authorization for remote files
147 . If `True`, will use the token generated when running
148 `hf auth login` (stored in `~/.cache/huggingface/token`).
149 hf_overrides: If a dictionary, contains arguments to be forwarded to the
150 HuggingFace config. If a callable, it is called to update the
151 HuggingFace config.
152 mm_processor_kwargs: Arguments to be forwarded to the model's processor
153 for multi-modal data, e.g., image processor. Overrides for the
154 multi-modal processor obtained from `AutoProcessor.from_pretrained`.
155 The available overrides depend on the model that is being run.
156 For example, for Phi-3-Vision: `{"num_crops": 4}`.
157 pooler_config: Initialize non-default pooling config for the pooling model,
158 e.g., `PoolerConfig(seq_pooling_type="MEAN", use_activation=False)`.
159 compilation_config: Either an integer or a dictionary. If it is an
160 integer, it is used as the mode of compilation optimization. If it
161 is a dictionary, it can specify the full compilation configuration.
162 attention_config: Configuration for attention mechanisms. Can be a
163 dictionary or an AttentionConfig instance. If a dictionary, it will
164 be converted to an AttentionConfig. Allows specifying the attention
165 backend and other attention-related settings.
166 return_sampling_mask: Return each sampled token's post-processing
167 support set. Requires Model Runner V2 and processed log probabilities.
168 spec_method: Top-level alias for `speculative_config["method"]`.
169 spec_model: Top-level alias for `speculative_config["model"]`.
170 spec_tokens: Top-level alias for
171 `speculative_config["num_speculative_tokens"]`.
172 **kwargs: Arguments for [`EngineArgs`][vllm.EngineArgs].
173
174 Note:
175 This class is intended to be used for offline inference. For online
176 serving, use the [AsyncLLMEngine][vllm.AsyncLLMEngine] class instead.
177 """

418 def generate(
419 self,
420 prompts: PromptType | Sequence[PromptType],
421 sampling_params: SamplingParams | Sequence[SamplingParams] | None = None,
422 *,
423 use_tqdm: bool | Callable[..., tqdm] = True,
424 lora_request: Sequence[LoRARequest] | LoRARequest | None = None,
425 priority: list[int] | None = None,
426 tokenization_kwargs: dict[str, Any] | None = None,
427 mm_processor_kwargs: dict[str, Any] | None = None,
428 ) -> list[RequestOutput]:
429 """Generates the completions for the input prompts.
430
431 This class automatically batches the given prompts, considering
432 the memory constraint. For the best performance, put all of your prompts
433 into a single list and pass it to this method.
434
435 Args:
436 prompts: The prompts to the LLM. You may pass a sequence of prompts
437 for batch inference. See [PromptType][vllm.inputs.PromptType]
438 for more details about the format of each prompt.
439 sampling_params: The sampling parameters for text generation. If
440 None, we use the default sampling parameters.
441 When it is a single value, it is applied to every prompt.
442 When it is a list, the list must have the same length as the
443 prompts and it is paired one by one with the prompt.
444 use_tqdm: If `True`, shows a tqdm progress bar.
445 If a callable (e.g., `functools.partial(tqdm, leave=False)`),
446 it is used to create the progress bar.
447 If `False`, no progress bar is created.
448 lora_request: LoRA request to use for generation, if any.
449 priority: The priority of the requests, if any.
450 Only applicable when priority scheduling policy is enabled.
451 If provided, must be a list of integers matching the length
452 of `prompts`, where each priority value corresponds to the prompt
453 at the same index.
454 tokenization_kwargs: Overrides for `tokenizer.encode`.
455 mm_processor_kwargs: Overrides for `processor.__call__`.
456
457 Returns:
458 A list of `RequestOutput` objects containing the
459 generated completions in the same order as the input prompts.
460 """
461 runner_type = self.model_config.runner_type
462 if runner_type != "generate":
463 raise ValueError(
464 "LLM.generate() is only supported for generative models. "
465 "Try passing `--runner generate` to use the model as a "
466 "generative model."
467 )
468
469 if sampling_params is None:
470 sampling_params = self.get_default_sampling_params()
471
472 return self._run_completion(
473 prompts=prompts,
474 params=sampling_params,
475 output_type=RequestOutput,
476 use_tqdm=use_tqdm,
477 lora_request=lora_request,
478 tokenization_kwargs=tokenization_kwargs,
479 priority=priority,
480 mm_processor_kwargs=mm_processor_kwargs,
481 )

483 def enqueue(
484 self,
485 prompts: PromptType | Sequence[PromptType],
486 sampling_params: SamplingParams | Sequence[SamplingParams] | None = None,
487 lora_request: Sequence[LoRARequest] | LoRARequest | None = None,
488 priority: list[int] | None = None,
489 use_tqdm: bool | Callable[..., tqdm] = True,
490 tokenization_kwargs: dict[str, Any] | None = None,
491 mm_processor_kwargs: dict[str, Any] | None = None,
492 ) -> list[str]:
493 """Enqueue prompts for generation without waiting for completion.
494
495 This method adds requests to the engine queue but does not start
496 processing them. Use wait_for_completion() to process the queued
497 requests and get results.
498
499 Args:
500 prompts: The prompts to the LLM. See generate() for details.
501 sampling_params: The sampling parameters for text generation.
502 lora_request: LoRA request to use for generation, if any.
503 priority: The priority of the requests, if any.
504 use_tqdm: If True, shows a tqdm progress bar while adding requests.
505 tokenization_kwargs: Overrides for `tokenizer.encode`.
506 mm_processor_kwargs: Overrides for `processor.__call__`.
507
508 Returns:
509 A list of request IDs for the enqueued requests.
510 """
511 runner_type = self.model_config.runner_type
512 if runner_type != "generate":
513 raise ValueError("LLM.enqueue() is only supported for generative models.")
514
515 if sampling_params is None:
516 sampling_params = self.get_default_sampling_params()
517
518 return self._add_completion_requests(
519 prompts=prompts,
520 params=sampling_params,
521 use_tqdm=use_tqdm,
522 lora_request=lora_request,
523 priority=priority,
524 tokenization_kwargs=tokenization_kwargs,
525 mm_processor_kwargs=mm_processor_kwargs,
526 )

567 def collective_rpc(
568 self,
569 method: str | Callable[..., _R],
570 timeout: float | None = None,
571 args: tuple = (),
572 kwargs: dict[str, Any] | None = None,
573 ) -> list[_R]:
574 """
575 Execute an RPC call on all workers.
576
577 Args:
578 method: Name of the worker method to execute, or a callable that
579 is serialized and sent to all workers to execute.
580
581 If the method is a callable, it should accept an additional
582 `self` argument, in addition to the arguments passed in `args`
583 and `kwargs`. The `self` argument will be the worker object.
584 timeout: Maximum time in seconds to wait for execution. Raises a
585 [`TimeoutError`][] on timeout. `None` means wait indefinitely.
586 args: Positional arguments to pass to the worker method.
587 kwargs: Keyword arguments to pass to the worker method.
588
589 Returns:
590 A list containing the results from each worker.
591
592 Note:
593 It is recommended to use this API to only pass control messages,
594 and set up data-plane communication to pass data.
595 """
596
597 return self.llm_engine.collective_rpc(method, timeout, args, kwargs)

599 def apply_model(self, func: Callable[[nn.Module], _R]) -> list[_R]:
600 """
601 Run a function directly on the model inside each worker,
602 returning the result for each of them.
603
604 !!! warning
605 To reduce the overhead of data transfer, avoid returning large
606 arrays or tensors from this method. If you must return them,
607 make sure you move them to CPU first to avoid taking up additional
608 VRAM!
609 """
610 return self.llm_engine.apply_model(func)

618 lora_request: Sequence[LoRARequest] | LoRARequest | None = None,
619 chat_template: str | None = None,

626 mm_processor_kwargs: dict[str, Any] | None = None,
627 ) -> list[RequestOutput]:
628 """
629 Generate responses for a chat conversation.
630
631 The chat conversation is converted into a text prompt using the
632 tokenizer and calls the [generate][vllm.LLM.generate] method to generate
633 the responses.
634
635 Multi-modal inputs can be passed in the same way you would pass them
636 to the OpenAI API.
637
638 Args:
639 messages: A sequence of conversations or a single conversation.
640
641 - Each conversation is represented as a list of messages.
642 - Each message is a dictionary with 'role' and 'content' keys.
643
644 sampling_params: The sampling parameters for text generation.
645 If None, we use the default sampling parameters. When it
646 is a single value, it is applied to every prompt. When it
647 is a list, the list must have the same length as the
648 prompts and it is paired one by one with the prompt.
649 use_tqdm: If `True`, shows a tqdm progress bar.
650 If a callable (e.g., `functools.partial(tqdm, leave=False)`),
651 it is used to create the progress bar.
652 If `False`, no progress bar is created.
653 lora_request: LoRA request to use for generation, if any.
654 chat_template: The template to use for structuring the chat.
655 If not provided, the model's default chat template will be used.
656 chat_template_content_format: The format to render message content.
657
658 - "string" will render the content as a string.
659 Example: `"Who are you?"`
660 - "openai" will render the content as a list of dictionaries,
661 similar to OpenAI schema.
662 Example: `[{"type": "text", "text": "Who are you?"}]`
663
664 add_generation_prompt: If True, adds a generation template
665 to each message.
666 continue_final_message: If True, continues the final message in
667 the conversation instead of starting a new one. Cannot be
668 `True` if `add_generation_prompt` is also `True`.
669 chat_template_kwargs: Additional kwargs to pass to the chat
670 template.
671 tokenization_kwargs: Overrides for `tokenizer.encode`.
672 mm_processor_kwargs: Overrides for `processor.__call__`.
673
674 Returns:
675 A list of `RequestOutput` objects containing the generated
676 responses in the same order as the input messages.
677 """
678 model_config = self.model_config
679 runner_type = model_config.runner_type
680 if runner_type != "generate":
681 raise ValueError(
682 "LLM.chat() is only supported for generative models. "
683 "Try passing `--runner generate` to use the model as a "
684 "generative model."
685 )
686
687 if sampling_params is None:
688 sampling_params = self.get_default_sampling_params()
689
690 return self._run_chat(
691 messages=messages,
692 params=sampling_params,
693 output_type=RequestOutput,
694 use_tqdm=use_tqdm,
695 lora_request=lora_request,
696 chat_template=chat_template,
697 chat_template_content_format=chat_template_content_format,

700 continue_final_message=continue_final_message,
701 tools=tools,

803 def sleep(self, level: int = 1, mode: PauseMode = "abort"):
804 """
805 Put the engine to sleep. The engine should not process any requests.
806 The caller should guarantee that no requests are being processed
807 during the sleep period, before `wake_up` is called.
808
809 Args:
810 level: The sleep level.
811 - Level 0: Pause scheduling but continue accepting requests.
812 Requests are queued but not processed.
813 - Level 1: Offload model weights to CPU, discard KV cache.
814 The content of kv cache is forgotten. Good for
815 sleeping and waking up the engine to run the same
816 model again. Please make sure there's enough CPU
817 memory to store the model weights.
818 - Level 2: Discard all GPU memory (weights + KV cache).
819 Good for sleeping and waking up the engine to run
820 a different model or update the model, where
821 previous model weights are not needed. It reduces
822 CPU memory pressure.
823 mode: How to handle any existing requests, can be "abort", "wait",
824 or "keep".
825 """
826 self.llm_engine.sleep(level=level, mode=mode)

828 def wake_up(self, tags: list[str] | None = None):
829 """
830 Wake up the engine from sleep mode. See the [sleep][vllm.LLM.sleep]
831 method for more details.
832
833 Args:
834 tags: An optional list of tags to reallocate the engine memory
835 for specific memory allocations. Values must be in
836 `("weights", "kv_cache", "scheduling")`. If None, all memory
837 is reallocated. wake_up should be called with all tags
838 (or None) before the engine is used again.
839 Use tags=["scheduling"] to resume from level 0 sleep.
840 """
841 self.llm_engine.wake_up(tags)
