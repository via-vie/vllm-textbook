46class ServeSubcommand(CLISubcommand):
47 """The `serve` subcommand for the vLLM CLI."""
48
49 name = "serve"
50
51 @staticmethod
52 def cmd(args: argparse.Namespace) -> None:
53 # If model is specified in CLI (as positional arg), it takes precedence
54 if hasattr(args, "model_tag") and args.model_tag is not None:
55 args.model = args.model_tag
56
57 if getattr(args, "grpc", False):
58 from vllm.entrypoints.launchers.grpc_server import serve_grpc
59
60 uvloop.run(serve_grpc(args))
61 return
62
63 rust_frontend_path = (
64 envs.VLLM_RUST_FRONTEND_PATH if envs.VLLM_USE_RUST_FRONTEND else None
65 )
66
67 if args.headless:
68 if args.api_server_count is not None and args.api_server_count > 0:
69 raise ValueError(
70 f"--api-server-count={args.api_server_count} cannot be "
71 "used with --headless (no API servers are started in "
72 "headless mode)."
73 )
74 # Default to 0 in headless mode (no API servers)
75 args.api_server_count = 0
76
77 # Detect LB mode for defaulting api_server_count.
78 # Multi-port: --data-parallel-multi-port-external-lb
79 # External LB: --data-parallel-external-lb or --data-parallel-rank
80 # Hybrid LB: --data-parallel-hybrid-lb or --data-parallel-start-rank
81 is_external_lb = (
82 args.data_parallel_external_lb or args.data_parallel_rank is not None
83 )
84
85 # If --data_parallel_multi_port_external_lb and --data_parallel_hybrid_lb
86 # are unset, default to hybrid if --data-parallel-start-rank is set
87 is_hybrid_lb = is_multi_port = False
88 if (
89 not args.data_parallel_hybrid_lb
90 and not args.data_parallel_multi_port_external_lb
91 ):
92 is_hybrid_lb = args.data_parallel_start_rank is not None
93 else:
94 is_hybrid_lb = args.data_parallel_hybrid_lb
95 is_multi_port = args.data_parallel_multi_port_external_lb
96
97 if sum([is_multi_port, is_external_lb, is_hybrid_lb]) > 1:
98 raise ValueError(
99 "Cannot use more than one data parallel load balancing mode. "
100 "Choose one of: --data-parallel-multi-port-external-lb, "
101 "--data-parallel-external-lb (or --data-parallel-rank), "
102 "--data-parallel-hybrid-lb (or --data-parallel-start-rank)."
103 )
104
105 # Default api_server_count if not explicitly set.
106 # - Multi-port: 1 (supervisor spawns one server per local DP rank)
107 # - Rust frontend: 1 (not applicable as it's multithreaded)
108 # - External LB: 1 (external LB handles distribution)
109 # - Hybrid LB: Use local DP size (internal LB for local ranks only)
110 # - Internal LB: Use full DP size
111 if args.api_server_count is None:
112 if is_multi_port or is_external_lb or rust_frontend_path:
113 args.api_server_count = 1
114 elif is_hybrid_lb:
115 args.api_server_count = args.data_parallel_size_local or 1
116 if args.api_server_count > 1:
117 logger.info(
118 "Defaulting api_server_count to data_parallel_size_local "
119 "(%d) for hybrid LB mode.",
120 args.api_server_count,
121 )
122 else:
123 args.api_server_count = args.data_parallel_size
124 if args.api_server_count > 1:
125 logger.info(
126 "Defaulting api_server_count to data_parallel_size (%d).",
127 args.api_server_count,
128 )
129 elif rust_frontend_path and args.api_server_count > 1:
130 logger.warning(
131 "Ignoring --api-server-count=%d when using rust front-end process",
132 args.api_server_count,
133 )
134 args.api_server_count = 1
135
136 # Elastic EP currently only supports running with at most one API server.
137 if getattr(args, "enable_elastic_ep", False) and args.api_server_count > 1:
138 logger.warning(
139 "Elastic EP only supports running with with at most one API server. "
140 "Capping api_server_count from %d to 1.",
141 args.api_server_count,
142 )
143 args.api_server_count = 1
144
145 if is_multi_port:
146 run_dp_supervisor(args)
147 elif args.api_server_count < 1:
148 run_headless(args)
149 elif args.api_server_count > 1 or rust_frontend_path:
150 run_multi_api_server(args)
151 else:
152 # Single API server (this process).
153 args.api_server_count = None
154 uvloop.run(run_server(args))

157 validate_parsed_serve_args(args)

159 def subparser_init(
160 self, subparsers: argparse._SubParsersAction

162 serve_parser = subparsers.add_parser(
163 self.name,
164 help="Launch a local OpenAI-compatible API server to serve LLM "
165 "completions via HTTP.",
166 description=DESCRIPTION,
167 usage="vllm serve [model_tag] [options]",
168 )
169
170 serve_parser = make_arg_parser(serve_parser)
171 serve_parser.epilog = VLLM_SUBCMD_PARSER_EPILOG.format(subcmd=self.name)
172 return serve_parser

179def run_headless(args: argparse.Namespace):
180 if args.api_server_count > 1:
181 raise ValueError("api_server_count can't be set in headless mode")
182
183 if args.reasoning_parser_plugin and len(args.reasoning_parser_plugin) > 3:
184 ReasoningParserManager.import_reasoning_parser(args.reasoning_parser_plugin)
185
186 # Create the EngineConfig.
187 engine_args = vllm.AsyncEngineArgs.from_cli_args(args)
188 usage_context = UsageContext.OPENAI_API_SERVER
189 vllm_config = engine_args.create_engine_config(
190 usage_context=usage_context, headless=True
191 )
192
193 if engine_args.data_parallel_hybrid_lb:
194 raise ValueError("data_parallel_hybrid_lb is not applicable in headless mode")
195
196 parallel_config = vllm_config.parallel_config
197 local_engine_count = parallel_config.data_parallel_size_local
198
199 if local_engine_count <= 0:
200 raise ValueError("data_parallel_size_local must be > 0 in headless mode")
201
202 shutdown_requested = False
203
204 # Catch SIGTERM and SIGINT to allow graceful shutdown.
205 def signal_handler(signum, frame):
206 nonlocal shutdown_requested
207 logger.debug("Received %d signal.", signum)
208 if not shutdown_requested:
209 shutdown_requested = True
210 raise SystemExit
211
212 signal.signal(signal.SIGTERM, signal_handler)
213 signal.signal(signal.SIGINT, signal_handler)
214
215 if parallel_config.node_rank_within_dp > 0:
216 from vllm.version import __version__ as VLLM_VERSION
217
218 # Run headless workers (for multi-node PP/TP).
219 host = parallel_config.master_addr
220 head_node_address = f"{host}:{parallel_config.master_port}"
221 logger.info(
222 "Launching vLLM (v%s) headless multiproc executor, "
223 "with head node address %s for torch.distributed process group.",
224 VLLM_VERSION,
225 head_node_address,
226 )
227
228 executor = MultiprocExecutor(vllm_config, monitor_workers=False)
229 executor.start_worker_monitor(inline=True)
230 return
231
232 host = parallel_config.data_parallel_master_ip
233 port = parallel_config.data_parallel_rpc_port
234 handshake_address = get_tcp_uri(host, port)
235
236 logger.info(
237 "Launching %d data parallel engine(s) in headless mode, "
238 "with head node address %s.",
239 local_engine_count,
240 handshake_address,
241 )
242
243 # Create the engines.
244 engine_manager = CoreEngineProcManager(
245 local_engine_count=local_engine_count,
246 start_index=vllm_config.parallel_config.data_parallel_rank,
247 local_start_index=0,
248 vllm_config=vllm_config,
249 local_client=False,
250 handshake_address=handshake_address,
251 executor_class=Executor.get_class(vllm_config),
252 log_stats=not engine_args.disable_log_stats,
253 )
254
255 try:
256 engine_manager.monitor_engine_liveness()
257 finally:
258 timeout = None
259 if shutdown_requested:
260 timeout = vllm_config.shutdown_timeout
261 logger.info("Waiting up to %d seconds for processes to exit", timeout)
262 engine_manager.shutdown(timeout=timeout)
263 logger.info("Shutting down.")
264

266def run_multi_api_server(args: argparse.Namespace):
267 assert not args.headless
268 rust_frontend_path = (
269 envs.VLLM_RUST_FRONTEND_PATH if envs.VLLM_USE_RUST_FRONTEND else None
270 )
271 num_api_servers: int = args.api_server_count
272 assert num_api_servers > 0
273
274 if rust_frontend_path and num_api_servers > 1:
275 raise ValueError(
276 "VLLM_RUST_FRONTEND_PATH does not support api_server_count > 1"
277 )
278
279 if num_api_servers > 1:
280 setup_multiprocess_prometheus()
281
282 shutdown_requested = False
283
284 # Catch SIGTERM and SIGINT to allow graceful shutdown.
285 def signal_handler(signum, frame):
286 nonlocal shutdown_requested
287 logger.debug("Received %d signal.", signum)
288 if not shutdown_requested:
289 shutdown_requested = True
290 raise SystemExit
291
292 signal.signal(signal.SIGTERM, signal_handler)
293 signal.signal(signal.SIGINT, signal_handler)
294
295 listen_address, sock = setup_server(args, reuse_port=num_api_servers > 1)
296
297 engine_args = vllm.AsyncEngineArgs.from_cli_args(args)
298 engine_args._api_process_count = num_api_servers
299 engine_args._api_process_rank = -1
300
301 usage_context = UsageContext.OPENAI_API_SERVER
302 vllm_config = engine_args.create_engine_config(usage_context=usage_context)
303
304 if num_api_servers > 1 and envs.VLLM_ALLOW_RUNTIME_LORA_UPDATING:
305 raise ValueError(
306 "VLLM_ALLOW_RUNTIME_LORA_UPDATING cannot be used with api_server_count > 1"
307 )
308
309 executor_class = Executor.get_class(vllm_config)
310 log_stats = not engine_args.disable_log_stats
311
312 parallel_config = vllm_config.parallel_config
313 dp_rank = parallel_config.data_parallel_rank
314 assert parallel_config.local_engines_only or dp_rank == 0
315
316 api_server_manager: APIServerProcessManager | RustFrontendProcessManager | None = (
317 None
318 )
319
320 from vllm.v1.engine.utils import get_engine_zmq_addresses
321
322 # Defer port allocation to the child's bind() to avoid TOCTOU, except
323 # for Rust front-end and Ray DP, which can't see the post-bind rebind
324 # (CLI-arg subprocess / pickled-into-actor snapshot respectively) and
325 # so pre-allocate driver-side -- reintroducing the original race only
326 # there.
327 is_ray_dp = parallel_config.data_parallel_backend == "ray"
328 addresses = get_engine_zmq_addresses(
329 vllm_config,
330 num_api_servers,
331 defer_api_server_ports=not (rust_frontend_path or is_ray_dp),
332 )
333
334 with launch_core_engines(
335 vllm_config, executor_class, log_stats, addresses
336 ) as engine_launch:
337 local_engine_manager = engine_launch.engine_manager
338 coordinator = engine_launch.coordinator
339 addresses = engine_launch.addresses
340 stats_update_address = (
341 coordinator.get_stats_publish_address() if coordinator else None
342 )
343
344 if rust_frontend_path:
345 if parallel_config.local_engines_only:
346 expected_engine_start_index = parallel_config.data_parallel_rank
347 expected_engine_count = parallel_config.data_parallel_size_local
348 else:
349 expected_engine_start_index = 0
350 expected_engine_count = parallel_config.data_parallel_size
351 # Start rust front-end process.
352 api_server_manager = RustFrontendProcessManager(
353 binary_path=rust_frontend_path,
354 sock=sock,
355 args=args,
356 input_address=addresses.inputs[0],
357 output_address=addresses.outputs[0],
358 engine_start_index=expected_engine_start_index,
359 engine_count=expected_engine_count,
360 data_parallel_size=parallel_config.data_parallel_size,
361 stats_update_address=stats_update_address,
362 )
363 else:
364 # Start API server(s).
365 api_server_manager = APIServerProcessManager(
366 listen_address=listen_address,
367 sock=sock,
368 args=args,
369 num_servers=num_api_servers,
370 input_addresses=addresses.inputs,
371 output_addresses=addresses.outputs,
372 stats_update_address=stats_update_address,
373 tensor_queue=engine_launch.tensor_queue,
374 )
375
376 if not is_ray_dp:
377 # Forward each child's bound endpoints to the engine handshake
378 # (runs on ``with`` exit). Skipped for Ray DP, where addresses
379 # are pre-allocated above and Ray actors already hold them.
380 actual_inputs, actual_outputs = (
381 api_server_manager.gather_actual_addresses()
382 )
383 addresses.inputs = actual_inputs
384 addresses.outputs = actual_outputs
385
386 # Set frontend processes to watch during engine startup.
387 # If any of these processes exit before the engines are up, the engine startup
388 # will be aborted with an error.
389 engine_launch.watched_frontend_processes = api_server_manager.processes
390
391 # Wait for API servers.
392 try:
393 wait_for_completion_or_failure(
394 api_server_manager=api_server_manager,
395 engine_manager=local_engine_manager,
396 coordinator=coordinator,
397 )
398 finally:
399 timeout = shutdown_by = None
400 if shutdown_requested:
401 timeout = vllm_config.shutdown_timeout
402 shutdown_by = time.monotonic() + timeout
403 logger.info("Waiting up to %d seconds for processes to exit", timeout)
404
405 def to_timeout(deadline: float | None) -> float | None:
406 return (
407 deadline if deadline is None else max(deadline - time.monotonic(), 0.0)
408 )
409
410 api_server_manager.shutdown(timeout=timeout)
411 if local_engine_manager:
412 local_engine_manager.shutdown(timeout=to_timeout(shutdown_by))
413 if coordinator:
414 coordinator.shutdown(timeout=to_timeout(shutdown_by))
