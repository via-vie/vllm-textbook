3"""The CLI entrypoints of vLLM
4
5Note that all future modules must be lazily loaded within main
6to avoid certain eager import breakage."""
7
8import importlib.metadata
9import sys
10from importlib.util import find_spec
11
12from vllm.logger import init_logger
13
14logger = init_logger(__name__)
15
16
17def maybe_run_omni() -> bool:
18 # If `--omni` arg is passed to the CLI, delegate to vLLM Omni's entrypoint handling
19 if "--omni" not in sys.argv:
20 return False
21
22 # NOTE: Check the spec instead of importing directly here, since things could
23 # fail with ImportError due to mismatched versions if things are moved around.
24 spec = find_spec("vllm_omni")
25 if spec is None:
26 logger.error(
27 "--omni flag requires a valid instance of vllm-omni to be installed."
28 )
29 sys.exit(1)
30
31 from vllm_omni.entrypoints.cli.main import main as omni_main
32
33 logger.info("Delegating entrypoint handling to vllm-omni")
34 omni_main()
35
36 return True
37
38
39def main():
40 if maybe_run_omni():
41 return
42
43 import vllm.entrypoints.cli.benchmark.main
44 import vllm.entrypoints.cli.collect_env
45 import vllm.entrypoints.cli.launch
46 import vllm.entrypoints.cli.openai
47 import vllm.entrypoints.cli.run_batch
48 import vllm.entrypoints.cli.serve
49 from vllm.entrypoints.serve.utils.api_utils import (
50 VLLM_SUBCMD_PARSER_EPILOG,
51 cli_env_setup,
52 )
53 from vllm.utils.argparse_utils import FlexibleArgumentParser
54
55 CMD_MODULES = [
56 vllm.entrypoints.cli.openai,
57 vllm.entrypoints.cli.serve,
58 vllm.entrypoints.cli.launch,
59 vllm.entrypoints.cli.benchmark.main,
60 vllm.entrypoints.cli.collect_env,
61 vllm.entrypoints.cli.run_batch,
62 ]
63
64 cli_env_setup()
65
66 vllm.entrypoints.cli.benchmark.main.maybe_exec_rust_bench()
67
68 # For 'vllm bench *': use CPU instead of UnspecifiedPlatform by default
69 if len(sys.argv) > 1 and sys.argv[1] == "bench":
70 logger.debug(
71 "Bench command detected, must ensure current platform is not "
72 "UnspecifiedPlatform to avoid device type inference error"
73 )
74 from vllm import platforms
75
76 if platforms.current_platform.is_unspecified():
77 from vllm.platforms.cpu import CpuPlatform
78
79 platforms.current_platform = CpuPlatform()
80 logger.info(
81 "Unspecified platform detected, switching to CPU Platform instead."
82 )
83
84 parser = FlexibleArgumentParser(
85 description="vLLM CLI",
86 epilog=VLLM_SUBCMD_PARSER_EPILOG.format(subcmd="[subcommand]"),
87 )
88 parser.add_argument(
89 "-v",
90 "--version",
91 action="version",
92 version=importlib.metadata.version("vllm"),
93 )
94 subparsers = parser.add_subparsers(required=False, dest="subparser")
95 cmds = {}
96 for cmd_module in CMD_MODULES:
97 new_cmds = cmd_module.cmd_init()
98 for cmd in new_cmds:
99 cmd.subparser_init(subparsers).set_defaults(dispatch_function=cmd.cmd)
100 cmds[cmd.name] = cmd
101 args = parser.parse_args()
102 if args.subparser in cmds:
103 cmds[args.subparser].validate(args)
104
105 if hasattr(args, "dispatch_function"):
106 args.dispatch_function(args)
107 else:
108 parser.print_help()
109
110
111if __name__ == "__main__":
112 main()
