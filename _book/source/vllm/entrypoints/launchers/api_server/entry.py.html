35@asynccontextmanager
36async def build_async_engine_client(
37 args: Namespace,
38 *,
39 usage_context: UsageContext = UsageContext.OPENAI_API_SERVER,
40 client_config: dict[str, Any] | None = None,
41) -> AsyncIterator[EngineClient]:
42 if os.getenv("VLLM_WORKER_MULTIPROC_METHOD") == "forkserver":
43 # The executor is expected to be mp.

49 logger.debug("Forkserver setup complete!")

66@asynccontextmanager
67async def build_async_engine_client_from_engine_args(
68 engine_args: AsyncEngineArgs,
69 *,
70 usage_context: UsageContext = UsageContext.OPENAI_API_SERVER,
71 client_config: dict[str, Any] | None = None,
72) -> AsyncIterator[EngineClient]:
73 """
74 Create EngineClient, either:
75 - in-process using the AsyncLLMEngine Directly
76 - multiprocess using AsyncLLMEngine RPC
77
78 Returns the Client or None if the creation failed.
79 """
80
81 # Create the EngineConfig (determines if we can use V1).
82 vllm_config = engine_args.create_engine_config(usage_context=usage_context)
83
84 from vllm.v1.engine.async_llm import AsyncLLM
85
86 async_llm: AsyncLLM | None = None
87
88 # Don't mutate the input client_config
89 client_config = dict(client_config) if client_config else {}
90 client_count = client_config.pop("client_count", 1)
91 client_index = client_config.pop("client_index", 0)
92
93 try:
94 async_llm = AsyncLLM.from_vllm_config(
95 vllm_config=vllm_config,
96 usage_context=usage_context,
97 enable_log_requests=engine_args.enable_log_requests,
98 aggregate_engine_logging=engine_args.aggregate_engine_logging,
99 disable_log_stats=engine_args.disable_log_stats,
100 client_addresses=client_config,
101 client_count=client_count,
102 client_index=client_index,
103 )
104
105 # Don't keep the dummy data in memory
106 assert async_llm is not None
107 await async_llm.reset_mm_cache()
108
109 yield async_llm
110 finally:
111 if async_llm:
112 async_llm.shutdown(timeout=vllm_config.shutdown_timeout)

115async def build_and_serve(
116 engine_client: EngineClient,
117 listen_address: str,
118 sock: socket.socket,
119 args: Namespace,
120 **uvicorn_kwargs,
121) -> asyncio.Task:
122 """Build FastAPI app, initialize state, and start serving.
123
124 Returns the shutdown task for the caller to await.
125 """
126
127 # Get uvicorn log config (from file or with endpoint filter)
128 log_config = get_uvicorn_log_config(args)
129 if log_config is not None:
130 uvicorn_kwargs["log_config"] = log_config
131
132 supported_tasks = await engine_client.get_supported_tasks()
133 model_config = engine_client.model_config
134
135 logger.info("Supported tasks: %s", supported_tasks)
136 app = build_app(args, supported_tasks, model_config)
137 await init_app_state(engine_client, app.state, args, supported_tasks)
138
139 logger.info("Starting vLLM server on %s", listen_address)
140
141 return await serve_http(
142 app,
143 sock=sock,
144 enable_ssl_refresh=args.enable_ssl_refresh,
145 host=args.host,
146 port=args.port,
147 log_level=args.uvicorn_log_level,
148 # NOTE: When the 'disable_uvicorn_access_log' value is True,
149 # no access log will be output.
150 access_log=not args.disable_uvicorn_access_log,
151 timeout_keep_alive=envs.VLLM_HTTP_TIMEOUT_KEEP_ALIVE,
152 ssl_keyfile=args.ssl_keyfile,
153 ssl_certfile=args.ssl_certfile,
154 ssl_ca_certs=args.ssl_ca_certs,
155 ssl_cert_reqs=args.ssl_cert_reqs,
156 ssl_ciphers=args.ssl_ciphers,
157 h11_max_incomplete_event_size=args.h11_max_incomplete_event_size,
158 h11_max_header_count=args.h11_max_header_count,
159 **uvicorn_kwargs,
160 )

163async def run_server(args, **uvicorn_kwargs) -> None:
164 """Run a single-worker API server."""
165
166 decorate_logs("APIServer", skip_if_decorated=True)
167
168 # Interrupt initialization if SIGTERM arrives before uvicorn installs its
169 # own signal handlers. Once uvicorn is running it replaces this.
170 def _interrupt_init(*_) -> None:
171 raise KeyboardInterrupt("terminated")
172
173 signal.signal(signal.SIGTERM, _interrupt_init)
174
175 listen_address, sock = setup_server(args, reuse_port=False)
176 await run_server_worker(listen_address, sock, args, **uvicorn_kwargs)
177

179async def run_server_worker(
180 listen_address, sock, args, client_config=None, **uvicorn_kwargs
181) -> None:
182 """Run a single API server worker."""
183
184 if args.tool_parser_plugin and len(args.tool_parser_plugin) > 3:
185 ToolParserManager.import_tool_parser(args.tool_parser_plugin)
186
187 if args.reasoning_parser_plugin and len(args.reasoning_parser_plugin) > 3:
188 ReasoningParserManager.import_reasoning_parser(args.reasoning_parser_plugin)
189
190 async with build_async_engine_client(
191 args,
192 client_config=client_config,
193 ) as engine_client:
194 shutdown_task = await build_and_serve(
195 engine_client, listen_address, sock, args, **uvicorn_kwargs
196 )
197 # NB: Await server shutdown only after the backend context is exited
198 try:
199 await shutdown_task
200 finally:
201 sock.close()
