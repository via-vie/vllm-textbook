58@config
59class EPLBConfig:
60 """Configuration for Expert Parallel Load Balancing (EP)."""
61
62 window_size: int = Field(default=1000, gt=0)
63 """Window size for expert load recording."""
64 step_interval: int = Field(default=3000, gt=0)
65 """
66 Interval for rearranging experts in expert parallelism.
67
68 Note that if this is greater than the EPLB window size, only the metrics
69 of the last `lb_window_size` steps will be used for rearranging experts.
70 """
71
72 num_redundant_experts: int = Field(default=0, ge=0)
73 """Number of redundant experts to use for expert parallelism."""
74
75 log_balancedness: bool = False
76 """
77 Log the balancedness each step of expert parallelism.
78 This is turned off by default since it will cause communication overhead.
79 """
80 log_balancedness_interval: int = Field(default=1, gt=0)
81 """
82 Interval for logging the balancedness.
83 """
84 use_async: bool = True
85 """
86 Whether to use non-blocking EPLB.
87 """
88
89 policy: EPLBPolicyOption = "default"
90 """The policy type for expert parallel load balancing (EPLB)."""
91
92 communicator: EPLBCommunicatorBackend | None = None
93 """
94 Backend for EPLB expert weight communication:
95 - "torch_nccl": Use torch.distributed on the device process group
96 - "torch_gloo": Use torch.distributed gloo with CPU staging
97 - "nixl": Use NIXL with staged send/recv buffers
98 - "pynccl": Use PyNccl send/recv
99 - None: Auto-select backend (prefers "nixl", falls back to "torch_gloo")
100 """
101
102 @model_validator(mode="after")
103 def _validate_eplb_config(self) -> Self:
104 if self.use_async and self.policy != "default":
105 raise ValueError("Async EPLB is only supported with the default policy.")
106 if self.use_async and self.communicator in ("torch_nccl", "pynccl"):
107 raise ValueError(
108 f"{self.communicator} communicator is incompatible with "
109 "async EPLB due to NCCL multi-stream conflicts. Use "
110 "'torch_gloo' or 'nixl' instead, or leave communicator "
111 "unset for automatic selection."
112 )
113 if self.log_balancedness and self.log_balancedness_interval <= 0:
114 raise ValueError("log_balancedness_interval must be greater than 0.")
115 return self

448 @field_validator("numa_bind_nodes")
449 @classmethod
450 def _validate_numa_bind_nodes(cls, value: list[int] | None) -> list[int] | None:
451 if value is None:
452 return None
453 if not value:
454 raise ValueError("numa_bind_nodes must not be empty.")
455 if any(node < 0 for node in value):
456 raise ValueError("numa_bind_nodes must contain non-negative integers.")
457 return value
458
459 @field_validator("numa_bind_cpus")
460 @classmethod
461 def _validate_numa_bind_cpus(cls, value: list[str] | None) -> list[str] | None:
462 if value is None:
463 return None
464 if not value:
465 raise ValueError("numa_bind_cpus must not be empty.")
466
467 for cpuset in value:
468 if not cpuset:
469 raise ValueError("numa_bind_cpus entries must not be empty.")
470 if not _NUMACTL_CPUSET_PATTERN.fullmatch(cpuset):
471 raise ValueError(
472 "numa_bind_cpus entries must use numactl CPU list syntax, "
473 "for example '0-3' or '0,2,4-7'."
474 )
475 for part in cpuset.split(","):
476 if "-" not in part:
477 continue
478 start_str, end_str = part.split("-", 1)
479 if int(start_str) > int(end_str):
480 raise ValueError(
481 f"numa_bind_cpus ranges must be ascending, but got '{cpuset}'."
482 )
483 return value

485 @model_validator(mode="after")
486 def _validate_parallel_config(self) -> Self:
487 if self._api_process_rank >= self._api_process_count:
488 raise ValueError(
489 "Invalid value of `_api_process_rank`. "
490 f"Expected to be `-1` or `[0, {self._api_process_count})`, "
491 f"but found: {self._api_process_rank}"
492 )
493
494 if self.enable_fault_tolerance and self._api_process_count > 1:
495 raise ValueError(
496 "Fault tolerance requires a single API server process "
497 f"(--api-server-count=1), but got {self._api_process_count}. "
498 "The FT system assumes one AsyncMPClient manages all engines."
499 )
500
501 if self.all2all_backend in ["pplx", "naive"]:
502 logger.warning(
503 "The '%s' all2all backend has been removed. "
504 "Falling back to 'allgather_reducescatter'.",
505 self.all2all_backend,
506 )
507 self.all2all_backend = "allgather_reducescatter"
508
509 if self.data_parallel_size_local > self.data_parallel_size:
510 raise ValueError(
511 f"data_parallel_size_local ({self.data_parallel_size_local}) "
512 f"must be <= data_parallel_size ({self.data_parallel_size})"
513 )
514
515 if self.data_parallel_size <= 1 and self.data_parallel_external_lb:
516 raise ValueError(
517 "data_parallel_external_lb can only be set when data_parallel_size > 1"
518 )
519
520 if not self.numa_bind and (
521 self.numa_bind_nodes is not None or self.numa_bind_cpus is not None
522 ):
523 raise ValueError(
524 "numa_bind_nodes and numa_bind_cpus require numa_bind=True."
525 )
526
527 if self.enable_eplb:
528 if not current_platform.is_cuda_alike():
529 raise ValueError(
530 "Expert parallelism load balancing is only supported on "
531 "CUDA devices or ROCm devices now."
532 )
533 if not self.enable_expert_parallel:
534 raise ValueError("enable_expert_parallel must be True to use EPLB.")
535 # The EP group spans the TP x PCP x DP ranks. EPLB therefore needs
536 # TP, PCP, or DP > 1.
537 if (
538 self.tensor_parallel_size
539 * self.prefill_context_parallel_size
540 * self.data_parallel_size
541 <= 1
542 ):
543 raise ValueError(
544 "EPLB requires tensor, prefill-context, or data parallelism, "
545 f"but got TP={self.tensor_parallel_size}, "
546 f"PCP={self.prefill_context_parallel_size}, "
547 f"DP={self.data_parallel_size}."
548 )
549 else:
550 if self.eplb_config.num_redundant_experts != 0:
551 raise ValueError(
552 "num_redundant_experts is set to "
553 f"{self.eplb_config.num_redundant_experts} but EPLB is not "
554 "enabled. Either enable EPLB or unset "
555 "num_redundant_experts."
556 )
557
558 tp = self.tensor_parallel_size
559 pcp = self.prefill_context_parallel_size
560 dcp = self.decode_context_parallel_size
561 if pcp == 1:
562 # DCP reuses the TP ranks when PCP is disabled.
563 if tp % dcp != 0:
564 raise ValueError(f"tp_size={tp} must be divisible by dcp_size={dcp}.")
565 elif dcp not in (1, pcp, tp * pcp):
566 raise ValueError(
567 "When PCP is enabled, DCP must be disabled, span the PCP "
568 "axis, or span the full TP x PCP axis. "
569 f"Got TP={tp}, PCP={pcp}, DCP={dcp}; valid DCP sizes are "
570 f"{sorted({1, pcp, tp * pcp})}."
571 )
572
573 return self

575 def set_dcp_defaults(
576 self,
577 comm_backend: DCPCommBackend = "ag_rs",
578 q_replicate: bool = False,
579 ) -> None:

655 @overload
656 def stateless_init_dp_group(
657 self, return_store: Literal[False] = ...
658 ) -> ProcessGroup: ...
659 @overload
660 def stateless_init_dp_group(
661 self, return_store: Literal[True] = ...
662 ) -> tuple[ProcessGroup, Store]: ...
663 def stateless_init_dp_group(
664 self, return_store: bool = False
665 ) -> ProcessGroup | tuple[ProcessGroup, Store]:
666 # NOTE: In high-concurrency scenarios multiple processes
667 # can pick the same (currently free) port through a race
668 # condition when calling `get_open_port()`. When the first
669 # process binds the port the others will subsequently fail
670 # with `torch.distributed.DistNetworkError: EADDRINUSE`.
671 # To make the initialization more robust we retry a few times
672 # with a fresh port whenever this specific error is observed.
673 from torch.distributed import DistNetworkError
674
675 from vllm.distributed.utils import (
676 stateless_init_torch_distributed_process_group,
677 )
678
679 max_retries = 5
680 last_exc: Exception | None = None
681 for _ in range(max_retries):
682 try:
683 port, listen_socket = self._pick_stateless_dp_port()
684 # use gloo since the engine process might not have cuda device
685 return stateless_init_torch_distributed_process_group(
686 self.data_parallel_master_ip,
687 port,
688 self.data_parallel_rank,
689 self.data_parallel_size,
690 backend="gloo",
691 return_store=return_store,
692 listen_socket=listen_socket,
693 )
694 except DistNetworkError as e:
695 # We only want to retry when the root cause is EADDRINUSE.
696 if "EADDRINUSE" in str(e):
697 logger.warning("Address already in use. Retrying with a new port.")
698 last_exc = e
699 continue # try again with a new port
700 raise e
701
702 # If we get here all retries have failed.
703 assert last_exc is not None
704 raise last_exc

769 @staticmethod
770 def has_unfinished_dp(dp_group: ProcessGroup, has_unfinished: bool) -> bool:
771 tensor = torch.tensor([has_unfinished], dtype=torch.int32, device="cpu")

773 # dp rank 1: has_unfinished_seqs=False
774 # aggregated: has_unfinished_seqs=True
775 # so this is an OR operation, i.e. MAX in integers

777 aggregated_has_unfinished = bool(tensor.item())

780 @staticmethod
781 def sync_dp_state(
782 dp_group: ProcessGroup, has_unfinished: bool, pending_pause: bool
783 ) -> tuple[bool, bool]:
784 """Combined all-reduce for DP state synchronization.
785
786 Uses a single SUM all-reduce on a 2-element tensor:
787 [0] = 1 if this rank has unfinished work, else 0.
788 SUM > 0 ≡ logical OR across ranks → any rank has work.
789 [1] = 1 if this rank has a pending pause request, else 0.
790 SUM == dp_size ≡ all ranks reached pause consensus.
791
792 has_unfinished_global is true if any rank has unfinished work,
793 or if some ranks are waiting for a pause consensus.
794
795 Returns:
796 (has_unfinished_global, pause_consensus)
797 """
798 tensor = torch.tensor(
799 [int(has_unfinished), int(pending_pause)], dtype=torch.int32, device="cpu"
800 )
801 torch.distributed.all_reduce(tensor, op=ReduceOp.SUM, group=dp_group)
802 dp_size = dp_group.size()
803 pause_count = tensor[1].item()
804 has_unfinished_global = tensor[0].item() > 0 or pause_count % dp_size != 0
805 return has_unfinished_global, pause_count == dp_size

818 """
819 Provide a hash that uniquely identifies all the configs
820 that affect the structure of the computation

826 to prevent hangs from mismatched collective communication patterns.

830 "data_parallel_rank",
831 "data_parallel_rank_local",
832 "data_parallel_size_local",
833 "data_parallel_index",
834 "data_parallel_backend",
835 "data_parallel_external_lb",

837 "data_parallel_master_ip",

839 "_data_parallel_master_port_list",
840 "_coord_store_port",
841 "data_parallel_rpc_port",
842 "rank",

844 "master_port",
845 "node_rank",

847 "max_parallel_loading_workers",

849 "ray_workers_use_nsight",

852 "distributed_executor_backend",
853 "worker_cls",
