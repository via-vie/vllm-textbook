37class CompilationMode(enum.IntEnum):
38 """The compilation approach used for torch.compile-based compilation of the
39 model."""

42 """No torch.compile compilation is applied, model runs in fully eager pytorch mode.
43 The model runs as-is."""

46 DYNAMO_TRACE_ONCE = 2

53class CUDAGraphMode(enum.Enum):
54 """Constants for the cudagraph mode in CompilationConfig.
55 Meanwhile, the subset enum `NONE`, `PIECEWISE` and `FULL` are also
56 treated as concrete runtime mode for cudagraph runtime dispatching.
57 """
58
59 NONE = 0
60 PIECEWISE = 1
61 FULL = 2
62 FULL_DECODE_ONLY = (FULL, NONE)
63 FULL_AND_PIECEWISE = (FULL, PIECEWISE)
64
65 def decode_mode(self) -> "CUDAGraphMode":
66 return CUDAGraphMode(self.value[0]) if self.separate_routine() else self
67
68 def mixed_mode(self) -> "CUDAGraphMode":
69 return CUDAGraphMode(self.value[1]) if self.separate_routine() else self
70
71 def has_mode(self, mode: "CUDAGraphMode") -> bool:
72 assert not mode.separate_routine()
73 if self.separate_routine():
74 return mode.value in self.value
75 return self == mode
76
77 def requires_piecewise_compilation(self) -> bool:
78 return self.has_mode(CUDAGraphMode.PIECEWISE)
79
80 def without_piecewise(self) -> "CUDAGraphMode":
81 if self == CUDAGraphMode.PIECEWISE:
82 return CUDAGraphMode.NONE
83 if self == CUDAGraphMode.FULL_AND_PIECEWISE:
84 return CUDAGraphMode.FULL_DECODE_ONLY
85 return self
86
87 def max_cudagraph_mode(self) -> "CUDAGraphMode":
88 return CUDAGraphMode(max(self.value)) if self.separate_routine() else self
89
90 def has_full_cudagraphs(self) -> bool:
91 return self.max_cudagraph_mode() == CUDAGraphMode.FULL
92
93 def has_piecewise_cudagraphs(self) -> bool:
94 return self.requires_piecewise_compilation()
95
96 def separate_routine(self) -> bool:
97 return isinstance(self.value, tuple)
98
99 @classmethod
100 def valid_runtime_modes(cls) -> frozenset["CUDAGraphMode"]:
101 return frozenset({cls.NONE, cls.PIECEWISE, cls.FULL})
102
103 def is_valid_runtime_mode(self) -> bool:
104 return self in CUDAGraphMode.valid_runtime_modes()
105
106 def __str__(self) -> str:
107 return self.name
108
109 def __bool__(self) -> bool:
110 return self != CUDAGraphMode.NONE

341class DynamicShapesType(str, enum.Enum):
342 """Types of dynamic shapes handling in torch.compile().
343 see Dynamic shapes and vllm guard dropping in torch_compile.md
344 for more details."""
345
346 BACKED = "backed"
347 """Use backed dynamic shapes. torch.compile() guards on backed dynamic
348 shapes and may add guards. Symbols are specialized to 0, 1, or >=2 even
349 without encountering branching on those ranges."""
350
351 UNBACKED = "unbacked"
352 """Use unbacked dynamic shapes. Guaranteed not to be guarded on and not
353 0/1 specialized, but may throw data dependent errors when branches require
354 their value without explicit unbacked handling."""
355
356 BACKED_SIZE_OBLIVIOUS = "backed_size_oblivious"
357 """Experimental flag that treats backed symbols as unbacked when explicit
358 unbacked handling is defined."""

404@config
405class CompilationConfig:
406 """Configuration for compilation.
407
408 You must pass CompilationConfig to VLLMConfig constructor.
409 VLLMConfig's post_init does further initialization. If used outside of the
410 VLLMConfig, some fields will be left in an improper state.
411
412 It contains PassConfig, which controls the custom fusion/transformation passes.
413 The rest has three parts:
414
415 - Top-level Compilation control:
416 - [`mode`][vllm.config.CompilationConfig.mode]
417 - [`debug_dump_path`][vllm.config.CompilationConfig.debug_dump_path]
418 - [`cache_dir`][vllm.config.CompilationConfig.cache_dir]
419 - [`backend`][vllm.config.CompilationConfig.backend]
420 - [`custom_ops`][vllm.config.CompilationConfig.custom_ops]
421 - [`splitting_ops`][vllm.config.CompilationConfig.splitting_ops]
422 - [`compile_mm_encoder`][vllm.config.CompilationConfig.compile_mm_encoder]
423 - CudaGraph capture:
424 - [`cudagraph_mode`][vllm.config.CompilationConfig.cudagraph_mode]
425 - [`cudagraph_capture_sizes`]
426 [vllm.config.CompilationConfig.cudagraph_capture_sizes]
427 - [`max_cudagraph_capture_size`]
428 [vllm.config.CompilationConfig.max_cudagraph_capture_size]
429 - [`cudagraph_num_of_warmups`]
430 [vllm.config.CompilationConfig.cudagraph_num_of_warmups]
431 - [`cudagraph_copy_inputs`]
432 [vllm.config.CompilationConfig.cudagraph_copy_inputs]
433 - Inductor compilation:
434 - [`compile_sizes`][vllm.config.CompilationConfig.compile_sizes]
435 - [`compile_ranges_endpoints`]
436 [vllm.config.CompilationConfig.compile_ranges_endpoints]
437 - [`inductor_compile_config`]
438 [vllm.config.CompilationConfig.inductor_compile_config]
439 - [`inductor_passes`][vllm.config.CompilationConfig.inductor_passes]
440 - custom inductor passes
441
442 Why we have different sizes for cudagraph and inductor:
443 - cudagraph: a cudagraph captured for a specific size can only be used
444 for the same size. We need to capture all the sizes we want to use.
445 - inductor: a graph compiled by inductor for a general shape can be used
446 for different sizes. Inductor can also compile for specific sizes,
447 where it can have more information to optimize the graph with fully
448 static shapes. However, we find the general shape compilation is
449 sufficient for most cases. It might be beneficial to compile for
450 certain small batchsizes, where inductor is good at optimizing.
451 """
452
453 # Top-level Compilation control
454 mode: CompilationMode = None # type: ignore[assignment]
455 """The compilation approach used for torch.compile-based compilation of the
456 model.
457
458 - None: If None, we will select the default compilation mode.
459 For V1 engine this is 3.
460 - 0: NONE: No torch.compile compilation is applied, model runs in fully
461 eager pytorch mode. The model runs as-is.
462 - 1: STOCK_TORCH_COMPILE: The standard `torch.compile` compilation pipeline.
463 - 2: DYNAMO_TRACE_ONCE: Single Dynamo trace through the model, avoiding
464 recompilation by removing guards.
465 Requires no dynamic-shape-dependent control-flow.
466 - 3: VLLM_COMPILE: Custom vLLM Inductor-based backend with caching,
467 piecewise compilation, shape specialization, and custom passes."""
468 debug_dump_path: Path | None = None
469 """The path to dump the debug information."""
470 cache_dir: str = ""
471 """The directory to store the compiled graph, to accelerate Inductor
472 compilation. By default, it will use model-related information to generate
473 a cache directory."""
474 compile_cache_save_format: Literal["binary", "unpacked"] = field(
475 default_factory=lambda: envs.VLLM_COMPILE_CACHE_SAVE_FORMAT
476 )
477 """Format for saving torch compile cache:\n
478 - "binary": saves as binary file (multiprocess safe)\n
479 - "unpacked": saves as directory structure for inspection/debugging
480 (NOT multiprocess safe)\n
481 Defaults to `VLLM_COMPILE_CACHE_SAVE_FORMAT` if not specified.
482 """
483 backend: str = ""
484 """The backend for compilation. It needs to be a string:
485
486 - "" (empty string): use the default backend ("inductor" on CUDA-alike
487 platforms).
488 - "eager"/"openxla"/...: use the specified backend registered in PyTorch.
489 - "full.module.name": a qualified name which can be used to import the
490
491 backend function.
492 We use string to avoid serialization issues when using compilation in a
493 distributed setting. When the compilation mode is 1 or 2, the backend is
494 used for the compilation directly (it sees the whole graph). When the
495 compilation mode is 3, the backend supports both whole graph and piecewise
496 compilation, available backends include eager, inductor, and custom backends,
497 the latter of which can be defined via `get_compile_backend`. Furthermore,
498 compilation is only piecewise if splitting ops is set accordingly and
499 use_inductor_graph_partition is off. Note that the default options for
500 splitting ops are sufficient for piecewise compilation.
501 """
502 custom_ops: list[str] = field(default_factory=list)
503 """Fine-grained control over which custom ops to enable/disable. Use 'all'
504 to enable all, 'none' to disable all. Also specify a list of custom op
505 names to enable (prefixed with a '+'), or disable (prefixed with a '-').
506 Examples:
507
508 - 'all,-op1' to enable all except op1
509 - 'none,+op1,+op2' to enable only op1 and op2
510
511 By default, all custom ops are enabled when running without Inductor and
512 disabled when running with Inductor: mode>CompilationMode.NONE and
513 backend="inductor".
514 Inductor generates (fused) Triton kernels for disabled custom ops."""
515
516 ir_enable_torch_wrap: bool = None # type: ignore[assignment]

520 Defaults to True when using Inductor with vllm-compile
521 (backend=="inductor" and mode == VLLM_COMPILE), False otherwise.
522 """

525 """A list of ops to exclude from cudagraphs, used in piecewise compilation.
526
527 The behavior depends on use_inductor_graph_partition:

1323 def custom_op_log_check(self):
1324 """
1325 This method logs the enabled/disabled custom ops and checks that the
1326 passed custom_ops field only contains relevant ops.
1327 It is called at the end of set_current_vllm_config,
1328 after the custom ops have been instantiated.
1329 """
1330
1331 if len(self.enabled_custom_ops) + len(self.disabled_custom_ops) == 0:
1332 logger.debug("No custom ops found in model.")
1333 return
1334
1335 logger.debug("enabled custom ops: %s", self.enabled_custom_ops)
1336 logger.debug("disabled custom ops: %s", self.disabled_custom_ops)
1337
1338 all_ops_in_model = self.enabled_custom_ops | self.disabled_custom_ops
1339 for op in self.custom_ops:
1340 if op in {"all", "none"}:
1341 continue
1342
1343 assert op[0] in {"+", "-"}, (
1344 "Invalid custom op syntax (should be checked during init)"
1345 )
1346
1347 # check if op name exists in model
1348 op_name = op[1:]
1349 if op_name not in all_ops_in_model:
1350 from vllm.model_executor.custom_op import op_registry
1351
1352 # Does op exist at all or is it just not present in this model?
1353 # Note: Only imported op classes appear in the registry.
1354 missing_str = (
1355 "doesn't exist (or wasn't imported/registered)"
1356 if op_name not in op_registry
1357 else "not present in model"
1358 )
1359
1360 enable_str = "enabling" if op[0] == "+" else "disabling"
1361 logger.warning_once(
1362 "Op '%s' %s, %s with '%s' has no effect",
1363 op_name,
1364 missing_str,
1365 enable_str,
1366 op,
1367 )

1369 def is_custom_op_enabled(self, op: str) -> bool:
1370 count_all = self.custom_ops.count("all")
1371 count_none = self.custom_ops.count("none")
1372 if count_all + count_none != 1:
1373 raise ValueError(
1374 "custom_ops must contain exactly one base mode: 'all' or 'none'"
1375 )
1376
1377 if count_all:
1378 return f"-{op}" not in self.custom_ops
1379
1380 return f"+{op}" in self.custom_ops

1382 def resolve_cudagraph_mode_and_sizes(
1383 self,
1384 min_cg_support: "AttentionCGSupport",
1385 min_cg_attn_backend: str | None,
1386 uniform_decode_query_len: int = 1,
1387 use_v2_model_runner: bool = False,
1388 tensor_parallel_size: int = 1,
1389 kv_cache_config: "KVCacheConfig | None" = None,
1390 max_num_reqs: int | None = None,
1391 is_profiling: bool = False,
1392 piecewise_capture_available: bool = True,
1393 ) -> CUDAGraphMode:
1394 from vllm.v1.attention.backend import AttentionCGSupport
1395
1396 cudagraph_mode = self.cudagraph_mode
1397 if cudagraph_mode is None or cudagraph_mode == CUDAGraphMode.NONE:
1398 self.cudagraph_mode = CUDAGraphMode.NONE
1399 return CUDAGraphMode.NONE
1400
1401 # Check cudagraph for mixed batch is supported
1402 if (
1403 cudagraph_mode.mixed_mode() == CUDAGraphMode.FULL
1404 and min_cg_support != AttentionCGSupport.ALWAYS
1405 ):
1406 msg = (
1407 f"CUDAGraphMode.{cudagraph_mode.name} is not supported "
1408 f"with {min_cg_attn_backend} backend (support: "
1409 f"{min_cg_support})"
1410 )
1411 if min_cg_support == AttentionCGSupport.NEVER:
1412 # if not supported any full cudagraphs, just raise it.
1413 msg += (
1414 "; please try cudagraph_mode=PIECEWISE, and "
1415 "make sure compilation mode is VLLM_COMPILE"
1416 )
1417 raise ValueError(msg)
1418
1419 # attempt to resolve the full cudagraph related mode
1420 if self.splitting_ops_contain_attention():
1421 msg += "; setting cudagraph_mode=FULL_AND_PIECEWISE"
1422 cudagraph_mode = CUDAGraphMode.FULL_AND_PIECEWISE
1423 else:
1424 msg += "; setting cudagraph_mode=FULL_DECODE_ONLY"
1425 cudagraph_mode = CUDAGraphMode.FULL_DECODE_ONLY
1426 logger.warning(msg)
1427
1428 # check that if we are doing decode full-cudagraphs it is supported
1429 if (
1430 cudagraph_mode.decode_mode() == CUDAGraphMode.FULL
1431 and min_cg_support == AttentionCGSupport.NEVER
1432 ):
1433 msg = (
1434 f"CUDAGraphMode.{cudagraph_mode.name} is not supported "
1435 f"with {min_cg_attn_backend} backend (support: "
1436 f"{min_cg_support})"
1437 )
1438 if self.mode == CompilationMode.VLLM_COMPILE and (
1439 self.splitting_ops_contain_attention()
1440 or self.use_inductor_graph_partition
1441 ):
1442 msg += (
1443 "; setting cudagraph_mode=PIECEWISE because "
1444 "attention is compiled piecewise"
1445 )
1446 cudagraph_mode = CUDAGraphMode.PIECEWISE
1447 else:
1448 msg += (
1449 "; setting cudagraph_mode=NONE because "
1450 "attention is not compiled piecewise"
1451 )
1452 cudagraph_mode = CUDAGraphMode.NONE
1453 logger.warning(msg)
1454
1455 # check that if we are doing spec-decode + decode full-cudagraphs it is
1456 # supported
1457 if (
1458 cudagraph_mode.decode_mode() == CUDAGraphMode.FULL
1459 and uniform_decode_query_len > 1
1460 and min_cg_support.value < AttentionCGSupport.UNIFORM_BATCH.value
1461 ):
1462 msg = (
1463 f"CUDAGraphMode.{cudagraph_mode.name} is not supported"
1464 f" with spec-decode for attention backend "
1465 f"{min_cg_attn_backend} (support: {min_cg_support})"
1466 )
1467 if self.splitting_ops_contain_attention():
1468 msg += "; setting cudagraph_mode=PIECEWISE"
1469 cudagraph_mode = CUDAGraphMode.PIECEWISE
1470 else:
1471 msg += "; setting cudagraph_mode=NONE"
1472 cudagraph_mode = CUDAGraphMode.NONE
1473 logger.warning(msg)
1474
1475 if (
1476 not piecewise_capture_available
1477 and cudagraph_mode.requires_piecewise_compilation()
1478 ):
1479 fallback_mode = cudagraph_mode.without_piecewise()
1480 logger.warning_once(
1481 "Cudagraph mode %s requires piecewise capture, but the loaded "
1482 "model provides neither a compiled submodule nor breakable CUDA "
1483 "graphs. Overriding to %s.",
1484 cudagraph_mode,
1485 fallback_mode,
1486 )
1487 cudagraph_mode = fallback_mode
1488
1489 # double check that we can support full cudagraph if they are requested
1490 # even after automatic downgrades
1491 if (
1492 cudagraph_mode.has_full_cudagraphs()
1493 and min_cg_support == AttentionCGSupport.NEVER
1494 ):
1495 raise ValueError(
1496 f"CUDAGraphMode.{cudagraph_mode.name} is not "
1497 f"supported with {min_cg_attn_backend} backend ("
1498 f"support:{min_cg_support}) "
1499 "; please try cudagraph_mode=PIECEWISE, "
1500 "and make sure compilation mode is VLLM_COMPILE"
1501 )
1502
1503 # MRV1 adjusts cudagraph sizes to be a multiple of uniform_decode_query_len
1504 # to avoid: https://github.com/vllm-project/vllm/issues/28207 and temp-fix:
1505 # https://github.com/vllm-project/vllm/issues/28207#issuecomment-3504004536
1506 # Will be removed in the near future when we have separate cudagraph capture
1507 # sizes for decode and mixed prefill-decode.
1508 # MRV2 handles cudagraph capture sizing in cudagraph_utils.py
1509 # and doesn't need below: https://github.com/vllm-project/vllm/pull/45953
1510 if (
1511 not use_v2_model_runner
1512 and cudagraph_mode.decode_mode() == CUDAGraphMode.FULL
1513 and uniform_decode_query_len > 1
1514 ):
1515 self.adjust_cudagraph_sizes_for_spec_decode(
1516 uniform_decode_query_len,
1517 tensor_parallel_size,
1518 )
1519
1520 # For Mamba models with FULL decode cudagraphs, each decode
1521 # sequence needs one Mamba cache block. The decode cudagraph
1522 # dispatcher already caps batch sizes at max_num_seqs, so we just
1523 # need to verify that enough blocks exist. Raising here instead
1524 # of silently capping cudagraph_capture_sizes avoids unintended
1525 # restrictions on PIECEWISE (prefill) cudagraphs.
1526 # See: https://github.com/vllm-project/vllm/issues/34094
1527 if (
1528 kv_cache_config is not None
1529 and max_num_reqs is not None
1530 and cudagraph_mode.has_full_cudagraphs()
1531 and not is_profiling
1532 and kv_cache_config.has_mamba_layers
1533 and max_num_reqs > kv_cache_config.num_blocks
1534 ):
1535 raise ValueError(
1536 f"max_num_seqs ({max_num_reqs}) exceeds available Mamba cache "
1537 f"blocks ({kv_cache_config.num_blocks}). Each decode sequence "
1538 "requires one Mamba cache block, so CUDA graph capture cannot "
1539 "proceed. Please lower max_num_seqs to at most "
1540 f"{kv_cache_config.num_blocks} or increase "
1541 "gpu_memory_utilization."
1542 )
1543
1544 self.cudagraph_mode = cudagraph_mode
1545 return cudagraph_mode

1547 def adjust_cudagraph_sizes_for_spec_decode(
1548 self, uniform_decode_query_len: int, tensor_parallel_size: int
1549 ):
1550 multiple_of = uniform_decode_query_len
1551 if tensor_parallel_size > 1 and self.pass_config.enable_sp:
1552 multiple_of = max(uniform_decode_query_len, tensor_parallel_size)
1553 if (
1554 multiple_of % uniform_decode_query_len != 0
1555 or multiple_of % tensor_parallel_size != 0
1556 ):
1557 raise ValueError(
1558 f"Can't determine cudagraph shapes that are both a "
1559 f"multiple of {uniform_decode_query_len} "
1560 f"(num_speculative_tokens + 1) required by spec-decode "
1561 f"and {tensor_parallel_size} (tensor_parallel_size) "
1562 f"required by sequence parallelism please adjust "
1563 f"num_speculative_tokens or disable sequence parallelism"
1564 )
1565
1566 if not self.cudagraph_capture_sizes or multiple_of <= 1:
1567 return
1568
1569 assert self.max_cudagraph_capture_size is not None
1570 rounded_sizes = sorted(
1571 set(
1572 round_up(size, multiple_of)
1573 for size in self.cudagraph_capture_sizes
1574 if round_up(size, multiple_of) <= self.max_cudagraph_capture_size
1575 )
1576 )
1577
1578 if len(rounded_sizes) == 0 and multiple_of <= self.max_cudagraph_capture_size:
1579 # if one valid but would be round_down use that
1580 rounded_sizes = [multiple_of]
1581
1582 if len(rounded_sizes) == 0:
1583 raise ValueError(
1584 f"No valid cudagraph sizes after rounding to multiple of {multiple_of} "
1585 f"(num_speculative_tokens + 1 or tp if sequence parallelism is enabled)"
1586 f" please adjust num_speculative_tokens ({uniform_decode_query_len - 1}"
1587 f") or max_cudagraph_capture_size ({self.max_cudagraph_capture_size})"
1588 f" or cudagraph_capture_sizes ({self.cudagraph_capture_sizes})"
1589 )
1590
1591 self.max_cudagraph_capture_size = rounded_sizes[-1]
1592 self.cudagraph_capture_sizes = rounded_sizes
