151 """Validate the standalone Qwen3-Omni DSpark checkpoint contract."""
152 if not _is_qwen3_omni_target(target_model_config):
153 return

155 draft_hf_config = draft_model_config.hf_config
156 draft_architectures = set(getattr(draft_model_config, "architectures", ()) or ())
157 draft_architectures.update(getattr(draft_hf_config, "architectures", ()) or ())
158 if _QWEN3_OMNI_DSPARK_ARCHITECTURE not in draft_architectures:
159 raise ValueError(
160 "Qwen3-Omni DSpark requires a standalone draft checkpoint with "
161 f"architectures=['{_QWEN3_OMNI_DSPARK_ARCHITECTURE}']; generic "
162 "Qwen3DSparkModel checkpoints must be converted first."

166 if block_size is None:
167 block_size = _get_qwen3_dspark_value(draft_hf_config, "dspark_block_size")
168 if (
169 not isinstance(block_size, int)
170 or isinstance(block_size, bool)
171 or block_size <= 0
172 ):
173 raise ValueError(
174 "Qwen3-Omni DSpark requires a positive integer block_size in the "
175 "draft config."
176 )
177 if num_speculative_tokens != block_size:
178 raise ValueError(
179 "Qwen3-Omni DSpark requires num_speculative_tokens to match the "
180 f"trained block_size ({block_size}); got {num_speculative_tokens}."

184 if target_text_config is None:
185 raise ValueError(
186 "Qwen3-Omni DSpark could not resolve the target thinker text_config."
187 )

189 target_text_config,
190 "hidden_size",
191 target_model_config.get_hidden_size(),
192 )

194 if draft_target_hidden_size != target_hidden_size:
195 raise ValueError(
196 "Qwen3-Omni DSpark draft target_hidden_size must match the target "
197 f"text hidden size ({target_hidden_size}); got {draft_target_hidden_size}."
198 )
199 draft_hidden_size = getattr(draft_hf_config, "hidden_size", None)
200 if draft_hidden_size != target_hidden_size:
201 raise ValueError(
202 "Qwen3-Omni DSpark draft hidden_size must match the target thinker "
203 f"hidden size ({target_hidden_size}); got {draft_hidden_size}."
204 )
205

207 "num_attention_heads": getattr(target_text_config, "num_attention_heads", None),

209 target_text_config,

212 ),

216 "num_attention_heads": getattr(draft_hf_config, "num_attention_heads", None),

221 ),
222 "head_dim": _get_attention_head_dim(draft_hf_config),
223 }
224 for name, target_value in target_attention.items():
225 if (
226 not isinstance(target_value, int)
227 or isinstance(target_value, bool)
228 or target_value <= 0
229 ):

231 "Qwen3-Omni DSpark requires a positive integer "
232 f"{name} in the target thinker text_config; got {target_value}."
233 )
234 draft_value = draft_attention[name]
235 if draft_value != target_value:
236 raise ValueError(
237 f"Qwen3-Omni DSpark draft {name} must match the target thinker "
238 f"value ({target_value}); got {draft_value}."
239 )

242 draft_hf_config, "dflash_config", "target_layer_ids"
243 )
244 if target_layer_ids is None:
245 target_layer_ids = getattr(draft_hf_config, "dspark_target_layer_ids", None)
246 if target_layer_ids is None:
247 target_layer_ids = getattr(draft_hf_config, "target_layer_ids", None)

249 raise ValueError(
250 "Qwen3-Omni DSpark requires a non-empty target_layer_ids list in "
251 "the draft config."
252 )
253 if any(
254 not isinstance(layer_id, int) or isinstance(layer_id, bool)
255 for layer_id in target_layer_ids
256 ):

258 "Qwen3-Omni DSpark target_layer_ids must contain only integers."
259 )

276 raise ValueError(
277 "Qwen3-Omni DSpark requires use_aux_hidden_state=true because Omni "
278 "conditioning is supplied through thinker hidden states."
279 )
280
281 markov_rank = getattr(draft_hf_config, "markov_rank", None)
282 if (
283 not isinstance(markov_rank, int)
284 or isinstance(markov_rank, bool)
285 or markov_rank <= 0
286 ):
287 raise ValueError(

290 )

296 )

302 "Qwen3-Omni DSpark requires sample_from_anchor=true and "
303 "dspark_bonus_anchor=false."
304 )

543 @staticmethod
544 def _acceptance_length_to_rates(length: float, n: int) -> list[float]:
545 """Mean acceptance length to unconditional per-position rates, using
546 the minimum-variance schedule."""
547 num_drafts = length - 1 # expected number of accepted draft tokens
548 num_full = int(num_drafts)
549 return (
550 [1.0] * num_full + [num_drafts - num_full] + [0.0] * (n - num_full - 1)
551 )[:n]
552
553 @staticmethod
554 def _resolve_synthetic_acceptance_rates(
555 n: int,
556 rates: list[float] | None,
557 length: float | None,
558 ) -> list[float]:
559 """Return per-position unconditional acceptance rates from exactly one
560 of `rates` or `length` (validates range, length, and monotonicity)."""
561 if (rates is None) == (length is None):
562 raise ValueError(
563 "rejection_sample_method='synthetic' requires exactly one of "
564 "synthetic_acceptance_rates or synthetic_acceptance_length."
565 )
566 if rates is not None:
567 if len(rates) != n:
568 raise ValueError(
569 f"synthetic_acceptance_rates must have length {n}, got {rates}."
570 )
571 if not all(0.0 <= r <= 1.0 for r in rates):
572 raise ValueError(
573 f"synthetic_acceptance_rates entries must be in [0, 1], "
574 f"got {rates}."
575 )
576 if any(rates[i] > rates[i - 1] for i in range(1, n)):
577 raise ValueError(
578 f"synthetic_acceptance_rates must be non-increasing, got {rates}."
579 )
580 return list(rates)
581 assert length is not None
582 if not 1.0 <= length <= float(n + 1):
583 raise ValueError(
584 f"synthetic_acceptance_length must be in [1, {n + 1}], got {length}."
585 )
586 return SpeculativeConfig._acceptance_length_to_rates(length, n)

589 """How the draft model samples tokens. 'greedy' always picks the argmax
590 token, and the draft probabilities are treated as one-hot during rejection
591 sampling. 'probabilistic' samples stochastically from the draft

600 def compute_hash(self) -> str:
601 """
602 WARNING: Whenever a new field is added to this config,
603 ensure that it is included in the factors list if
604 it affects the computation graph.
605
606 Provide a hash that uniquely identifies all the configs
607 that affect the structure of the computation
608 graph from input ids/embeddings to the final hidden states,
609 excluding anything before input ids/embeddings and after
610 the final hidden states.
611 """
612 factors: list[Any] = []
613 # Eagle3 and extract_hidden_states affect the computation graph because
614 # they return intermediate hidden states in addition to the final hidden state.
615 uses_aux_hidden_states = self.method in (
616 "eagle3",
617 "extract_hidden_states",
618 "dflash",
619 "dspark",
620 )
621 factors.append(uses_aux_hidden_states)
622
623 if self.draft_model_config is not None:
624 factors.append(self.draft_model_config.compute_hash())
625 # The specific layers used also affect the computation graph.
626 layer_ids = getattr(
627 self.draft_model_config.hf_config,
628 "eagle_aux_hidden_state_layer_ids",
629 None,
630 )
631 if layer_ids is not None and uses_aux_hidden_states:
632 # Convert to tuple to make it hashable
633 factors.append(tuple(layer_ids))
634
635 if self.method == "mtp" and self.draft_model_config is not None:
636 factors.append(
637 getattr(
638 self.draft_model_config.hf_config,
639 "index_share_for_mtp_iteration",
640 False,
641 )
642 )
643
644 hash_str = safe_hash(str(factors).encode(), usedforsecurity=False).hexdigest()
645 return hash_str
