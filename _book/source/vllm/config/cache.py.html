73 block_size: int = Field(default=None, gt=0) # type: ignore[assignment]
74 """Size of a contiguous cache block in number of tokens.
75 Accepts None (meaning "use default"). After construction, always int."""
76 user_specified_block_size: bool = field(default=False, init=False)
77 """Whether block_size was explicitly provided. Derived automatically."""
78 user_specified_mamba_block_size: bool = field(default=False, init=False)
79 """Whether mamba_block_size was explicitly provided. Derived automatically."""
80 kv_cache_layout: str | None = field(default=None, init=False)
81 """Resolved physical KV cache layout name (a ``KVCacheLayout`` member).
82
83 ``None`` means the layout has not been resolved yet. The engine core
84 resolves it once (``resolve_kv_cache_layout``) before memory profiling, and
85 every worker process adopts the resolved name — via the
86 ``set_kv_cache_layout`` RPC, or ``KVCacheConfig.kv_cache_layout`` for
87 workers spawned after resolution — before the KV cache is allocated. Once
88 set the value is final and read with ``get_resolved_kv_cache_layout``,
89 which raises on ``None``. Tests and standalone tools may pre-set a value,
90 which resolution then honors as-is."""
91 prefix_match_unit: int | None = Field(default=None, gt=0)
92 """The finest token boundary (in tokens) a prefix-cache hit can land on.
93
94 Prefix-cache keys are computed every `prefix_match_unit` tokens. It can
95 be set finer than the physical KV cache block sizes (e.g. 32 vs a
96 1024-token hybrid-model block) as long as every KV cache group's
97 `block_size` is divisible by it, enabling cache hits at boundaries
98 inside a physical block. It controls matching granularity only, not how
99 often states are stored.
100
101 This equals to the `hash_block_size` used throughout the KV cache code.
102 """
103 gpu_memory_utilization: float = Field(default=0.92, gt=0, le=1)
104 """The fraction of GPU memory to be used for the model executor, which can
105 range from 0 to 1. For example, a value of 0.5 would imply 50% GPU memory
106 utilization. If unspecified, will use the default value of 0.92. This is a
107 per-instance limit, and only applies to the current vLLM instance. It does
108 not matter if you have another vLLM instance running on the same GPU. For
109 example, if you have two vLLM instances running on the same GPU, you can
110 set the GPU memory utilization to 0.5 for each instance."""
111 cache_dtype: CacheDType = "auto"
112 """Data type for kv cache storage. If "auto", will use model data type.
113 CUDA 11.8+ supports fp8 (=fp8_e4m3) and fp8_e5m2. ROCm (AMD GPU) supports
114 fp8 (=fp8_e4m3). Intel Gaudi (HPU) supports fp8 (using fp8_inc).
115 Some models (namely DeepSeekV3.2) default to fp8, set to bfloat16 to use
116 bfloat16 instead, this is an invalid option for models that do not default
117 to fp8.
118 "nvfp4_4over6" uses the NVFP4 layout and selects between max/6 and max/4
119 scales per 16 values by minimizing squared reconstruction error.
120 """
121 is_attention_free: bool = False
122 """Whether the model is attention-free. This is primarily set in
123 `ModelConfig` and that value should be manually duplicated here."""
124 num_gpu_blocks_override: int | None = None
125 """Number of GPU blocks to use. This overrides the profiled `num_gpu_blocks`
126 if specified. Does nothing if `None`. Used for testing preemption."""
127 sliding_window: int | None = None
128 """Sliding window size for the KV cache. This is primarily set in
129 `ModelConfig` and that value should be manually duplicated here."""
130 enable_prefix_caching: bool = True
131 """Whether to enable prefix caching."""
132 prefix_caching_hash_algo: PrefixCachingHashAlgo = "sha256"
133 """Set the hash algorithm for prefix caching:
134
135 - "sha256" uses Pickle for object serialization before hashing. This is the current
136 default, as SHA256 is the most secure choice to avoid potential hash collisions.
137 - "sha256_cbor" provides a reproducible, cross-language compatible hash. It
138 serializes objects using canonical CBOR and hashes them with SHA-256.
139 - "xxhash" uses Pickle serialization with xxHash (128-bit) for faster,
140 non-cryptographic hashing. Requires the optional ``xxhash`` package.
141 IMPORTANT: Use of a hashing algorithm that is not considered cryptographically
142 secure theoretically increases the risk of hash collisions, which can cause
143 undefined behavior or even leak private information in multi-tenant environments.
144 Even if collisions are still very unlikely, it is important to consider your
145 security risk tolerance against the performance benefits before turning this on.
146 - "xxhash_cbor" combines canonical CBOR serialization with xxHash for
147 reproducible hashing. Requires the optional ``xxhash`` package."""
148 prefix_cache_retention_interval: int | None = Field(default=0, ge=0)
149 """Token interval between retained sliding-window and Mamba prefix-cache
150 checkpoints. ``0`` retains only semantic checkpoints, including the latest
151 replay boundary and shared-prefix junctions. Positive values additionally
152 retain periodic checkpoints at the specified interval, which must be a
153 multiple of the scheduler block size. ``None`` retains checkpoints densely.
154 Applies only to sliding-window and Mamba cache groups."""
155 kv_cache_dtype_skip_layers: list[str] = field(default_factory=list)
156 """Layer patterns to skip KV cache quantization. Accepts layer indices
157 (e.g., '0', '2', '4') or attention type names (e.g., 'sliding_window')."""
158 mamba_page_size_padded: int | None = None
159 """ Optional override for mamba page size; used by hybrid mamba/attention
160 models to ensure exact alignment with attention page size."""
161 skip_page_size_padded: int | None = None
162 """Optional override for the page size of layers skipped from KV cache
163 quantization (``--kv-cache-dtype-skip-layers``); set during block-size
164 alignment so unquantized skip layers pad up to the quantized primary's
165 page."""
166 mamba_block_size: int | None = Field(default=None, gt=0)
167 """Size of a contiguous cache block in number of tokens for mamba cache.
168 Can be set only when prefix caching is enabled.
169 Value must be a multiple of 8 to align with causal_conv1d kernel."""
170 mamba_cache_dtype: MambaDType = "auto"
171 """The data type to use for the Mamba cache (both the conv as well as the
172 ssm state). If set to 'auto', the data type will be inferred from the model
173 config."""
174 mamba_ssm_cache_dtype: MambaDType = "auto"
175 """The data type to use for the Mamba cache (ssm state only, conv state will
176 still be controlled by mamba_cache_dtype). If set to 'auto', the data type
177 for the ssm state will be determined by mamba_cache_dtype."""
178 mamba_cache_mode: MambaCacheMode = "none"
179 """The cache strategy for Mamba layers:
180
181 - "none": set when prefix caching is disabled.
182 - "all": cache the mamba state of all tokens at position i * block_size.
183 - "align": only cache the mamba state of the last token of each scheduler step and
184 when the token is at position i * block_size. This is the default when prefix
185 caching is enabled.
186 """
187 enable_mamba_fine_grained_prefix_cache: bool = False
188 """Also register a Mamba "align" checkpoint at the shared-prefix junction --
189 where an EAGLE/MTP sibling was observed to resume -- instead of only at the
190 prompt tail. Off by default; only takes effect with `mamba_cache_mode`
191 "align", EAGLE on the Mamba group, and a prefix match unit smaller than the
192 Mamba block size."""
193 replayssm_buffer_len: int = Field(default=16, gt=0)
194 """ReplaySSM logical history length B for Mamba2. Triton uses B physical
195 rows and FlashInfer uses B+1. Kimi-K3 speculative decode does not use B.
196 Default 16."""
197 use_replayssm: bool = False
198 """Use the ReplaySSM Mamba2 decode kernel: cache recent SSM inputs and skip
199 the per-step full-state store, writing the checkpoint back only on flush.
200 Requires mamba_cache_mode 'none' or 'align' (prefix caching) and the Triton
201 or FlashInfer mamba backend; standard (non-speculative) decode only. In align
202 mode flushes are most efficient when mamba_block_size is a multiple of
203 replayssm_buffer_len, but this is not required."""
204 use_kda_recoverssm: bool = field(default=False, init=False)
205 """Whether Kimi-K3 KDA uses RecoverSSM speculative decode."""

207 # Will be set after profiling.
208 num_gpu_blocks: int | None = field(default=None, init=False)

229 """Size of KV Cache per GPU in bytes. By default, this is set to None

248 def compute_hash(self) -> str:
249 """
250 WARNING: Whenever a new field is added to this config,
251 ensure that it is included in the factors list if
252 it affects the computation graph.
253
254 Provide a hash that uniquely identifies all the configs
255 that affect the structure of the computation
256 graph from input ids/embeddings to the final hidden states,
257 excluding anything before input ids/embeddings and after
258 the final hidden states.
259 """
260 ignored_factors = {
261 # Runtime/derived knobs that don't affect compiled graph shape
262 "gpu_memory_utilization",
263 "kv_cache_memory_bytes",
264 "is_attention_free",
265 "num_gpu_blocks_override",
266 "enable_prefix_caching",
267 "prefix_caching_hash_algo",
268 "prefix_cache_retention_interval",
269 # Prefix-caching implementation detail (doesn't affect compiled graph).
270 "prefix_match_unit",
271 "enable_mamba_fine_grained_prefix_cache",
272 "mamba_page_size_padded",
273 "skip_page_size_padded",
274 "user_specified_block_size",
275 "user_specified_mamba_block_size",
276 "_block_size_resolved",
277 # Post-init/derived counters
278 "num_gpu_blocks",
279 "num_cpu_blocks",
280 "kv_cache_size_tokens",
281 "kv_cache_max_concurrency",
282 # WIP feature toggle not impacting compiled graph shape
283 "kv_sharing_fast_prefill",
284 }
285
286 from vllm.config.utils import get_hash_factors, hash_factors
287
288 factors = get_hash_factors(self, ignored_factors)
289 return hash_factors(factors)

296 _block_size_resolved: bool = field(default=False, init=False)
297 """Guard against pydantic re-running _apply_block_size_default."""
298
299 @field_validator("block_size", mode="wrap")
300 @classmethod
301 def _skip_none_validation(cls, value: Any, handler: Callable) -> Any:
302 if value is None:
303 return value
304 return handler(value)
305
306 @model_validator(mode="after")
307 def _apply_block_size_default(self) -> "CacheConfig":
308 # Pydantic re-runs validators when CacheConfig is nested inside
309 # another pydantic model (e.g. VllmConfig). Guard against that.
310 if self._block_size_resolved:
311 return self
312 self._block_size_resolved = True
313 if self.block_size is None:
314 self.block_size = self.DEFAULT_BLOCK_SIZE
315 else:
316 self.user_specified_block_size = True
317 if self.mamba_block_size is not None:
318 self.user_specified_mamba_block_size = True
319 return self
320
321 @field_validator("mamba_cache_mode", mode="after")
322 @classmethod
323 def _validate_mamba_cache_mode(cls, mode: MambaCacheMode) -> MambaCacheMode:
324 if mode == "all":
325 logger.warning_once(
326 "Mamba cache mode 'all' is deprecated and will be removed in an "
327 "upcoming release. If this is a problem, please open an issue "
328 "at https://github.com/vllm-project/vllm/issues."
329 )
330 return mode

332 @field_validator("cache_dtype", mode="after")
333 @classmethod
334 def _validate_cache_dtype(cls, cache_dtype: CacheDType) -> CacheDType:
335 if kv_cache_uses_per_token_head_scales(cache_dtype):
336 logger.info(
337 "Using %s data type to store kv cache. It reduces the GPU "
338 "memory footprint and boosts the performance. "
339 "Dynamic per-token-head scales will be computed at runtime.",
340 str(cache_dtype),
341 )
342 elif is_quantized_kv_cache(cache_dtype):
343 logger.info(
344 "Using %s data type to store kv cache. It reduces the GPU "
345 "memory footprint and boosts the performance. "
346 "Meanwhile, it may cause accuracy drop without a proper "
347 "scaling factor",
348 str(cache_dtype),
349 )
350 return cache_dtype

352 def get_resolved_kv_cache_layout(self) -> KVCacheLayout:
353 if self.kv_cache_layout is None:
354 raise ValueError(
355 "KV cache layout has not been resolved yet; it is resolved once "
356 "by the engine core (resolve_kv_cache_layout) unless explicitly "
357 "set by the user."
358 )
359 return _layout_from_name(self.kv_cache_layout)
