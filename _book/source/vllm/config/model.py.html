480 def _update_nested(
481 self,
482 target: PretrainedConfig | dict[str, Any],
483 updates: dict[str, Any],
484 ) -> None:
485 """Recursively updates a config or dict with nested updates."""
486 for key, value in updates.items():
487 if isinstance(value, dict):
488 # Get the nested target
489 if isinstance(target, dict):
490 nested_target = target.get(key)
491 else:
492 nested_target = getattr(target, key, None)
493
494 # If nested target exists and can be updated recursively
495 if nested_target is not None and (
496 isinstance(nested_target, dict)
497 or hasattr(nested_target, "__dict__")
498 ):
499 self._update_nested(nested_target, value)
500 continue
501
502 # Set the value (base case)
503 if isinstance(target, dict):
504 target[key] = value
505 else:
506 setattr(target, key, value)

508 def _apply_dict_overrides(
509 self,
510 config: PretrainedConfig,
511 overrides: dict[str, Any],
512 ) -> None:
513 """Apply dict overrides, handling both nested configs and dict values."""
514 from transformers import PretrainedConfig
515
516 for key, value in overrides.items():
517 attr = getattr(config, key, None)
518 if attr is not None and isinstance(attr, PretrainedConfig):
519 # It's a nested config - recursively update it
520 self._update_nested(attr, value)
521 else:
522 # It's a dict-valued parameter - set it directly
523 setattr(config, key, value)

871 def _supports_multimodal_for_mm_prefix(self) -> bool:
872 """Whether multimodal inputs can still appear for this deployment.
873

877 ``with_hf_config``, which deep-copies this ``ModelConfig`` and swaps

889 return cached

892 # Early call before multimodal init — do not clear mm_prefix yet.
893 return True

895 from vllm.multimodal import MULTIMODAL_REGISTRY
896
897 supports_mm = MULTIMODAL_REGISTRY.supports_multimodal_inputs(self)

902 "are configuration-disabled. Attention backends without "

1130 def maybe_untie_word_embeddings(self) -> None:
1131 """Stop trusting `tie_word_embeddings` when the checkpoint disagrees.

1135 that tensor, so build the `lm_head` as if untied and let it load. The
1136 two are compared once loaded, and re-tied if they turn out to match, by
1137 [maybe_retie_word_embeddings][vllm.model_executor.model_loader.weight_tying.maybe_retie_word_embeddings].
1138
1139 Transformers makes the same decision in `PreTrainedModel.tie_weights`,
1140 where it can compare the two tensors directly.
1141 """

1143 return

1145 return
1146
1147 logger.debug(
1148 "The config for %s says the word embeddings are tied, but the checkpoint "
1149 "contains an lm_head. Loading it to find out whether they really are tied.",
1150 self.model,
1151 )

1153 # level, and the text config is what the language model itself sees.
1154 self.hf_config.tie_word_embeddings = False
1155 self.hf_config.get_text_config().tie_word_embeddings = False
1156 self.word_embeddings_untied_by_checkpoint = True

1258 def _verify_quantization(self) -> None:
1259 supported_quantization = me_quant.QUANTIZATION_METHODS
1260 if self.quantization is not None:
1261 self.quantization = cast(me_quant.QuantizationMethods, self.quantization)
1262
1263 # Parse quantization method from the HF model config, if available.
1264 quant_cfg = self.model_arch_config.quantization_config
1265
1266 if quant_cfg is not None:
1267 quant_method = quant_cfg["quant_method"]
1268 # Quantization methods which are overrides (i.e. they have a
1269 # `override_quantization_method` method) must be checked in order
1270 # of preference (this is particularly important for GPTQ).
1271 overrides = [
1272 "auto_gptq",
1273 "gptq",
1274 "gptq_marlin",
1275 "auto_awq",
1276 "awq",
1277 "awq_marlin",
1278 "inc",
1279 "moe_wna16",
1280 "modelopt",
1281 "modelopt_fp4",
1282 "modelopt_mxfp8",
1283 "mxfp8",
1284 "modelopt_mixed",

1286 # imports during override detection (e.g., MXFP4 imports Triton)
1287 "mxfp4",
1288 "gpt_oss_mxfp4",
1289 "deepseek_v4_fp8",
1290 "humming",
1291 ]
1292 # if the user specifies humming, we should always use humming
1293 if self.quantization == "humming":
1294 overrides = ["humming"] + overrides
1295 quantization_methods = [
1296 q for q in supported_quantization if q not in overrides
1297 ]
1298 # Any custom overrides will be in quantization_methods so we place
1299 # them at the start of the list so custom overrides have preference
1300 # over the built-in ones.
1301 quantization_methods = quantization_methods + overrides
1302
1303 # Detect which checkpoint is it
1304 for name in quantization_methods:
1305 method = me_quant.get_quantization_config(name)
1306 quantization_override = method.override_quantization_method(
1307 quant_cfg, self.quantization, hf_config=self.hf_config
1308 )
1309 if quantization_override is not None:
1310 # Raise error if the override is not custom (custom would
1311 # be in QUANTIZATION_METHODS but not QuantizationMethods)
1312 # and hasn't been added to the overrides list.
1313 if (
1314 name in get_args(me_quant.QuantizationMethods)
1315 and name not in overrides
1316 ):
1317 raise ValueError(
1318 f"Quantization method {name} is an override but "
1319 "is has not been added to the `overrides` list "
1320 "above. This is necessary to ensure that the "
1321 "overrides are checked in order of preference."
1322 )

1324 self.quantization = quantization_override

1326
1327 quant_method = quant_method if quant_method != "" else None
1328 # Verify quantization configurations.
1329 if self.quantization is None:
1330 self.quantization = quant_method
1331 elif self.quantization != quant_method:

1333 "Quantization method specified in the model config "
1334 f"({quant_method}) does not match the quantization "
1335 f"method specified in the `quantization` argument "
1336 f"({self.quantization})."
1337 )
1338
1339 if self.quantization is not None:
1340 if self.quantization not in supported_quantization:
1341 raise ValueError(
1342 f"Unknown quantization method: {self.quantization}. Must "
1343 f"be one of {supported_quantization}."
1344 )
1345 current_platform.verify_quantization(self.quantization)
1346
1347 if self.quantization in me_quant.DEPRECATED_QUANTIZATION_METHODS:
1348 if self.allow_deprecated_quantization:
1349 logger.warning(
1350 "The quantization method %s is deprecated "
1351 "and will be removed in future versions of vLLM.",

1357 "and will be removed in future versions of vLLM. To bypass, "

1600 # for w/o this attribute, we will need to have workarounds like so

1606

1752 "temperature",
1753 "top_k",
1754 "top_p",
1755 "min_p",
