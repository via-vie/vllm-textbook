25@config
26class SchedulerConfig:
27 """Scheduler configuration."""
28
29 max_model_len: InitVar[int]
30 """Maximum length of a sequence (including prompt and generated text).
31
32 Note: This is stored in the ModelConfig, and is used only here to
33 provide fallbacks and validate other attributes."""
34
35 is_encoder_decoder: InitVar[bool]
36 """True if the model is an encoder-decoder model.
37
38 Note: This is stored in the ModelConfig, and is used only here to
39 disable chunked prefill and prefix caching for encoder-decoder models.
40 """
41
42 DEFAULT_MAX_NUM_BATCHED_TOKENS: ClassVar[int] = 2048
43 DEFAULT_MAX_NUM_BATCHED_TOKENS_FOR_BATCHED_DP: ClassVar[int] = 256
44 DEFAULT_MAX_NUM_SEQS: ClassVar[int] = 128
45
46 runner_type: RunnerType = "generate"
47 """The runner type to launch for the model."""
48
49 max_num_batched_tokens: int = Field(default=DEFAULT_MAX_NUM_BATCHED_TOKENS, ge=1)
50 """Maximum number of tokens that can be processed in a single iteration.
51
52 The default value here is mainly for convenience when testing.
53 In real usage, this should be set in `EngineArgs.create_engine_config`.
54 """
55
56 max_num_scheduled_tokens: int | None = Field(default=None, ge=0)
57 """Maximum number of tokens that the scheduler may issue in a single iteration.
58
59 This is usually equal to max_num_batched_tokens, but can be smaller in cases
60 when the model might append tokens into the batch (such as speculative decoding).
61 Defaults to max_num_batched_tokens."""
62
63 max_num_seqs: int = Field(default=DEFAULT_MAX_NUM_SEQS, ge=1)
64 """Maximum number of sequences to be processed in a single iteration.
65
66 The default value here is mainly for convenience when testing.
67 In real usage, this should be set in `EngineArgs.create_engine_config`.
68 """
69
70 long_prefill_token_threshold: int = Field(default=0, ge=0)
71 """For chunked prefill, a request is considered long if the prompt is
72 longer than this number of tokens. 0 disables the cap (default)."""
73
74 max_num_queued_reqs: int | None = Field(default=None, ge=0)
75 """Maximum number of requests that can be in-flight (waiting or running)
76 at the same time, or None for no limit. When the limit is reached, new
77 requests are rejected with HTTP 503 so the client can retry on another
78 instance. This bounds vLLM's otherwise unbounded request queue and is
79 primarily a coarse capacity valve.
80
81 Unlike ``max_num_seqs``, which applies per data-parallel rank, this
82 limit is enforced in the API server process and counts in-flight
83 requests across all DP ranks it routes to. Size it as roughly
84 ``data_parallel_size * max_num_seqs`` plus the desired queue depth if
85 it should not bind before per-rank admission does."""
86
87 max_num_queued_tokens: int | None = Field(default=None, ge=0)
88 """Maximum total prompt tokens of requests currently in the prefill
89 phase, or None for no limit. When the limit is reached, new requests
90 are rejected with HTTP 503.
91
92 This is a TTFT QoS mechanism: by setting it to
93 ``target_TTFT * prefill_throughput`` you reject requests when the
94 prefill backlog would exceed the latency target. In a disaggregated
95 prefill-decode setup this maps directly to the prefill pool's
96 capacity.
97
98 Like ``max_num_queued_reqs``, this limit is enforced in the API
99 server process and covers the prefill backlog across all DP ranks it
100 routes to, so ``prefill_throughput`` in the formula above is the
101 aggregate throughput of the deployment.
102
103 Note: the count is conservative. A partially prefilled request
104 still contributes its full ``prompt_len`` until it transitions out
105 of the prefill phase, because the scheduler's per-iteration
106 ``num_computed_tokens`` progress is not propagated to the API
107 server process during prefill (``EngineCoreOutput`` is only
108 emitted once the request starts producing tokens). Similarly,
109 prefix-cache hits (``num_cached_tokens``) are only known to the
110 OutputProcessor after prefill completes. This overestimates the
111 real backlog, causing earlier rejection than strictly necessary
112 — the safe direction for QoS. The impact is limited to long
113 prompts under chunked prefill; short prompts that prefill in a
114 single iteration are unaffected."""
115
116 enable_chunked_prefill: bool = True
117 """If True, prefill requests can be chunked based
118 on the remaining `max_num_batched_tokens`.
119
120 The default value here is mainly for convenience when testing.
121 In real usage, this should be set in `EngineArgs.create_engine_config`.
122 """
123
124 is_multimodal_model: bool = False
125 """True if the model is multimodal."""
126
127 # TODO (ywang96): Make this configurable.
128 max_num_encoder_input_tokens: int = Field(init=False)
129 """Multimodal encoder compute budget, only used in V1.
130
131 NOTE: This is not currently configurable. It will be overridden by
132 max_num_batched_tokens in case max multimodal embedding size is larger."""
133
134 # TODO (ywang96): Make this configurable.
135 encoder_cache_size: int = Field(init=False)
136 """Multimodal encoder cache size, only used in V1.
137
138 NOTE: This is not currently configurable. It will be overridden by
139 max_num_batched_tokens in case max multimodal embedding size is larger."""
140
141 policy: SchedulerPolicy = "fcfs"
142 """The scheduling policy to use:
143
144 - "fcfs" means first come first served, i.e. requests are handled in order
145 of arrival.
146 - "priority" means requests are handled based on given priority (lower
147 value means earlier handling) and time of arrival deciding any ties)."""
148
149 disable_chunked_mm_input: bool = False
150 """If set to true and chunked prefill is enabled, we do not want to
151 partially schedule a multimodal item. Only used in V1
152 This ensures that if a request has a mixed prompt
153 (like text tokens TTTT followed by image tokens IIIIIIIIII) where only
154 some image tokens can be scheduled (like TTTTIIIII, leaving IIIII),
155 it will be scheduled as TTTT in one step and IIIIIIIIII in the next."""
156
157 # scheduler class or path. "vllm.v1.core.sched.scheduler.Scheduler"
158 # (default) or "mod.custom_class".
159 scheduler_cls: str | type[object] | None = None
160 """The scheduler class to use. "vllm.v1.core.sched.scheduler.Scheduler" is
161 the default scheduler. Can be a class directly or the path to a class of
162 form "mod.custom_class"."""
163
164 disable_hybrid_kv_cache_manager: bool | None = None
165 """If set to True, KV cache manager will allocate the same size of KV cache
166 for all attention layers even if there are multiple type of attention layers
167 like full attention and sliding window attention.
168 If set to None, the default value will be determined based on the environment
169 and starting configuration.
170 """
171
172 scheduler_reserve_full_isl: bool = True
173 """If True, the scheduler checks whether the full input sequence length
174 fits in the KV cache before admitting a new request, rather than only
175 checking the first chunk. Prevents over-admission and KV cache thrashing
176 with chunked prefill."""
177
178 watermark: float = Field(default=0.0, ge=0.0, lt=1.0)
179 """Fraction of total KV cache blocks to keep free (the watermark) when
180 admitting waiting or preempted requests into the running queue. This headroom
181 helps avoid frequent KV cache eviction and the resulting repeated preemption
182 of requests when GPU memory is scarce. Must be in the range [0.0, 1.0); 0.0
183 (the default) disables the watermark."""
184
185 prefill_schedule_interval: int = Field(default=1, ge=1)
186 """For data-parallel deployments, only admit new prefill requests
187 once every N engine steps, aligned across DP ranks, to better balance
188 per-step forward-pass times."""
189
190 async_scheduling: bool | None = None
191 """If set to False, disable async scheduling. Async scheduling helps to
192 avoid gaps in GPU utilization, leading to better latency and throughput.
193 """
194
195 stream_interval: int = Field(default=1, ge=1)
196 """The interval (or buffer size) for streaming in terms of token length.
197 A smaller value (1) makes streaming smoother by sending each token immediately,
198 while a larger value (e.g., 10) reduces host overhead and may increase throughput
199 by batching multiple tokens before sending."""

212 def get_scheduler_cls(self) -> type["SchedulerInterface"]:
213 if self.scheduler_cls is None:
214 if self.async_scheduling:
215 from vllm.v1.core.sched.async_scheduler import AsyncScheduler
216
217 return AsyncScheduler
218 from vllm.v1.core.sched.scheduler import Scheduler

220 return Scheduler

222 # The first half of this warning can be removed once the Scheduler interface is
223 # finalized and we can maintain support for scheduler classes that implement it
224 logger.warning_once(
225 "Using custom scheduler class %s. This scheduler interface is not public "
226 "and compatibility may not be maintained. If you have subclassed Scheduler "
227 "instead of AsyncScheduler, you will see degraded performance due to async "
228 "scheduling being disabled.",
229 self.scheduler_cls, # type: ignore[arg-type]
230 )
231 if not isinstance(self.scheduler_cls, str):
232 return cast(type["SchedulerInterface"], self.scheduler_cls)
233 return resolve_obj_by_qualname(self.scheduler_cls)

235 def compute_hash(self) -> str:
236 """
237 WARNING: Whenever a new field is added to this config,
238 ensure that it is included in the factors list if
239 it affects the computation graph.
240
241 Provide a hash that uniquely identifies all the configs
242 that affect the structure of the computation
243 graph from input ids/embeddings to the final hidden states,
244 excluding anything before input ids/embeddings and after
245 the final hidden states.
246 """
247 factors: list[Any] = []
248
249 # max_num_batched_tokens need to be included in the hash due
250 # to two reasons:
251 # 1. LoRA creates static buffers based on max_num_batched_tokens.
252 # The tensor sizes and strides get captured in the torch.compile
253 # graph explicitly.
254 # 2. Inductor decides whether using 32-bit or 64-bit indexing integer
255 # based on the data sizes. `max_num_batched_tokens` has an
256 # impact on that. For more details, please check
257 # https://github.com/vllm-project/vllm/issues/29585
258 factors.append(self.max_num_batched_tokens)
259
260 # PLE and other model components allocate static per-request buffers.
261 # Their shapes are captured in compiled graphs.
262 factors.append(self.max_num_seqs)
263
264 hash_str = safe_hash(str(factors).encode(), usedforsecurity=False).hexdigest()
265 return hash_str

273 def __post_init__(self, max_model_len: int, is_encoder_decoder: bool) -> None:
274 if is_encoder_decoder:
275 # Chunked prefill should be disabled for encoder-decoder models.
276 self.disable_chunked_mm_input = True
277 self.enable_chunked_prefill = False
278 self.long_prefill_token_threshold = 0
279 logger.info(
280 "Encoder-decoder models do not support chunked prefill nor"
281 " prefix caching; disabling both."
282 )
283
284 self.max_num_encoder_input_tokens = self.max_num_batched_tokens
285 self.encoder_cache_size = self.max_num_batched_tokens
286
287 if self.enable_chunked_prefill:
288 logger.info_once(
289 "Chunked prefill is enabled with max_num_batched_tokens=%d.",
290 self.max_num_batched_tokens,
291 )
292
293 self.verify_max_model_len(max_model_len)
294
295 def verify_max_model_len(self, max_model_len: int) -> Self:
296 if (
297 self.max_num_batched_tokens < max_model_len
298 and not self.enable_chunked_prefill
299 ):
300 raise ValueError(
301 f"max_num_batched_tokens ({self.max_num_batched_tokens}) is "
302 f"smaller than max_model_len ({max_model_len}). "
303 "This effectively limits the maximum sequence length to "
304 "max_num_batched_tokens and makes vLLM reject longer "
305 "sequences. Please increase max_num_batched_tokens or "
306 "decrease max_model_len."
307 )
308
309 if self.max_num_batched_tokens < self.max_num_seqs:
310 raise ValueError(
311 f"max_num_batched_tokens ({self.max_num_batched_tokens}) must "
312 "be greater than or equal to max_num_seqs "
313 f"({self.max_num_seqs})."
314 )
315
316 if self.max_num_batched_tokens > self.max_num_seqs * max_model_len:
317 logger.warning(
318 "max_num_batched_tokens (%d) exceeds max_num_seqs "
319 "* max_model_len (%d). This may lead to unexpected behavior.",
320 self.max_num_batched_tokens,
321 self.max_num_seqs * max_model_len,
322 )
323
324 if self.long_prefill_token_threshold > max_model_len:
325 raise ValueError(
326 "long_prefill_token_threshold "
327 f"({self.long_prefill_token_threshold}) cannot be greater "
328 f"than the max_model_len ({max_model_len})."
329 )
330
331 return self
