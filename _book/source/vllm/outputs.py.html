32@dataclass
33class CompletionOutput:
34 """The output data of one completion output of a request.
35
36 Args:
37 index: The index of the output in the request.
38 text: The generated output text.
39 token_ids: The token IDs of the generated output text.
40 cumulative_logprob: The cumulative log probability of the generated
41 output text.
42 logprobs: The log probabilities of the top probability words at each
43 position if the logprobs are requested.
44 sampling_mask: The post-processing token support set for each generated
45 token, if requested.
46 finish_reason: The reason why the sequence is finished.
47 stop_reason: The stop string or token id that caused the completion
48 to stop, None if the completion finished for some other reason
49 including encountering the EOS token.
50 lora_request: The LoRA request that was used to generate the output.
51 spec_decode_metrics: Per-sequence speculative-decoding acceptance metrics,
52 populated on finish when speculative decoding ran and
53 ``--per-request-spec-decode-metrics`` is enabled; None otherwise.
54 Surfaced in the response as ``metrics.speculative_decoding`` for
55 single-sequence (``n == 1``) requests.
56 """
57
58 index: int
59 text: str
60 token_ids: GenericSequence[int]
61 cumulative_logprob: float | None
62 logprobs: SampleLogprobs | None
63 routed_experts: np.ndarray | None = None # [seq_len,layer_num,topk]
64 finish_reason: str | None = None
65 stop_reason: int | str | None = None
66 lora_request: LoRARequest | None = None
67 sampling_mask: SamplingMask | None = None
68 spec_decode_metrics: RequestSpecDecodeMetrics | None = None
69
70 def finished(self) -> bool:
71 return self.finish_reason is not None
72
73 def __repr__(self) -> str:
74 return (
75 f"CompletionOutput(index={self.index}, "
76 f"text={self.text!r}, "
77 f"token_ids={self.token_ids}, "
78 f"routed_experts={self.routed_experts}, "
79 f"sampling_mask={self.sampling_mask}, "
80 f"cumulative_logprob={self.cumulative_logprob}, "
81 f"logprobs={self.logprobs}, "
82 f"finish_reason={self.finish_reason}, "
83 f"stop_reason={self.stop_reason})"

106class RequestOutput:
107 """The output data of a completion request to the LLM.
108
109 Args:
110 request_id: The unique ID of the request.
111 prompt: The prompt string of the request.
112 For encoder/decoder models, this is the
113 decoder input prompt.
114 prompt_token_ids: The token IDs of the prompt.
115 For encoder/decoder models, this is the
116 decoder input prompt token ids.
117 prompt_logprobs: The log probabilities to return per prompt token.
118 outputs: The output sequences of the request.
119 finished: Whether the whole request is finished.
120 metrics: Metrics associated with the request.
121 lora_request: The LoRA request that was used to generate the output.
122 encoder_prompt: The encoder prompt string of the request.
123 None if decoder-only.
124 encoder_prompt_token_ids: The token IDs of the encoder prompt.
125 None if decoder-only.
126 num_cached_tokens: The number of tokens with prefix cache hit.
127 num_cache_creation_tokens: Prompt tokens currently counted as local
128 prefix-cache writes for this request.
129 kv_transfer_params: The params for remote K/V transfer.
130 ec_transfer_params: The params for remote encoder-cache transfer.
131 """
132
133 def __init__(
134 self,
135 request_id: str,
136 prompt: str | None,
137 prompt_token_ids: list[int] | None,
138 prompt_logprobs: PromptLogprobs | None,
139 outputs: list[CompletionOutput],
140 finished: bool,
141 metrics: RequestStateStats | None = None,
142 lora_request: LoRARequest | None = None,
143 encoder_prompt: str | None = None,
144 encoder_prompt_token_ids: list[int] | None = None,
145 num_cached_tokens: int | None = None,
146 num_cache_creation_tokens: int | None = None,
147 *,
148 kv_transfer_params: dict[str, Any] | None = None,
149 ec_transfer_params: dict[str, Any] | None = None,
150 # Forward compatibility, code that uses args added in new release can
151 # still run with older versions of vLLM without breaking.
152 **kwargs: Any,
153 ) -> None:
154 if kwargs:
155 logger.warning_once(
156 "RequestOutput: Ignoring extra arguments: %s", str(kwargs)
157 )
158 self.request_id = request_id
159 self.prompt = prompt
160 self.prompt_token_ids = prompt_token_ids
161 self.prompt_logprobs = prompt_logprobs
162 self.outputs = outputs
163 self.finished = finished
164 self.metrics = metrics
165 self.lora_request = lora_request
166 self.encoder_prompt = encoder_prompt
167 self.encoder_prompt_token_ids = encoder_prompt_token_ids
168 self.num_cached_tokens = num_cached_tokens
169 self.num_cache_creation_tokens = num_cache_creation_tokens
170 self.kv_transfer_params = kv_transfer_params
171 self.ec_transfer_params = ec_transfer_params
172
173 def add(self, next_output: "RequestOutput", aggregate: bool) -> None:
174 """Merge subsequent RequestOutput into this one"""
175
176 self.finished |= next_output.finished
177 self.kv_transfer_params = next_output.kv_transfer_params
178 self.ec_transfer_params = next_output.ec_transfer_params
179
180 for next_completion in next_output.outputs:
181 for i, completion in enumerate(self.outputs):
182 if completion.index == next_completion.index:
183 if aggregate:
184 # Merge outputs with same index
185 completion.text += next_completion.text
186 if not isinstance(completion.token_ids, MutableSequence):
187 completion.token_ids = list(completion.token_ids)
188 completion.token_ids.extend(next_completion.token_ids)
189 if next_completion.logprobs:
190 assert completion.logprobs is not None
191 completion.logprobs.extend(next_completion.logprobs) # type: ignore[arg-type]
192 completion.cumulative_logprob = (
193 next_completion.cumulative_logprob
194 )
195 completion.finish_reason = next_completion.finish_reason
196 completion.stop_reason = next_completion.stop_reason
197 else:
198 # Replace the output with the new one
199 self.outputs[i] = next_completion
200 break
201 else:
202 self.outputs.append(next_completion)
203
204 def __repr__(self) -> str:
205 return (
206 f"RequestOutput(request_id={self.request_id}, "
207 f"prompt={self.prompt!r}, "
208 f"prompt_token_ids={self.prompt_token_ids}, "
209 f"encoder_prompt={self.encoder_prompt!r}, "
210 f"encoder_prompt_token_ids={self.encoder_prompt_token_ids}, "
211 f"prompt_logprobs={self.prompt_logprobs}, "
212 f"outputs={self.outputs}, "
213 f"finished={self.finished}, "
214 f"metrics={self.metrics}, "
215 f"lora_request={self.lora_request}, "
216 f"num_cached_tokens={self.num_cached_tokens}, "
217 f"num_cache_creation_tokens={self.num_cache_creation_tokens})"
218 )

270@dataclass
271class EmbeddingOutput:
272 """The output data of one embedding output of a request.
273
274 Args:
275 embedding: The embedding vector, which is a list of floats.
276 Its length depends on the hidden dimension of the model.
277 """
278
279 embedding: list[float]
280
281 @staticmethod
282 def from_base(pooling_output: PoolingOutput) -> "EmbeddingOutput":
283 pooled_data = pooling_output.data
284 if pooled_data.ndim != 1:
285 raise ValueError("pooled_data should be a 1-D embedding vector")
286
287 return EmbeddingOutput(pooled_data.tolist())
288
289 @property
290 def hidden_size(self) -> int:
291 return len(self.embedding)
292
293 def __repr__(self) -> str:
294 return f"EmbeddingOutput(hidden_size={self.hidden_size})"

354class ScoringOutput:

359 """
360
361 score: float
362
363 @staticmethod
364 def from_base(pooling_output: PoolingOutput) -> "ScoringOutput":
365 # pooling_output shape:
366 # classify task: (num_classes) num_classes == 1
