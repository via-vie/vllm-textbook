29@dataclass(frozen=True)
30class BatchDescriptor:
31 """
32 Batch descriptor for cudagraph dispatching. We should keep the num of
33 items as minimal as possible to properly and uniquely describe the padded
34 batch for cudagraph.
35 """
36
37 num_tokens: int
38 num_reqs: int | None = None
39 """
40 Number of requests in the batch. Can be None for PIECEWISE cudagraphs where
41 the cudagraphs can handle any number of requests.
42 """
43 uniform: bool = False
44 """
45 True if all the requests in the batch have the same number of tokens.
46 """
47 has_lora: bool = False
48 """
49 Whether this batch has active LoRA adapters.
50 """
51 num_active_loras: int = 0
52 """
53 Number of distinct active LoRA adapters in this batch.
54 When cudagraph_specialize_lora_count is enabled, separate CUDA graphs
55 are captured for each num_active_loras value. This allows kernels
56 (like fused_moe_lora) whose grid size depends on num_active_loras
57 to be properly captured.
58 """

61def _compute_sp_num_tokens(
62 num_tokens_across_dp_cpu: torch.Tensor, sequence_parallel_size: int
63) -> list[int]:
64 sp_tokens = (
65 num_tokens_across_dp_cpu + sequence_parallel_size - 1
66 ) // sequence_parallel_size
67
68 sp_tokens = sp_tokens.repeat_interleave(sequence_parallel_size)
69 return sp_tokens.tolist()

72@dataclass
73class DPMetadata:
74 num_tokens_across_dp_cpu: torch.Tensor
75
76 # NOTE: local_sizes should only be set by the chunked_sizes context manager
77 local_sizes: list[int] | None = None
78
79 @staticmethod
80 def make(
81 parallel_config: ParallelConfig,
82 num_tokens: int,
83 num_tokens_across_dp_cpu: torch.Tensor,
84 ) -> "DPMetadata":
85 assert num_tokens_across_dp_cpu is not None
86 assert (
87 parallel_config.data_parallel_size > 1
88 or parallel_config.use_sequence_parallel_moe
89 )
90 assert parallel_config.is_moe_model is not False
91 dp_rank = parallel_config.data_parallel_rank
92 batchsize = num_tokens
93
94 # If num_tokens_across_dp is None, it will be computed by all_reduce
95 # Otherwise, num_tokens_across_dp[dp_rank] should be equal to batchsize
96 assert num_tokens_across_dp_cpu[dp_rank] == batchsize, (
97 f"{num_tokens_across_dp_cpu[dp_rank]} {batchsize}"
98 )
99 return DPMetadata(num_tokens_across_dp_cpu)

101 @contextmanager
102 def sp_local_sizes(self, sequence_parallel_size: int):
103 """
104 Context manager for setting self.local_sizes. Same as self.chunked_sizes
105 but without any chunking.
106 """
107 self.local_sizes = _compute_sp_num_tokens(
108 self.num_tokens_across_dp_cpu, sequence_parallel_size
109 )
110 try:
111 yield self.local_sizes
112 finally:
113 self.local_sizes = None
114
115 def get_chunk_sizes_across_dp_rank(self) -> list[int] | None:
116 assert self.local_sizes is not None
117 return self.local_sizes
118
119 # Get the cumulative tokens across sequence parallel ranks.
120 # In this case the input to the MoEs will be distributed w.r.t both
121 # DP and TP rank.
122 # When sp_size==1, this is just the cumulative num tokens across DP.
123 def cu_tokens_across_sp(self, sp_size: int) -> torch.Tensor:
124 num_tokens_across_sp_cpu = (
125 self.num_tokens_across_dp_cpu - 1 + sp_size
126 ) // sp_size
127 num_tokens_across_sp_cpu = num_tokens_across_sp_cpu.repeat_interleave(sp_size)
128 return torch.cumsum(num_tokens_across_sp_cpu, dim=0)

131@dataclass
132class ForwardContext:
133 # copy from vllm_config.compilation_config.static_forward_context
134 no_compile_layers: dict[str, Any]
135 attn_metadata: dict[str, AttentionMetadata] | list[dict[str, AttentionMetadata]]
136 slot_mapping: dict[str, torch.Tensor] | list[dict[str, torch.Tensor]]
137 """
138 Type Dict[str, AttentionMetadata] for v1, map from layer_name of each
139 attention layer to its attention metadata
140 Type List[Dict[str, AttentionMetadata]] for DBO. List of size two, one
141 for each microbatch.
142 Set dynamically for each forward pass
143 """
144 # set dynamically for each forward pass
145 dp_metadata: DPMetadata | None = None
146 # determine the cudagraph style at runtime to be FULL, PIECEWISE, or NONE.
147 # by default NONE, no cudagraph is used.
148 cudagraph_runtime_mode: CUDAGraphMode = CUDAGraphMode.NONE
149 batch_descriptor: BatchDescriptor | None = None
150
151 ubatch_slices: UBatchSlices | None = None
152
153 # Boolean mask over the token axis: True for padding rows that are not real
154 # tokens. Consumers can use it to skip work for padded tokens. None when
155 # the producer does not set it.
156 is_padding: torch.Tensor | None = None
157
158 # If True, bypass the compiled model call, e.g. by using .forward() directly
159 skip_compiled: bool = False
160
161 # For torch.compile cold start times, we need to avoid hard-coding
162 # any strings into the graph. Right now, the vllm.moe_forward
163 # and vllm.moe_forward_shared custom operators hard-code strings into
164 # the graph.
165 #
166 # The workaround is to store a list of the strings that each of those
167 # custom ops needs in the ForwardContext (all_moe_layers)
168 # as well as a counter (moe_layer_index).
169 # The ForwardContext object is alive for the duration of the forward pass.
170 # When the custom op needs a layer string, get the next string
171 # from all_moe_layers and increment the counter.
172 #
173 # This assumes that the custom operators will always be executed in
174 # order and that torch.compile will not try to reorder these
175 # operations with respect to each other.
176 #
177 # TODO(https://github.com/vllm-project/vllm/issues/31985):
178 # There are longer-term solutions, like unwrapping the moe custom operator,
179 # that aren't ready yet.
180 # We could also treat the string as a "symbolic input" to the graph but
181 # the PyTorch-side bits for that aren't ready yet either.
182 #
183 # If this value is None (like in some tests), then we end up baking the string
184 # into the graph. Otherwise, the moe custom ops will pop a string from this list.
185 all_moe_layers: list[str] | None = None
186 moe_layer_index: int = 0
187
188 additional_kwargs: dict[str, Any] = field(default_factory=dict)
189
190 def __post_init__(self):
191 assert self.cudagraph_runtime_mode.is_valid_runtime_mode(), (
192 f"Invalid cudagraph runtime mode: {self.cudagraph_runtime_mode}"
193 )

247 skip_compiled=skip_compiled,

253@contextmanager
254def override_forward_context(forward_context: ForwardContext | None):
255 """A context manager that overrides the current forward context.
256 This is used to override the forward context for a specific
257 forward pass.
258 """
259 global _forward_context
260 prev_context = _forward_context
261 _forward_context = forward_context
262 try:
263 yield
264 finally:
265 _forward_context = prev_context

268@contextmanager
269def set_forward_context(
270 attn_metadata: Any,
271 vllm_config: VllmConfig,
272 num_tokens: int | None = None,
273 num_tokens_across_dp: torch.Tensor | None = None,
274 cudagraph_runtime_mode: CUDAGraphMode = CUDAGraphMode.NONE,
275 batch_descriptor: BatchDescriptor | None = None,
276 ubatch_slices: UBatchSlices | None = None,
277 slot_mapping: dict[str, torch.Tensor] | list[dict[str, torch.Tensor]] | None = None,
278 skip_compiled: bool = False,
279 is_padding: torch.Tensor | None = None,
280):
281 """A context manager that stores the current forward context,
282 can be attention metadata, etc.
283 Here we can inject common logic for every model forward pass.
284 """
285 global forward_start_time
286 need_to_track_batchsize = track_batchsize and attn_metadata is not None
287 if need_to_track_batchsize:
288 forward_start_time = time.perf_counter()
289
290 dp_metadata: DPMetadata | None = None
291 if (
292 (
293 vllm_config.parallel_config.data_parallel_size > 1
294 or vllm_config.parallel_config.use_sequence_parallel_moe
295 )
296 and vllm_config.parallel_config.is_moe_model is not False
297 and (attn_metadata is not None or num_tokens is not None)
298 ):
299 # If num_tokens_across_dp hasn't already been initialized, then
300 # initialize it here. Both DP padding and Microbatching will be
301 # disabled.
302 if (
303 num_tokens_across_dp is None
304 and vllm_config.parallel_config.data_parallel_size > 1
305 ):
306 assert ubatch_slices is None
307 assert num_tokens is not None
308 _, num_tokens_across_dp, _ = coordinate_batch_across_dp(
309 num_tokens_unpadded=num_tokens,
310 parallel_config=vllm_config.parallel_config,
311 allow_microbatching=False,
312 )
313 assert num_tokens_across_dp is not None
314 elif num_tokens_across_dp is None:
315 assert num_tokens is not None
316 num_tokens_across_dp = torch.tensor([num_tokens], dtype=torch.int32)
317 dp_metadata = DPMetadata.make(
318 vllm_config.parallel_config, num_tokens or 0, num_tokens_across_dp
319 )
320
321 # Convenience: if cudagraph is used and num_tokens is given, we can just
322 # create a batch descriptor here if not given (there's no harm since if it
323 # doesn't match in the wrapper it'll fall through).
324 if cudagraph_runtime_mode != CUDAGraphMode.NONE and num_tokens is not None:
325 batch_descriptor = batch_descriptor or BatchDescriptor(num_tokens=num_tokens)
326
327 additional_kwargs = current_platform.set_additional_forward_context(
328 attn_metadata=attn_metadata,
329 vllm_config=vllm_config,
330 dp_metadata=dp_metadata,
331 num_tokens=num_tokens,
332 num_tokens_across_dp=num_tokens_across_dp,
333 cudagraph_runtime_mode=cudagraph_runtime_mode,
334 batch_descriptor=batch_descriptor,
335 ubatch_slices=ubatch_slices,
336 )
337
338 forward_context = create_forward_context(
339 attn_metadata,
340 vllm_config,
341 dp_metadata,
342 cudagraph_runtime_mode,
343 batch_descriptor,
344 ubatch_slices,
345 slot_mapping,
346 additional_kwargs,
347 skip_compiled,
348 is_padding=is_padding,
349 )
350
351 try:
352 with override_forward_context(forward_context):
353 yield
354 finally:
355 global last_logging_time, batchsize_logging_interval
356 if need_to_track_batchsize:
357 batchsize = num_tokens
358 # we use synchronous scheduling right now,
359 # adding a sync point here should not affect
360 # scheduling of the next batch
361 synchronize = current_platform.synchronize
362 if synchronize is not None:
363 synchronize()
364 now = time.perf_counter()
365 # time measurement is in milliseconds
366 batchsize_forward_time[batchsize].append((now - forward_start_time) * 1000)
367 if now - last_logging_time > batchsize_logging_interval:
368 last_logging_time = now
369 forward_stats = []
370 for bs, times in batchsize_forward_time.items():
371 if len(times) <= 1:
372 # can be cudagraph / profiling run
373 continue
374 medium = torch.quantile(torch.tensor(times), q=0.5).item()
375 medium = round(medium, 2)
376 forward_stats.append((bs, len(times), medium))
377 forward_stats.sort(key=lambda x: x[1], reverse=True)
378 if forward_stats:
379 logger.info(
380 (
381 "Batchsize forward time stats "
382 "(batchsize, count, median_time(ms)): %s"
383 ),
384 forward_stats,
385 )
