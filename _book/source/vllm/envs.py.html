389def env_with_choices(
390 env_name: str,
391 default: str | None,
392 choices: list[str] | Callable[[], list[str]],
393 case_sensitive: bool = True,
394) -> Callable[[], str | None]:
395 """
396 Create a lambda that validates environment variable against allowed choices
397
398 Args:
399 env_name: Name of the environment variable
400 default: Default value if not set (can be None)
401 choices: List of valid string options or callable that returns list
402 case_sensitive: Whether validation should be case sensitive
403
404 Returns:
405 Lambda function for environment_variables dict
406 """
407
408 def _get_validated_env() -> str | None:
409 value = os.getenv(env_name)
410 if value is None:
411 return default
412
413 # Resolve choices if it's a callable (for lazy loading)
414 actual_choices = choices() if callable(choices) else choices
415
416 if not case_sensitive:
417 check_value = value.lower()
418 check_choices = [choice.lower() for choice in actual_choices]
419 else:
420 check_value = value
421 check_choices = actual_choices
422
423 if check_value not in check_choices:
424 raise ValueError(
425 f"Invalid value '{value}' for {env_name}. "
426 f"Valid options: {actual_choices}."
427 )
428
429 return value
430
431 return _get_validated_env

436 default: list[str],
437 choices: list[str] | Callable[[], list[str]],
438 case_sensitive: bool = True,
439) -> Callable[[], list[str]]:
440 """
441 Create a lambda that validates environment variable
442 containing comma-separated values against allowed choices
443
444 Args:
445 env_name: Name of the environment variable
446 default: Default list of values if not set
447 choices: List of valid string options or callable that returns list
448 case_sensitive: Whether validation should be case sensitive
449
450 Returns:
451 Lambda function for environment_variables
452 dict that returns list of strings
453 """
454
455 def _get_validated_env_list() -> list[str]:
456 value = os.getenv(env_name)
457 if value is None:
458 return default
459
460 # Split comma-separated values and strip whitespace
461 values = [v.strip() for v in value.split(",") if v.strip()]
462
463 if not values:
464 return default
465
466 # Resolve choices if it's a callable (for lazy loading)
467 actual_choices = choices() if callable(choices) else choices
468
469 # Validate each value
470 for val in values:
471 if not case_sensitive:
472 check_value = val.lower()
473 check_choices = [choice.lower() for choice in actual_choices]
474 else:
475 check_value = val
476 check_choices = actual_choices
477
478 if check_value not in check_choices:
479 raise ValueError(
480 f"Invalid value '{val}' in {env_name}. "
481 f"Valid options: {actual_choices}."
482 )
483
484 return values
485
486 return _get_validated_env_list
