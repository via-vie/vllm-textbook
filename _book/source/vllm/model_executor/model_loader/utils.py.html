37@instrument(span_name="Initialize model")
38def initialize_model(
39 vllm_config: VllmConfig,
40 *,
41 prefix: str = "",
42 model_class: type[nn.Module] | None = None,
43 model_config: ModelConfig | None = None,
44) -> nn.Module:
45 """Initialize a model with the given configurations."""
46 if model_config is None:
47 model_config = vllm_config.model_config
48 if model_class is None:
49 model_class, _ = get_model_architecture(model_config)
50
51 if vllm_config.quant_config is not None:
52 configure_quant_config(vllm_config.quant_config, model_class)
53
54 signatures = inspect.signature(model_class.__init__)
55 all_params = [param.name for param in signatures.parameters.values()]
56 if "vllm_config" in all_params and "prefix" in all_params:
57 # new-style model class
58 with set_current_vllm_config(vllm_config, check_compile=True, prefix=prefix):
59 model = model_class(vllm_config=vllm_config, prefix=prefix)
60 record_metadata_for_reloading(model)
61 return model
62
63 msg = (
64 "vLLM model class should accept `vllm_config` and `prefix` as "
65 "input arguments. Possibly you have an old-style model class"
66 " registered from out of tree and it is used for new vLLM version. "
67 "Check https://docs.vllm.ai/en/latest/design/arch_overview.html "
68 "for the design and update the model class accordingly."
69 )
70 warnings.warn(msg, DeprecationWarning, stacklevel=2)
71
72 logger.warning(
73 "Trying to guess the arguments for old-style model class %s",
74 model_class,
75 )
76 # try to be compatible with old-style model class
77 kwargs: dict[str, Any] = {}
78 if "prefix" in all_params:
79 kwargs["prefix"] = prefix
80 if "config" in all_params:
81 kwargs["config"] = model_config.hf_config
82 if "cache_config" in all_params:
83 kwargs["cache_config"] = vllm_config.cache_config
84 if "quant_config" in all_params:
85 kwargs["quant_config"] = vllm_config.quant_config
86 if "lora_config" in all_params:
87 kwargs["lora_config"] = vllm_config.lora_config
88 if "scheduler_config" in all_params:
89 kwargs["scheduler_config"] = vllm_config.scheduler_config
90 with set_current_vllm_config(vllm_config, check_compile=True, prefix=prefix):
91 model = model_class(**kwargs)
92 record_metadata_for_reloading(model)
93
94 return model

97def process_weights_after_loading(
98 model: nn.Module, model_config: ModelConfig, target_device: torch.device
99) -> None:
100 """Post-process loaded weights into runtime format.
101
102 Under ``weights_already_processed`` (weight cache IPC loader), quant
103 methods skip tensor transforms and must declare
104 ``supports_pre_processed_weights``, otherwise this raises ``RuntimeError``.
105 """
106 # Reclaim memory when an explicit lm_head has been
107 # loaded, but it is identical to the input embeddings.
108 maybe_retie_word_embeddings(model, model_config)
109
110 for name, module in model.named_modules():
111 quant_method = getattr(module, "quant_method", None)
112 if isinstance(quant_method, QuantizeMethodBase):
113 if (
114 is_weights_pre_processed()
115 and not quant_method.supports_pre_processed_weights
116 ):
117 raise RuntimeError(
118 f"layer {name or '<root>'}: {type(quant_method).__name__} "
119 "does not support pre-processed weights"
120 )
121 # When quant methods need to process weights after loading
122 # (for repacking, quantizing, etc), they typically expect parameters
123 # to be on the global target device. This scope is for the
124 # case where cpu offloading is used, where we will move the
125 # parameters onto device for processing and back off after.
126 # Methods that can process weights in place (e.g. PLE scale
127 # validation) set requires_device_loading=False to skip this move.
128 loading_context = (
129 device_loading_context(module, target_device)
130 if quant_method.requires_device_loading
131 else nullcontext()
132 )
133 with loading_context:
134 quant_method.process_weights_after_loading(module)
135 # process_weights_after_loading may swap in freshly-created
136 # Parameters (e.g. FP8 requantization), which are stamped with the
137 # global rank in BasevLLMParameter.__init__. Re-reconcile their TP
138 # state to the layer so a later weight reload / RL weight-refit
139 # narrows replicated (disable_tp) weights at the correct offset.
140 if hasattr(module, "update_param_tp_status"):
141 module.update_param_tp_status()
142 # Repacking transients above can leave large amounts of memory in
143 # the caching allocator, which starves the OS on UMA devices.
144 release_device_memory_under_pressure(target_device)
145
146 # Initialize post-load attention weights for any attention layer and MM
147 # encoder. NOTE: Happens after other modules so we can easily decompress
148 # weights.
149 for _, module in model.named_modules():
150 if is_deferred_attention_layer(module):
151 # TODO(lucas): see if there is a way to unify the signatures
152 # of process_weights_after_loading
153 with device_loading_context(module, target_device):
154 module.process_weights_after_loading(model_config.dtype)
155
156 # Model-level post-load hook, after the per-layer quant finalize.
157 if hasattr(model, "process_weights_after_loading"):
158 model.process_weights_after_loading()
159
160 # Needed for torchao model reloading via model.reload_weights
161 # @kylesayrs @jerryzh168 this can be removed if callers move to `reload_weights`
162 if model_config.quantization == "torchao":
163 set_torchao_reload_attrs(model, model_config)
164

166@contextmanager
167def device_loading_context(module: torch.nn.Module, target_device: torch.device):
168 if target_device.type == "cpu":
169 # If target is CPU, no need to move anything
170 yield module
171 return
172
173 cpu_params: set[str] = set()
174 uva_offloaded_parameters: list[str] = []
175
176 # Store which parameters are on CPU and move them to the GPU
177 for name, p in module.named_parameters():
178 if p.device.type == "cpu":
179 cpu_params.add(name)
180 p.data = p.data.to(target_device)
181 if getattr(p, "_vllm_is_uva_offloaded", False):
182 uva_offloaded_parameters.append(name)
183 # Parameters already on target device are not touched
184
185 try:
186 yield module
187
188 finally:
189 use_pin_memory = (
190 is_pin_memory_available()
191 and not envs.VLLM_WEIGHT_OFFLOADING_DISABLE_PIN_MEMORY
192 )
193 # Restore the CPU-resident parameters, ignoring new parameters.
194 for name, p in module.named_parameters():
195 if name in cpu_params:
196 p.data = torch.empty_like(
197 p.data, device="cpu", pin_memory=use_pin_memory
198 ).copy_(p.data)
199
200 # parameter is UVA offloaded, but was replaced with a new device tensor
201 # re-offload it to CPU using UVA
202 if name in uva_offloaded_parameters and not getattr(
203 p, "_vllm_is_uva_offloaded", False
204 ):
205 cpu_data = torch.empty_like(
206 p.data, device="cpu", pin_memory=use_pin_memory
207 ).copy_(p.data)
208 p.data = get_accelerator_view_from_cpu_tensor(cpu_data)
209 p._vllm_is_uva_offloaded = True

212_MODEL_ARCH_BY_HASH = dict[int, tuple[type[nn.Module], str]]()
213"""Caches the outputs of `_get_model_architecture`."""

216def _get_model_architecture(model_config: ModelConfig) -> tuple[type[nn.Module], str]:
217 from vllm.model_executor.models.adapters import as_embedding_model, as_seq_cls_model
218
219 architectures = getattr(model_config.hf_config, "architectures", None) or []
220
221 model_cls, arch = model_config.registry.resolve_model_cls(
222 architectures,
223 model_config=model_config,
224 )
225
226 if arch == model_config._get_transformers_backend_cls():
227 assert model_config.model_impl != "vllm"
228 if model_config.model_impl == "auto":
229 logger.warning_once(
230 "%s has no vLLM implementation, falling back to Transformers "
231 "implementation. Some features may not be supported and "
232 "performance may not be optimal.",
233 arch,
234 )
235
236 convert_type = model_config.convert_type
237 if convert_type == "none":
238 pass
239 elif convert_type == "embed":
240 logger.debug_once("Converting to embedding model.")
241 model_cls = as_embedding_model(model_cls)
242 elif convert_type == "classify":
243 logger.debug_once("Converting to sequence classification model.")
244 model_cls = as_seq_cls_model(model_cls)
245 else:
246 assert_never(convert_type)
247
248 return model_cls, arch

251def get_model_architecture(model_config: ModelConfig) -> tuple[type[nn.Module], str]:
252 key = hash(
253 (
254 model_config.model,
255 model_config.convert_type,
256 model_config.runner_type,
257 model_config.trust_remote_code,
258 model_config.model_impl,
259 tuple(getattr(model_config.hf_config, "architectures", None) or []),
260 )
261 )
262 if key in _MODEL_ARCH_BY_HASH:
263 return _MODEL_ARCH_BY_HASH[key]
264
265 model_cls_and_arch = _get_model_architecture(model_config)
266 _MODEL_ARCH_BY_HASH[key] = model_cls_and_arch
267 return model_cls_and_arch

270def get_model_cls(model_config: ModelConfig) -> type[nn.Module]:
271 return get_model_architecture(model_config)[0]

274def get_architecture_class_name(model_config: ModelConfig) -> str:
275 return get_model_architecture(model_config)[1]

278def configure_quant_config(
279 quant_config: QuantizationConfig, model_class: type[nn.Module]
280):
281 """
282 Pass packed_modules_mapping by reference to quant_config so that
283 quant_config can properly match fused modules
284
285 Note that model attributes are passed by reference to quant_config,
286 enabling them to be updated by model_class.__new__ (ex. chatglm, qwen)
287
288 Once the `SupportsQuant` mixin has been added to all models, this
289 function can be removed
290 """
291 if not issubclass(model_class, SupportsQuant):
292 hf_to_vllm_mapper = getattr(model_class, "hf_to_vllm_mapper", None)
293 packed_mapping = getattr(model_class, "packed_modules_mapping", None)
294
295 # pass mappings by reference to quant_config
296 if hf_to_vllm_mapper is not None:
297 quant_config.apply_vllm_mapper(hf_to_vllm_mapper.get_rename_mapper())
298 if packed_mapping is not None:
299 quant_config.packed_modules_mapping = packed_mapping
