69def register_model_loader(load_format: str):
70 """Register a customized vllm model loader.
71
72 When a load format is not supported by vllm, you can register a customized
73 model loader to support it.
74
75 Args:
76 load_format (str): The model loader format name.
77
78 Examples:
79 >>> from vllm.config.load import LoadConfig
80 >>> from vllm.model_executor.model_loader import (
81 ... get_model_loader,
82 ... register_model_loader,
83 ... )
84 >>> from vllm.model_executor.model_loader.base_loader import BaseModelLoader
85 >>>
86 >>> @register_model_loader("my_loader")
87 ... class MyModelLoader(BaseModelLoader):
88 ... def download_model(self):
89 ... pass
90 ...
91 ... def load_weights(self):
92 ... pass
93 >>>
94 >>> load_config = LoadConfig(load_format="my_loader")
95 >>> type(get_model_loader(load_config))
96 <class 'MyModelLoader'>
97 """ # noqa: E501
98
99 def _wrapper(model_loader_cls):
100 if load_format in _LOAD_FORMAT_TO_MODEL_LOADER:
101 logger.warning(
102 "Load format `%s` is already registered, and will be "
103 "overwritten by the new loader class `%s`.",
104 load_format,
105 model_loader_cls,
106 )
107 if not issubclass(model_loader_cls, BaseModelLoader):
108 raise ValueError(
109 "The model loader must be a subclass of `BaseModelLoader`."
110 )
111 _LOAD_FORMAT_TO_MODEL_LOADER[load_format] = model_loader_cls
112 logger.info(
113 "Registered model loader `%s` with load format `%s`",
114 model_loader_cls,
115 load_format,
116 )
117 return model_loader_cls
118
119 return _wrapper

122def get_model_loader(load_config: LoadConfig) -> BaseModelLoader:
123 """Get a model loader based on the load format."""
124 load_format = load_config.load_format
125 if load_format not in _LOAD_FORMAT_TO_MODEL_LOADER:
126 raise ValueError(f"Load format `{load_format}` is not supported")
127 return _LOAD_FORMAT_TO_MODEL_LOADER[load_format](load_config)

130def get_model(
131 *,
132 vllm_config: VllmConfig,
133 model_config: ModelConfig | None = None,
134 prefix: str = "",
135 load_config: LoadConfig | None = None,
136) -> nn.Module:
137 loader = get_model_loader(load_config or vllm_config.load_config)
138 if model_config is None:
139 model_config = vllm_config.model_config
140 return loader.load_model(
141 vllm_config=vllm_config, model_config=model_config, prefix=prefix
142 )
