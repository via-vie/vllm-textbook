74 def __init__(self, load_config: LoadConfig):
75 super().__init__(load_config)
76 self.local_expert_ids: set[int] | None = None

78 extra_config = load_config.model_loader_extra_config
79 if not isinstance(extra_config, dict):
80 raise ValueError(
81 f"model_loader_extra_config must be a dict for load format "
82 f"{load_config.load_format}, got {type(extra_config).__name__}"
83 )
84 allowed_keys = {
85 "enable_multithread_load",
86 "num_threads",
87 "enable_weights_track",
88 }
89 unexpected_keys = set(extra_config.keys()) - allowed_keys
90
91 if unexpected_keys:
92 raise ValueError(
93 f"Unexpected extra config keys for load format "
94 f"{load_config.load_format}: "
95 f"{unexpected_keys}"
96 )
97
98 enable_multithread_load = extra_config.get("enable_multithread_load", False)
99 if not isinstance(enable_multithread_load, bool):
100 raise ValueError(
101 f"enable_multithread_load must be a bool, got "
102 f"{type(enable_multithread_load).__name__}"
103 )
104 num_threads = extra_config.get("num_threads")
105 if num_threads is not None and not (
106 isinstance(num_threads, int) and num_threads > 0
107 ):
108 raise ValueError(
109 f"num_threads must be a positive integer, got {num_threads!r}"
110 )
111
112 self.enable_weights_track: bool | None = extra_config.get(
113 "enable_weights_track", None
114 )
115
116 # The multi-thread loader ignores safetensors_load_strategy, so reject
117 # the combination instead of silently dropping the requested strategy.
118 if extra_config.get("enable_multithread_load") and (

121 raise ValueError(
122 "enable_multithread_load does not support "
123 "safetensors_load_strategy="
124 f"{load_config.safetensors_load_strategy!r}; the multi-thread "
125 "loader only implements the default lazy strategy."

128 def _prepare_weights(
129 self,
130 model_name_or_path: str,
131 subfolder: str | None,
132 revision: str | None,
133 fall_back_to_pt: bool,
134 allow_patterns_overrides: list[str] | None,
135 ) -> tuple[str, list[str], bool]:
136 """Prepare weights for the model.
137
138 If the model is not local, it will be downloaded."""
139 model_name_or_path = (
140 maybe_download_from_modelscope(model_name_or_path, revision)
141 or model_name_or_path
142 )
143
144 is_local = os.path.isdir(model_name_or_path)
145 load_format = self.load_config.load_format
146 use_safetensors = False
147 index_file = SAFE_WEIGHTS_INDEX_NAME
148
149 # First check for 'auto' format that mistral files format are present.
150 # This is to load mistral models with official format by default.
151 if load_format == "auto":
152 load_format = (
153 "mistral"
154 if len(
155 list_filtered_repo_files(
156 model_name_or_path=model_name_or_path,
157 allow_patterns=["consolidated*.safetensors"],
158 revision=revision,
159 )
160 )
161 > 0
162 else "hf"
163 )
164
165 # Some quantized models use .pt files for storing the weights.
166 if load_format == "hf":
167 allow_patterns = ["*.safetensors", "*.bin"]
168 elif (
169 load_format == "safetensors"
170 or load_format == "fastsafetensors"
171 or load_format == "instanttensor"
172 ):
173 use_safetensors = True
174 allow_patterns = ["*.safetensors"]
175 elif load_format == "mistral":
176 use_safetensors = True
177 allow_patterns = ["consolidated*.safetensors"]
178 index_file = "consolidated.safetensors.index.json"
179 elif load_format == "pt":
180 allow_patterns = ["*.pt"]
181 elif load_format == "npcache":
182 allow_patterns = ["*.bin"]
183 else:
184 raise ValueError(f"Unknown load_format: {load_format}")
185
186 # Don't fall back to .pt for explicit safetensors formats; otherwise a
187 # .pt file is matched and later opened as safetensors.
188 if fall_back_to_pt and not use_safetensors:
189 allow_patterns += ["*.pt"]
190
191 if allow_patterns_overrides is not None:
192 allow_patterns = allow_patterns_overrides
193
194 if not is_local:
195 hf_folder = download_weights_from_hf(
196 model_name_or_path,
197 self.load_config.download_dir,
198 allow_patterns,
199 revision,
200 subfolder=subfolder,
201 ignore_patterns=self.load_config.ignore_patterns,
202 )
203 else:
204 hf_folder = model_name_or_path
205
206 if subfolder is not None:
207 hf_folder = os.path.join(hf_folder, subfolder)
208
209 hf_weights_files: list[str] = []
210 for pattern in allow_patterns:
211 hf_weights_files += glob.glob(os.path.join(hf_folder, pattern))
212 if len(hf_weights_files) > 0:
213 if pattern.endswith(".safetensors"):
214 use_safetensors = True
215 break
216
217 if use_safetensors:
218 # For models like Mistral-7B-Instruct-v0.3
219 # there are both sharded safetensors files and a consolidated
220 # safetensors file. Using both breaks.
221 # Here, we download the `model.safetensors.index.json` and filter
222 # any files not found in the index.
223 if not is_local and len(hf_weights_files) > 1:
224 download_safetensors_index_file_from_hf(
225 model_name_or_path,
226 index_file,
227 cache_dir=self.load_config.download_dir,
228 subfolder=subfolder,
229 revision=revision,
230 )
231 hf_weights_files = filter_duplicate_safetensors_files(
232 hf_weights_files, hf_folder, index_file
233 )
234 else:
235 hf_weights_files = filter_files_not_needed_for_inference(hf_weights_files)
236
237 if len(hf_weights_files) == 0:
238 raise RuntimeError(
239 f"Cannot find any model weights with `{model_name_or_path}`"
240 )
241
242 return hf_folder, hf_weights_files, use_safetensors

244 def _get_weights_iterator(
245 self, source: "Source"
246 ) -> Generator[tuple[str, torch.Tensor], None, None]:
247 """Get an iterator for the model weights based on the load format."""
248 extra_config = self.load_config.model_loader_extra_config
249 hf_folder, hf_weights_files, use_safetensors = self._prepare_weights(
250 source.model_or_path,
251 source.subfolder,
252 source.revision,
253 source.fall_back_to_pt,
254 source.allow_patterns_overrides,
255 )
256 if self.load_config.load_format == "npcache":
257 # Currently np_cache only support *.bin checkpoints
258 assert use_safetensors is False
259 weights_iterator = np_cache_weights_iterator(
260 source.model_or_path,
261 self.load_config.download_dir,
262 hf_folder,
263 hf_weights_files,
264 self.load_config.use_tqdm_on_load,
265 )
266 elif use_safetensors:
267 if self.load_config.load_format == "fastsafetensors":
268 weights_iterator = fastsafetensors_weights_iterator(
269 hf_weights_files,
270 self.load_config.use_tqdm_on_load,
271 )
272 elif self.load_config.load_format == "instanttensor":
273 weights_iterator = instanttensor_weights_iterator(
274 hf_weights_files,
275 self.load_config.use_tqdm_on_load,
276 )
277 else:
278 if extra_config.get("enable_multithread_load"):
279 weights_iterator = multi_thread_safetensors_weights_iterator(
280 hf_weights_files,
281 self.load_config.use_tqdm_on_load,
282 max_workers=extra_config.get(
283 "num_threads", self.DEFAULT_NUM_THREADS
284 ),
285 )
286 else:
287 weights_iterator = safetensors_weights_iterator(
288 hf_weights_files,
289 self.load_config.use_tqdm_on_load,
290 self.load_config.safetensors_load_strategy,
291 local_expert_ids=self.local_expert_ids,
292 safetensors_prefetch_num_threads=(
293 self.load_config.safetensors_prefetch_num_threads
294 ),
295 safetensors_prefetch_block_size=(
296 self.load_config.safetensors_prefetch_block_size
297 ),
298 )
299 else:
300 if extra_config.get("enable_multithread_load"):
301 weights_iterator = multi_thread_pt_weights_iterator(
302 hf_weights_files,
303 self.load_config.use_tqdm_on_load,
304 self.load_config.pt_load_map_location,
305 max_workers=extra_config.get(
306 "num_threads", self.DEFAULT_NUM_THREADS
307 ),
308 )
309 else:
310 weights_iterator = pt_weights_iterator(
311 hf_weights_files,
312 self.load_config.use_tqdm_on_load,
313 self.load_config.pt_load_map_location,
314 )
315
316 if self.counter_before_loading_weights == 0.0:
317 self.counter_before_loading_weights = time.perf_counter()
318 # Apply the prefix.
319 return ((source.prefix + name, tensor) for (name, tensor) in weights_iterator)

321 def get_all_weights(
322 self,
323 model_config: ModelConfig,
324 model: nn.Module,
325 ) -> Generator[tuple[str, torch.Tensor], None, None]:
326 primary_weights = DefaultModelLoader.Source(
327 model_config.model,
328 model_config.revision,
329 prefix="",
330 fall_back_to_pt=getattr(model, "fall_back_to_pt_during_load", True),
331 allow_patterns_overrides=getattr(model, "allow_patterns_overrides", None),
332 )
333 yield from self._get_weights_iterator(primary_weights)
334
335 secondary_weights = cast(
336 Iterable[DefaultModelLoader.Source],
337 getattr(model, "secondary_weights", ()),
338 )
339 for source in secondary_weights:
340 yield from self._get_weights_iterator(source)

351 def _init_ep_weight_filter(self, model_config: ModelConfig) -> None:
352 """Compute local expert ids for EP weight filtering.
353
354 When expert parallelism is active, each rank only needs a subset of
355 expert weights. By computing the set upfront we can skip non-local
356 expert tensors *before* reading them from disk.
357 """
358 from vllm.config import get_current_vllm_config
359
360 vllm_config = get_current_vllm_config()
361 parallel_config = vllm_config.parallel_config
362
363 if not (
364 model_config.is_moe
365 and parallel_config.enable_expert_parallel
366 and parallel_config.enable_ep_weight_filter
367 ):
368 return
369
370 # When EPLB is enabled, redundant physical expert slots may map to
371 # logical experts that belong to other ranks in the default partition.
372 # The weight loader needs to see ALL logical expert weights so it can
373 # populate these redundant slots. Skip the filter entirely.
374 if parallel_config.enable_eplb:
375 return
376
377 num_experts = model_config.get_num_experts()
378 if num_experts <= 0:
379 return
380
381 # EP size/rank computation mirrors FusedMoEParallelConfig.make():
382 # ep_size = dp_size * pcp_size * tp_size (flattened)
383 # ep_rank = dp_rank * pcp_size * tp_size + pcp_rank * tp_size + tp_rank
384 from vllm.distributed import (
385 get_dp_group,
386 get_pcp_group,
387 get_tensor_model_parallel_rank,
388 )
389
390 dp_size = parallel_config.data_parallel_size
391 tp_size = parallel_config.tensor_parallel_size
392 pcp_size = parallel_config.prefill_context_parallel_size
393 dp_rank = get_dp_group().rank_in_group if dp_size > 1 else 0
394 tp_rank = get_tensor_model_parallel_rank() if tp_size > 1 else 0
395 pcp_rank = get_pcp_group().rank_in_group if pcp_size > 1 else 0
396 ep_size = dp_size * pcp_size * tp_size
397 ep_rank = dp_rank * pcp_size * tp_size + pcp_rank * tp_size + tp_rank
398
399 self.local_expert_ids = compute_local_expert_ids(
400 num_experts,
401 ep_size,
402 ep_rank,
403 placement=parallel_config.expert_placement_strategy,
404 )
405 if self.local_expert_ids is not None:
406 logger.info_once(
407 "EP weight filter: ep_size=%d, ep_rank=%d, loading %d/%d experts",
408 ep_size,
409 ep_rank,
410 len(self.local_expert_ids),
411 num_experts,
412 )

414 @instrument(span_name="Load weights")
415 def load_weights(self, model: nn.Module, model_config: ModelConfig) -> None:
416 if model_config.quantization == "torchao":
417 quant_config = get_quant_config(model_config, self.load_config)
418 if (
419 hasattr(quant_config, "is_checkpoint_torchao_serialized")
420 and quant_config.is_checkpoint_torchao_serialized
421 and torchao_version_at_least("0.15.0")
422 ):
423 self.load_config.safetensors_load_strategy = "torchao"
424
425 self._init_ep_weight_filter(model_config)
426
427 loaded_weights = model.load_weights(self.get_all_weights(model_config, model))
428
429 self.counter_after_loading_weights = time.perf_counter()
430 logger.info_once(
431 "Loading weights took %.2f seconds",
432 self.counter_after_loading_weights - self.counter_before_loading_weights,
433 )
434 # We only enable strict check for non-quantized models
435 # that have loaded weights tracking by default.
436 default_enable_weights_track = (
437 model_config.quantization is None and loaded_weights is not None
438 )
439 enable_weights_track = (
440 self.enable_weights_track
441 if self.enable_weights_track is not None
442 else default_enable_weights_track
443 )
444 if enable_weights_track:
445 self.track_weights_loading(model, loaded_weights)

447 def track_weights_loading(
448 self, model: nn.Module, loaded_weights: set[str] | None
449 ) -> None:
450 weights_to_load = {name for name, _ in model.named_parameters()}
451 if loaded_weights is not None:
452 # ignore online quantization scales
453 for name, module in model.named_modules():
454 quant_method = getattr(module, "quant_method", None)
455 has_online_quant = getattr(quant_method, "uses_meta_device", False)
456 has_postprocess_quant = getattr(
457 quant_method, "process_weights_after_loading", None
458 )
459 # ignore kv_cache scale and online quant scale,
460 # which can be missing in checkpoints
461 if has_online_quant or has_postprocess_quant:
462 for param_name, _ in module.named_parameters():
463 full_name = f"{name}.{param_name}" if name else param_name
464 loaded_weights.add(full_name)
465 weights_not_loaded = weights_to_load - loaded_weights
466 if weights_not_loaded:
467 raise ValueError(
468 "Following weights were not initialized from "
469 f"checkpoint: {weights_not_loaded}"
470 )
