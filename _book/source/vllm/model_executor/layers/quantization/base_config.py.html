23class QuantizeMethodBase(ABC):
24 """Base class for different quantized methods."""
25
26 requires_device_loading: bool = True
27 """Whether post-load processing requires parameters on the target device."""
28
29 uses_meta_device: bool = False
30 """
31 Whether this method creates weights on meta device for online quantization.
32 When True, weights are created on meta device and quantized layer-wise
33 in process_weights_after_loading, reducing peak memory during loading.
34 """
35
36 supports_pre_processed_weights: bool = False
37 """
38 Whether ``process_weights_after_loading`` supports running under
39 ``weights_already_processed``. Methods must skip tensor transforms in
40 that mode; the loader driver rejects methods that do not declare support.
41 """
42
43 @abstractmethod
44 def create_weights(
45 self, layer: torch.nn.Module, *weight_args, **extra_weight_attrs
46 ):
47 """Create weights for a layer.
48
49 The weights will be set as attributes of the layer."""
50 raise NotImplementedError
51
52 @abstractmethod
53 def apply(self, layer: torch.nn.Module, *args, **kwargs) -> torch.Tensor:
54 """Apply the weights in layer to the input tensor.
55
56 Expects create_weights to have been called before on the layer."""
57 raise NotImplementedError
58
59 # Not required functions
60 def embedding(self, layer: torch.nn.Module, *args, **kwargs) -> torch.Tensor:
61 """Gather embeddings in the layer based on indices in the input tensor.
62
63 Expects create_weights to have been called before on the layer."""
64 raise NotImplementedError
65
66 # Not required functions
67 def tie_weights(self, layer: torch.nn.Module, embed_tokens: torch.nn.Module):
68 """Tie ``layer``'s weight to ``embed_tokens``' weight.
69
70 The default shares the weight tensor, which is the standard behavior for
71 tied word embeddings and matches what ``ParallelLMHead.tie_weights`` did
72 directly before quantization methods became responsible for it.
73 Quantization methods that need special weight handling (e.g. repacked
74 weights) override this.
75
76 Expects create_weights to have been called before on the layer."""
77 layer.weight = embed_tokens.weight
78 return layer
79
80 def process_weights_after_loading(self, layer: nn.Module) -> None:
81 """Process the weight after loading.
82
83 This can be used for example, to transpose weights for computation.
84 """
85 return
86

88def method_has_implemented_embedding(method_class: type[QuantizeMethodBase]) -> bool:
89 """
90 Not all quant methods have embedding implemented, so we need to check that
91 it exists for our given method. We check this by making sure the function
92 has been changed from the base implementation.
93 """
94 base_embedding = inspect.getattr_static(QuantizeMethodBase, "embedding", None)
95 class_embedding = inspect.getattr_static(method_class, "embedding", None)
96
97 return class_embedding is not None and class_embedding is not base_embedding

100class QuantizationConfig(ABC):
101 """Base class for quantization configs."""
102
103 _ignore_unexpected_suffixes = (
104 ".q_scale",
105 ".k_scale",
106 ".v_scale",
107 ".q_zero_point",
108 ".k_zero_point",
109 ".v_zero_point",
110 )
111 """Suffixes of quantization parameters that may be present in the checkpoint but
112 not in the model, and should be ignored if unexpected during loading. These are used
113 after remapping, so should be in vLLM format (e.g. .q_scale, not .q.scale)."""
114 online_quantization_config: "OnlineQuantizationConfig | None" = None
115
116 def __init__(self):
117 super().__init__()
118 # mapping is updated by models as they initialize
119 self.packed_modules_mapping: dict[str, list[str]] = dict()
120
121 @abstractmethod
122 def get_name(self) -> QuantizationMethods:
123 """Name of the quantization method."""
124 raise NotImplementedError
125
126 @abstractmethod
127 def get_supported_act_dtypes(self) -> list[torch.dtype]:
128 """List of supported activation dtypes."""
129 raise NotImplementedError
130
131 @classmethod
132 @abstractmethod
133 def get_min_capability(cls) -> int:
134 """Minimum GPU capability to support the quantization method.
135
136 E.g., 70 for Volta, 75 for Turing, 80 for Ampere.
137 This requirement is due to the custom CUDA kernels used by the
138 quantization method.
139 """
140 raise NotImplementedError
141
142 @staticmethod
143 @abstractmethod
144 def get_config_filenames() -> list[str]:
145 """List of filenames to search for in the model directory."""
146 raise NotImplementedError
147
148 @classmethod
149 @abstractmethod
150 def from_config(cls, config: dict[str, Any]) -> "QuantizationConfig":
151 """Create a config class from the model's quantization config."""
152 raise NotImplementedError
153
154 @classmethod
155 def override_quantization_method(
156 cls,
157 hf_quant_cfg: dict[str, Any],
158 user_quant: str | None,
159 hf_config: Any = None,
160 ) -> QuantizationMethods | None:
161 """
162 Detects if this quantization method can support a given checkpoint
163 format by overriding the user specified quantization method --
164 this method should only be overwritten by subclasses in exceptional
165 circumstances.
166
167 Args:
168 hf_quant_cfg: The checkpoint's quantization config dict.
169 user_quant: The user-specified quantization method string.
170 hf_config: The HuggingFace model config object (e.g. for
171 model_type checks). May be None if not available.
172 """
173 return None
174
175 @staticmethod
176 def get_from_keys(config: dict[str, Any], keys: list[str]) -> Any:
177 """Get a value from the model's quantization config."""
178 for key in keys:
179 if key in config:
180 return config[key]
181 raise ValueError(
182 f"Cannot find any of {keys} in the model's quantization config."
183 )
184
185 @staticmethod
186 def get_from_keys_or(config: dict[str, Any], keys: list[str], default: Any) -> Any:
187 """Get an optional value from the model's quantization config."""
188 try:
189 return QuantizationConfig.get_from_keys(config, keys)
190 except ValueError:
191 return default

209 @staticmethod
210 def get_cache_scale_mapper() -> "WeightsMapper":
211 """Mapping from checkpoint KV-cache scale names to vLLM scale names.
212
213 Returning a mapper here causes `AutoWeightsLoader` to apply it to the
214 weight stream automatically; individual model `load_weights` methods
215 do not need to know about KV-cache scales.
216 """
217 from vllm.model_executor.models.utils import WeightsMapper
218
219 orig_to_new_regex = {
220 # Deprecated fused kv_scale -> attn.k_scale
221 re.compile(r"\.kv_scale$"): r".attn.k_scale",
222 # ModelOpt: .self_attn.{k,v}_proj.{k,v}_scale -> .self_attn.attn.*
223 re.compile(r"\.self_attn\.[kv]_proj\.([kv])_scale$"): (
224 r".self_attn.attn.\1_scale"
225 ),
226 # Fused QKV / qkqkv proj: .self_attn.qk(qk)v_proj.{k,v}_scale -> attn
227 re.compile(r"\.self_attn\.qk(?:qk)?v_proj\.([kv])_scale$"): (
228 r".self_attn.attn.\1_scale"
229 ),
230 # NemotronH: .mixer.{k,v}_proj.{k,v}_scale -> .mixer.attn.*
231 re.compile(r"\.mixer\.[kv]_proj\.([kv])_scale$"): r".mixer.attn.\1_scale",
232 # HYV3: .self_attn.q.scale -> .self_attn.attn.q_scale
233 re.compile(r"\.self_attn\.q\.scale$"): r".self_attn.attn.q_scale",
234 # HYV3: .self_attn.{k,v}_cache.scale -> .self_attn.attn.{k,v}_scale
235 re.compile(r"\.self_attn\.([kv])_cache\.scale$"): (
236 r".self_attn.attn.\1_scale"
237 ),
238 # Default: .{q,k,v}_scale -> .attn.{q,k,v}_scale (unless already .attn)
239 re.compile(r"(?<!\.attn)\.([qkv])_scale$"): r".attn.\1_scale",
240 re.compile(r"(?<!\.attn)\.([qkv])_zero_point$"): r".attn.\1_zero_point",
241 }
242 return WeightsMapper(orig_to_new_regex=orig_to_new_regex)
243

245 def get_checkpoint_weight_mapper() -> "WeightsMapper":

247 from vllm.model_executor.models.utils import WeightsMapper

285def resolve_quant_method(
286 quant_config: QuantizationConfig, layer: torch.nn.Module, prefix: str
287) -> QuantizeMethodBase | None:
288 """Return the checkpoint method with configured online quantization."""
289 from vllm.model_executor.layers.fused_moe import RoutedExperts
290 from vllm.model_executor.layers.fused_moe.unquantized_fused_moe_method import (
291 UnquantizedFusedMoEMethod,
292 )
293 from vllm.model_executor.layers.linear import (
294 LinearBase,
295 UnquantizedLinearMethod,
296 )
297
298 base_quant_method = quant_config.get_quant_method(layer, prefix)
299 if quant_config.online_quantization_config is None:
300 return base_quant_method
301 # Online quantization currently supports only LinearBase and RoutedExperts.
302 # Embeddings and ParallelLMHead retain their checkpoint quantization method.
303 if not isinstance(layer, (LinearBase, RoutedExperts)):
304 return base_quant_method
305
306 quant_config.online_quantization_config.packed_modules_mapping = (
307 quant_config.packed_modules_mapping
308 )
309 checkpoint_is_quantized = base_quant_method is not None and not isinstance(
310 base_quant_method, (UnquantizedLinearMethod, UnquantizedFusedMoEMethod)
311 )
312 online_target = quant_config.online_quantization_config.resolve_quant_method_cls(
313 layer, prefix
314 )
315 if checkpoint_is_quantized:
316 if online_target is not None:
317 raise ValueError(
318 f"Cannot apply requested online quantization {online_target[3]} to "
319 f"pre-quantized layer {prefix}: {base_quant_method} was already "
320 "selected by the checkpoint quantization config."
321 )
322 return base_quant_method
323 if online_target is None:
324 return base_quant_method
325 return quant_config.online_quantization_config.get_quant_method(layer, prefix)
