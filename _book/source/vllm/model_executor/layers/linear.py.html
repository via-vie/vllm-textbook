165 """Linear method without quantization."""

169 def __init__(self) -> None:

172 config.kernel_config.linear_backend if config is not None else "auto"

176 def create_weights(
177 self,
178 layer: torch.nn.Module,
179 input_size_per_partition: int,
180 output_partition_sizes: list[int],
181 input_size: int,
182 output_size: int,
183 params_dtype: torch.dtype,

192 data=torch.empty(
193 sum(output_partition_sizes),
194 input_size_per_partition,
195 dtype=params_dtype,
196 ),

200 )

205 def process_weights_after_loading(self, layer: torch.nn.Module) -> None:
206 if current_platform.is_cpu():
207 # MLA's kv_b_proj (see `skip_weight_relayout`): not perf-critical,
208 # so skip packing and use a plain fallback.
209 if getattr(layer, "skip_weight_relayout", False):
210 layer.cpu_linear = torch.nn.functional.linear
211 return
212
213 from vllm.model_executor.layers.utils import dispatch_cpu_unquantized_gemm
214
215 dispatch_cpu_unquantized_gemm(layer, remove_weight=True)
216 elif current_platform.is_xpu():
217 # Opt-in: F.linear on XPU is faster with an N-contiguous (N, K) weight
218 # when K > N, but oneDNN's ab-weights matmul is not run-to-run bitwise
219 # bitwise reproducible. Off by default.
220 weight = layer.weight.data
221 if (
222 envs.VLLM_XPU_FORCE_N_CONTIG_WEIGHT
223 and weight.ndim == 2
224 and weight.stride(0) != 1
225 ):
226 layer.weight.data = weight.t().contiguous().t()

229 self,
230 layer: torch.nn.Module,
231 x: torch.Tensor,
232 bias: torch.Tensor | None = None,
233 ) -> torch.Tensor:
234 if envs.VLLM_BATCH_INVARIANT and (
235 current_platform.is_cuda_alike() or current_platform.is_xpu()
236 ):
237 return linear_batch_invariant(x, layer.weight, bias)
238 return self._gemm_impl(layer, x, layer.weight, bias)

432

628class DCPGroupColumnParallelLinear(ColumnParallelLinear):
629 """Column-parallel linear whose weight is sharded across DCP groups.
630
631 With Decode Context Parallelism (DCP) the KV cache is sharded across a DCP
632 group, so MLA decode must attend the group's full head set. This layer shards
633 its output across DCP *groups* (effective tp size ``tp_size //
634 dcp_world_size``) rather than across every rank, so each rank in a group
635 holds the whole group's heads, letting decode skip the query all-gather.
636
637 :meth:`forward` returns the group's full head set. :meth:`_local_view`
638 extracts this rank's TP shard for prefill.
639 """
640
641 def __init__(self, *args, **kwargs):
642 dcp_world_size = (
643 get_current_vllm_config().parallel_config.decode_context_parallel_size
644 )
645 rank = get_tensor_model_parallel_rank()
646 world_size = get_tensor_model_parallel_world_size()
647 self.group_size = max(dcp_world_size, 1)
648 self.qrep_active = self.group_size > 1
649 self.rank_in_group = rank % self.group_size
650 super().__init__(
651 *args,
652 **kwargs,
653 tp_rank=rank // self.group_size,
654 tp_size=world_size // self.group_size,
655 )
656
657 def _local_view(self, out: torch.Tensor) -> torch.Tensor:
658 """Slice this rank's tp head shard from a group-heads output.
659
660 ``out`` is head-shaped, i.e. ``(..., group_heads, head_dim)``.
661 """
662 if self.group_size == 1:
663 return out
664 n = out.shape[-2] // self.group_size
665 start = self.rank_in_group * n
666 return out[..., start : start + n, :].contiguous()

670 """Packed linear layers with column parallelism.
671
672 Similar to ColumnParallelLinear, but the weight matrix is concatenated
673 along the output dimension. When the weight matrix is loaded, the
674 different partitions are sharded separately.
675
676 Args:
677 input_size: input dimension of the linear layer.
678 output_sizes: list of output dimensions of the linear layer.
679 bias: If true, add bias.
680 gather_output: If true, call all-gather on output and make the output
681 available to all GPUs, otherwise, every GPU will have
682 its own output.
683 skip_bias_add: This was added to enable performance optimizations where
684 bias can be fused with other element-wise operations. we
685 skip adding bias but instead return it.
686 params_dtype: Data type for the parameters.
687 quant_config: Quantization configure.
688 prefix: The name of the layer in the state dict, including all parents
689 (e.g. model.layers.0.qkv_proj)
690 return_bias: If true, return bias together with outputs in forward pass.
