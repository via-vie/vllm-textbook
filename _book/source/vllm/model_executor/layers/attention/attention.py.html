58def validate_kv_sharing_target(
59 current_layer_name, target_layer_name, static_forward_context
60):
61 error_msg = (
62 f"Specified KV sharing target layer for {current_layer_name} "
63 f"is not valid: target layer {target_layer_name} "
64 )
65
66 if current_layer_name == target_layer_name:
67 raise ValueError(error_msg + "cannot be the same as the current layer.")
68
69 if target_layer_name not in static_forward_context:
70 from vllm.model_executor.models.utils import extract_layer_index
71
72 # If target layer name is not in the static fwd context, it means either
73 # a) the target layer does not come BEFORE the current layer, or
74 # b) the target layer is not an Attention layer that exists in the model
75 current_layer_idx = extract_layer_index(current_layer_name)
76 target_layer_idx = extract_layer_index(target_layer_name)
77 if current_layer_idx <= target_layer_idx:
78 raise ValueError(error_msg + "must come before the current layer.")
79 else:
80 raise ValueError(error_msg + "is not a valid Attention layer in the model.")
81
82 # Currently KV sharing is only supported between layers of the same type
83 target_layer_attn_type = static_forward_context[target_layer_name].attn_type
84 expected = static_forward_context[current_layer_name].attn_type
85 if target_layer_attn_type != expected:
86 raise ValueError(
87 error_msg + f"must be the same type as the current layer ({expected})."
88 )

98def _largest_kernel_block_within(
99 attn_backend: "type[AttentionBackend]",
100 per_token_bytes: int,
101 page_budget: int,
102 fallback: int,
103) -> int:
104 """Largest supported kernel block size whose page fits in ``page_budget``.
105
106 A padded spec (e.g. skip-quant layer) that pads its page up to a large shared page
107 wastes ``page_budget - block*per_token`` bytes per block. Picking the largest kernel
108 block whose natural page still fits under ``page_budget`` minimizes that waste.
109 ``MultipleOf`` declarations are expanded to the largest aligned block that fits.
110 Falls back to the smallest supported block when nothing fits.
111 """
112 from vllm.v1.attention.backend import MultipleOf
113
114 sizes = attn_backend.get_supported_kernel_block_sizes()
115 max_block_size = page_budget // per_token_bytes
116 candidates = [s for s in sizes if isinstance(s, int)]
117 candidates.extend(
118 max(s.base, max_block_size // s.base * s.base)
119 for s in sizes
120 if isinstance(s, MultipleOf)
121 )
122 if not candidates:
123 return fallback
124 smallest = min(candidates)
125 fitting = [b for b in candidates if b * per_token_bytes <= page_budget]
126 return max(fitting) if fitting else smallest

129def set_default_quant_scales(layer: nn.Module, register_buffer: bool = False) -> None:
130 """Sets default quantization scales for the layer."""
131 if register_buffer:
132 layer.register_buffer("_k_scale", torch.tensor(1.0, dtype=torch.float32))
133 layer.register_buffer("_v_scale", torch.tensor(1.0, dtype=torch.float32))
134 layer.register_buffer("_q_scale", torch.tensor(1.0, dtype=torch.float32))
135 layer.register_buffer("_prob_scale", torch.tensor(1.0, dtype=torch.float32))
136 else:
137 layer._k_scale.fill_(1.0)
138 layer._v_scale.fill_(1.0)
139 layer._q_scale.fill_(1.0)
140 layer._prob_scale.fill_(1.0)
141
142 # We also keep q/k/v_scale on host (cpu) memory for attention
143 # backends that require the scales to be on host instead of on device.
144 # e.g. Flashinfer & AITER
145 layer._q_scale_float = 1.0
146 layer._k_scale_float = 1.0
147 layer._v_scale_float = 1.0
148 layer._k_scale_cpu = torch.tensor(1.0, dtype=torch.float32)
149 layer._v_scale_cpu = torch.tensor(1.0, dtype=torch.float32)
150 layer._prob_scale_float = 1.0

153def _init_kv_cache_quant(
154 layer: nn.Module,
155 quant_config: QuantizationConfig | None,
156 prefix: str,
157) -> None:
158 """Initializes KV cache scaling factors and quantization method.
159
160 This helper function sets up the KV cache quantization attributes that are
161 shared between Attention and MLAAttention layers. It initializes scale
162 tensors for query, key, value, and probability, and configures the
163 quantization method if applicable.
164
165 Args:
166 layer: The attention layer instance to initialize.
167 quant_config: Optional quantization configuration.
168 prefix: Layer name prefix for quantization method lookup.
169 """
170
171 # Note [Register q/k/v/prob scales in state dict]
172 # When calling model.to(device), only parameters/buffers in state dict are
173 # moved. If not registering q/k/v/prob scales in state dict, there would
174 # be an IMA error when a cuda kernel (e.g., quant_fp8) accesses the tensor
175 # on cpu.
176 # Registering in state dict means it interacts with weight loading. One edge
177 # case is when quant_method is None, or quant_method is UnquantizedLinearMethod
178 # (i.e., should_load_quant_weights(quant_method) == False).
179 # In this case, the checkpoint does not have the scales. We need to
180 # initialize the scales to 1.0 and update the scales after weight loading.
181 # This is espectially important when we load dummy weights first (providing
182 # wrong scales) and then load real weights (which misses scales and keeps the
183 # wrong scales from dummy load).
184 set_default_quant_scales(layer, register_buffer=True)
185
186 # The output scale on host memory. This should be the input scale of
187 # the quant op after this attention layer.
188 layer._o_scale_float = None
189
190 quant_method = (
191 resolve_quant_method(quant_config, layer, prefix=prefix)
192 if quant_config
193 else None
194 )
195
196 # See [Note: Register q/k/v/prob scales in state dict]
197 if should_load_quant_weights(quant_method):
198 assert isinstance(quant_method, BaseKVCacheMethod)
199 # TODO (mgoin): kv cache dtype should be specified in the FP8
200 # checkpoint config and become the "auto" behavior
201 if layer.kv_cache_dtype == "fp8_e5m2":
202 # A compressed-tensors checkpoint stores fp8 KV scales only when it
203 # declares a kv_cache_scheme; weight-only ones declare none and must
204 # keep fp8_e5m2, the only fp8 KV dtype usable on Ampere.
205 from vllm.model_executor.layers.quantization.compressed_tensors.compressed_tensors import ( # noqa: E501
206 CompressedTensorsConfig,
207 CompressedTensorsKVCacheMethod,
208 )
209
210 if not isinstance(quant_method, CompressedTensorsKVCacheMethod) or (
211 cast(CompressedTensorsConfig, quant_method.quant_config).kv_cache_scheme
212 is not None
213 ):
214 raise ValueError(
215 "fp8_e5m2 kv-cache is not supported with fp8 checkpoints."
216 )
217 # If quantization is enabled, we make "k_scale" and "v_scale"
218 # parameters so that it can be loaded from the model checkpoint.
219 # The k/v_scale will then be converted back to native float32
220 # values after weight loading.
221 layer.quant_method = quant_method
222 layer.quant_method.create_weights(layer)

225class Attention(nn.Module, AttentionLayerBase):
226 """Attention layer.
227
228 This class takes query, key, and value tensors as input. The input tensors
229 can either contain prompt tokens or generation tokens.
230 The class does the following:
231
232 1. Store the input key and value tensors in the KV cache.
233 2. Perform (multi-head/multi-query/grouped-query) attention.
234 3. Return the output tensor.
235 """
236
237 def __init__(
238 self,
239 num_heads: int,
240 head_size: int,
241 scale: float,
242 num_kv_heads: int | None = None,

258 The KV cache is stored inside this class and is accessed via

285 if kv_cache_scheme is not None and kv_cache_dtype == "auto":
286 kv_cache_dtype = "fp8"

314 "Layer %s: kv_cache_dtype=%s, sliding_window=%s",

316 kv_cache_dtype,
317 sliding_window,

327 f"num_heads ({num_heads}) is not divisible by num_kv_heads ({num_kv_heads})"

332 self.num_heads = num_heads
333 self.head_size = head_size
334 self.head_size_v = self.head_size if head_size_v is None else head_size_v

336 self.sliding_window = sliding_window
337 self.has_sink = extra_impl_args.get("sinks") is not None

485 def forward(
486 self,
487 query: torch.Tensor,
488 key: torch.Tensor,
489 value: torch.Tensor,
490 # For some alternate attention backends like MLA the attention output
491 # shape does not match the query shape, so we optionally let the model
492 # definition specify the output tensor shape.
493 output_shape: torch.Size | None = None,
494 output_dtype: torch.dtype | None = None,
495 ) -> torch.Tensor:
496 """
497 The KV cache is stored inside this class and is accessed via
498 `self.kv_cache`.
499
500 Attention metadata (`attn_metadata`) is set using a context manager in
501 the model runner's `execute_model` method. It is accessed via forward
502 context using
503 `vllm.forward_context.get_forward_context().attn_metadata`.
504 """
505 if output_dtype is None:
506 output_dtype = query.dtype
507 if self.query_quant is not None:
508 # quantizing with a simple torch operation enables
509 # torch.compile to fuse this into previous ops
510 # which reduces overheads during decoding.
511 # Otherwise queries are quantized using custom ops
512 # which causes decoding overheads
513 assert self.kv_cache_dtype in {"fp8", "fp8_e4m3"} or (
514 self.kv_cache_dtype.startswith("nvfp4")
515 )
516
517 # check if query quantization is supported
518 if self.impl.supports_quant_query_input:
519 query, _ = self.query_quant(query, self._q_scale)
520
521 if output_shape is None:
522 # Handle both 2D [num_tokens, hidden] and
523 # 3D [num_tokens, heads, head_dim] query
524 num_tokens = query.shape[0]
525 output_shape = torch.Size((num_tokens, self.num_heads * self.head_size_v))
526 output = torch.empty(output_shape, dtype=output_dtype, device=query.device)
527 hidden_size = output_shape[-1]
528 # Reshape the query, key, and value tensors.
529 # NOTE(woosuk): We do this outside the custom op to minimize the
530 # CPU overheads from the non-CUDA-graph regions.
531 query = query.view(-1, self.num_heads, self.head_size)
532 output = output.view(-1, self.num_heads, self.head_size_v)
533 if key is not None:
534 key = key.view(-1, self.num_kv_heads, self.head_size)
535 if value is not None:
536 value = value.view(-1, self.num_kv_heads, self.head_size_v)
537 kv_cache_dummy_dep = None
538 if self.use_direct_call:
539 # Skip this if sharing KV cache with an earlier attention layer.
540 if (
541 not self.attn_backend.forward_includes_kv_cache_update
542 and self.kv_sharing_target_layer_name is None
543 and key is not None
544 and value is not None
545 ):
546 kv_cache_dummy_dep = unified_kv_cache_update(
547 key, value, self.layer_name
548 )
549 unified_attention_with_output(
550 query,
551 key,
552 value,
553 output,
554 self.layer_name,
555 kv_cache_dummy_dep=kv_cache_dummy_dep,
556 )
557 else:
558 # Skip this if sharing KV cache with an earlier attention layer.
559 encoded = _encode_layer_name(self.layer_name)
560 if (
561 not self.attn_backend.forward_includes_kv_cache_update
562 and self.kv_sharing_target_layer_name is None
563 and key is not None
564 and value is not None
565 ):
566 kv_cache_dummy_dep = torch.ops.vllm.unified_kv_cache_update(
567 key, value, encoded
568 )
569 torch.ops.vllm.unified_attention_with_output(
570 query,
571 key,
572 value,
573 output,
574 encoded,
575 kv_cache_dummy_dep=kv_cache_dummy_dep,
576 )
577 return output.view(-1, hidden_size)

604 def get_kv_cache_spec(self, vllm_config: VllmConfig) -> KVCacheSpec | None:
605 # Block size may get updated after model loading, refresh it
606 block_size = vllm_config.cache_config.block_size
607 # Encoder-only attention is prefill-only and keeps no autoregressive KV
608 # cache. In hybrid models (e.g. Qwen3.5 / ColQwen3.5: GatedDeltaNet
609 # linear_attention interleaved with full_attention) the runner iterates
610 # every attention module to build the KV-cache spec, so an ENCODER_ONLY
611 # full_attention layer reaches here; it contributes no KV cache group.
612 if self.attn_type in (AttentionType.ENCODER_ONLY, AttentionType.ENCODER):
613 return None
614 # Should not be called for enc-dec attention.
615 assert self.attn_type == AttentionType.DECODER
616 quant_mode = get_kv_quant_mode(self.kv_cache_dtype)
617 if self.sliding_window is not None:
618 assert not self.attn_backend.is_mla(), (
619 "MLA is not supported for sliding window"
620 )
621 # SW chooses its own block_size, decoupled from the user's
622 # ``--block-size`` (which only constrains primary attention).
623 # When this SW layer is a padded spec (skip-quant: its page is
624 # padded up to ``skip_page_size_padded``), pick the largest kernel
625 # block that still fits the shared page so we waste fewer padding
626 # bytes per block. Otherwise (page_size_padded is None) take the
627 # primary block size when the backend can run it unsplit: if this
628 # page does not divide the primary page, ``unify`` then pads it
629 # (padded pages cannot be split) instead of scaling a small block
630 # to a size coprime with the primary one, which inflates the
631 # scheduler LCM (e.g. a 1024 B/token SWA draft next to a 1152
632 # B/token MLA target: 1728 vs 1536 gives LCM 13824). Backends that
633 # cannot run the primary block start from their smallest block and
634 # ``unify`` scales it up by an integer ratio.
635 shared_page = vllm_config.cache_config.skip_page_size_padded
636 # The backend owns its packing
637 sw_per_token = self.attn_backend.customize_spec(
638 SlidingWindowSpec(
639 block_size=1,
640 num_kv_heads=self.num_kv_heads,
641 head_size=self.head_size,
642 head_size_v=self.head_size_v,
643 dtype=self.kv_cache_torch_dtype,
644 kv_quant_mode=quant_mode,
645 sliding_window=self.sliding_window,
646 )
647 ).real_page_size_bytes
648 page_budget = shared_page or sw_per_token * block_size
649 sw_block_size = _largest_kernel_block_within(
650 self.attn_backend, sw_per_token, page_budget, block_size
651 )
652 return SlidingWindowSpec(
653 block_size=sw_block_size,
654 num_kv_heads=self.num_kv_heads,
655 head_size=self.head_size,
656 head_size_v=self.head_size_v,
657 dtype=self.kv_cache_torch_dtype,
658 kv_quant_mode=quant_mode,
659 sliding_window=self.sliding_window,
660 page_size_padded=shared_page,
661 )
662 else:
663 return FullAttentionSpec(
664 block_size=block_size,
665 num_kv_heads=self.num_kv_heads,
666 head_size=self.head_size,
667 head_size_v=self.head_size_v,
668 dtype=self.kv_cache_torch_dtype,
669 kv_quant_mode=quant_mode,
670 )

673def get_attention_context(
674 layer_name: str,
675) -> tuple[Any, "Attention | MLAAttention", torch.Tensor, torch.Tensor]:
676 """Extract attention context for a given layer.
677
678 This helper function extracts the attention metadata, attention layer
679 instance, KV cache tensor, and slot mapping for a specific layer.
680
681 Args:
682 layer_name: The name/identifier of the attention layer.
683
684 Returns:
685 A tuple containing:
686 - attn_metadata: Attention metadata for this specific layer, or None if
687 no metadata available
688 - attn_layer: The attention layer instance (Attention or MLAAttention)
689 - kv_cache: The KV cache tensor for current forward pass
690 - slot_mapping: The slot mapping for this specific layer
691
692 Note: attn_metadata may be None, but attn_layer and kv_cache are always
693 extracted from the forward context.
694 """
695 forward_context: ForwardContext = get_forward_context()
696 attn_metadata_raw = forward_context.attn_metadata
697 attn_metadata: AttentionMetadata
698 if isinstance(attn_metadata_raw, dict):
699 attn_metadata = attn_metadata_raw[layer_name]
700 elif isinstance(attn_metadata_raw, list):
701 # list[dict[str, AttentionMetadata]]: used in speculative decoding
702 # where [0] is the base-model (non-speculative) metadata dict.
703 attn_metadata = attn_metadata_raw[0][layer_name]
704 else:
705 attn_metadata = attn_metadata_raw
706 attn_layer: Attention | MLAAttention = forward_context.no_compile_layers[layer_name]
707 kv_cache = attn_layer.kv_cache
708 slot_mapping = forward_context.slot_mapping
709 assert isinstance(slot_mapping, dict), (
710 f"Expected slot_mapping to be a dict, got {type(slot_mapping)}. "
711 )
712 layer_slot_mapping = slot_mapping.get(layer_name)
713 return attn_metadata, attn_layer, kv_cache, layer_slot_mapping

717 key: torch.Tensor,
718 value: torch.Tensor,
719 layer_name: LayerNameType,
720) -> torch.Tensor:
721 """
722 Returns a dummy that is passed to unified_attention to signal a side effect and

724 """

726 _, attn_layer, kv_cache, layer_slot_mapping = get_attention_context(layer_name)

729 f"{attn_layer.impl.__class__.__name__} does not support kv cache update"
730 )
731 attn_layer.impl.do_kv_cache_update( # type: ignore[attr-defined]
732 attn_layer,
733 key,
734 value,
735 kv_cache,
736 layer_slot_mapping,
737 )
738
739 return key.new_empty(0)
