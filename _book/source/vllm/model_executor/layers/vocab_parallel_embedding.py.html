175@torch.compile(dynamic=True, backend=current_platform.simple_compile_backend)
176def get_masked_input_and_mask(
177 input_: torch.Tensor,
178 org_vocab_start_index: int,
179 org_vocab_end_index: int,
180 num_org_vocab_padding: int,
181 added_vocab_start_index: int,
182 added_vocab_end_index: int,
183) -> tuple[torch.Tensor, torch.Tensor]:
184 # torch.compile will fuse all of the pointwise ops below
185 # into a single kernel, making it very fast
186 org_vocab_mask = (input_ >= org_vocab_start_index) & (input_ < org_vocab_end_index)
187 added_vocab_mask = (input_ >= added_vocab_start_index) & (
188 input_ < added_vocab_end_index
189 )
190 added_offset = (
191 added_vocab_start_index
192 - (org_vocab_end_index - org_vocab_start_index)
193 - num_org_vocab_padding
194 )
195 valid_offset = (org_vocab_start_index * org_vocab_mask) + (
196 added_offset * added_vocab_mask
197 )
198 vocab_mask = org_vocab_mask | added_vocab_mask
199 input_ = vocab_mask * (input_ - valid_offset)
200 return input_, ~vocab_mask

204@PluggableLayer.register("vocab_parallel_embedding")
205class VocabParallelEmbedding(PluggableLayer):
206 """Embedding parallelized in the vocabulary dimension.
207
208 Adapted from torch.nn.Embedding, note that we pad the vocabulary size to
209 make sure it is divisible by the number of model parallel GPUs.
210
211 In order to support various loading methods, we ensure that LoRA-added
212 embeddings are always at the end of TP-sharded tensors. In other words,
213 we shard base embeddings and LoRA embeddings separately (both padded),
214 and place them in the same tensor.
215 In this example, we will have the original vocab size = 1010,
216 added vocab size = 16 and padding to 64. Therefore, the total
217 vocab size with padding will be 1088 (because we first pad 1010 to
218 1024, add 16, and then pad to 1088).
219 Therefore, the tensor format looks like the following:
220 TP1, rank 0 (no sharding):
221 |< --------BASE-------- >|< -BASE PADDING-- >|< -----LORA------ >|< -LORA PADDING-- >|
222 corresponding token_id: | 0 | 1 | ... | 1009 | -1 | ... | -1 | 1010 | ... | 1025 | -1 | ... | -1 |
223 index: | 0 | 1 | ... | 1009 | 1010 | ... | 1023 | 1024 | ... | 1039 | 1040 | ... | 1087 |
224
225 TP2, rank 0:
226 |< --------------------BASE--------------------- >|< -----LORA------ >|< -LORA PADDING- >|
227 corresponding token_id: | 0 | 1 | 2 | ... | 497 | 498 | ... | 511 | 1010 | ... | 1025 | -1 | ... | -1 |
228 index: | 0 | 1 | 2 | ... | 497 | 498 | ... | 511 | 512 | ... | 527 | 528 | ... | 543 |
229 TP2, rank 1:
230 |< -----------BASE----------- >|< -BASE PADDING- >|< -----------LORA PADDING----------- >|
231 corresponding token_id: | 512 | 513 | 514 | ... | 1009 | -1 | ... | -1 | -1 | ... | -1 | -1 | ... | -1 |
232 index: | 0 | 1 | 2 | ... | 497 | 498 | ... | 511 | 512 | ... | 527 | 528 | ... | 543 |
233
234 Args:
235 num_embeddings: vocabulary size.
236 embedding_dim: size of hidden state.
237 params_dtype: type of the parameters.
238 org_num_embeddings: original vocabulary size (without LoRA).
239 padding_size: padding size for the vocabulary.
240 quant_config: quant config for the layer
241 prefix: full name of the layer in the state dict
242 disable_tp: If true, tensor parallelism will be disabled for this layer.
243 quant_method: Preselected quantization method for model-specific layers.
244 parallel_group: Process group used to shard and reduce the embedding.
245 """ # noqa: E501

249 def __init__(
250 self,
251 num_embeddings: int,
252 embedding_dim: int,
253 params_dtype: torch.dtype | None = None,
254 org_num_embeddings: int | None = None,
255 padding_size: int = DEFAULT_VOCAB_PADDING_SIZE,
256 quant_config: QuantizationConfig | None = None,
257 prefix: str = "",
258 *,
259 disable_tp: bool = False,
260 quant_method: QuantizeMethodBase | None = None,
261 parallel_group: GroupCoordinator | None = None,
262 ):
263 super().__init__()
264
265 # Keep the input dimensions.
266 self.disable_tp = disable_tp
267 self.parallel_group = parallel_group
268 if disable_tp:
269 tp_rank, self.tp_size = 0, 1
270 elif parallel_group is not None:
271 tp_rank = parallel_group.rank_in_group
272 self.tp_size = parallel_group.world_size
273 else:
274 tp_rank = get_tensor_model_parallel_rank()
275 self.tp_size = get_tensor_model_parallel_world_size()
276 self.tp_rank = tp_rank
277 self.num_embeddings = num_embeddings
278 self.padding_size = padding_size
279 self.org_vocab_size = org_num_embeddings or num_embeddings
280 num_added_embeddings = num_embeddings - self.org_vocab_size
281 self.org_vocab_size_padded = pad_vocab_size(
282 self.org_vocab_size, self.padding_size
283 )
284 self.num_embeddings_padded = pad_vocab_size(
285 self.org_vocab_size_padded + num_added_embeddings, self.padding_size
286 )
287 assert self.org_vocab_size_padded <= self.num_embeddings_padded
288
289 self.shard_indices = self._get_indices(
290 self.num_embeddings_padded,
291 self.org_vocab_size_padded,
292 self.num_embeddings,
293 self.org_vocab_size,
294 tp_rank,
295 self.tp_size,
296 )
297 self.embedding_dim = embedding_dim
298
299 # Avoid overriding a preselected model-specific method with generic
300 # config-based dispatch.
301 if quant_method is None and quant_config is not None:
302 quant_method = resolve_quant_method(quant_config, self, prefix=prefix)
303 if quant_method is None:
304 quant_method = UnquantizedEmbeddingMethod()
305
306 # If we are making an embedding layer, then our quantization linear
307 # method must implement the embedding operation. If we are another
308 # layer type like ParallelLMHead, this is not important.
309 is_embedding_layer = not isinstance(self, ParallelLMHead)
310 quant_method_implements_embedding = method_has_implemented_embedding(
311 type(quant_method)
312 )
313 if is_embedding_layer and not quant_method_implements_embedding:
314 raise NotImplementedError(
315 f"The class {type(quant_method).__name__} must implement "
316 "the 'embedding' method, see UnquantizedEmbeddingMethod."
317 )
318
319 self.quant_method: QuantizeMethodBase = quant_method
320
321 if params_dtype is None:
322 params_dtype = torch.get_default_dtype()
323 self.params_dtype = params_dtype
324 # Divide the weight matrix along the vocabulary dimension.
325 self.num_added_embeddings = self.num_embeddings - self.org_vocab_size
326 self.num_embeddings_per_partition = divide(
327 self.num_embeddings_padded, self.tp_size
328 )
329 assert (
330 self.shard_indices.num_elements_padded == self.num_embeddings_per_partition
331 )
332 self.num_org_embeddings_per_partition = (
333 self.shard_indices.org_vocab_end_index
334 - self.shard_indices.org_vocab_start_index
335 )
336 self.num_added_embeddings_per_partition = (
337 self.shard_indices.added_vocab_end_index
338 - self.shard_indices.added_vocab_start_index
339 )
340
341 # The fused kernel masks, shifts and gathers in one pass; it assumes a
342 # plain row-major shard, so it only applies to unquantized weights.
343 self.use_fused_embedding = (
344 self.tp_size > 1
345 and current_platform.is_cuda()
346 and isinstance(quant_method, UnquantizedEmbeddingMethod)
347 )
348
349 self.quant_method.create_weights(
350 self,
351 self.embedding_dim,
352 [self.num_embeddings_per_partition],
353 self.embedding_dim,
354 self.num_embeddings_padded,
355 params_dtype=params_dtype,
356 weight_loader=self.weight_loader,
357 )
358 self.update_param_tp_status()
359
360 def update_param_tp_status(self):
361 for param in self.parameters():
362 if isinstance(param, BasevLLMParameter):
363 param.tp_rank = self.tp_rank
364 param.tp_size = self.tp_size

366 @classmethod
367 def _get_indices(
368 cls,
369 vocab_size_padded: int,
370 org_vocab_size_padded: int,
371 vocab_size: int,
372 org_vocab_size: int,
373 tp_rank: int,
374 tp_size: int,
375 ) -> VocabParallelEmbeddingShardIndices:
376 """Get start and end indices for vocab parallel embedding, following the
377 layout outlined in the class docstring, based on the given tp_rank and
378 tp_size."""
379 num_added_embeddings_padded = vocab_size_padded - org_vocab_size_padded
380 padded_org_vocab_start_index, padded_org_vocab_end_index = (
381 vocab_range_from_global_vocab_size(org_vocab_size_padded, tp_rank, tp_size)
382 )
383 padded_added_vocab_start_index, padded_added_vocab_end_index = (
384 vocab_range_from_global_vocab_size(
385 num_added_embeddings_padded, tp_rank, tp_size, offset=org_vocab_size
386 )
387 )
388 # remove padding
389 org_vocab_start_index = min(padded_org_vocab_start_index, org_vocab_size)
390 org_vocab_end_index = min(padded_org_vocab_end_index, org_vocab_size)
391 added_vocab_start_index = min(padded_added_vocab_start_index, vocab_size)
392 added_vocab_end_index = min(padded_added_vocab_end_index, vocab_size)
393 return VocabParallelEmbeddingShardIndices(
394 padded_org_vocab_start_index,
395 padded_org_vocab_end_index,
396 padded_added_vocab_start_index,
397 padded_added_vocab_end_index,
398 org_vocab_start_index,
399 org_vocab_end_index,
400 added_vocab_start_index,
401 added_vocab_end_index,
402 )

404 def get_sharded_to_full_mapping(self) -> list[int] | None:
405 """Get a mapping that can be used to reindex the gathered
406 logits for sampling.
407
408 During sampling, we gather logits from all ranks. The relationship
409 of index->token_id will follow the same format as outlined in the class
410 docstring. However, after the gather, we want to reindex the final
411 logits tensor to map index->token_id one-to-one (the index is always
412 equal the token_id it corresponds to). The indices returned by this
413 method allow us to do that.
414 """
415 if self.tp_size < 2:
416 return None
417
418 base_embeddings: list[int] = []
419 added_embeddings: list[int] = []
420 padding: list[int] = []
421 for tp_rank in range(self.tp_size):
422 shard_indices = self._get_indices(
423 self.num_embeddings_padded,
424 self.org_vocab_size_padded,
425 self.num_embeddings,
426 self.org_vocab_size,
427 tp_rank,
428 self.tp_size,
429 )
430 range_start = self.num_embeddings_per_partition * tp_rank
431 range_end = self.num_embeddings_per_partition * (tp_rank + 1)
432 base_embeddings.extend(
433 range(range_start, range_start + shard_indices.num_org_elements)
434 )
435 padding.extend(
436 range(
437 range_start + shard_indices.num_org_elements,
438 range_start + shard_indices.num_org_elements_padded,
439 )
440 )
441 added_embeddings.extend(
442 range(
443 range_start + shard_indices.num_org_elements_padded,
444 range_start
445 + shard_indices.num_org_elements_padded
446 + shard_indices.num_added_elements,
447 )
448 )
449 padding.extend(
450 range(
451 range_start
452 + shard_indices.num_org_elements_padded
453 + shard_indices.num_added_elements,
454 range_start
455 + shard_indices.num_org_elements_padded
456 + shard_indices.num_added_elements_padded,
457 )
458 )
459 assert (
460 range_start
461 + shard_indices.num_org_elements_padded
462 + shard_indices.num_added_elements_padded
463 == range_end
464 )
465 ret = base_embeddings + added_embeddings + padding
466 assert len(ret) == self.num_embeddings_padded
467 return ret

469 def weight_loader(self, param: Parameter, loaded_weight: torch.Tensor):
470 output_dim = getattr(param, "output_dim", None)
471 packed_dim = getattr(param, "packed_dim", None)
472
473 # Parameters without an output dimension are copied onto all GPUs.
474 if output_dim is None:
475 if (
476 loaded_weight.ndim == 0
477 and param.data.ndim == 1
478 and param.data.numel() == 1
479 ):
480 loaded_weight = loaded_weight.reshape(1)
481 assert param.data.shape == loaded_weight.shape
482 param.data.copy_(loaded_weight)
483 return
484
485 # Shard indexes for loading the weight
486 start_idx = self.shard_indices.org_vocab_start_index
487 shard_size = self.shard_indices.org_vocab_end_index - start_idx
488
489 # If param packed on the same dim we are sharding on, then
490 # need to adjust offsets of loaded weight by pack_factor.
491 if packed_dim is not None and packed_dim == output_dim:
492 packed_factor = (
493 param.packed_factor
494 if isinstance(param, BasevLLMParameter)
495 else param.pack_factor
496 )
497 assert loaded_weight.shape[output_dim] == (
498 self.org_vocab_size // param.packed_factor
499 )
500 start_idx = start_idx // packed_factor
501 shard_size = shard_size // packed_factor
502 else:
503 assert loaded_weight.shape[output_dim] == self.org_vocab_size
504
505 # Copy the data. Select chunk corresponding to current shard.
506 loaded_weight = loaded_weight.narrow(output_dim, start_idx, shard_size)
507 param[: loaded_weight.shape[0]].data.copy_(loaded_weight)
508 param[loaded_weight.shape[0] :].data.fill_(0)
509
510 def forward(self, input_):
511 if self.tp_size == 1:
512 return self.quant_method.embedding(self, input_.long())
513
514 if self.use_fused_embedding:
515 output_parallel = ops.vocab_parallel_embedding(
516 input_ if input_.ndim == 1 else input_.reshape(-1),
517 self.weight,
518 self.shard_indices.org_vocab_start_index,
519 self.shard_indices.org_vocab_end_index,
520 self.shard_indices.num_org_vocab_padding,
521 self.shard_indices.added_vocab_start_index,
522 self.shard_indices.added_vocab_end_index,
523 )
524 if input_.ndim != 1:
525 output_parallel = output_parallel.view(
526 *input_.shape, self.embedding_dim
527 )
528 else:
529 masked_input, input_mask = get_masked_input_and_mask(
530 input_,
531 self.shard_indices.org_vocab_start_index,
532 self.shard_indices.org_vocab_end_index,
533 self.shard_indices.num_org_vocab_padding,
534 self.shard_indices.added_vocab_start_index,
535 self.shard_indices.added_vocab_end_index,
536 )
537 output_parallel = self.quant_method.embedding(self, masked_input.long())
538 if output_parallel.dtype in (
539 torch.float8_e4m3fn,
540 torch.float8_e5m2,
541 ):
542 # Each vocab token has one owner, so FP8 bytes can use int8 SUM.
543 comm_output = output_parallel.view(torch.int8)
544 comm_output.masked_fill_(input_mask.unsqueeze(-1), 0)
545 output = (
546 self.parallel_group.all_reduce(comm_output)
547 if self.parallel_group is not None
548 else tensor_model_parallel_all_reduce(comm_output)
549 )
550 return output.view(output_parallel.dtype)
551 output_parallel.masked_fill_(input_mask.unsqueeze(-1), 0)
552 # Reduce across all the model parallel GPUs.
553 if self.parallel_group is not None:
554 return self.parallel_group.all_reduce(output_parallel)
555 return tensor_model_parallel_all_reduce(output_parallel)
556
557 def extra_repr(self) -> str:
558 s = f"num_embeddings={self.num_embeddings}"
