1126 def register_model(
1127 self,
1128 model_arch: str,
1129 model_cls: type[nn.Module] | str,
1130 ) -> None:
1131 """
1132 Register an external model to be used in vLLM.
1133
1134 `model_cls` can be either:
1135
1136 - A [`torch.nn.Module`][] class directly referencing the model.
1137 - A string in the format `<module>:<class>` which can be used to
1138 lazily import the model. This is useful to avoid initializing CUDA
1139 when importing the model and thus the related error
1140 `RuntimeError: Cannot re-initialize CUDA in forked subprocess`.
1141 """
1142 if not isinstance(model_arch, str):
1143 msg = f"`model_arch` should be a string, not a {type(model_arch)}"
1144 raise TypeError(msg)
1145
1146 if model_arch in self.models:
1147 logger.debug(
1148 "Model architecture %s is already registered, and will be "
1149 "overwritten by the new model class %s.",
1150 model_arch,
1151 model_cls,
1152 )
1153
1154 if isinstance(model_cls, str):
1155 split_str = model_cls.split(":")
1156 if len(split_str) != 2:
1157 msg = "Expected a string in the format `<module>:<class>`"
1158 raise ValueError(msg)
1159
1160 model = _LazyRegisteredModel(*split_str)
1161 elif isinstance(model_cls, type) and issubclass(model_cls, nn.Module):
1162 model = _RegisteredModel.from_model_cls(model_cls)
1163 else:
1164 msg = (
1165 "`model_cls` should be a string or PyTorch model class, "
1166 f"not a {type(model_cls)}"
1167 )
1168 raise TypeError(msg)
1169
1170 self.models[model_arch] = model

1217 def _try_resolve_transformers(
1218 self,
1219 architecture: str,
1220 model_config: ModelConfig,
1221 ) -> str | None:
1222 if architecture in _TRANSFORMERS_BACKEND_MODELS:
1223 return architecture
1224
1225 auto_map: dict[str, str] = (
1226 getattr(model_config.hf_config, "auto_map", None) or dict()
1227 )
1228
1229 # Make sure that config class is always initialized before model class,
1230 # otherwise the model class won't be able to access the config class,
1231 # the expected auto_map should have correct order like:
1232 # "auto_map": {
1233 # "AutoConfig": "<your-repo-name>--<config-name>",
1234 # "AutoModel": "<your-repo-name>--<config-name>",
1235 # "AutoModelFor<Task>": "<your-repo-name>--<config-name>",
1236 # },
1237 for prefix in ("AutoConfig", "AutoModel"):
1238 for name, module in auto_map.items():
1239 if name.startswith(prefix):
1240 try_get_class_from_dynamic_module(
1241 module,
1242 model_config.model,
1243 revision=model_config.revision,
1244 code_revision=model_config.code_revision,
1245 trust_remote_code=model_config.trust_remote_code,
1246 warn_on_fail=False,
1247 )
1248
1249 model_module = getattr(transformers, architecture, None)
1250
1251 if model_module is None:
1252 for name, module in auto_map.items():
1253 if name.startswith("AutoModel"):
1254 model_module = try_get_class_from_dynamic_module(
1255 module,
1256 model_config.model,
1257 revision=model_config.revision,
1258 code_revision=model_config.code_revision,
1259 trust_remote_code=model_config.trust_remote_code,
1260 warn_on_fail=True,
1261 )
1262 if model_module is not None:
1263 break
1264 else:
1265 if model_config.model_impl != "transformers":
1266 return None
1267
1268 raise ValueError(
1269 f"Cannot find model module. {architecture!r} is not a "
1270 "registered model in the Transformers library (only "
1271 "relevant if the model is meant to be in Transformers) "
1272 "and 'AutoModel' is not present in the model config's "
1273 "'auto_map' (relevant if the model is custom)."
1274 )
1275
1276 assert issubclass(model_module, transformers.PreTrainedModel)
1277 transformers_model_cls: type[transformers.PreTrainedModel] = model_module
1278 if not (
1279 transformers_model_cls.is_backend_compatible()
1280 or transformers_model_cls._can_set_attn_implementation()
1281 ):
1282 if model_config.model_impl != "transformers":
1283 return None
1284
1285 raise ValueError(
1286 f"The Transformers implementation of {architecture!r} "
1287 "is not compatible with vLLM."
1288 )

1290 return model_config._get_transformers_backend_cls()

1318 def inspect_model_cls(
1319 self,
1320 architectures: str | list[str],
1321 model_config: ModelConfig,
1322 ) -> tuple[_ModelInfo, str]:
1323 if isinstance(architectures, str):
1324 architectures = [architectures]
1325 if not architectures:
1326 raise ValueError("No model architectures are specified")
1327
1328 # Require transformers impl
1329 if model_config.model_impl == "transformers":
1330 arch = self._try_resolve_transformers(architectures[0], model_config)
1331 if arch is not None:
1332 model_info = self._try_inspect_model_cls(arch)
1333 if model_info is not None:
1334 return (model_info, arch)
1335 elif model_config.model_impl == "terratorch":
1336 model_info = self._try_inspect_model_cls("Terratorch")
1337 assert model_info is not None
1338 return (model_info, "Terratorch")
1339
1340 # Fallback to transformers impl (after resolving convert_type)
1341 if (
1342 all(arch not in self.models for arch in architectures)
1343 and model_config.model_impl == "auto"
1344 and getattr(model_config, "convert_type", "none") == "none"
1345 ):
1346 arch = self._try_resolve_transformers(architectures[0], model_config)
1347 if arch is not None:
1348 model_info = self._try_inspect_model_cls(arch)
1349 if model_info is not None:
1350 return (model_info, arch)
1351
1352 for arch in architectures:
1353 normalized_arch = self._normalize_arch(arch, model_config)
1354 model_info = self._try_inspect_model_cls(normalized_arch)
1355 if model_info is not None:
1356 return (model_info, arch)
1357
1358 # Fallback to transformers impl (before resolving runner_type)
1359 if (
1360 all(arch not in self.models for arch in architectures)
1361 and model_config.model_impl == "auto"
1362 ):
1363 arch = self._try_resolve_transformers(architectures[0], model_config)
1364 if arch is not None:
1365 model_info = self._try_inspect_model_cls(arch)
1366 if model_info is not None:
1367 return (model_info, arch)
1368
1369 return self._raise_for_unsupported(architectures)

1536def _run_in_subprocess(fn: Callable[[], _T]) -> _T:
1537 # NOTE: We use a temporary directory instead of a temporary file to avoid
1538 # issues like https://stackoverflow.com/questions/23212435/permission-denied-to-write-to-my-temporary-file
1539 with tempfile.TemporaryDirectory() as tempdir:
1540 output_filepath = os.path.join(tempdir, "registry_output.tmp")
1541
1542 # `cloudpickle` allows pickling lambda functions directly
1543 import cloudpickle
1544
1545 input_bytes = cloudpickle.dumps((fn, output_filepath))
1546
1547 # cannot use `sys.executable __file__` here because the script
1548 # contains relative imports
1549 returned = subprocess.run(
1550 _SUBPROCESS_COMMAND, input=input_bytes, capture_output=True
1551 )
1552
1553 # check if the subprocess is successful
1554 try:
1555 returned.check_returncode()
1556 except Exception as e:
1557 # wrap raised exception to provide more information
1558 raise RuntimeError(
1559 f"Error raised in subprocess:\n{returned.stderr.decode()}"
1560 ) from e
1561
1562 with open(output_filepath, "rb") as f:
1563 return pickle.load(f)
