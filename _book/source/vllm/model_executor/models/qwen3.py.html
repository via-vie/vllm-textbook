65class Qwen3Attention(nn.Module):
66 def __init__(
67 self,
68 hidden_size: int,
69 num_heads: int,
70 num_kv_heads: int,
71 rope_parameters: dict,
72 max_position: int = 4096 * 32,
73 head_dim: int | None = None,
74 rms_norm_eps: float = 1e-06,
75 qkv_bias: bool = False,
76 cache_config: CacheConfig | None = None,
77 quant_config: QuantizationConfig | None = None,
78 prefix: str = "",
79 attn_type: str = AttentionType.DECODER,
80 dual_chunk_attention_config: dict[str, Any] | None = None,
81 per_layer_sliding_window: int | None = None,
82 ) -> None:
83 super().__init__()
84 self.hidden_size = hidden_size
85 tp_size = get_tensor_model_parallel_world_size()
86 self.total_num_heads = num_heads
87 assert self.total_num_heads % tp_size == 0
88 self.num_heads = self.total_num_heads // tp_size
89 self.total_num_kv_heads = num_kv_heads
90 if self.total_num_kv_heads >= tp_size:
91 # Number of KV heads is greater than TP size, so we partition
92 # the KV heads across multiple tensor parallel GPUs.
93 assert self.total_num_kv_heads % tp_size == 0
94 else:
95 # Number of KV heads is less than TP size, so we replicate
96 # the KV heads across multiple tensor parallel GPUs.
97 assert tp_size % self.total_num_kv_heads == 0
98 self.num_kv_heads = max(1, self.total_num_kv_heads // tp_size)
99 self.head_dim = head_dim or hidden_size // self.total_num_heads
100 self.q_size = self.num_heads * self.head_dim
101 self.kv_size = self.num_kv_heads * self.head_dim
102 self.scaling = self.head_dim**-0.5
103 self.dual_chunk_attention_config = dual_chunk_attention_config
104
105 self.qkv_proj = QKVParallelLinear(
106 hidden_size,
107 self.head_dim,
108 self.total_num_heads,
109 self.total_num_kv_heads,
110 bias=qkv_bias,
111 quant_config=quant_config,
112 prefix=f"{prefix}.qkv_proj",
113 )
114 self.o_proj = RowParallelLinear(
115 self.total_num_heads * self.head_dim,
116 hidden_size,
117 bias=False,
118 quant_config=quant_config,
119 prefix=f"{prefix}.o_proj",
120 )
121
122 self.rotary_emb = get_rope(
123 self.head_dim,
124 max_position=max_position,
125 rope_parameters=rope_parameters,
126 dual_chunk_attention_config=dual_chunk_attention_config,
127 )
128 attn_cls = (
129 EncoderOnlyAttention
130 if attn_type == AttentionType.ENCODER_ONLY
131 else Attention
132 )
133 self.attn = attn_cls(
134 self.num_heads,
135 self.head_dim,
136 self.scaling,
137 num_kv_heads=self.num_kv_heads,
138 cache_config=cache_config,
139 quant_config=quant_config,
140 per_layer_sliding_window=per_layer_sliding_window,
141 prefix=f"{prefix}.attn",
142 attn_type=attn_type,
143 **{
144 "layer_idx": extract_layer_index(prefix),
145 "dual_chunk_attention_config": dual_chunk_attention_config,
146 }
147 if dual_chunk_attention_config
148 else {},
149 )
150 self.q_norm = RMSNorm(self.head_dim, eps=rms_norm_eps)
151 self.k_norm = RMSNorm(self.head_dim, eps=rms_norm_eps)
152
153 def forward(
154 self,
155 positions: torch.Tensor,
156 hidden_states: torch.Tensor,
157 ) -> torch.Tensor:
158 qkv, _ = self.qkv_proj(hidden_states)
159 q, k, v = qkv.split([self.q_size, self.kv_size, self.kv_size], dim=-1)
160 # Add qk-norm
161 q_by_head = q.view(*q.shape[:-1], q.shape[-1] // self.head_dim, self.head_dim)
162 q_by_head = self.q_norm(q_by_head)
163 q = q_by_head.view(q.shape)
164 k_by_head = k.view(*k.shape[:-1], k.shape[-1] // self.head_dim, self.head_dim)
165 k_by_head = self.k_norm(k_by_head)
166 k = k_by_head.view(k.shape)
167 q, k = self.rotary_emb(positions, q, k)
168 attn_output = self.attn(q, k, v)
169 output, _ = self.o_proj(attn_output)
170 return output

173class Qwen3DecoderLayer(nn.Module):
174 def __init__(
175 self,
176 config: Qwen3Config,
177 cache_config: CacheConfig | None = None,
178 quant_config: QuantizationConfig | None = None,
179 prefix: str = "",
180 per_layer_sliding_window: int | None = None,
181 ) -> None:
182 super().__init__()
183 self.hidden_size = config.hidden_size
184 set_default_rope_theta(config, default_theta=1000000)
185 dual_chunk_attention_config = getattr(
186 config, "dual_chunk_attention_config", None
187 )
188
189 # By default, Qwen3 uses causal attention as it is a decoder-only model.
190 # You can override the HF config with `is_causal=False` to enable
191 # bidirectional attention, which is used in some embedding models
192 # (e.g. Alibaba-NLP/gte-Qwen3-7B-instruct)
193 if getattr(config, "is_causal", True):
194 attn_type = AttentionType.DECODER
195 else:
196 attn_type = AttentionType.ENCODER_ONLY
197
198 self.self_attn = Qwen3Attention(
199 hidden_size=self.hidden_size,
200 num_heads=config.num_attention_heads,
201 max_position=config.max_position_embeddings,
202 num_kv_heads=config.num_key_value_heads,
203 rms_norm_eps=config.rms_norm_eps,
204 qkv_bias=getattr(config, "attention_bias", False),
205 head_dim=getattr(config, "head_dim", None),
206 cache_config=cache_config,
207 quant_config=quant_config,
208 rope_parameters=config.rope_parameters,
209 prefix=f"{prefix}.self_attn",
210 attn_type=attn_type,
211 dual_chunk_attention_config=dual_chunk_attention_config,
212 per_layer_sliding_window=per_layer_sliding_window,
213 )
214 self.mlp = Qwen3MLP(
215 hidden_size=self.hidden_size,
216 intermediate_size=config.intermediate_size,
217 hidden_act=config.hidden_act,
218 quant_config=quant_config,
219 prefix=f"{prefix}.mlp",
220 )
221 self.input_layernorm = RMSNorm(config.hidden_size, eps=config.rms_norm_eps)
222 self.post_attention_layernorm = RMSNorm(
223 config.hidden_size, eps=config.rms_norm_eps
224 )
225
226 def forward(
227 self,
228 positions: torch.Tensor,
229 hidden_states: torch.Tensor,
230 residual: torch.Tensor | None,
231 ) -> tuple[torch.Tensor, torch.Tensor]:
232 # Self Attention
233 if residual is None:
234 residual = hidden_states
235 hidden_states = self.input_layernorm(hidden_states)
236 else:
237 hidden_states, residual = self.input_layernorm(hidden_states, residual)
238 hidden_states = self.self_attn(
239 positions=positions,
240 hidden_states=hidden_states,
241 )
242
243 # Fully Connected
244 hidden_states, residual = self.post_attention_layernorm(hidden_states, residual)
245 hidden_states = self.mlp(hidden_states)
246 return hidden_states, residual

271class Qwen3ForCausalLM(
272 LocalArgmaxMixin, nn.Module, SupportsLoRA, SupportsPP, SupportsEagle, SupportsEagle3
273):
274 hf_to_vllm_mapper = Qwen3Model.hf_to_vllm_mapper
275 packed_modules_mapping = {
276 "qkv_proj": ["q_proj", "k_proj", "v_proj"],
277 "gate_up_proj": ["gate_proj", "up_proj"],
278 }
279 embedding_modules = {
280 "embed_tokens": "input_embeddings",
281 "lm_head": "output_embeddings",
282 }
283
284 def __init__(self, *, vllm_config: VllmConfig, prefix: str = ""):
285 super().__init__()
286 config = vllm_config.model_config.hf_config
287 quant_config = vllm_config.quant_config
288
289 self.config = config
290
291 self.vllm_config = vllm_config
292 self.quant_config = quant_config
293 self.model = Qwen3Model(
294 vllm_config=vllm_config, prefix=maybe_prefix(prefix, "model")
295 )
296
297 if get_pp_group().is_last_rank:
298 self.lm_head = ParallelLMHead(
299 config.vocab_size,
300 config.hidden_size,
301 quant_config=quant_config,
302 prefix=maybe_prefix(prefix, "lm_head"),
303 )
304 if config.tie_word_embeddings:
305 self.lm_head = self.lm_head.tie_weights(self.model.embed_tokens)
306 else:
307 self.lm_head = PPMissingLayer()
308
309 self.logits_processor = LogitsProcessor(config.vocab_size)
310
311 self.make_empty_intermediate_tensors = (
312 self.model.make_empty_intermediate_tensors
313 )
314
315 def embed_input_ids(self, input_ids: torch.Tensor) -> torch.Tensor:
316 return self.model.embed_input_ids(input_ids)
317
318 def forward(
319 self,
320 input_ids: torch.Tensor | None,
321 positions: torch.Tensor,
322 intermediate_tensors: IntermediateTensors | None = None,
323 inputs_embeds: torch.Tensor | None = None,
324 ) -> torch.Tensor | IntermediateTensors:
325 hidden_states = self.model(
326 input_ids, positions, intermediate_tensors, inputs_embeds
327 )
328 return hidden_states
329
330 def compute_logits(
331 self,
332 hidden_states: torch.Tensor,
333 ) -> torch.Tensor | None:
334 logits = self.logits_processor(self.lm_head, hidden_states)
335 return logits
336
337 def load_weights(self, weights: Iterable[tuple[str, torch.Tensor]]) -> set[str]:
338 loader = AutoWeightsLoader(self)
339 return loader.load_weights(weights)
